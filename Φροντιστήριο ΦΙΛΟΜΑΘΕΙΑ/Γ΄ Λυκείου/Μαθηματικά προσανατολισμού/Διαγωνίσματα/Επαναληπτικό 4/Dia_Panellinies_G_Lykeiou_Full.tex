\titlos{Μαθηματικά Προσανατολισμού}{Γ' Λυκείου}{Εφ' όλης της Ύλης}
\begin{thema}
\item\mbox{}\\
\begin{erwthma}
\item Έστω μια συνάρτηση $f$ ορισμένη σε ένα διάστημα $\Delta$. Αν $F$ είναι μια παράγουσα της $f$ στο $\Delta$, να αποδείξετε ότι:
\begin{itemize}
    \item Όλες οι συναρτήσεις της μορφής $G(x) = F(x) + c$, όπου $c \in \mathbb{R}$, είναι παράγουσες της $f$ στο $\Delta$.
    \item Κάθε άλλη παράγουσα $G$ της $f$ στο $\Delta$ παίρνει τη μορφή $G(x) = F(x) + c$, όπου $c \in \mathbb{R}$.
\end{itemize}
\monades{7}

\item  Πότε η ευθεία $y = \lambda x + \beta$ λέγεται ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$;
\monades{4}

\item  Θεωρήστε τον ισχυρισμό: <<Αν μια συνάρτηση $f$ είναι συνεχής σε ένα σημείο $x_0$ του πεδίου ορισμού της, τότε είναι και παραγωγίσιμη σε αυτό>>.
\begin{itemize}
    \item[(α)] Να χαρακτηρίσετε τον ισχυρισμό ως Αληθή ή Ψευδή.
    \item[(β)] Να δικαιολογήσετε την απάντησή σας με ένα αντιπαράδειγμα.
\end{itemize}
\monades{1+3}

\item  Να χαρακτηρίσετε τις προτάσεις που ακολουθούν, γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό}, αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη.
\begin{itemize}
    \item[(α)] Αν $\lim_{x \to x_0} f(x) = 0$ και $\lim_{x \to x_0} g(x) = +\infty$, τότε το $\lim_{x \to x_0} [f(x) \cdot g(x)]$ είναι πάντοτε ίσο με $0$.
    \item[(β)] Αν η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και γνησίως αύξουσα, τότε υποχρεωτικά $f'(x) > 0$ για κάθε $x \in \mathbb{R}$.
    \item[(γ)] Ισχύει $\int_{\alpha}^{\beta} f(x) dx = -\int_{\beta}^{\alpha} f(x) dx$ για κάθε συνεχή συνάρτηση $f$ στο $[\alpha, \beta]$.
    \item[(δ)] Ένα τοπικό μέγιστο μιας συνάρτησης μπορεί να είναι μικρότερο από ένα τοπικό ελάχιστο της ίδιας συνάρτησης.
    \item[(ε)] Αν $f'(x) = g'(x)$ για κάθε $x \in \Delta$, τότε $f(x) = g(x)$ για κάθε $x \in \Delta$.
\end{itemize}
\monades{10}
\end{erwthma}
\item\mbox{}\\
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2}{x-1}, \quad x \neq 1$.
\begin{erwthma}
\item Να βρείτε τα διαστήματα μονοτονίας και τα τοπικά ακρότατα της συνάρτησης $f$.
\monades{8}
\item Να βρείτε τις ασύμπτωτες της γραφικής παράστασης της $f$.
\monades{7}
\item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο $M(2, f(2))$.
\monades{5}
\item Να υπολογίσετε το όριο $\lim_{x \to 1} [f(x) \cdot (x-1)^2]$.
\monades{5}
\end{erwthma}
\item\mbox{}\\
Δίνεται η συνάρτηση $f(x) = e^x + x - 1, \quad x \in \mathbb{R}$.
\begin{erwthma}
\item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$ και να βρείτε το σύνολο τιμών της.
\monades{6}
\item Να αποδείξετε ότι η εξίσωση $f(x) = 0$ έχει μοναδική ρίζα την $x = 0$.
\monades{5}
\item Να λύσετε την ανίσωση $e^{x^2} - e^x > x - x^2$.
\monades{7}
\item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις ευθείες $x=0$ και $x=1$.
\monades{7}
\end{erwthma}
\item\mbox{}\\
Δίνεται παραγωγίσιμη συνάρτηση $f: (0, +\infty) \to \mathbb{R}$ για την οποία ισχύουν:
$$ x f'(x) - f(x) = x^2 \ln x, \quad \text{για κάθε } x > 0 $$
και $f(1) = 0$.
\begin{erwthma}
\item Να αποδείξετε ότι ο τύπος της συνάρτησης είναι $f(x) = x^2 \ln x - x^2 + x, \quad x > 0$.
\monades{7}
\item Να μελετήσετε την $f$ ως προς τη μονοτονία και την κυρτότητα.
\monades{6}
\item Να βρείτε το πλήθος των ριζών της εξίσωσης $f(x) = \alpha$ για τις διάφορες τιμές του $\alpha \in \mathbb{R}$.
\monades{6}
\item Να αποδείξετε ότι υπάρχει ένα τουλάχιστον $\xi \in (1, e)$ τέτοιο ώστε:
$$ f'(\xi) = \dfrac{f(e)}{\xi (e-1)} $$
\monades{6}
\end{erwthma}
\end{thema}
