\item\mbox{}\\
\begin{erwthma}
\item Θεωρία σχολικού βιβλίου.
\item Θεωρία σχολικού βιβλίου (Ορισμός ασύμπτωτης).
\item (α) Ψευδής.\\
(β) Αντιπαράδειγμα: Η συνάρτηση $f(x) = |x|$ είναι συνεχής στο $x_0 = 0$ ($\lim_{x \to 0} |x| = 0 = f(0)$) αλλά δεν είναι παραγωγίσιμη σε αυτό, καθώς $\lim_{x \to 0^+} \frac{|x|-0}{x} = 1$ ενώ $\lim_{x \to 0^-} \frac{|x|-0}{x} = -1$.
\item 
\begin{alist}
\item \textbf{Λάθος} (Απροσδιόριστη μορφή $0 \cdot \infty$).
\item \textbf{Λάθος} (Μπορεί να μηδενίζεται σε μεμονωμένα σημεία, π.χ. $f(x)=x^3$ στο $0$).
\item \textbf{Σωστό} (Ιδιότητα ορισμένου ολοκληρώματος).
i\textbf{Σωστό} (Τοπικά ακρότατα δεν συγκρίνονται άμεσα μεταξύ τους, π.χ. $f(x) = \frac{1}{x}$).
\item \textbf{Λάθος} (Ισχύει $f(x) = g(x) + c$).
\end{alist}
\item\mbox{}\\
$f(x) = \dfrac{x^2}{x-1}, \quad x \neq 1$.
\end{erwthma}
\item\mbox{}\\
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(-\infty, 1) \cup (1, +\infty)$ με:
$$ f'(x) = \dfrac{(x^2)'(x-1) - x^2(x-1)'}{(x-1)^2} = \dfrac{2x(x-1) - x^2}{(x-1)^2} = \dfrac{2x^2 - 2x - x^2}{(x-1)^2} = \dfrac{x^2 - 2x}{(x-1)^2} = \dfrac{x(x-2)}{(x-1)^2} $$
Το πρόσημο της $f'$ εξαρτάται από τον αριθμητή $x(x-2)$:
\begin{itemize}
    \item $f'(x) > 0 \Leftrightarrow x < 0$ ή $x > 2$. Άρα $f$ γνησίως αύξουσα στα intervals $(-\infty, 0]$ και $[2, +\infty)$.
    \item $f'(x) < 0 \Leftrightarrow 0 < x < 1$ ή $1 < x < 2$. Άρα $f$ γνησίως φθίνουσα στα διαστήματα $[0, 1)$ και $(1, 2]$.
\end{itemize}
Τοπικά ακρότατα:
Στο $x = 0$, τοπικό μέγιστο το $f(0) = 0$.
Στο $x = 2$, τοπικό ελάχιστο το $f(2) = \frac{4}{1} = 4$.

\item Ασύμπτωτες:
\textbf{Κατακόρυφη:}
$\lim_{x \to 1^+} f(x) = \lim_{x \to 1^+} \frac{x^2}{x-1} = +\infty$ (αφού $x-1 \to 0^+$).
Άρα η $x = 1$ είναι κατακόρυφη ασύμπτωτη.

\textbf{Πλάγιες/Οριζόντιες:}
$\lim_{x \to \pm\infty} \frac{f(x)}{x} = \lim_{x \to \pm\infty} \frac{x^2}{x^2-x} = 1 = \lambda$.
$\lim_{x \to \pm\infty} (f(x) - \lambda x) = \lim_{x \to \pm\infty} (\frac{x^2}{x-1} - x) = \lim_{x \to \pm\infty} \frac{x^2 - x^2 + x}{x-1} = \lim_{x \to \pm\infty} \frac{x}{x-1} = 1 = \beta$.
Άρα η ευθεία $y = x + 1$ είναι πλάγια ασύμπτωτη στο $+\infty$ και στο $-\infty$.

\item Εφαπτομένη στο $M(2, f(2)) = (2, 4)$.
Η κλίση είναι $f'(2) = \frac{2(2-2)}{(2-1)^2} = 0$.
Άρα η εφαπτομένη είναι η οριζόντια ευθεία $y - 4 = 0(x-2) \Rightarrow \boxed{y = 4}$.

\item Όριο $\lim_{x \to 1} [f(x) \cdot (x-1)^2] = \lim_{x \to 1} \left[ \frac{x^2}{x-1} \cdot (x-1)^2 \right] = \lim_{x \to 1} [x^2 (x-1)] = 1^2 \cdot 0 = 0$.
\end{erwthma}
\item\mbox{}\\
$f(x) = e^x + x - 1, \quad x \in \mathbb{R}$.

\begin{erwthma}
\item $f'(x) = e^x + 1 > 0$ για κάθε $x \in \mathbb{R}$.
Άρα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$.
Σύνολο τιμών:
$\lim_{x \to -\infty} f(x) = 0 + (-\infty) - 1 = -\infty$.
$\lim_{x \to +\infty} f(x) = +\infty + \infty - 1 = +\infty$.
Άρα $f(\mathbb{R}) = (-\infty, +\infty) = \mathbb{R}$.

\item Παρατηρούμε ότι $f(0) = e^0 + 0 - 1 = 1 - 1 = 0$.
Άρα το $x=0$ είναι ρίζα.
Επειδή η $f$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

\item Ανίσωση $e^{x^2} - e^x > x - x^2 \Leftrightarrow e^{x^2} + x^2 - 1 > e^x + x - 1 \Leftrightarrow f(x^2) > f(x)$.
Επειδή η $f$ είναι γνησίως αύξουσα:
$$ x^2 > x \Leftrightarrow x^2 - x > 0 \Leftrightarrow x(x-1) > 0 \Leftrightarrow x < 0 \text{ ή } x > 1. $$

\item Εμβαδόν:
Στο $[0, 1]$, επειδή $x \geq 0$ και η $f$ αύξουσα, $f(x) \geq f(0) = 0$.
Άρα $E = \int_0^1 f(x) dx = \int_0^1 (e^x + x - 1) dx$.
$$ E = [e^x + \frac{x^2}{2} - x]_0^1 = (e^1 + \frac{1}{2} - 1) - (e^0 + 0 - 0) = e - \frac{1}{2} - 1 = e - \frac{3}{2} \text{ τ.μ.} $$
\end{erwthma}
\item\mbox{}\\
$x f'(x) - f(x) = x^2 \ln x, \quad x > 0$.

\begin{erwthma}
\item Διαιρούμε με $x^2$ (αφού $x>0$):
$$ \dfrac{x f'(x) - f(x)}{x^2} = \ln x \Rightarrow \left( \dfrac{f(x)}{x} \right)' = (x \ln x - x)' $$
(Mια παράγουσα του $\ln x$ είναι $x \ln x - x$).
Άρα υπάρχιε $c \in \mathbb{R}$ ώστε $\frac{f(x)}{x} = x \ln x - x + c$.
Για $x=1$: $\frac{0}{1} = 1 \ln 1 - 1 + c \Rightarrow 0 = -1 + c \Rightarrow c = 1$.
Άρα $\frac{f(x)}{x} = x \ln x - x + 1 \Rightarrow \boxed{f(x) = x^2 \ln x - x^2 + x}$.

\item $f'(x) = 2x \ln x + x^2 \frac{1}{x} - 2x + 1 = 2x \ln x + x - 2x + 1 = 2x \ln x - x + 1$.
$f''(x) = 2 \ln x + 2x \frac{1}{x} - 1 = 2 \ln x + 2 - 1 = 2 \ln x + 1$.
Κυρτότητα:
$f''(x) = 0 \Leftrightarrow \ln x = -1/2 \Leftrightarrow x = e^{-1/2} = 1/\sqrt{e}$.
$f''(x) > 0 \Leftrightarrow x > e^{-1/2}$ (κυρτή).
$f''(x) < 0 \Leftrightarrow 0 < x < e^{-1/2}$ (κοίλη).
Σημείο καμπής: $K(e^{-1/2}, f(e^{-1/2}))$.

Μονοτονία:
Παρατηρούμε ότι $f'(1) = 2(1)(0) - 1 + 1 = 0$.
Επειδή $f''$ μηδενίζεται στο $e^{-1/2} < 1$, στο $(1, +\infty)$ η $f'' > 0 \Rightarrow f'$ αύξουσα. Άρα για $x > 1 \Rightarrow f'(x) > f'(1) = 0$.
Στο $(0, 1)$ έχουμε αλλαγή μονοτονίας της $f'$;
Στο $(e^{-1/2}, 1)$, $f'' > 0 \Rightarrow f'$ αύξουσα, άρα $f'(x) < f'(1) = 0$.
Στο $(0, e^{-1/2}]$, $f'' < 0 \Rightarrow f'$ φθίνουσα.
Όριο $f'(x)$ στο $0^+$: $\lim_{x \to 0^+} (2x \ln x - x + 1) = 0 - 0 + 1 = 1 > 0$.
(Το $\lim x \ln x = 0$).
Άρα η $f'$ ξεκινά από θετικές τιμές, μειώνεται, γίνεται αρνητική, και μετά ξανά θετική; Όχι.
$f''(x)$ έχει ρίζα μόνο στο $e^{-1/2}$.
$f'$ έχει ελάχιστο στο $e^{-1/2}$;
$f'$ φθίνει στο $(0, e^{-1/2}]$. $f'(e^{-1/2}) = 2 e^{-1/2} (-1/2) - e^{-1/2} + 1 = -e^{-1/2} - e^{-1/2} + 1 = 1 - 2/\sqrt{e} < 0$ (αφού $\sqrt{e} < 2$).
Άρα η $f'$ έχει ρίζα $x_1$ στο $(0, e^{-1/2})$ και ρίζα $x=1$.
Επομένως η $f$ έχει:
Τοπικό μέγιστο στο $x_1$.
Τοπικό ελάχιστο στο $x=1$ (με τιμή $f(1)=0$).

\item Πλήθος ριζών $f(x) = \alpha$.
Από μελέτη μονοτονίας:
$f$ γν. αύξουσα στο $(0, x_1]$, γν. φθίνουσα στο $[x_1, 1]$, γν. αύξουσα στο $[1, +\infty)$.
Ελάχιστο: $f(1) = 0$.
Όρια: $\lim_{x \to 0^+} f(x) = 0$. $\lim_{x \to +\infty} f(x) = +\infty$.
Μέγιστο: $f(x_1) > 0$.
Διερεύνηση:
\begin{itemize}
    \item Αν $\alpha < 0$: 0 ρίζες.
    \item Αν $\alpha = 0$: 1 ρίζα ($x=1$, διπλή; όχι απλά ελάχιστο, ρίζα είναι). Επίσης το όριο στο 0 είναι 0, αλλά δεν το πιάνει.
    \item Αν $0 < \alpha < f(x_1)$: 3 ρίζες.
    \item Αν $\alpha = f(x_1)$: 2 ρίζες.
    \item Αν $\alpha > f(x_1)$: 1 ρίζα.
\end{itemize}

\item Θεωρούμε τη συνάρτηση $h(x) = \xi (e-1) f'(\xi) - f(e)$ (ως προς $\xi$).
Ή καλύτερα: Ζητάμε $\xi \in (1, e)$ ώστε $f'(\xi) = \frac{f(e)}{\xi(e-1)}$.
Ισοδύναμα $\xi f'(\xi) = \frac{f(e)}{e-1}$.
Έστω $g(x) = f(x)$.
Θ.Μ.Τ. για την $f$ στο $[1, e]$: υπάρχει $\xi_1$ ώστε $f'(\xi_1) = \frac{f(e)-0}{e-1}$.
Αυτό είναι κοντά αλλά έχουμε το $\xi$ στον παρονομαστή.
Θεωρούμε τη συνάρτηση $K(x) = f(x) - \frac{f(e)}{e-1} \ln x$.
$K'(x) = f'(x) - \frac{f(e)}{e-1} \frac{1}{x}$.
$K(1) = 0 - 0 = 0$.
$K(e) = f(e) - \frac{f(e)}{e-1} \cdot 1$? Όχι αυτό δεν μηδενίζεται.
Ας δούμε το Θεώρημα \eng{Bolzano} για την $h(x) = x f'(x) - \frac{f(e)}{e-1}$ στο $[1, e]$.
$h(1) = 1 f'(1) - A = 0 - A < 0$ (αφού $f(e) = e^2-e^2+e = e > 0$, άρα $A > 0$).
$h(e) = e f'(e) - A$.
$f'(e) = 2e \ln e - e + 1 = 2e - e + 1 = e + 1$.
Οπότε $h(e) = e(e+1) - \frac{e}{e-1} = e^2 + e - \frac{e}{e-1} > 0$.
Άρα υπάρχει $\xi \in (1, e)$ ώστε $h(\xi)=0 \Leftrightarrow \xi f'(\xi) = \frac{f(e)}{e-1}$.
\end{erwthma}
