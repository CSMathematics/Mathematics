\paragraph{Ερωτήσεις θεωρίας}
\askhsh
Έστω ότι η συνάρτηση $f$ είναι συνεχής στο διάστημα $[a,b]$.
\begin{alist}
    \item Αν $f(a) \cdot f(b) > 0$, είναι δυνατόν η εξίσωση $f(x)=0$ να έχει ρίζα στο $(a,b)$; Δώστε παράδειγμα.
    \item Αν η $f$ δεν έχει ρίζα στο $[a,b]$, να αποδείξετε ότι διατηρεί σταθερό πρόσημο στο διάστημα αυτό.
    \item Αν $f(x) \ne 0$ για κάθε $x \in [a,b]$ και $f(a)=1$, να βρείτε το πρόσημο του $f(b)$.
    \item Να διατυπώσετε το αντίστροφο του θεωρήματος \eng{Bolzano}. Ισχύει; (Δώστε αντιπαράδειγμα).
\end{alist}

\askhsh
Δίνεται συνεχής συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ με την ιδιότητα $f(f(x)) = x$ για κάθε $x \in \mathbb{R}$.
\begin{alist}
    \item Να αποδείξετε ότι η $f$ είναι «1-1».
    \item Αν η $f$ είναι γνησίως αύξουσα, να δείξετε ότι $f(x)=x$ για κάθε $x$.
    \item Αν η $f$ είναι γνησίως φθίνουσα, να δώσετε ένα παράδειγμα τέτοιας συνάρτησης (π.χ. $f(x)=-x+k$).
\end{alist}

\askhsh
Έστω $f, g$ συνεχείς συναρτήσεις στο $[0,1]$ με $f(0)=g(1)=0$ και $f(1)=g(0)=1$.
\begin{alist}
    \item Να ορίσετε τη συνάρτηση διαφοράς $h(x)=f(x)-g(x)$.
    \item Να αποδείξετε ότι οι γραφικές παραστάσεις των $f$ και $g$ τέμνονται σε τουλάχιστον ένα σημείο.
    \item Αν επιπλέον $f(x) \cdot g(x) \ne 0$ για $x \in (0,1)$, τι συμπεραίνετε για τη θέση της ρίζας;
\end{alist}

\askhsh
Έστω συνεχής συνάρτηση $f: [0,1] \to [0,1]$.
\begin{alist}
    \item Να αποδείξετε ότι η εξίσωση $f(x)=x$ έχει τουλάχιστον μία ρίζα στο $[0,1]$. (Θεώρημα Σταθερού Σημείου).
    \item Αν η $f$ είναι γνησίως φθίνουσα, να δείξετε ότι το σταθερό σημείο είναι μοναδικό.
    \item Να ερμηνεύσετε γεωμετρικά το αποτέλεσμα ως προς την τομή με την ευθεία $y=x$.
\end{alist}

\askhsh
Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (\textbf{Σ}) ή Λανθασμένες (\textbf{Λ}):
\begin{alist}
    \item Αν η συνάρτηση $f$ είναι συνεχής στο $[a,b]$ και $f(a)f(b)<0$, τότε η εξίσωση $f(x)=0$ έχει τουλάχιστον μία ρίζα στο $(a,b)$.
    \item Αν $f(a)f(b)<0$, τότε υπάρχει πάντα ρίζα της $f(x)=0$ στο $(a,b)$, ακόμα και αν η $f$ δεν είναι συνεχής.
    \item Το θεώρημα \eng{Bolzano} εξασφαλίζει τη μοναδικότητα της ρίζας στο διάστημα $(a,b)$.
    \item Αν η $f$ είναι συνεχής στο $[a,b]$ και δεν έχει ρίζα στο $(a,b)$, τότε ισχύει υποχρεωτικά $f(a)f(b)>0$.
    \item Αν η γραφική παράσταση μιας συνεχούς συνάρτησης περνάει από τα σημεία $A(1, -2)$ και $B(3, 4)$, τότε η $C_f$ τέμνει τον άξονα $x'x$ σε ένα τουλάχιστον σημείο με τετμημένη $x_0 \in (1,3)$.
    \item Αν $f(a)f(b)>0$, τότε η $f$ δεν έχει ρίζα στο $(a,b)$.
    \item Αν η $f$ είναι συνεχής στο $[a,b]$ και έχει ρίζα στο $(a,b)$, τότε κατ' ανάγκη ισχύει $f(a)f(b)<0$.
    \item Το θεώρημα \eng{Bolzano} εφαρμόζεται και σε διαστήματα της μορφής $(a,b)$.
    \item Αν η $f$ είναι συνεχής στο $[a,b]$ με $f(a)\ne 0$ και $f(x)\ne 0$ για κάθε $x\in(a,b)$, τότε $f(a)f(b)>0$.
    \item Η συνάρτηση $f(x)=1/x$ ικανοποιεί τις προϋποθέσεις του θεωρήματος \eng{Bolzano} στο $[-1,1]$. 
\end{alist}
