\paragraph{Ομάδα Α: Εφαρμογή του Θεωρήματος Rolle}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^2 - 4x + 3$.
\begin{alist}
  \item Να δείξετε ότι η $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο διάστημα $[1, 3]$.
  \item Να βρείτε το $\xi \in (1, 3)$ για το οποίο ισχύει $f'(\xi) = 0$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^3 - 3x$.
\begin{alist}
  \item Να εξετάσετε αν η $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο διάστημα $[-\sqrt{3}, \sqrt{3}]$.
  \item Αν ναι, να βρείτε όλα τα $\xi$ του θεωρήματος.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \hm x$.
\begin{alist}
  \item Να δείξετε ότι η $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο διάστημα $[0, \pi]$.
  \item Να βρείτε το $\xi \in (0, \pi)$ του θεωρήματος.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^3 - 6x^2 + 11x - 6$.
\begin{alist}
  \item Να δείξετε ότι $f(1) = f(2) = f(3) = 0$.
  \item Εφαρμόζοντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση $f'(x) = 0$ έχει τουλάχιστον δύο ρίζες στο $(1, 3)$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = e^x \cdot \hm x$.
\begin{alist}
  \item Να δείξετε ότι $f(0) = f(\pi)$.
  \item Να δείξετε ότι υπάρχει $\xi \in (0, \pi)$ ώστε $\ef\xi = -1$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^2 - 2x$ στο διάστημα $[0, 2]$.
\begin{alist}
  \item Να εξετάσετε αν ικανοποιούνται οι προϋποθέσεις του Θεωρήματος Rolle.
  \item Να βρείτε το $\xi$ του θεωρήματος και να το ερμηνεύσετε γεωμετρικά.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \syn 2x$ στο διάστημα $\left[0, \pi\right]$.
\begin{alist}
  \item Να ελέγξετε αν ικανοποιούνται οι προϋποθέσεις του Θεωρήματος Rolle.
  \item Να βρείτε όλα τα $\xi \in (0, \pi)$ για τα οποία ισχύει $f'(\xi) = 0$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \ln x - \dfrac{x-1}{x+1}$ για $x > 0$.
\begin{alist}
  \item Να δείξετε ότι $f(1) = 0$.
  \item Να υπολογίσετε το $\displaystyle\lim_{x \to +\infty} f(x)$.
  \item Χρησιμοποιώντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση $f'(x) = 0$ έχει τουλάχιστον μία ρίζα στο $(1, +\infty)$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^4 - 2x^2$.
\begin{alist}
  \item Να δείξετε ότι $f(-1) = f(0) = f(1)$.
  \item Πόσες ρίζες έχει η $f'(x) = 0$ στο $(-1, 1)$; Να τις βρείτε.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = (x-1)^2(x+1)^2$.
\begin{alist}
  \item Να βρείτε τις ρίζες της $f$.
  \item Εφαρμόζοντας διαδοχικά το Θεώρημα Rolle, να δείξετε ότι η $f'(x) = 0$ έχει ακριβώς 3 ρίζες.
  \item Να τις υπολογίσετε.
\end{alist}

% --- Θεωρητικές ασκήσεις ---

\askhsh
Έστω $f$ συνεχής στο $[0, 1]$ και παραγωγίσιμη στο $(0, 1)$ με $f(0) = f(1) = 0$.
\begin{alist}
  \item Να δείξετε ότι υπάρχει $\xi \in (0, 1)$ ώστε $f'(\xi) = 0$.
  \item Αν επιπλέον $f'(0) = 2$, μπορούμε να συμπεράνουμε κάτι για το πρόσημο του $f(\xi)$;
\end{alist}

\askhsh
Έστω $f$ συνεχής στο $[a, b]$ και παραγωγίσιμη στο $(a, b)$ με $f(a) = f(b) = 0$. Επίσης, $f(x) > 0$ για κάθε $x \in (a, b)$.
\begin{alist}
  \item Να δείξετε ότι υπάρχει $\xi \in (a, b)$ ώστε $f'(\xi) = 0$.
  \item Να αποδείξετε ότι το σημείο $(\xi, f(\xi))$ είναι τοπικό μέγιστο της $C_f$.
\end{alist}

\askhsh
Έστω $f$ παραγωγίσιμη στο $\mathbb{R}$ με $f(x_1) = f(x_2) = f(x_3)$ για $x_1 < x_2 < x_3$.
\begin{alist}
  \item Να δείξετε ότι η $f'(x) = 0$ έχει τουλάχιστον 2 ρίζες στο $(x_1, x_3)$.
  \item Αν $f''$ υπάρχει, να δείξετε ότι η $f''(x) = 0$ έχει τουλάχιστον 1 ρίζα στο $(x_1, x_3)$.
\end{alist}

\askhsh
Έστω $f, g$ συνεχείς στο $[0, 1]$ και παραγωγίσιμες στο $(0, 1)$ με $f(0) = g(0)$ και $f(1) = g(1)$.
\begin{alist}
  \item Θεωρώντας τη συνάρτηση $h = f - g$, να δείξετε ότι υπάρχει $\xi \in (0, 1)$ με $f'(\xi) = g'(\xi)$.
  \item Να δώσετε γεωμετρική ερμηνεία του αποτελέσματος.
\end{alist}

\askhsh
Έστω $f$ συνεχής στο $[0, 2]$ και παραγωγίσιμη στο $(0, 2)$ με $f(0) = 1$, $f(1) = 2$, $f(2) = 1$.
\begin{alist}
  \item Να δείξετε ότι υπάρχει $\xi \in (0, 2)$ με $f'(\xi) = 0$.
  \item Πόσα τέτοια $\xi$ υπάρχουν τουλάχιστον;
\end{alist}

\askhsh
Έστω $f$ συνεχής στο $[a, b]$ και παραγωγίσιμη στο $(a, b)$. Αν η $C_f$ τέμνει τον άξονα $x'x$ στα σημεία $A(a, 0)$ και $B(b, 0)$:
\begin{alist}
  \item Να δείξετε ότι υπάρχει σημείο της $C_f$ με οριζόντια εφαπτομένη.
  \item Αν $f(x) > 0$ για $x \in (a, b)$, πού βρίσκεται αυτό το σημείο σε σχέση με τον άξονα $x'x$;
\end{alist}

\paragraph{Ομάδα Β: Ύπαρξη ρίζας εξίσωσης μέσω εύρεσης αρχικής}

\askhsh
Δίνεται η εξίσωση $3x^2 + 2x - 4 = 0$.
\begin{alist}
  \item Να βρείτε μια συνάρτηση $F$ της οποίας η παράγωγος ισούται με $3x^2 + 2x - 4$.
  \item Να δείξετε ότι η εξίσωση έχει τουλάχιστον μία ρίζα στο διάστημα $(0, 1)$.
  \item Να μελετήσετε τη συνάρτηση $f(x) = 3x^2 + 2x - 4$ ως προς τη μονοτονία και να δείξετε ότι η εξίσωση έχει ακριβώς μία ρίζα στο $(0, 1)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $e^x - x - 1 = 0$.
\begin{alist}
  \item Να βρείτε μια αρχική συνάρτηση $F$ ώστε $F'(x) = e^x - x - 1$.
  \item Να υπολογίσετε τα $F(0)$ και $\displaystyle\lim_{x \to +\infty} F(x)$.
  \item Χρησιμοποιώντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση έχει τουλάχιστον μία ρίζα στο $(0, +\infty)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $\syn x + x \cdot \hm x = 0$.
\begin{alist}
  \item Να δείξετε ότι $(x \cdot \syn x)' = \syn x - x \cdot \hm x$.
  \item Να υπολογίσετε τα $F(0)$ και $F(\pi)$ για $F(x) = x \cdot \syn x$.
  \item Χρησιμοποιώντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση έχει τουλάχιστον μία ρίζα στο $(0, \pi)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $2x + \syn x = 0$.
\begin{alist}
  \item Να βρείτε μια συνάρτηση $F$ ώστε $F'(x) = 2x + \syn x$.
  \item Να δείξετε ότι η εξίσωση έχει ακριβώς μία ρίζα (την $x = 0$).
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = 2x \cdot e^{x^2}$.
\begin{alist}
  \item Να βρείτε μια αρχική συνάρτηση $F$ της $f$.
  \item Να δείξετε ότι η εξίσωση $f(x) = 0$ έχει μοναδική ρίζα την $x = 0$.
  \item Να μελετήσετε τη $f$ ως προς τη μονοτονία.
\end{alist}

\askhsh
Δίνεται η εξίσωση $e^x(x - 1) + 1 = 0$.
\begin{alist}
  \item Να δείξετε ότι $(x \cdot e^x)' = e^x(x+1)$.
  \item Θεωρώντας κατάλληλη συνάρτηση $F$, να δείξετε ότι η εξίσωση $e^x(x - 1) + 1 = 0$ έχει τουλάχιστον μία ρίζα στο $(0, +\infty)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $\dfrac{1}{x} + \ln x - 1 = 0$ για $x > 0$.
\begin{alist}
  \item Να βρείτε το πεδίο ορισμού της συνάρτησης $f(x) = \dfrac{1}{x} + \ln x - 1$.
  \item Να βρείτε μια αρχική $F$ της $f$ και να υπολογίσετε τα $F(1)$ και $F(e)$.
  \item Χρησιμοποιώντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση έχει τουλάχιστον μία ρίζα στο $(1, e)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $\hm x - x \cdot \syn x = 0$.
\begin{alist}
  \item Να βρείτε μια συνάρτηση $F$ ώστε $F'(x) = \hm x - x \cdot \syn x$.
  \item Χρησιμοποιώντας το Θεώρημα Rolle, να δείξετε ότι η εξίσωση έχει τουλάχιστον μία ρίζα στο διάστημα $(0, \pi)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $x^2 \cdot e^x + 2x \cdot e^x = 0$.
\begin{alist}
  \item Να παραγοντοποιήσετε το αριστερό μέλος.
  \item Να βρείτε τις ρίζες της εξίσωσης.
  \item Να δείξετε ότι η συνάρτηση $F(x) = x^2 \cdot e^x$ είναι γνησίως αύξουσα για $x > 0$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $3x^2 \cdot \ln x + x^2 = 0$ για $x > 0$.
\begin{alist}
  \item Να βρείτε μια συνάρτηση $F$ ώστε $F'(x) = 3x^2 \cdot \ln x + x^2$.
  \item Να δείξετε ότι η εξίσωση έχει μοναδική ρίζα στο $(0, 1)$.
\end{alist}

\]
Τι παρατηρείτε για τη σχέση αυτή;

  \item Να δείξετε ότι υπάρχει $\xi \in (0, c)$ με $f'(\xi) = 0$.
\end{alist}

\askhsh

  \item Για $\lambda = 1$, να υπολογίσετε το $\displaystyle\lim_{x \to 0^+} f(x)$.
\end{alist}

\askhsh
Δίνεται η εξίσωση $\hm x = \lambda$ όπου $\lambda \in \mathbb{R}$.
\begin{alist}
  \item Να βρείτε τις τιμές του $\lambda$ για τις οποίες η εξίσωση έχει ρίζα στο $[0, \pi]$.
  \item Για $\lambda = \dfrac{1}{2}$, να βρείτε τις ρίζες της εξίσωσης στο $[0, 2\pi]$.
  \item Να δείξετε ότι για $|\lambda| < 1$, η εξίσωση $\hm x = \lambda$ έχει άπειρες ρίζες.
\end{alist}

\paragraph{Ομάδα Θ: Σωστό -- Λάθος}

\askhsh
Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (\textbf{Σ}) ή Λανθασμένες (\textbf{Λ}):
\begin{alist}
  \item Αν η $f$ είναι συνεχής στο $[a, b]$ με $f(a) = f(b)$, τότε υπάρχει $\xi \in (a, b)$ με $f'(\xi) = 0$.
  \item Αν η $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο $[a, b]$, τότε το $\xi$ του θεωρήματος είναι μοναδικό.
  \item Αν $f(a) = f(b) = 0$, τότε η $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο $[a, b]$.
  \item Το Θεώρημα Rolle εφαρμόζεται μόνο σε πολυωνυμικές συναρτήσεις.
  \item Αν η $f$ είναι σταθερή στο $[a, b]$, τότε κάθε $x \in (a, b)$ είναι $\xi$ του Θεωρήματος Rolle.
\end{alist}

\askhsh
Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (\textbf{Σ}) ή Λανθασμένες (\textbf{Λ}):
\begin{alist}
  \item Αν μια εξίσωση $f(x) = 0$ έχει $n$ διακεκριμένες ρίζες, τότε η $f'(x) = 0$ έχει τουλάχιστον $n-1$ ρίζες.
  \item Αν η $f'(x) = 0$ έχει ακριβώς 2 ρίζες, τότε η $f(x) = 0$ έχει το πολύ 3 ρίζες.
  \item Για να εφαρμοστεί το Θεώρημα Rolle, η $f$ πρέπει να είναι παραγωγίσιμη στο κλειστό $[a, b]$.
  \item Αν $f(a) \neq f(b)$, τότε δεν μπορεί να υπάρχει $\xi \in (a, b)$ με $f'(\xi) = 0$.
  \item Αν η $f$ είναι παραγωγίσιμη σε κάθε σημείο, τότε είναι και συνεχής.
\end{alist}

\askhsh

  \item 2
  \item 3
  \item δεν υπάρχει
\end{alist}
\end{multicols}

\askhsh
Η εξίσωση $x^5 + x + 1 = 0$ έχει:
\begin{multicols}{4}
\begin{alist}

  \item ακριβώς 1 ρίζα
  \item ακριβώς 3 ρίζες
  \item 5 ρίζες
\end{alist}
\end{multicols}

\askhsh

\begin{multicols}{4}
\begin{alist}

  \item 2 ρίζες
  \item 3 ρίζες
  \item 4 ρίζες
\end{alist}
\end{multicols}

\askhsh

\begin{alist}
  \item Να ελέγξουμε μόνο ότι $f(0) = f(1)$
  \item Να ελέγξουμε συνέχεια στο $[0, 1]$ και παραγωγισιμότητα στο $(0, 1)$
  \item Η $f$ δεν ικανοποιεί τις προϋποθέσεις του θεωρήματος
  \item Να ελέγξουμε μόνο ότι $f'(x) = 0$ έχει ρίζα
\end{alist}

  \item καμία ρίζα
  \item ακριβώς 1 ρίζα
  \item ακριβώς 2 ρίζες
  \item άπειρες ρίζες
\end{alist}
\end{multicols}
