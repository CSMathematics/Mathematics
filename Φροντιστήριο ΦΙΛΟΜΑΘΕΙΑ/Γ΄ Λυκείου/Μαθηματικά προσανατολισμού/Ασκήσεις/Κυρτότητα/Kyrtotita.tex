\paragraph{Έλεγχος κυρτότητας και σημείων καμπής}

\askhsh
Δίνεται η συνάρτηση $f(x)=x^4-6x^2+8x+1$.
\begin{alist}
    \item Να βρείτε την πρώτη παράγωγο $f'(x)$ και τη δεύτερη παράγωγο $f''(x)$.
    \item Να βρείτε τις ρίζες της $f''(x)=0$.
    \item Να μελετήσετε την $f$ ως προς την κυρτότητα και να βρείτε τα σημεία καμπής.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \frac{x}{x^2+1}$.
\begin{alist}
    \item Να αποδείξετε ότι $f''(x) = \frac{2x(x^2-3)}{(x^2+1)^3}$.
    \item Να λύσετε την ανίσωση $f''(x)>0$.
    \item Να βρείτε τα διαστήματα στα οποία η $f$ είναι κοίλη.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = e^{-x^2}$.
\begin{alist}
    \item Να υπολογίσετε τη δεύτερη παράγωγο της $f$.
    \item Να βρείτε τα σημεία στα οποία μηδενίζεται η $f''$.
    \item Να μελετήσετε την $f$ ως προς την κυρτότητα.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x \ln x$, $x>0$.
\begin{alist}
    \item Να βρείτε την $f''(x)$.
    \item Να εξετάσετε αν η $f$ έχει σημεία καμπής.
    \item Να αποδείξετε ότι η $f$ είναι κυρτή σε όλο το πεδίο ορισμού της.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \cos x$ στο διάστημα $[0, 2\pi]$.
\begin{alist}
    \item Να λύσετε την εξίσωση $f''(x)=0$ στο δοσμένο διάστημα.
    \item Να φτιάξετε πίνακα προσήμου της $f''$.
    \item Να βρείτε τα σημεία καμπής της $C_f$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^3 - 3x^2 + 6x - 2$.
\begin{alist}

    \item Να αποδείξετε ότι η $f$ έχει μοναδικό σημείο καμπής.
    \item Να δείξετε ότι στο σημείο καμπής η $f'$ παρουσιάζει ολικό ελάχιστο.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \arctan x$.
\begin{alist}
    \item Να δείξετε ότι $f''(x) = -\frac{2x}{(1+x^2)^2}$.
    \item Να βρείτε το πρόσημο της $f''$ για $x \ne 0$.
    \item Να βρείτε το σημείο καμπής.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \sqrt{x^2+4}$.
\begin{alist}

    \item Να εξετάσετε αν η εξίσωση $f''(x)=0$ έχει λύση.
    \item Να βρείτε το πρόσημο της $f''$ και την κυρτότητα της $f$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \ln(x^2+2)$.

\askhsh

\askhsh

\paragraph{Κυρτότητα και συναρτήσεις πολλαπλού τύπου}

\askhsh
Δίνεται η συνάρτηση $f(x) = \begin{cases} x^2+2x, & x \le 0 \\ 2x-x^2, & x > 0 \end{cases}$.
\begin{alist}
    \item Να αποδείξετε ότι η $f$ είναι συνεχής και παραγωγίσιμη στο $x_0=0$.
    \item Να βρείτε την $f''(x)$ για $x \ne 0$.
    \item Να εξετάσετε αν η $f$ στρέφει τα κοίλα άνω ή κάτω στο $(-\infty, 0)$ και στο $(0, +\infty)$.
    \item Είναι το $0$ σημείο καμπής;
\end{alist}

\askhsh
Δίνεται η συνάρτηση $g(x) = \begin{cases} \sqrt{-x}, & x \le 0 \\ \sqrt{x}, & x > 0 \end{cases}$.
\begin{alist}
    \item Να βρείτε την $g'(x)$ για $x \ne 0$.
    \item Να βρείτε την $g''(x)$ για $x \ne 0$.
    \item Να μελετήσετε την κυρτότητα στα διαστήματα $(-\infty, 0)$ και $(0, +\infty)$.
    \item Έχει η $g$ σημείο καμπής;
\end{alist}

\paragraph{Κυρτότητα και εφαπτόμενη ευθεία}

\askhsh
Δίνεται η συνάρτηση $f(x) = e^x$.
\begin{alist}
    \item Να αποδείξετε ότι η $f$ είναι κυρτή στο $\mathbb{R}$.
    \item Να βρείτε την εξίσωση της εφαπτομένης της $C_f$ στο σημείο $A(0, 1)$.
    \item Να αποδείξετε την ανισότητα $e^x \ge x+1$ για κάθε $x \in \mathbb{R}$.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \ln x$.
\begin{alist}
    \item Να μελετήσετε την $f$ ως προς την κυρτότητα.
    \item Να βρείτε την εφαπτομένη στο $x_0=1$.
    \item Να δείξετε ότι $\ln x \le x-1$ για κάθε $x>0$.
\end{alist}

\askhsh
Δίνεται παραγωγίσιμη συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ η οποία είναι κυρτή.
\begin{alist}
    \item Αν η εφαπτομένη της $C_f$ στο $x_0=2$ είναι η $y=3x-1$, να αποδείξετε ότι $f(x) \ge 3x-1$ για κάθε $x \in \mathbb{R}$.
    \item Να βρείτε το ελάχιστο της συνάρτησης $g(x) = f(x) - 3x$.
    \item Αν επιπλέον $f(2)=5$, να δείξετε ότι $f(3) > 8$.
\end{alist}

\paragraph{Εύρεση παραμέτρων}

\askhsh
Δίνεται η συνάρτηση $f(x) = x^3 + \alpha x^2 + \beta x + 1$.
\begin{alist}
    \item Να βρείτε τη δεύτερη παράγωγο της $f$.
    \item Αν η $C_f$ έχει σημείο καμπής το $M(1, 2)$, να βρείτε τις τιμές των $\alpha, \beta$.
    \item Για τις τιμές που βρήκατε, να χαρακτηρίσετε το είδος της κυρτότητας εκατέρωθεν του 1.
\end{alist}

\askhsh
Δίνεται η συνάρτηση $f(x) = \alpha \ln x + \beta x^2$, $x>0$.
\begin{alist}
    \item Να βρείτε την $f''(x)$.
    \item Να βρείτε τα $\alpha, \beta$ ώστε η $C_f$ να έχει σημείο καμπής στο $x_0=1$ και η εφαπτομένη εκεί να είναι παράλληλη στην ευθεία $y=x$.
    \item Να ελέγξετε αν για αυτές τις τιμές το $x_0=1$ είναι πράγματι σημείο καμπής.
\end{alist}

\paragraph{Ερωτήσεις Σωστό - Λάθος}

\askhsh
Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (\textbf{Σ}) ή Λανθασμένες (\textbf{Λ}):
\begin{alist}
    \item Αν μια συνάρτηση $f$ είναι κυρτή σε ένα διάστημα $\Delta$, τότε η $f'$ είναι γνησίως αύξουσα στο $\Delta$.
    \item Αν $f''(x_0)=0$, τότε το $M(x_0, f(x_0))$ είναι υποχρεωτικά σημείο καμπής της $C_f$.
    \item Αν η $f$ είναι κοίλη στο $\mathbb{R}$, τότε η γραφική της παράσταση βρίσκεται κάτω από την εφαπτομένη της σε οποιοδήποτε σημείο (εκτός του σημείου επαφής).
    \item Η συνάρτηση $f(x)=x^4$ έχει σημείο καμπής το $(0,0)$.
    \item Αν μια συνάρτηση $f$ έχει σημείο καμπής στο $x_0$, και είναι δύο φορές παραγωγίσιμη, τότε ισχύει $f''(x_0)=0$.
    \item Η συνάρτηση $f(x) = |x|$ έχει σημείο καμπής το $(0,0)$.
    \item Αν $f''(x) > 0$ για κάθε $x \in \mathbb{R}$, τότε η εξίσωση $f(x)=0$ έχει το πολύ δύο ρίζες.
    \item Υπάρχει πολυώνυμο 3ου βαθμού χωρίς σημεία καμπής.
    \item Αν μια συνάρτηση είναι κυρτή στο $(a,b)$ και κυρτή στο $(b,c)$, τότε είναι υποχρεωτικά κυρτή στο $(a,c)$.
    \item Η συνάρτηση $f(x)=e^x$ δεν έχει σημεία καμπής.
\end{alist}
