\paragraph{Εύρεση Παραμέτρου}
\askhsh Δίνεται η συνάρτηση $f(x)=(\lambda-2)x+3$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της $f$.
    \item Να βρείτε την τιμή του $\lambda$ ώστε η $f$ να είναι γνησίως αύξουσα.
    \item Αν $\lambda=4$, να βρείτε τα σημεία τομής της γραφικής παράστασης της $f$ με τους άξονες $x'x$ και $y'y$.
\end{alist}



\askhsh Δίνεται η συνάρτηση $f(x)=\dfrac{\lambda}{x}$ με $x>0$.
\begin{alist}
    \item Να βρείτε το $\lambda$ αν η γραφική παράσταση της $f$ διέρχεται από το σημείο $A(1,2)$.
    \item Για την τιμή του $\lambda$ που βρήκατε, να μελετήσετε την $f$ ως προς τη μονοτονία.
    \item Να συγκρίνετε τις τιμές $f(2)$ και $f(3)$.
\end{alist}



\askhsh Δίνεται η συνάρτηση $f(x)=\lambda-|x-3|$.
\begin{alist}
    \item Να βρείτε τη μέγιστη τιμή της $f$ ως συνάρτηση του $\lambda$.
    \item Αν η μέγιστη τιμή της $f$ είναι $5$, να βρείτε το $\lambda$.
    \item Για $\lambda=5$, να σχεδιάσετε τη γραφική παράσταση της $f$.
\end{alist}

\askhsh Δίνονται οι συναρτήσεις $f(x)=x^2+\lambda$ και $g(x)=2$.
\begin{alist}
    \item Να βρείτε το $\lambda$ ώστε οι γραφικές παραστάσεις των $f$ και $g$ να τέμνονται στο σημείο με τετμημένη $x_0=1$.
    \item Για την τιμή του $\lambda$ που βρήκατε, να δείξετε ότι $f(x)\ge 1$ για κάθε $x\in\mathbb{R}$.
    \item Να βρείτε τα σημεία τομής της $C_f$ με τους άξονες.
\end{alist}



\askhsh Δίνεται η συνάρτηση $f(x)=\dfrac{\lambda x+1}{x-\mu}$.
\begin{alist}
    \item Να βρείτε το $\mu$ ώστε το πεδίο ορισμού της $f$ να είναι το $A=\mathbb{R}-\{2\}$.
    \item Αν $\mu=2$, να βρείτε το $\lambda$ ώστε $f(3)=4$.
    \item Για τις τιμές των $\lambda,\mu$ που βρήκατε, να βρείτε τα σημεία τομής της $C_f$ με τους άξονες.
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία και να συγκρίνετε τις τιμές $f(4)$ και $f(3)$.
\end{alist}

\askhsh Δίνεται η συνάρτηση $f(x)=\alpha x+\beta$.

    \item Να μελετήσετε την $f$ ως προς τη μονοτονία.
    \item Να λύσετε την ανίσωση $f(x)>0$.
\end{alist}
