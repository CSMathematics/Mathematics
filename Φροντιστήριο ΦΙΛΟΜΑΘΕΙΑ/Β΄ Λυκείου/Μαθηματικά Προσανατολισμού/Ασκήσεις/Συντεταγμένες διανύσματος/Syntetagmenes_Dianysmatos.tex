\paragraph{Ίσα διανύσματα}
\askhsh Να βρείτε, αν υπάρχουν, τις τιμές των $x,y \in \mathbb{R}$ για τις οποίες τα διανύσματα $\vec{a}$ και $\vec{b}$ είναι ίσα, σε καθεμία από τις παρακάτω περιπτώσεις:
\begin{alist}
    \item $\vec{a} = (x, 2), \vec{b} = (3, y)$
    \item $\vec{a} = (x-1, 4), \vec{b} = (2, y+1)$
    \item $\vec{a} = (2x, y-3), \vec{b} = (4, x)$
    \item $\vec{a} = (x^2, y), \vec{b} = (1, |x|)$
\end{alist}

\askhsh Δίνονται τα διανύσματα $\vec{u}=(2x-y, 4)$ και $\vec{v}=(5, x+y)$. Να βρείτε τα $x,y$ ώστε $\vec{u}=\vec{v}$.
\begin{alist}
    \item Να σχηματίσετε το σύστημα.
    \item Να λύσετε το σύστημα.
    \item Να βρείτε το διάνυσμα $\vec{u}$.
    \item Να επαληθεύσετε το αποτέλεσμα.
\end{alist}

\askhsh Έστω τα διανύσματα $\vec{a}=(\lambda, 2)$ και $\vec{b}=(3, \mu-1)$.
\begin{alist}
    \item Να βρείτε τα $\lambda,\mu$ αν $\vec{a}=\vec{b}$.
    \item Να βρείτε τα $\lambda,\mu$ αν $\vec{a}=-\vec{b}$.
    \item Αν $\vec{a}+\vec{\beta}=(3,2)$, να βρείτε τα $\lambda,\mu$.
    \item Αν $\vec{a}-2\vec{b}=()$, είναι $\mu=3$.
\end{alist}

\askhsh Θεωρούμε τα διανύσματα $\vec{x}=(\alpha^2-1, \beta)$ και $\vec{y}=(3, \alpha)$. Αν $\vec{x}=\vec{y}$:

\begin{alist}
    \item Να δείξετε ότι $\alpha=2$ ή $\alpha=-2$.
    \item Για $\alpha=2$, να βρείτε το $\beta$.
    \item Για $\alpha=-2$, να βρείτε το $\beta$.
    \item Να γράψετε τα διανύσματα για κάθε περίπτωση.
\end{alist}


\paragraph{Μηδενικό και μη μηδενικό διάνυσμα}
\askhsh Να εξετάσετε ποιες από τις παρακάτω προτάσεις είναι σωστές:
\begin{alist}
    \item Το $\vec{0}$ έχει συντεταγμένες $(0,0)$.
    \item Αν $\vec{a}=(x,y)$ και $x^2+y^2=0$, τότε $\vec{a}=\vec{0}$.
    \item Αν $\vec{a}\neq\vec{0}$, τότε $x\neq 0$ ή $y\neq 0$.
    \item Το διάνυσμα $(0,5)$ είναι το μηδενικό.
\end{alist}

\askhsh Δίνεται το διάνυσμα $\vec{a}=(\lambda-2, \lambda^2-4)$.

\begin{alist}
    \item Να βρείτε το $\lambda$ ώστε $\vec{a}=\vec{0}$.
    \item Να βρείτε το $\lambda$ ώστε $\vec{a}\neq\vec{0}$.
    \item Για $\lambda=2$, ποιο είναι το διάνυσμα;
    \item Για $\lambda=0$, ποιο είναι το διάνυσμα;
\end{alist}

\askhsh Δίνονται τα διανύσματα $\vec{u}=(2k-1, k+3)$ και $\vec{v}=(k^2, 1)$.

\begin{alist}
    \item Να βρείτε το $k$ ώστε το $\vec{u}$ να είναι μηδενικό.
    \item Να βρείτε το $k$ ώστε το $\vec{v}$ να είναι μηδενικό.
    \item Υπάρχει τιμή του $k$ ώστε $\vec{u}=\vec{v}=\vec{0}$;
    \item Να βρείτε το $k$ ώστε $\vec{u}=(1,4)$.
\end{alist}
\askhsh Έστω $\vec{w} = (x^2-3x+2, y^2-4)$.

\begin{alist}
    \item Να βρείτε τα $x$ ώστε η τετμημένη να είναι $0$.
    \item Να βρείτε τα $y$ ώστε η τεταγμένη να είναι $0$.
    \item Να βρείτε τα $x,y$ ώστε $\vec{w}=\vec{0}$.
    \item Να βρείτε ένα ζεύγος $(x,y)$ ώστε $\vec{w}\neq\vec{0}$.
\end{alist}

\paragraph{Γραμμικός συνδυασμός διανυσμάτων}

    \item $\vec{z} = 3(\vec{a}-2\vec{b})$
\end{alist}

\begin{alist}
    \item $x\vec{u} + y\vec{v} = (0, 13)$
    \item $x\vec{u} + y\vec{v} = (5, 1)$
    \item $x\vec{u} - \vec{v} = \vec{v}$
    \item $2\vec{u} + y\vec{v} = (1, 11)$
\end{alist}


\askhsh Να γραφεί το διάνυσμα $\vec{w}=(-4, 1)$ ως γραμμικός συνδυασμός των $\vec{a}=(1, 2)$ και $\vec{b}=(2, -1)$.

\begin{alist}
    \item Γράφουμε $\vec{w} = \kappa \vec{a} + \lambda \vec{b}$.
    \item Καταστρώστε το σύστημα για τα $\kappa, \lambda$.
    \item Λύστε το σύστημα.
    \item Επαληθεύστε το αποτέλεσμα.
\end{alist}


\askhsh Αν $\vec{a}=\vec{i}+2\vec{j}$, $\vec{b}=3\vec{i}- \vec{j}$ και $\vec{\gamma}=2\vec{i}+5\vec{j}$:

\begin{alist}

    \item Να υπολογίσετε το $\vec{u} = 2\vec{a}+\vec{b}$.
    \item Να βρείτε τα $\kappa, \lambda$ ώστε $\vec{\gamma} = \kappa \vec{a} + \lambda \vec{b}$.
    \item Να δείξετε ότι τα $\vec{a}, \vec{b}$ δεν είναι παράλληλα.
\end{alist}
