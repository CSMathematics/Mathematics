\begin{thema}
\item\mbox{}\\\vspace{-2mm}
\begin{erwthma}
\item Δίνονται δύο διανύσματα $\vec{a},\vec{\beta}\nparallel y'y$ με συντελεστές διεύθυνσης $\lambda_1,\lambda_2$ αντίστοιχα. Να αποδείξετε ότι \[\vec{a}\parallel\vec{\beta}\Leftrightarrow \lambda_1=\lambda_2\]\monades{15}
\item \swstolathospan
\begin{alist}
\item Τα διανύσματα $\vec{a}=(-1,2)$ και $\vec{\beta}=(2,-4)$ έχουν ίσους συντελεστές διεύθυνσης.
\item Το μέτρο ενός διανύσματος $\overrightarrow{AB}$ δίνεται από τον τύπο $\left|\overrightarrow{AB}\right|=\sqrt{(x_A-x_B)^2+(y_A-y_B)^2}$.
\item Αν ένα διάνυσμα $\vec{a}=(x,y)$ είναι παράλληλο στον άξονα $x'x$ τότε $x=0$.
\item Αν για δύο διανύσματα $\vec{a},\vec{\beta}$ ισχύει $\vec{a}=\lambda\vec{\beta}$, τότε $\det{\left(\vec{a},\vec{\beta}\right)=0}$.
\item Το διάνυσμα $\vec{a}=(\kappa-1,\kappa-1)$ έχει έχει συντελεστή διεύθυνσης $\lambda=1$ για κάθε τιμή του $\kappa\in\mathbb{R}$.
\end{alist}
\end{erwthma}
\item\mbox{}\\
Δίνονται τα διανύσματα $\vec{a}=(-4,5),\vec{\beta}=(-1,3)$ και $\vec{\gamma}=(\lambda-1,2\lambda)$, με $\left|\vec{\gamma}\right|=2\sqrt{10}$.
\begin{erwthma}
\item Να βρείτε τον πραγματικό αριθμό $\lambda$. \monades{10}
\item Αν $\lambda=3$ να γράψετε το διάνυσμα $\vec{\gamma}$ ως γραμμικό συνδυασμό των $\vec{a}$ και $\vec{\beta}$.\monades{15}
\end{erwthma}
\item\mbox{}\\
Δίνονται τα σημεία $A(-3,4),B(-2,-3)$ και $\varGamma(3-a,-a)$. Αν γνωρίζουμε ότι το διάνυσμα $\overrightarrow{B\varGamma}$ έχει συντελεστή διεύθυνσης $\lambda_{\overrightarrow{B\varGamma}}=\frac{1}{2}$.
\begin{erwthma}
\item να αποδείξτε ότι $a=1$.\monades{7}
\item να βρείτε τις συντεταγμένες των διανυσμάτων $\overrightarrow{AB}$ και $\overrightarrow{\varGamma M}$, όπου $M$ το μέσο του $AB$.\monades{8}
\item να αποδείξετε ότι το τρίγωνο $AB\varGamma$ είναι ισοσκελές.\monades{10}
\end{erwthma}
