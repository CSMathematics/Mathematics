% --- ΘΕΜΑ Α ---
\begin{thema}
\item \textbf{Θεωρία}
\begin{erwthma}
    \item Να συμπληρώσετε τις παρακάτω ταυτότητες: \monades{10}
    \item \begin{multicols}{3}
    \begin{enumerate}[label=\alph*.]
        \item $(a+\beta)^2 = $ \vspace{2mm}
        \item $a^2-\beta^2 = $ \vspace{2mm}
        \item $(a-\beta)^3 = $ \vspace{2mm}
    \end{enumerate}
    \end{multicols}
    \item Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (\textbf{Σ}) ή Λανθασμένες (\textbf{Λ}): \monades{15}
    \begin{enumerate}[label=\alph*.]
        \item Η διαδικασία με την οποία μια παράσταση από άθροισμα μετατρέπεται σε γινόμενο λέγεται παραγοντοποίηση.
        \item Ισχύει ότι $(a-\beta)^2 = a^2 - \beta^2$.
        \item Το τριώνυμο $x^2+2x+1$ είναι ανάπτυγμα τετραγώνου.
        \item Στην παραγοντοποίηση με ομαδοποίηση, όλοι οι όροι πρέπει να έχουν κοινό παράγοντα.
        \item Η παράσταση $x^2-9$ παραγοντοποιείται ως $(x-3)(x+3)$.
    \end{enumerate}
\end{erwthma}

% --- ΘΕΜΑ Β ---
\item \textbf{Βασικές Εφαρμογές}
\begin{erwthma}
    \item Να αναπτύξετε τις παρακάτω ταυτότητες: \monades{12.5}
    \begin{multicols}{4}
    \begin{enumerate}[label=\alph*.]
        \item $(3x+2)^2$
        \item $(x-4)^3$
        \item $(2x-\sqrt{3})(2x+\sqrt{3})$
        \item $(y^2-1)^2$
    \end{enumerate}
    \end{multicols}
    \item Να παραγοντοποιήσετε τις παρακάτω παραστάσεις: \monades{12.5}
    \begin{multicols}{2}
    \begin{enumerate}[label=\alph*.]
        \item $5x^2 - 10x$
        \item $x^2 - 16$
        \item $3a + 3\beta + ax + \beta x$
        \item $x^2 - 25y^2$
    \end{enumerate}
    \end{multicols}
\end{erwthma}

% --- ΘΕΜΑ Γ ---
\item \textbf{Σύνθετες Ασκήσεις}
\begin{erwthma}
    \item Να παραγοντοποιήσετε πλήρως τις παρακάτω παραστάσεις: \monades{15}
    \begin{enumerate}[label=\alph*.]
        \item $x^3 - x^2 + x - 1$
        \item $x^2 - 5x + 6$
        \item $3x^2 - 12$
    \end{enumerate}
    \item Να μετατρέψετε τα παρακάτω κλάσματα σε ισοδύναμα με ρητό παρονομαστή: \monades{10}
    \begin{multicols}{2}
    \begin{enumerate}[label=\alph*.]
        \item $\dfrac{1}{\sqrt{5}-1}$
        \item $\dfrac{2}{\sqrt{7}+\sqrt{3}}$
    \end{enumerate}
    \end{multicols}
\end{erwthma}

% --- ΘΕΜΑ Δ ---
\item \textbf{Σύνθεση \& Εξισώσεις}
\begin{erwthma}
    \item Να αποδείξετε την ταυτότητα: \monades{12}
    \[ (a+\beta)^2 - (a-\beta)^2 = 4a\beta \]
    \item Να λύσετε τις παρακάτω εξισώσεις με τη βοήθεια της παραγοντοποίησης: \monades{13}
    \begin{enumerate}[label=\alph*.]
        \item $x^2 - 4x = 0$
        \item $x^2 - 9 = 0$
        \item $(x-2)(x+3) = 0$
    \end{enumerate}
\end{erwthma}
\end{thema}
