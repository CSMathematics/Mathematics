\paragraph{ΘΕΩΡΙΑ}

\begin{enumerate}
\item \textbf{Λύση}
\begin{rlist}
\item Ρητή αλγεβρική παράσταση ονομάζεται κάθε αλγεβρική παράσταση που περιέχει μία τουλάχιστον μεταβλητή στον παρονομαστή της.
\item Για να ορίζεται μια ρητή αλγεβρική παράσταση, πρέπει οι τιμές των μεταβλητών να είναι τέτοιες, ώστε ο παρονομαστής να είναι διάφορος του μηδενός (να μην μηδενίζεται).
\item Μια ρητή αλγεβρική παράσταση μπορεί να απλοποιηθεί αν ο αριθμητής και ο παρονομαστής της είναι γινόμενα παραγόντων και έχουν κοινιούς παράγοντες.
\item Οι περιορισμοί που αφορούν μια ρητή αλγεβρική παράσταση είναι συνήθως περισσότεροι, ή στην καλύτερη περίπτωση ίδιοι, μ' αυτούς που αφορούν την απλοποιημένη μορφή της. Κατά την απλοποίηση μπορεί να χαθεί πληροφορία για το πεδίο ορισμού, γι' αυτό τους περιορισμούς τους βρίσκουμε \textbf{πάντα στην αρχική παράσταση}.
\end{rlist}

\item \textbf{Λύση}
\begin{rlist}
\item \textbf{Σ}
\item \textbf{Λ} (Ο παρονομαστής 3 είναι αριθμός, όχι μεταβλητή).
\item \textbf{Σ}
\item \textbf{Σ} (Το $x^2+1$ είναι πάντα θετικό, άρα ποτέ μηδέν).
\item \textbf{Λ} (Ισχύει μόνο για $x \neq 0$).
\item \textbf{Σ}
\item \textbf{Σ}
\item \textbf{Σ} (Ο αριθμητής μηδέν και ο παρονομαστής όχι).
\item \textbf{Σ} ($x-y = -(y-x)$).
\item \textbf{Λ} ($x^2-1 = (x-1)(x+1)$, άρα το Ε.Κ.Π. είναι $(x-1)(x+1) = x^2-1$. Η πρόταση φαίνεται σωστή, αλλά έλέγχουμε αν εννοεί κάτι άλλο. Είναι σωστή).
\item \textbf{Λ} (Εξαιρούνται οι τιμές που μηδενίζουν τον παρονομαστή).
\item \textbf{Σ}

\paragraph{ΑΣΚΗΣΕΙΣ}

\begin{center}
\textbf{ΠΕΡΙΟΡΙΣΜΟΙ}
\end{center}

\begin{enumerate}[resume]
\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item Πρέπει $x-3 \neq 0 \Rightarrow \mathbf{x \neq 3}$.
\item Πρέπει $x+4 \neq 0 \Rightarrow \mathbf{x \neq -4}$.
\item Πρέπει $2x-6 \neq 0 \Rightarrow 2x \neq 6 \Rightarrow \mathbf{x \neq 3}$.
\item Πρέπει $x^2-1 \neq 0 \Rightarrow x^2 \neq 1 \Rightarrow \mathbf{x \neq \pm 1}$.
\item Πρέπει $x(x-2) \neq 0$
$\Rightarrow x \neq 0$ και $x-2 \neq 0 \Rightarrow \mathbf{x \neq 0, x \neq 2}$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{rlist}
\item Πρέπει $x^2-5x+6 \neq 0$.
Παραγοντοποιούμε: $(x-2)(x-3) \neq 0 \Rightarrow \mathbf{x \neq 2, x \neq 3}$.
\item Πρέπει $x^2-4x+4 \neq 0 \Rightarrow (x-2)^2 \neq 0 \Rightarrow x-2 \neq 0 \Rightarrow \mathbf{x \neq 2}$.
\item Πρέπει $x^3-4x \neq 0 \Rightarrow x(x^2-4) \neq 0 \Rightarrow x(x-2)(x+2) \neq 0$.
Άρα $\mathbf{x \neq 0, x \neq 2, x \neq -2}$.
\item Πρέπει $x^2+1 \neq 0$. Ισχύει για κάθε $\mathbf{x \in \mathbb{R}}$ (αφού $x^2 \ge 0$).
\item Πρέπει $|x|-3 \neq 0 \Rightarrow |x| \neq 3 \Rightarrow \mathbf{x \neq 3, x \neq -3}$.
\end{rlist}

\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item $x \neq 0$.
\item $x-1 \neq 0 \Rightarrow x \neq 1$.
\item $2y-4 \neq 0 \Rightarrow 2y \neq 4 \Rightarrow y \neq 2$.
\item $4-z \neq 0 \Rightarrow z \neq 4$.
\item $3x-6 \neq 0 \Rightarrow 3x \neq 6 \Rightarrow x \neq 2$.
\item $x-y \neq 0 \Rightarrow x \neq y$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item $x^2 \neq 0 \Rightarrow x \neq 0$.
\item $x(x-3) \neq 0 \Rightarrow x \neq 0$ και $x \neq 3$.
\item $x(x+1) \neq 0 \Rightarrow x \neq 0$ και $x \neq -1$.
\item $(x-2)(x+1) \neq 0 \Rightarrow x \neq 2$ και $x \neq -1$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item $x^2-x \neq 0 \Rightarrow x(x-1) \neq 0 \Rightarrow x \neq 0, x \neq 1$.
\item $x^3-4x^2 \neq 0 \Rightarrow x^2(x-4) \neq 0 \Rightarrow x \neq 0, x \neq 4$.
\item $2x-x^2 \neq 0 \Rightarrow x(2-x) \neq 0 \Rightarrow x \neq 0, x \neq 2$.
\item $xy-y \neq 0 \Rightarrow y(x-1) \neq 0 \Rightarrow y \neq 0, x \neq 1$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{rlist}
\item $x^2-4 \neq 0 \Rightarrow x^2 \neq 4 \Rightarrow x \neq \pm 2$.
\item $4y^2-25 \neq 0 \Rightarrow 4y^2 \neq 25 \Rightarrow y^2 \neq \frac{25}{4} \Rightarrow y \neq \pm \frac{5}{2}$.
\item $3x^2-18 \neq 0 \Rightarrow 3(x^2-6) \neq 0 \Rightarrow x^2 \neq 6 \Rightarrow x \neq \pm \sqrt{6}$.
\item $x^3-16x \neq 0 \Rightarrow x(x^2-16) \neq 0 \Rightarrow x \neq 0, x \neq \pm 4$.
\item $y^2-x^2 \neq 0 \Rightarrow (y-x)(y+x) \neq 0 \Rightarrow y \neq x, y \neq -x$.
\item $z^4-1 \neq 0 \Rightarrow (z^2-1)(z^2+1) \neq 0 \Rightarrow z^2 \neq 1 \Rightarrow z \neq \pm 1$ (το $z^2+1 \neq 0$ ισχύει πάντα).
\end{rlist}

\item \textbf{Λύση}
\begin{rlist}
\item $x^2-4x+4 \neq 0 \Rightarrow (x-2)^2 \neq 0 \Rightarrow x \neq 2$.
\item $x^2+2x+1 \neq 0 \Rightarrow (x+1)^2 \neq 0 \Rightarrow x \neq -1$.
\item $25y^2+10y+1 \neq 0 \Rightarrow (5y+1)^2 \neq 0 \Rightarrow 5y \neq -1 \Rightarrow y \neq -\frac{1}{5}$.
\item $z^3-6z^2+9z \neq 0 \Rightarrow z(z^2-6z+9) \neq 0 \Rightarrow z(z-3)^2 \neq 0 \Rightarrow z \neq 0, z \neq 3$.
\end{rlist}

\item \textbf{Λύση}
\begin{enumerate}
\item Περιορισμοί: $3x-9 \neq 0 \Rightarrow x \neq 3$.
$\dfrac{x-3}{3(x-3)} = \dfrac{1}{3}$.
\item Περιορισμοί: $4-x^2 \neq 0 \Rightarrow x \neq \pm 2$.
$\dfrac{2-x}{(2-x)(2+x)} = \dfrac{1}{x+2}$.
\item Περιορισμοί: $y(y^2-1) \neq 0 \Rightarrow y \neq 0, y \neq \pm 1$.
$\dfrac{y^2}{y(y-1)(y+1)} = \dfrac{y}{(y-1)(y+1)}$.
\item Περιορισμοί: $z(16z^2-9) \neq 0 \Rightarrow z \neq 0, z \neq \pm \frac{3}{4}$.
$\dfrac{z(4z+3)}{z(4z-3)(4z+3)} = \dfrac{1}{4z-3}$.
\end{enumerate}

\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item $\dfrac{2(x+2)}{(x+2)^2} = \dfrac{2}{x+2}$.
\item $\dfrac{(y-5)(y+5)}{y^2(y-5)} = \dfrac{y+5}{y^2}$.
\item $\dfrac{(2z-3)(2z+3)}{(2z-3)^2} = \dfrac{2z+3}{2z-3}$.
\item $\dfrac{(t+5)^2}{2(t^2-25)} = \dfrac{(t+5)^2}{2(t-5)(t+5)} = \dfrac{t+5}{2(t-5)}$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{multicols}{2}
\begin{rlist}
\item $\dfrac{(x+y)^2}{(x-y)(x+y)} = \dfrac{x+y}{x-y}$.
\item $\dfrac{2(2z-y)}{3y(y-2z)} = \dfrac{-2(y-2z)}{3y(y-2z)} = -\dfrac{2}{3y}$.
\item $\dfrac{xy(y-x)}{(y-x)^2} = \dfrac{xy}{y-x}$.
\item $\dfrac{(z-3x)(z+3x)}{2x(z-3x)} = \dfrac{z+3x}{2x}$.
\end{rlist}
\end{multicols}

\item \textbf{Λύση}
\begin{enumerate}
\item Αριθμητής: Τριώνυμο $x^2-3x+2 = (x-1)(x-2)$.
Άρα $\dfrac{(x-1)(x-2)}{(x-1)(x-2)} = 1$.
\item Παρονομαστής: $z^2-z-12 = (z-4)(z+3)$.
Άρα $\dfrac{z(z-4)}{(z-4)(z+3)} = \dfrac{z}{z+3}$.
\item Αριθμητής: $(x-5)^2$. Παρονομαστής: $(x-5)(x+2)$.
Άρα $\dfrac{(x-5)^2}{(x-5)(x+2)} = \dfrac{x-5}{x+2}$.
\item Αριθμητής: $(y-7)(y+2)$ (Ρίζες 7, -2). Παρονομαστής: $(y-7)(y+7)$.
Άρα $\dfrac{(y-7)(y+2)}{(y-7)(y+7)} = \dfrac{y+2}{y+7}$.
\end{enumerate}

\begin{center}
\textbf{ΣΥΝΘΕΤΕΣ ΑΣΚΗΣΕΙΣ}
\end{center}

\item \textbf{Λύση}
\begin{enumerate}
\item Περιορισμοί: $x \neq 0, x \neq -1$.
$= \dfrac{x}{x+1} \cdot \dfrac{(x-1)(x+1)}{x} = x-1$.
\item Περιορισμοί: $x \neq 0, x \neq -2$.
$= \dfrac{(x-2)(x+2)}{x} \cdot \dfrac{2x}{x+2} = 2(x-2)$.
\item Περιορισμοί: $x \neq 0, x \neq 1$.
Ε.Κ.Π. = $x(x-1)$.
$= \dfrac{2(x-1)}{x(x-1)} + \dfrac{3x}{x(x-1)} = \dfrac{2x-2+3x}{x(x-1)} = \dfrac{5x-2}{x(x-1)}$.
\item Περιορισμοί: $x \neq \pm 1$.
$= \dfrac{x}{(x-1)(x+1)} - \dfrac{x+1}{(x-1)(x+1)} = \dfrac{x-(x+1)}{(x-1)(x+1)} = \dfrac{-1}{x^2-1}$.
\item Περιορισμοί: $x \neq 0, x \neq -1$.
$= \dfrac{x+1}{x} \cdot \dfrac{x}{x+1} = 1$.
\end{enumerate}

\item \textbf{Λύση}
\begin{rlist}
\item $x^2-8x+16 \neq 0 \Rightarrow (x-4)^2 \neq 0 \Rightarrow \mathbf{x \neq 4}$.
$A = \dfrac{(x-4)(x+4)}{(x-4)^2} = \dfrac{x+4}{x-4}$.
\item $x(x+1) \neq 0 \Rightarrow \mathbf{x \neq 0, x \neq -1}$.
$P(x) = \dfrac{x(x^2-1)}{x(x+1)} = \dfrac{x(x-1)(x+1)}{x(x+1)} = x-1$.
\item Αριθμητής: $a(x+y)+b(x+y) = (x+y)(a+b)$.
Άρα $\dfrac{(x+y)(a+b)}{x+y} = a+b$ (για $x+y \neq 0$).
\item $B(x) = (x-3)(x+3)$. $A(x) = (x-2)(x-3)$.
Άρα $\dfrac{(x-2)(x-3)}{(x-3)(x+3)} = \dfrac{x-2}{x+3}$ (για $x \neq \pm 3$).
\end{rlist}

\item \textbf{Λύση}
\begin{enumerate}
\item $A(x) = 2(x-3)$, $B(x)=(x-3)(x+3)$. Περιορισμός $x \neq \pm 3$.
$\dfrac{2(x-3)}{(x-3)(x+3)} = \dfrac{2}{x+3}$.
\item $\varGamma(x) = (x-3)(x-4)$. Περιορισμός $x \neq 3, x \neq 4$.
$\dfrac{(x-3)(x+3)}{(x-3)(x-4)} = \dfrac{x+3}{x-4}$.
\item (Είναι ίδιο με το i - Τυπογραφικό).
\end{enumerate}

\item \textbf{Λύση}
\begin{rlist}
\item $P(2) = \dfrac{4-2}{2^2-16} = \dfrac{2}{4-16} = -\dfrac{2}{12} = -\dfrac{1}{6}$.
$P(-1) = \dfrac{4-(-1)}{(-1)^2-16} = \dfrac{5}{1-16} = -\dfrac{5}{15} = -\dfrac{1}{3}$.
$P(3) = \dfrac{4-3}{3^2-16} = \dfrac{1}{9-16} = -\dfrac{1}{7}$.
\item Για να ορίζεται πρέπει $B(x) \neq 0 \Rightarrow x^2-16 \neq 0 \Rightarrow x \neq \pm 4$.
Άρα το $P(4)$ \textbf{δεν ορίζεται} (διότι $4^2-16=0$).
\item $P(x) = \dfrac{4-x}{(x-4)(x+4)} = \dfrac{-(x-4)}{(x-4)(x+4)} = -\dfrac{1}{x+4}$. (Για $x \neq \pm 4$).
Ακόμα και μετά την απλοποίηση, ο περιορισμός $x \neq 4$ ισχύει από την αρχική. Άρα το $P(4)$ συνεχίζει να \textbf{μην ορίζεται}.
\end{rlist}

\item \textbf{Λύση}
\begin{enumerate}
\item $B(x) \neq 0 \Rightarrow x \neq \pm 2$.
Αριθμητής: $x^2-4 - (4x-8) = x^2-4-4x+8 = x^2-4x+4 = (x-2)^2$.
$\dfrac{(x-2)^2}{(x-2)(x+2)} = \dfrac{x-2}{x+2}$.
\item $A(x) \neq 0 \Rightarrow 4(x-2) \neq 0 \Rightarrow x \neq 2$.
Αριθμητής: $x^2-4 - (x^2-5x+6) = x^2-4-x^2+5x-6 = 5x-10 = 5(x-2)$.
$\dfrac{5(x-2)}{4(x-2)} = \dfrac{5}{4}$.
\item $B(x) \neq 0 \Rightarrow x \neq \pm 2$.
Αριθμητής: $4x-8 + x^2-5x+6 = x^2-x-2 = (x-2)(x+1)$.
$\dfrac{(x-2)(x+1)}{(x-2)(x+2)} = \dfrac{x+1}{x+2}$.
\item Πρέπει $B(-2x) \neq 0 \Rightarrow (-2x)^2-4 \neq 0 \Rightarrow 4x^2-4 \neq 0 \Rightarrow x^2 \neq 1 \Rightarrow x \neq \pm 1$.
$A(2x) = 4(2x)-8 = 8x-8 = 8(x-1)$.
$B(-2x) = 4x^2-4 = 4(x^2-1) = 4(x-1)(x+1)$.
$\dfrac{8(x-1)}{4(x-1)(x+1)} = \dfrac{2}{x+1}$.
\end{enumerate}
