        \tkzDefPoint(0,\s){A} % A

        \tkzDefPoint(\s,2*\s){K}

        % Draw the square
        \tkzDrawPolygon(A,B,G,D)
        
        % Draw the line segments DK and GK, which represent the lines DM and GB extended to K.
        \tkzDrawSegment(D,K)
        \tkzDrawSegment(G,K)

        % Draw and label points
        \tkzDrawPoints(A,B,G,D,M,K)

\begin{alist}
\item Τα τρίγωνα $MKB$ και $\Delta K\Gamma$ είναι όμοια. \monades{5}
\item $(MKB) = \frac{1}{4} (\Delta K\Gamma)$. \monades{5}
\item $(M\Gamma B\Delta) = \frac{3}{4} (AB\Gamma\Delta)$. \monades{10}
\item Αν $(M\Gamma B\Delta) = 75 \, \text{m}^2$ να υπολογίσετε την πλευρά του τετραγώνου. \monades{5}
\end{alist}
