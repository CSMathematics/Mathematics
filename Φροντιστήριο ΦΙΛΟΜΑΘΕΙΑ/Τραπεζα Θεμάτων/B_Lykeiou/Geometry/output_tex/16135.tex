Δίνεται ορθογώνιο τρίγωνο $AB\Gamma$ με υποτείνουσα $B\Gamma = 10$ και έστω ότι $\Delta$ είναι η προβολή της κορυφής $A$ στην $B\Gamma$.
\begin{alist}
    \item Αν $\Delta B = 2$ να υπολογίσετε:
    \begin{rlist}
        \item το ύψος $A\Delta$ του τριγώνου $AB\Gamma$ , \monades{7}
        \item το εμβαδόν του τριγώνου $AB\Gamma$. \monades{5}
    \end{rlist}
    \item Υποθέστε ότι το σημείο $A$ κινείται πάνω στο ημικύκλιο με διάμετρο την $B\Gamma$.
    \begin{rlist}
        \item Να αποδείξετε ότι το εμβαδόν του τριγώνου $AB\Gamma$ είναι $(AB\Gamma) = 5A\Delta$. \monades{7}
        \item Θεωρήστε τον παρακάτω ισχυρισμό:
        «Για όλες τις θέσεις του $A$ πάνω στο ημικύκλιο με διάμετρο την $B\Gamma$, το εμβαδόν του τριγώνου $AB\Gamma$ δεν υπερβαίνει το $25$».
        Είναι αληθής ή Ψευδής ο παραπάνω ισχυρισμός; Να αιτιολογήσετε την απάντησή σας. \monades{6}
    \end{rlist}
\end{alist}
