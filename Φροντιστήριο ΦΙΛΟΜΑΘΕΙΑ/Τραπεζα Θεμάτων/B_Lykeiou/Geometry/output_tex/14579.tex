    \item 7.7. Θεώρημα του Θαλή.
\end{ylh}
\begin{wrapfigure}[9]{r}{5.5cm}
    \centering
    \begin{tikzpicture}[scale=0.8]
        % Define points A, B, Gamma (C)
        \tkzDefPoint(0,0){B}
        \tkzDefPoint(6,0){C}
        \tkzDefPoint(2,4){A}

        % Define Delta (D) on AB such that AB = 3AD.
        % This implies AD = (1/3)AB, so BD = (2/3)AB.
        % So D is 2/3 of the way from B to A.
        \tkzDefPointWith[linear, K=2/3](B,A) \tkzGetPoint{D}

        % Define E on BC such that DE is parallel to AC.
        % By Thales' theorem, BD/DA = BE/EC. Since BD/DA = 2, then BE/EC = 2.
        % So E is 2/3 of the way from B to C.
        \tkzDefPointWith[linear, K=2/3](B,C) \tkzGetPoint{E}

        % Define Zeta (Z) on AC.
        % For part (b), AG = 3.9 and GZ = 1.3, so GZ = (1/3)AG.
        % So Z is 1/3 of the way from C to A.
        \tkzDefPointWith[linear, K=1/3](C,A) \tkzGetPoint{Z}

        % Draw triangle ABG
        \tkzDrawPolygon(A,B,C)

        % Draw segments DE and EZ
        \tkzDrawSegment(D,E)
        \tkzDrawSegment(E,Z)

        % Draw points
        \tkzDrawPoints(A,B,C,D,E,Z)

        % Labels
        \tkzLabelPoint[above left](A){A}
        \tkzLabelPoint[below left](B){B}
        \tkzLabelPoint[below right](C){$\Gamma$}
        \tkzLabelPoint[above left](D){$\Delta$}
        \tkzLabelPoint[below](E){E}
        \tkzLabelPoint[right](Z){Z}
    \end{tikzpicture}
\end{wrapfigure}
Δίνεται το τρίγωνο $ΑΒΓ$ και τα σημεία $Δ, Ε$ και $Ζ$ των πλευρών του $ΑΒ, ΒΓ$ και $ΑΓ$ αντίστοιχα, ώστε η $ΔΕ$ να είναι παράλληλη στην $ΑΓ$. Επίσης $ΑΒ = 3ΑΔ$.
\begin{alist}
\item Να βρείτε τους λόγους $\frac{ΒΔ}{ΑΔ}$ και $\frac{ΒΕ}{ΕΓ}$. \monades{15}
\item Αν επιπλέον γνωρίζετε ότι $ΑΓ = 3,9$ και $ΓΖ = 1,3$ να αποδείξετε ότι η $ΖΕ$ είναι παράλληλη της $ΑΒ$. \monades{10}
\end{alist}
