\begin{alist}
\item Να σχεδιάσετε την προβολή $\Alpha\Delta$, της $\Alpha\Beta$ στην $\Alpha\Gamma$ και να υπολογίσετε το μήκος της. \monades{13}
\item Αν $\Alpha\Delta = 3$, να υπολογίσετε το μήκος του ύψους $\Beta\Delta$. \monades{12}
\end{alist}
