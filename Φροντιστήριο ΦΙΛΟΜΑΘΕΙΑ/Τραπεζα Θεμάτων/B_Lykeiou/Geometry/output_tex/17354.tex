Στο παρακάτω τρίγωνο $\Delta EZ$ φέρουμε τα ύψη του $\Delta K$ και $ZI$.
\begin{alist}
    \item Να συμπληρώσετε τα κενά στις παρακάτω προτάσεις: \monades{15}
    \begin{rlist}
        \item Η προβολή της πλευράς $\Delta E$ στην πλευρά $EZ$ είναι το τμήμα $EK$.
        \item Η προβολή της πλευράς $\Delta Z$ στην πλευρά $EZ$ είναι το τμήμα $KZ$.
        \item Το τμήμα $\Delta I$ είναι η προβολή της πλευράς $\Delta Z$ στην πλευρά $\Delta E$.
        \item Το τμήμα $EI$ είναι η προβολή της πλευράς $ZE$ στην πλευρά $\Delta E$.
        \item $\Delta Z^2 = \Delta E^2 + EZ^2 + 2 \cdot EZ \cdot EK$.
        \item $EZ^2 = \Delta E^2 + \Delta Z^2 - 2 \cdot \Delta E \cdot \Delta I$.
    \end{rlist}
    \item Αν $\Delta E = 2$, $EZ = 4$ και $\Delta Z = 5$, να υπολογίσετε το μήκος του τμήματος $\Delta I$. \monades{10}
\end{alist}
