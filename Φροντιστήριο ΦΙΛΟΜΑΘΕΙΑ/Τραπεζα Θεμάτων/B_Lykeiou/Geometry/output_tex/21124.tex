    \item 9.4. Γενίκευση του Πυθαγόρειου θεωρήματος
    \item 10.3. Εμβαδόν βασικών ευθύγραμμων σχημάτων.
\end{ylh}
\begin{alist}
\item Δίνεται ισοσκελές τρίγωνο $AB\Gamma$ με πλευρές $\alpha = 40$, $\beta = 25$, $\gamma = 25$ και αντίστοιχα ύψη $v_{\alpha}$, $v_{\beta}$, $v_{\gamma}$. Να αποδείξετε ότι:
    \begin{rlist}
    \item Το τρίγωνο $AB\Gamma$ είναι αμβλυγώνιο. \monades{6}
    \item Το εμβαδόν του τριγώνου $AB\Gamma$ είναι $E = 300$ και τα ύψη του είναι $v_{\alpha} = 15$ και $v_{\beta} = v_{\gamma} = 24$. \monades{7}
    \item Το τρίγωνο που κατασκευάζεται με πλευρές ίσες με τα ύψη $v_{\alpha}$, $v_{\beta}$, $v_{\gamma}$ είναι οξυγώνιο. \monades{7}
    \end{rlist}
\item Θεωρήστε τον ισχυρισμό: «Το τρίγωνο που κατασκευάζεται με πλευρές ίσες με τα ύψη οποιουδήποτε ισοσκελούς και αμβλυγωνίου τριγώνου, είναι ισοσκελές και οξυγώνιο.» Είναι αληθής ή Ψευδής ο παραπάνω ισχυρισμός; Να αιτιολογήσετε την απάντησή σας. \monades{5}
\end{alist}
