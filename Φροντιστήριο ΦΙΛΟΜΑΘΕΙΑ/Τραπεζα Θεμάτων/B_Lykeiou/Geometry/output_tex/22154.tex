Δίνονται τα τρίγωνα ΑΒΓ και ΑΔΕ, που η κοινή κορυφή τους Α βρίσκεται στο κέντρο τριών ομόκεντρων κύκλων $(A, \rho_1)$, $(A, \rho_2)$ και $(A, \rho_3)$, η κορυφή Γ βρίσκεται στον κύκλο $(A, \rho_3)$, οι κορυφές Β και Ε στον κύκλο $(A, \rho_2)$ και η κορυφή Δ στον κύκλο $(A, \rho_1)$, όπως στο σχήμα, με $\rho_1 < \rho_2 < \rho_3$. Ονομάζουμε $E_{ΕΓ}$ το εμβαδόν του σκιασμένου δακτυλίου μεταξύ των κύκλων $(A, \rho_2)$ και $(A, \rho_3)$, $E_1$ το εμβαδόν του κύκλου $(A, \rho_1)$, $E_2$ το εμβαδόν του κύκλου $(A, \rho_2)$ και $E_3$ το εμβαδόν του κύκλου $(A, \rho_3)$.
\begin{alist}
\item Αν $\frac{E_{ΕΓ}}{E_2} = \frac{7}{9}$, να αποδείξετε ότι:
\begin{rlist}
    \item $\frac{\rho_2}{\rho_3} = \frac{3}{4}$. \monades{07}
    \item $\frac{E_2}{E_3} = \frac{9}{16}$. \monades{05}
    \item Αν επιπλέον οι ΔΕ και ΒΓ είναι παράλληλες να αποδείξετε ότι $\frac{\rho_1}{\rho_2} = \frac{3}{4}$. \monades{08}
\end{rlist}
\item Αν $E_{ΕΓ} = E_2$ και επιπλέον οι ΔΕ και ΒΓ είναι παράλληλες, να αποδείξετε ότι $E_{\Delta Β} = E_1$, όπου $E_{\Delta Β}$ είναι το εμβαδόν του δακτυλίου μεταξύ των κύκλων $(A, \rho_1)$ και $(A, \rho_2)$. \monades{05}
\end{alist}
