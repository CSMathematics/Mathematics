Δίνεται κύκλος με κέντρο $O$, ακτίνα $\rho$ και εμβαδόν ίσο με $16\pi$.
\begin{alist}
    \item Να υπολογίσετε την ακτίνα $\rho$ του κύκλου. \monades{07}
    \item Αν η ακτίνα $\rho$ του κύκλου είναι $4$ να υπολογίσετε:
    \begin{rlist}
        \item Την πλευρά του εγγεγραμμένου τετραγώνου στον κύκλο. \monades{09}
        \item Το εμβαδόν του χωρίου που περικλείεται ανάμεσα στο τετράγωνο και στον κύκλο. \monades{09}
    \end{rlist}
\end{alist}
