Δίνεται τρίγωνο $AB\Gamma$. Θεωρούμε $AM$ τη διάμεσό του και $E$ τυχαίο σημείο του τμήματος $BM$. Από το $E$ φέρουμε ευθεία παράλληλη στην $AM$ που τέμνει την πλευρά $AB$ στο $\Delta$ και την προέκταση της $\Gamma A$ στο $Z$.

\begin{alist}
\item Να συμπληρώσετε τα κενά στις παρακάτω ισότητες και να αιτιολογήσετε την επιλογή σας:
    \begin{rlist}
    \item $\frac{\Delta E}{\mathbf{AM}} = \frac{\mathbf{BE}}{BM} = \frac{B\Delta}{\mathbf{BA}}$ \\
    Αιτιολόγηση: Εφαρμόζουμε το Θεώρημα του Θαλή στο τρίγωνο $BAM$ για την παράλληλη ευθεία $\Delta E$ προς την $AM$.
    \item $\frac{\mathbf{EZ}}{AM} = \frac{\Gamma E}{\mathbf{\Gamma M}} = \frac{\mathbf{\Gamma Z}}{\Gamma A}$ \\
    Αιτιολόγηση: Εφαρμόζουμε το Θεώρημα του Θαλή στα όμοια τρίγωνα $\Gamma EZ$ και $\Gamma AM$, καθώς η ευθεία $ZE$ είναι παράλληλη προς την $AM$.
    \end{rlist}
    \monades{12}
\item Να αποδείξετε ότι $\Delta E + EZ = 2AM$ για οποιαδήποτε θέση του $E$ στο $BM$. \monades{13}
\end{alist}
