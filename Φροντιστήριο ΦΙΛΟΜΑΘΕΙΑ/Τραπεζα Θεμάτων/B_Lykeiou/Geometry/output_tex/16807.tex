    \item 9.2. Το Πυθαγόρειο θεώρημα
    \item 10.3. Εμβαδόν βασικών ευθύγραμμων σχημάτων.
\end{ylh}
Δίνεται ορθογώνιο $ΑΒΓΔ$ με διαστάσεις $ΑΒ = 24$, $ΒΓ = 12$ και σημείο $Ε$ στην ευθεία $ΑΒ$.
\begin{alist}
\item Να υπολογίσετε την περίμετρο και το εμβαδό του τριγώνου $ΓΕΔ$ όταν :
    \begin{rlist}
    \item Το σημείο $Ε$ είναι το μέσο της πλευράς $ΑΒ$.
    \item Το σημείο $Ε$ ταυτιστεί με την κορυφή $Α$ του ορθογωνίου.
    \end{rlist}
    \monades{16}
\item Αφήνουμε το σημείο $Ε$ να κινηθεί στην προέκταση του τμήματος $ΑΒ$ προς το $Β$, απομακρυνόμενο από το σημείο $Β$.
    \begin{rlist}
    \item Να εξετάσετε αν η περίμετρος του τριγώνου $ΓΕΔ$ αυξάνεται ή μειώνεται. \monades{05}
    \item Για το εμβαδό του τριγώνου $ΓΕΔ$ συμβαίνει η ίδια μεταβολή με αυτή που απαντήσατε για την περίμετρο του τριγώνου $ΓΕΔ$ στο ερώτημα β)i.; Να αιτιολογήσετε την απάντησή σας. \monades{04}
    \end{rlist}
\end{alist}
