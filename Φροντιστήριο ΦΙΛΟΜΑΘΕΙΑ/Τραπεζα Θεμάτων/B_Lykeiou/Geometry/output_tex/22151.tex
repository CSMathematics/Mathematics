Δίνεται τρίγωνο $\text{AB}\Gamma$ με $\text{AB} < \text{A}\Gamma$. Στην πλευρά $\text{AB}$ παίρνουμε σημείο $\Delta$ και στην πλευρά $\text{A}\Gamma$ σημείο $\text{E}$ ώστε $\text{AE} = \text{AB}$. Με κέντρο το σημείο $\text{A}$ και ακτίνες $\rho = \text{A}\Delta$, $r = \text{AB} = \text{AE}$ και $R = \text{A}\Gamma$ γράφουμε τρεις ομόκεντρους κύκλους $(\text{A}, \rho)$, $(\text{A}, r)$ και $(\text{A}, R)$ όπως στο σχήμα. Έστω $E_{E\Gamma}$ το εμβαδόν του σκιασμένου δακτυλίου μεταξύ των κύκλων $(\text{A}, r)$ και $(\text{A}, R)$, $E_{\Delta B}$ το εμβαδόν του δακτυλίου μεταξύ των κύκλων $(\text{A}, \rho)$ και $(\text{A}, r)$, $E_{AE}$ το εμβαδόν του κύκλου $(\text{A}, r)$ και $E_{A\Delta}$ το εμβαδόν του κύκλου $(\text{A}, \rho)$.

\begin{alist}
\item Να αποδείξετε ότι:
  \begin{rlist}
  \item \[ \frac{E_{E\Gamma}}{E_{AE}} = \frac{R^2-r^2}{r^2} \] \monades{10}
  \item \[ \frac{E_{\Delta B}}{E_{A\Delta}} = \frac{r^2-\rho^2}{\rho^2} \] \monades{07}
  \end{rlist}
\item Αν επιπλέον οι $\Delta \text{E}$ και $\text{B}\Gamma$ είναι παράλληλες, να αποδείξετε ότι:
  \[ \frac{E_{E\Gamma}}{E_{AE}} = \frac{E_{\Delta B}}{E_{A\Delta}} \] \monades{08}
\end{alist}
