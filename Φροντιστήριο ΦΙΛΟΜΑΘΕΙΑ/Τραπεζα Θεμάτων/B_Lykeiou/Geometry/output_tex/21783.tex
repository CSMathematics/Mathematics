Δίνεται ορθογώνιο τρίγωνο $\text{AB}\Gamma$, με $\widehat{\Gamma} = 90^\circ$, $\widehat{\Alpha} = 30^\circ$ και $\text{AB} = 2$.
\begin{alist}
    \item Να αποδείξετε ότι $\text{A}\Gamma = \sqrt{3}$. \monades{7}
    \item Φέρνουμε κάθετη στην $\text{AB}$, στο σημείο $\text{B}$, που τέμνει την προέκταση της $\text{A}\Gamma$ στο $\Delta$. Να αποδείξετε ότι $\text{A}\Delta = \frac{4\sqrt{3}}{3}$. \monades{10}
    \item Αν $\text{K}$ είναι το μέσο της $\text{A}\Delta$, να αποδείξετε ότι $(\text{KAB}) = \frac{\sqrt{3}}{3}$. \monades{8}
\end{alist}
