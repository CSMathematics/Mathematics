\begin{tikzpicture}[scale=0.8]
    % Ορίζουμε τις παραμέτρους α και β για τη σχεδίαση.
    % Επιλέγουμε α=2, β=3 ώστε η διάταξη των σημείων στην ευθεία να είναι A,Z,M,E,Δ,B,
    % όπως φαίνεται στο αρχικό σχήμα και να είναι συμβατή με τις ιδιότητες (α<β).
    \pgfmathsetmacro{\valpha}{2}
    \pgfmathsetmacro{\vbeta}{3}

    % Ορίζουμε τα σημεία A, M, B
    \tkzDefPoint(0,0){A}
    \tkzDefPoint(2*\valpha,0){M}
    \tkzDefPoint(2*\valpha+2*\vbeta,0){B}

    % Ορίζουμε τα σημεία Z, Δ, E στην ευθεία, όπως εμφανίζονται στο σχήμα.
    % Σημείωση: Το Z είναι το μέσο του AM. Το E είναι το μέσο του AB. Το Δ είναι το μέσο του MB.
    % Αυτή η αντιστοίχιση των συμβόλων Z, E, Δ με τα μέσα, αντικατοπτρίζει τη διάταξη στο σχήμα.
    % Ωστόσο, στην εκφώνηση, τα Ε και Δ στην αναφορά "όπου Ζ, Ε, Δ είναι τα μέσα των AM, MB και AB αντίστοιχα"
    % έχουν διαφορετική σειρά (Ε για MB, Δ για AB).
    % Για τη σχεδίαση, ακολουθούμε πιστά τις ετικέτες του σχήματος.

    \tkzDefMidPoint(A,B) \tkzGetPoint{E_axis} % Το σημείο που φέρει την ετικέτα E στο σχήμα
    \tkzDefMidPoint(M,B) \tkzGetPoint{Delta_axis} % Το σημείο που φέρει την ετικέτα Δ στο σχήμα

    % Ορίζουμε το σημείο Γ (τομή του μεγάλου ημικυκλίου και της κάθετης από το Μ)
    % Η y-συντεταγμένη του Γ είναι 2*sqrt(α*β)
    \tkzDefPoint(2*\valpha, {2*sqrt(\valpha*\vbeta)}){Gamma}

    % Σχεδίαση του ευθύγραμμου τμήματος ΑΒ
    \tkzDrawSegment[thick](A,B)

    % Σχεδίαση των ημικυκλίων
    % Ημικύκλιο με διάμετρο AM (κέντρο Z_axis)
    \tkzDrawArc[thick,R](Z_axis,A)(M)
    % Ημικύκλιο με διάμετρο MB (κέντρο Delta_axis)
    \tkzDrawArc[thick,R](Delta_axis,M)(B)
    % Ημικύκλιο με διάμετρο AB (κέντρο E_axis)
    \tkzDrawArc[thick,R](E_axis,A)(B)

    % Σχεδίαση της διακεκομμένης γραμμής από το M στο Γ
    \tkzDrawSegment[densely dashed](M,Gamma)

    % Τοποθέτηση ετικετών για τα σημεία
    \tkzLabelPoint[below](A){A}
    \tkzLabelPoint[below](Z_axis){Z}
    \tkzLabelPoint[below](M){M}
    \tkzLabelPoint[below](E_axis){E}
    \tkzLabelPoint[below](Delta_axis){$\Delta$}
    \tkzLabelPoint[below](B){B}
    \tkzLabelPoint[above left](Gamma){$\Gamma$}

    % Υπολογίζουμε τη μέση θέση: x-συντεταγμένη μεταξύ M και E_axis, y-συντεταγμένη στο μισό του Γ
    \tkzText[above right]({\pgfmathparse{1.5*\valpha+0.5*\vbeta}\pgfmathresult}, {\pgfmathparse{sqrt(\valpha*\vbeta)}\pgfmathresult}){$\tau$}

Δίνεται ευθύγραμμο τμήμα ΑΒ και τυχαίο σημείο του Μ, τέτοιο ώστε ΑΜ = 2α και ΜΒ = 2β. Με διαμέτρους ΑΜ, ΜΒ και ΑΒ γράφουμε ημικύκλια προς το ίδιο μέρος του ΑΒ, όπως φαίνεται στο σχήμα. Έστω Γ το σημείο τομής του ημικυκλίου ΑΒ και της κάθετης από το Μ στο ΑΒ.
\begin{alist}
\item Να υπολογίσετε το εμβαδόν καθενός από τα τρία ημικύκλια $\overset{\frown}{ZAM}$, $\overset{\frown}{EMB}$ και $\overset{\frown}{\Delta AB}$, όπου Ζ, Ε, Δ είναι τα μέσα των AM, MB και AB αντίστοιχα. \monades{09}
\item Να υπολογίσετε το εμβαδόν καμπυλόγραμμου σχήματος τ που περικλείεται μεταξύ των τριών ημικυκλίων. \monades{06}
\item Να αποδείξετε ότι το καμπυλόγραμμο σχήμα τ που περικλείεται μεταξύ των τριών ημικυκλίων έχει το ίδιο εμβαδόν με τον κύκλο διαμέτρου ΜΓ. \monades{05}
\item Για ποια θέση του Μ μεγιστοποιείται το εμβαδόν του καμπυλόγραμμου σχήματος τ; \monades{05}
\end{alist}
