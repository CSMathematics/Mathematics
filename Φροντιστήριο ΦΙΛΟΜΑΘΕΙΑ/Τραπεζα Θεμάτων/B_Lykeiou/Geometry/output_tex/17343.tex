\begin{ylh}{1}
    \item 9.4. Γενίκευση του Πυθαγόρειου θεωρήματος.
\end{ylh}
\begin{wrapfigure}[5]{r}{5cm}
\centering
\begin{tikzpicture}[scale=0.8]
    % Ορισμός σημείων σύμφωνα με τη διάταξη του σχήματος
    % Τοποθετούμε το A στην αρχή (0,2) και το Δ στην (3,2) για να έχουμε ΑΔ=3 οριζόντια στην κορυφή.
    \tkzDefPoint(0,2){A} 
    \tkzDefPoint(3,2){D} 
    
    % Το σημείο Γ τοποθετείται σε σχέση με το Δ, λαμβάνοντας υπόψη τη γωνία ΑΔΓ = 120 μοίρες και μήκος ΔΓ = 5.
    % Ο διάνυσμα ΔΑ είναι προς τα αριστερά (από (3,2) προς (0,2)).
    % Για να σχηματιστεί γωνία 120 μοιρών εσωτερικά και το Γ να βρίσκεται κάτω από την ΑΔ (όπως στο σχήμα),
    % χρησιμοποιούμε γωνία -60 μοιρών σε σχέση με τον θετικό άξονα x, ξεκινώντας από το Δ.
    \tkzDefPoint(3 + 5*cos(-60), 2 + 5*sin(-60)){C}
    
    % Το σημείο Β ορίζεται από την τομή δύο κύκλων:
    % Ένας κύκλος με κέντρο Α και ακτίνα ΑΒ=5.
    % Ένας κύκλος με κέντρο Γ και ακτίνα ΒΓ=8.
    \tkzInterCC(A,5cm)(C,8cm) \tkzGetPoints{B_upper}{B_lower}
    % Το σχήμα δείχνει το Β κάτω από το Α και αριστερά από το Γ, επομένως επιλέγουμε το κάτω σημείο τομής.
    \tkzGetPoint{B}{B_lower} 
    
    % Σχεδίαση του τετράπλευρου
    \tkzDrawPolygon(A,B,C,D)
    
    % Σχεδίαση της διαγωνίου ΑΓ
    \tkzDrawSegment(A,C)
    
    % Ετικέτες σημείων
    \tkzLabelPoint[above left](A){A}
    \tkzLabelPoint[above right](D){$\Delta$}
    \tkzLabelPoint[below right](C){$\Gamma$}
    \tkzLabelPoint[below left](B){B}
    
    % Σήμανση της γωνίας ω (ΒΑΓ)
    \tkzMarkAngle[fill=maincolor!30,size=0.8cm](C,A,B)
    \tkzLabelAngle[pos=0.5](C,A,B){$\omega$}
\end{tikzpicture}
\end{wrapfigure}
Στο τετράπλευρο ΑΒΓΔ του παρακάτω σχήματος, είναι ΑΔ = 3, ΑΒ = ΓΔ = 5, ΒΓ = 8 και $\hat{\Delta} = 120^\circ$.
Δίνεται ότι $\sigmaυν120^\circ = -\frac{1}{2}$.

\begin{alist}
    \item Να αποδείξετε ότι ΑΓ = 7. \monades{10}
    \item Να αποδείξετε ότι $\sigmaυν\omega = \frac{1}{7}$, όπου $\omega$ είναι η γωνία $\hat{\text{Β}\text{Α}\text{Γ}}$. \monades{15}
\end{alist}
