\end{ylh}
\begin{wrapfigure}[7]{r}{5.5cm}
    \centering
    \begin{tikzpicture}
        % Define points based on the geometry
        % Assuming D is the origin for convenience in right-angled triangle ADG
        \tkzDefPoint(0,0){D} % Delta
        \tkzDefPoint(0,4){A} % A (AD = 4)
        \tkzDefPoint(4,0){G} % Gamma (for a 45-degree angle at G, DG=AD=4)
        % For point B, we know BG = 7. If DG = 4, then BD = BG - DG = 7 - 4 = 3
        \tkzDefPoint(-3,0){B} % B

        % Draw the triangle sides
        \tkzDrawSegments(A,B A,G B,G)

        % Draw the altitude
        \tkzDrawSegment(A,D)

        % Mark the right angle at Delta (AΔ ⊥ BΓ)
        \tkzMarkRightAngle[fill=maincolor!20](A,D,G)

        % Label the angle Gamma as 45 degrees
        \tkzMarkAngle[fill=maincolor!20,size=0.8cm](A,G,D)
        \tkzLabelAngle[pos=1.1](A,G,D){$45^\circ$}

        % Label the points
        \tkzLabelPoint[above](A){A}
        \tkzLabelPoint[below left](B){B}
        \tkzLabelPoint[below right](G){$\Gamma$}
        \tkzLabelPoint[below](D){$\Delta$}
    \end{tikzpicture}
\end{wrapfigure}
Δίνεται τρίγωνο ΑΒΓ με ΒΓ = $7$, $\hat{\Gamma} = 45^\circ$ και ύψος ΑΔ = $4$.
