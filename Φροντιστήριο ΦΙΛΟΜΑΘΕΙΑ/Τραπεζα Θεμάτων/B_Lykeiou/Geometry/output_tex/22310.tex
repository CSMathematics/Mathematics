Το παρακάτω σχήμα ($\Sigma$) αποτελείται από το ημικύκλιο διαμέτρου ΑΒ και το ορθογώνιο ΑΒΓΔ. Το ημικύκλιο και το ορθογώνιο έχουν ίσα εμβαδά. Δίνεται $ΑΒ = 8 \text{ cm}$.
\begin{alist}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item το εμβαδόν του ημικυκλίου είναι $E = 8\pi \text{ cm}^2$, \monades{8}
        \item το μήκος του ημικυκλίου είναι $L = 4\pi \text{ cm}$. \monades{8}
    \end{rlist}
    \item Να βρείτε:
    \begin{rlist}
        \item το μήκος της πλευράς ΑΔ του ορθογωνίου, \monades{5}
        \item την περίμετρο του σχήματος ($\Sigma$). \monades{4}
    \end{rlist}
\end{alist}
