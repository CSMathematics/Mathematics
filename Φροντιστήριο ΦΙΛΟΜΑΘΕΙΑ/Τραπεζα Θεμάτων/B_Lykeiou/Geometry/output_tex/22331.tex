    \begin{tikzpicture}[scale=0.35]
        % Ορισμός σημείων με βάση τις δεδομένες τιμές και τα ζητούμενα
        % Έστω H η αρχή των αξόνων (0,0).
        % Το AH είναι ύψος, άρα τα A και Δ βρίσκονται στον άξονα y.
        % Τα Γ και B βρίσκονται στον άξονα x.
        \tkzDefPoint(0,0){H}
        \tkzDefPoint(-5,0){G} % Γ (Gamma), αφού ΓH = 5
        \tkzDefPoint(12,0){B} % B, αφού BH = 12
        \tkzDefPoint(0,16){A} % A, υποθέτοντας AH = 16 από το ερώτημα (α)
        \tkzDefPoint(0,12){D} % Δ (Delta), υποθέτοντας AΔ = 4 από το ερώτημα (β), οπότε ΔH = AH - AΔ = 16 - 4 = 12

        % Σχεδίαση τμημάτων
        \tkzDrawSegment(A,G) % AΓ
        \tkzDrawSegment(A,B) % AB
        \tkzDrawSegment(G,B) % ΓB
        \tkzDrawSegment(A,H) % AH (το ύψος)
        \tkzDrawSegment(D,B) % ΔB
        \tkzDrawSegment(G,D) % ΓΔ

        % Σήμανση ορθής γωνίας στο H
        \tkzMarkRightAngle[fill=thirdcolor!30,size=0.4](A,H,G) 

        % Σχεδίαση σημείων
        \tkzDrawPoints(A,B,G,H,D)

        % Ετικέτες σημείων
        \tkzLabelPoint[above left](A){A}
        \tkzLabelPoint[below right](B){B}
        \tkzLabelPoint[below left](G){$\Gamma$}
        \tkzLabelPoint[below](H){H}
        \tkzLabelPoint[left](D){$\Delta$}
    \end{tikzpicture}

Στο τρίγωνο $AB\Gamma$ του σχήματος το $AH$ είναι ύψος και το $\Delta$ σημείο του $AH$. Δίνονται $AB = 20$, $BH = 12$, $\Gamma H = 5$ και ότι το εμβαδόν του τριγώνου $AB\Delta$ είναι $(AB\Delta) = 24$.
\begin{alist}
    \item Να αποδείξετε ότι $AH = 16$. \monades{13}
    \item Να αποδείξετε ότι $A\Delta = 4$. \monades{6}
    \item Να βρείτε το εμβαδόν του τριγώνου $\Gamma\Delta H$. \monades{6}
\end{alist}
