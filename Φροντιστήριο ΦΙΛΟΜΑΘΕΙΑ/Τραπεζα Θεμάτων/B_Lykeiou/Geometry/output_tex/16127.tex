\begin{ylh}{1}
    \item 10.3. Εμβαδόν βασικών ευθύγραμμων σχημάτων
    \item 10.5. Λόγος εμβαδών όμοιων τριγώνων - πολυγώνων.
\end{ylh}
Ένα τρίγωνο $AB\Gamma$ έχει πλευρά $B\Gamma = 9$ και αντίστοιχο ύψος $A\Delta = 8$.
\begin{alist}
\item Να υπολογίσετε το εμβαδόν του τριγώνου $AB\Gamma$. \monades{9}
\item Ένα άλλο τρίγωνο $A'B'\Gamma'$ είναι όμοιο με το τρίγωνο $AB\Gamma$ και η ομόλογη πλευρά της $B\Gamma$ είναι η $B'\Gamma' = 6$. Να υπολογίσετε:
    \begin{rlist}
    \item τον λόγο ομοιότητας των τριγώνων $AB\Gamma$ και $A'B'\Gamma'$, \monades{7}
    \item το εμβαδόν του τριγώνου $A'B'\Gamma'$. \monades{9}
    \end{rlist}
\end{alist}
