\begin{alist}
\item Να υπολογίσετε το μήκος του τμήματος $AE$. \monades{7}
\item Να αποδείξετε ότι τα τρίγωνα $AB\Gamma$ και $E\Gamma\Delta$ είναι όμοια. \monades{7}
\item Έστω ότι το σημείο τομής των ευθειών $AB$ και $E\Delta$ είναι το $Z$ και $ZH$ είναι το ύψος του τριγώνου $ZAE$ από την κορυφή του $Z$. Να αποδείξετε ότι:

  \item $EH = 13$, \monades{6}
  \item $ZH = \frac{52}{3}$. \monades{5}

\end{alist}
