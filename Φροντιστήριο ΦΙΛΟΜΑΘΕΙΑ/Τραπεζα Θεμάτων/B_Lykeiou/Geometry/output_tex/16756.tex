\begin{alist}
\item Να αποδείξετε ότι:
\[ \frac{(\text{ΑΒΓ})}{(\text{ΔΒΓ})} = \frac{\text{ΑΒ}}{\text{ΔΒ}} \] \monades{15}
\item Αν $(\text{ΑΒΓ}) = 25$ και $\text{ΑΒ} = 5\text{ΑΔ}$, τότε να υπολογίσετε το εμβαδόν του τριγώνου ΔΒΓ. \monades{10}
\end{alist}
