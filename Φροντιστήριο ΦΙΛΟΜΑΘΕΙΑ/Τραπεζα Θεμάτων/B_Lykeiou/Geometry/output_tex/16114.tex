    \item 10.3. Εμβαδόν βασικών ευθύγραμμων σχημάτων
    \item 10.5. Λόγος εμβαδών όμοιων τριγώνων - πολυγώνων.
\end{ylh}
Δίνεται τρίγωνο $\mathrm{AB\Gamma}$ και σημείο $\mathrm{E}$ στην $\mathrm{A\Gamma}$ τέτοιο ώστε $\mathrm{\Gamma E} = \frac{1}{4} \mathrm{\Gamma A}$.
\begin{alist}
    \item Αν $\Delta$ σημείο της $\mathrm{AB}$ τέτοιο ώστε $\mathrm{A\Delta} = \frac{1}{3} \mathrm{AB}$:
    \begin{rlist}
        \item Να αποδείξετε ότι $(\mathrm{AB\Gamma}) = 4(\mathrm{A\Delta E})$ \monades{10}
        \item Αν από τα $\mathrm{E}$ και $\mathrm{\Gamma}$ φέρουμε τις κάθετες $\mathrm{EZ}$ και $\mathrm{\Gamma H}$ προς την $\mathrm{AB}$, να υπολογίσετε τον λόγο $\frac{\mathrm{EZ}}{\mathrm{\Gamma H}}$. \monades{09}
    \end{rlist}
    \item Θεωρώντας ότι το $\mathrm{E}$ παραμένει ακίνητο, ενώ το $\Delta$ κινείται στο εσωτερικό της $\mathrm{AB}$, να βρείτε σε ποιο σημείο πρέπει να βρεθεί το $\Delta$ ώστε $(\mathrm{AB\Gamma}) = 2(\mathrm{A\Delta E})$. \monades{06}
\end{alist}
