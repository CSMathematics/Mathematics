\begin{ylh}{1}
    \item 9.4. Γενίκευση του Πυθαγόρειου θεωρήματος
    \item 10.4. Άλλοι τύποι για το εμβαδόν τριγώνου
\end{ylh}
Σε τρίγωνο $\mathrm{AB}\Gamma$ ισχύει $\mathrm{AB} = \mathrm{A}\Gamma = 5$ και $\mathrm{B}\Gamma = 6$.
\begin{alist}
\item Να βρεθεί το είδος του τριγώνου ως προς τις γωνίες του. \monades{5}
\item Να υπολογιστεί το εμβαδόν του. \monades{10}
\item Στην προέκταση της πλευράς $\Gamma\mathrm{A}$ παίρνουμε σημείο $\Delta$ τέτοιο ώστε $\mathrm{A}\Delta = 10$.
Να υπολογισθεί το εμβαδόν του τριγώνου $\mathrm{AB}\Delta$. \monades{10}
\end{alist}
