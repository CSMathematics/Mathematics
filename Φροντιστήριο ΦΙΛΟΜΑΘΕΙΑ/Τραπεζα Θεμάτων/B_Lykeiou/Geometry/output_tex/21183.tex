\begin{alist}
    \item Να αποδείξετε ότι ΑΓ = 2. \monades{6}
    \item Να αποδείξετε ότι
    \begin{rlist}
        \item $AZ^2 = 4 + 2\sqrt{2}$. \monades{7}
        \item $\Gamma Z^2 = 4 - 2\sqrt{2}$. \monades{7}
    \end{rlist}
    \item Να υπολογίσετε σε μοίρες το μέτρο της γωνίας $\hat{A\text{Z}\Gamma} = \hat{\omega}$. \monades{5}
\end{alist}
