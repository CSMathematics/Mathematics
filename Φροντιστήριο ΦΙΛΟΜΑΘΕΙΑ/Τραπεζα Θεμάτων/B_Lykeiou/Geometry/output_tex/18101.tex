\begin{alist}
    \item Να αποδείξετε ότι οι γωνίες $\hat{\text{B}}$ και $\text{B}\hat{\text{A}}\Delta$ είναι ίσες. \monades{8}
    \item Να αποδείξετε ότι τα τρίγωνα $\text{ΑΒΓ}$ και $\text{Β}\Delta\text{Α}$ είναι όμοια. \monades{9}
    \item Να υπολογίσετε τον λόγο $\frac{(\text{ΑΒΓ})}{(\text{Β}\Delta\text{Α})}$ των εμβαδών των δύο τριγώνων. \monades{8}
\end{alist}
