    \item 10.4. Άλλοι τύποι για το εμβαδόν τριγώνου.

\begin{alist}
    \item Να βρεθεί το είδος του τριγώνου ως προς τις γωνίες του. \monades{5}
    \item Να υπολογιστεί το εμβαδό του. \monades{8}
    \item Να υπολογίσετε:
    \begin{rlist}
        \item την ακτίνα $ρ$ του εγγεγραμμένου κύκλου του τριγώνου $ΑΒΓ$.
        \item την ακτίνα $R$ του περιγεγραμμένου κύκλου του τριγώνου $ΑΒΓ$.
    \end{rlist} \monades{12}
\end{alist}
