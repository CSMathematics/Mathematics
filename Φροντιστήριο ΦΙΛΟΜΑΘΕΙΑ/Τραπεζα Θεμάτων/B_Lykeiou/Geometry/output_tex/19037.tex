Θεωρούμε τρίγωνο ΑΒΓ και τα σημεία \(\Delta\), E, Z των πλευρών AB, B\(\Gamma\), A\(\Gamma\) αντίστοιχα τέτοια ώστε
\[ \Delta B = \frac{1}{5} AB, \quad E\Gamma = \frac{1}{4} B\Gamma, \quad Z\Gamma = \frac{1}{2} A\Gamma \]
\begin{alist}
\item Να υπολογίσετε τους λόγους
\[ \frac{(\Delta BE)}{(AB\Gamma)}, \quad \frac{(E\Gamma Z)}{(AB\Gamma)}, \quad \frac{(ZA\Delta)}{(AB\Gamma)} \] \monades{15}
\item Αν είναι \( (AB\Gamma) = 120 \), να υπολογίσετε το εμβαδόν του τριγώνου \(\Delta EZ\). \monades{10}
\end{alist}
