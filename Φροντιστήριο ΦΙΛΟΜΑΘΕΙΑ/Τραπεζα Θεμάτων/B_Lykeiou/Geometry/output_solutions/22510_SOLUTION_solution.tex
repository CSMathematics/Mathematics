    \item [ΛΥΣΗ]
    \begin{alist}
        \item Στο τρίγωνο ΑΒΔ η ΒΟ είναι διάμεσος, οπότε χωρίζει το τρίγωνο ΑΒΔ σε δύο ισεμβαδικά τρίγωνα. Επομένως $(ΑΟΒ) = (ΒΟΔ)$.
        \par Ομοίως στο τρίγωνο ΑΓΔ η ΓΟ είναι διάμεσος, οπότε $(ΑΟΓ) = (ΓΟΔ)$.
        
        \item Από το α) ερώτημα έχουμε $(ΑΟΒ) = (ΒΟΔ) \ (1)$ και $(ΑΟΓ) = (ΓΟΔ) \ (2)$.
        \par Προσθέτοντας τις (1), (2) κατά μέλη παίρνουμε $(ΑΟΒ) + (ΑΟΓ) = (ΒΟΔ) + (ΓΟΔ)$, οπότε $(ΑΟΒ) + (ΑΟΓ) = (ΒΟΓΔ)$.
        \par Αφαιρώντας από τα δύο μέλη το (ΒΟΓ) έχουμε $(ΑΟΒ) + (ΑΟΓ) - (ΒΟΓ) = (ΒΔΓ)$.
        
        \item Από το β) ερώτημα έχουμε $(ΑΟΒ) + (ΑΟΓ) - (ΒΟΓ) = (ΒΔΓ) \ (3)$.
        \par Ομοίως αποδεικνύεται ότι $(ΒΟΓ) + (ΑΟΒ) - (ΑΟΓ) = (ΓΕΑ) \ (4)$
        \par και $(ΒΟΓ) + (ΑΟΓ) - (ΒΟΑ) = (ΑΖΒ) \ (5)$.
        \par Προσθέτοντας κατά μέλη τις (3), (4), (5) παίρνουμε $(ΑΟΒ) + (ΒΟΓ) + (ΑΟΓ) = (ΔΒΓ) + (ΕΓΑ) + (ΖΑΒ)$, άρα $(ΑΒΓ) = (ΒΔΓ) + (ΓΕΑ) + (ΑΖΒ)$.
        \par Επομένως $2(ΑΒΓ)=(ΑΒΓ) + (ΔΒΓ) + (ΕΓΑ) + (ΖΑΒ) = (ΑΖΒΔΓΕ)$.
    \end{alist}
\end{ylh}
