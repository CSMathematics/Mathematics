    \tkzDefPoint(5,0){B}

    % For subquestion a), M is defined by MB/MG = 1/3, which means M is at 3/4 of GB from G.
    \tkzDefPoint(3.75,0){M}

Δίνεται τρίγωνο ΝΓΒ. Στην πλευρά ΓΒ θεωρούμε σημείο Μ και στο ευθύγραμμο τμήμα ΝΜ θεωρούμε σημείο Α.
\begin{alist}
    \item Αν ισχύουν $\dfrac{\text{MB}}{\text{MΓ}} = \dfrac{1}{3}$ και $\dfrac{\text{NA}}{\text{NM}} = \dfrac{1}{4}$, να αποδείξετε ότι τα τρίγωνα ΑΜΒ και ΑΝΓ είναι ισεμβαδικά, δηλαδή $(ΑΜΒ) = (ΑΝΓ)$. \monades{13}
    \item Αν το Μ είναι το μέσο της πλευράς ΒΓ και ισχύει $(ΑΜΒ) = (ΑΝΓ)$, να αποδείξετε ότι το Α είναι το μέσο του τμήματος ΝΜ. \monades{12}
\end{alist}
