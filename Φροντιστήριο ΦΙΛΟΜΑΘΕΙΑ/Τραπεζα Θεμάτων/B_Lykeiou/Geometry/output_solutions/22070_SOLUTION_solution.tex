\begin{alist}
    \item Να αποδείξετε ότι το τρίγωνο $ΑΒΓ$ είναι ορθογώνιο στο $A$. \monades{7}
    
    \item Φέρουμε το ύψος $ΑΔ$ από την κορυφή $Α$ προς την πλευρά $ΒΓ$.
    \begin{rlist}
        \item Να αποδείξετε ότι τα τρίγωνα $ΑΒΔ$ και $ΓΑΔ$ είναι όμοια και να βρείτε το λόγο ομοιότητάς τους. \monades{9}
        \item Να υπολογίσετε το λόγο $\frac{(\mathrm{AB\Delta})}{(\mathrm{A\Gamma\Delta})}$. \monades{9}
    \end{rlist}
\end{alist}
