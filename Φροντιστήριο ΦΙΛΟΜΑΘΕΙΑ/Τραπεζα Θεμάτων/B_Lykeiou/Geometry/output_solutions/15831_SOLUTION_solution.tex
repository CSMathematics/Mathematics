    \item [Ύλη από εικόνα]

ΛΥΣΗ
\begin{alist}
\item Από τα δεδομένα, το $\Delta$ να είναι το μέσο του $\text{BM}$ και το $\text{M}$ μέσο της $\text{B}\Gamma$.
    Άρα $\text{M}\Delta = \frac{1}{2} \text{MB} = \frac{1}{2} \text{M}\Gamma$ (1).

    Επίσης η $\text{ME} // \text{A}\Delta$, οπότε από το Θ. Θαλή θα είναι:
    \[ \frac{\text{EA}}{\text{E}\Gamma} = \frac{\text{M}\Delta}{\text{M}\Gamma} \]
    Η τελευταία ισότητα λόγω της (1) γράφεται:
    \[ \frac{\text{EA}}{\text{E}\Gamma} = \frac{\frac{1}{2}\text{M}\Gamma}{\text{M}\Gamma} = \frac{1}{2} \]

\item Από τα δεδομένα η $\text{ZE} // \text{B}\Gamma$, οπότε από το Θ. Θαλή θα είναι:
    \[ \frac{\text{ZA}}{\text{ZB}} = \frac{\text{EA}}{\text{E}\Gamma} \]
    Η τελευταία ισότητα λόγω του ερωτήματος (α) δίνει:
    \[ \frac{\text{ZA}}{\text{ZB}} = \frac{1}{2} \]
\end{alist}
