\begin{alist}
    \item Να αποδείξετε ότι τα τρίγωνα ΑΔΕ και ΑΒΓ είναι όμοια.
    \monades{7}

    \item Αν το σημείο Δ είναι το μέσο της πλευράς ΑΒ, να υπολογίσετε τον λόγο των εμβαδών $\frac{(\text{ΑΔΕ})}{(\text{ΑΒΓ})}$.
    \monades{8}

    \item Να προσδιορίσετε τη θέση του σημείου Δ στην πλευρά ΑΒ, ώστε το εμβαδόν του τραπεζίου ΔΕΓΒ να είναι ίσο με $\frac{2}{9}$ του εμβαδού του τριγώνου ΑΒΓ.
    \monades{10}
\end{alist}
