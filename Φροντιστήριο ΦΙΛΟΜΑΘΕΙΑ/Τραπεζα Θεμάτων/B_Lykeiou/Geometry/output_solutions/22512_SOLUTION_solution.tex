\Askhsh{22512_SOLUTION}{1}
\begin{ylh}{1}
    \item [ΛΥΣΗ]
\end{ylh}
\begin{alist}
    \item [α)] Σύμφωνα με το νόμο των συνημιτόνων στο τρίγωνο ΑΒΓ έχουμε
    \[
        \mathrm{AB}^2 = \mathrm{A}\Gamma^2 + \mathrm{B}\Gamma^2 - 2 \cdot \mathrm{A}\Gamma \cdot \mathrm{B}\Gamma \cdot \text{συν}\,\Gamma = 2^2 + 4^2 - 2 \cdot 2 \cdot 4 \cdot \frac{1}{2} = 12,
    \]
    άρα $\mathrm{AB} = \sqrt{12} = 2\sqrt{3}$.

    \item [β)] Έχουμε $\mathrm{B}\Gamma^2 = 16$ και $\mathrm{AB}^2 + \mathrm{A}\Gamma^2 = 12 + 4 = 16$.

    Επομένως $\mathrm{B}\Gamma^2 = \mathrm{AB}^2 + \mathrm{A}\Gamma^2$, οπότε το τρίγωνο ΑΒΓ είναι ορθογώνιο με υποτείνουσα τη ΒΓ.

    \item [γ)] Επειδή $\hat{\mathrm{A}} = 90^\circ$ έχουμε $(\mathrm{AB}\Gamma) = \frac{1}{2} \cdot \mathrm{AB} \cdot \mathrm{A}\Gamma = \frac{2 \cdot 2\sqrt{3}}{2} = 2\sqrt{3}$.
\end{alist}
