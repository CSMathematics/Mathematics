\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
    \item Η γωνία $\hat{\text{ΑΚΔ}}$ είναι η κεντρική γωνία $\hat{\omega}_4$ του τετραγώνου ΑΒΓΔ. Άρα $\hat{\text{ΑΚΔ}} = \frac{360^\circ}{4} = 90^\circ$. Επομένως το τρίγωνο ΑΚΔ είναι ορθογώνιο με ορθή γωνία την $\hat{\text{ΑΚΔ}}$ και υποτείνουσα την ΑΔ.
    \item
    \begin{rlist}
        \item Το εμβαδόν του ορθογωνίου τριγώνου ΑΚΔ είναι $(\text{ΑΚΔ}) = \frac{1}{2} \cdot \text{ΚΔ} \cdot \text{ΚΑ} = \frac{1}{2} \cdot \rho \cdot \rho = \frac{1}{2} \rho^2$. Όμως $(\text{ΑΚΔ})=4$, οπότε $\frac{1}{2}\rho^2 = 4$ ή $\rho^2 = 8$ ή $\rho = \sqrt{8}$.
        \item Το εμβαδόν του κύκλου είναι $E = \pi\rho^2 = \pi \cdot (\sqrt{8})^2 = 8\pi$.
    \end{rlist}
\end{alist}
