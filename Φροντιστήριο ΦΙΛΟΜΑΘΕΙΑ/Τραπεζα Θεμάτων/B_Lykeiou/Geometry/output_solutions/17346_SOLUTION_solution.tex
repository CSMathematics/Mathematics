\Askhsh{17346_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Σύμφωνα με το νόμο των συνημιτόνων στο τρίγωνο ΑΒΓ ισχύει:
    \[ A\Gamma^2 = AB^2 + B\Gamma^2 - 2AB \cdot B\Gamma \cdot \sigma\upsilon\nu \hat{B} \]
    Από τα δεδομένα έχουμε ότι $AB = 6$, $B\Gamma = 4$ και $\hat{B} = 60^\circ$, άρα:
    \[ A\Gamma^2 = 6^2 + 4^2 - 2 \cdot 6 \cdot 4 \sigma\upsilon\nu 60^\circ \text{ ή } A\Gamma^2 = 36 + 16 - 48 \cdot \frac{1}{2} \text{ ή } A\Gamma^2 = 28 \text{ ή } A\Gamma = 2\sqrt{7}. \]
    \monades{}
\item Η μεγαλύτερη πλευρά του τριγώνου ΑΒΓ είναι η ΑΒ, αφού $AB = 6 = 2 \cdot 3 > 2\sqrt{7} = A\Gamma$, άρα η μεγαλύτερη γωνία του τριγώνου είναι η $\hat{\Gamma}$, γιατί βρίσκεται απέναντι από τη μεγαλύτερη πλευρά. Όμως $AB^2 = 6^2 = 36$ και $A\Gamma^2 + B\Gamma^2 = (2\sqrt{7})^2 + 4^2 = 28 + 16 = 44$, άρα $AB^2 < A\Gamma^2 + B\Gamma^2$. Επομένως $\hat{\Gamma} < 90^\circ$, άρα το τρίγωνο ΑΒΓ είναι οξυγώνιο.
    \monades{}
\item Για τον υπολογισμό του εμβαδού του τριγώνου ΑΒΓ χρησιμοποιούμε τον τύπο:
    \[ (AB\Gamma) = \frac{1}{2} AB \cdot B\Gamma \cdot \eta\mu\hat{B}. \]
    Όμως $AB = 6$, $B\Gamma = 4$ και $\hat{B} = 60^\circ$, άρα:
    \[ (AB\Gamma) = \frac{1}{2} \cdot 6 \cdot 4 \cdot \eta\mu 60^\circ = 12 \cdot \frac{\sqrt{3}}{2} = 6\sqrt{3}. \]
    \monades{}
\end{alist}
