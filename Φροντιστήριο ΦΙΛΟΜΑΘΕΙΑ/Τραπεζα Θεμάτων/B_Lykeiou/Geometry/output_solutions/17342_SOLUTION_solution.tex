    \item [Ύλη από εικόνα]
\end{ylh}
\textbf{ΛΥΣΗ}
\begin{alist}
    \item \begin{rlist}
        \item Το τρίγωνο $\text{A}\Gamma\Delta$ είναι ορθογώνιο με $\text{A}\hat{\Delta}\Gamma = 90^\circ$, γιατί το ύψος $\text{A}\Delta$ του τριγώνου $\text{AB}\Gamma$ είναι κάθετο στη $\text{B}\Gamma$. Στο ορθογώνιο τρίγωνο $\text{A}\Gamma\Delta$ οι οξείες γωνίες του $\Gamma\hat{\text{A}}\Delta$ και $\hat{\Gamma} = 45^\circ$ είναι συμπληρωματικές, άρα $\Gamma\hat{\text{A}}\Delta + 45^\circ = 90^\circ$ ή $\Gamma\hat{\text{A}}\Delta = 45^\circ$.
        Το τρίγωνο $\text{A}\Gamma\Delta$ είναι ισοσκελές, γιατί έχει δύο γωνίες ίσες, τις $\hat{\Gamma} = \Gamma\hat{\text{A}}\Delta = 45^\circ$. Οπότε θα είναι $\text{A}\Delta = \Gamma\Delta$ ως πλευρές του ισοσκελούς τριγώνου $\text{A}\Gamma\Delta$ που βρίσκονται απέναντι από τις ίσες γωνίες του $\hat{\Gamma}, \Gamma\hat{\text{A}}\Delta$. Όμως από την υπόθεση έχουμε ότι $\text{A}\Delta = 4$, άρα $\Gamma\Delta = 4$.
        \item Σύμφωνα με το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο $\text{A}\Gamma\Delta$ έχουμε ότι
        \[ \text{A}\Gamma^2 = \text{A}\Delta^2 + \Gamma\Delta^2 \]
        Όμως $\text{A}\Delta = \Gamma\Delta = 4$, άρα $\text{A}\Gamma^2 = 4^2 + 4^2$ ή $\text{A}\Gamma^2 = 32$ ή $\text{A}\Gamma = \sqrt{32} = 4\sqrt{2}$.
    \end{rlist}
    \item Είναι $\text{B}\Delta = \text{B}\Gamma - \Gamma\Delta = 7 - 4 = 3$. Σύμφωνα με το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο $\text{AB}\Delta$ έχουμε ότι $\text{AB}^2 = \text{A}\Delta^2 + \text{B}\Delta^2$.
    Όμως $\text{A}\Delta = 4$ και $\text{B}\Delta = 3$, άρα $\text{AB}^2 = 4^2 + 3^2$ ή $\text{AB}^2 = 25$ ή $\text{AB} = 5$.
\end{alist}
