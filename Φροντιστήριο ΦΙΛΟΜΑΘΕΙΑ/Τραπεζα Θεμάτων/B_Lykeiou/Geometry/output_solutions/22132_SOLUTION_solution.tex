    \item[Γεωμετρία Β' Λυκείου, Θεώρημα Θαλή]
\end{ylh}
ΛΥΣΗ

\begin{alist}
    \item Οι βάσεις ΑΔ και ΒΓ του τραπεζίου ΑΒΓΔ είναι παράλληλες, άρα σύμφωνα με το πόρισμα του θεωρήματος του Θαλή η ΑΔ χωρίζει σε μέρη ανάλογα τις πλευρές ΕΒ και ΕΓ του τριγώνου ΕΒΓ τις οποίες τέμνει. Επομένως:
    \[
        \frac{ΕΑ}{ΕΖ} = \frac{ΑΒ}{ΖΓ}
    \]
    Αντικαθιστώντας τα γνωστά μήκη $ΕΑ = 1$ και $ΕΖ = 1,5$ έχουμε:
    \[
        \frac{1}{1,5} = \frac{ΑΒ}{ΖΓ} \text{ ή } ΖΓ = 1,5 \cdot ΑΒ
    \]

    \item Έχουμε $ΑΒ = 4$. Επομένως, από το α) $ΖΓ = 1,5 \cdot 4 = 6$.

    \item Η ΑΖ είναι παράλληλη στην ΒΓ, γιατί το Ζ είναι σημείο της βάσης ΑΔ του τραπεζίου ΑΒΓΔ. Από εφαρμογή του θεωρήματος του Θαλή το τρίγωνο ΕΒΓ που ορίζεται από τις προεκτάσεις των πλευρών ΕΑ και ΕΖ του τριγώνου ΕΑΖ και την παράλληλη ΒΓ στην ΑΖ έχει πλευρές ανάλογες προς τις πλευρές του ΕΑΖ, άρα:
    \[
        \frac{ΕΑ}{ΕΒ} = \frac{ΕΖ}{ΕΓ} = \frac{ΑΖ}{ΒΓ}
    \]
    Όμως $ΕΒ = ΕΑ + ΑΒ = 1 + 4 = 5$ και $ΒΓ = 10$. Αντικαθιστώντας στη σχέση $\frac{ΕΑ}{ΕΒ} = \frac{ΑΖ}{ΒΓ}$ έχουμε:
    \[
        \frac{1}{5} = \frac{ΑΖ}{10} \text{ ή } 5 \cdot ΑΖ = 10 \text{ ή } ΑΖ = 2
    \]
\end{alist}
