\Askhsh{22260_SOLUTION}{1}
\begin{ylh}{1}
    \item [ΛΥΣΗ]
\end{ylh}
Από τα δεδομένα έχουμε: $\text{ΑΒ}=4$, $\text{ΑΓ}=6$ και $\hat{\text{Α}} = 150^{\circ}$ και $\eta\mu150^{\circ} = \frac{1}{2}$.
\begin{alist}
\item Είναι $(\text{ΑΒΓ}) = \frac{1}{2} \text{ΑΒ}\cdot\text{ΑΓ}\cdot\eta\mu\text{Α} = \frac{1}{2}\cdot 4 \cdot 6 \cdot\eta\mu150^{\circ} = 12\cdot\frac{1}{2} = 6$.

\item Από τα δεδομένα η $\text{ΓΔ} = \frac{1}{2}\text{ΓΑ}$. Επίσης το $\text{ΓΕ} = \frac{2}{3}\text{ΓΒ}$.

Έτσι, τα τρίγωνα ΓΔΕ και ΑΒΓ έχουν τη γωνία Γ κοινή. Άρα ο λόγος των εμβαδών τους θα ισούται με το λόγο των γινομένων των πλευρών που περιέχουν την κοινή γωνία Γ.

Δηλαδή:
\[ \frac{(\text{ΓΔΕ})}{(\text{ΑΒΓ})} = \frac{\text{ΓΔ}\cdot\text{ΓΕ}}{\text{ΓΑ}\cdot\text{ΓΒ}} \quad \text{ή} \quad \frac{(\text{ΓΔΕ})}{(\text{ΑΒΓ})} = \frac{\frac{1}{2}\text{ΓΑ}\cdot\frac{2}{3}\text{ΓΒ}}{\text{ΓΑ}\cdot\text{ΓΒ}} \quad \text{ή} \quad \frac{(\text{ΓΔΕ})}{(\text{ΑΒΓ})} = \frac{1}{3}. \]

\item Από το ερώτημα (β) έχουμε $\frac{(\text{ΓΔΕ})}{(\text{ΑΒΓ})} = \frac{1}{3}$ ή $(\text{ΓΔΕ}) = \frac{1}{3}(\text{ΑΒΓ})$
και λόγω του ερωτήματος (α) θα είναι: $(\text{ΓΔΕ}) = \frac{1}{3}\cdot 6 = 2$.
\end{alist}
