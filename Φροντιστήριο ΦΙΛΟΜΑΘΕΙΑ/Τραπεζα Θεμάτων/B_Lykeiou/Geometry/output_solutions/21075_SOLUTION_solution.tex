\begin{alist}
    \item Το εμβαδόν του κύκλου δίνεται από τον τύπο $E = \pi \rho^2$. Επομένως $16\pi = \pi \rho^2$, άρα $\rho = 4$.
    \item
    \begin{rlist}
        \item Η πλευρά τετραγώνου εγγεγραμμένου σε κύκλο με ακτίνα $\rho$ ισούται με $\rho \sqrt{2}$. Επομένως για $\rho = 4$ έχουμε ότι η πλευρά του τετραγώνου είναι $AB = \rho \sqrt{2} = 4\sqrt{2}$.
        \item Το εμβαδόν του κύκλου είναι $16\pi$, ενώ το εμβαδόν του τετραγώνου είναι $E = (AB)^2 = (4\sqrt{2})^2 = 16 \cdot 2 = 32$. Επομένως το εμβαδόν του χωρίου που περικλείεται από τον κύκλο και το τετράγωνο θα είναι ίσο με $16\pi - 32$.
    \end{rlist}
\end{alist}
