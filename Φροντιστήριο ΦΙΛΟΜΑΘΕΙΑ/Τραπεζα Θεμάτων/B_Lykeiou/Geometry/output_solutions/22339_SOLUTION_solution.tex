\Askhsh{22339_SOLUTION}{1}
\begin{ylh}{1}
	\begin{alist}
		\item[α)]
		\begin{rlist}
			\item i) Σε κάθε ορθογώνιο τρίγωνο, το τετράγωνο μιας κάθετης πλευράς του είναι ίσο με το γινόμενο της υποτείνουσας επί την προβολή της πλευράς αυτής στην υποτείνουσα.

			Άρα στο ορθογώνιο τρίγωνο ΑΒΓ ($\hat{A} = 90^\circ$), όπου είναι $ΑΒ = 15$ και $ΔΒ = 9$, έχουμε
			\[
				ΑΒ^2 = ΔΒ \cdot ΒΓ \quad \text{ή} \quad 15^2 = 9 \cdot ΒΓ \quad \text{ή} \quad ΒΓ = \frac{225}{9} \quad \text{ή} \quad ΒΓ = 25.
			\]
			\item ii) Από το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο ΑΒΓ ($\hat{A} = 90^\circ$), όπου είναι $ΑΒ = 15$ και $ΒΓ = 25$, έχουμε
			\[
				ΑΓ^2 = ΒΓ^2 - ΑΒ^2 \quad \text{ή} \quad ΑΓ^2 = 25^2 - 15^2 \quad \text{ή} \quad ΑΓ^2 = 625 - 225 \quad \text{ή} \quad ΑΓ^2 = 400 \quad \text{ή} \quad ΑΓ = 20.
			\]
		\end{rlist}
		\item[β)] Επειδή το τρίγωνο ΑΒΓ είναι ορθογώνιο, το εμβαδόν του είναι
		\[
			(ΑΒΓ) = \frac{ΑΒ \cdot ΑΓ}{2} = \frac{15 \cdot 20}{2} = 150.
		\]
	\end{alist}
\end{ylh}
