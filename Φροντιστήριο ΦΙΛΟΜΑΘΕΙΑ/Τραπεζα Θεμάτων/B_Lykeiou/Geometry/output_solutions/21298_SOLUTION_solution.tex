    \item [Λύση από εικόνα]
\end{ylh}
\textbf{ΛΥΣΗ}

\begin{alist}
    \item[α)] Το μήκος του κύκλου $(K,\rho)$ είναι $L = 2\pi \cdot \rho$. Άρα, $\rho = \frac{L}{2\pi}$.
    Εφόσον $L=10\pi$ θα είναι $\rho = \frac{10\pi}{2\pi}$ ή $\rho=5$.
    \item[β)]
    \begin{rlist}
        \item Το ορθογώνιο τρίγωνο $\text{ΑΒΓ}$ έχει κάθετες πλευρές τις $\text{ΑΒ}$ και $\text{ΑΓ}$ και υποτείνουσα τη $\text{ΒΓ}$, που είναι διάμετρος του κύκλου.
        Για τη διάμετρο $\text{ΒΓ}$ ισχύει ότι $\text{ΒΓ} = 2\rho = 10$.
        Εφαρμόζοντας το Πυθαγόρειο Θεώρημα στο ορθογώνιο τρίγωνο $\text{ΑΒΓ}$ έχουμε:
        \[
        \text{ΒΓ}^2 = \text{ΑΒ}^2 + \text{ΑΓ}^2 \text{ ή } \text{ΑΓ}^2 = \text{ΒΓ}^2 - \text{ΑΒ}^2 \text{ ή } \text{ΑΓ}^2 = 100-36 \text{ ή } \text{ΑΓ}^2 = 64 \text{ ή } \text{ΑΓ}=8.
        \]
        \item Το εμβαδόν του τριγώνου $\text{ΑΒΓ}$ είναι ίσο με $(\text{ΑΒΓ}) = \frac{\text{ΑΒ} \cdot \text{ΑΓ}}{2} = \frac{6 \cdot 8}{2} = \frac{48}{2} = 24$.
    \end{rlist}
\end{alist}
