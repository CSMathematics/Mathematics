    \item[β)] $A\Delta^2 = \Delta B \cdot \Delta E$, άρα $9 = 3\sqrt{3} \Delta E \Leftrightarrow \Delta E = \sqrt{3}$
    
    $B\Gamma^2 = B\Delta \cdot BZ$, άρα $9 = 3\sqrt{3} BZ \Leftrightarrow BZ = \sqrt{3}$
    \item[γ)] Επειδή $EZ = B\Delta - \Delta E - BZ = 3\sqrt{3} - \sqrt{3} - \sqrt{3} = \sqrt{3}$, έχουμε $\Delta E = EZ = ZB$.
\end{alist}
