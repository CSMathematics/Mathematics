\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Τα τρίγωνα $ABΓ$ και $ΓBΔ$ έχουν:
  \begin{rlist}
    \item $\hat{B}$ κοινή γωνία
    \item $\frac{AB}{ΓB} = \frac{BΓ}{BΔ}$ ή $\frac{36}{24} = \frac{24}{16}$ ή $\frac{3}{2} = \frac{3}{2}$, που ισχύει,
  \end{rlist}
  επομένως είναι όμοια αφού έχουν δύο πλευρές ανάλογες μία προς μία και την περιεχόμενη
  στις πλευρές αυτές γωνία κοινή. Ο λόγος ομοιότητας των τριγώνων $ABΓ$ και $ΓBΔ$ είναι $\frac{3}{2}$.
\item Αφού τα τρίγωνα $ABΓ$ και $ΓBΔ$ είναι όμοια, θα έχουν τις πλευρές τους ανάλογες, οπότε
  \[ \frac{AΓ}{ΓΔ} = \frac{AB}{ΓB} \text{ ή } \frac{36}{ΓΔ} = \frac{36}{24} \text{ ή } ΓΔ = 24. \]
\end{alist}
