\begin{alist}
    \item Το εμβαδόν του τριγώνου $\Alpha\Beta\Gamma$ είναι
    \[ (\Alpha\Beta\Gamma) = \frac{1}{2} \cdot \Alpha\Beta \cdot \Alpha\Gamma \cdot \eta\mu\Phi = \frac{1}{2} \cdot 5 \cdot 5 \cdot \frac{2}{5} = 5. \]
    \item Σχεδιάζουμε το ύψος $\Beta\Eta$ του $\Alpha\Beta\Gamma$ από την κορυφή $\Beta$, κάθετα στην πλευρά $\Alpha\Gamma$.
    
    Για το εμβαδόν του $\Alpha\Beta\Gamma$ ισχύει ότι
    \[ (\Alpha\Beta\Gamma) = \frac{1}{2} \cdot \Alpha\Gamma \cdot \Beta\Eta = \frac{1}{2} \cdot 5 \cdot \Beta\Eta = \frac{5}{2} \cdot \Beta\Eta. \]
    Από τη λύση του ερωτήματος α) έχουμε ότι $(\Alpha\Beta\Gamma)=5.$
    Άρα $\frac{5}{2} \cdot \Beta\Eta=5$ ή $\Beta\Eta=\frac{2}{5} \cdot 5$ ή $\Beta\Eta=2.$
\end{alist}
