\begin{alist}
    \item Στο τρίγωνο \text{ΒΕΔ}, η \text{ΕΚ} είναι διάμεσος, οπότε τα τρίγωνα \text{ΒΕΚ} και \text{ΕΚΔ} έχουν ίσες βάσεις \text{ΒΚ} και \text{ΚΔ} και κοινό ύψος από το \text{Ε}. Επομένως τα εμβαδά τους θα είναι ίσα, δηλαδή $(\text{ΒΕΚ}) = (\text{ΕΚΔ})$.
    Άρα
    \[ (\text{ΕΚΔ}) = \frac{(\text{ΒΕΔ})}{2} \quad (1). \]
    \item Το \text{ΑΕΔΖ} είναι παραλληλόγραμμο γιατί έχει τις απέναντι πλευρές του παράλληλες, αφού \text{ΔΖ}//\text{ΑΕ} και \text{ΔΕ}//\text{ΑΖ}. Η διαγώνιος \text{ΕΖ} του παραλληλογράμμου \text{ΑΕΔΖ} το χωρίζει σε δύο ίσα άρα και ισοδύναμα τρίγωνα, δηλαδή $(\text{ΕΖΔ}) = (\text{ΑΕΖ})$.
    Επομένως,
    \[ (\text{ΕΖΔ}) = \frac{(\text{ΑΕΔΖ})}{2} \quad (2). \]
    \item Στο τρίγωνο \text{ΔΖΓ}, η \text{ΖΛ} είναι διάμεσος, οπότε τα τρίγωνα \text{ΔΖΛ} και \text{ΓΖΛ} έχουν ίσες βάσεις \text{ΔΛ} και \text{ΓΛ} και κοινό ύψος από το \text{Ζ}. Επομένως τα εμβαδά τους θα είναι ίσα, δηλαδή $(\text{ΔΖΛ}) = (\text{ΓΖΛ})$.
    Άρα
    \[ (\text{ΔΖΛ}) = \frac{(\text{ΔΖΓ})}{2} \quad (3). \]
\end{alist}
Προσθέτοντας κατά μέλη τις σχέσεις (1), (2), (3) έχουμε:
\[ (\text{ΕΚΔ})+(\text{ΕΖΔ})+(\text{ΔΖΛ})=\frac{(\text{ΒΕΔ})}{2}+\frac{(\text{ΑΕΔΖ})}{2}+\frac{(\text{ΔΖΓ})}{2} \]
Άρα:
\[ (\text{ΚΕΖΛ}) = \frac{(\text{ΒΕΔ})+(\text{ΑΕΔΖ})+(\text{ΔΖΓ})}{2} = \frac{(\text{ΑΒΓ})}{2} \]
Επομένως, το εμβαδόν του \text{ΚΕΖΛ} θα ισούται με το μισό του εμβαδού του τριγώνου \text{ΑΒΓ} ανεξάρτητα από τη θέση του σημείου \text{Δ} πάνω στη \text{ΒΓ}.
