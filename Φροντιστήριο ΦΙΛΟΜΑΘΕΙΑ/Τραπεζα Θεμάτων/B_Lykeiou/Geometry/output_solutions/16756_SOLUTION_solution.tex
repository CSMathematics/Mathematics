\begin{alist}
\item Τα τρίγωνα $\text{ΑΒ}\Gamma$ και $\Delta\text{В}\Gamma$ έχουν κοινή τη γωνία $\hat{\text{B}}$. Οπότε, ο λόγος των εμβαδών τους ισούται με τον λόγο των γινομένων των πλευρών που περιέχουν τη γωνία $\hat{\text{B}}$ σε καθένα από τα τρίγωνα. Έτσι είναι:
\[ \frac{(\text{ΑΒ}\Gamma)}{(\Delta\text{В}\Gamma)} = \frac{\text{ΑΒ} \cdot \text{В}\Gamma}{\Delta\text{В} \cdot \text{В}\Gamma} \text{ ή } \frac{(\text{ΑΒ}\Gamma)}{(\Delta\text{В}\Gamma)} = \frac{\text{ΑΒ}}{\Delta\text{В}} \]

\item Αφού είναι $\text{ΑВ} = 5\text{Α}\Delta$, τότε:
\[ \Delta\text{В} = \text{ΑВ} - \text{Α}\Delta = 5\text{Α}\Delta - \text{Α}\Delta = 4\text{Α}\Delta \]
Οπότε, είναι:
\[ \frac{25}{(\Delta\text{В}\Gamma)} = \frac{5\text{Α}\Delta}{4\text{Α}\Delta} \text{ ή } \frac{25}{(\Delta\text{В}\Gamma)} = \frac{5}{4} \text{ ή } (\Delta\text{В}\Gamma) = \frac{25 \cdot 4}{5} \]
Άρα, $(\Delta\text{В}\Gamma) = 20.$
\end{alist}
