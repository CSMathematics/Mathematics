\Askhsh{22513_SOLUTION}{1}
\begin{ylh}{1}
    \item[ΛΥΣΗ]
    \begin{alist}
        \item Για να αποδείξουμε ότι το τρίγωνο είναι ορθογώνιο, αρκεί να αποδείξουμε ότι
        \[
            ΒΓ^2 = ΑΓ^2 + ΑΒ^2.
        \]
        Έχουμε $ΒΓ^2 = 13^2 = 169$ και $ΑΓ^2 + ΑΒ^2 = 5^2 + 12^2 = 25 + 144 = 169$. Άρα $ΒΓ^2 = ΑΒ^2 + ΑΓ^2$, οπότε $\hat{A} = 90^\circ$.
        
        \item Επειδή $\hat{A} = 90^\circ$ έχουμε
        \[
            (ΑΒΓ) = \frac{1}{2}ΑΒ \cdot ΑΓ = \frac{5 \cdot 12}{2} = 30.
        \]
        
        \item Ισχύει $(ΑΒΓ) = \frac{1}{2} \cdot α \cdot υ_α$, άρα
        \[
            υ_α = \frac{(ΑΒΓ)}{2α} = \frac{2 \cdot 30}{13} = \frac{60}{13}.
        \]
    \end{alist}
\end{ylh}
