\Askhsh{22509_SOLUTION}{1}
\textbf{ΛΥΣΗ}

Από τα δεδομένα έχουμε: $ΑΒ = ΔΓ = 2α$, $ΑΔ = ΒΓ = α$, $ΜΒ = x$ και $ΑΜ = 2α - x$.

\begin{alist}
    \item Από το ορθογώνιο τρίγωνο ΜΒΓ έχουμε: $ΜΓ^2 = ΜΒ^2 + ΒΓ^2 = α^2 + x^2$.
    
    Από το ορθογώνιο τρίγωνο ΔΝΓ έχουμε:
    \[ ΝΓ^2 = ΝΔ^2 + ΔΓ^2 = (2x)^2 + (2α)^2 = 4x^2 + 4α^2. \]
    Από το ορθογώνιο τρίγωνο ΑΜΝ έχουμε:
    \[ ΜΝ^2 = ΑΝ^2 + ΑΜ^2 = (α + 2x)^2 + (2α - x)^2 = α^2 + 4x^2 + 4αx + 4α^2 + x^2 - 4αx = 5α^2 + 5x^2. \]

    \item Από το α) έπεται $ΜΓ^2 + ΓΝ^2 = α^2 + x^2 + 4x^2 + 4α^2 = 5α^2 + 5x^2 = ΜΝ^2$, κατά συνέπεια το τρίγωνο ΜΝΓ είναι ορθογώνιο με υποτείνουσα τη ΜΝ.

    \item Από τα δεδομένα και το ερώτημα α) τα τρίγωνα ΑΜΝ και ΓΜΝ είναι ορθογώνια οπότε:
    \begin{align*}
        (ΑΜΝ) &= \frac{1}{2} ΑΜ \cdot ΑΝ = \frac{1}{2} (2α - x)(α + 2x) = \frac{1}{2} (2α^2 + 4αx - αx - 2x^2) = \\
        &= \frac{1}{2} (2α^2 + 3αx - 2x^2).
    \end{align*}
    \begin{align*}
        (ΜΓΝ) &= \frac{1}{2} ΜΓ \cdot ΓΝ = \frac{1}{2} \sqrt{α^2 + x^2} \cdot \sqrt{4α^2 + 4x^2} = \frac{1}{2} \sqrt{4(α^2 + x^2)^2} \\
        &= \frac{1}{2} \cdot 2 (α^2 + x^2) = α^2 + x^2.
    \end{align*}

    \item Λόγω του ερωτήματος β) έχουμε:
    \begin{align*}
        (ΑΜΝ) = (ΜΝΓ) &\iff \frac{1}{2} (2α^2 + 3αx - 2x^2) = α^2 + x^2 \iff 2α^2 + 3αx - 2x^2 = 2α^2 + 2x^2 \\
        &\iff 4x^2 = 3αx \stackrel{x \neq 0}{\iff} x = \frac{3}{4} α,
    \end{align*}
    οπότε $ΑΜ = \frac{3}{4} α$, άρα γνωστή η θέση του Μ.
\end{alist}
