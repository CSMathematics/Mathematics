\Askhsh{20667_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Για το εμβαδόν του τριγώνου \text{ΑΒΓ} ισχύει ο τύπος
    \[ (\text{ABΓ}) = \frac{1}{2} \text{BΓ} \cdot \text{AΓ} \cdot \text{ημΓ}. \]
    Από τα δεδομένα έχουμε ότι $\text{BΓ} = 8$, επομένως
    \[ (\text{ABΓ}) = \frac{1}{2} \cdot 8 \cdot \text{AΓ} \cdot \text{ημΓ} = 4 \text{AΓ} \cdot \text{ημΓ}. \]
\item Για το εμβαδόν του τριγώνου \text{ΓΔΕ} ισχύει ο τύπος
    \[ (\text{ΓΔΕ}) = \frac{1}{2} \text{ΓΔ} \cdot \text{ΓΕ} \cdot \text{ημΓ}. \]
    Από τα δεδομένα έχουμε ότι $\text{BΓ} = 8$ και $\text{ΔB} = 4$, επομένως $\text{ΓΔ} = \text{BΓ} + \text{ΔB} = 8 + 4 = 12$.
    Επιπλέον $\text{ΓΕ} = \frac{\text{AΓ}}{2}$, γιατί \text{E} είναι το μέσο της \text{ΑΓ}. Άρα
    \[ (\text{ΓΔΕ}) = \frac{1}{2} \cdot 12 \cdot \frac{\text{AΓ}}{2} \cdot \text{ημΓ} = 3 \text{AΓ} \cdot \text{ημΓ}. \]
\item Από τα ερωτήματα α) και β) προκύπτει ότι ο λόγος των εμβαδών των τριγώνων \text{ΑΒΓ} και \text{ΓΔΕ} είναι
    \[ \frac{(\text{ABΓ})}{(\text{ΓΔΕ})} = \frac{4 \text{AΓ} \cdot \text{ημΓ}}{3 \text{AΓ} \cdot \text{ημΓ}} = \frac{4}{3}. \]
    Όμως $(\text{ΓΔΕ}) = 12$ τ.μ. επομένως
    \[ \frac{(\text{ABΓ})}{12} = \frac{4}{3} \text{ ή } (\text{ABΓ}) = 16 \text{ τ.μ.} \]
\end{alist}
