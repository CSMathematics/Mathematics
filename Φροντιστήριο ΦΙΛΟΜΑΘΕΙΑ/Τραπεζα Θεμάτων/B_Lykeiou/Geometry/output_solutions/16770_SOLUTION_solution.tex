\noindent \OnPar{ΛΥΣΗ} % Χρησιμοποιούμε OnPar για τον τίτλο "ΛΥΣΗ"

\begin{alist}
    \item Τα τρίγωνα OEB και O$\Delta$A έχουν:
    \[ \widehat{EOB} = \widehat{\Delta OA} \text{, επειδή η ΟΑ είναι διχοτόμος της γωνίας } \widehat{EO\Delta} \]
    \[ \widehat{E} = \widehat{\Delta} = 70^\circ \]
    Επομένως τα τρίγωνα είναι όμοια, γιατί έχουν δύο γωνίες τους ίσες μια προς μια αντίστοιχα. \monades{7}

    \item Αφού τα δύο τρίγωνα είναι όμοια θα έχουν τις ομόλογες πλευρές τους ανάλογες. Οι ομόλογες πλευρές των δύο τριγώνων σημειώνονται στον ακόλουθο πίνακα:
    \[
    \begin{array}{|l|c|c|c|}
        \hline
        \text{Ίσες γωνίες} & \widehat{E} = \widehat{\Delta} & \widehat{EOB} = \widehat{\Delta OA} & \widehat{B} = \widehat{A} \\
        \hline
        \text{Απέναντι πλευρά στο τρίγωνο OEB} & OB & EB & OE \\
        \hline
        \text{Απέναντι πλευρά στο τρίγωνο O}\Delta\text{A} & OA & A\Delta & O\Delta \\
        \hline
    \end{array}
    \]
    Επομένως, θα είναι:
    \[ \frac{OB}{OA} = \frac{EB}{A\Delta} = \frac{OE}{O\Delta} \] \monades{8}

    \item Τα τρίγωνα OEB και O$\Delta$A είναι όμοια, οπότε ο λόγος των εμβαδών τους ισούται με το τετράγωνο του λόγου ομοιότητας, δηλαδή:
    \[ \frac{\left(OEB\right)}{\left(O\Delta A\right)} = \PARENS{\frac{OB}{OA}}^2 . \]
    Άρα $\frac{\left(OEB\right)}{28} = \PARENS{\frac{3}{2}}^2 = \frac{9}{4}$, δηλαδή $\left(OEB\right) = \frac{9}{4} \cdot 28 = 63$ τ.μ. \monades{10}
\end{alist}
