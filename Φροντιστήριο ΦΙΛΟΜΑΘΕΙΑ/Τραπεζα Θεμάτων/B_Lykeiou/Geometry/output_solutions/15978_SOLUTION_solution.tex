\subsection*{ΛΥΣΗ}

Από τα δεδομένα έχουμε: $A\Delta = \frac{1}{4}AB$, οπότε $B\Delta = \frac{3}{4}AB$, $BE = \frac{2}{3}B\Gamma$, άρα $\Gamma E = \frac{1}{3}B\Gamma$ και $\Gamma Z = \frac{1}{2}A\Gamma$, επομένως $AZ = \frac{1}{2}A\Gamma$.

\begin{alist}
\item Τα τρίγωνα $A\Delta Z$ και $AB\Gamma$ έχουν κοινή τη γωνία $A$, οπότε ο λόγος των εμβαδών τους θα ισούται με το λόγο των γινομένων των πλευρών που περιέχουν τις γωνίες αυτές.
\[
\text{Δηλαδή:} \quad \frac{\PARENS{A\Delta Z}}{\PARENS{AB\Gamma}} = \frac{A\Delta \cdot AZ}{AB \cdot A\Gamma} = \frac{\frac{1}{4}AB \cdot \frac{1}{2}A\Gamma}{AB \cdot A\Gamma} = \frac{1}{8}, \quad \text{οπότε } \PARENS{A\Delta Z} = \frac{1}{8}\PARENS{AB\Gamma}.
\]
\[
\text{Επίσης:} \quad \frac{\PARENS{BE\Delta}}{\PARENS{AB\Gamma}} = \frac{B\Delta \cdot BE}{BA \cdot B\Gamma} = \frac{\frac{3}{4}AB \cdot \frac{2}{3}B\Gamma}{AB \cdot B\Gamma} = \frac{1}{2}, \quad \text{οπότε } \PARENS{BE\Delta} = \frac{1}{2}\PARENS{AB\Gamma}.
\]
\[
\text{Όμοια:} \quad \frac{\PARENS{\Gamma EZ}}{\PARENS{AB\Gamma}} = \frac{\Gamma Z \cdot \Gamma E}{\Gamma A \cdot \Gamma B} = \frac{\frac{1}{2}\Gamma A \cdot \frac{1}{3}B\Gamma}{\Gamma A \cdot \Gamma B} = \frac{1}{6}, \quad \text{οπότε } \PARENS{\Gamma EZ} = \frac{1}{6}\PARENS{AB\Gamma}.
\]

\item Λόγω του ερωτήματος (α) έχουμε:
\[
\PARENS{\Delta EZ} = \PARENS{AB\Gamma} - \PARENS{A\Delta Z} - \PARENS{BE\Delta} - \PARENS{\Gamma EZ} = \PARENS{AB\Gamma} - \frac{1}{8}\PARENS{AB\Gamma} - \frac{1}{2}\PARENS{AB\Gamma} - \frac{1}{6}\PARENS{AB\Gamma} = \frac{5}{24}\PARENS{AB\Gamma}.
\]
\end{alist}
