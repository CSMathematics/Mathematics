\noindent\textbf{ΛΥΣΗ}

\begin{alist}
    \item Φέρουμε το ύψος $\text{ΒΔ}$. Η προβολή της $\text{ΑΒ}$ στην $\text{ΑΓ}$ είναι η $\text{ΑΔ}$.
    Εφαρμόζοντας το γενικευμένο Πυθαγόρειο θεώρημα για την οξεία γωνία $\text{Α}$, έχουμε:
    \[
    \text{ΒΓ}^2 = \text{ΑΒ}^2 + \text{ΑΓ}^2 - 2\text{ΑΓ} \cdot \text{ΑΔ}
    \]
    Αντικαθιστούμε τα γνωστά τμήματα,
    \[
    (\sqrt{41})^2 = 5^2 + 8^2 - 2 \cdot 8 \cdot \text{ΑΔ} \text{, άρα } 16\text{ΑΔ} = 25+64-41=48 \text{, άρα } \text{ΑΔ} = 3
    \]

    \item $\text{ΑΔ} = 3$.
    Εφαρμόζοντας το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο $\text{ΑΒΔ}$ έχουμε:
    \[
    \text{ΑΒ}^2 = \text{ΒΔ}^2 + \text{ΑΔ}^2
    \]
    Αντικαθιστώντας έχουμε: $5^2 = \text{ΒΔ}^2 + 3^2 \text{, άρα } \text{ΒΔ}^2 = 25-9=16 \text{, άρα } \text{ΒΔ} = 4$.
\end{alist}
