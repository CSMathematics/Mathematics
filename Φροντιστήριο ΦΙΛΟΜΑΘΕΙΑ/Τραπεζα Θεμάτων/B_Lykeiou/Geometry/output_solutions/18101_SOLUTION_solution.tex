\begin{alist}
\item Το τρίγωνο $\text{AB}\Gamma$ είναι ισοσκελές με $\text{A}\Gamma = \text{B}\Gamma$. Άρα, οι γωνίες $\widehat{\text{B}}$ και $\text{B}\widehat{\text{A}}\Gamma$ θα είναι ίσες, ως προσκείμενες στη βάση $\text{AB}$.
\item Τα τρίγωνα $\text{AB}\Gamma$ και $\text{B}\Delta\text{A}$ είναι ισοσκελή και έχουν μία αντίστοιχη γωνία ίση, αφού $\text{B}\widehat{\text{A}}\Gamma = \widehat{\text{B}}$ (η γωνία $\text{B}\widehat{\text{A}}\Gamma$ είναι προσκείμενη στη βάση $\text{AB}$ του ισοσκελούς $\text{AB}\Gamma$ και η γωνία $\widehat{\text{B}}$ είναι προσκείμενη στη βάση $\text{B}\Delta$ του ισοσκελούς $\text{AB}\Delta$).
Επομένως, τα τρίγωνα $\text{AB}\Gamma$ και $\text{AB}\Delta$ είναι όμοια.
\item Αφού τα τρίγωνα $\text{AB}\Gamma$ και $\text{AB}\Delta$ είναι όμοια, τότε ο λόγος των εμβαδών τους θα ισούται με το τετράγωνο του λόγου ομοιότητας τους. Ο λόγος ομοιότητας $\lambda$ των δύο τριγώνων ισούται με τον λόγο των πλευρών που βρίσκονται απέναντι από τις ίσες γωνίες $\text{B}\widehat{\text{A}}\Gamma$ και $\widehat{\text{B}}$ αντίστοιχα, δηλαδή είναι
\[ \frac{\text{B}\Gamma}{\text{A}\Delta} = \frac{3}{2} \]
Επομένως, έχουμε:
\[ \frac{(\text{AB}\Gamma)}{(\text{B}\Delta\text{A})} = \left(\frac{\text{B}\Gamma}{\text{A}\Delta}\right)^2 = \left(\frac{3}{2}\right)^2 = \frac{9}{4} \]
\end{alist}
