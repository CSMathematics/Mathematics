    \item [ΛΥΣΗ]
\end{ylh}
\begin{alist}
    \item Εφαρμόζουμε το νόμο συνημιτόνων στο τρίγωνο ΑΒΔ:
    \[ \text{ΒΔ}^2 = \text{ΑΒ}^2 + \text{ΑΔ}^2 - 2 \cdot \text{ΑΒ} \cdot \text{ΑΔ} \cdot \text{συν}(\text{Δ}\hat{\text{Α}}\text{Β}). \]
    Μετά από αντικατάσταση $\text{ΒΔ}^2 = 3^2 + 5^2 - 2 \cdot 3 \cdot 5 \cdot \text{συν}120^\circ$ και στη συνέχεια μετά από πράξεις: $\text{ΒΔ} = 7$ μονάδες μήκους.
    
    \item Εφαρμόζουμε το νόμο συνημιτόνων στο τρίγωνο ΔΒΓ:
    \[ \text{ΔΓ}^2 = \text{ΒΓ}^2 + \text{ΒΔ}^2 - 2 \cdot \text{ΒΓ} \cdot \text{ΒΔ} \cdot \text{συν}(\text{Δ}\hat{\text{Β}}\text{Γ}). \]
    Μετά από αντικατάσταση $\text{ΔΓ}^2 = 8^2 + 7^2 - 2 \cdot 8 \cdot 7 \cdot \text{συν}120^\circ$ και στη συνέχεια μετά από πράξεις: $\text{ΔΓ} = 13$ μονάδες μήκους.
    
    \item Για το τρίγωνο ΓΒΔ ισχύει: $(\text{ΓΒΔ}) = \frac{1}{2} \cdot \text{ΒΔ} \cdot \text{ΒΓ} \cdot \text{ημ}(\text{Δ}\hat{\text{Β}}\text{Γ})$ δηλαδή, $(\text{ΓΒΔ}) = \frac{1}{2} \cdot 7 \cdot 8 \cdot \text{ημ}120^\circ = \frac{56\sqrt{3}}{4}$ τετραγωνικές μονάδες.
\end{alist}
