\begin{alist}
    \item 
    \[ \left. \begin{aligned}
        \text{ΒΓ}^2 &= 6^2 = 36 \\
        \text{ΑΒ}^2 + \text{ΑΓ}^2 &= 5^2 + 5^2 = 50
    \end{aligned} \right\} \text{ άρα } \text{ΒΓ}^2 < \text{ΑΒ}^2 + \text{ΑΓ}^2 \text{ οπότε και } \hat{\text{A}} < 90^\circ. \]
    Όμως η $\hat{\text{A}}$ είναι η μεγαλύτερη γωνία του αφού βρίσκεται απέναντι από την μεγαλύτερη πλευρά του, οπότε και οι άλλες γωνίες θα είναι οξείες, άρα το τρίγωνο είναι οξυγώνιο.
    
    \item 
    Με εφαρμογή του τύπου του Ήρωνα $(\text{ΑΒΓ}) = \sqrt{\tau(\tau-\alpha)(\tau-\beta)(\tau-\gamma)}$ και
    \[ \tau = \frac{\alpha+\beta+\gamma}{2} = \frac{5+6+5}{2} = 8 \]
    προκύπτει
    \[ (\text{ΑΒΓ}) = \sqrt{8 \cdot (8-5) \cdot (8-6) \cdot (8-5)} = \sqrt{8 \cdot 3 \cdot 2 \cdot 3} = 12 \text{ τ.μ.} \]
    
    \item 
    Τα τρίγωνα ΑΒΓ και ΑΒΔ έχουν παραπληρωματικές τις γωνίες $\widehat{\text{ΓΑΒ}}$ και $\widehat{\text{ΒΑΔ}}$, οπότε ο λόγος των εμβαδών τους είναι ίσος με το λόγο των γινομένων των πλευρών που περιέχουν τις γωνίες αυτές.
    \[ \frac{(\text{ΑΒΔ})}{(\text{ΑΒΓ})} = \frac{\text{ΑΒ} \cdot \text{ΑΔ}}{\text{ΑΒ} \cdot \text{ΑΓ}} = \frac{10}{5} = 2 \]
    οπότε θα ισχύει $(\text{ΑΒΔ}) = 2 \cdot (\text{ΑΒΓ}) = 24 \text{ τ.μ.}$
\end{alist}
