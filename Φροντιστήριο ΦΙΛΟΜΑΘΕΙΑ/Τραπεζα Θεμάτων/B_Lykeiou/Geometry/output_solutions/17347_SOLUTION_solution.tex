\textbf{ΛΥΣΗ}

\begin{alist}
    \item Για τον υπολογισμό του εμβαδού του τριγώνου \text{ΑΓΔ} χρησιμοποιούμε τον τύπο
    \[ \text{(ΑΓΔ)} = \frac{1}{2} \text{ΑΔ} \cdot \text{ΑΓ} \cdot \eta\mu\hat{A}. \]
    Από τα δεδομένα έχουμε ότι \text{ΑΔ} = 6, \text{ΑΓ} = 10. Επιπλέον, είναι $\hat{A} = 60^\circ$, γιατί το τρίγωνο \text{ΑΒΓ} είναι ισόπλευρο. Άρα
    \[ \text{(ΑΓΔ)} = \frac{1}{2} \cdot 6 \cdot 10 \cdot \eta\mu60^\circ = 30 \cdot \frac{\sqrt{3}}{2} = 15\sqrt{3}. \]
    \item Το εμβαδόν ισόπλευρου τριγώνου πλευράς $\alpha$ ισούται με $\frac{\alpha^2\sqrt{3}}{4}$, άρα το εμβαδόν του
    ισόπλευρου τριγώνου \text{ΑΔΕ} που έχει πλευρά 6 είναι
    \[ \text{(ΑΔΕ)} = \frac{6^2\sqrt{3}}{4} = 9\sqrt{3}. \]
    Το εμβαδόν του τετραπλεύρου \text{ΑΓΔΕ} είναι
    \[ \text{(ΑΓΔΕ)} = \text{(ΑΓΔ)} + \text{(ΑΔΕ)} = 15\sqrt{3} + 9\sqrt{3} = 24\sqrt{3}. \]
\end{alist}
