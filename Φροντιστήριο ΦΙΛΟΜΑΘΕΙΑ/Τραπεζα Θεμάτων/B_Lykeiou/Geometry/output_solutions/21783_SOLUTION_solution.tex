Δίνεται ορθογώνιο τρίγωνο $\mathrm{AB}\Gamma$ ($\hat{\Gamma} = 90^\circ$) με $\mathrm{AB}=2$ και $\hat{\mathrm{A}}=30^\circ$.
\begin{alist}
    \item Να υπολογίσετε τα μήκη των πλευρών $\mathrm{A}\Gamma$ και $\mathrm{B}\Gamma$. \monades{7}
    \item Θεωρούμε σημείο $\Delta$ τέτοιο ώστε το τρίγωνο $\mathrm{A}\Delta\mathrm{B}$ να είναι ορθογώνιο στο $\mathrm{B}$ και τα σημεία $\Gamma, \Delta$ να βρίσκονται στο ίδιο ημιεπίπεδο ως προς την ευθεία $\mathrm{AB}$.
        \begin{rlist}
            \item Να αποδείξετε ότι τα τρίγωνα $\mathrm{AB}\Gamma$ και $\mathrm{A}\Delta\mathrm{B}$ είναι όμοια. \monades{9}
            \item Να υπολογίσετε το μήκος του τμήματος $\mathrm{A}\Delta$. \monades{4}
        \end{rlist}
    \item Αν $\mathrm{K}$ είναι το μέσο του τμήματος $\mathrm{A}\Delta$, να υπολογίσετε το εμβαδόν του τριγώνου $\mathrm{KAB}$. \monades{5}
\end{alist}
