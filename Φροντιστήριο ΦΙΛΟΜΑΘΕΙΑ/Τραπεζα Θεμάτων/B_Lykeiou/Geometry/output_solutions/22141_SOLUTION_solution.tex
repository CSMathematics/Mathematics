Δίνεται το παρακάτω σχήμα για το οποίο ισχύουν:
\begin{itemize}
    \item Τα σημεία Β, Α, Ε και Γ, Α, Δ είναι συνευθειακά.
    \item Η πλευρά ΒΓ είναι παράλληλη στην ΔΕ (ΒΓ // ΔΕ).
    \item ΑΒ = ΑΓ = 4 και ΒΓ = 5.
\end{itemize}

\begin{alist}
    \item Να αποδείξετε ότι ο λόγος ομοιότητας του τριγώνου ΑΒΓ προς το τρίγωνο ΑΔΕ είναι $\lambda = \dfrac{1}{2}$. \monades{7}
    \item Αν το εμβαδόν του τριγώνου ΑΒΓ είναι ίσο με 10, να βρείτε το εμβαδόν του τριγώνου ΑΔΕ. \monades{8}
    \item Θεωρούμε σημείο Ζ της πλευράς ΑΔ τέτοιο, ώστε το εμβαδόν του τριγώνου ΑΒΖ να είναι ίσο με το εμβαδόν του τριγώνου ΑΒΓ. Να αποδείξετε ότι το σημείο Ζ είναι το μέσο του τμήματος ΑΔ. \monades{10}
\end{alist}
