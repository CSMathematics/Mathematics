        \node[above] at ($(A)!0.5!(E)$) {5}; % AE

\begin{alist}
    \item [α)] Είναι:
    \[
    \frac{B\Delta}{A\Gamma} = \frac{12}{4} = 3, \quad \frac{\Delta E}{E\Gamma} = \frac{6}{2} = 3, \quad \frac{BE}{AE} = \frac{15}{5} = 3
    \]
    \item [β)] Τα τρίγωνα ΑΕΓ και ΒΕΔ είναι όμοια διότι έχουν τις πλευρές τους ανάλογες μία προς μία.
    \item [γ)] Αφού τα τρίγωνα ΑΕΓ και ΒΕΔ είναι όμοια, τότε θα έχουν ίσες τις γωνίες που βρίσκονται απέναντι από ομόλογες πλευρές τους. Επομένως, έχουμε:
    \[
    \hat{A} = \hat{B}, \quad \text{αφού βρίσκονται απέναντι από τις πλευρές } E\Gamma \text{ και } \Delta E \text{ αντίστοιχα}
    \]
    \[
    \hat{\Gamma} = \hat{\Delta}, \quad \text{αφού βρίσκονται απέναντι από τις πλευρές } AE \text{ και } BE \text{ αντίστοιχα}
    \]
    \[
    A\hat{E}\Gamma = B\hat{E}\Delta, \quad \text{αφού βρίσκονται απέναντι από τις πλευρές } A\Gamma \text{ και } B\Delta \text{ αντίστοιχα}
    \]
\end{alist}
