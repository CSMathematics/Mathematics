\Askhsh{22331_SOLUTION}{1}
\begin{ylh}{1}
    \item [ΛΥΣΗ]
\end{ylh}
\begin{alist}
    \item Στο ορθογώνιο τρίγωνο ΑΗΒ ($\hat{H} = 90^\circ$), είναι $AB = 20$ και $BH = 12$, άρα από το Πυθαγόρειο θεώρημα έχουμε
    \[ AH^2 = AB^2 - BH^2 = 20^2 - 12^2 = 400 - 144 = 256 = 16^2 \text{ ή } AH = 16. \]

    \item Το ύψος που αντιστοιχεί στη βάση ΑΔ του τριγώνου ΑΒΔ είναι το $BH = 12$, και επειδή το εμβαδόν του είναι $(\text{ΑΒΔ}) = 24$, θα έχουμε
    \[ (\text{ΑΒΔ}) = \frac{ΑΔ \cdot ΒΗ}{2} \text{ ή } 24 = \frac{ΑΔ \cdot 12}{2} \text{ ή } 6ΑΔ = 24 \text{ ή } ΑΔ = 4. \]

    \item Το τρίγωνο ΓΔΗ είναι ορθογώνιο με κάθετες πλευρές $ΔΗ = ΑΗ - ΑΔ = 16 - 4 = 12$ και $ΓΗ = 5$.

    Επομένως το εμβαδόν του τριγώνου ΓΔΗ είναι 
    \[ (\text{ΓΔΗ}) = \frac{ΗΓ \cdot ΗΔ}{2} = \frac{5 \cdot 12}{2} = 30. \]
\end{alist}
