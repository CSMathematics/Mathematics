Δίνεται τρίγωνο $ΑΒΓ$ και σημεία $Δ, Ε$ στις πλευρές του $ΑΒ, ΑΓ$ αντίστοιχα, τέτοια ώστε $ΔΕ \parallel ΒΓ$. Στην προέκταση της $ΕΑ$ (προς το μέρος του $Α$) θεωρούμε σημείο $Ζ$ τέτοιο ώστε $ΑΖ = ΑΕ$.
\begin{alist}
\item Αν $\dfrac{ΑΕ}{ΑΓ} = \dfrac{1}{3}$:
    \begin{rlist}
        \item Να αποδείξετε ότι το εμβαδόν του τριγώνου $ΑΔΕ$ είναι ίσο με το $\dfrac{1}{9}$ του εμβαδού του τριγώνου $ΑΒΓ$. \monades{9}
        \item Να αποδείξετε ότι το εμβαδόν του τριγώνου $ΔΕΖ$ είναι ίσο με το $\dfrac{2}{9}$ του εμβαδού του τριγώνου $ΑΒΓ$. \monades{8}
    \end{rlist}
\item Να βρείτε το λόγο $\lambda = \dfrac{ΑΕ}{ΑΓ}$, ώστε το εμβαδόν του τριγώνου $ΔΕΖ$ να είναι ίσο με το μισό του εμβαδού του τριγώνου $ΑΒΓ$. \monades{8}
\end{alist}
