\Askhseis{ΛΥΣΗ}

\begin{alist}
    \item Τα τρίγωνα ΑΔΕ και ΑΒΓ έχουν εμβαδόν τους ίσο με τον λόγο των γινομένων των πλευρών που περιέχουν τις γωνίες αυτές, δηλαδή:
    \[ \frac{(\text{A}\Delta \text{E})}{(\text{AB}\Gamma)} = \frac{\frac{1}{2} A\Delta \cdot AE \sin(\widehat{\Delta AE})}{\frac{1}{2} AB \cdot A\Gamma \sin(\widehat{BA\Gamma})} = \frac{A\Delta \cdot AE}{AB \cdot A\Gamma} \]

    \item Αν προεκτείνουμε τις πλευρές ΒΑ και ΓΑ κατά τμήματα είναι $A\Delta = \mu \cdot AB$ και $AE = \nu \cdot A\Gamma$ αντίστοιχα, όπου $\mu, \nu$ είναι θετικοί πραγματικοί αριθμοί, τότε θα έχουμε:
    \[ \frac{(\text{A}\Delta \text{E})}{(\text{AB}\Gamma)} = \frac{A\Delta \cdot AE}{AB \cdot A\Gamma} = \mu \cdot \nu \]

    \item Τα τρίγωνα ΑΔΕ και ΑΒΓ έχουν $\widehat{A\Delta E} = \widehat{BA\Gamma}$ (ως κατακορυφήν), οπότε θα είναι όμοια αν και μόνο αν έχουν τις προσκείμενες πλευρές σε αυτές τις γωνίες ανάλογες. Επομένως, θα πρέπει:
    \[ \frac{A\Delta}{AB} = \frac{AE}{A\Gamma} \Leftrightarrow \frac{A\Delta}{AB} = \frac{AE}{A\Gamma} \quad \text{(1) ή} \quad \frac{A\Delta}{AE} = \frac{AB}{A\Gamma} \quad \text{(2)} \]
\end{alist}
