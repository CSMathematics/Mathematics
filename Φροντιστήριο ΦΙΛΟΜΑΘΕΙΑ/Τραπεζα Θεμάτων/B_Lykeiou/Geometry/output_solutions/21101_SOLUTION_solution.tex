\noindent \textsf{\bfseries ΛΥΣΗ}

\begin{alist}
    \item Έχουμε:
    \[ ΒΓ^2 = (\sqrt{3})^2 = 3 \]
    \[ ΑΒ^2 + ΑΓ^2 = (\sqrt{2})^2 + 1^2 = 2 + 1 = 3 \]
    Επομένως, είναι $ΒΓ^2 = ΑΒ^2 + ΑΓ^2$, άρα η γωνία που βρίσκεται απέναντι από την πλευρά ΒΓ είναι ορθή, δηλαδή $\hat{Α} = 90^\circ$.
    
    \item Το εμβαδόν του τριγώνου είναι
    \[ (ΑΒΓ) = \frac{1}{2}ΑΒ \cdot ΑΓ = \frac{1}{2}\sqrt{2} \cdot 1 = \frac{1}{2}\sqrt{2} \]
    
    \item Φέρουμε το ύψος ΑΔ. Το εμβαδόν του τριγώνου μπορεί να δοθεί και από τον τύπο
    \[ (ΑΒΓ) = \frac{1}{2} ΒΓ \cdot ΑΔ \]
    Στο ερώτημα β) βρήκαμε ότι $(ΑΒΓ) = \frac{1}{2}\sqrt{2}$.
    Έτσι έχουμε:
    \[ \frac{1}{2}\sqrt{2} = \frac{1}{2}\sqrt{3} \cdot ΑΔ \quad \text{ή} \quad \sqrt{2} = \sqrt{3} \cdot ΑΔ \quad \text{ή} \quad ΑΔ = \frac{\sqrt{2}}{\sqrt{3}} = \frac{\sqrt{2}\sqrt{3}}{\sqrt{3}\sqrt{3}} = \frac{\sqrt{6}}{3} \]
\end{alist}
