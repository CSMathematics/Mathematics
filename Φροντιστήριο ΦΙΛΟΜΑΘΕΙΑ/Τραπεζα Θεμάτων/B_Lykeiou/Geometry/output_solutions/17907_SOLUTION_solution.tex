Από τα δεδομένα έχουμε: $(\mathrm{A\Gamma\Theta I}) = \text{E}$, $(\mathrm{ABHZ}) = \text{E}_1$, $(\mathrm{B\Gamma P \Delta}) = \text{E}_2$ και ισχύουν: $\text{E}_1=4\text{E} \ (1)$, $\text{E}_2=5\text{E} \ (2)$.
\begin{alist}
    \item Θέτουμε $\mathrm{B\Gamma} = \alpha$, $\mathrm{A\Gamma} = \beta$, $\mathrm{AB} = \gamma$, τις πλευρές του τριγώνου $\mathrm{AB\Gamma}$, τότε: $\text{E} = \beta^2 \ (3)$, $\text{E}_1 = \gamma^2 \ (4)$ και $\text{E}_2 = \alpha^2 \ (5)$. Οι (1), (2) λόγω των (3), (4), (5) γράφονται:
    \begin{itemize}
        \item $\gamma^2 = 4\beta^2$
        \item $\alpha^2 = 5\beta^2$
    \end{itemize}
    \item Οι τελευταίες δύο ισότητες έχουν τα δεύτερα μέλη τους ίσα, οπότε θα είναι και $\beta^2 + \gamma^2 = \alpha^2$, που σημαίνει ότι το τρίγωνο $\mathrm{AB\Gamma}$ είναι ορθογώνιο, με υποτείνουσα την πλευρά $\alpha$, άρα ορθή γωνία την $\mathrm{A}$.
\end{alist}
