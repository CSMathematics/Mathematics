\selectlanguage{greek}
\begin{alist}
    \item Το τμήμα $BE$ είναι το μισό της πλευράς $B\Gamma$, αφού η $AE$ είναι διάμεσος στην πλευρά $B\Gamma$, άρα $E\Gamma=5$. Στο τρίγωνο $A\Gamma E$ μεγαλύτερη πλευρά του είναι η $\Gamma E$ και θα εξετάσουμε αν το τετράγωνο της μεγαλύτερης πλευράς του ισούται με το άθροισμα των τετραγώνων των άλλων δύο πλευρών του. Δηλαδή αν $\Gamma E^2 = A\Gamma^2 + AE^2 \Leftrightarrow 5^2 = 4^2 + 3^2 \Leftrightarrow 25 = 16+9 \Leftrightarrow 25 = 25$ που ισχύει. Άρα το τρίγωνο $A\Gamma E$ είναι ορθογώνιο με $\widehat{EAG} = 90^{\circ}$, οπότε $AE \perp A\Gamma$. \monades{X}
    \item 
    \begin{rlist}
        \item Γνωρίζουμε ότι η διάμεσος ενός τριγώνου χωρίζει το τρίγωνο σε δύο ισοδύναμα τρίγωνα, άρα $(ABE) = (A\Gamma E)$.
        \item Το $A\Gamma E$ είναι ορθογώνιο τρίγωνο, άρα το εμβαδό του θα ισούται με το ημιγινόμενο των κάθετων πλευρών του, δηλαδή $(A\Gamma E) = \frac{1}{2} \cdot AE \cdot A\Gamma = \frac{1}{2} \cdot 3 \cdot 4 = 6 \text{ τ.μ.}$
    \end{rlist}
    Από το β) i. ερώτημα έχουμε ότι $(ABE)=(A\Gamma E)$, άρα $(AB\Gamma)=2(A\Gamma E)$. Βρήκαμε ότι $(A\Gamma E)=6$, επομένως $(AB\Gamma) = 2 \cdot 6 = 12 \text{ τ.μ.}$ \monades{X}
\end{alist}
