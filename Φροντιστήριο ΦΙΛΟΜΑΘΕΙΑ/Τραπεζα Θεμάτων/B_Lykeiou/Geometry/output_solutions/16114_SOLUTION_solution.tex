\noindent ΛΥΣΗ

\begin{alist}
    \item
    \begin{rlist}
        \item Τα τρίγωνα $\mathrm{A}\Delta\mathrm{E}$ και $\mathrm{AB}\Gamma$ έχουν την γωνία $\hat{\mathrm{A}}$ κοινή οπότε ο λόγος των εμβαδών τους θα ισούται με το λόγο των γινομένων των πλευρών που περιέχουν την γωνία.

        Επίσης $\Gamma\mathrm{E} = \frac{1}{4} \mathrm{A}\Gamma$, οπότε $\mathrm{A}\mathrm{E} = \frac{3}{4} \mathrm{A}\Gamma$.
        \[ \frac{(\mathrm{A}\Delta\mathrm{E})}{(\mathrm{AB}\Gamma)} = \frac{\mathrm{A}\Delta \cdot \mathrm{A}\mathrm{E}}{\mathrm{AB} \cdot \mathrm{A}\Gamma} = \frac{\mathrm{A}\Delta}{\mathrm{AB}} \cdot \frac{\mathrm{A}\mathrm{E}}{\mathrm{A}\Gamma} = \frac{1}{3} \cdot \frac{3}{4} = \frac{1}{4} \]
        άρα $(\mathrm{AB}\Gamma) = 4(\mathrm{A}\Delta\mathrm{E})$.

        \item Τα $\mathrm{EZ}$ και $\Gamma\mathrm{H}$ είναι ύψη των τριγώνων $\mathrm{A}\Delta\mathrm{E}$ και $\mathrm{AB}\Gamma$ αντίστοιχα.
        \[ \frac{(\mathrm{A}\Delta\mathrm{E})}{(\mathrm{AB}\Gamma)} = \frac{\frac{1}{2} \mathrm{A}\Delta \cdot \mathrm{EZ}}{\frac{1}{2} \mathrm{AB} \cdot \Gamma\mathrm{H}} = \frac{\mathrm{A}\Delta}{\mathrm{AB}} \cdot \frac{\mathrm{EZ}}{\Gamma\mathrm{H}} = \frac{1}{3} \frac{\mathrm{EZ}}{\Gamma\mathrm{H}} \]
        Επειδή όμως $\frac{(\mathrm{A}\Delta\mathrm{E})}{(\mathrm{AB}\Gamma)} = \frac{1}{4}$ από το ερώτημα (\textalpha), θα έχουμε
        \[ \frac{1}{4} = \frac{1}{3} \cdot \frac{\mathrm{EZ}}{\Gamma\mathrm{H}} \Rightarrow \frac{\mathrm{EZ}}{\Gamma\mathrm{H}} = \frac{1}{4} \cdot \frac{3}{1} = \frac{3}{4} \]
    \end{rlist}

    \item Για να ισχύει $(\mathrm{AB}\Gamma) = 2(\mathrm{A}\Delta\mathrm{E})$ πρέπει $\frac{(\mathrm{A}\Delta\mathrm{E})}{(\mathrm{AB}\Gamma)} = \frac{1}{2}$. Σκεπτόμενοι ανάλογα με το πρώτο ερώτημα $\frac{(\mathrm{A}\Delta\mathrm{E})}{(\mathrm{AB}\Gamma)} = \frac{\mathrm{A}\Delta \cdot \mathrm{A}\mathrm{E}}{\mathrm{AB} \cdot \mathrm{A}\Gamma} = \frac{\mathrm{A}\Delta}{\mathrm{AB}} \cdot \frac{\mathrm{A}\mathrm{E}}{\mathrm{A}\Gamma} = \frac{1}{2}$ και επειδή ο λόγος $\frac{\mathrm{A}\mathrm{E}}{\mathrm{A}\Gamma} = \frac{3}{4}$ παραμένει σταθερός θα έχουμε
    \[ \frac{\mathrm{A}\Delta}{\mathrm{AB}} \cdot \frac{3}{4} = \frac{1}{2} \Rightarrow \frac{\mathrm{A}\Delta}{\mathrm{AB}} = \frac{1}{2} : \frac{3}{4} = \frac{1}{2} \cdot \frac{4}{3} = \frac{2}{3} \]
    $\Rightarrow \mathrm{A}\Delta = \frac{2}{3} \mathrm{AB}$.
\end{alist}
