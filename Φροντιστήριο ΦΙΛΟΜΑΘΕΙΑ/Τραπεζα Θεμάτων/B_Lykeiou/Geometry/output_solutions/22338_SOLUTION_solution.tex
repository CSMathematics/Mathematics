ΛΥΣΗ
\begin{alist}
    \item Στο ορθογώνιο τρίγωνο ΑΔΓ το ΑΚ είναι το ύψος του, που αντιστοιχεί στην υποτείνουσα ΔΓ και οι προβολές των κάθετων πλευρών ΑΔ και ΑΓ στην υποτείνουσα ΔΓ είναι αντίστοιχα $ΚΔ = 9$ και $ΚΓ = 16$. Άρα
    \[ ΑΚ^2 = ΚΔ \cdot ΚΓ \text{ ή } ΑΚ^2 = 9 \cdot 16 \text{ ή } ΑΚ^2 = 144 \text{ ή } ΑΚ^2 = 12^2 \text{ ή } ΑΚ = 12. \]
    
    \item Το τραπέζιο ΑΒΓΔ έχει ύψος $ΑΚ = 12$ και βάσεις
    $ΔΓ = ΚΔ + ΚΓ = 9 + 16 = 25$ και
    $ΑΒ = ΚΓ = 16$, αφού το ΑΒΓΚ είναι ορθογώνιο ($\hat{Β} = \hat{Γ} = \hat{Κ} = 90^\circ$).
    
    Επομένως το εμβαδόν του τραπεζίου ΑΒΓΔ είναι
    \[ (ΑΒΓΔ) = \frac{(ΔΓ+ΑΒ) \cdot ΑΚ}{2} = \frac{(25+16) \cdot 12}{2} = 246. \]
\end{alist}
