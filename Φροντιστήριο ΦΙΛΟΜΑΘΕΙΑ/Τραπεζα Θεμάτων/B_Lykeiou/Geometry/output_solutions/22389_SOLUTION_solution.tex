\Askhsh{22389_SOLUTION}{1}
\begin{ylh}{1}
    \item [Λύση από την εικόνα]
\end{ylh}
\begin{alist}
    \item Είναι $ΒΑ = ΒΔ = R$ και $ΓΔ = ΓΕ = ρ$.

    Επειδή το Ε είναι το μέσο της ΑΓ, είναι $ΑΓ = 2ΓΕ = 2ρ$.

    Επίσης είναι $ΒΓ = ΒΔ + ΔΓ = R + ρ$.

    Από το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο ΑΒΓ ($\hat{Α} = 90^\circ$) έχουμε
    \[ ΒΓ^2 = ΑΒ^2 + ΑΓ^2 \]
    \[ (R + ρ)^2 = R^2 + (2ρ)^2 \]
    \[ R^2 + 2Rρ + ρ^2 = R^2 + 4ρ^2 \]
    \[ 2Rρ = 3ρ^2 \]
    \[ ρ = \frac{2}{3}R . \]

    \item Το εμβαδόν του ορθογωνίου τριγώνου ΑΒΓ είναι
    \[ E_1 = \frac{1}{2} ΑΓ \cdot ΑΒ = \frac{1}{2} \cdot 2ρ \cdot R = \frac{2}{3}R \cdot R = \frac{2}{3}R^2 \]
    και του κύκλου $(Β, R)$, $E_2 = πR^2$, άρα
    \[ \frac{E_2}{E_1} = \frac{πR^2}{\frac{2}{3}R^2} = \frac{3π}{2} . \]

    \item Ο κυκλικός τομέας $Β\overarc{ΑΔ}$ είναι ακτίνας $R$ και γωνίας $\hat{Β} = μ^\circ$, οπότε το εμβαδόν του είναι
    \[ E_3 = \frac{πR^2μ}{360} . \]
    Ο κυκλικός τομέας $Γ\overarc{ΔΕ}$ είναι ακτίνας $ρ = \frac{2}{3}R$ και γωνίας $\hat{Γ} = 90^\circ - μ^\circ$, οπότε το εμβαδόν του είναι
    \[ E_4 = \frac{πρ^2(90-μ)}{360} = \frac{4πR^2(90-μ)}{9 \cdot 360} . \]
    Άρα
    \[ \frac{E_4}{E_3} = \frac{4πR^2(90-μ) \cdot 360}{9 \cdot 360 \cdot πR^2μ} = \frac{4(90-μ)}{9μ} . \]
\end{alist}
