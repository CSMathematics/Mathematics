\Askhsh{22243_SOLUTION}{1}
\begin{ylh}{1}
\noindent\textbf{ΛΥΣΗ}
\begin{alist}
    \item[α)] Σύμφωνα με το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο ΑΒΖ έχουμε ότι
    \[ BZ^2 = AB^2 + AZ^2. \]
    Από την υπόθεση έχουμε ότι $AZ = \frac{3}{4}AB$, οπότε
    \[ BZ^2 = AB^2 + \left(\frac{3}{4}AB\right)^2 \quad \text{ή} \quad BZ^2 = \frac{25}{16}AB^2 \quad \text{ή} \quad BZ = \frac{5}{4}AB. \]

    \item[β)] Αφού το ΑΒΓΔ είναι τετράγωνο, θα είναι $ΑΔ = ΒΓ = ΑΒ$.
    \par Επιπλέον,
    \[ \Delta Z = A\Delta - AZ = AB - \frac{3}{4}AB = \frac{1}{4}AB. \]
    \begin{rlist}
        \item[i.] Σύμφωνα με το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο ΒΓΕ έχουμε ότι
        \[ BE^2 = B\Gamma^2 + \Gamma E^2 \quad \text{ή} \quad BE^2 = AB^2 + \left(\frac{1}{2}AB\right)^2 \quad \text{ή} \quad BE^2 = \frac{5}{4}AB^2. \]
        Σύμφωνα με το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο ΔΕΖ έχουμε ότι
        \[ ZE^2 = \Delta Z^2 + \Delta E^2 \quad \text{ή} \quad ZE^2 = \left(\frac{1}{4}AB\right)^2 + \left(\frac{1}{2}AB\right)^2 \quad \text{ή} \quad ZE^2 = \frac{5}{16}AB^2. \]

        \item[ii.] Από το ερωτήματα α και βi έχουμε ότι
        \[ BE^2 + ZE^2 = \frac{5}{4}AB^2 + \frac{5}{16}AB^2 = \frac{25}{16}AB^2 = BZ^2. \]
        Σύμφωνα με το αντίστροφο του Πυθαγορείου θεωρήματος το τρίγωνο ΒΕΖ είναι ορθογώνιο με $B\hat{E}Z = 90^\circ$.
    \end{rlist}

    \item[γ)] Είναι
    \[ \frac{BE}{B\Gamma} = \frac{\sqrt{\frac{5}{4}}AB}{AB} = \frac{\sqrt{5}}{2} \quad \text{και} \quad \frac{ZE}{E\Gamma} = \frac{\sqrt{\frac{5}{16}}AB}{\frac{1}{2}AB} = \frac{\sqrt{5}}{2}. \]
    Τα τρίγωνα ΒΕΖ και ΒΓΕ έχουν:
    \begin{itemize}
        \item $\displaystyle \frac{BE}{B\Gamma} = \frac{ZE}{E\Gamma}$ και
        \item $B\hat{E}Z = B\hat{\Gamma}E = 90^\circ$.
    \end{itemize}
    Άρα τα τρίγωνα ΒΕΖ και ΒΓΕ είναι όμοια, γιατί έχουν δύο πλευρές ανάλογες μία προς μία και τις περιεχόμενες στις πλευρές αυτές γωνίες ίσες.
    \par Ο λόγος των εμβαδών τους είναι ίσος με το τετράγωνο του λόγου ομοιότητας, οπότε
    \[ \frac{(BEZ)}{(B\Gamma E)} = \left(\frac{BE}{B\Gamma}\right)^2 = \left(\frac{\sqrt{5}}{2}\right)^2 = \frac{5}{4}. \]
\end{alist}
\end{ylh}
