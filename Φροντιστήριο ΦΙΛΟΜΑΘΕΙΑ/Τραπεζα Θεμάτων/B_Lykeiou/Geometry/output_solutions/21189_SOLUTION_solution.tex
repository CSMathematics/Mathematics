\onoma{ΛΥΣΗ}

\begin{alist}
\item
α) Για τα τρίγωνα ΑΒΓ και ΑΓΔ έχουμε
\begin{itemize}
    \item ΑΓ είναι κοινή πλευρά.
    \item ΑΒ = ΓΔ αφού το τετράπλευρο ΑΒΓΔ είναι παραλληλόγραμμο.
    \item ΒΓ = ΑΔ αφού το τετράπλευρο ΑΒΓΔ είναι παραλληλόγραμμο.
\end{itemize}
συνεπώς τα τρίγωνα είναι ίσα, άρα έχουν ίσα εμβαδά, δηλαδή $(ΑΒ\Gamma) = (Α\Gamma\Delta)$
τότε
\[ (ΑΒ\Gamma\Delta) = (ΑΒ\Gamma) + (Α\Gamma\Delta) = (ΑΒ\Gamma) + (ΑΒ\Gamma) = 2(ΑΒ\Gamma) \]
δηλαδή
\[ (ΑΒ\Gamma) = (Α\Gamma\Delta) = \frac{1}{2} \cdot (ΑΒ\Gamma\Delta) \]

\item
β) Αφού τα Μ, Ν είναι τα μέσα των πλευρών ΑΒ και ΒΓ αντίστοιχα, τότε
\[ ΜΑ = ΜΒ = \frac{ΑΒ}{2} \quad \text{και} \quad ΒΝ = ΝΓ = \frac{ΒΓ}{2} \]
Τα τρίγωνα BMN και ΑΒΓ έχουν την γωνία $\hat{Β}$ κοινή, επομένως
\[ \frac{(ΒΜΝ)}{(ΑΒ\Gamma)} = \frac{ΜΒ \cdot ΒΝ}{ΑΒ \cdot ΒΓ} = \frac{\frac{ΑΒ}{2} \cdot \frac{ΒΓ}{2}}{ΑΒ \cdot ΒΓ} = \frac{\frac{ΑΒ \cdot ΒΓ}{4}}{ΑΒ \cdot ΒΓ} = \frac{1}{4}, \quad \text{δηλαδή} \quad \frac{(ΒΜΝ)}{(ΑΒ\Gamma)} = \frac{1}{4} \]

\item
γ) Από το (β) ερώτημα έχουμε
\[ \frac{(ΒΜΝ)}{(ΑΒ\Gamma)} = \frac{1}{4} \quad \text{ή} \quad (ΒΜΝ) = \frac{1}{4} \cdot (ΑΒ\Gamma) \]
επειδή από το (α) ερώτημα έχουμε $(ΑΒ\Gamma) = \frac{1}{2} \cdot (ΑΒ\Gamma\Delta)$, τότε θα είναι
\[ (ΒΜΝ) = \frac{1}{4} \cdot (ΑΒ\Gamma) = \frac{1}{4} \cdot \frac{1}{2} \cdot (ΑΒ\Gamma\Delta) = \frac{1}{8} \cdot (ΑΒ\Gamma\Delta) \]
\end{alist}
