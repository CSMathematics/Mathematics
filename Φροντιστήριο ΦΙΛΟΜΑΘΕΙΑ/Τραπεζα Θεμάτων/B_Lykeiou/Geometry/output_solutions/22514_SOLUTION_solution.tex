	ΛΥΣΗ
	\begin{alist}
		\item Σύμφωνα με το Πυθαγόρειο θεώρημα στο τρίγωνο ΑΒΓ έχουμε
		      \[ ΑΓ^2 = ΒΓ^2 - ΑΒ^2 = 5^2 - 4^2 = 25 - 16 = 9 \text{, άρα } ΑΓ = 3. \]
		
		\item Έχουμε $ΑΒ^2 = ΒΔ \cdot ΒΓ$ , οπότε $ΒΔ = \dfrac{ΑΒ^2}{ΒΓ} = \dfrac{16}{5}$.
		
		\item Από το ορθογώνιο τρίγωνο ΑΒΔ έχουμε $ΑΒ^2 = ΑΔ^2 + ΒΔ^2$, οπότε
		      \begin{align*}
			      ΑΔ^2 &= ΑΒ^2 - ΒΔ^2 = 4^2 - \frac{16^2}{5^2} = 16 - \frac{16 \cdot 16}{25} = \frac{16 \cdot 25 - 16 \cdot 16}{25} =                          \\
			           &= \frac{16 \cdot (25-16)}{25} = \frac{16 \cdot 9}{25} \text{, άρα } ΒΔ = \frac{4 \cdot 3}{5} = \frac{12}{5}.
		      \end{align*}
	\end{alist}
\end{ylh}
