\noindent \textbf{ΛΥΣΗ}

\begin{alist}
    \item Τα τρίγωνα $\Delta OZ$ και $BOE$ είναι ίσα διότι έχουν:
    \begin{itemize}
        \item $\Delta O=BO$ (το $O$ είναι μέσο της $\Delta B$)
        \item $\Delta \hat{O}Z = B \hat{O}E$ (ως κατακορυφήν)
        \item $Z \hat{\Delta}O = E \hat{B}O$ (ως εντός εναλλάξ των παραλλήλων $AB$ και $\Gamma \Delta$ τεμνόμενων από τη $B\Delta$)
    \end{itemize}
    Επομένως, τα τρίγωνα $\Delta OZ$ και $BOE$ θα είναι ισοδύναμα, δηλαδή $(\Delta OZ) = (BOE)$.

    \item Τα τρίγωνα $A\Delta B$ και $\Gamma B \Delta$ είναι ίσα, οπότε:
    \[ (A\Delta B) = (\Gamma B \Delta) \]
    Από το σχήμα έχουμε ότι $(A\Delta B) = (\Delta OEA) + (BOE)$ και $(\Gamma B \Delta) = (B\Gamma ZO) + (\Delta OZ)$.
    Οπότε, είναι:
    \[ (\Delta OEA) + (BOE) = (B\Gamma ZO) + (\Delta OZ) \]
    Αφού $(BOE) = (\Delta OZ)$, τότε είναι $(\Delta OEA) = (B\Gamma ZO)$.
\end{alist}
