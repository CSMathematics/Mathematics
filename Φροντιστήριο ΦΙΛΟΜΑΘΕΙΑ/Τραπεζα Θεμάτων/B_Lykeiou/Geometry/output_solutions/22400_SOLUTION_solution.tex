\Askhsh{22400_SOLUTION}{1}
\begin{ylh}{1}
	\item[α)] 
	\begin{rlist}
		\item Στο τρίγωνο ΑΒΓ είναι $ΑΒ = 9$, $ΑΓ = 12$ και $ΒΓ = 15$, άρα έχουμε
		\[ ΒΓ^2 = 15^2 = 225 \text{ και } ΑΒ^2 + ΑΓ^2 = 9^2 + 12^2 = 81 + 144 = 225, \]
		άρα $ΒΓ^2 = ΑΒ^2 + ΑΓ^2$.
		
		Επομένως από το αντίστροφο του Πυθαγορείου θεωρήματος είναι $\hat{Α} = 90^\circ$, άρα το τρίγωνο ΑΒΓ είναι ορθογώνιο.
		
		\item Επειδή $\hat{Α} = 90^\circ$, το τρίγωνο ΑΔΕ είναι ορθογώνιο, οπότε από το Πυθαγόρειο θεώρημα έχουμε $ΔΕ^2 = ΑΔ^2 + ΑΕ^2$ ή $ΔΕ^2 = 4^2 + 3^2$ ή $ΔΕ^2 = 25$ ή $ΔΕ^2 = 5^2$ ή $ΔΕ = 5$.
	\end{rlist}
	
	\item[β)] 
	\begin{rlist}
		\item Στο τρίγωνο ΑΒΓ είναι $ΑΒ = 9$, $ΑΓ = 12$ και $ΒΓ = 10$, άρα έχουμε
		\[ ΑΓ^2 = 12^2 = 144 \text{ και } ΑΒ^2 + ΒΓ^2 = 9^2 + 10^2 = 81 + 100 = 181 \]
		άρα $ΑΓ^2 < ΑΒ^2 + ΒΓ^2$, οπότε $\hat{Β} < 90^\circ$. Η οξεία γωνία $\hat{Β}$ είναι η μεγαλύτερη γωνία του τριγώνου ΑΒΓ αφού βρίσκεται απέναντι από τη μεγαλύτερη πλευρά την ΑΓ. Άρα το τρίγωνο ΑΒΓ είναι οξυγώνιο και όχι ορθογώνιο.
		
		\item Τα τρίγωνα ΑΔΕ και ΑΓΒ έχουν
		\[ \frac{ΑΔ}{ΑΓ} = \frac{4}{12} = \frac{1}{3} \text{, } \frac{ΑΕ}{ΑΒ} = \frac{3}{9} = \frac{1}{3} \]
		και τη γωνία $\hat{Α}$ κοινή,
		
		άρα έχουν δύο πλευρές ανάλογες μία προς μία και τις περιεχόμενες στις πλευρές αυτές γωνίες ίσες, οπότε είναι όμοια.
		
		Άρα τα τρίγωνα ΑΔΕ και ΑΓΒ θα έχουν και τις τρίτες πλευρές ανάλογες με λόγο $\displaystyle\frac{1}{3}$.
		
		Επομένως $\displaystyle\frac{ΔΕ}{ΒΓ} = \frac{1}{3}$ ή $\displaystyle\frac{ΔΕ}{10} = \frac{1}{3}$ ή $ΔΕ = \displaystyle\frac{10}{3}$.
	\end{rlist}
\end{ylh}
