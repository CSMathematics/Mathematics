\begin{rlist}
\item Τα τρίγωνα $\Delta \text{E}\Gamma$ και ΑΒΓ έχουν τις γωνίες τους $\hat{\Gamma}=\hat{\Delta}$, γιατί είναι εντός εναλλάξ των παραλλήλων ΔΕ και ΒΓ που τέμνονται από την ΑΓ. Επομένως ο λόγος των εμβαδών τους θα ισούται με το λόγο των γινομένων των πλευρών που περιέχουν τις γωνίες αυτές:
    \[ \frac{(\Delta \text{E}\Gamma)}{(\text{AB}\Gamma)} = \frac{\Delta \text{E}\cdot \Delta \Gamma}{\text{A}\Gamma \cdot \text{B}\Gamma} \]
    \[ \frac{(\text{A}\Delta \text{E})}{(\text{AB}\Gamma)} = \frac{\Delta \text{E} \cdot \text{A}\Delta}{\text{A}\Gamma \cdot \text{B}\Gamma} = \frac{\Delta \text{E} \cdot \text{A}\Delta}{2\text{B}\Gamma \cdot \text{B}\Gamma} \quad (1), \text{ γιατί } \text{A}\Gamma = 2\text{B}\Gamma \]
    \[ \frac{(\text{A}\Delta \text{E})}{(\text{AB}\Gamma)} = \frac{2\Delta \Gamma}{\text{AB}} = 2 \quad (1) \text{, γιατί } \text{A}\Gamma = 2\Delta \Gamma \text{ και } \text{A}\Gamma = 2\text{B}\Gamma \] \monades{1}
\item Αν $\Delta \text{E}\Gamma$ παραλληλόγραμμο, τότε $\Delta \text{E} = \text{B}\Gamma$. Επομένως η (1) γίνεται
    \[ \frac{(\text{A}\Delta \text{E})}{(\text{AB}\Gamma)} = \frac{\Delta \text{E}}{2\text{B}\Gamma} = \frac{1}{2} \] \monades{1}
\item Στο τρίγωνο ΑΒΓ η διάμεσος ΒΔ χωρίζει το τρίγωνο σε δύο ισοδύναμα τρίγωνα, τα ΑΒΔ και ΒΔΓ. Επομένως το καθένα από αυτά θα έχει το μισό εμβαδόν του ΑΒΓ.
    \[ \frac{(\text{AB}\Delta)}{(\text{AB}\Gamma)} = \frac{1}{2} \quad (3) \] \monades{1}
\end{rlist}
\begin{alist}
\item [\beta)] Ο μαθητής για να αποδείξει ότι τα τρίγωνα είναι όμοια χρησιμοποιεί το επιχείρημα ότι έχουν δύο πλευρές τους ανάλογες μία προς μία και τις γωνίες $\hat{\Delta}$ και $\hat{\Gamma}$ ίσες. Για να εξασφαλίσουμε όμως την ομοιότητα αυτό το κριτήριο θα έπρεπε οι γωνίες να είναι οι περιεχόμενες των αναλόγων πλευρών, πράγμα το οποίο εδώ δεν συμβαίνει. Οι περιεχόμενες γωνίες στις πλευρές $\Delta \text{E}, \Delta \Gamma, \text{A}\text{B}, \text{A}\Gamma$ είναι οι $\hat{\Delta}$ και $\hat{\text{A}}$.
\end{alist}
