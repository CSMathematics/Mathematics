    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ

\begin{alist}
\item Επειδή τα δύο κανονικά πολύγωνα με πλήθος πλευρών $v_1$ και $v_2$ έχουν λόγο ίσο με $\frac{1}{2}$, ισχύει $\frac{v_1}{v_2} = \frac{1}{2}$ ή $v_2 = 2v_1$, τότε ο λόγος των αντίστοιχων κεντρικών γωνιών $\omega_1$ και $\omega_2$ αυτών των πολυγώνων είναι:
\[
\frac{\omega_1}{\omega_2} = \frac{\frac{360^\circ}{v_1}}{\frac{360^\circ}{v_2}} = \frac{360^\circ \cdot v_2}{360^\circ \cdot v_1} = \frac{v_2}{v_1} = \frac{2v_1}{v_1} = 2
\]

\item Επειδή $v_2 = 2v_1$ και $v_1 = 5$, τότε το πλήθος των πλευρών $v_2$ του άλλου κανονικού πολυγώνου είναι: $v_2 = 2v_1 = 2 \cdot 5 = 10$, τότε ο λόγος των γωνιών τους $\frac{\Phi_1}{\Phi_2}$ είναι:
\[
\frac{\Phi_1}{\Phi_2} = \frac{180^\circ - \frac{360^\circ}{v_1}}{180^\circ - \frac{360^\circ}{v_2}} = \frac{180^\circ - \frac{360^\circ}{5}}{180^\circ - \frac{360^\circ}{10}} = \frac{108^\circ}{144^\circ} = \frac{3}{4}
\]
\end{alist}
