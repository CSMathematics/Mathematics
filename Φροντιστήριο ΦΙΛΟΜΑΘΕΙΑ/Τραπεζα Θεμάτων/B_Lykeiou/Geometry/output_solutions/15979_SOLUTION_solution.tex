\textbf{ΛΥΣΗ}

Έχουμε: $AB = A\Gamma = 5$ και $\hat{A} = 120^\circ$.

\begin{alist}
    \item Στο τρίγωνο $AB\Gamma$, από το νόμο των συνημιτόνων είναι:
    \[
    B\Gamma^2 = AB^2 + A\Gamma^2 - 2 \cdot AB \cdot A\Gamma \cdot \sigma\upsilon\nu A = 5^2 + 5^2 - 2 \cdot 5 \cdot 5 \cdot \sigma\upsilon\nu 120^\circ = 25 + 25 - 50 \cdot \left(-\frac{1}{2}\right) = 75,
    \]
    οπότε $B\Gamma = \sqrt{75} = 5\sqrt{3}$. \monades{X}
    \item Επίσης $(AB\Gamma) = \frac{1}{2} \cdot AB \cdot A\Gamma \cdot \eta\mu A = \frac{1}{2} \cdot 5 \cdot 5 \cdot \eta\mu 120^\circ = \frac{1}{2} \cdot 25 \cdot \frac{\sqrt{3}}{2} = \frac{25\sqrt{3}}{4}$. \monades{X}
\end{alist}
