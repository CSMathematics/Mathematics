\begin{alist}
    \item Τα τρίγωνα \text{A}\Delta\text{E} και \Delta\text{E}\text{B} έχουν τις γωνίες τους $\hat{\Delta}_1$ και $\hat{\Delta}_2$ παραπληρωματικές όπως και τα τρίγωνα \text{A}\Delta\text{E} και \Delta\text{E}\Gamma έχουν παραπληρωματικές τις γωνίες $\hat{\text{E}}_1$ και $\hat{\text{E}}_2$. Άρα οι λόγοι των εμβαδών τους θα είναι ίσοι με το λόγο των γινομένων των πλευρών που περιέχουν τις γωνίες αυτές. Δηλαδή
    \[ \frac{(\text{A}\Delta\text{E})}{(\Delta\text{E}\text{B})} = \frac{\text{A}\Delta \cdot \Delta\text{E}}{\Delta\text{B} \cdot \Delta\text{E}} = \frac{\text{A}\Delta}{\Delta\text{B}} \quad \text{και} \quad \frac{(\text{A}\Delta\text{E})}{(\Delta\text{E}\Gamma)} = \frac{\text{A}\text{E} \cdot \Delta\text{E}}{\text{E}\Gamma \cdot \Delta\text{E}} = \frac{\text{A}\text{E}}{\text{E}\Gamma} \cdot \]

    \item Το τμήμα \Delta\text{E} είναι παράλληλο στην πλευρά \text{B}\Gamma του τριγώνου \text{A}\text{B}\Gamma, οπότε εφαρμόζοντας το
    \text{Θεώρημα του Θαλή}, έχουμε ότι $\frac{\text{A}\Delta}{\Delta\text{B}} = \frac{\text{A}\text{E}}{\text{E}\Gamma}$ . Επομένως από το α) ερώτημα θα ισχύει ότι
    \[ \frac{(\text{A}\Delta\text{E})}{(\Delta\text{E}\text{B})} = \frac{(\text{A}\Delta\text{E})}{(\Delta\text{E}\Gamma)} \quad \text{, άρα} \quad (\Delta\text{E}\text{B}) = (\Delta\text{E}\Gamma). \]
\end{alist}
