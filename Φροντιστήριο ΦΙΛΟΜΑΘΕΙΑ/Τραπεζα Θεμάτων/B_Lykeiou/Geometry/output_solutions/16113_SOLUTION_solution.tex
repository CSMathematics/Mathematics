\kefalaio{ΛΥΣΗ}

\begin{alist}
    \item[α)] Τα τρίγωνα $\mathrm{AEB}$ και $\mathrm{\Gamma E\Delta}$ έχουν:
    \[ \widehat{\mathrm{A}}_1 = \widehat{\mathrm{\Gamma}}_1 \]
    ως εντός εναλλάξ των παραλλήλων $\mathrm{AB}$ και $\mathrm{\Gamma\Delta}$ που τέμνονται από την $\mathrm{A\Gamma}$.
    \[ \widehat{\mathrm{\Delta}}_1 = \widehat{\mathrm{B}}_1 \]
    σαν εντός εναλλάξ των παραλλήλων $\mathrm{AB}$ και $\mathrm{\Gamma\Delta}$ που τέμνονται από την $\mathrm{B\Delta}$.
    Επομένως τα τρίγωνα είναι όμοια, αφού έχουν δύο γωνίες τους ίσες μία προς μία.

    \item[β)] Από την ομοιότητα των τριγώνων $\mathrm{ABE}$ και $\mathrm{\Gamma E\Delta}$ συμπεραίνουμε ότι οι αντίστοιχες πλευρές θα είναι ανάλογες. Οι ομόλογες πλευρές φαίνονται στον πίνακα.
    \begin{center}
    \begin{tabular}{|c|c|c|c|}
    \hline
    \multicolumn{1}{|c|}{} & \multicolumn{3}{c|}{\textbf{Ίσες γωνίες}} \\
    \cline{2-4}
    \multicolumn{1}{|c|}{} & $\widehat{\mathrm{A}}_1 = \widehat{\mathrm{\Gamma}}_1$ & $\widehat{\mathrm{B}}_1 = \widehat{\mathrm{\Delta}}_1$ & $\widehat{\mathrm{AEB}} = \widehat{\mathrm{\Gamma E\Delta}}$ \\
    \hline
    Απέναντι πλευρά στο τρίγωνο $\mathrm{ABE}$ & $\mathrm{BE}$ & $\mathrm{AE}$ & $\mathrm{AB}$ \\
    \hline
    Απέναντι πλευρά στο τρίγωνο $\mathrm{\Gamma E\Delta}$ & $\mathrm{\Delta E}$ & $\mathrm{\Gamma E}$ & $\mathrm{\Delta\Gamma}$ \\
    \hline
    \end{tabular}
    \end{center}
    Επομένως θα ισχύει:
    \[ \frac{\mathrm{AB}}{\mathrm{\Gamma\Delta}} = \frac{\mathrm{AE}}{\mathrm{\Gamma E}} = \frac{\mathrm{BE}}{\mathrm{\Delta E}} \quad (1). \]

    \item[γ)] Από την (1) έχουμε
    \[ \frac{\mathrm{AB}}{\mathrm{\Gamma\Delta}} = \frac{\mathrm{AE}}{\mathrm{\Gamma E}} = \frac{\mathrm{BE}}{\mathrm{\Delta E}} \text{ ή } \frac{8}{\mathrm{\Gamma\Delta}} = \frac{6}{15} = \frac{\mathrm{BE}}{10}. \]
    Από $\frac{8}{\mathrm{\Gamma\Delta}} = \frac{6}{15}$ προκύπτει $6\mathrm{\Gamma\Delta} = 8 \cdot 15$, άρα $6\mathrm{\Gamma\Delta} = 120$, άρα $\mathrm{\Gamma\Delta} = 20$. \\
    Από $\frac{6}{15} = \frac{\mathrm{BE}}{10}$ προκύπτει $15\mathrm{BE} = 6 \cdot 10$, άρα $15\mathrm{BE} = 60$, άρα $\mathrm{BE} = 4$.
\end{alist}
