\Askhsh{21183_SOLUTION}{1}
\begin{ylh}{1}
    \item [Λύση από εικόνα]
\end{ylh}

\OnKef{ΛΥΣΗ}

\begin{alist}
    \item Στο ορθογώνιο τρίγωνο ΑΒΓ με εφαρμογή του πυθαγορείου θεωρήματος έχουμε:
    \[
        \text{ΑΓ}^2 = \text{ΑΒ}^2 + \text{ΒΓ}^2 = \left(\sqrt{2}\right)^2 + \left(\sqrt{2}\right)^2 = 4 \text{ επομένως } \text{ΑΓ} = 2.
    \]
    \item
    \begin{rlist}
        \item Στο ορθογώνιο τρίγωνο ΑΗΖ είναι $\text{ΑΗ} = \text{ΑΔ} + \text{ΔΗ} = \sqrt{2} + 1$, επομένως έχουμε:
        \[
            \text{ΑΖ}^2 = \text{ΑΗ}^2 + \text{ΗΖ}^2 = \left(\sqrt{2} + 1\right)^2 + 1^2 = 2 + 2\sqrt{2} + 1 + 1 = 4 + 2\sqrt{2}.
        \]
        \item Είναι $\text{ΕΓ} = \text{ΔΓ} - \text{ΔΕ} = \sqrt{2} - 1$. Τότε στο ορθογώνιο τρίγωνο ΖΕΓ έχουμε:
        \[
            \text{ΖΓ}^2 = \text{ΖΕ}^2 + \text{ΕΓ}^2 = 1^2 + \left(\sqrt{2} - 1\right)^2 = 1 + 2 - 2\sqrt{2} + 1 = 4 - 2\sqrt{2}.
        \]
    \end{rlist}
    \item Από το β ερώτημα προκύπτει ότι
    \[
        \text{ΑΖ} = \sqrt{4 + 2\sqrt{2}} \text{ και } \text{ΖΓ} = \sqrt{4 - 2\sqrt{2}}
    \]
    Στο τρίγωνο ΑΖΓ με εφαρμογή του νόμου των συνημιτόνων προκύπτει
    \[
        \text{ΑΓ}^2 = \text{ΑΖ}^2 + \text{ΖΓ}^2 - 2\text{ΑΖ} \cdot \text{ΖΓ} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        2^2 = 4 + 2\sqrt{2} + 4 - 2\sqrt{2} - 2\sqrt{4 + 2\sqrt{2}}\sqrt{4 - 2\sqrt{2}} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 2\sqrt{\left(4 + 2\sqrt{2}\right)\left(4 - 2\sqrt{2}\right)} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 2\sqrt{4^2 - \left(2\sqrt{2}\right)^2} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 2\sqrt{16 - 8} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 2\sqrt{8} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 2 \cdot 2\sqrt{2} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4 = 8 - 4\sqrt{2} \cdot \text{συν}\omega \text{ ή}
    \]
    \[
        4\sqrt{2} \cdot \text{συν}\omega = 8 - 4 \text{ ή}
    \]
    \[
        4\sqrt{2} \cdot \text{συν}\omega = 4 \text{ ή}
    \]
    \[
        \text{συν}\omega = \frac{4}{4\sqrt{2}} = \frac{1}{\sqrt{2}} = \frac{\sqrt{2}}{2} \text{ άρα } \hat{\omega} = 45^\circ.
    \]
\end{alist}
