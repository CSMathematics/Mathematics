\begin{alist}
    \item Τα τρίγωνα $\Gamma\Theta\text{Z}$ και $\text{KZ}\Delta$ είναι ίσα γιατί έχουν
    \[
    \text{ΓZ} = \text{Z}\Delta \quad \text{αφού Z μέσο της } \text{Γ}\Delta
    \]
    \[
    \widehat{\Gamma\text{Z}\Theta} = \widehat{\text{KZ}\Delta} \quad \text{ως κατακορυφήν γωνίες}
    \]
    \[
    \widehat{\text{Z}\Gamma\Theta} = \widehat{\text{Z}\Delta\text{K}} \quad \text{ως εντός εναλλάξ γωνίες}
    \]
    Άρα $(\Gamma\Theta\text{Z}) = (\text{KZ}\Delta)$ και συνεπώς
    \[
    (\text{AB}\Gamma\Delta) = (\text{AB}\Gamma\text{ZK}) + (\text{KZ}\Delta) = (\text{AB}\Gamma\text{ZK}) + (\Gamma\Theta\text{Z}) = (\text{AB}\Theta\text{K})
    \]

    \item Το $\text{AB}\Theta\text{K}$ είναι παραλληλόγραμμο με $(\text{AB}\Theta\text{K}) = \text{AB} \cdot \text{HZ}$
    Από α) είναι $(\text{AB}\Gamma\Delta) = (\text{AB}\Theta\text{K})$
    Άρα $(\text{AB}\Gamma\Delta) = \text{AB} \cdot \text{HZ}$

    \item Για το τραπέζιο $\text{AB}\Gamma\Delta$ ισχύει $(\text{AB}\Gamma\Delta) = \frac{(\text{A}\Delta+\text{B}\Gamma)\cdot \text{υ}}{2}$
    Από β) είναι $(\text{AB}\Gamma\Delta) = \text{AB} \cdot \text{HZ}$
    Άρα
    \[
    \frac{(\text{A}\Delta+\text{B}\Gamma)\cdot \text{υ}}{2} = \text{AB} \cdot \text{HZ} \Rightarrow \text{υ} = \frac{2 \cdot \text{AB} \cdot \text{HZ}}{\text{A}\Delta+\text{B}\Gamma}
    \]

    \item Φέρνουμε τη διαγώνιο $\text{BK}$ του παραλληλογράμμου $\text{AB}\Theta\text{K}$.
    Τα τρίγωνα $\text{ABK}$ και $\text{B}\Theta\text{K}$ είναι ίσα οπότε $(\text{ABK})=(\text{B}\Theta\text{K})$
    Επιπλέον $(\text{ABK})+(\text{B}\Theta\text{K}) = (\text{AB}\Theta\text{K})$ συνεπώς $(\text{ABK}) = \frac{1}{2}(\text{AB}\Theta\text{K}) .$
    Η $\text{KE}$ είναι διάμεσος του $\text{ABK}$ συνεπώς $(\text{AEK}) = \frac{1}{2}(\text{ABK}) .$
    Άρα $(\text{AEK}) = \frac{1}{4}(\text{AB}\Theta\text{K})$
    Όμως από α) είναι $(\text{AB}\Gamma\Delta) = (\text{AB}\Theta\text{K})$
    Συνεπώς $(\text{AEK}) = \frac{1}{4}(\text{AB}\Gamma\Delta)$
\end{alist}
