\begin{alist}
    \item Έστω $\Gamma\Delta=x$, τότε $B\Delta=20-x$ και επειδή
    \[ A\Delta^2 = B\Delta \cdot \Delta\Gamma \Leftrightarrow 8^2 = (20 - x) \cdot x \]
    \[ x^2 - 20x + 64 = 0 \]
    ή
    \[ x = 4 \text{ ή } x = 16 \]
    Άρα $\Gamma\Delta=4$ και $B\Delta=16$. Η λύση $\Gamma\Delta=16$ και $B\Delta=4$ απορρίπτεται γιατί $B\Delta > \Gamma\Delta$.

    \item $A\Gamma^2 = B\Gamma \cdot \Gamma\Delta$, άρα $A\Gamma^2 = 20 \cdot 4 = 80$
    και $A\Gamma = \sqrt{80} = 4\sqrt{5}.$
    Ομοίως $AB^2 = B\Gamma \cdot B\Delta = 16 \cdot 20 = 320$
    και $AB = \sqrt{320} = 8\sqrt{5}$
\end{alist}
