    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.

Δίνονται τα διανύσματα $\vec{\alpha}=(1,-2)$ και $\vec{\beta}=(2,3)$.
\begin{alist}
\item Να βρείτε το διάνυσμα $\vec{u}=2\vec{\alpha}+\vec{\beta}$. \monades{8}
\item Αν $\vec{u}=(4,-1)$ να βρείτε την τιμή του $\kappa\in R$ ώστε το διάνυσμα $\vec{u}$ να είναι κάθετο στο διάνυσμα $\vec{v}=(1,\kappa)$. \monades{9}
\item Για $\kappa=4$ να υπολογίσετε το μέτρο του διανύσματος $\vec{v}$ του προηγούμενου ερωτήματος. \monades{8}
\end{alist}
