\begin{alist}
\item Να δείξετε ότι τα διανύσματα $\vec{a}, \vec{b}$ είναι αντίρροπα και ότι $|\vec{b}| = 4|\vec{a}|$. \monades{12}
\item Να βρείτε τη γωνία που σχηματίζουν τα διανύσματα $\vec{a}, \vec{b}$ και να δικαιολογήσετε την απάντησή σας. \monades{6}

\end{alist}
