\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα διανύσματα: $\vec{a} = 3\vec{i} + 3\sqrt{3}\vec{j}$, $\vec{β} = \sqrt{2}\vec{i}$, $\vec{γ} = -3\vec{j}$ και $\vec{δ} = (-1, 1)$.
\begin{alist}
\item Να βρείτε το συντελεστή διεύθυνσης καθενός από τα διανύσματα $\vec{a}$, $\vec{β}$ και $\vec{δ}$. \monades{09}
\item Να γράψετε τη γωνία που σχηματίζει καθένα από τα διανύσματα $\vec{a}$, $\vec{β}$, $\vec{γ}$ και $\vec{δ}$ με το θετικό ημιάξονα Ox. \monades{10}
\item Να βρείτε τα μέτρα των διανυσμάτων $\vec{a}$ και $\vec{γ}$. \monades{06}
\end{alist}
