\begin{ylh}{1}
    \item 3.2 Η Παραβολή.
\end{ylh}
Δίνεται η παραβολή (C) με εξίσωση
\[ y^2 = x \quad (1) \]
\begin{alist}
\item Να βρείτε τις συντεταγμένες της εστίας $E$ και την εξίσωση της διευθετούσας (δ). \monades{12}
\item Να αποδείξετε ότι το σημείο $A(1,-1)$ είναι σημείο της παραβολής. \monades{05}
\item Να βρείτε την εξίσωση της εφαπτομένης παραβολής στο σημείο της $A(1,-1)$. \monades{08}
\end{alist}
