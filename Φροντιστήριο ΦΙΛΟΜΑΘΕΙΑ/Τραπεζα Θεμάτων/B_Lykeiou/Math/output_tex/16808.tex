\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία του επιπέδου $A(-8, 1)$, $B(4, 5)$ και $\Gamma(-4, 9)$.
\begin{alist}
\item Να βρείτε τις συντεταγμένες του μέσου $K$ του ευθύγραμμου τμήματος $AB$. \monades{8}
\item Να δείξετε ότι ο κύκλος $(C)$ που έχει κέντρο το σημείο $K$ και διάμετρο το τμήμα $AB$ διέρχεται από το σημείο $\Gamma$. \monades{9}
\item Να βρείτε την εξίσωση του κύκλου $(C)$. \monades{8}
\end{alist}
