\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία $A(1,0)$ και $B(0,-1)$ και ο κύκλος $c_1$ με εξίσωση
\[
c_1: \left(x - \frac{1}{2}\right)^2 + \left(y + \frac{1}{2}\right)^2 = 2.
\]
\begin{alist}
\item Να αποδείξετε ότι το σύνολο των σημείων $N(x,y)$ του επιπέδου, τα οποία ικανοποιούν τη σχέση $\overrightarrow{NA}^2 - \overrightarrow{NB}^2 = 4$, ανήκουν στην ευθεία $(\varepsilon)$ με εξίσωση $y = -x - 2$. \monades{07}
\item Να αποδείξετε ότι το σύνολο των σημείων $P$ του επιπέδου, τα οποία ικανοποιούν την εξίσωση $2x^2 + 2y^2 + 10x + 14y + 21 = 0$, ανήκουν σε κύκλο $c_2$ κέντρου $\Lambda\left(-\frac{5}{2}, -\frac{7}{2}\right)$ και ακτίνας $R = 2\sqrt{2}$. \monades{06}
\item \begin{rlist}
    \item Να αποδείξετε ότι οι δύο κύκλοι $c_1$ και $c_2$ εφάπτονται εξωτερικά και στη συνέχεια να βρείτε την ελάχιστη και τη μέγιστη απόσταση των σημείων τους. \monades{06}
    \item Να αποδείξετε ότι η ευθεία $(\varepsilon)$ είναι κοινή εφαπτομένη των κύκλων $c_1$ και $c_2$. \monades{06}
\end{rlist}
\end{alist}
