    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου
\end{ylh}
Δίνονται τα σημεία $\mathrm{A}(2,1)$, $\mathrm{B}(3,-1)$ και $\Gamma(-2,0)$.
\begin{alist}
    \item[α)]
    \begin{rlist}
        \item Να αποδείξετε ότι τα σημεία $\mathrm{A}$, $\mathrm{B}$ και $\Gamma$ δεν είναι συνευθειακά. \monades{07}
        \item Να αποδείξετε ότι το εμβαδόν του τριγώνου $\mathrm{AB} \Gamma$ ισούται με $\frac{9}{2}$ τετραγωνικές μονάδες. \monades{03}
    \end{rlist}
    \item[β)] Να βρεθεί ο γεωμετρικός τόπος των σημείων $\Delta(x,y)$ για τα οποία ισχύει $(\Delta \mathrm{A} \Gamma)=(\mathrm{AB} \Gamma)$. \monades{07}
    \item[γ)] Αν ο γεωμετρικός τόπος των σημείων $\Delta$ του ερωτήματος (β) αποτελείται από τις ευθείες $\varepsilon_{1}: x-4 y-7=0$ και $\varepsilon_{2}: x-4 y+11=0$, τότε:
    \begin{rlist}
        \item Να αποδείξετε ότι οι ευθείες $\mathrm{A} \Gamma, \varepsilon_{1}$ και $\varepsilon_{2}$ είναι παράλληλες. \monades{03}
        \item Να εξετάσετε αν είναι αληθής ή Ψευδής ο ισχυρισμός « οι ευθείες $x-4 y-7=0$ και $x-4 y+11=0$ έχουν ως μεσοπαράλληλο την ευθεία $\mathrm{A} \Gamma$ ». \monades{05}
    \end{rlist}
\end{alist}
