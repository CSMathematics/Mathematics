\Askhsh{22273}{4}
\begin{ylh}{1}
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Δίνεται η έλλειψη με εξίσωση $\frac{x^2}{9} + \frac{y^2}{4} = 1$ (1).

\begin{alist}
\item Να προσδιορίσετε δικαιολογώντας την απάντησή σας τις συντεταγμένες :
    \begin{rlist}
    \item Των σημείων που η έλλειψη τέμνει τους άξονες $x'x$ και $y'y$.
    \item Των εστιών $E$ και $E'$ της έλλειψης.
    \end{rlist}
    \monades{12}
\item Να βρείτε τις εξισώσεις των ευθειών που διέρχονται από το σημείο $A(0 , 4)$ και εφάπτονται στη καμπύλη που περιγράφει η εξίσωση (1).
    \monades{13}
\end{alist}
