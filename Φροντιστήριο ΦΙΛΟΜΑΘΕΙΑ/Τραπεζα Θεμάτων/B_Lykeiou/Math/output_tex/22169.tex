\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται η υπερβολή $\frac{x^2}{\alpha^2} - \frac{y^2}{\beta^2} = 1$ με ασύμπτωτη την $y = \frac{3}{4}x$. Η απόσταση των κορυφών της $A$ και $A'$ είναι 8.
\begin{alist}
\item
    \begin{rlist}
    \item Να βρείτε την εξίσωση της υπερβολής. \monades{10}
    \item Ποιες είναι οι εστίες της υπερβολής; \monades{05}
    \end{rlist}
\item Να βρείτε την εφαπτομένη της $\frac{x^2}{16} - \frac{y^2}{9} = 1$ στο σημείο $\left(5, \frac{9}{4}\right)$. \monades{10}
\end{alist}
