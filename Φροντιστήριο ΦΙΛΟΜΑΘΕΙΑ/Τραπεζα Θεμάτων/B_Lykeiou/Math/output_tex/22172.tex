\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Θεωρούμε την ευθεία $\varepsilon: 3x - 4y = 0$ και το σημείο $A(-2,1)$.
\begin{alist}
\item Να αποδείξετε ότι η απόσταση του σημείου $A$ από την ευθεία $(\varepsilon)$ είναι 2. \monades{8}
\item Να βρείτε την εξίσωση ευθείας $(\eta)$ κάθετης στην $(\varepsilon)$ που διέρχεται από το σημείο $A$. \monades{10}
\item Να βρείτε την εξίσωση του κύκλου που έχει κέντρο το σημείο $A$ και εφάπτεται στην ευθεία $(\varepsilon)$. \monades{7}
\end{alist}
