\Askhsh{16003}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Θεωρούμε την οικογένεια των ευθειών $\varepsilon_\alpha : (\alpha-4)x - 2\alpha y + \alpha + 4 = 0, \alpha \in \mathbb{R}.$
\begin{alist}
\item Να βρείτε τις ευθείες που προκύπτουν όταν $\alpha=0$ και όταν $\alpha=1$ και κατόπιν να προσδιορίσετε το κοινό τους σημείο M. \monades{8}
\item Να αποδείξετε ότι όλες οι ευθείες της οικογένειας διέρχονται από το M. \monades{6}
\item Έστω ότι μια ευθεία της παραπάνω οικογένειας τέμνει τους θετικούς ημιάξονες Ox, Oy στα σημεία A και B αντίστοιχα.
\begin{rlist}
\item Να αποδείξετε ότι $0 < \alpha < 4$. \monades{6}
\item Να βρείτε για ποια τιμή του $\alpha$ ισχύει $(\text{OA}) = 2(\text{OB})$. \monades{5}
\end{rlist}
\end{alist}
