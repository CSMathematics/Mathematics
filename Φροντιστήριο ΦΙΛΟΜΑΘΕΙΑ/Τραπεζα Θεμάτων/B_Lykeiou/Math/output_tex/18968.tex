\Askhsh{18968}{2}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
    \item 3.2 Απόσταση Σημείου από Ευθεία.
\end{ylh}
Δίνεται ο κύκλος C με εξίσωση $x^2 + y^2 - 6x - 8y = 0 \quad :(1)$.
\begin{alist}
\item Να δείξετε ότι ο κύκλος C έχει κέντρο $K(3,4)$ και ακτίνα $\rho=5$.  \monades{10}
\item Να δείξετε ότι η απόσταση του κέντρου K του κύκλου από την ευθεία $\varepsilon: 3x + 4y = 0$ ισούται με 5 μονάδες μήκους.  \monades{8}
\item Να δικαιολογήσετε αν είναι αληθής ή ψευδής ο ισχυρισμός: «Ο κύκλος C και η ευθεία $\varepsilon$ εφάπτονται».  \monades{7}
\end{alist}
