\Askhsh{15432}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ x^2 + y^2 - 4\kappa x - 2\kappa y + 4 = 0 \quad (1) \]
με $\kappa \in \mathbb{R}$.

\begin{alist}
\item Να βρείτε τις τιμές του $\kappa \in \mathbb{R}$ ώστε η εξίσωση (1) να παριστάνει κύκλο. \monades{7}
\item Να βρείτε τις συντεταγμένες του κέντρου και την ακτίνα του κάθε κύκλου. \monades{3}
\item Να βρείτε την ευθεία στην οποία ανήκουν τα κέντρα των παραπάνω κύκλων. \monades{7}
\item Για $\kappa = 1$ να βρείτε την εξίσωση εφαπτομένης του αντίστοιχου κύκλου της εξίσωσης (1) στο σημείο $\Gamma(2,2)$. \monades{8}
\end{alist}
