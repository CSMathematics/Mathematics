\Askhsh{20718}{2}
\begin{ylh}{1}
    \item 3.3 Η Έλλειψη.
\end{ylh}
Δίνεται η έλλειψη $C$ με εξίσωση $\frac{x^2}{9} + \frac{y^2}{25} = 1$.
\begin{alist}
\item Να βρείτε τις εστίες της. \monades{8}
\item Να σχεδιάσετε την έλλειψη $C$ σε ορθοκανονικό σύστημα συντεταγμένων. \monades{8}
\item Να σχεδιάσετε στο ίδιο ορθοκανονικό σύστημα τις εφαπτόμενες στις κορυφές της $C$ και να γράψετε τις εξισώσεις τους. \monades{9}
\end{alist}
