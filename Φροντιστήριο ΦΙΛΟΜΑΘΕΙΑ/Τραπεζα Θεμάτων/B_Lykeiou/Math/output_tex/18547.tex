\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
    \item 1.6. Εμβαδόν Τριγώνου.
\end{ylh}
Δίνονται τα σημεία $A(0,-1)$, $B(\lambda, 1)$ και $\Gamma(\lambda-2, \lambda-3)$, όπου $\lambda \in \mathbb{R}$.
\begin{alist}
    \item Να βρείτε τις τιμές του $\lambda \in \mathbb{R}$ ώστε:
    \begin{rlist}
        \item Τα σημεία $A, B$ και $\Gamma$ να είναι κορυφές τριγώνου. \monades{8}
        \item Το τρίγωνο $AB\Gamma$ να είναι ορθογώνιο με $\widehat{A} = 90^\circ$. \monades{7}
    \end{rlist}
    \item Για $\lambda = -2$, να βρείτε:
    \begin{rlist}
        \item Το εσωτερικό γινόμενο $\vec{AB} \cdot \vec{A\Gamma}$. \monades{4}
        \item Το εμβαδό του τριγώνου $AB\Gamma$. \monades{6}
    \end{rlist}
\end{alist}
