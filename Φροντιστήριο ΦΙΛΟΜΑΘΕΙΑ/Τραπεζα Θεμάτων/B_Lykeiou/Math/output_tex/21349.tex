\Askhsh{21349}{4}
\begin{ylh}{1}
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων
    \item 3.1 Ο Κύκλος.
\end{ylh}
Σε ορθοκανονικό σύστημα αξόνων με αρχή το σημείο Ο θεωρούμε κύκλο (C) και ευθεία (ε) με εξισώσεις $x^2 + y^2 - 9x - 3y + 10 = 0$ (1) και $4x + 3y - 10 = 0$ (2) αντίστοιχα.

\begin{alist}
\item
\begin{rlist}
    \item Να βρείτε το κέντρο Κ και την ακτίνα R του κύκλου (C). \monades{5}
    \item Να υπολογίσετε την απόσταση του κέντρου Κ από την ευθεία (ε) και να αποδείξετε ότι η ευθεία (ε) τέμνει τον κύκλο (C) σε δύο σημεία. \monades{4}
    \item Να προσδιορίσετε τα σημεία Α και Β στα οποία η ευθεία (ε) τέμνει τον κύκλο (C). \monades{5}
\end{rlist}
\item Αν είναι $A(1,2)$ και $B(4, -2)$, τότε:
\begin{rlist}
    \item Να υπολογίσετε το εσωτερικό γινόμενο $\vec{OA} \cdot \vec{OB}$. \monades{5}
    \item Να αποδείξετε ότι ο κύκλος με διάμετρο ΑΒ διέρχεται από το σημείο Ο. \monades{6}
\end{rlist}
\end{alist}
