\Askhsh{18416}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ x(x-4)+y(y-2)=2(x+y-4)\quad (1) . \]
\begin{alist}
\item Να δείξετε ότι η εξίσωση (1) παριστάνει κύκλο με κέντρο $K(3,2)$ και ακτίνα $\rho=\sqrt{5}$. \monades{6}
\item Δίνονται τα σημεία $A(4,4)$ και $B(2,0)$.
    \begin{rlist}
    \item Να δείξετε ότι τα σημεία A και B είναι αντιδιαμετρικά σημεία του κύκλου. \monades{4}
    \item Να βρείτε τις εξισώσεις των εφαπτόμενων του κύκλου οι οποίες είναι παράλληλες στην διάμετρο AB. \monades{9}
    \end{rlist}
\item Να βρείτε την τιμή της παραμέτρου $\lambda$ ώστε η ευθεία $(\eta)$ με εξίσωση $y=\lambda x+4$ να τέμνει τον παραπάνω κύκλο σε δύο σημεία $\Gamma$ και $\Delta$ ώστε $(\Gamma\Delta)=\sqrt{20}$. \monades{6}
\end{alist}
