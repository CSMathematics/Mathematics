\Askhsh{15178}{3}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Δίνεται η εξίσωση $(\mu + 1)x + (\mu + 2)y = 0, \mu \in \mathbb{R}$ (1).
\begin{alist}
    \item \begin{rlist}
        \item Να αποδείξετε ότι η εξίσωση (1) παριστάνει ευθεία για κάθε $\mu \in \mathbb{R}$. \monades{08}
        \item Να αποδείξετε ότι όλες οι ευθείες που ορίζονται από την εξίσωση (1) διέρχονται από την αρχή των αξόνων. \monades{02}
    \end{rlist}
    \item \begin{rlist}
        \item Πότε η εξίσωση (1) παριστάνει ευθεία που έχει συντελεστή διεύθυνσης $0$; Ποια είναι η εξίσωσή της; \monades{03}
        \item Πότε η εξίσωση (1) παριστάνει ευθεία για την οποία δεν ορίζεται συντελεστής διεύθυνσης; Ποια είναι η εξίσωσή της; \monades{03}
    \end{rlist}
    \item Να βρείτε για ποια τιμή του πραγματικού αριθμού $\mu$, προκύπτει ευθεία η οποία σχηματίζει γωνία $45^\circ$ με τον άξονα $x'x$. Ποια είναι η εξίσωσή της; \monades{09}
\end{alist}
