\Askhsh{15680}{2}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος $C: x^2 + y^2 - 2x - 4y + 1 = 0$ με κέντρο $K(1,2)$ και η ευθεία $\varepsilon: 3x+4y+1=0$.
\begin{alist}
\item Να αποδείξετε ότι η ακτίνα του κύκλου $C$ είναι $\rho = 2$. \monades{10}
\item Να αποδείξετε ότι η απόσταση του κέντρου $K$ από την ευθεία $\varepsilon$ είναι $\frac{12}{5}$. \monades{10}
\item Να αιτιολογήσετε γιατί η ευθεία $\varepsilon$ και ο κύκλος $C$ δεν έχουν κοινά σημεία. \monades{5}
\end{alist}
