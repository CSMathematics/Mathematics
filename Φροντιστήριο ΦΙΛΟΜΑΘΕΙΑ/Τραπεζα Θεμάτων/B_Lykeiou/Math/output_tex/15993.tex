\Askhsh{15993}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση $(x - 2)^2 + (y - \lambda)^2 = \lambda^2 + 1$ (1), όπου $\lambda \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι για κάθε τιμή του $\lambda$ η (1) παριστάνει κύκλο, του οποίου να βρείτε το κέντρο και την ακτίνα. \monades{03}
\item Να αποδείξετε ότι όλοι οι κύκλοι που ορίζονται από την (1) για τις διάφορες τιμές του $\lambda$ διέρχονται από δύο σταθερά σημεία. \monades{10}
\item Αν $A(1,0)$ και $B(3,0)$ είναι τα μοναδικά σημεία από τα οποία διέρχονται όλοι οι κύκλοι, τότε να βρείτε την εξίσωση της κοινής χορδής τους και να αποδείξετε ότι είναι κάθετη στην ευθεία που διέρχεται από τα κέντρα των κύκλων. \monades{07}
\item Αν ένα σημείο $M(\alpha, \beta)$ επαληθεύει την (1) για κάθε $\lambda \in \mathbb{R}$, τότε να αποδείξετε ότι $\alpha \cdot \beta = 0$. \monades{05}
\end{alist}
