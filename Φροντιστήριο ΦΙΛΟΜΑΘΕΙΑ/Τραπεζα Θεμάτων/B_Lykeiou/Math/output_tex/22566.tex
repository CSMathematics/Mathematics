\Askhsh{22566}{2}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται η υπερβολή με εξίσωση $4x^2 - y^2 = 4$.
\begin{alist}
\item Να αποδείξετε ότι οι συντεταγμένες της κορυφής της υπερβολής είναι $A(1,0)$ και $A'(-1,0)$. \monades{8}
\item Να αποδείξετε ότι οι ασύμπτωτες της υπερβολής είναι οι $y = 2x$ και $y = -2x$. \monades{9}
\item Να αποδείξετε ότι η ευθεία που διέρχεται από την κορυφή $A$ και είναι παράλληλη προς την ασύμπτωτη $y = -2x$ έχει εξίσωση $y = -2x + 2$. \monades{8}
\end{alist}
