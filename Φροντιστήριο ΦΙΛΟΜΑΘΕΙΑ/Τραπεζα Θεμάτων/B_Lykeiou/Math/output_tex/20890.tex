\begin{alist}
\item την εξίσωση της πλευράς $\mathrm{B\Gamma}$. \monades{10}
\item την εξίσωση του κύκλου που έχει κέντρο το $\mathrm{A}$ και εφάπτεται στην πλευρά $\mathrm{B\Gamma}$. \monades{15}
\end{alist}
