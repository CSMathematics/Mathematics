    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία $A(1,1)$ και $B(5,5)$.
\begin{alist}
\item Αν για το σημείο $M(x, \psi)$ ισχύει $\overrightarrow{AM}^2 + \overrightarrow{BM}^2 = 32$, να αποδείξετε ότι:
    \begin{rlist}
    \item Το σημείο $M$ βρίσκεται πάνω στην καμπύλη με εξίσωση $x^2 + \psi^2 - 6\psi - 6x + 10 = 0 \ (1)$ \monades{08}
    \item Η εξίσωση (1) παριστάνει κύκλο. \monades{03}
    \end{rlist}
\item Αν το κέντρο του κύκλου είναι το $K(3,3)$ και η ακτίνα του $\rho = 2\sqrt{2}$:
    \begin{rlist}
    \item Να διερευνήσετε για ποιες τιμές του $\lambda$ η ευθεία $(\varepsilon): \lambda x + \psi = 2$ εφάπτεται του κύκλου (1). \monades{07}
    \item Υπάρχει τιμή του $\lambda$ για την οποία η ευθεία $(\varepsilon)$ σχηματίζει με την $AB$ γωνία $45^\circ$; \monades{07}
    \end{rlist}
\end{alist}
