\Askhsh{15080}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται οι εξισώσεις
\[
    C_1 : x^2 + y^2 - 2x - 8 = 0 \quad (1)
\]
και
\[
    C_2 : x^2 + y^2 - 6x + 8 = 0 \quad (2)
\]
\begin{alist}
\item Να δείξετε ότι οι (1) και (2) είναι εξισώσεις κύκλων, με κέντρα $K(1,0)$, $\Lambda(3,0)$ και ακτίνες $\rho_1=3$, $\rho_2=1$ αντίστοιχα. \monades{6}
\item
    \begin{rlist}
    \item Να βρείτε το μήκος της διακέντρου $(K\Lambda)$. \monades{5}
    \item Να δείξετε ότι ο κύκλος $C_2$ εφάπτεται εσωτερικά του κύκλου $C_1$. \monades{5}
    \end{rlist}
\item Να βρείτε τις εξισώσεις των ακτίνων του κύκλου $C_1$ που εφάπτονται στον κύκλο $C_2$. \monades{9}
\end{alist}
