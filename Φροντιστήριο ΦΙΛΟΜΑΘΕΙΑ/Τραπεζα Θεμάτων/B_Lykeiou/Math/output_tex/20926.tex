\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο.
    \item 1.5. Η Ευθεία.
\end{ylh}
Δίνεται η ευθεία $\varepsilon: x - 2y = 1$ και τα σημεία $A(0,2)$, $B(1,0)$.
\begin{alist}
\item Να αποδείξετε ότι το σημείο $B$ ανήκει στην ευθεία $\varepsilon$ ενώ το σημείο $A$ δεν είναι σημείο της $\varepsilon$. \monades{08}
\item Να βρείτε την απόσταση του σημείου $A$ από την ευθεία $\varepsilon$. \monades{08}
\item Να υπολογίσετε την απόσταση του $A$ από το $B$ και να αποδείξετε ότι η προβολή του $A$ στην ευθεία $\varepsilon$ είναι το $B$. \monades{09}
\end{alist}
