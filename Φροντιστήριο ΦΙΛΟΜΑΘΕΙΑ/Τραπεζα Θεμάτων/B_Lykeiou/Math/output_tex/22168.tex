\begin{ylh}{1}
    \item 3.2 Η Παραβολή
    \item 3.3 Η Έλλειψη.
\end{ylh}
Δίνονται η παραβολή $y^2 = 2px$ και η έλλειψη $\frac{x^2}{\alpha^2} + \frac{y^2}{\beta^2} = 1$.
\begin{alist}
\item Αν η παραβολή διέρχεται από το σημείο $A(1,2)$, να βρείτε:
    \begin{rlist}
    \item Την εξίσωση της παραβολής. \monades{10}
    \item Την εστία $E$ της παραβολής. \monades{05}
    \end{rlist}
\item Να βρεθεί η εξίσωση της έλλειψης με κέντρο το $O$, αν η μια εστία της είναι το σημείο $E(1,0)$ και ο μεγάλος άξονας της έχει μήκος ίσο με 4. \monades{10}
\end{alist}
