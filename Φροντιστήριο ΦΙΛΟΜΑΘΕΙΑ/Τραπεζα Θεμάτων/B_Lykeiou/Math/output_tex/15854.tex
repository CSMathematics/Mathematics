\Askhsh{15854}{2}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα διανύσματα $\vec{a} = (2, 1)$ και $\vec{b} = (-8, -4)$.
\begin{alist}
\item Να δείξετε ότι $\vec{a} // \vec{b}$. \monades{10}
\item Να δείξετε ότι για τα διανύσματα $\vec{a}, \vec{b}$ ισχύει $\vec{b} = -4\vec{a}$. \monades{7}
\item Να αποδείξετε ότι το μέτρο του διανύσματος $\vec{b}$ είναι τετραπλάσιο του μέτρου του διανύσματος $\vec{a}$. \monades{8}
\end{alist}
