\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Σε μια σύγχρονη πόλη, κατασκευάζεται σιδηροδρομικό δίκτυο που περιλαμβάνει:
\begin{itemize}
    \item τη γραμμή $\gamma_1$, κάθε σημείο της οποίας στο ορθοκανονικό σύστημα συντεταγμένων είναι της μορφής: $A(\lambda-1, 2\lambda+1)$, $\lambda \in \mathbb{R}$.
    \item τη γραμμή $\gamma_2$, που περνάει από το σταθμό $Σ(-4, 2)$ και είναι παράλληλη στο διάνυσμα $\vec{u} = (-1, 3)$.
\end{itemize}
\begin{alist}
    \item Να βρείτε τις εξισώσεις των ευθειών πάνω στις οποίες βρίσκονται οι γραμμές $\gamma_1$ και $\gamma_2$. \monades{10}
    \item Η είσοδος του αθλητικού σταδίου μιας συνοικίας θα βρίσκεται στο σημείο $K(1, 1)$ του ορθοκανονικού συστήματος συντεταγμένων. Οι κατασκευαστές θέλουν να συνδέσουν την είσοδο του σταδίου απ' ευθείας με κάθετο δρόμο, με μια από τις γραμμές $\gamma_1$ και $\gamma_2$. Να βρείτε με ποια από τις δύο γραμμές είναι πιο συμφέρουσα η σύνδεση. Δίνεται ότι το κόστος σύνδεσης ανά μονάδα μήκους, είναι το ίδιο και για τις δύο γραμμές. \monades{9}
    \item Γύρω από το στάδιο θα δημιουργηθεί κυκλικό πάρκο. Να βρείτε την εξίσωση του κύκλου, που θα ορίζει το πάρκο, αν το κέντρο του είναι το σημείο $K$ και επιπλέον ο κύκλος αυτός εφάπτεται της γραμμής $\gamma_1$. \monades{6}
\end{alist}
