\Askhsh{15646}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται οι κύκλοι $C_1 : (x-1)^2+(y-1)^2=9$ και $C_2 : (x-4)^2+(y-4)^2=9$.
\begin{alist}
\item Να δείξετε ότι τα κέντρα K, Λ των κύκλων $C_1$ και $C_2$ αντίστοιχα βρίσκονται στην διχοτόμο της γωνίας $x\hat{O}y$ του συστήματος συντεταγμένων. \monades{8}
\item Να βρείτε τα σημεία τομής B, Γ των κύκλων $C_1$ και $C_2$. \monades{7}
\item Να βρείτε τα σημεία της ευθείας $y=x$ ώστε το τρίγωνο που σχηματίζεται με τα B, Γ να έχει εμβαδόν $\frac{21}{2} \tau.\mu.$ \monades{10}
\end{alist}
