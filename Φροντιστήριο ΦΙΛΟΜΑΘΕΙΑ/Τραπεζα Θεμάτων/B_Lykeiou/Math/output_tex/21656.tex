\Askhsh{21656}{4}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Έστω υπερβολή $C$ με κέντρο το $(0,0)$, εστίες τα σημεία $E(5,0)$, $E'(-5,0)$ και κορυφές τα σημεία $A(4,0)$, $A'(-4,0)$.
\begin{alist}
\item Να βρείτε:
    \begin{rlist}
    \item τις εξισώσεις των ασυμπτώτων της υπερβολής $C$. \monades{3}
    \item την εξίσωση της υπερβολής $C$. \monades{3}
    \end{rlist}
\item Να σχεδιάσετε στο ίδιο ορθοκανονικό σύστημα, την υπερβολή $C$, τις ασύμπτωτες της $C$ και το ορθογώνιο βάσης της $C$. \monades{9}
\item Αν $M$ τυχαίο σημείο της $C$, να βρείτε την τιμή της παράστασης $(ME)-(ME')$. \monades{5}
\item Αν $M(\sqrt{80},6)$ σημείο της $C$, να βρείτε την εξίσωση της διχοτόμου της γωνίας $\widehat{EME'}$. \monades{5}
\end{alist}
