    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος C: $x^2 + y^2 = 4$ και το σημείο A$(2\sqrt{2}, 0)$.
\begin{alist}
\item
    \begin{rlist}
    \item Να αποδείξετε ότι το σημείο A είναι εξωτερικό του κύκλου C. \monades{05}
    \item Να βρείτε τις εξισώσεις των εφαπτόμενων του κύκλου C που διέρχονται από το σημείο A και να αποδείξετε ότι είναι μεταξύ τους κάθετες. \monades{12}
    \end{rlist}
\item Αν B, $\Gamma$ τα σημεία επαφής του κύκλου C με τις εφαπτόμενες ευθείες από το σημείο A, να υπολογίσετε το εμβαδό του τετραπλεύρου $ABO\Gamma$. \monades{08}
\end{alist}
