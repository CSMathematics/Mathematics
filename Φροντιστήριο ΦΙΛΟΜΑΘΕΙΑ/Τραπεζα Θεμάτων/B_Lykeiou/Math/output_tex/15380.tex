\begin{alist}
\item Να βρείτε για ποια τιμή του $\alpha$, η απόσταση του σημείου $A$ από το σημείο $B$ είναι ίση με την απόσταση του σημείου $A$ από την ευθεία $\varepsilon$. \monades{8}
\item Για $\alpha = 4$
    \begin{rlist}
    \item Να βρείτε το εμβαδόν του τριγώνου $AB\Gamma$, όπου $\Gamma$ το σημείο τομής της ευθείας $\varepsilon$ με τον άξονα $y'y$. \monades{8}
    \item Να βρείτε το σημείο της ευθείας $\varepsilon$ που απέχει την μικρότερη απόσταση από την αρχή των αξόνων. \monades{9}
    \end{rlist}
\end{alist}
