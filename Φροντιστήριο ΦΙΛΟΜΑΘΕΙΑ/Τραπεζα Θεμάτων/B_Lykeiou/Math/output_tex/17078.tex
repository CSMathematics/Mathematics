\Askhsh{17078}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας.
\end{ylh}
Δίνονται τα σημεία $A(3, 2\alpha)$, $B(4, \alpha)$, $\Gamma(\alpha + 1, 1 - \alpha)$ και $\Delta(\alpha, 1)$, με $\alpha \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Η ευθεία που διέρχεται από τα σημεία $A$ και $B$ έχει εξίσωση $y = -\alpha x + 5\alpha$. \monades{6}
    \item Τα σημεία $\Gamma$ και $\Delta$ ανήκουν στην ευθεία $AB$ αν και μόνο αν $\alpha = \frac{5 \pm \sqrt{21}}{2}$. \monades{7}
    \item Το τετράπλευρο $AB\Gamma\Delta$ είναι παραλληλόγραμμο όταν $\alpha \ne \frac{5 \pm \sqrt{21}}{2}$. \monades{7}
    \end{rlist}
\item Θεωρήστε τον ισχυρισμό: «Υπάρχει πραγματικός αριθμός $\alpha$ ώστε το τετράπλευρο $AB\Gamma\Delta$ να είναι τετράγωνο.» Είναι αληθής ή ψευδής ο παραπάνω ισχυρισμός; Να αιτιολογήσετε την απάντησή σας. \monades{5}
\end{alist}
