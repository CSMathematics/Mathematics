\Askhsh{15692}{4}
Δίνεται η εξίσωση $x^2+y^2+y=x+2xy+6$.
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item $(x-y)^2-(x-y)-6=0$. \monades{4}
    \item Η εξίσωση παριστάνει ένα ζεύγος παράλληλων ευθειών, τις οποίες να βρείτε. \monades{4}
    \end{rlist}
\item Έστω $\varepsilon_1: x-y-3=0$ και $\varepsilon_2: x-y+2=0$ οι δύο παράλληλες ευθείες. Να αποδείξετε ότι όλα τα σημεία $M\left(\alpha, \alpha-\frac{1}{2}\right), \alpha \in \mathbb{R}$ ισαπέχουν από τις δύο ευθείες. \monades{10}
\item Να βρείτε την μεσοπαράλληλη των δύο ευθειών. \monades{7}
\end{alist}
