    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ (y - 1)^2 = (3 + x)(1 - x) \quad (1) \]
Να αποδείξετε ότι:
\begin{alist}
    \item Η εξίσωση (1) παριστάνει κύκλο με κέντρο $K(-1,1)$ και ακτίνα $R = 2$. \monades{9}
    \item Η αρχή $O(0,0)$ των αξόνων είναι εσωτερικό σημείο του κύκλου $(K,R)$. \monades{7}
    \item Η ευθεία $(\varepsilon): x + y = 2$ είναι τέμνουσα του κύκλου $(K,R)$. \monades{9}
\end{alist}
