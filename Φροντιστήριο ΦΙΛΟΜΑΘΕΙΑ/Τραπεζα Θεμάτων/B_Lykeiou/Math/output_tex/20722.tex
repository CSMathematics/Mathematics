\begin{alist}
\item Να βρείτε το είδος της καμπύλης $C$ πάνω στην οποία κινείται το σημείο $K$ και να γράψετε την εξίσωσή της, αιτιολογώντας την απάντησή σας. \monades{6}
\end{alist}
Έστω $C: \frac{x^2}{25} + \frac{y^2}{16} = 1$ και $(\varepsilon): 3x+5y = 25$.

\begin{alist}
\item Να αποδείξετε ότι $C$ και $(\varepsilon)$ έχουν ένα μόνο κοινό σημείο $M$ και να βρείτε τις συντεταγμένες του σημείου $M$. \monades{7}

\item Να ερμηνεύσετε γραφικά το συμπέρασμα του ερωτήματος β) και να σχεδιάσετε στο ίδιο ορθοκανονικό σύστημα την έλλειψη $C$ και την ευθεία $\varepsilon$. \monades{6}

\item Να σχεδιάσετε τη διχοτόμο της γωνίας $\widehat{EME'}$ και να βρείτε την εξίσωσή της. \monades{6}
\end{alist}
