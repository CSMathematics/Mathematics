\Askhsh{22508}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Οι κορυφές $A$, $\Gamma$ ενός τετραγώνου $AB\Gamma\Delta$ είναι τα σημεία $(1,4)$ και $(3,0)$ αντιστοίχως.
\begin{alist}
\item Να αποδείξετε ότι η εξίσωση της μεσοκαθέτου του ευθυγράμμου τμήματος $A\Gamma$ γράφεται στη μορφή $y - 2 = \frac{1}{2}(x - 2)$. \monades{8}
\item Να αποδείξετε ότι η εξίσωση του κύκλου, ο οποίος έχει διάμετρο το ευθύγραμμο τμήμα $A\Gamma$ γράφεται στη μορφή $(x - 2)^2 + (y - 2)^2 = 5$. \monades{8}
\item Να υπολογίσετε τις συντεταγμένες των δύο άλλων κορυφών $B,\Delta$ του τετραγώνου. \monades{9}
\end{alist}
