\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία $O(0,0)$, $A(\alpha,0)$ και $B(0,\beta)$, όπου $\alpha,\beta > 0$.
\begin{alist}
\item Να βρείτε συναρτήσει των $\alpha,\beta$
    \begin{rlist}
    \item τις συντεταγμένες του μέσου $M$ του τμήματος $AB$. \monades{5}
    \item την απόσταση $(OM)$. \monades{5}
    \end{rlist}
\item Αν $(OM) = \frac{\sqrt{\alpha^2 + \beta^2}}{2}$, τότε:
    \begin{rlist}
    \item να αποδείξετε ότι $(OM) = \frac{(AB)}{2}$. \monades{5}
    \item να γράψετε την πρόταση της Ευκλείδειας Γεωμετρίας που έχει αποδειχθεί. \monades{3}
    \end{rlist}
\item Να βρείτε την εξίσωση του περιγεγραμμένου κύκλου του τριγώνου $OAB$. \monades{7}
\end{alist}
