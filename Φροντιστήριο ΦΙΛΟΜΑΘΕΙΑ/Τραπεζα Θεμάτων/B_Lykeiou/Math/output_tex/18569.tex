\Askhsh{18569}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος $C$: $x^2+y^2=1$.
\begin{alist}
\item Αν $A$ και $A'$ είναι τα σημεία τομής του κύκλου $C$ με τους ημιάξονες $Ox$ και $Ox'$ αντίστοιχα, τότε:
    \begin{rlist}
    \item Να αποδείξετε ότι οι συντεταγμένες των σημείων $A$ και $A'$ είναι $A(1,0)$ και $A'(-1,0)$. \monades{5}
    \item Να βρείτε την εξίσωση της ευθείας $\varepsilon$ που διέρχεται από το $A$ και σχηματίζει με τον άξονα $x'x$ γωνία $150^\circ$. \monades{6}
    \end{rlist}
\item Αν η ευθεία $\varepsilon$ τέμνει τον κύκλο $C$ και στο σημείο $B$, να αποδείξετε ότι η χορδή $AB$ έχει μήκος $\sqrt{3}$. \monades{8}
\item Αν η ευθεία $\varepsilon$ έχει εξίσωση $y = - \frac{\sqrt{3}}{3} (x-1)$, να βρείτε την εξίσωση της ευθείας ($\zeta$) που διέρχεται από τα σημεία $A'$ και $B$. \monades{6}
\end{alist}
