\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου
\end{ylh}
Δίνονται τα σημεία $A(1,3)$, $B(-2,2)$ και η ευθεία $\varepsilon: 3x+y+\alpha = 0$ με $\alpha \in \mathbb{R}$.
\begin{alist}
\item Να βρεθεί η απόσταση του σημείου $A$ από το σημείο $B$. \monades{5}
\item Για ποιες τιμές του $\alpha$, η απόσταση $AB$ είναι ίση με την απόσταση του σημείου $A$ από την ευθεία $\varepsilon$. \monades{8}
\item Για $\alpha = 4$ να βρεθεί το εμβαδόν του τριγώνου $AB\Gamma$, όπου $\Gamma$ το σημείο τομής της ευθείας $\varepsilon$ με τον άξονα $y'y$. \monades{12}
\end{alist}
