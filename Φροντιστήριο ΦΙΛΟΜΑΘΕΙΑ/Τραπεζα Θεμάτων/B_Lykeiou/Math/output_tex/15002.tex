\begin{ylh}{1}
    \item 1.3. Πολλαπλασιασμός Αριθμού με Διάνυσμα
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα σημεία $A(0,5)$ και $\Delta(4,5)$ και τα διανύσματα $\vec{AB} = (3,-3)$ και $\vec{A\Gamma} = (3,1)$.
\begin{alist}
    \item Να αποδείξετε ότι το σημείο $\Gamma$ έχει συντεταγμένες $\Gamma(3,6)$. \monades{11}
    \item
    \begin{rlist}
        \item Να βρείτε τις συντεταγμένες του διανύσματος $\vec{\Gamma\Delta}$. \monades{6}
        \item Να αποδείξετε ότι $\vec{AB} // \vec{\Gamma\Delta}$. \monades{8}
    \end{rlist}
\end{alist}
