\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Δίνεται το σημείο $A(1, -3)$ και η ευθεία $\varepsilon: 4x + 6y = 1$.
\begin{alist}
\item Να εξηγήσετε γιατί το $A$ δεν είναι σημείο της ευθείας $\varepsilon$. \monades{10}
\item Να βρείτε την εξίσωση της ευθείας που διέρχεται από το σημείο $A$ και είναι παράλληλη στην ευθεία $\varepsilon$. \monades{15}
\end{alist}
