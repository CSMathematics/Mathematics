    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.

Δίνονται τα διανύσματα $\vec{α} = (-1,3)$, $\vec{β} = \left(-2, - \frac{1}{2}\right)$ και $\vec{v} = (x^2, x - 1)$.

\begin{alist}
    \item Να βρείτε τις συντεταγμένες του διανύσματος $\vec{u} = \vec{α} - 2\vec{β}$. \monades{07}
    \item Να βρείτε τους αριθμούς $x$ για τους οποίους τα διανύσματα $\vec{u} = (3,4)$ και $\vec{v}$ είναι κάθετα. \monades{09}
    \item Να βρείτε τους αριθμούς $x$ για τους οποίους τα διανύσματα $\vec{v}$ και $\vec{β}$ είναι συγγραμμικά; \monades{09}
\end{alist}
