\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1. Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία $A(1,0)$ και $B(3,0)$.
\begin{alist}
\item Να βρείτε την εξίσωση της μεσοκαθέτης ευθείας ($\zeta$) του ευθύγραμμου τμήματος $AB$.  \monades{07}
\item Αν $K$ είναι ένα τυχαίο σημείο της ευθείας ($\zeta$), να βρείτε την εξίσωση ($c$) όλων των κύκλων, οι οποίοι έχουν κέντρο $K$ και διέρχονται από τα σημεία $A$ και $B$, συναρτήσει μιας παραμέτρου $\lambda \in \mathbb{R}$.  \monades{08}
\item Αν η εξίσωση $(x - 2)^2 + (y - \lambda)^2 = \lambda^2 + 1, \lambda \in \mathbb{R}$, παριστάνει όλους τους κύκλους ($c$) του ερωτήματος β), τότε:
    \begin{rlist}
    \item Να σχεδιάσετε τον κύκλο, ο οποίος έχει διάμετρο το ευθύγραμμο τμήμα $AB$.  \monades{05}
    \item Να αποδείξετε ότι η ευθεία ($\varepsilon$): $x + \lambda y - 1 = 0$ εφάπτεται σε όλους τους κύκλους ($c$) στο σημείο $A(1,0)$.  \monades{05}
    \end{rlist}
\end{alist}
