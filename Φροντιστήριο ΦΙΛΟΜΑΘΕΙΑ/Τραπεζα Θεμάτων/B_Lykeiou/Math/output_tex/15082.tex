\Askhsh{15082}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται δύο κύκλοι με εξισώσεις:
\[
C_1:(x-2)^2+(y-3)^2=8 \quad \text{και} \quad C_2:(x-7)^2+(y+2)^2=18.
\]
\begin{alist}
\item Να υπολογίσετε το μήκος της διακέντρου (ΚΛ), όπου Κ,Λ τα κέντρα των κύκλων $C_1, C_2$ αντίστοιχα. Ακολούθως να δείξετε ότι οι δύο κύκλοι εφάπτονται εξωτερικά. \monades{5}
\item
    \begin{rlist}
    \item Να βρείτε την εξίσωση της ευθείας ΚΛ . \monades{5}
    \item Να βρείτε τα σημεία τομής της ευθείας ΚΛ με τον κύκλο $C_1$ και το σημείο επαφής των δύο κύκλων. \monades{7}
    \end{rlist}
\item Να βρείτε την εξίσωση της κοινής εσωτερικής εφαπτομένης των κύκλων. \monades{8}
\end{alist}
