\Askhsh{21154}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ x^2 + y^2 - 4ax - 4ay = 0 \quad (1) \]
όπου $a$ είναι πραγματικός αριθμός.

\begin{alist}
\item Να βρείτε τις τιμές του $a$ για τις οποίες η εξίσωση (1) παριστάνει κύκλο. \monades{8}
\item Να προσδιορίσετε το κέντρο $K$ και την ακτίνα $R$ των κύκλων ως συνάρτηση του $a$. \monades{6}
\item Να βρείτε τον γεωμετρικό τόπο των κέντρων των κύκλων για τις διάφορες τιμές του $a$ του ερωτήματος (α). \monades{5}
\item Να εξετάσετε αν υπάρχει τιμή του $a$ ώστε ο αντίστοιχος κύκλος που ορίζεται από την εξίσωση (1) να εφάπτεται στον άξονα $x'x$. \monades{6}
\end{alist}
