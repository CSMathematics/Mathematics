\Askhsh{22063}{4}
\begin{ylh}{1}
    \item 1.1. Η Έννοια του Διανύσματος
    \item 1.2. Πρόσθεση και Αφαίρεση Διανυσμάτων
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
\end{ylh}
\begin{alist}
    \item Έστω $\vec{\alpha}, \vec{\beta}$ δύο μη μηδενικά διανύσματα. Να αποδείξετε ότι:
    \begin{rlist}
        \item $|\vec{\alpha} + \vec{\beta}| = |\vec{\alpha}| + |\vec{\beta}| \Leftrightarrow \vec{\alpha} \uparrow\uparrow \vec{\beta}$. \monades{5}
        \item $|\vec{\alpha} + \vec{\beta}| = ||\vec{\alpha}| - |\vec{\beta}|| \Leftrightarrow \vec{\alpha} \uparrow\downarrow \vec{\beta}$. \monades{5}
    \end{rlist}
    \item Θεωρούμε τρία διανύσματα $\vec{\alpha}, \vec{\beta}, \vec{\gamma}$ για τα οποία ισχύουν ότι:
    \[ \vec{\alpha} + \vec{\beta} + \vec{\gamma} = \vec{0}, \quad |\vec{\alpha}| = 1, \quad |\vec{\beta}| = 2, \quad |\vec{\gamma}| = 1. \]
    Να αποδείξετε ότι:
    \begin{rlist}
        \item $\vec{\alpha} \uparrow\uparrow \vec{\gamma}$. \monades{5}
        \item $\vec{\alpha} \uparrow\downarrow \vec{\beta}$. \monades{5}
        \item $\vec{\alpha} = \vec{\gamma}$ και $\vec{\beta} = -2\vec{\alpha}$. \monades{5}
    \end{rlist}
\end{alist}
