    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1. Ο Κύκλος
\end{ylh}
Δίνονται τα σημεία $A(1,2)$, $B(2,4)$ και $\Gamma(3,1)$.
\begin{alist}
\item Να αποδείξετε ότι η $\sphericalangle B\hat{A}\Gamma = 90^\circ$. \monades{6}
\item Να βρείτε την εξίσωση του κύκλου $c$, ο οποίος διέρχεται από τα σημεία $A, B$ και $\Gamma$. \monades{9}
\item Αν ο κύκλος $c$ έχει εξίσωση $\left(x - \frac{5}{2}\right)^2 + \left(y - \frac{5}{2}\right)^2 = \frac{5}{2}$, τότε να βρείτε τις εξισώσεις των εφαπτομένων του, οι οποίες διέρχονται από την αρχή των αξόνων. \monades{10}
\end{alist}
