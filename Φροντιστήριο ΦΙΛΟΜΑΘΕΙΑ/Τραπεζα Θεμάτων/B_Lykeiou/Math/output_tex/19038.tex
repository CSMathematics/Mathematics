\begin{ylh}{1}
    \item 1.3. Πολλαπλασιασμός Αριθμού με Διάνυσμα
    \item 1.4. Συντεταγμένες στο Επίπεδο
\end{ylh}
Δίνεται τα διανύσματα $\vec{a} = (2,3)$, $\vec{b} = (-1,1)$ και $\vec{y} = (-5,-5)$.
\begin{alist}
\item Να υπολογίσετε τη γωνία που σχηματίζει το διάνυσμα $\vec{b}$ με τον άξονα $x'x$. \monades{9}
\item Να αποδείξετε ότι $|\vec{y}| = 5|\vec{b}|$. \monades{8}
\item Να υπολογίσετε τους πραγματικούς αριθμούς $\lambda, \mu$ ώστε το διάνυσμα $\vec{y}$ να γραφεί στην μορφή $\vec{y} = \lambda\vec{a} + \mu\vec{b}$. \monades{8}
\end{alist}
