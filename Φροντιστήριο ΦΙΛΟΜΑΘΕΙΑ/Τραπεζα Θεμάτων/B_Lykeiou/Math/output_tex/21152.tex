\OnKef{ΘΕΜΑ 1}

\begin{alist}
    \item Να χαρακτηρίσετε καθεμία από τις προτάσεις που ακολουθούν ως \textbf{Σωστή (Σ)} ή \textbf{Λανθασμένη (Λ)}, γράφοντας στην κόλλα σας, δίπλα στον αριθμό που αντιστοιχεί σε καθεμιά από αυτές το γράμμα Σ αν η πρόταση είναι Σωστή, ή το γράμμα Λ αν αυτή είναι Λάθος.
    \begin{rlist}
        \item Κάθε διάνυσμα στον χώρο είναι ίσο με τη διανυσματική ακτίνα του πέρατος μείον τη διανυσματική ακτίνα της αρχής.
        \item Η ευθεία που διέρχεται από το σημείο $A(x_0,y_0)$ και είναι παράλληλη στον άξονα $x'x$ έχει εξίσωση $x = x_0$.
        \item Η ευθεία με εξίσωση $Ax + By + \Gamma = 0$ είναι παράλληλη στο διάνυσμα $\vec{\eta} = (A,B)$.
        \item Η παραβολή με εξίσωση $y^2 = 4x$ έχει εστία το σημείο $E(1,0)$.
        \item Η εφαπτομένη του κύκλου $x^2 + y^2 = \rho^2$ στο σημείο του $A(x_1,y_1)$ έχει εξίσωση $xx_1 + yy_1 = \rho^2$.
    \end{rlist} \monades{10}
    \item Να αποδείξετε ότι η εξίσωση της ευθείας που διέρχεται από το σημείο $A(x_0,y_0)$ και έχει συντελεστή διεύθυνσης $\lambda$ είναι
    \[ y - y_0 = \lambda(x - x_0) \] \monades{15}
\end{alist}
