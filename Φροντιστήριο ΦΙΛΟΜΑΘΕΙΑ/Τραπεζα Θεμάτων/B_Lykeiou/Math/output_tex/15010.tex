\begin{ylh}{1}
    \item 1.2. Πρόσθεση και Αφαίρεση Διανυσμάτων
    \item 1.3. Πολλαπλασιασμός Αριθμού με Διάνυσμα.
\end{ylh}
Δίνονται τα μη συνευθειακά σημεία του επιπέδου $A,B,\Gamma$ και τα διανύσματα $\vec{B\Delta}$ και $\vec{\Gamma E}$ τέτοια ώστε $\vec{B\Delta} = \vec{BA} + \vec{B\Gamma}$ και $\vec{\Gamma E} = \vec{\Gamma A} + \vec{\Gamma B}$.

\begin{alist}
\item
    \begin{rlist}
    \item Να δείξετε ότι $\vec{A\Delta} = \vec{B\Gamma}$ και $\vec{AE} = \vec{\Gamma B}$. \monades{8}
    \item Να δείξετε ότι τα διανύσματα $\vec{A\Delta}$ και $\vec{AE}$ είναι αντίθετα. \monades{8}
    \end{rlist}
\item Να δικαιολογήσετε γιατί τα σημεία $A, \Delta$ και $E$ είναι συνευθειακά. \monades{9}
\end{alist}
