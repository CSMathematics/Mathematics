\Askhsh{22264}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ x^2 + y^2 + \lambda x + \lambda y + \lambda - 1 = 0 \quad (1), \quad \lambda \in \mathbb{R}. \]
\begin{alist}
    \item Να αποδείξετε ότι η εξίσωση (1), παριστάνει κύκλο για κάθε $\lambda \in \mathbb{R}$. \monades{7}
    \item Να βρείτε το κέντρο και την ακτίνα του κύκλου που ορίζεται από την εξίσωση (1), ο οποίος εφάπτεται της ευθείας $\varepsilon: x+y+2=0$. \monades{9}
    \item Για $\lambda = 1$, στον κύκλο που προκύπτει από την εξίσωση (1), να βρείτε τις εξισώσεις των εφαπτομένων του, που διέρχονται από το σημείο $M\left(-\frac{3}{2}, -\frac{1}{2}\right)$. \monades{9}
\end{alist}
