\Askhsh{14978}{4}
\begin{ylh}{1}
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Δίνονται τα σημεία $A(1,1)$, $B(3,3)$.
\begin{alist}
\item Αν $M(x,y)$ σημείο του επιπέδου, να βρείτε τις αποστάσεις $d_1, d_2$ του $M$ από τα $A$ και $B$ αντίστοιχα. \monades{6}
\item Να γράψετε τη σχέση που πρέπει να πληρούν οι $d_1, d_2$, ώστε το σημείο $M$ να ανήκει στη μεσοκάθετο του $AB$. \monades{4}
\item Να βρείτε την εξίσωση της μεσοκαθέτου του $AB$. \monades{8}
\item Να βρείτε σημείο $Σ$ τέτοιο ώστε το τρίγωνο $ΣAB$ να είναι ισόπλευρο. \monades{7}
\end{alist}
