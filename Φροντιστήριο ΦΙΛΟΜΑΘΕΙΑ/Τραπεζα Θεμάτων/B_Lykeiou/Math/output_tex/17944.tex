\Askhsh{17944}{3}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται η υπερβολή με εξίσωση της μορφής $(C): \frac{x^2}{a^2} - \frac{y^2}{\beta^2}=1$, εστιακή απόσταση $EE'=2\sqrt{7}$ και εκκεντρότητα $\varepsilon=\frac{\sqrt{7}}{2}$.
\begin{alist}
\item Να αποδείξετε ότι $a=2, \beta=\sqrt{3}$. \monades{8}
\item
    \begin{rlist}
    \item Να βρείτε τις συντεταγμένες των κορυφών $A, A'$ της υπερβολής (C).
    \item Να βρείτε τις εξισώσεις των ασύμπτωτων ευθειών της υπερβολής (C). \monades{8}
    \end{rlist}
\item Να σχεδιάσετε στο ίδιο σύστημα αξόνων την υπερβολή (C), τις ασύμπτωτές της, τις εστίες της και τις κορυφές της. \monades{9}
\end{alist}
