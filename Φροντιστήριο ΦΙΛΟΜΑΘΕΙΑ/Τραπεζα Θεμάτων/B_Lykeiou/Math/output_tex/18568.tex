\Askhsh{18568}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας.
\end{ylh}
Δίνονται τα σημεία $A(2,4)$, $B(-1,0)$ και $\Gamma(3,-2)$.
\begin{alist}
\item Να αποδείξετε ότι τα σημεία A, B, $\Gamma$ αποτελούν κορυφές τριγώνου ΑΒ$\Gamma$. \monades{04}
\item Αν η ευθεία ΑΒ τέμνει τον άξονα $y'y$ σε ένα σημείο $\Delta$ και η ευθεία Α$\Gamma$ τέμνει τον άξονα $x'x$ σε ένα σημείο $E$, τότε:
    \begin{rlist}
    \item Να βρείτε τις συντεταγμένες των σημείων $\Delta$ και $E$. \monades{10}
    \item Να αποδείξετε ότι $\vec{A\Delta} = 2\vec{\Delta B}$ και $\vec{AE} = 2\vec{E\Gamma}$. \monades{06}
    \end{rlist}
\item Να αποδείξετε ότι η ευθεία $\Delta E$ είναι παράλληλη της B$\Gamma$. \monades{05}
\end{alist}
