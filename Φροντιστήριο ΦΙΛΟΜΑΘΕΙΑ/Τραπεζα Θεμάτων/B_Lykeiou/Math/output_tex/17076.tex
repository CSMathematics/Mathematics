\begin{ylh}{1}
    \item 1.2. Πρόσθεση και Αφαίρεση Διανυσμάτων
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα σημεία $A(-3, -1)$, $B(0, 3)$ και $M(x, y)$ του καρτεσιανού επιπέδου $Oxy$.
\begin{alist}
\item Να βρείτε τις συντεταγμένες των διανυσμάτων $\overrightarrow{AM}$, $\overrightarrow{MB}$ και $\overrightarrow{AB}$. \monades{6}
\item Να βρείτε τα μέτρα των διανυσμάτων $\overrightarrow{AM}$, $\overrightarrow{MB}$ και $\overrightarrow{AB}$. \monades{6}
\item Να αποδείξετε ότι $|\overrightarrow{AM}| + |\overrightarrow{MB}| \geq 5$. \monades{6}
\item Θεωρήστε τον ισχυρισμό: «Υπάρχει ζεύγος πραγματικών αριθμών $(x,y)$ τέτοιο ώστε να ισχύει $\sqrt{(x + 3)^2+(y + 1)^2} + \sqrt{x^2+(y - 3)^2} = 4$.»\\ Είvαι αληθής ή Ψευδής ο παραπάνω ισχυρισμός; Να αιτιολογήσετε την απάντησή σας. \monades{7}
\end{alist}
