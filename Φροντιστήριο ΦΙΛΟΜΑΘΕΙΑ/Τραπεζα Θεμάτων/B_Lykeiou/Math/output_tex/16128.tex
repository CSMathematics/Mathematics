\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται υπερβολή $(C):$
\[ \frac{x^2}{16} - \frac{y^2}{9} = 1. \]
\begin{alist}
\item Να βρείτε τις συντεταγμένες των εστιών $E'$ και $E$.  \monades{10}
\item Αν το $N$ είναι τυχαίο σημείο της $(C)$, να βρείτε την τιμή της διαφοράς $|(NE') - (NE)|$.  \monades{5}
\item Να σχεδιάσετε την υπερβολή $(C)$.  \monades{10}
\end{alist}
