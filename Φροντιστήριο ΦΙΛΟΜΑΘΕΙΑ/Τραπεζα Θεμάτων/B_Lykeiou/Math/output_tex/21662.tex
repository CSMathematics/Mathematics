Δίνεται η ευθεία $\varepsilon: - x + y - 2 = 0$ και τα σημεία $A(-5,1)$ και $B(-3,5)$.
\begin{alist}
    \item Να βρείτε το συμμετρικό του σημείου $A$ ως προς το σημείο $B$. \monades{10}
    \item Να βρείτε:
    \begin{rlist}
        \item την εξίσωση της ευθείας $\varepsilon'$ που διέρχεται από το $B$ και είναι κάθετη στην $\varepsilon$. \monades{5}
        \item το σημείο τομής των ευθειών $\varepsilon$ και $\varepsilon'$. \monades{5}
        \item το συμμετρικό του σημείου $B$ ως προς την ευθεία $\varepsilon$. \monades{5}
    \end{rlist}
\end{alist}
