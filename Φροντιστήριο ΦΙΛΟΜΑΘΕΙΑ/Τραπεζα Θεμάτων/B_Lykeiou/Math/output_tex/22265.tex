\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου.
\end{ylh}
Στο καρτεσιανό επίπεδο δίνονται τα σημεία $A(1, -1)$, $B(2, 2)$ και $\Gamma(\mu-1, 3\mu-2)$, $\mu\in\mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι καθώς το $\mu$ διατρέχει το $\mathbb{R}$, το σημείο $\Gamma$ κινείται στην ευθεία
$\varepsilon: y = 3x + 1$. \monades{6}
\item Να αποδείξετε ότι καθώς το $\mu$ διατρέχει το $\mathbb{R}$, τα σημεία $A, B, \Gamma$ είναι κορυφές τριγώνου. \monades{6}
\item Να αποδείξετε ότι το εμβαδόν του τριγώνου $AB\Gamma$ είναι σταθερό. \monades{5}
\item Να βρείτε τις εξισώσεις των ευθειών που διέρχονται από το σημείο $B$ και από τις
οποίες το σημείο $A$, απέχει απόσταση ίση με $1$. \monades{8}
\end{alist}
