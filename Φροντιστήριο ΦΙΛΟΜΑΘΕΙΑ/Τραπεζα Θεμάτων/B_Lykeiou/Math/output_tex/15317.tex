\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
\end{ylh}
Δίνονται τα διανύσματα $\vec{v}=(3,0)$ και $\vec{w}=(-3,4)$.
\begin{alist}
\item Να δείξετε ότι τα διανύσματα δεν είναι παράλληλα. \monades{12}
\item
    \begin{rlist}
    \item Στο παρακάτω σύστημα συντεταγμένων να σχεδιάσετε τα διανύσματα $\vec{v}$ και $\vec{w}$. \monades{10}
    \begin{center}
    \begin{tikzpicture}
    \begin{axis}[
        axis lines = middle,
        xlabel = {}, % No explicit labels in the image for x/y axis
        ylabel = {}, % No explicit labels in the image for x/y axis
        xmin = -4.5, xmax = 4.5,
        ymin = -4.5, ymax = 4.5,
        xtick = {-3,-2,-1,0,1,2,3,4},
        ytick = {-4,-3,-2,-1,0,1,2,3,4},
        grid = both,
        grid style = {gray!30},
        width = 10cm, height = 10cm, % Square plot area
        tick label style = {font=\scriptsize},
    ]
        % Σχεδίαση του διανύσματος v = (3,0)
        \addplot[->, very thick, maincolor] coordinates {(0,0) (3,0)};
        \node[maincolor, above right] at (axis cs:3,0) {$\vec{v}$};

        % Σχεδίαση του διανύσματος w = (-3,4)
        \addplot[->, very thick, secondarycolor] coordinates {(0,0) (-3,4)};
        \node[secondarycolor, above left] at (axis cs:-3,4) {$\vec{w}$};
    \end{axis}
    \end{tikzpicture}
    \end{center}
    \item Να προσδιορίσετε το είδος της γωνίας $\theta$ που σχηματίζουν τα διανύσματα. \monades{3}
    \end{rlist}
\end{alist}
