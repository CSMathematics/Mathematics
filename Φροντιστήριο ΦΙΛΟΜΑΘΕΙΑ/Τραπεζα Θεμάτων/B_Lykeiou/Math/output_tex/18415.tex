\Askhsh{18415}{4}
\begin{ylh}{1}
    \item 2.3. Εμβαδόν Τριγώνου
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση $(x-3\lambda)^2+(y+2\lambda)^2=1$ (1) όπου $\lambda \in \mathbb{R}$ και η ευθεία $\varepsilon: 2x+3y=0$.
\begin{alist}
\item Να αποδείξετε ότι για κάθε $\lambda \in \mathbb{R}$ τα κέντρα των κύκλων που προκύπτουν από την (1) ανήκουν στην ευθεία $\varepsilon$. \monades{6}
\item Να βρείτε τις εξισώσεις των ευθειών $\varepsilon_1, \varepsilon_2$ που απέχουν μεταξύ τους 2 μονάδες και έχουν μεσοπαράλληλη την ευθεία $\varepsilon$. \monades{7}
\item Να αποδείξετε ότι όλοι οι κύκλοι που προκύπτουν από την (1) εφάπτονται σε δύο σταθερές ευθείες. \monades{6}
\item Να βρείτε το εμβαδόν ενός τετραγώνου του οποίου δύο απέναντι πλευρές ανήκουν στις ευθείες $\varepsilon_1, \varepsilon_2$ αντίστοιχα. \monades{6}
\end{alist}
