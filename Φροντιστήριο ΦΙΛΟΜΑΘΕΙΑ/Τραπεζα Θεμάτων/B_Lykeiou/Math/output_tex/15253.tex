\Askhsh{15253}{4}
\begin{ylh}{1}
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας.
\end{ylh}
Δίνεται η εξίσωση
\[ (\mu^2-1)x+(3\mu^2-2\mu-1)y-5\mu^2+4\mu+1=0 \quad (1), \]
όπου $\mu \in \mathbb{R}$.
\begin{alist}
\item Να βρείτε για ποιες τιμές του $\mu$ η (1) παριστάνει ευθεία $\varepsilon$. \monades{5}
\item Να βρείτε για ποιες τιμές του $\mu$ οι ευθείες $\varepsilon$ :
    \begin{rlist}
    \item είναι παράλληλες στον $xx'$. \monades{4}
    \item είναι παράλληλες στον $yy'$. \monades{4}
    \item διέρχονται από το $(0,0)$. \monades{4}
    \end{rlist}
\item Να δείξετε ότι όλες οι ευθείες $\varepsilon$ που προκύπτουν από την (1) διέρχονται από σταθερό σημείο. \monades{8}
\end{alist}
