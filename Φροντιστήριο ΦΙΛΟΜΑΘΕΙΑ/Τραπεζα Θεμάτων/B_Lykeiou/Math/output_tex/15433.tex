\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Απόσταση Σημείου από Ευθεία
    \item 2.3. Εμβαδόν Τριγώνου
\end{ylh}
Δύο οικισμοί Α και Β βρίσκονται στις θέσεις που ορίζουν τα σημεία $A(-1, -2)$ και $B(3,1)$. Εξωτερικά των οικισμών υπάρχει ευθύγραμμος δρόμος με εξίσωση $\delta: x + y - 1 = 0$.
\begin{alist}
\item Να βρείτε σε ποια θέση του δρόμου $\delta$:
    \begin{rlist}
    \item Ο οικισμός Α έχει τη μικρότερη απόσταση από τον δρόμο. \monades{8}
    \item Υπάρχει το Κέντρο Υγείας της περιοχής, αν είναι γνωστό ότι ισαπέχει από τους δύο οικισμούς. \monades{7}
    \end{rlist}
\item Να βρείτε τη θέση $\Gamma$ ενός αυτοκινήτου πάνω στο δρόμο, αν είναι γνωστό, ότι το εμβαδόν του τριγώνου που σχηματίζουν τα τρία σημεία Α, Β και $\Gamma$ είναι ίσο με 8. \monades{10}
\end{alist}
