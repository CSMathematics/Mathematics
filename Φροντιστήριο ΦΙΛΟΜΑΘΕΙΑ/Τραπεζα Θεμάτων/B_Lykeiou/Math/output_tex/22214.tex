    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων
    \item 3.1 Ο Κύκλος.
\end{ylh}
Σε καρτεσιανό επίπεδο $Oxy$ θεωρούμε τα σημεία $M(x,y)$, $A(-1,0)$, $B(1,0)$ για τα οποία ισχύει
\[ \vec{AM}^2 + \vec{BM}^2 = 9|\vec{AB}|. \]
\begin{alist}
\item Να αποδείξετε ότι ο γεωμετρικός τόπος των σημείων $M$ είναι ο κύκλος με εξίσωση
\[ x^2 + y^2 = 8. \] \monades{10}
\item Έστω $\Gamma$ και $\Delta$ δύο σημεία του κύκλου τέτοια ώστε $\Gamma\Delta^2 = 32$.
    \begin{rlist}
    \item Να δείξετε ότι τα σημεία $\Gamma$ και $\Delta$ και η αρχή των αξόνων είναι συνευθειακά σημεία. \monades{08}
    \item Αν το σημείο $M$ κινείται στον κύκλο, να υπολογίσετε το $\vec{M\Gamma} \cdot \vec{M\Delta}$. \monades{07}
    \end{rlist}
\end{alist}
