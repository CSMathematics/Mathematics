\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ x^2 + y^2 - x - y - \frac{7}{2} = 0 \quad (1) \]
\begin{alist}
\item Να αποδείξετε ότι η εξίσωση (1) παριστάνει κύκλο με κέντρο $K\left(\frac{1}{2}, \frac{1}{2}\right)$ και ακτίνα $R = 2$. \monades{9}
\item Να αποδείξετε ότι το σημείο $A\left(\frac{1}{2}, -\frac{3}{2}\right)$ είναι σημείο του κύκλου $(K,R)$. \monades{8}
\item Να βρείτε την εξίσωση της εφαπτομένης του κύκλου $(K,R)$ στο A. \monades{8}
\end{alist}
