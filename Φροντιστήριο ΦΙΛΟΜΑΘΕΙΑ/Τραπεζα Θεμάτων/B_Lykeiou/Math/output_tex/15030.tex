\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος $C: (x-2)^2+(y+3)^2=5$ και η ευθεία $\varepsilon: 2x+y+5=0$.
\begin{alist}
\item Να βρείτε το κέντρο και την ακτίνα του κύκλου $C$. \monades{6}
\item Να δείξετε ότι ο κύκλος $C$ και η ευθεία $(\varepsilon)$ δεν έχουν κοινά σημεία. \monades{6}
\item Να δείξετε ότι υπάρχουν δύο ευθείες $(\eta_1),(\eta_2)$ που είναι παράλληλες στην ευθεία $(\varepsilon)$ και εφάπτονται του κύκλου $C$ και να βρείτε τις εξισώσεις τους. \monades{7}
\item Να βρείτε τη μεσοπαράλληλη των ευθειών $(\eta_1),(\eta_2)$. \monades{6}
\end{alist}
