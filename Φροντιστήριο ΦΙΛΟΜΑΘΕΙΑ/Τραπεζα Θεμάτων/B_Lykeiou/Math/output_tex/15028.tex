\Askhsh{15028}{2}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Έστω κύκλος $C$ με κέντρο $K(1,2)$ και ακτίνα $\rho=2$ και ευθεία $(\varepsilon)$ με εξίσωση $3x+4y-1=0$.
\begin{alist}
\item Να γράψετε την εξίσωση του κύκλου $C$. \monades{8}
\item Να δείξετε ότι η απόσταση του κέντρου $K(1,2)$ από την ευθεία $(\varepsilon)$ είναι ίση με 2. \monades{9}
\item Να δείξετε ότι η ευθεία $(\varepsilon)$ εφάπτεται στον κύκλο $C$. \monades{8}
\end{alist}
