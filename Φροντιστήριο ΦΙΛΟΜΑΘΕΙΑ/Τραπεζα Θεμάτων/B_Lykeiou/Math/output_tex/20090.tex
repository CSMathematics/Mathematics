    \item 3.2 Η Παραβολή.
\end{ylh}
Δίνεται η παραβολή $C: y^2 = 4x$ και $M(x_0, y_0)$, $y_0 > 0$, ένα σημείο της.
\begin{alist}
    \item Αν $A$ είναι η προβολή του $M$ στη διευθετούσα της παραβολής,
    \begin{rlist}
        \item Να εκφράσετε τις συντεταγμένες των σημείων $M$ και $A$ συναρτήσει της τεταγμένης $y_0$ του σημείου $M$. \monades{05}
        \item Αν $E$ είναι η εστία της παραβολής, να βρείτε το σημείο $M$ για το οποίο $(MAE) = \frac{5}{8}$ τ.μ. \monades{12}
    \end{rlist}
    \item Αν $M\left(\frac{1}{4}, 1\right)$ και $\varepsilon$ η εφαπτομένη της παραβολής στο σημείο $M$, να αποδείξετε ότι το τετράπλευρο $AMEM'$ είναι ρόμβος, όπου $E$ είναι η εστία της παραβολής και $M'$ το σημείο που η ευθεία $\varepsilon$ τέμνει τον άξονα $x'x$. \monades{08}
\end{alist}
