\Askhsh{20862}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.2. Η Παραβολή.
\end{ylh}
Δίνονται τα σημεία $M(-2,2)$, $E\PARENS{0,-\frac{1}{2}}$ και η ευθεία $(\zeta)$ με εξίσωση $y=\frac{1}{2}$.
\begin{alist}
\item Να βρείτε την εξίσωση της ευθείας $(\varepsilon_1)$ που διέρχεται από το σημείο $M$ και σχηματίζει γωνία $45^\circ$ με τον άξονα $x'x$.  \monades{05}
\item Να βρείτε την εξίσωση, που εκφράζει το σύνολο των σημείων του επιπέδου, που απέχουν ίση απόσταση από το σημείο $E$ και την ευθεία $(\zeta)$. \monades{06}
\item \begin{rlist}
\item Να βρείτε την εξίσωση της εφαπτομένης $(\eta)$ της καμπύλης $C: x^2+2y=0$, που είναι παράλληλη στην ευθεία $(\varepsilon_1)$, με εξίσωση $y=x+4$. \monades{07}
\item Να σχεδιάσετε την γραφική παράσταση της καμπύλης $C$ και των ευθειών $(\varepsilon_1)$ και $(\eta)$. Με τη βοήθεια του σχήματος (ή με οποιονδήποτε άλλο τρόπο) να αποδείξετε ότι η ελάχιστη απόσταση των σημείων της $C$ από την ευθεία $(\varepsilon_1)$ είναι $\frac{7\sqrt{2}}{4}$. \monades{07}
\end{rlist}
\end{alist}
