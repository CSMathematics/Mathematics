\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα σημεία $A(0, 2)$, $B(3, 0)$, $\Gamma(6,2)$ και $\Delta(3,4)$.
\begin{alist}
\item Να βρείτε τις συντεταγμένες των διανυσμάτων $\overrightarrow{A\Delta}$, $\overrightarrow{B\Gamma}$, $\overrightarrow{BA}$ και να επιβεβαιώσετε ότι: $\overrightarrow{A\Delta} = \overrightarrow{B\Gamma}$. \monades{12}
\item Να δείξετε ότι $|\overrightarrow{BA}| = |\overrightarrow{B\Gamma}|$. Ποιο είναι το σχήμα του τετραπλεύρου $AB\Gamma\Delta$; \monades{13}
\end{alist}
