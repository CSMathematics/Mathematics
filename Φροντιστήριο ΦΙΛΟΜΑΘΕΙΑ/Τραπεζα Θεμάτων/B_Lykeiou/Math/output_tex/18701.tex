\Askhsh{18701}{2}
\begin{ylh}{1}
    \item 3.2 Η Παραβολή.
\end{ylh}
Δίνεται η παραβολή με εξίσωση $y = \frac{1}{2}x^2$ (1).
\begin{alist}
\item Να βρείτε την εστία και τη διευθετούσα της παραβολής. \monades{6}
\item Να βρείτε την εξίσωση της εφαπτομένης της παραβολής στο σημείο $A(2,2)$. \monades{10}
\item Να σχεδιάσετε στο ίδιο ορθοκανονικό σύστημα συντεταγμένων την παραβολή (1), την εστία, τη διευθετούσα και την εφαπτομένη της παραβολής. \monades{9}
\end{alist}
