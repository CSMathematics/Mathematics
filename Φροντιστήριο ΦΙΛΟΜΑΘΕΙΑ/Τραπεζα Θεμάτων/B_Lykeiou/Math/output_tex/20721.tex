\Askhsh{20721}{2}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται η υπερβολή $C$ με εξίσωση
\[ \frac{x^2}{9} - \frac{y^2}{16} = 1. \]
\begin{alist}
    \item Να βρείτε τις εστίες της $C$. \monades{8}
    \item Να βρείτε τις εξισώσεις των ασυμπτώτων της $C$. \monades{8}
    \item Να σχεδιάσετε την υπερβολή $C$ και τις ασύμπτωτες της στο ίδιο σύστημα αξόνων. \monades{9}
\end{alist}
