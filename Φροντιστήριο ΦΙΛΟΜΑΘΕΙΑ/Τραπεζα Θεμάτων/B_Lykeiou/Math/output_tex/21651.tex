\Askhsh{21651}{2}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Η υπερβολή $C$ έχει εστίες τα σημεία $E(5,0), E'(-5,0)$ και διέρχεται από το σημείο $A(4,0)$.
\begin{alist}
\item Να αποδείξετε ότι έχει εκκεντρότητα $ \frac{5}{4} $. \monades{10}
\item Να βρείτε την εξίσωση της $C$. \monades{8}
\item Να βρείτε την εξίσωση της εφαπτομένης της $C$ στο σημείο $M(5, \frac{9}{4})$. \monades{7}
\end{alist}
