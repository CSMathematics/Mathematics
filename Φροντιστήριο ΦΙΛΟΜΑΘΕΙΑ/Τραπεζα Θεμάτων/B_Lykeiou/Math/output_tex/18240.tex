    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Απόσταση Σημείου από Ευθεία
    \item 2.3. Γραφική Παράσταση Ευθείας
\end{ylh}
Δίνεται το σημείο $A(1,2)$ και η ευθεία $(\varepsilon): y = x+3$.
\begin{alist}
\item Να βρείτε την απόσταση του σημείου $A$ από την ευθεία $(\varepsilon)$. \monades{7}
\item Να βρείτε την εξίσωση της ευθείας $(\eta)$ που διέρχεται από το $A$ και είναι παράλληλη στην $(\varepsilon)$. \monades{8}
\item Να σχεδιάσετε στο ίδιο ορθοκανονικό σύστημα συντεταγμένων τις ευθείες $(\eta)$, $(\varepsilon)$. \monades{10}
\end{alist}
