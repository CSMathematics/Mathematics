\Askhsh{21218}{2}
\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνονται οι υπερβολές $(C_1):x^2-y^2=1,(C_2):y^2-x^2=1.$
\begin{alist}
\item Να αποδείξετε ότι οι εστίες της $C_1$ είναι οι $E_1(\sqrt{2},0), E'_1(-\sqrt{2},0).$ \monades{12}
\item Αν $E_2,E'_2$ οι εστίες της $C_2$ τότε να αποδείξετε ότι το $E_1E_2E'_1E'_2$ είναι τετράγωνο. \monades{13}
\end{alist}
