\begin{wrapfigure}[10]{r}{5cm}

    \begin{tikzpicture}[scale=0.6]
        % Define points for an equilateral triangle with side 10
        \tkzDefPoint(0,0){M}
        \tkzDefPoint(-5,0){B}
        \tkzDefPoint(5,0){C}
        % Altitude for side 10 equilateral triangle is 5*sqrt(3)
        \tkzDefPoint(0,{5*sqrt(3)}){A}

        % Draw the triangle ABC
        \tkzDrawPolygon(A,B,C)

        % Draw the median AM as a dashed line
        \tkzDrawSegment[dashed](A,M)

        % Draw points
        \tkzDrawPoints(A,B,C,M)

        % Label points
        \tkzLabelPoint[above](A){A}
        \tkzLabelPoint[below left](B){B}
        \tkzLabelPoint[below right](C){$\Gamma$}
        \tkzLabelPoint[below](M){M}
    \end{tikzpicture}
\end{wrapfigure}
Δίνεται ισόπλευρο τρίγωνο ΑΒΓ πλευράς 10 και το μέσο M της πλευράς ΒΓ.
\begin{alist}
\item Να βρεθούν τα μέτρα των γωνιών:
    \begin{rlist}
    \item $(\overrightarrow{AB}, \overrightarrow{A\Gamma})$
    \item $(\overrightarrow{AM}, \overrightarrow{B\Gamma})$
    \item $(\overrightarrow{AM}, \overrightarrow{\Gamma A})$
    \item $(\overrightarrow{BM}, \overrightarrow{\Gamma M})$
    \item $(\overrightarrow{\Gamma M}, \overrightarrow{\Gamma B})$
    \end{rlist} \monades{10}
\item Να υπολογιστούν τα εσωτερικά γινόμενα:
    \begin{rlist}
    \item $\overrightarrow{AM} \cdot \overrightarrow{B\Gamma}$
    \item $\overrightarrow{AM} \cdot \overrightarrow{\Gamma A}$
    \item $\overrightarrow{\Gamma M} \cdot \overrightarrow{\Gamma B}$
    \end{rlist} \monades{15}
\end{alist}
