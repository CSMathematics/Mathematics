\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 3.2 Η Παραβολή.
\end{ylh}
Δίνεται το σημείο $E(2,0)$, η ευθεία $\delta_1: x = -2$ και τυχαίο σημείο $M(x,y)$ του επιπέδου.
\begin{alist}
\item
    \begin{rlist}
    \item Να βρείτε την απόσταση $(ME)$ του σημείου $M(x,y)$ από το $E(2,0)$ ως συνάρτηση των $x, y$. \monades{8}
    \item Να βρείτε την απόσταση $d(M,\delta)$ του σημείου $M$ από την ευθεία $\delta$ ως συνάρτηση των $x, y$. \monades{8}
    \end{rlist}
\item Αν ισχύει $(ME) = d(M,\delta)$ να δείξετε ότι το σημείο $M$ ανήκει στην παραβολή $y^2 = 8x$. \monades{9}
\end{alist}
