\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου.
\end{ylh}
Δίνεται το σημείο $M(-2, 2)$.
\begin{alist}
\item Να βρείτε τις εξισώσεις των ευθειών ($\varepsilon$) που διέρχονται από το σημείο $M$. \monades{06}
\item
    \begin{rlist}
    \item Να βρείτε ποιες από τις παραπάνω εξισώσεις ευθειών σχηματίζουν τρίγωνο με τον αρνητικό ημιάξονα $Ox'$ και τον θετικό ημιάξονα $Oy$. \monades{04}
    \item Να βρείτε την εξίσωση της ευθείας ($\varepsilon_1$), η οποία διέρχεται από το σημείο $M$ και σχηματίζει με τον αρνητικό ημιάξονα $Ox'$ και τον θετικό ημιάξονα $Oy$ τρίγωνο, με εμβαδόν $E = 8$. \monades{10}
    \end{rlist}
\item Αν ($\varepsilon_1$): $y = x + 4$, να βρείτε το μήκος του ύψους του ορθογωνίου τριγώνου, που σχηματίζει η ($\varepsilon_1$) με τους άξονες, το οποίο φέρεται από την κορυφή $O$. \monades{05}
\end{alist}
