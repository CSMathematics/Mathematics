\Askhsh{22064}{4}
\begin{ylh}{1}
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
\end{ylh}
\begin{alist}
    \item[α)] Αν $\vec{AB}$, $\vec{AΓ}$, $\vec{AΔ}$ είναι τρία διανύσματα, τότε οι ποσότητες $\vec{AB} \cdot \vec{AΓ}$ και $\vec{AB} \cdot \vec{AΔ}$ :
    \begin{rlist}
        \item i) είναι αριθμοί ή διανύσματα;
        \item ii) μπορούν να συγκριθούν;
    \end{rlist} \monades{5}

    \item[β)] Για τα διανύσματα του παρακάτω σχήματος να επιλέξετε την σωστή απάντηση:
    \begin{wrapfigure}[9]{r}{5cm}
    \centering
    \begin{tikzpicture}
        \tkzDefPoint(0,0){A}
        \tkzDefPoint(4,0){B}
        \tkzDefPoint(2,0){P} % Projection point
        \tkzDefPoint(2,2){G} % Gamma
        \tkzDefPoint(2,3){D} % Delta

        \tkzDrawSegment[->,thick](A,B)
        \tkzDrawSegment[->,thick](A,G)
        \tkzDrawSegment[->,thick](A,D)

        \tkzDrawSegment[dashed](G,P)
        \tkzDrawSegment[dashed](D,P)

        \tkzMarkRightAngle[size=0.3cm,fill=green!20](A,P,G)

        \tkzDrawPoints(A,B,G,D)
        \tkzLabelPoint[left](A){A}
        \tkzLabelPoint[right](B){B}
        \tkzLabelPoint[above right](G){$\Gamma$}
        \tkzLabelPoint[above right](D){$\Delta$}
    \end{tikzpicture}
    \end{wrapfigure}
    \begin{rlist}
        \item i) $\vec{AB} \cdot \vec{AΓ} > \vec{AB} \cdot \vec{AΔ}$
        \item ii) $\vec{AB} \cdot \vec{AΓ} < \vec{AB} \cdot \vec{AΔ}$
        \item iii) $\vec{AB} \cdot \vec{AΓ} = \vec{AB} \cdot \vec{AΔ}$
    \end{rlist} \monades{10}

    \item[γ)] Για τα διανύσματα του παρακάτω σχήματος (η διακεκομμένη γραμμή είναι τμήμα κύκλου με κέντρο A) να επιλέξετε την σωστή απάντηση:
    \begin{wrapfigure}[9]{r}{5cm}
    \centering
    \begin{tikzpicture}
        \tkzDefPoint(0,0){A}
        \tkzDefPoint(4,0){B} % Point on the line AB

        \pgfmathsetmacro{\Radius}{3}
        \tkzDefPointWith[polar](A,\Radius:25){G} % Gamma
        \tkzDefPointWith[polar](A,\Radius:40){D} % Delta (25 + 15 = 40)

        \tkzDrawSegment[->,thick](A,B)
        \tkzDrawSegment[->,thick](A,G)
        \tkzDrawSegment[->,thick](A,D)

        \tkzDrawArc[dashed](A,D)(G)

        \tkzMarkAngle[fill=green!20,size=0.5cm](B,A,G)
        \tkzLabelAngle[pos=0.7](B,A,G){$25^\circ$}
        \tkzMarkAngle[fill=green!20,size=0.7cm](G,A,D)
        \tkzLabelAngle[pos=0.7](G,A,D){$15^\circ$}

        \tkzDrawPoints(A,B,G,D)
        \tkzLabelPoint[left](A){A}
        \tkzLabelPoint[right](B){B}
        \tkzLabelPoint[above right](G){$\Gamma$}
        \tkzLabelPoint[above](D){$\Delta$}
    \end{tikzpicture}
    \end{wrapfigure}
    \begin{rlist}
        \item i) $\vec{AB} \cdot \vec{AΓ} > \vec{AB} \cdot \vec{AΔ}$
        \item ii) $\vec{AB} \cdot \vec{AΓ} < \vec{AB} \cdot \vec{AΔ}$
        \item iii) $\vec{AB} \cdot \vec{AΓ} = \vec{AB} \cdot \vec{AΔ}$
    \end{rlist} \monades{10}
\end{alist}
Σε όλα τα ερωτήματα, να αιτιολογήσετε τις απαντήσεις σας.
