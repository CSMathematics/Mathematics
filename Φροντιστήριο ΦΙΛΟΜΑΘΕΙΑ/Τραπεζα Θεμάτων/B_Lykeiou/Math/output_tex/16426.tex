\begin{ylh}{1}
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
\end{ylh}
Δίνονται τα διανύσματα $\vec{\alpha} = (2, -1)$ και $\vec{\beta} = (-3, 2)$.
\begin{alist}
\item Να υπολογίσετε το γινόμενο $\vec{\alpha} \cdot (2\vec{\alpha} - \vec{\beta})$.  \monades{10}
\item Να βρείτε το διάνυσμα $\vec{\gamma} = (x, y)$ όταν $\vec{\gamma} \perp \vec{\alpha}$ και $|\vec{\gamma}| = \sqrt{5}$.  \monades{15}
\end{alist}
