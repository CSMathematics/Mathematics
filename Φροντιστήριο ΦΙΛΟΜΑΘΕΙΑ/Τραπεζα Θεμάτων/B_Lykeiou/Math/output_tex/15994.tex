\Askhsh{15994}{2}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση $x^2 + y^2 - 4x + 3 = 0$ (1).
\begin{alist}
\item Να αποδείξετε ότι η εξίσωση (1) παριστάνει κύκλο του οποίου να βρείτε το κέντρο και την ακτίνα. \monades{13}
\item Να σχεδιάσετε τον κύκλο $(c)$ και να βρείτε, χρησιμοποιώντας το σχήμα ή με οποιονδήποτε άλλο τρόπο, τα κοινά του σημεία με τους άξονες. \monades{12}
\end{alist}
