\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Θεωρούμε τα σημεία $A(1, 2)$, $B(-3, 4)$, $\Gamma(2, 5)$.
\begin{alist}
\item Να βρείτε σημείο $\Delta$ ώστε $\vec{\Delta\Gamma} = \vec{AB}$. \monades{10}
\item Να αιτιολογήσετε γιατί το τετράπλευρο $AB\Gamma\Delta$ είναι παραλληλόγραμμο. \monades{8}
\item Να βρείτε το κέντρο $O$ του παραλληλογράμμου. \monades{7}
\end{alist}
