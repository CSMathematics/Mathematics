    \item 3.3 Η Έλλειψη.
\end{ylh}
Δίνεται η έλλειψη $(c)$ με εξίσωση
\[ \frac{x^2}{4} + \frac{y^2}{16} = 1. \]
\begin{alist}
\item Να δικαιολογήσετε ότι $α = 4, β = 2$ και $\gamma = 2\sqrt{3}$. \monades{08}
\item Να βρείτε τα μήκη των αξόνων και τις εστίες της έλλειψης $(c)$. \monades{08}
\item Να σχεδιάσετε την έλλειψη $(c)$ και τον κύκλο $x^2 + y^2 = 16$ στο ίδιο σύστημα αξόνων. \monades{09}
\end{alist}
