\begin{ylh}{1}
    \item 3.3 Η Έλλειψη.
\end{ylh}
Δίνεται η έλλειψη (C) με εξίσωση
\[ \frac{x^2}{225} + \frac{y^2}{81} = 1 \quad (1) \]
\begin{alist}
\item Να βρείτε τις συντεταγμένες των εστιών $E$ και $E'$. \monades{10}
\item Να αποδείξετε ότι το σημείο $B(0,9)$ είναι σημείο της έλλειψης. \monades{5}
\item Να βρείτε την εξίσωση της εφαπτομένης έλλειψης στο σημείο της $B(0,9)$. \monades{10}
\end{alist}
