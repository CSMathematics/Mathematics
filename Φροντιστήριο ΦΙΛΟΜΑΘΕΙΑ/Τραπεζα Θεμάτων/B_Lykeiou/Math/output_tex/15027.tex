    \centering
    \begin{tikzpicture}[scale=0.8]
        % Ορισμός σημείων A και B
        \tkzDefPoint(0,0){A}
        \tkzDefPoint(4,2){B}
        % Ορισμός του μέσου M
        \tkzDefMidPoint(A,B) \tkzGetPoint{M}
        
        % Σχεδίαση του τμήματος AB
        \tkzDrawSegment[very thick,maincolor](A,B)
        
        % Σχεδίαση της μεσοκαθέτου
        \tkzDefLine[orthogonal through M](A,B) \tkzGetPoint{P_ortho}
        \tkzDrawLine[add=1.2 and 1.2, secondarycolor, very thick](M,P_ortho)
        
        % Σημεία και ετικέτες
        \tkzDrawPoint[fill=maincolor](A) \tkzLabelPoint[left,maincolor](A){A}
        \tkzDrawPoint[fill=maincolor](B) \tkzLabelPoint[right,maincolor](B){B}
        \tkzDrawPoint[fill=black](M) \tkzLabelPoint[below](M){M}
        
        % Σήμανση ορθής γωνίας
        \tkzMarkRightAngle[fill=thirdcolor!20,size=0.3](A,M,P_ortho)
    \end{tikzpicture}
\end{wrapfigure}
