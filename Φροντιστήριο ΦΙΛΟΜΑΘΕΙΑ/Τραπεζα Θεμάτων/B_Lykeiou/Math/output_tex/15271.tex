\begin{ylh}{2}
    \item 2.1. Εξίσωση Ευθείας.
\end{ylh}
Δίνονται τα σημεία $A(-3, 2)$, $B(1, 6)$ και $\Gamma(-13, -7)$.
\begin{alist}
\item Να βρείτε το συντελεστή διεύθυνσης της ευθείας που διέρχεται από τα $A, B$. \monades{8}
\item Να αποδείξετε ότι η ευθεία που διέρχεται από τα $A, B$ έχει εξίσωση $y=x+5$. \monades{7}
\item Να αιτιολογήσετε γιατί το σημείο $\Gamma$ δεν είναι πάνω στην $AB$. \monades{10}
\end{alist}
