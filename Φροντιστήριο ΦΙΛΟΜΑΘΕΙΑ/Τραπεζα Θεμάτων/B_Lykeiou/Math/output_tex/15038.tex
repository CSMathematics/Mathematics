\begin{ylh}{1}
    \item 1.5. Εσωτερικό Γινόμενο Διανυσμάτων.
\end{ylh}
Θεωρούμε διανύσματα $\vec{α}, \vec{β}$ τέτοια ώστε $|\vec{α}| = 3, |\vec{β}| = 4$ και $(\widehat{\vec{α}, \vec{β}}) = \frac{\pi}{3}$.
\begin{alist}
    \item Να βρείτε το εσωτερικό γινόμενο των διανυσμάτων $\vec{α}, \vec{β}$. \monades{9}
    \item Να βρείτε τα $\vec{α}^2$ και $\vec{β}^2$. \monades{6}
    \item Να αποδείξετε ότι $(3\vec{α} - \vec{β}) \cdot (\vec{α} - 3\vec{β}) = 15$. \monades{10}
\end{alist}
