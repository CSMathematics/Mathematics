\Askhsh{15275}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας.
\end{ylh}
Σε ορθοκανονικό σύστημα συντεταγμένων θεωρούμε το σημείο $M(2, 1)$.
\begin{alist}
    \item Μια ευθεία $(\varepsilon)$ με συντελεστή διεύθυνσης $\lambda$ διέρχεται από το $M$. Να βρείτε:
    \begin{rlist}
        \item Την εξίσωση της. \monades{2}
        \item Για ποιες τιμές του $\lambda$ η ευθεία σχηματίζει τρίγωνο με τους άξονες. \monades{5}
    \end{rlist}
    \item Έστω ότι η ευθεία $(\varepsilon)$ τέμνει τους άξονες $x'x$ και $y'y$ στα σημεία $A$, $B$ αντίστοιχα.
    \begin{rlist}
        \item Να βρείτε, με τη βοήθεια του $\lambda$, τα μήκη των τμημάτων $OA$, $OB$. \monades{6}
        \item Να βρείτε για ποιες τιμές του $\lambda$ η ευθεία σχηματίζει με τους άξονες ισοσκελές τρίγωνο. \monades{6}
        \item Να υπολογίσετε, σε κάθε περίπτωση, το εμβαδόν του ισοσκελούς τριγώνου που σχηματίζεται. \monades{6}
    \end{rlist}
\end{alist}
