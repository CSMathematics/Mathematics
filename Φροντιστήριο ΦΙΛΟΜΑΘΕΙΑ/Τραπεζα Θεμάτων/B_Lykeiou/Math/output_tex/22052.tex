    \item 1.4. Συντεταγμένες στο Επίπεδο.
\end{ylh}
Δίνονται τα διανύσματα $\vec{a} = (-2,5)$ και $\vec{b} = (1,-3)$.
\begin{alist}
\item Να αποδείξετε ότι τα διανύσματα $\vec{a}, \vec{b}$ δεν είναι παράλληλα. \monades{12}
\item Να εκφράσετε το διάνυσμα $\vec{v} = (8,-21)$ ως γραμμικό συνδυασμό των διανυσμάτων $\vec{a}$ και $\vec{b}$. \monades{13}
\end{alist}
