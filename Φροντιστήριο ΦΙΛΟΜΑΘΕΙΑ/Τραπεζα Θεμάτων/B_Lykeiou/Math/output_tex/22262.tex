\Askhsh{22262}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου.
\end{ylh}
Δίνεται τρίγωνο ΑΒΓ με κορυφές τα σημεία $A(-2, 1)$, $B(1, 5)$ και $\Gamma(5, -1)$.
\begin{alist}
\item Να υπολογίσετε το εμβαδόν του τριγώνου ΑΒΓ. \monades{5}
\item Να βρείτε την εξίσωση της ευθείας ΒΓ. \monades{5}
\item Να βρείτε την εξίσωση του ύψους του τριγώνου από την κορυφή Α. Στη συνέχεια να βρείτε το σημείο $\Delta$ της ευθείας ΒΓ, από το οποίο, το Α απέχει την ελάχιστη απόσταση. \monades{8}
\item Να βρείτε το σύνολο των σημείων $M$ του επιπέδου για τα οποία ισχύει: $(MAB) = \frac{1}{2} (AB\Gamma)$. \monades{7}
\end{alist}
