\Askhsh{18467}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ (x-1)^2+(y+2)^2=2(x+3) \quad : (1). \]
\begin{alist}
\item Να δείξετε ότι η εξίσωση (1) παριστάνει κύκλο με κέντρο $K(2,-2)$ και ακτίνα $\rho=3$. \monades{06}
\item Να δείξετε ότι η αρχή $O$ των αξόνων είναι εσωτερικό σημείο του κύκλου. \monades{04}
\item Να βρείτε την εξίσωση της ευθείας $(\varepsilon)$ η οποία τέμνει τον κύκλο σε δύο σημεία $A$ και $B$ ώστε η αρχή των αξόνων να είναι το μέσο της χορδής $AB$. \monades{08}
\item Αν η ευθεία $(\varepsilon)$ του προηγούμενου ερωτήματος έχει εξίσωση $y=x$ τότε να βρείτε το εμβαδό του τριγώνου $KAB$. \monades{07}
\end{alist}
