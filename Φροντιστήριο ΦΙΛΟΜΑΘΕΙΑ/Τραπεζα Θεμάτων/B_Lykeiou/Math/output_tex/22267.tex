\Askhsh{22267}{2}
\begin{ylh}{1}
    \item 3.2 Η Παραβολή.
\end{ylh}
Δίνεται η εξίσωση $y^2 = 4x$ (1).
\begin{alist}
\item Να γράψετε στο τετράδιό σας συμπληρωμένη την παρακάτω πρόταση :
«Τα σημεία του επιπέδου που επαληθεύουν την εξίσωση (1) βρίσκονται σε μια καμπύλη που ονομάζεται \dots. Η εστία της Ε, έχει συντεταγμένες Ε(\dots, \dots) και η διευθετούσα έχει εξίσωση \dots.» \monades{09}
\item Να βρείτε την εξίσωση της ευθείας $\varepsilon$ που εφάπτεται στην παραπάνω καμπύλη στο σημείο $A(1, -2)$. \monades{08}
\item Να αποδείξετε ότι το σημείο τομής της ευθείας $\varepsilon$ με τον άξονα $x'x$ είναι σημείο της διευθετούσας της παραβολής. \monades{08}
\end{alist}
