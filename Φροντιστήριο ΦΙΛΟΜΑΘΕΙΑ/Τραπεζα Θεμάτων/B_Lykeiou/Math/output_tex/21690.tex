\begin{ylh}{1}
    \item 3.2 Η Παραβολή.
    \item 2.2 Η Ευθεία.
\end{ylh}
Δίνεται η παραβολή $C: y^2 = 3x$ και η ευθεία $\epsilon: 3x + 4y + 10 = 0$.
\begin{alist}
\item Να αποδείξετε ότι η ευθεία και η παραβολή δεν έχουν κοινά σημεία και να τις σχεδιάσετε. \monades{8}
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = {$x$},
    ylabel = {$y$},
    xmin = -5, xmax = 15,
    ymin = -7, ymax = 7,
    xtick = {-4,-2,0,2,4,6,8,10,12,14},
    ytick = {-6,-4,-2,0,2,4,6},
    grid = both,
    grid style = {gray!30},
    samples = 100,
    width = 10cm, height = 7cm,
    tick label style = {font=\scriptsize},
    every axis x label/.style = {at={(current axis.right of origin)}, anchor=north west},
    every axis y label/.style = {at={(current axis.above origin)}, anchor=south},
]
    % Parabola C: y^2 = 3x => y = sqrt(3x) and y = -sqrt(3x)
    \addplot[very thick, maincolor, domain=0:15] {sqrt(3*x)};
    \addplot[very thick, maincolor, domain=0:15] {-sqrt(3*x)};
    \node[maincolor, above right] at (axis cs:10,5) {$C: y^2 = 3x$};

    % Line epsilon: 3x + 4y + 10 = 0 => y = -3/4 x - 10/4 = -0.75x - 2.5
    \addplot[very thick, secondarycolor, domain=-5:15] {-0.75*x - 2.5};
    \node[secondarycolor, above] at (axis cs:8,-8) {$\epsilon: 3x + 4y + 10 = 0$};
\end{axis}
\end{tikzpicture}
\end{center}
\item Έστω $M(x_o, y_o)$ ένα σημείο της παραβολής. Να αποδείξετε ότι η απόσταση του $M$ από την ευθεία $\epsilon$ είναι $d(M, \epsilon) = \frac{(y_o+2)^2+6}{5}$. \monades{8}
\item Να βρείτε το σημείο της παραβολής που είναι το πιο κοντινό στην ευθεία. \monades{5}
\item Να αποδείξετε ότι η εφαπτομένη της παραβολής στο σημείο που βρήκατε στο ερώτημα γ) είναι παράλληλη στην ευθεία $\epsilon$. \monades{4}
\end{alist}
