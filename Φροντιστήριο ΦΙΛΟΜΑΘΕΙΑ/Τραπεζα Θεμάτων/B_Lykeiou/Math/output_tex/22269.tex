\begin{ylh}{1}
    \item 3.4 Η Υπερβολή.
\end{ylh}
Δίνεται η υπερβολή με εξίσωση : $\dfrac{x^2}{4} - y^2 = 1$ (1).
\begin{alist}
\item Να προσδιορίσετε δικαιολογώντας την απάντησή σας :
\begin{rlist}
\item Τις συντεταγμένες των εστιών της.
\item Την εκκεντρότητά της.
\item Τις εξισώσεις των ασύμπτωτων της υπερβολής.
\end{rlist} \monades{15}
\item Να βρείτε την εξίσωση της ευθείας $\varepsilon$ που εφάπτεται στην υπερβολή στο σημείο της, $A\left(\sqrt{5}, \dfrac{1}{2}\right).$ \monades{10}
\end{alist}
