\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση
\[ (x - 1)(x + 3) + (y + 1)(y - 3) = -4 \quad (1) \]
\begin{alist}
    \item Να αποδείξετε ότι η εξίσωση (1) παριστάνει κύκλο με κέντρο $K(-1,1)$ και ακτίνα $R = 2$. \monades{9}
    \item
    \begin{rlist}
        \item Να βρείτε τα σημεία $A$ και $B$ του κύκλου $(K,R)$ τα οποία έχουν τετμημένη ίση με $-1$. \monades{8}
        \item Να αποδείξετε ότι τα σημεία $A$ και $B$ είναι αντιδιαμετρικά. \monades{8}
    \end{rlist}
\end{alist}
