Η υπερβολή στο παρακάτω σχήμα έχει εστίες τα σημεία $E'(-10,0)$ και $E(10,0)$ και κορυφές τα σημεία $A'(-8,0)$ και $A(8,0)$.
\begin{alist}
    \item Να αποδείξετε ότι η υπερβολή έχει εξίσωση
    \[ \frac{x^2}{64} - \frac{y^2}{36} = 1 \] \monades{12}
    \item Έστω $M$ ένα σημείο της υπερβολής.
    \begin{rlist}
        \item Να αποδείξετε ότι $|(ME') - (ME)| = 16$. \monades{8}
        \item Αν $(ME) = 9$, να βρείτε την απόσταση του σημείου $M$ από την εστία $E'$. \monades{5}
    \end{rlist}
\end{alist}
