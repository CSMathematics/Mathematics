\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.2. Γενική Μορφή Εξίσωσης Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Σε καρτεσιανό σύστημα αξόνων Oxy η εξίσωση $3x + 4y = 25$ περιγράφει τη θέση ενός αγωγού ύδρευσης. Σε αυτό το σύστημα θέλουμε να σχεδιάσουμε ένα κυκλικό συντριβάνι με κέντρο το $O(0,0)$ και ακτίνα $2$.

\begin{alist}
\item
    \begin{rlist}
    \item Ποια είναι η εξίσωση του κύκλου που περιγράφει την θέση του συντριβανιού; \monades{04}
    \item Να εξετάσετε αν ο αγωγός ύδρευσης διέρχεται από το κέντρο του συντριβανιού, προκειμένου να ενωθεί με αυτό. \monades{05}
    \item Αν ο αγωγός ύδρευσης δεν διέρχεται από το κέντρο του συντριβανιού, ποιο σημείο του αγωγού ύδρευσης πρέπει να ενωθεί με το κέντρο του συντριβανιού ώστε να έχουμε την μικρότερη δυνατή απόσταση, άρα και οικονομικότερη κατασκευή; \monades{08}
    \end{rlist}
\item Ο μηχανικός που θέλει να χαράξει έναν ευθύγραμμο δρόμο, κατέληξε στην εξίσωση $\lambda x + y + \lambda - 2 = 0$, με $\lambda \neq 0$. Μπορείτε να τον βοηθήσετε να βρει για ποια τιμή του $\lambda$ ο δρόμος αυτός εφάπτεται του συντριβανιού; \monades{08}
\end{alist}
