\Askhsh{15081}{4}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται οι κύκλοι
\[
C_1 : x^2 + y^2 + 2\sqrt{2}x + 1 = 0 \quad \text{και}
\]
\[
C_2 : x^2 + y^2 - 6\sqrt{2}x + 9 = 0.
\]
\begin{alist}
\item Να δείξετε ότι οι κύκλοι $C_1$ και $C_2$ έχουν κέντρα $\mathrm{K}(-\sqrt{2},0)$, $\Lambda(3\sqrt{2},0)$ και ακτίνες $\rho_1=1$, $\rho_2=3$ αντίστοιχα. \monades{8}
\item
    \begin{rlist}
    \item Να δείξετε ότι από την αρχή των αξόνων διέρχονται δύο κοινές εφαπτόμενες των κύκλων $C_1$ και $C_2$. \monades{10}
    \item Να σχεδιάσετε ένα πρόχειρο σχήμα όπου να φαίνονται οι κύκλοι και οι δύο αυτές εφαπτόμενες. \monades{7}
    \end{rlist}
\end{alist}
