\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
\end{ylh}
Δίνονται οι ευθείες: $\varepsilon_1 : 2x+y-6=0$ και $\varepsilon_2 : 2x+y+2=0$.
\begin{alist}
\item Να δείξετε ότι οι ευθείες $\varepsilon_1$ και $\varepsilon_2$ είναι παράλληλες. \monades{12}
\item
    \begin{rlist}
    \item Να δείξετε ότι το σημείο $A(0,6)$ ανήκει στην ευθεία $\varepsilon_1$. \monades{5}
    \item Να υπολογίσετε την απόσταση των ευθειών $\varepsilon_1$ και $\varepsilon_2$. \monades{8}
    \end{rlist}
\end{alist}
