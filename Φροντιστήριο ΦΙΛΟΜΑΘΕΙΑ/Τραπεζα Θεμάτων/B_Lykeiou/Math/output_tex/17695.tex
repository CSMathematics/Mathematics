\Askhsh{17695}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας
\end{ylh}
Υποθέτουμε, ότι σε ένα επίπεδο που έχουμε εφοδιάσει με ορθοκανονικό σύστημα συντεταγμένων, κινούνται δύο σημεία A και B. Κάθε χρονική στιγμή $t$ με $t \ge 0$ η θέση του πρώτου σημείου είναι $A(t-1, 2t-1)$ και του δεύτερου $B(3t-1, -4t-1)$.
\begin{alist}
    \item Να βρείτε τις εξισώσεις των γραμμών πάνω στις οποίες κινούνται τα δύο σημεία. \monades{8}
    \item Υπάρχει χρονική στιγμή κατά την οποία τα δύο σημεία ταυτίζονται; \monades{7}
    \item Να υπολογιστεί η απόσταση των δύο σημείων την χρονική στιγμή $t=2$. \monades{5}
    \item Να βρεθεί η χρονική στιγμή $t$ κατά την οποία η απόσταση του σημείου $A$ από την ευθεία $\epsilon: 4x+3y+7=0$ ισούται με $6$. \monades{5}
\end{alist}
