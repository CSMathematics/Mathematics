\Askhsh{16057}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 2.3. Εμβαδόν Τριγώνου.
\end{ylh}
Δίνονται τα σημεία $A(2,0)$, $B(3,4)$ και $\lambda \in \mathbb{R}$.
\begin{alist}
\item
    \begin{rlist}
    \item Να βρείτε την εξίσωση που περιγράφει όλες τις ευθείες που διέρχονται από το σημείο $A$ και έχουν κλίση $\lambda$. \monades{5}
    \item Να αποδείξετε ότι η ευθεία, η οποία διέρχεται από το σημείο $A$, έχει κλίση $\lambda$ και απέχει απόσταση ίση με 1 από το σημείο $B$, έχει εξίσωση ($\varepsilon$): $15x - 8y - 30 = 0$. \monades{8}
    \end{rlist}
\item Να αποδείξετε ότι υπάρχει και άλλη ευθεία ($\zeta$), εκτός από την ($\varepsilon$), η οποία διέρχεται από το σημείο $A$ και απέχει απόσταση ίση με 1 από το σημείο $B$. \monades{5}
\item Να βρείτε τις εξισώσεις των διχοτόμων των γωνιών που σχηματίζουν οι ευθείες ($\varepsilon$) και ($\zeta$). \monades{7}
\end{alist}
