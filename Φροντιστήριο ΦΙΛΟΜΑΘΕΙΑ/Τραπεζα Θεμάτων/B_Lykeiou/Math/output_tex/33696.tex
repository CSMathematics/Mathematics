\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
    \item 2.3. Εμβαδόν Τριγώνου
\end{ylh}
Το κέντρο $K$ ενός κύκλου $(c)$ βρίσκεται στο πρώτο τεταρτημόριο και είναι σημείο της ευθείας $(\varepsilon): y=2x-1$. Ο κύκλος $(c)$ έχει ακτίνα $\rho=3\sqrt{2}$ και η ευθεία $(\zeta): x+y-2=0$ εφάπτεται στον κύκλο στο σημείο $A$.
\begin{alist}
\item Να αποδείξετε ότι η εξίσωση του κύκλου $(c)$ είναι $(x-3)^2 + (y-5)^2 = 18$. \monades{09}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Η εξίσωση της ευθείας $KA$ είναι η $x-y+2=0$. \monades{05}
    \item Οι συντεταγμένες του $A$ είναι $(0,2)$. \monades{04}
    \end{rlist}
\item Να υπολογισθεί το εμβαδόν του τριγώνου $A\Lambda M$, όπου $M$ και $\Lambda$ είναι τα σημεία τομής της ευθείας $(\varepsilon)$ με τον κύκλο $(c)$. \monades{07}
\end{alist}
