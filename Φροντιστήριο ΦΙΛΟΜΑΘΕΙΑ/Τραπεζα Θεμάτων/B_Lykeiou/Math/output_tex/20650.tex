\Askhsh{20650}{4}
\begin{ylh}{1}
    \item 1.4. Συντεταγμένες στο Επίπεδο
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
\begin{alist}
\item Δίνονται οι ευθείες $\varepsilon_1: y=x+2$, $\varepsilon_2: y=x-2$ και τα σημεία $A(-2,0)$, $B(2,0)$ των ευθειών $\varepsilon_1, \varepsilon_2$ αντίστοιχα.
\begin{rlist}
\item Να αποδειχθεί ότι $\varepsilon_1 // \varepsilon_2$. \monades{04}
\item Να βρεθούν οι συντεταγμένες του μέσου $M$, του $AB$. \monades{02}
\item Να βρεθεί η εξίσωση της μεσοπαραλλήλου των ευθειών $\varepsilon_1, \varepsilon_2$. \monades{06}
\end{rlist}
\item Ο κύκλος $(K, \rho)$ έχει την ιδιότητα να εφάπτεται των ευθειών $\varepsilon_1$ και $\varepsilon_2$. Αν το κέντρο $K$ του κύκλου $(K, \rho)$ ανήκει στην ευθεία $(\eta): x=\lambda$, όπου $\lambda \in \mathbb{R}$, τότε:
\begin{rlist}
\item Να βρεθούν οι συντεταγμένες του κέντρου $K$, συναρτήσει του $\lambda$. \monades{05}
\item Να αποδείξετε ότι η ακτίνα $\rho$ είναι ανεξάρτητη του $\lambda$ και να γράψετε την εξίσωση που παριστάνει όλους τους κύκλους $(K, \rho)$, για τις διάφορες τιμές του $\lambda \in \mathbb{R}$. \monades{08}
\end{rlist}
\end{alist}
