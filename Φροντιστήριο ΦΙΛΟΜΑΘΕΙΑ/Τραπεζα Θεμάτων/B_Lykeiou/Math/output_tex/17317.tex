\Askhsh{17317}{2}
\begin{ylh}{1}
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος $C: (x - 1)^2 + (y - 2)^2 = 4$ και η ευθεία $\varepsilon: 3x - 4y = 8$.
\begin{alist}
    \item Να βρείτε το κέντρο $K$ του κύκλου $C$ και την ακτίνα του. \monades{5}
    \item Αν $K(1,2)$, να δείξετε ότι η απόσταση του κέντρου του κύκλου $C$ από την ευθεία $\varepsilon$ είναι $d(K,\varepsilon) = \frac{13}{5}$. \monades{13}
    \item Να αιτιολογήσετε γιατί η ευθεία και ο κύκλος δεν έχουν κανένα κοινό σημείο. \monades{7}
\end{alist}
