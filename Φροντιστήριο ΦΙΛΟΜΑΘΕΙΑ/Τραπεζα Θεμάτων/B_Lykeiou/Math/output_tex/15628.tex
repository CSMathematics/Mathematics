\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται η εξίσωση $x^2 + y^2 + (4 - 2k)x - 2(1 + k)y + 5 - 2k = 0 \ (I)$, όπου $k \in (0, +\infty)$.
\begin{alist}
\item Να αποδείξετε ότι η $(I)$ παριστάνει κύκλο με κέντρο $M(k - 2, k + 1)$ και ακτίνα $k\sqrt{2}$ για κάθε $k > 0$. \monades{10}
\item Να αποδείξετε ότι το σημείο $M$ ανήκει σε μια σταθερή ευθεία για κάθε $k > 0$. \monades{7}
\item Να αποδείξετε ότι η ευθεία $(\varepsilon): y = -x - 1$ είναι εφαπτομένη του παραπάνω κύκλου για κάθε $k > 0$. \monades{8}
\end{alist}
