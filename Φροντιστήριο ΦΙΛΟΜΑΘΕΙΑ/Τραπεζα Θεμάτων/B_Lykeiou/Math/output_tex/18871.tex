\Askhsh{18871}{4}
\begin{ylh}{1}
    \item 2.1 Η Ευθεία.
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνεται ο κύκλος $(c)$ με κέντρο $O(0,0)$ και ακτίνα $\rho = \sqrt{5}$ και το σημείο $A(3,1)$.
\begin{alist}
\item Να βρείτε την εξίσωση του κύκλου και να αποδείξετε οτι το σημείο $A$ είναι εξωτερικό του κύκλου. \monades{07}
\item \begin{rlist}
    \item Να αποδείξετε οτι οι ευθείες που εφάπτονται στον κύκλο $(c)$ και διέρχονται από το σημείο $A$ έχουν εξισώσεις $(\varepsilon_1): 2x - y = 5$ και $(\varepsilon_2): x + 2y = 5$. \monades{10}
    \item Να βρείτε την εξίσωση της διχοτόμου της γωνίας $\widehat{B\kern-0.2em A\kern-0.2em \Gamma}$, όπου $B$ και $\Gamma$ είναι αντίστοιχα τα σημεία επαφής των ευθειών $(\varepsilon_1)$ και $(\varepsilon_2)$ με τον κύκλο. \monades{08}
\end{rlist}
\end{alist}
