\Askhsh{21307}{2}
\begin{ylh}{1}
    \item 3.2 Η Παραβολή.
\end{ylh}
Σε καρτεσιανό επίπεδο Oxy δίνεται η παραβολή με εξίσωση $x^2 = 12y$.
\begin{alist}
\item Να αποδείξετε ότι η εστία της παραβολής είναι το σημείο $E(0,3)$ και να βρείτε τα σημεία της παραβολής που έχουν τεταγμένη 3. \monades{12}
\item Να αποδείξετε ότι οι εφαπτομένες $(\varepsilon_1)$ και $(\varepsilon_2)$ της παραβολής στα σημεία $A(6,3)$ και $B(-6,3)$, αντίστοιχα, έχουν εξισώσεις $y = x - 3$ και $y = -x - 3$. \monades{08}
\item Να βρείτε το σημείο τομής των $(\varepsilon_1)$ και $(\varepsilon_2)$. \monades{05}
\end{alist}
