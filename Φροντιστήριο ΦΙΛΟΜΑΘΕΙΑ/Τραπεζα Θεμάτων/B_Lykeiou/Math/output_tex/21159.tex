\Askhsh{21159}{4}
\begin{ylh}{1}
    \item 2.1. Εξίσωση Ευθείας
    \item 3.1 Ο Κύκλος.
\end{ylh}
Δίνονται τα σημεία $A(\alpha, 0)$ και $B(0, \beta)$ με $\alpha, \beta > 0$ και $\alpha + \beta = 10$.
\begin{alist}
\item Να αποδείξετε ότι η εξίσωση των κύκλων με διάμετρο την $AB$, για κάθε τιμή των $\alpha$ και $\beta$ είναι $x^2 + y^2 - \alpha x - (10 - \alpha)y = 0$. \monades{8}
\item Να αποδείξετε ότι όλοι οι κύκλοι με διάμετρο την $AB$, για τις διάφορες τιμές των $\alpha$ και $\beta$ διέρχονται από δύο σταθερά σημεία, την αρχή $O$ των αξόνων και ένα σημείο $P$ του οποίου να βρείτε τις συντεταγμένες. \monades{9}
\item Να βρείτε τον γεωμετρικό τόπο των κέντρων όλων των κύκλων με διάμετρο την $AB$ για τις διάφορες τιμές των $\alpha$ και $\beta$. \monades{8}
\end{alist}
