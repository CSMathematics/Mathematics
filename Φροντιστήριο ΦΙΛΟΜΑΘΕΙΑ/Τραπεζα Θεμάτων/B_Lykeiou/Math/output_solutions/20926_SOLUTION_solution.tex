\lysh{20926_SOLUTION}{1}
\textgreek{ΛΥΣΗ}

\textgreek{α) Για να αποδείξουμε ότι το σημείο $B(1,0)$ ανήκει στην $\varepsilon$, αρκεί να δείξουμε ότι}
\[
1 - 2 \cdot 0 = 1 \text{ ή } 1 = 1,
\]
\textgreek{που ισχύει.}

\textgreek{Για το σημείο $A(0,2)$ έχουμε: $0 - 2 \cdot 2 = -4 \neq 1$, επομένως το σημείο $A$ δεν είναι σημείο της $\varepsilon$, αφού οι συντεταγμένες του δεν επαληθεύουν την εξίσωση της ευθείας.}

\textgreek{β) Έχουμε:}
\[
d(A, \varepsilon) = \frac{|0 - 2 \cdot 2 - 1|}{\sqrt{1^2 + (-2)^2}} = \frac{5}{\sqrt{5}} = \sqrt{5}.
\]

\textgreek{γ) Έχουμε:}
\[
(AB) = \sqrt{(1 - 0)^2 + (0 - 2)^2} = \sqrt{5}.
\]
\textgreek{Επομένως $(AB) = d(A, \varepsilon)$, δηλαδή η προβολή του $A$ στην ευθεία $\varepsilon$ είναι το $B$.}
\end{lysh}
