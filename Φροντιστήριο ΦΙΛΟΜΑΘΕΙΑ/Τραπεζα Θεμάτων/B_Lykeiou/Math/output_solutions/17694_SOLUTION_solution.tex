\lysh{17694_SOLUTION}{1}
\text{ΛΥΣΗ}

\text{α) Εφόσον τα σημεία της ευθείας } $\varepsilon$ \text{ πάνω στην οποία βρίσκεται η σιδηροδρομική}
\text{γραμμή ισαπέχουν από τα A, B, αυτή θα είναι η μεσοκάθετος του ευθυγράμμου}
\text{τμήματος AB. Αν M το μέσο του AB τότε:}
\[ x_M = \frac{x_A+x_B}{2} = \frac{3+7}{2} = 5, y_M = \frac{y_A+y_B}{2} = \frac{-2+6}{2} = 2, \]
\text{άρα M(5, 2).}

\text{Επίσης } $\lambda_{AB} = \frac{y_B-y_A}{x_B-x_A} = \frac{-2-6}{7-3} = \frac{-8}{4} = -2,$

\text{οπότε: } $\varepsilon \perp AB \Leftrightarrow \lambda_{\varepsilon} \cdot \lambda_{AB} = -1 \Leftrightarrow \lambda_{\varepsilon} = \frac{-1}{\lambda_{AB}} = \frac{-1}{-2} = \frac{1}{2},$

\text{άρα } $\varepsilon: y - y_M = \lambda_{\varepsilon}(x - x_M)$ \text{ ή } $\varepsilon: y - 2 = \frac{1}{2}(x - 5)$ \text{ ή } $\varepsilon: 2y - 4 = x - 5$ \text{ ή } $\varepsilon: x - 2y - 1 = 0,$
\text{η ζητούμενη εξίσωση.}

\text{β) Θέλουμε } $(\Sigma AB) = 20$ \text{ τ.μ. (1)}

\text{Αν } $\Sigma(x, y)$, \text{ τότε το } $\Sigma \in \varepsilon \Leftrightarrow x - 2y - 1 = 0 \Leftrightarrow x = 2y + 1$ \text{ (2), οπότε } $\Sigma(2y + 1, y).$

\text{Ακόμη } $\vec{AB} = (x_B - x_A, y_B - y_A) = (7-3, -2-6) = (4, -8)$ \text{ και}

$\vec{A\Sigma} = (x_{\Sigma} - x_A, y_{\Sigma} - y_A) = (2y + 1 - 3, y - 6) = (2y - 2, y - 6)$, \text{ οπότε από την (1) είναι:}
\[ \frac{1}{2} |\det(\vec{AB}, \vec{A\Sigma})| = 20 \Leftrightarrow \left|\begin{vmatrix} X_{AB} & Y_{AB} \\ X_{A\Sigma} & Y_{A\Sigma} \end{vmatrix}\right| = 40 \Leftrightarrow \left|\begin{vmatrix} 4 & -8 \\ 2y - 2 & y - 6 \end{vmatrix}\right| = 40 \Leftrightarrow \]
\[ |4(y - 6) - (-8)(2y - 2)| = 40 \Leftrightarrow |4y - 24 + 16y - 16| = 40 \Leftrightarrow |20y - 40| = 40 \Leftrightarrow \]
\[ 20|y - 2| = 40 \Leftrightarrow |y - 2| = 2 \Leftrightarrow y - 2 = -2 \text{ ή } y - 2 = 2 \Leftrightarrow y = 0 \text{ ή } y = 4. \]
\text{Για } $y = 0$ \text{ από την (2) παίρνουμε: } $x = 1$. \text{ Για } $y = 4$ \text{ από την (2) παίρνουμε: } $x = 9$.

\text{Άρα } $\Sigma(1, 0)$ \text{ ή } $\Sigma(9, 4)$.
\end{lysh}
