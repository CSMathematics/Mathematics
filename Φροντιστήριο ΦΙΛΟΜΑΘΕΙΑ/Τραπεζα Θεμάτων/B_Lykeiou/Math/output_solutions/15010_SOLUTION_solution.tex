\textnormal{ΛΥΣΗ}

\textnormal{α)}
\textnormal{i. Είναι } $\vec{\mathrm{A}\Delta} = \vec{\mathrm{A}\mathrm{B}} + \vec{\mathrm{B}\Delta} = \vec{\mathrm{A}\mathrm{B}} + \left(\vec{\mathrm{B}\mathrm{A}} + \vec{\mathrm{B}\Gamma}\right) = \vec{\mathrm{B}\Gamma} \quad (1)$

\textnormal{και } $\vec{\mathrm{A}\mathrm{E}} = \vec{\mathrm{A}\mathrm{\Gamma}} + \vec{\mathrm{\Gamma}\mathrm{E}} = \vec{\mathrm{A}\mathrm{\Gamma}} + \left(\vec{\mathrm{\Gamma}\mathrm{A}} + \vec{\mathrm{\Gamma}\mathrm{B}}\right) = \vec{\mathrm{\Gamma}\mathrm{B}} \quad (2)$

\textnormal{ii. Από τις σχέσεις (1) και (2) συμπεραίνουμε ότι } $\vec{\mathrm{A}\Delta} = -\vec{\mathrm{A}\mathrm{E}} \textnormal{ δηλαδή τα διανύσματα } \vec{\mathrm{A}\Delta} \textnormal{ και } \vec{\mathrm{A}\mathrm{E}} \textnormal{ είναι αντίθετα.}$

\textnormal{β) Από το α) ερώτημα είναι } $\vec{\mathrm{A}\Delta} = (-1)\vec{\mathrm{A}\mathrm{E}}\textnormal{, οπότε τα διανύσματα } \vec{\mathrm{A}\Delta} \textnormal{ και } \vec{\mathrm{A}\mathrm{E}} \textnormal{ είναι παράλληλα. Επιπλέον έχουν κοινό σημείο το } \mathrm{A}\textnormal{, άρα τα } \mathrm{A},\Delta \textnormal{ και } \mathrm{E} \textnormal{ είναι συνευθειακά.}$
