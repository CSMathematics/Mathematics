\lysh{17805_SOLUTION}{1}
\text{ΛΥΣΗ}

\vspace{0.5em} % Small vertical space for readability between parts

\noindent α) Είναι σύμφωνα με το σχήμα $\vec{OA} = (3,4)$.
Ακόμα
\[
\vec{A\Delta} = (x_\Delta - x_A, y_\Delta - y_A) = \left(\frac{12}{5}-3, \frac{16}{5}-4\right) = \left(-\frac{3}{5}, -\frac{4}{5}\right).
\]

\noindent β) Έχουμε
\[
\vec{A\Delta} = \left(-\frac{3}{5}, -\frac{4}{5}\right) = -\frac{1}{5}(3,4) = -\frac{1}{5}\vec{OA}.
\]

\noindent γ) Είναι $\vec{AB} = (x_B - x_A, y_B - y_A) = (4,-3)$.

\vspace{0.5em} % Small vertical space

\noindent Έχουμε
\[
(\text{A}\Delta\text{B}) = \frac{1}{2}\left|\det(\vec{A\Delta}, \vec{AB})\right| \stackrel{\beta)}{=} \frac{1}{2}\begin{vmatrix} -\frac{3}{5} & -\frac{4}{5} \\ 4 & -3 \end{vmatrix} = \frac{1}{2}\left|\frac{9}{5} + \frac{16}{5}\right| = \frac{1}{2}\left|\frac{25}{5}\right| = \frac{1}{2} \cdot 5 = \frac{5}{2} \text{ τετραγωνικές μονάδες}.
\]
Έχει δοθεί $(\text{OAB}) = \frac{25}{2}$ τετραγωνικές μονάδες, επομένως $(\text{A}\Delta\text{B}) = \frac{5}{2} = \frac{1}{5} \cdot \frac{25}{2} = \frac{1}{5}(\text{OAB})$.
\end{lysh}
