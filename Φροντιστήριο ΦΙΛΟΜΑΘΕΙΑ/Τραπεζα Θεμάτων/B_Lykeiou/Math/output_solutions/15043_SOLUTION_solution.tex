\lysh{15043_SOLUTION}{1}
\text{ΛΥΣΗ}

\textbf{α)}
\begin{enumerate}[i.]
    \item \text{Είναι } $\vec{AB} = (4 - 0, 1 - 5) = (4,-4)$ \text{ και } $\vec{B\Gamma} = (6 - 4, -1 - 1) = (2, -2)$.
    \item \text{Είναι } $\det(\vec{AB}, \vec{B\Gamma}) = \begin{vmatrix} 4 & -4 \\ 2 & -2 \end{vmatrix} = -8 + 8 = 0$.
\end{enumerate}
\text{Επομένως } $\vec{AB} // \vec{B\Gamma}$ \text{ και αφού υπάρχει κοινό σημείο το } $B$, \text{ τότε τα σημεία } $A, B, \Gamma$ \text{ είναι συνευθειακά.}

\textbf{β) Είναι:}
\[
|\vec{AB}| = \sqrt{4^2 + (-4)^2} = \sqrt{16 + 16} = \sqrt{32} = \sqrt{2 \cdot 16} = 4\sqrt{2}
\]
\text{ και }
\[
|\vec{B\Gamma}| = \sqrt{2^2 + (-2)^2} = \sqrt{4 + 4} = \sqrt{8} = \sqrt{2 \cdot 4} = 2\sqrt{2}.
\]
\text{Επομένως έχουμε ότι } $|\vec{AB}| = 2 |\vec{B\Gamma}|$.

\textbf{2\textsuperscript{ος} τρόπος:}
\[
\text{Επειδή } \vec{AB} = (4, -4) = 2 \cdot (2, -2) = 2 \vec{B\Gamma} \Rightarrow |\vec{AB}| = 2 |\vec{B\Gamma}|.
\]
\end{lysh}
