\text{ΛΥΣΗ}

\text{α)}
\begin{enumerate}
    \item[i.] Οι συντεταγμένες του μέσου $M$ του τμήματος $AB$ είναι $M\left(\frac{\alpha}{2}, \frac{\beta}{2}\right)$.
    \item[ii.] Η απόσταση $(OM)$ είναι
    \[ (OM) = \sqrt{\left(\frac{\alpha}{2}-0\right)^{2}+\left(\frac{\beta}{2}-0\right)^{2}} = \sqrt{\frac{\alpha^{2}}{4}+\frac{\beta^{2}}{4}} = \sqrt{\frac{\alpha^{2}+\beta^{2}}{4}} = \frac{\sqrt{\alpha^{2}+\beta^{2}}}{2} . \]
\end{enumerate}

\text{β)}
\begin{enumerate}
    \item[i.] $(AB)=\sqrt{(\alpha-0)^{2}+(0-\beta)^{2}}=\sqrt{\alpha^{2}+\beta^{2}}=2 \cdot(OM) \text{ ή } (OM)=\frac{(AB)}{2}$.
    \item[ii.] Η πρόταση της Ευκλείδειας Γεωμετρίας που έχει αποδειχθεί είναι η εξής:

    Σε κάθε ορθογώνιο τρίγωνο, η διάμεσος που αντιστοιχεί στην υποτείνουσα ισούται με το μισό της υποτείνουσας.
\end{enumerate}

\text{γ) Αφού } $(OM)=\frac{(AB)}{2}=(AM)=(BM)$ \text{ συμπεραίνουμε ότι το σημείο } $M$ \text{ ισαπέχει από τις κορυφές του τριγώνου } $OAB$ \text{ και επομένως είναι το κέντρο του ζητούμενου περιγεγραμμένου κύκλου. Επίσης η ακτίνα του ζητούμενου κύκλου είναι η}
$(OM) = \frac{\sqrt{\alpha^{2}+\beta^{2}}}{2} . \text{ Συνεπώς ο ζητούμενος κύκλος έχει εξίσωση}$
\[ \left(x-\frac{\alpha}{2}\right)^{2}+\left(x-\frac{\beta}{2}\right)^{2}=\frac{\alpha^{2}+\beta^{2}}{4} . \]
