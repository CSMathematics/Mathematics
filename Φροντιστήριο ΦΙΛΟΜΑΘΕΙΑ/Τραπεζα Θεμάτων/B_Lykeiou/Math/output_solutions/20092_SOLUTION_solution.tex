\textbf{\alpha)}
\begin{enumerate}
    \item Αρκεί να δείξουμε ότι το σύστημα των εξισώσεων της παραβολής $y^2 = 4x$ και της ευθείας $\frac{x}{3} - \frac{y}{4} + 1 = 0$ είναι αδύνατο.
    \[
    \left\{
    \begin{array}{l}
    \frac{x}{3} - \frac{y}{4} + 1 = 0 \\
    y^2 = 4x
    \end{array}
    \right.
    \Leftrightarrow
    \left\{
    \begin{array}{l}
    4x - 3y + 12 = 0 \\
    y^2 = 4x
    \end{array}
    \right.
    \Leftrightarrow
    \left\{
    \begin{array}{l}
    y^2 - 3y + 12 = 0 \\
    y^2 = 4x
    \end{array}
    \right.
    \]
    Η δευτέρου βαθμού εξίσωση $y^2 - 3y + 12 = 0$ δεν έχει πραγματικές ρίζες, αφού $\Delta = 9 - 48 < 0$, άρα το σύστημα είναι αδύνατο.

    Η ευθεία $\varepsilon$ ισοδύναμα γράφεται $4x - 3y + 12 = 0$.
    $d(M, \varepsilon) = \frac{|4\cdot 1 - 3\cdot \frac{1}{4} + 12|}{\sqrt{4^2+(-3)^2}} = \frac{|16 - \frac{3}{4} + 12|}{\sqrt{16+9}} = \frac{|\frac{64-3+48}{4}|}{5} = \frac{|\frac{109}{4}|}{5} = \frac{109}{20}$.

    \item Τα σημεία τομής της ευθείας $\varepsilon$ με τους άξονες τα βρίσκουμε θέτοντας $y=0$ και $x=0$ στην εξίσωσή της.

    Για $y=0$ έχουμε $\frac{x}{3} = -1$ ή $x=-3$, άρα $\Gamma(-3, 0)$.
    Για $x=0$ έχουμε $-\frac{y}{4} = 1$ ή $y=-4$, άρα $\Delta(0, -4)$.

    Για το εμβαδό του τριγώνου $M\Gamma\Delta$ θα βρούμε το μήκος του τμήματος $\Gamma\Delta$ γιατί το ύψος που αντιστοιχεί σε αυτό είναι η απόσταση του σημείου $M$ από την ευθεία $\varepsilon$.

    $(\Gamma\Delta) = \sqrt{(-3-0)^2 + (0-(-4))^2} = \sqrt{(-3)^2 + 4^2} = \sqrt{9+16} = \sqrt{25} = 5$.

    Άρα $(M\Gamma\Delta) = \frac{1}{2}(\Gamma\Delta) \cdot d(M, \varepsilon) = \frac{1}{2} \cdot 5 \cdot \frac{109}{20} = \frac{5 \cdot 109}{40} = \frac{109}{8}$ τ.μ.
\end{enumerate}

\textbf{\beta)}
\begin{enumerate}
    \item Η εφαπτομένη $\zeta$ της παραβολής σε τυχαίο σημείο της $(x_1, y_1)$ με $y_1 \neq 0$ έχει εξίσωση $yy_1 = 2(x+x_1)$ ή $y = \frac{2}{y_1}x + \frac{2x_1}{y_1}$ με συντελεστή διεύθυνσης $\lambda_\zeta = \frac{2}{y_1}$.

    Η ευθεία $\varepsilon$ έχει συντελεστή διεύθυνσης $\lambda_\varepsilon = \frac{4-0}{0-(-3)} = \frac{4}{3}$. Για να είναι η ευθεία $\varepsilon$ παράλληλη της εφαπτομένης $\zeta$ πρέπει και αρκεί $\lambda_\zeta = \lambda_\varepsilon$ ή $\frac{2}{y_1} = \frac{4}{3}$ ή $y_1 = \frac{3}{2}$.

    Το σημείο $(x_1, y_1)$ επαληθεύει την εξίσωση της παραβολής, επομένως $y_1^2 = 4x_1$ ή $(\frac{3}{2})^2 = 4x_1$ ή $\frac{9}{4} = 4x_1$ ή $x_1 = \frac{9}{16}$.

    Για $x_1 = \frac{9}{16}$, $y_1 = \frac{3}{2}$ η εφαπτομένη $\zeta$ που είναι παράλληλη της ευθείας $\varepsilon$ είναι:
    $y = \frac{2}{y_1}x + \frac{2x_1}{y_1}$
    $y = \frac{2}{3/2}x + \frac{2(9/16)}{3/2}$
    $y = \frac{4}{3}x + \frac{9/8}{3/2}$
    $y = \frac{4}{3}x + \frac{9}{8} \cdot \frac{2}{3}$
    $y = \frac{4}{3}x + \frac{3}{4}$
    ή τελικά $\zeta: 16x - 12y + 9 = 0$.

    \item Για να βρούμε την απόσταση των ευθειών $\zeta$ και $\varepsilon$, αρκεί να βρούμε ένα σημείο, έστω $K$, της $\zeta$ και να υπολογίσουμε την απόσταση του σημείου αυτού από την ευθεία $\varepsilon$.

    Θέτοντας $x=0$ στην εξίσωση της ευθείας $\zeta$ έχουμε: $y = \frac{9}{12} = \frac{3}{4}$. Άρα το σημείο $K$ της ευθείας $\zeta$ είναι το $K(0, \frac{3}{4})$.

    Η εξίσωση της ευθείας $\varepsilon$ είναι: $4x - 3y + 12 = 0$, επομένως:
    $d(\zeta, \varepsilon) = d(K, \varepsilon) = \frac{|4\cdot 0 - 3\cdot \frac{3}{4} + 12|}{\sqrt{4^2+(-3)^2}} = \frac{|-\frac{9}{4} + 12|}{\sqrt{25}} = \frac{|\frac{-9+48}{4}|}{5} = \frac{|\frac{39}{4}|}{5} = \frac{39}{20}$.
\end{enumerate}
