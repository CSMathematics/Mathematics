\noindent \textbf{ΛΥΣΗ}
\begin{enumerate}[α)]
    \item Η υπερβολή $C$ έχει εστίες τα σημεία $E(5,0),E'(-5,0)$ οπότε έχει εξίσωση της μορφής $\frac{x^2}{\alpha^2} - \frac{y^2}{\beta^2}=1$ και $\gamma=5$. Αφού έχει εκκεντρότητα $\frac{5}{4}$ συμπεραίνουμε ότι
    \[ \frac{\gamma}{\alpha} = \frac{5}{4} \Leftrightarrow \frac{5}{\alpha} = \frac{5}{4} \Leftrightarrow \alpha = 4. \]
    Από τη σχέση $\gamma^2 = \alpha^2 + \beta^2$ έχουμε ότι
    \[ 5^2 = 4^2 + \beta^2 \Leftrightarrow \beta^2 = 9 \xrightarrow{\beta>0} \beta = 3. \]
    Τελικά η ζητούμενη εξίσωση είναι η $\frac{x^2}{16} - \frac{y^2}{9}=1$.
    \item Οι εξισώσεις των ασύμπτωτων της $C$ είναι $y = \frac{\beta}{\alpha}x \Leftrightarrow y = \frac{3}{4}x$ και $y = -\frac{\beta}{\alpha}x \Leftrightarrow y = -\frac{3}{4}x$.
    \item Η εφαπτόμενη στο $M(5,\frac{9}{4})$ έχει εξίσωση
    \[ \frac{5 \cdot x}{25} - \frac{\frac{9}{4} \cdot y}{9} = 1 \Leftrightarrow \frac{x}{5} - \frac{y}{4}=1. \]
\end{enumerate}
\end{lysh}
