\lysh{20864_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\textbf{α)} Η ευθεία $\varepsilon_1: 2x+y-6=0$ γράφεται και $\varepsilon_1: y=-2x+6$ με συντελεστή διεύθυνσης $\lambda_1 = -2$.
Η ευθεία $\varepsilon_2: 2x+y+2=0$ γράφεται και $\varepsilon_2: y=-2x-2$ με συντελεστή διεύθυνσης $\lambda_2 = -2$.
Επειδή $\lambda_1 = \lambda_2$, οι ευθείες $\varepsilon_1$ και $\varepsilon_2$ είναι παράλληλες.

\textbf{β)}
\begin{enumerate}
    \item[i.] Το σημείο $A(0,6)$ είναι σημείο της $\varepsilon_1$, αφού $2 \cdot 0 + 6 - 6 = 0$.
    \item[ii.] Η απόσταση των ευθειών $\varepsilon_1$ και $\varepsilon_2$ είναι ίση με την απόσταση του σημείου $A$ από την ευθεία $\varepsilon_2: 2x+y+2=0$. Άρα:
    \[
    d(\varepsilon_1, \varepsilon_2) = d(A, \varepsilon_2) = \frac{|2 \cdot 0 + 1 \cdot 6 + 2|}{\sqrt{2^2+1^2}} = \frac{8}{\sqrt{5}} = \frac{8\sqrt{5}}{5}.
    \]
\end{enumerate}
