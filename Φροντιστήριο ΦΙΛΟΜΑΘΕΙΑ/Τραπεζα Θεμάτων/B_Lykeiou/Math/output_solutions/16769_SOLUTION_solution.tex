\textbf{ΛΥΣΗ}

\noindent
\textbf{α) Το εμβαδόν του τριγώνου ΑΒΓ είναι:}
\[ (\mathrm{AB}\Gamma) = \frac{1}{2}|\det(\overrightarrow{\mathrm{AB}}, \overrightarrow{\mathrm{B}\Gamma})| \]
όπου
\[ \overrightarrow{\mathrm{AB}} = (x_{\mathrm{B}} - x_{\mathrm{A}}, y_{\mathrm{B}} - y_{\mathrm{A}}) = (-1 - 1, 5 - 7) = (-2, -2) \]
\[ \overrightarrow{\mathrm{B}\Gamma} = (x_{\Gamma} - x_{\mathrm{B}}, y_{\Gamma} - y_{\mathrm{B}}) = (3 - (-1), 3 - 5) = (3 + 1, 3 - 5) = (4, -2) \]
Οπότε:
\[ (\mathrm{AB}\Gamma) = \frac{1}{2}\left|\det\begin{pmatrix} -2 & -2 \\ 4 & -2 \end{pmatrix}\right| = \frac{1}{2}|(-2)(-2) - (-2)(4)| = \frac{1}{2}|4 + 8| = 6 \]

\noindent
\textbf{β)}
\
\noindent
\textbf{i. Το μέσο Μ της πλευράς ΒΓ είναι:}
\[ \mathrm{M} \left(\frac{x_{\mathrm{B}} + x_{\Gamma}}{2}, \frac{y_{\mathrm{B}} + y_{\Gamma}}{2}\right) = \left(\frac{-1 + 3}{2}, \frac{5 + 3}{2}\right) = (1, 4) \]

\noindent
\textbf{ii.} Παρατηρούμε ότι για τα σημεία Α και Μ είναι $x_{\mathrm{M}} = x_{\mathrm{A}} = 1$. Επομένως, η ευθεία ΑΜ είναι κατακόρυφη (δεν ορίζεται ο συντελεστής διεύθυνσης της ευθείας), οπότε έχει εξίσωση $x = 1$.
