\lysh{18732_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\textbf{α)}
\begin{enumerate}
    \item Οι συντεταγμένες του διανύσματος $\vec{\Lambda\Gamma}$ δίνονται από την σχέση $\vec{\Lambda\Gamma} = (x_\Gamma - x_\Lambda, y_\Gamma - y_\Lambda)$, οπότε αντικαθιστώντας παίρνουμε $\vec{\Lambda\Gamma} = \left(-1, -\frac{4}{3}\right)$. Όμοια παίρνουμε $\vec{\Lambda O} = (-3, -4)$, δηλαδή έχουμε $\vec{\Lambda\Gamma} = \left(-1, -\frac{4}{3}\right) = \frac{1}{3}(-3, -4) = \frac{1}{3}\vec{\Lambda O}$.

    Με όμοιο τρόπο βρίσκουμε ότι $\vec{\Lambda\Delta} = \left(\frac{4}{3}, -1\right)$ και $\vec{\Lambda\mathrm{B}} = (4, -3)$, δηλαδή έχουμε $\vec{\Lambda\Delta} = \left(\frac{4}{3}, -1\right) = \frac{1}{3}(4, -3) = \frac{1}{3}\vec{\Lambda\mathrm{B}}$.

    \item Είναι $\lambda_{\Gamma\Delta} = \frac{3-3}{3-3} = \frac{0}{0}$ και $\lambda_{\mathrm{OB}} = \frac{1-0}{7-0} = \frac{1}{7}$ επομένως $\Gamma\Delta \parallel \mathrm{OB}$.

    \item Είναι
    \[
    (\mathrm{A}\Gamma\Delta) = \frac{1}{2}\left|\det(\vec{\Lambda\Gamma}, \vec{\Lambda\Delta})\right| = \frac{1}{2}\left|\det\begin{pmatrix} -1 & 4/3 \\ -4/3 & -1 \end{pmatrix}\right|
    \]
    \[
    = \frac{1}{2}\left|\left(-1\right)\left(-1\right) - \left(\frac{4}{3}\right)\left(-\frac{4}{3}\right)\right| = \frac{1}{2}\left|1 + \frac{16}{9}\right| = \frac{1}{2}\left|\frac{25}{9}\right| = \frac{25}{18} \text{ τ.μ.}
    \]
    Ακόμα
    \[
    (\mathrm{ABO}) = \frac{1}{2}\left|\det(\vec{\Lambda O}, \vec{\Lambda\mathrm{B}})\right| = \frac{1}{2}\left|\det\begin{pmatrix} -3 & 4 \\ -4 & -3 \end{pmatrix}\right|
    \]
    \[
    = \frac{1}{2}\left|(-3)(-3) - (4)(-4)\right| = \frac{1}{2}\left|9 + 16\right| = \frac{1}{2}\left|25\right| = \frac{25}{2} \text{ τ.μ.}
    \]
    Επομένως έχουμε
    \[
    (\mathrm{A}\Gamma\Delta) = \frac{25}{18} = \frac{1}{9} \cdot \frac{25}{2} = \left(\frac{1}{3}\right)^2 (\mathrm{ABO}).
    \]
\end{enumerate}

\textbf{β)} Έχουμε
\[
\vec{\Lambda\mathrm{A}} = \frac{1}{\nu}\vec{\Lambda\mathrm{O}} \Rightarrow \vec{\Lambda\mathrm{A}} = \frac{1}{\nu}(x_\mathrm{O} - x_\Lambda, y_\mathrm{O} - y_\Lambda) \Rightarrow \vec{\Lambda\mathrm{A}} = \frac{1}{\nu}(0-3, 0-4) \Rightarrow \vec{\Lambda\mathrm{A}} = \frac{1}{\nu}(-3, -4).
\]
Όμοια,
\[
\vec{\Lambda\mathrm{B}} = \frac{1}{\nu}\vec{\Lambda\Delta} \Rightarrow \vec{\Lambda\mathrm{B}} = \frac{1}{\nu}(x_\mathrm{B} - x_\Lambda, y_\mathrm{B} - y_\Lambda) \Rightarrow \vec{\Lambda\mathrm{B}} = \frac{1}{\nu}(7-3, 1-4) \Rightarrow \vec{\Lambda\mathrm{B}} = \frac{1}{\nu}(4, -3).
\]
Επομένως
\[
(\mathrm{A}\Gamma\Delta) = \frac{1}{2}\left|\det(\vec{\Lambda\mathrm{A}}, \vec{\Lambda\Delta})\right| = \frac{1}{2}\left|\det\begin{pmatrix} -3/\nu & 4/\nu \\ -4/\nu & -3/\nu \end{pmatrix}\right|
\]
\[
= \frac{1}{2}\left|\left(-\frac{3}{\nu}\right)\left(-\frac{3}{\nu}\right) - \left(\frac{4}{\nu}\right)\left(-\frac{4}{\nu}\right)\right| = \frac{1}{2}\left|\frac{9}{\nu^2} + \frac{16}{\nu^2}\right| = \frac{1}{2}\left|\frac{25}{\nu^2}\right| = \frac{25}{2\nu^2} \text{ τ.μ.}
\]
Από το α)iii. έχουμε $(\mathrm{ABO}) = \frac{25}{2}$ τ.μ., οπότε τελικά,
\[
(\mathrm{A}\Gamma\Delta) = \frac{25}{2\nu^2} = \frac{1}{\nu^2} \cdot \frac{25}{2} = \left(\frac{1}{\nu}\right)^2 (\mathrm{ABO}).
\]
