\lysh{21683_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\text{α) Η εξίσωση της εφαπτομένης του τεταρτοκυκλίου στο σημείο \( M \) είναι: } $xx_1 + yy_1 = 4$.
\begin{itemize}
    \item \text{Αν \( y = 0 \), τότε έχουμε \( x = \frac{4}{x_1} \), οπότε \( K \left(\frac{4}{x_1}, 0\right) \).}
    \item \text{Αν \( x = 0 \), τότε έχουμε \( y = \frac{4}{y_1} \) (προφανώς \( y_1 \neq 0 \)), οπότε \( \Lambda \left(0, \frac{4}{y_1}\right) \).}
\end{itemize}

\text{β) Το μήκος του τμήματος \( K\Lambda \), δεδομένου ότι \( x_1^2 + y_1^2 = 4 \), είναι:}
\[
d = \sqrt{\frac{16}{x_1^2} + \frac{16}{y_1^2}} = 4 \sqrt{\frac{1}{x_1^2} + \frac{1}{y_1^2}} = 4 \sqrt{\frac{x_1^2 + y_1^2}{x_1^2 y_1^2}} = 4 \frac{\sqrt{4}}{x_1 y_1} = \frac{8}{x_1 y_1}
\]
\text{που είναι το ζητούμενο.}

\text{γ) Όταν \( x_1 = \sqrt{2} \), τότε από την ισότητα \( x_1^2 + y_1^2 = 4 \) με \( y_1 > 0 \) έχουμε:}
\[
y_1^2 = 4 - x_1^2 \Leftrightarrow y_1^2 = 4 - 2 \Leftrightarrow y_1^2 = 2 \Leftrightarrow y_1 = \sqrt{2}
\]
\text{οπότε } $d_0 = \frac{8}{\sqrt{2}\sqrt{2}} = \frac{8}{2} = 4$.

\text{δ) Αρκεί να αποδείξουμε ότι } $\frac{8}{x_1 y_1} \ge 4$, \text{ή αρκεί, } $\frac{4}{x_1 y_1} \ge 2$, \text{για την απόδειξη της οποίας αρκεί}
\text{να δείξουμε ότι } $x_1^2 + y_1^2 \ge 2x_1 y_1$, \text{που ισχύει, αφού προκύπτει άμεσα από την προφανή}
\text{ανισότητα } $(x_1 - y_1)^2 \ge 0$.
\end{lysh}
