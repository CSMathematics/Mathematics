\lysh{15439_SOLUTION}{1}
{\large \alpha)}
\begin{enumerate}
    \item Βρίσκουμε την προβολή $Π$ του σημείου $A$ πάνω στην ευθεία $(\varepsilon)$:

    Αφού η κλίση της $(\varepsilon)$ είναι $\lambda_\varepsilon = -1 \Rightarrow \lambda_{A\Pi} = 1$ γιατί η $A\Pi$ είναι κάθετη στην $(\varepsilon)$.

    Έτσι η $A\Pi$ έχει εξίσωση $y - 3 = 1(x - 2)$, δηλαδή $A\Pi: y = x + 1$.

    Οι συντεταγμένες του σημείου $A$ προκύπτουν από την επίλυση του συστήματος:
    \[
    (\Sigma_1)
    \begin{cases}
    y = x + 1 \\
    x + y + 1 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    y = x + 1 \\
    x + (x + 1) + 1 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    y = x + 1 \\
    2x + 2 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    y = 0 \\
    x = -1
    \end{cases}
    \]
    Επομένως, $Π(-1,0)$.

    \item Βρίσκουμε το συμμετρικό σημείο $A'(x,y)$:

    Το $Π$ είναι το μέσον του ευθύγραμμου τμήματος $AA'$, και ως εκ τούτου είναι:
    \[
    \begin{cases}
    \frac{x+2}{2} = -1 \\
    \frac{y+3}{2} = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    x = -4 \\
    y = -3
    \end{cases}
    \text{, δηλαδή } A'(-4,-3).
    \]
\end{enumerate}

{\large \beta)}
\begin{enumerate}
    \item Βρίσκουμε την ανακλώμενη ακτίνα $(\varepsilon_2)$, δηλαδή την $A'M$:

    Είναι η ευθεία που διέρχεται από τα σημεία $A'(-4,-3)$ και $M(1,1)$, δηλαδή
    \[
    y - 1 = \frac{-3-1}{-4-1}(x - 1) \Leftrightarrow 4x - 5y + 1 = 0.
    \]

    \item Οι συντεταγμένες του σημείου $\Sigma$, δηλαδή του σημείου πρόσπτωσης της φωτεινής ακτίνας πάνω στην ευθεία $(\varepsilon)$, προκύπτουν από την επίλυση του συστήματος των ευθειών $(\varepsilon)$ και $(\varepsilon_2)$:
    \[
    (\Sigma_2)
    \begin{cases}
    x + y + 1 = 0 \\
    4x - 5y + 1 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    -4x - 4y - 4 = 0 \\
    4x - 5y + 1 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    x + y + 1 = 0 \\
    -9y - 3 = 0
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    x - \frac{1}{3} + 1 = 0 \\
    y = -\frac{1}{3}
    \end{cases}
    \Leftrightarrow
    \begin{cases}
    x = -\frac{2}{3} \\
    y = -\frac{1}{3}
    \end{cases}
    \text{, δηλαδή } \Sigma\left(-\frac{2}{3},-\frac{1}{3}\right).
    \]
\end{enumerate}

{\large \gamma)} Βρίσκουμε την προσπίπτουσα ακτίνα $(\varepsilon_1)$, δηλαδή την $A\Sigma$:

Είναι η ευθεία που διέρχεται από τα σημεία $A(2,3)$ και $\Sigma\left(-\frac{2}{3},-\frac{1}{3}\right)$, δηλαδή
\[
y - 3 = \frac{3 - \left(-\frac{1}{3}\right)}{2 - \left(-\frac{2}{3}\right)}(x - 2) \Leftrightarrow 5x - 4y + 2 = 0.
\]
