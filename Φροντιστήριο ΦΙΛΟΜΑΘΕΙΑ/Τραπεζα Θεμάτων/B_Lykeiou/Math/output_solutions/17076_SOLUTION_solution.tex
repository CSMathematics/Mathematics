\lysh{17076_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\textbf{α)} Οι συντεταγμένες των διανυσμάτων $\vec{AM}$, $\vec{MB}$ και $\vec{AB}$ είναι αντίστοιχα
\[
\vec{AM} = (x_M - x_A, y_M - y_A) = (x + 3, y + 1),
\]
\[
\vec{MB} = (x_B - x_M, y_B - y_M) = (-x, 3 - y),
\]
\[
\vec{AB} = (x_B - x_A, y_B - y_A) = (3, 4).
\]

\textbf{β)} Τα μέτρα των διανυσμάτων $\vec{AM}$, $\vec{MB}$ και $\vec{AB}$ είναι αντίστοιχα
\[
|\vec{AM}| = \sqrt{(x_M - x_A)^2 + (y_M - y_A)^2} = \sqrt{(x + 3)^2 + (y + 1)^2},
\]
\[
|\vec{MB}| = \sqrt{(x_B - x_M)^2 + (y_B - y_M)^2} = \sqrt{x^2 + (y - 3)^2},
\]
\[
|\vec{AB}| = \sqrt{(x_B - x_A)^2 + (y_B - y_A)^2} = \sqrt{3^2 + 4^2} = \sqrt{25} = 5.
\]

\textbf{γ)} Για το μέτρο του αθροίσματος δύο διανυσμάτων ισχύει
\[
|\vec{AM} + \vec{MB}| \leq |\vec{AM}| + |\vec{MB}| \quad \text{ή} \quad |\vec{AB}| \leq |\vec{AM}| + |\vec{MB}| \quad \text{ή} \quad 5 \leq |\vec{AM}| + |\vec{MB}|,
\]
άρα
\[
|\vec{AM}| + |\vec{MB}| \geq 5.
\]

\textbf{δ)} Από το γ) ερώτημα για οποιοδήποτε σημείο $M(x, y)$ του καρτεσιανού επιπέδου ισχύει
\[
|\vec{AM}| + |\vec{MB}| \geq 5,
\]
οπότε αντικαθιστώντας από το β) ερώτημα θα έχουμε
\[
\sqrt{(x + 3)^2 + (y + 1)^2} + \sqrt{x^2 + (y - 3)^2} \geq 5.
\]

Οπότε δεν υπάρχει ζεύγος $(x, y)$ πραγματικών αριθμών ώστε να ισχύει
\[
\sqrt{(x + 3)^2 + (y + 1)^2} + \sqrt{x^2 + (y - 3)^2} = 4.
\]

Επομένως ο ισχυρισμός είναι ψευδής.
\end{lysh}
