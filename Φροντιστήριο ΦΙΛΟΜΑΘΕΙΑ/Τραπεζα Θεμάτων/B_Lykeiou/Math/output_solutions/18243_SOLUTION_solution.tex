\text{ΛΥΣΗ}

\text{\textalpha) Είναι } $\vec{\alpha} \cdot \vec{\beta} = |\vec{\alpha}| \cdot |\vec{\beta}| \cdot \sigma\upsilon\nu \frac{\pi}{3} = 2 \cdot 4 \cdot \frac{1}{2} = 4$.

\text{\textbeta) Είναι}
\[
\vec{\gamma} \cdot \vec{\delta} = (\vec{\alpha} - \vec{\beta}) \cdot (2\vec{\alpha} + \vec{\beta}) = 2\vec{\alpha}^2 + \vec{\alpha} \cdot \vec{\beta} - 2\vec{\alpha} \cdot \vec{\beta} - \vec{\beta}^2 = 2|\vec{\alpha}|^2 - \vec{\alpha} \cdot \vec{\beta} - |\vec{\beta}|^2 = 2 \cdot 2^2 - 4 - 4^2 = -12.
\]

\text{\textgamma) Είναι}
\[
|\vec{\gamma}|^2 = \vec{\gamma}^2 = (\vec{\alpha} - \vec{\beta})^2 = \vec{\alpha}^2 - 2\vec{\alpha} \cdot \vec{\beta} + \vec{\beta}^2 = 12 \text{ οπότε } |\vec{\gamma}| = \sqrt{12} = 2\sqrt{3}.
\]
\text{Επίσης}
\[
|\vec{\delta}|^2 = \vec{\delta}^2 = (2\vec{\alpha} + \vec{\beta})^2 = 4\vec{\alpha}^2 + 4\vec{\alpha} \cdot \vec{\beta} + \vec{\beta}^2 = 48 \text{ οπότε } |\vec{\delta}| = \sqrt{48} = 4\sqrt{3}.
\]

\text{\textdelta) Είναι } $\sigma\upsilon\nu(\vec{\gamma} \hat{,} \vec{\delta}) = \frac{\vec{\gamma} \cdot \vec{\delta}}{|\vec{\gamma}| \cdot |\vec{\delta}|} = \frac{-12}{2\sqrt{3} \cdot 4\sqrt{3}} = -\frac{1}{2}\text{, οπότε } (\vec{\gamma} \hat{,} \vec{\delta}) = \frac{2\pi}{3}$.
