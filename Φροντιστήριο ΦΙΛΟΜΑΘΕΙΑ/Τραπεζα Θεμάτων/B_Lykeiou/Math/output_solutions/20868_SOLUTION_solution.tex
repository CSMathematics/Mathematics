\lysh{20868_SOLUTION}{1}
\textbf{ΛΥΣΗ}
\begin{enumerate}[α)]
    \item Αφού η ευθεία ($\varepsilon$) διέρχεται από τα σημεία $A, B$ έχει κλίση $\lambda = \frac{y_B-y_A}{x_B-x_A} = \frac{2-1}{3-1} = \frac{1}{2}$.
    \item Ως γνωστόν, η εξίσωση ευθείας που διέρχεται από το σημείο $M(x_0, y_0)$ και έχει κλίση $\lambda$ είναι η $y - y_0 = \lambda(x - x_0)$.
    Επομένως, η εξίσωση της ευθείας ($\varepsilon$), η οποία διέρχεται από το σημείο $A(1,1)$ και έχει κλίση $\lambda = \frac{1}{2}$, είναι: $y - 1 = \frac{1}{2}(x - 1) \Leftrightarrow y = \frac{1}{2}x + \frac{1}{2}$.
    \item Επειδή το σημείο $\Gamma$ της ευθείας ($\varepsilon$) ανήκει και στον άξονα $x'x$, είναι της μορφής $\Gamma(x, 0)$.
\end{enumerate}
Έτσι έχουμε:
\[ 0 = \frac{1}{2}x + \frac{1}{2} \Leftrightarrow x + 1 = 0 \Leftrightarrow x = -1 \]
Επομένως, το ζητούμενο σημείο είναι το $\Gamma(-1,0)$.
\end{lysh}
