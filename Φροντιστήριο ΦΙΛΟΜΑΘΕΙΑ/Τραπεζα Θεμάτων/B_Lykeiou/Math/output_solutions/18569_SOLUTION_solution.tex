\begin{enumerate}[\upshape α)]
    \item
    \begin{enumerate}[\upshape i.]
        \item Τα σημεία τομής του κύκλου C με τους ημιάξονες Ox και Ox' έχουν τεταγμένη μηδέν.

        Επομένως, για $y=0$ έχουμε: $x^2 = 1 \Leftrightarrow x = \pm 1$. Άρα $A'(-1,0)$ και $A(1,0)$.
        \item Ο συντελεστής διεύθυνσης της ευθείας $\varepsilon$ που διέρχεται από το σημείο $A(1,0)$, είναι
        \[ \lambda_\varepsilon = \varepsilon\varphi 150^\circ = - \frac{\sqrt{3}}{3} . \]
        Οπότε η εξίσωση της $\varepsilon$ είναι: $y-0 = - \frac{\sqrt{3}}{3} (x-1) \Leftrightarrow y = - \frac{\sqrt{3}}{3} (x-1)$.
    \end{enumerate}
    \item
    \begin{enumerate}[\upshape i.]
        \item Αν $OK$ το απόστημα της χορδής $AB$, τότε
        \[ OK=d(O, \varepsilon)=\frac{\left|-\frac{\sqrt{3}}{3} \cdot 0-0+\frac{\sqrt{3}}{3}\right|}{\sqrt{\left(-\frac{\sqrt{3}}{3}\right)^2+1^2}} = \frac{\left|\frac{\sqrt{3}}{3}\right|}{\sqrt{\frac{3}{9}+1}} = \frac{\frac{\sqrt{3}}{3}}{\sqrt{\frac{12}{9}}} = \frac{\frac{\sqrt{3}}{3}}{\frac{2\sqrt{3}}{3}} = \frac{1}{2}. \]
        Αν $AK = \mu = \frac{AB}{2}$, με Πυθαγόρειο θεώρημα στο τρίγωνο $OAK$ έχουμε:
        \[ OK^2+\mu^2=OA^2 \Leftrightarrow \mu^2 = 1^2 - \left(\frac{1}{2}\right)^2 \Leftrightarrow \mu^2 = 1 - \frac{1}{4} \Leftrightarrow \mu^2 = \frac{3}{4} \Leftrightarrow \mu = \frac{\sqrt{3}}{2}. \]
        Άρα $AB=2\mu=\sqrt{3}$.
        \item Η γωνία $A'BA$ είναι εγγεγραμμένη που βαίνει σε ημικύκλιο, άρα $A'B \perp BA$ δηλαδή
        \[ \zeta \perp \varepsilon, \text{ οπότε } \lambda_\zeta = - \frac{1}{\lambda_\varepsilon} = - \frac{1}{-\frac{\sqrt{3}}{3}} = \frac{3}{\sqrt{3}} = \sqrt{3} \text{ και } (\zeta): y-0 = \sqrt{3}(x-(-1)) \Leftrightarrow y = \sqrt{3}(x+1). \]
    \end{enumerate}
\end{enumerate}
