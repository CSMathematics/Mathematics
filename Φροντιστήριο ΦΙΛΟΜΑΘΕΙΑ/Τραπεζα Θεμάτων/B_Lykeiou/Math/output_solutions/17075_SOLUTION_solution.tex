\lysh{17075_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\begin{enumerate}[\textalpha)]
    \item Παρατηρούμε ότι τα σημεία $A, B$ και $\Gamma$ έχουν συντεταγμένες $A(2,3)$, $B(6,0)$ και $\Gamma(2,1)$. Επομένως οι συντεταγμένες των διανυσμάτων $\vec{AB}$ και $\vec{A\Gamma}$ θα είναι
    \[
    \vec{AB} = (x_B - x_A, y_B - y_A) = (6-2, 0-3) = (4, -3) \text{ και}
    \]
    \[
    \vec{A\Gamma} = (x_\Gamma - x_A, y_\Gamma - y_A) = (2-2, 1-3) = (0, -2).
    \]

    \item Το εσωτερικό γινόμενο των διανυσμάτων $\vec{AB}$ και $\vec{A\Gamma}$ είναι $\vec{AB} \cdot \vec{A\Gamma} = x_1 \cdot x_2 + y_1 \cdot y_2$,
    όπου $(x_1, y_1)$ και $(x_2, y_2)$ είναι οι συντεταγμένες των διανυσμάτων $\vec{AB}$ και $\vec{A\Gamma}$ αντίστοιχα.
    Αντικαθιστώντας τις συντεταγμένες των διανυσμάτων $\vec{AB}$ και $\vec{A\Gamma}$, από το προηγούμενο ερώτημα, θα έχουμε
    \[
    \vec{AB} \cdot \vec{A\Gamma} = 4 \cdot 0 + (-3) \cdot (-2) = 0 + 6 = 6.
    \]
\end{enumerate}
\end{lysh}
