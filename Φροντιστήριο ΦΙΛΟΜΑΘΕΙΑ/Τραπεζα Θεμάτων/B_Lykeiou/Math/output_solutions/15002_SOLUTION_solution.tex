\lysh{15002_SOLUTION}{1}
\noindent Λύση
\begin{enumerate}[α)]
    \item Αν $\Gamma(x_\Gamma, y_\Gamma)$ τότε:
    \[
    \vec{A\Gamma} = (3,1) \Leftrightarrow (x_\Gamma - 0, y_\Gamma - 5) = (3,1) \Leftrightarrow
    \]
    \[
    x_\Gamma = 3 \text{ και } y_\Gamma - 5 = 1 \Leftrightarrow y_\Gamma = 6.
    \]
    Άρα, $\Gamma(3,6).$
    \item
    \begin{enumerate}[i.]
        \item Είναι
        \[
        \vec{\Gamma\Delta} = (4 - 3, 5 - 6) \Leftrightarrow \vec{\Gamma\Delta} = (1, -1).
        \]
        \item Έχουμε ότι:
        \[
        \det(\vec{AB}, \vec{\Gamma\Delta}) = \begin{vmatrix} 3 & -3 \\ 1 & -1 \end{vmatrix} = 3(-1) - (-3) \cdot 1 = -3 + 3 = 0.
        \]
        (Εναλλακτικά: $\vec{AB} = (3, -3) = 3(1, -1) = 3\vec{\Gamma\Delta}.$) \\
        Άρα, τα διανύσματα $\vec{AB}$ και $\vec{\Gamma\Delta}$ είναι παράλληλα.
    \end{enumerate}
\end{enumerate}
\end{lysh}
