\lysh{14953_SOLUTION}{1}
\text{ΛΥΣΗ}

\medskip

\textnormal{α) Βρίσκουμε τις συντεταγμένες των διανυσμάτων $\vec{AB}$ και $\vec{A\Gamma}$}
\[
\vec{AB} = (7 - (-2), 8 - 5) = (9,3)
\]
\[
\vec{A\Gamma} = (1 - (-2), -4 - 5) = (3, -9)
\]

\textnormal{β) Σύμφωνα με την αναλυτική έκφραση του εσωτερικού γινομένου, θα έχουμε:}
\[
\vec{AB} \cdot \vec{A\Gamma} = 9 \cdot 3 + 3 \cdot (-9) = 27 - 27 = 0.
\]

\textnormal{γ) Αν $\theta = \angle\left(\vec{AB}, \vec{A\Gamma}\right) = \text{B}\widehat{\text{A}}\text{Γ}$, γνωρίζουμε ότι $\vec{AB} \cdot \vec{A\Gamma} = |\vec{AB}||\vec{A\Gamma}|\sigma\upsilon\nu\theta$, άρα $\sigma\upsilon\nu\theta = 0.$}

\textnormal{Αλλά $0 \le \theta \le \pi$, έτσι $\theta = 90^\circ.$}
\end{lysh}
