\begin{enumerate}[label=\alph*)]
    \item Εἶναι $ \vec{\Delta E} = \vec{\Delta A} + \vec{A E} = -\frac{1}{\kappa} \vec{\alpha} + \frac{1}{\lambda} \vec{\beta} $
    \[ \vec{B \Gamma} = \vec{B A} + \vec{A \Gamma} = -\vec{\alpha} + \vec{\beta} \]
    \item
    \begin{enumerate}[label=\roman*)]
        \item Ἀν $\kappa=\lambda$ τότε $ \vec{B \Gamma} = -\vec{\alpha} + \vec{\beta} = \kappa \left(-\frac{1}{\kappa} \vec{\alpha} + \frac{1}{\kappa} \vec{\beta} \right) = \kappa \cdot \vec{\Delta E} $
        ἄρα $ \vec{B \Gamma} // \vec{\Delta E} $ καὶ $ \left|\vec{B \Gamma}\right| = \kappa \cdot \left|\vec{\Delta E}\right| $.
        \item Ἀν $\kappa=\lambda=2$ τότε τὰ σημεῖα $\Delta$ καὶ E εἶναι μέσα τῶν $AB$ καὶ $A \Gamma$ ἀντίστοιχα, ὀπότε $ \vec{B \Gamma} // \vec{\Delta E} $ καὶ $ \left|\vec{B \Gamma}\right| = 2 \cdot \left|\vec{\Delta E}\right| $, ἐπομένως $\Delta E // B \Gamma$ καὶ $B \Gamma = 2 \cdot \Delta E$.
        Ἀποδείξαμε δηλαδή ὅτι τὸ εὐθύγραμμο τμῆμα ποὺ ἑνώνει τὰ μέσα δύο πλευρῶν τριγώνου εἶναι παράλληλο μὲ τὴν τρίτη πλευρὰ τοῦ τριγώνου καὶ ἰσοῦται μὲ τὸ μισὸ τῆς τρίτης πλευρᾶς.
    \end{enumerate}
\end{enumerate}
