\textbf{ΛΥΣΗ}

\begin{enumerate}[α)]
    \item Είναι: $\vec{AB} = (x_B - x_A \, , \, y_B - y_A) = (0+2 \, , \, 8-3) = (2 \, , \, 5).$

    Επίσης: $\vec{\Gamma\Delta} = (x_\Delta - x_\Gamma \, , \, y_\Delta - y_\Gamma) = (10-5 \, , \, 5-3) = (5 \, , \, 2).$

    Οπότε: $\vec{AB}\cdot\vec{\Gamma\Delta} = (2 \, , \, 5)\cdot(5 \, , \, 2) = 2\cdot5 + 5\cdot2 = 10+10 = 20.$

    \item Αν $\omega$ είναι η γωνία που σχηματίζει το διάνυσμα $\vec{u}$ με τον άξονα $x'x$, τότε $\text{εφ}\omega = \lambda_{\vec{u}}$
    όπου $\lambda_{\vec{u}}$ είναι ο συντελεστής διεύθυνσης του διανύσματος $\vec{u}$.

    Είναι: $\vec{u} = \vec{AB} + \vec{\Gamma\Delta} = (2 \, , \, 5) + (5 \, , \, 2) = (7 \, , \, 7)$, άρα $\text{εφ}\omega = \lambda_{\vec{u}} = \frac{y_{\vec{u}}}{x_{\vec{u}}} = \frac{7}{7} = 1.$

    Αφού το διάνυσμα $\vec{u}$ έχει θετικές συντεταγμένες, θα βρίσκεται στο $1^\circ$ τεταρτημόριο.

    Έτσι από τις σχέσεις $\text{εφ}\omega = 1$ και $0 \le \omega < \pi/2$, προκύπτει ότι $\omega = \frac{\pi}{4}.$
\end{enumerate}
\end{lysh}
