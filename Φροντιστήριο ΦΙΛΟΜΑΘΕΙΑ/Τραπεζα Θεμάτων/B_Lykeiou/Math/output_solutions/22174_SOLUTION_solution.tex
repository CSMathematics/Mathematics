\begin{enumerate}[α)]
    \item Ο μεγάλος άξονας της έλλειψης είναι $2\alpha$, οπότε $2\alpha=10$, άρα $\alpha=5$.

    Η εκκεντρότητα της έλλειψης είναι $\varepsilon = \frac{\gamma}{\alpha}$, οπότε $\frac{\gamma}{\alpha} = 0,6$ ή $\gamma = 0,6 \cdot 5$ ή $\gamma = 3$.

    Όμως $\beta^2 = \alpha^2 - \gamma^2 = 5^2 - 3^2 = 25 - 9 = 16$.

    Η εξίσωση της έλλειψης δίνεται από τον τύπο $\frac{x^2}{\alpha^2} + \frac{y^2}{\beta^2} = 1$, άρα $\frac{x^2}{25} + \frac{y^2}{16} = 1$.

    \item
    \begin{enumerate}[i)]
        \item Η εφαπτομένη της έλλειψης σε σημείο $(x_1, y_1)$ δίνεται από τον τύπο $\frac{x x_1}{\alpha^2} + \frac{y y_1}{\beta^2} = 1$.

        Εφόσον το σημείο επαφής είναι το $\Gamma\left(3, \frac{16}{5}\right)$, η εξίσωση της εφαπτομένης θα γίνει
        \[ \frac{x \cdot 3}{25} + \frac{y \cdot \frac{16}{5}}{16} = 1 \Rightarrow \frac{3x}{25} + \frac{y}{5} = 1 \]
        Για να διέρχεται από το σημείο $\Delta(0,5)$ θα πρέπει οι συντεταγμένες του σημείου να την επαληθεύουν.
        \[ \frac{3 \cdot 0}{25} + \frac{5}{5} = 1 \Rightarrow 0 + 1 = 1 \]
        που σημαίνει ότι η εφαπτομένη διέρχεται από το σημείο $\Delta$.

        \item Σημεία συνάντησης των τροχιών είναι οι λύσεις του συστήματος των εξισώσεών τους, εφόσον υπάρχουν. Οι εξισώσεις είναι $\frac{x^2}{25} + \frac{y^2}{16} = 1$ και $\frac{x^2}{25} - \frac{y^2}{16} = 1$.

        Κατά μέλη προσθέτουμε τις δύο εξισώσεις:
        \[ \left(\frac{x^2}{25} + \frac{y^2}{16}\right) + \left(\frac{x^2}{25} - \frac{y^2}{16}\right) = 1 + 1 \]
        \[ \frac{2x^2}{25} = 2 \Rightarrow x^2 = 25 \Rightarrow x = \pm 5 \]
        επειδή $x > 0$. Η εξίσωση $\frac{x^2}{25} + \frac{y^2}{16} = 1$ για $x = 5$ μας δίνει:
        \[ \frac{5^2}{25} + \frac{y^2}{16} = 1 \Rightarrow 1 + \frac{y^2}{16} = 1 \Rightarrow \frac{y^2}{16} = 0 \Rightarrow y^2 = 0 \Rightarrow y = 0 \]
        Σημείο συνάντησης είναι το $(5,0)$.
    \end{enumerate}
\end{enumerate}
\end{lysh}
