\lysh{15194_SOLUTION}{1}
\text{ΛΥΣΗ}

\text{α) Θα δείξουμε ότι τα σημεία Α , Β και Γ δεν είναι συνευθειακά.}

\text{Είναι } $\lambda_{\text{AB}} = \frac{4-1}{4-1} = 1$ \text{ και } $\lambda_{\text{ΒΓ}} = \frac{1-4}{3-4} = 3$. \text{Αφού } $\lambda_{\text{AB}} \ne \lambda_{\text{ΒΓ}}$ \text{ τα σημεία Α , Β και Γ δεν είναι συνευθειακά.}

\text{β) Για το μέσον M της ΒΓ έχουμε } $x_{\text{M}} = \frac{x_{\text{B}}+x_{\text{Γ}}}{2} = \frac{4+3}{2} = \frac{7}{2}$ \text{ και } $y_{\text{M}} = \frac{y_{\text{B}}+y_{\text{Γ}}}{2} = \frac{4+1}{2} = \frac{5}{2}$. \text{Άρα} $M\left(\frac{7}{2}, \frac{5}{2}\right)$.

\text{Ακόμα } $\lambda_\varepsilon \cdot \lambda_{\text{ΒΓ}} = -1 \Leftrightarrow \lambda_\varepsilon \cdot 3 = -1 \Leftrightarrow \lambda_\varepsilon = -\frac{1}{3}$.

\text{Έχουμε } $(\varepsilon): y - \frac{5}{2} = -\frac{1}{3}\left(x - \frac{7}{2}\right) \Leftrightarrow y = -\frac{1}{3}x + \frac{11}{3}$.

\text{γ) Το σημείο K(x, y) ανήκει στην ευθεία ($\varepsilon$) αν και μόνο αν οι συντεταγμένες του επαληθεύουν την εξίσωση της, άρα } $K\left(x, -\frac{1}{3}x + \frac{11}{3}\right)$.

\text{Έχουμε}
\[
(\text{KA})=(\text{KB}) \Leftrightarrow \sqrt{(x-1)^2+\left(1+\frac{1}{3}x-\frac{11}{3}\right)^2} = \sqrt{(x-4)^2+\left(4+\frac{1}{3}x-\frac{11}{3}\right)^2}
\]
\[
(x-1)^2+\left(\frac{1}{3}x-\frac{8}{3}\right)^2 = (x-4)^2+\left(\frac{1}{3}x+\frac{1}{3}\right)^2 \Leftrightarrow (x-4)^2-(x-1)^2=\left(\frac{1}{3}x-\frac{8}{3}\right)^2-\left(\frac{1}{3}x+\frac{1}{3}\right)^2
\]
\[
-3(2x-5)=-3\left(\frac{2}{3}x-\frac{7}{3}\right) \Leftrightarrow 2x-5=\frac{2}{3}x-\frac{7}{3} \Leftrightarrow 6x-15=2x-7 \Leftrightarrow 4x=8 \Leftrightarrow x=2.
\]
\text{Οπότε } $y = -\frac{1}{3} \cdot 2 + \frac{11}{3} = 3$, \text{ δηλαδή K(2,3).}

\text{Το K(2,3) ως σημείο της μεσοκαθέτου του ΒΓ ισαπέχει από τα άκρα του Β και Γ , επιπλέον } $(\text{KA})=(\text{KB})$, \text{ άρα τελικά } $(\text{KA})=(\text{KB})=(\text{KΓ})$, \text{ οπότε το σημείο Κ είναι το κέντρο του περιγεγραμμένου κύκλου του τριγώνου ΑΒΓ.}
\end{lysh}
