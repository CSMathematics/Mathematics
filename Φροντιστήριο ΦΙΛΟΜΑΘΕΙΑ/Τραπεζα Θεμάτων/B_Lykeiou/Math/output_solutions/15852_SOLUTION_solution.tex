\lysh{15852_SOLUTION}{1}
\centering{\textbf{ΛΥΣΗ}}

\medskip

\noindent \textbf{α)} Δίνονται οι συντεταγμένες των διανυσμάτων $\vec{\alpha}=(3,2)$, $\vec{\beta}=(-2,1)$, τότε
\[
\vec{v} = 2\vec{\alpha} + 3\vec{\beta} = 2(3,2) + 3(-2,1) = (6,4) + (-6,3) = (0,7).
\]

\noindent \textbf{β)} Υπολογίζουμε το εσωτερικό γινόμενο των διανυσμάτων $\vec{\alpha}, \vec{\beta}$ και το μέτρο του διανύσματος $\vec{\alpha}$.
\begin{itemize}
    \item $\vec{\alpha} \cdot \vec{\beta} = x_1 \cdot x_2 + y_1 \cdot y_2 = 3 \cdot (-2) + 2 \cdot 1 = -6 + 2 = -4$.
    \item $|\vec{\alpha}| = \sqrt{3^2 + 2^2} = \sqrt{9+4} = \sqrt{13}$.
\end{itemize}

\noindent \textbf{γ)} Το εσωτερικό γινόμενο των διανυσμάτων $\vec{\alpha}, \vec{v}$ υπολογίζεται
\[
\vec{\alpha} \cdot \vec{v} = \vec{\alpha} \cdot (2\vec{\alpha} + 3\vec{\beta}) = 2\vec{\alpha} \cdot \vec{\alpha} + 3\vec{\alpha} \cdot \vec{\beta} = 2|\vec{\alpha}|^2 + 3\vec{\alpha} \cdot \vec{\beta} = 2 \cdot 13 + 3(-4) = 26 - 12 = 14.
\]
\end{lysh}
