\lysh{20869_SOLUTION}{1}
\textbf{ΛΥΣΗ}

\begin{enumerate}[a)]
    \item Η υπερβολή $\frac{x^2}{\alpha^2} - \frac{y^2}{\beta^2} = 1$, με $\beta = \sqrt{\gamma^2 - \alpha^2}$, έχει εστίες τις $E'(-\gamma,0)$, $E(\gamma,0)$ και ασύμπτωτες τις ευθείες $\epsilon_1: y = \frac{\beta}{\alpha}x$, $\epsilon_2: y = -\frac{\beta}{\alpha}x$.
    
    Η υπερβολή $C: x^2 - y^2 = 1$ είναι ισοσκελής, με $\alpha = \beta = 1$.
    
    Επιπλέον $\beta = \sqrt{\gamma^2 - \alpha^2} \Leftrightarrow 1 = \sqrt{\gamma^2 - 1} \Leftrightarrow \gamma^2 = 2 \xrightarrow{\gamma>0} \gamma = \sqrt{2}$.
    
    Επομένως, οι εστίες της $C$ είναι τα σημεία $E'(-\sqrt{2},0)$, $E(\sqrt{2},0)$ και ασύμπτωτες, οι ευθείες $\epsilon_1: y = x$, $\epsilon_2: y = -x$.

    \item 
    \begin{enumerate}[i)]
        \item Η εφαπτομένη της υπερβολής $\frac{x^2}{\alpha^2} - \frac{y^2}{\beta^2} = 1$ στο σημείο της $M(x_1,y_1)$ είναι η ευθεία
        \[ \frac{x \cdot x_1}{\alpha^2} - \frac{y \cdot y_1}{\beta^2} = 1. \]
        Η εφαπτομένη της $C$ στο σημείο $A(1,0)$ είναι η $\zeta: x \cdot 1 - y \cdot 0 = 1 \Leftrightarrow x = 1$.

        \item Το σημείο $\Gamma$ προκύπτει από την επίλυση του συστήματος $(\Sigma): \begin{cases} \epsilon_1: y = x \\ \zeta: x = 1 \end{cases} \Leftrightarrow \begin{cases} y = 1 \\ x = 1 \end{cases}$
        
        Επομένως $\Gamma(1,1)$.
    \end{enumerate}
\end{enumerate}
