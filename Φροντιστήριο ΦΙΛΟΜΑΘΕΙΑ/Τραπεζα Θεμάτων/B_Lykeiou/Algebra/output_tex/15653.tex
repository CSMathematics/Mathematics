\begin{ylh}{1}
\item 4.2 Διαίρεση πολυωνύμων
\item 4.3 Πολυωνυμικές εξισώσεις και ανισώσεις
\end{ylh}
Δίνεται το πολυώνυμο $P(x) = x^3 + x^2 + 2x + 2$.
\begin{alist}
\item
\begin{rlist}
\item Να κάνετε τη διαίρεση του $P(x)$ με το $(x + 1)$ .
\monades{8}
\item Να γράψετε την ταυτότητα της διαίρεσης $P(x) : (x + 1)$.
\monades{5}
\end{rlist}
\item Aν $P(x) = (x + 1)(x^2 + 2)$, να λύσετε την ανίσωση $P(x) < 0$.
\monades{12}
\end{alist}
