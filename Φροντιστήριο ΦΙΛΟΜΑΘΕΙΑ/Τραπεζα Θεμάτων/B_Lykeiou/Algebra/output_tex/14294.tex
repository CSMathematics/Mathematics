Στο σχήμα δίνονται οι γραφικές παραστάσεις μιας παραβολής $f(x) = \alpha x^{2} + \beta x + \gamma$ και της ευθείας $g(x) = - x + 2$.
\begin{center}
\begin{tikzpicture}[scale=0.8, >=latex]
    \draw[very thin, color=gray!30] (-5.5,-3.5) grid (5.5,7.5);
    \draw[->] (-5.5,0) -- (5.5,0) node[right] {$x$};
    \draw[->] (0,-3.5) -- (0,7.5) node[above] {$y$};
    \foreach \x in {-5,-4,-3,-2,-1,1,2,3,4,5}
        \draw (\x,2pt) -- (\x,-2pt) node[below, font=\small] {\x};
    \foreach \y in {-3,-2,-1,1,2,3,4,5,6,7}
        \draw (2pt,\y) -- (-2pt,\y) node[left, font=\small] {\y};
    \node[below left, font=\small] at (0,0) {$0$};
    % Parabola f(x) = 0.5*x^2 - 2
    \draw[very thick, color=maincolor, samples=100, domain=-4.2:4.2] plot (\x, {0.5*\x*\x - 2}) node[right] {$f$};
    % Line g(x) = -x + 2
    \draw[very thick, color=secondarycolor, samples=2, domain=-5:5] plot (\x, {-\x + 2}) node[above right] {$g$};
    % Points
    \filldraw (2,0) circle (2pt) node[below right] {$A(2,0)$};
    \filldraw (-2,0) circle (2pt) node[above left] {$B$};
    \filldraw (0,-2) circle (2pt) node[below right] {$\Gamma$};
    \filldraw (-4,6) circle (2pt) node[above right] {$\Delta(-4,6)$};
\end{tikzpicture}
\end{center}
\begin{alist}
\item Δεδομένου ότι η παραβολή διέρχεται από τα σημεία $Α,\ Β,\ \Gamma$, να βρείτε τις τιμές των $\alpha,\ \beta,\ \gamma.$
\monades{8}
\item Αν $\alpha = \frac{1}{2},\ \beta = 0$ και $\gamma = - 2$, να βρείτε αλγεβρικά τις συντεταγμένες των κοινών σημείων της ευθείας και της παραβολής.
\monades{8}
\item Αν μετατοπίσουμε την παραβολή κατά $4,5$ μονάδες προς τα πάνω, να δείξετε ότι η ευθεία και η παραβολή θα έχουν ένα μόνο κοινό σημείο.
\monades{9}
\end{alist}
