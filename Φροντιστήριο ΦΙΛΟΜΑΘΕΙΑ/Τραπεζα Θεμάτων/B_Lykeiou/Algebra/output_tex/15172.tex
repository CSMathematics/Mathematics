\item 3.3 Αναγωγή στο 1o Τεταρτημόριο
\item 3.4 Οι τριγωνομετρικές συναρτήσεις
\end{ylh}
Δίνεται η συνάρτηση
\begin{equation*}f(x) = 4\hm (11\pi - x),\ \ \ x\epsilon\mathbb{R}\end{equation*}
\begin{alist}
\item Να δείξετε ότι:
\begin{rlist}
\item $\ \hm (11\pi - x) = \hm  x,\ \ x\epsilon\mathbb{R}$.$\ $
\monades{6}
\item $f(x) = 4\hm  x,\ \ \ x\epsilon\mathbb{R.}$
\monades{4}
\end{rlist}
\item Να σχεδιάσετε τη γραφική παράσταση της συνάρτηση $f(x) = 4\hm  x$,$\ $όταν $x\epsilon\lbrack 0,2\pi\rbrack.$
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle, xlabel = {$x$}, ylabel = {$y$},
    xmin = -0.5, xmax = 7.5, ymin = -4.5, ymax = 4.5,
    xtick = {1.57, 3.14, 4.71, 6.28},
    xticklabels = {$\pi/2$, $\pi$, $3\pi/2$, $2\pi$},
    ytick = {-4,-2,2,4},
    grid = both, grid style = {gray!30},
    samples = 150, width = 10cm, height = 7cm,
    tick label style = {font=\small},
]
    \addplot[very thick, maincolor, domain=0:2*pi] {4*sin(deg(x))};
\end{axis}
\end{tikzpicture}
\end{center}
\monades{15}
\end{alist}
