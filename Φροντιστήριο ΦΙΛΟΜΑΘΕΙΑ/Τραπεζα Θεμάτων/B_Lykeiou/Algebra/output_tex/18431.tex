\begin{ylh}{1}
\item 1.1 Γραμμικά Συστήματα
\end{ylh}
Δίνεται το σύστημα \(\left\{ \begin{array}{r}
3x + y = 11 \\
6x + ky = 8
\end{array} \right.\ \) με αγνώστους $x,y$ και $k$ παράμετρος.
\begin{alist}
\item Να λύσετε το σύστημα όταν $k = 2$.
\monades{12}
\item Να λύσετε το σύστημα όταν $k = 1$.
\monades{13}
\end{alist}
