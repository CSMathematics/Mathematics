\begin{alist}
\item Να αποδείξετε ότι δεν υπάρχει γωνία θ ώστε $\eta\mu\theta = \frac{1}{2}$ και $\sigma\upsilon\nu\theta = \frac{1}{2}$.

\monades{12}

\monades{13}
\end{alist}
