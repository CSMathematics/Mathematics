\begin{alist}
\item Να υπολογίσετε την τιμή του πραγματικού αριθμού $\alpha$\emph{.}

\monades{8}

\item Για $\alpha = 2$,

\end{alist}
\begin{alist}
\item
  να υπολογίσετε το υπόλοιπο της διαίρεσης του $P(x)$ με το $x - 1.$

\monades{2}


\end{alist}

\end{alist}
