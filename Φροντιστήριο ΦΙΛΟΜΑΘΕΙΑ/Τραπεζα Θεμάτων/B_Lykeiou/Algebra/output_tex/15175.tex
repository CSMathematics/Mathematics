\begin{ylh}{1}
\item 4.3 Πολυωνυμικές εξισώσεις και ανισώσεις
\end{ylh}
Δίνεται το πολυώνυμο $P(x) = x^{3} - x^{2} + \ x - 1$.
\begin{alist}
\item Να αποδείξετε ότι το $1$ είναι μία ρίζα του πολυωνύμου.
\monades{5}
\item Να αποδείξετε ότι $P(x) = (x - 1\ ) \cdot (x^{2} + 1)$.
\monades{10}
\item Να λύσετε την εξίσωση $P(x) = 0$.
\monades{10}
\end{alist}
