\begin{center}
\begin{tikzpicture}[scale=1.5, >=latex]
    % Circle center K
    \coordinate (K) at (0,0);
    \draw[thick] (K) circle (1); % radius 1 scaled, real is 10cm
    
    % Radius KB
    \draw[thick] (K) -- (0:1) node[midway, below] {$10\text{cm}$};
    \filldraw (0:1) circle (1pt) node[right] {$B$};
    
    % Arc AB length 12 => angle = 1.2 rad = 68.75 deg
    \draw[very thick, red] (0:1) arc (0:68.75:1) node[midway, right] {$12\text{cm}$};
    \draw[thick] (K) -- (68.75:1);
    \filldraw (68.75:1) circle (1pt) node[above right] {$A$};
    
    % Angle omega
    \draw[->] (0.3,0) arc (0:68.75:0.3);
    \node at (34:0.45) {$\omega$};
    
    \filldraw (K) circle (1.5pt) node[below left] {$K$};
\end{tikzpicture}
\end{center}

Δίνεται ο κύκλος του παραπάνω σχήματος με κέντρο $K$ και ακτίνα $10\text{cm}$. Επίσης δίζεται το τόξο $\wideparen{AB}$ με μήκος $12\text{cm}$ και η αντίστοιχη επίκεντρη γωνία $\omega$.

\begin{alist}
\item \begin{rlist}
\item Να αιτιολογήσετε γιατί το μέτρο της γωνίας $\omega$ είναι $1,2\text{rad}$. \monades{6}
\item Με χρήση του αi) ερωτήματος, να αιτιολογήσετε γιατί η γωνία $\omega$ είναι οξεία. \monades{6}
\end{rlist}
\item Αν $\sigma\upsilon\nu\omega = \frac{9}{25}$, να βρείτε το $\eta\mu\omega$. (Δίνεται ότι $\sqrt{544} = 4\sqrt{34}$). \monades{13}
\end{alist}
