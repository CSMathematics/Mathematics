\begin{ylh}{1}
\item 3.1 Τριγωνομετρικοί Αριθμοί Γωνίας
\end{ylh}
\begin{center}
\begin{tikzpicture}[scale=2, >=latex]
    % Grid
    \draw[very thin, color=gray!30, step=0.2] (-1.2,-1.2) grid (1.2,1.2);
    % Unit Circle
    \draw[thick] (0,0) circle (1);
    % Axes
    \draw[->] (-1.3,0) -- (1.3,0) node[right] {$x$};
    \draw[->] (0,-1.3) -- (0,1.3) node[above] {$y$};
    % Point A
    \coordinate (A) at (1,0);
    \filldraw (A) circle (1.5pt) node[above right] {$A$};
    % Markers on axes
    \foreach \x in {-1,1} \draw (\x,2pt) -- (\x,-2pt) node[below] {\x};
    \foreach \y in {-1,1} \draw (2pt,\y) -- (-2pt,\y) node[left] {\y};
    \node[below left] at (0,0) {$0$};
\end{tikzpicture}
\end{center}
Στον τριγωνομετρικό κύκλο έχει σημειωθεί το σημείο $A$.
\begin{alist}
\item Να μεταφέρετε το σχήμα στην κόλλα σας και να τοποθετήσετε κατά προσέγγιση στον τριγωνομετρικό κύκλο σημεία $B, \Gamma, \Delta$ ώστε να δημιουργηθούν τόξα $\widearc{AB} = 1\text{rad}$, $\widearc{A\Gamma} = 2\text{rad}$ και $\widearc{A\Delta} = 4\text{rad}$. \monades{13}
\item Για κάθε ένα τόξο του α) ερωτήματος να αποφανθείτε αν το συνημίτονο της αντίστοιχης επίκεντρης γωνίας είναι θετικός ή αρνητικός αριθμός. Να αιτιολογήσετε τις απαντήσεις σας. \monades{12}
\end{alist}
