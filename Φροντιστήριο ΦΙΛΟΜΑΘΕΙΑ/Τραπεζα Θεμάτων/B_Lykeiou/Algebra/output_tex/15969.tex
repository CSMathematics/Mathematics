Δίνεται η συνάρτηση $f(x) = 2\sigma\upsilon\nu(13\pi + x) - 2\eta\mu(\ \frac{\pi}{2} - x).$

\begin{alist}
\item Να δείξετε ότι $\sigma\upsilon\nu(13\pi + x) = - \sigma\upsilon\nu x$.

\monades{5}

\item Να δείξετε ότι $f(x) = - 4\sigma\upsilon\nu x$.

\monades{8}

\item Να λύσετε την εξίσωση $f(x) = - 2$.

\monades{12}
\end{alist}
