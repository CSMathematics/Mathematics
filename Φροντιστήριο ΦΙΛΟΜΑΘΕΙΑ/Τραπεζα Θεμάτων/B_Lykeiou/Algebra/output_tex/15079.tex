\begin{ylh}{1}
\item 3.1 Τριγωνομετρικοί Αριθμοί Γωνίας
\end{ylh}
Στον παρακάτω τριγωνομετρικό κύκλο σχεδιάσαμε γωνία $\widehat{\omega} = B\widehat{O}A$.
\begin{center}
\begin{tikzpicture}[scale=2, >=latex]
    \draw[thick] (0,0) circle (1);
    \draw[->] (-1.2,0) -- (1.2,0) node[right] {$x$};
    \draw[->] (0,-1.2) -- (0,1.2) node[above] {$y$};
    \node[below left] at (0,0) {$O$};
    \filldraw (1,0) circle (0.5pt) node[below right] {$B(1,0)$};
    % Angle omega: cos(omega) = 3/5 = 0.6, sin(omega) = 4/5 = 0.8
    \coordinate (A) at (0.6,0.8);
    \draw[very thick] (0,0) -- (A) node[above right] {$A$};
    \draw[dashed] (A) -- (0.6,0) node[below, font=\tiny] {$3/5$};
    \draw[dashed] (A) -- (0,0.8) node[left, font=\tiny] {$4/5$};
    % Angle arc omega
    \draw[->, thick, maincolor] (0.3,0) arc (0:53.13:0.3);
    \node[maincolor] at (0.4,0.2) {$\omega$};
    % Point Gamma diametrically opposite to A
    \coordinate (G) at (-0.6,-0.8);
    \draw[very thick] (0,0) -- (G) node[below left] {$\Gamma$};
    % Angle phi = B O Gamma (omega + 180)
    \draw[->, thick, secondarycolor] (0.2,0) arc (0:233.13:0.2);
    \node[secondarycolor] at (-0.3,-0.1) {$\varphi$};
\end{tikzpicture}
\end{center}
\begin{alist}
\item Με βάση το σχήμα, να αιτιολογήσετε γιατί $\syn \omega = \frac{3}{5}$.
\monades{8}
\item Η προέκταση του τμήματος $AO$ τέμνει τον τριγωνομετρικό κύκλο στο σημείο $\Gamma$, όπως φαίνεται στο σχήμα.
\begin{rlist}
\item Να εκφράσετε την γωνία $\widehat{\varphi} = B\widehat{O}\Gamma$ με την βοήθεια της γωνίας $\widehat{\omega}$.
\monades{8}
\item Με την βοήθεια του τριγωνομετρικού κύκλου ή με οποιονδήποτε άλλο τρόπο θέλετε να υπολογίσετε το $\syn \varphi$.
\monades{9}
\end{rlist}
\end{alist}
