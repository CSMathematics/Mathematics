\item 3.4 Οι τριγωνομετρικές συναρτήσεις
\end{ylh}
Δίνεται η συνάρτηση $f(x) = -3\syn  2x, x \in \mathbb{R}$.
\begin{alist}
\item Να βρείτε την περίοδο, τη μέγιστη και την ελάχιστη τιμή της συνάρτησης $f$.
\monades{12}
\item Να συμπληρώσετε τον παρακάτω πίνακα τιμών και να παραστήσετε γραφικά τη συνάρτηση $f$ σε διάστημα μιας περιόδου.
\monades{13}
\begin{center}
\begin{mytblr}{}
$x$ & $0$ & $\frac{\pi}{4}$ & $\frac{\pi}{2}$ & $\frac{3\pi}{4}$ & $\pi$ \\
$2x$ & & & & & \\
$\syn  2x$ & & & & & \\
$f(x) = -3\syn  2x$ & & & & & \\
\end{mytblr}
\end{center}
\end{alist}
