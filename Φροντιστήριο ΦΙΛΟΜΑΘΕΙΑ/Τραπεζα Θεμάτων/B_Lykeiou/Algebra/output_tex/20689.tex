\begin{ylh}{1}
\item 4.4 Εξισώσεις και ανισώσεις που ανάγονται σε πολυωνυμικές
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
\begin{alist}
\item Να λυθεί η ανίσωση $\frac{x - 2}{x + 1} > 0$.
\monades{07}
\item Δίνεται η συνάρτηση $f(x) = \left(\frac{\alpha - 2}{\alpha + 1}\right)^x$, με $x \in \mathbb{R}$.
\begin{rlist}
\item
  Να βρεθούν οι τιμές του $\alpha \in \mathbb{R}$, για τις οποίες η συνάρτηση $f$ είναι καλώς ορισμένη.
\monades{03}
\item
  Για ποιες τιμές του $\alpha \in \mathbb{R}$ η συνάρτηση $f$ είναι γνησίως φθίνουσα;
\monades{10}
\item
  Να αποδείξετε ότι δεν υπάρχουν τιμές του πραγματικού αριθμού $\alpha$ για τις οποίες η συνάρτηση $f$ είναι σταθερή.
\monades{05}
\end{rlist}
\end{alist}
