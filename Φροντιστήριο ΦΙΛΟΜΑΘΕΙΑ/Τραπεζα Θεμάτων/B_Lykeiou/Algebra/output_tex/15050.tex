\begin{alist}
\item Να βρείτε τη μέγιστη και την ελάχιστη τιμή της.

\monades{5}

\item Να συγκρίνετε τους αριθμούς $f\left(\frac{2\pi}{5}\right)$ και $f\left(\frac{\pi}{3}\right)$.

\monades{6}

\item Να σχεδιάσετε τη γραφική της παράσταση, στο διάστημα $[0, 2\pi]$.
