\begin{alist}
\item Να δείξετε ότι το $- 2$ δεν είναι ρίζα του πολυωνύμου.
\monades{08}
\item Να βρείτε το πηλίκο της διαίρεσης $P(x):(x + 2)$
\monades{10}
\item Να γράψετε την ταυτότητα της διαίρεσης $P(x):(x + 2)$.
\monades{07}
\end{alist}
