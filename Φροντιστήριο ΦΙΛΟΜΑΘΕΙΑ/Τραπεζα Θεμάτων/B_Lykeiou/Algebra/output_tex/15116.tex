\item 2.1 Μονοτονία-Ακρότατα-Συμμετρίες Συνάρτησης
\end{ylh}

\begin{alist}
\item Να μεταφέρετε το σχήμα στην κόλλα σας και να χαράξετε τα υπόλοιπα τμήματα της γραφικής παράστασης της $f$.
\monades{8}
\item Να βρείτε
\begin{rlist}
\item τα διαστήματα στα οποία η συνάρτηση $f$ είναι γνησίως φθίνουσα. Να αιτιολογήσετε την απάντησή σας.
\monades{8}
\item τη μέγιστη και την ελάχιστη τιμή της $f$ καθώς και τις θέσεις των ακρότατων αυτών.
\monades{9}
\end{rlist}
\end{alist}
