ΘEMA 2
Δίνεται η συνάρτηση $f(x) = \hm (\pi - x) + \syn \left(\frac{\pi}{2} - x\right)$, $x \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι $f(x) = 2\hm  x$, για κάθε $x \in \mathbb{R}$.
\monades{12}
\item
\begin{rlist}
\item Να βρείτε την περίοδο καθώς και τη μέγιστη και ελάχιστη τιμή της $f$.
\monades{6}
\item Να κάνετε τη γραφική παράσταση της $f$ για $0 \le x \le 2\pi$.
\monades{7}
\end{rlist}
\end{alist}
