\item 5.2 Λογάριθμοι
\item 5.3 Λογαριθμική συνάρτηση
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \ln|x|$.
\begin{alist}
\item Να βρείτε το πεδίο ορισμού της $f$\emph{\textbf{.}}
\monades{3}
\item Να προσδιορίσετε το είδος της συμμετρίας της γραφικής παράστασης της $f$.
\monades{6}
\item Να κάνετε τη γραφική παράσταση της συνάρτησης $f$.
\monades{6}
\item Να αποδείξετε ότι η συνάρτηση $Ε(x) = \frac{1}{2}(x - 1)\ln x,\ \ \text{με  }x \in (0,1) \cup (1, + \infty)$ μπορεί να περιγράψει το εμβαδόν του τριγώνου $ΑΒ\Gamma$, όπου $Α(1,0),\ \ B(x,0)\text{  και}\ \ \Gamma(x,lnx)$.
\monades{10}
\end{alist}
