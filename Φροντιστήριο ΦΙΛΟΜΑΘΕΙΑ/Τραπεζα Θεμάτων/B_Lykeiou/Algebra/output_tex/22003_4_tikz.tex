\begin{tikzpicture}[scale=1.5]
    % Grid
    \draw[help lines, gray!30] (0,-1) grid[xstep=0.25, ystep=1] (1,1);
    
    % Axes
    \draw[->, thick] (-0.2,0) -- (1.2,0) node[right] {$x$};
    \draw[->, thick] (0,-1.5) -- (0,1.5) node[above] {$y$};
    
    % Function y = sin(4*pi*x)
    \draw[ultra thick, blue, domain=0:1.05, samples=150] plot (\x, {sin(4*\x*180)});
    
    % Labels
    \node[below left] at (0,0) {$0$};
    \foreach \x in {0.25, 0.5, 0.75, 1} \draw (\x, 0.05) -- (\x, -0.05) node[below] {$\x$};
    \draw (0.05, 1) -- (-0.05, 1) node[left] {$1$};
    \draw (0.05, -1) -- (-0.05, -1) node[left] {$-1$};
