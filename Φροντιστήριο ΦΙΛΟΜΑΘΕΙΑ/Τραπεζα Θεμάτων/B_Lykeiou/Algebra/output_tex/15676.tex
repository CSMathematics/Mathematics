Δίνεται η συνάρτηση $f(x) = \ln(e^x - 1)$.
\begin{alist}
\item Να βρείτε το πεδίο ορισμού της $f$.
\monades{7}
\item Να βρείτε τα σημεία τομής της γραφικής παράστασης της $f$ με τον άξονα $x'x$.
\monades{8}
\item Να βρείτε για ποιες τιμές του $x$ η γραφική παράσταση της $f$ είναι κάτω από τον $x'x$.
\monades{10}
\end{alist}
