\begin{ylh}{1}
\item 3.4 Οι τριγωνομετρικές συναρτήσεις
\end{ylh}
Στο παρακάτω σύστημα συντεταγμένων έχουμε σχεδιάσει δύο γραφικές παραστάσεις $C_1$ και $C_2$ για $x \in [0, 2\pi]$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle, xlabel = {$x$}, ylabel = {$y$},
    xmin = -0.5, xmax = 7, ymin = -1.5, ymax = 1.8,
    xtick = {0.785, 1.571, 2.356, 3.141, 3.927, 4.712, 5.498, 6.283},
    xticklabels = {$\pi/4$, $\pi/2$, $3\pi/4$, $\pi$, $5\pi/4$, $3\pi/2$, $7\pi/4$, $2\pi$},
    ytick = {-1,1},
    grid = both, grid style = {gray!30},
    samples = 150, width = 12cm, height = 7cm,
    tick label style = {font=\small},
]
    \addplot[very thick, maincolor, domain=0:2*pi] {cos(deg(x))} node[right, pos=1] {$C_1$};
    \addplot[very thick, secondarycolor, domain=0:2*pi] {sin(deg(x))} node[right, pos=1] {$C_2$};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{alist}
\item Αν οι γραφικές παραστάσεις είναι των συναρτήσεων $f(x) = \syn  x$ και $g(x) = \hm  x$ για $x \in [0, 2\pi]$, ποια από τις $C_1$, $C_2$ είναι η γραφική παράσταση της $f(x) = \syn  x$ και ποια της $g(x) = \hm  x$; Να αιτιολογήσετε την απάντησή σας.
\monades{10}
\item Με την βοήθεια του σχήματος να λύσετε την εξίσωση $\hm  x = \syn  x$ στο διάστημα $[0, 2\pi]$.
\monades{15}
\end{alist}
