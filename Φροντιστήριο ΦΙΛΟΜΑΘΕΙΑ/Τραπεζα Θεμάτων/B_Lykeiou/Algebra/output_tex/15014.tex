\Askhsh{15014}{4}
\begin{ylh}{1}
\item 3.4 Οι τριγωνομετρικές συναρτήσεις
\item 3.5 Βασικές τριγωνομετρικές εξισώσεις
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \alpha \cdot \hm (\beta x)$, με $\alpha, \beta$ ακέραιους θετικούς αριθμούς.\\
\begin{alist}
\item Να βρείτε την τιμή του $\alpha$, αν η μέγιστη τιμή της συνάρτησης είναι $2$.
\monades{6}
\item Αν $\alpha = 2$, να δείξετε ότι η μικρότερη τιμή του $\beta$ για την οποία είναι $f\left(\frac{\pi}{16}\right) = 2$ είναι $\beta = 8$.
\monades{10}
\item Αν $\alpha = 2$ και $\beta = 8$, να λύσετε την εξίσωση $f(x) = 1$ στο διάστημα $\left[0, \frac{\pi}{2}\right]$.
\monades{9}
\end{alist}
