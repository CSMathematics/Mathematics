Στο παρακάτω σχήμα δίνεται η γραφική παράσταση της συνάρτησης $f(x) = x^2 + 1$ και η γραφική παράσταση μιας συνάρτησης $g(x)$ με $x \in \mathbb{R}$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle, xlabel = {$x$}, ylabel = {$y$},
    xmin = -2.5, xmax = 4.5, ymin = -0.5, ymax = 5.5,
    xtick = {-2,-1,1,2,3,4}, ytick = {1,2,3,4,5},
    grid = both, grid style = {gray!30},
    samples = 150, width = 10cm, height = 7cm,
    tick label style = {font=\small},
]
    \addplot[very thick, maincolor, dashed, domain=-2.1:2.1] {x^2 + 1} node[above right, pos=1] {$C_f$};
    \addplot[very thick, secondarycolor, domain=-0.1:4.1] {(x-2)^2 + 1} node[above left, pos=1] {$C_g$};
    \filldraw (axis cs:0,1) circle (1.5pt);
    \filldraw (axis cs:2,1) circle (1.5pt);
\end{axis}
\end{tikzpicture}
\end{center}

\item Έχει η $f$ μέγιστη τιμή ή ελάχιστη; Να αιτιολογήσετε την απάντησή σας. \monades{7}

\item \begin{rlist}

\end{rlist}
