\begin{ylh}{1}
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
Στο σχήμα φαίνεται η γραφική παράσταση της συνάρτησης $f(x) = 2^{x},\ \ x\epsilon\mathbb{R}$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle, xlabel = {$x$}, ylabel = {$y$},
    xmin = -4, xmax = 3, ymin = -1, ymax = 5,
    xtick = {-4,-3,-2,-1,1,2,3}, ytick = {1,2,3,4,5},
    grid = both, grid style = {gray!30},
    samples = 150, width = 10cm, height = 7cm,
    tick label style = {font=\small},
]
    \addplot[very thick, maincolor, domain=-4:2.3] {pow(2,x)} node[above right, pos=1] {$C_f$};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{alist}
\item Να λύσετε την εξίσωση $2^{x} - 1 = 0$.
\monades{10}
\item Με τη βοήθεια της γραφικής παράστασης της συνάρτησης $f$, να σχεδιάσετε την γραφική παράσταση της συνάρτησης $g(x) = 2^{x} - 1,\ \ x\epsilon\mathbb{R}$.
\monades{10}
\item Να βρείτε τα σημεία τομής της γραφικής παράστασης της $g$ με τους άξονες συντεταγμένων.
\monades{05}
\end{alist}
