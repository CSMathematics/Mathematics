\item Να μεταφέρετε τον κύκλο στην κόλλα σας και να φέρετε σε αυτόν τις τελικές πλευρές των γωνιών $3\pi + \theta$ και $4\pi - \theta$. \monades{9}
\item \begin{rlist}
\item Να αιτιολογήσετε γιατί $\hm \theta = 0,4$. \monades{7}

\end{rlist}
\end{alist}
