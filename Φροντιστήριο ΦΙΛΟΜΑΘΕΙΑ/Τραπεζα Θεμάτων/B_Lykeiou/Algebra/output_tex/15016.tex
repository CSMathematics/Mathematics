\begin{alist}
\item Να αιτιολογήσετε γιατί το ζεύγος $(0, 4)$ δεν αποτελεί λύση του παραπάνω συστήματος.

\monades{8}

\item Να λύσετε το παραπάνω σύστημα.

\monades{10}

\item Να βρείτε τις συντεταγμένες του σημείου τομής των ευθειών $(\varepsilon_1): 3x + 2y = 8$ και $(\varepsilon_2): 2x - y = 3$.

\monades{7}
\end{alist}
