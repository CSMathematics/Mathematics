\begin{ylh}{1}
\item 2.1 Μονοτονία-Ακρότατα-Συμμετρίες Συνάρτησης
\end{ylh}
Στο παρακάτω σχήμα δίνεται η γραφική παράσταση μιας συνάρτησης $f$ με πεδίο ορισμού το διάστημα $[-2, 2]$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = {$x$},
    ylabel = {$y$},
    xmin = -2.5, xmax = 2.5,
    ymin = -2.5, ymax = 2.5,
    xtick = {-2,-1,1,2},
    ytick = {-2,-1,1,2},
    grid = both,
    grid style = {gray!30},
    samples = 150,
    width = 10cm, height = 7cm,
    tick label style = {font=\small},
    every axis x label/.style = {at={(current axis.right of origin)}, anchor=north west},
    every axis y label/.style = {at={(current axis.above origin)}, anchor=south east},
]
    \addplot[very thick, maincolor, domain=-2:2] {x^3 - 3*x};
    \addplot[only marks, mark=*, mark size=2pt, maincolor] coordinates {(-1,2) (1,-2)};
    \addplot[only marks, mark=o, mark size=2pt, maincolor] coordinates {(-2,-2) (2,2)};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{alist}
\item Να εξετάσετε αν η $f$ είναι άρτια ή περιττή και να αιτιολογήσετε την απάντησή σας.
\monades{7}
\item Να γράψετε τα διαστήματα στα οποία η $f$ είναι γνησίως αύξουσα.
\monades{8}
\item Να βρείτε τη μέγιστη και την ελάχιστη τιμή της $f$ καθώς και τις θέσεις των ακρότατων αυτών.
\monades{10}
\end{alist}
