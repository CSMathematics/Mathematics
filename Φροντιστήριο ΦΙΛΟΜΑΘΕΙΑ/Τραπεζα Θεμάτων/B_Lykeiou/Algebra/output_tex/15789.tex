Δίνεται η γραφική παράσταση της συνάρτησης $f(x) = \hm  x$ με $x \in [0, \pi]$.
\begin{center}
\begin{tikzpicture}[scale=1.5, >=latex]
    \draw[very thin, color=gray!30, xstep=pi/4, ystep=1] (-0.5,-1.5) grid (4.5,1.5);
    \draw[->] (-0.5,0) -- (5,0) node[right] {$x$};
    \draw[->] (0,-1.5) -- (0,2) node[above] {$y$};
    \foreach \x/\xtext in {0.785/\frac{\pi}{4}, 1.571/\frac{\pi}{2}, 2.356/\frac{3\pi}{4}, 3.141/\pi}
        \draw (\x,2pt) -- (\x,-2pt) node[below, font=\small, fill=white] {$\xtext$};
    \foreach \y in {-1,1}
        \draw (2pt,\y) -- (-2pt,\y) node[left, font=\small] {\y};
    \node[below left, font=\small] at (0,0) {$0$};
    % f(x) = sin(x) on [0, pi]
    \draw[very thick, color=maincolor, samples=100, smooth, domain=0:pi] plot (\x, {sin(deg(\x))}) node[above right] {$f$};
\end{tikzpicture}
\end{center}

\item Να μεταφέρετε στην κόλλα σας το σχήμα και μετατοπίζοντας κατάλληλα την $f$ να σχεδιάσετε την συνάρτηση $g(x) = f\left(x - \frac{\pi}{2}\right)$. \monades{8}
\item Ποιος είναι ο τύπος της $g$ και σε ποιο διάστημα ορίζεται; \monades{8}
\end{rlist}
\item Nα λύσετε την εξίσωση $f(x) = g(x)$. \monades{9}
\end{alist}
