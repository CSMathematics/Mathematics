Στο παρακάτω σχήμα δίνεται η γραφική παράσταση μιας συνάρτησης f με πεδίο ορισμού το $\mathbb{R}$.
\begin{center}
\begin{tikzpicture}[scale=1, >=latex]
    \draw[very thin, color=gray!30] (-2.5,-3.5) grid (5.5,5.5);
    \draw[->] (-2.5,0) -- (6,0) node[right] {$x$};
    \draw[->] (0,-3.5) -- (0,6) node[above] {$y$};
    \foreach \x in {-2,-1,1,2,3,4,5}
        \draw (\x,2pt) -- (\x,-2pt) node[below, font=\small] {\x};
    \foreach \y in {-3,-2,-1,1,2,3,4,5}
        \draw (2pt,\y) -- (-2pt,\y) node[left, font=\small] {\y};
    \node[below left, font=\small] at (0,0) {$0$};
    % Cubic function with local max at x=0 (y=0) and local min at x=2 (y=-2) 
    % => f(x) = 1/2 x^3 - 3/2 x^2
    \draw[very thick, color=maincolor, samples=100, smooth, domain=-1.2:3.5] plot (\x, {0.5*\x*\x*\x - 1.5*\x*\x});
    % Focus on x=2
    \draw[dashed, thick] (2,0) -- (2,-2) -- (0,-2);
    \filldraw (2,-2) circle (1.5pt);
    % Point P (generic)
    \filldraw (3.2, 1.4) circle (1.5pt) node[right] {$P(x_0, f(x_0))$};
\end{tikzpicture}
\end{center}
\begin{alist}
\item Να βρείτε, αν υπάρχουν:
\begin{rlist}
\item τα σημεία τομής της γραφικής παράστασης της f με τους άξονες \monades{3}
\item το πρόσημο των τιμών της συνάρτησης f \monades{4}
\item τα ακρότατά της, καθώς και τις θέσεις αυτών των ακροτάτων. \monades{5}
\end{rlist}
\item Ένας μαθητής ισχυρίζεται ότι η f είναι άρτια και ένας δεύτερος ότι είναι περιττή.
Ποιος από τους δυο μαθητές έχει δίκιο; Να αιτιολογήσετε την απάντησή σας.
\monades{5}
\item Στο παραπάνω σχήμα φαίνεται επίσης και ένα τυχαίο σημείο $Ρ(x_{0,}f(x_{0}))$ στην γραφική παράσταση της $f\ \mu\varepsilon\ x_{0} > 0.$
\begin{rlist}
\item Να δειχτεί ότι τα σημεία $Ρ(x_{0,}f(x_{0}))\ \ \kappa\alpha\iota\ \Sigma(x_{0,} - f(x_{0}))$ είναι συμμετρικά ως προς τον άξονα x'x. \monades{4}
\item Να βρεθεί ο τύπος της συνάρτησης -f και να γίνει η γραφική της παράσταση. \monades{4}
\end{rlist}
\end{alist}
