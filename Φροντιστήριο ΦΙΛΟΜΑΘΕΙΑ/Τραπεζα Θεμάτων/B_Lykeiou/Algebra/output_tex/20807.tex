ΘEMA 2
Δίνεται η συνάρτηση $f(x) = \hm (\pi + x) + \hm (-x)$, $x \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι $f(x) = -2\hm  x$, για κάθε $x \in \mathbb{R}$ και να βρείτε την περίοδο αυτής.
\monades{12}
\item
\begin{rlist}
\item Να μεταφέρετε στην κόλλα σας και να συμπληρώσετε τον παρακάτω πίνακα τιμών.
\begin{tblr}{hlines = {1pt,white},
vlines = {0.5pt,white},
column{odd} = {maincolor!40,c},
column{even} = {maincolor!10,c},
column{1} = {maincolor,c,fg={white}}}
$x$ & $0$ & $\frac{\pi}{2}$ & $\pi$ & $\frac{3\pi}{2}$ & $2\pi$ \\
$f(x) = -2\hm  x$ & & & & & \\
\end{tblr}
\monades{6}
\item Να σχεδιάσετε τη γραφική παράσταση της συνάρτησης $f$ για $0 \le x \le 2\pi$.
\monades{7}
\end{rlist}
\end{alist}
