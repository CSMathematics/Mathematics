\item 3.2 Βασικές Τριγωνομετρικές Ταυτότητες
\item 3.3 Αναγωγή στο 1o Τεταρτημόριο
\end{ylh}
Δίνεται η παράσταση $A = 2\syn \left(\frac{\pi}{2} - \theta\right) + \hm (-\theta)$.
\begin{alist}
\item Να αποδείξετε ότι $A = \hm \theta$.
\monades{12}
\item Να βρείτε την τιμή της παράστασης Α, όταν $\theta \in \left(\frac{3\pi}{2}, 2\pi\right)$ και $\syn \theta = \frac{12}{13}$.
\monades{13}
\end{alist}
