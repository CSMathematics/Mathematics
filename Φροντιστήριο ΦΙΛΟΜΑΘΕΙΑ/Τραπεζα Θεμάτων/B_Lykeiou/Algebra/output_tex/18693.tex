\begin{ylh}{1}
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
Δίνεται η συνάρτηση $f$ με $f(x) = {(\frac{2 - \lambda}{4})}^{x}$.
\begin{alist}
\item Να βρεθούν οι τιμές του πραγματικού αριθμού λ για τις οποίες η f είναι εκθετική συνάρτηση.
\monades{5}
\item Για ποιες τιμές του λ που βρήκατε στο προηγούμενο ερώτημα η συνάρτηση είναι γνησίως
φθίνουσα;
\monades{7}
\item Για $\lambda = 0$
\begin{rlist}
\item Nα σχεδιάσετε τη γραφική παράσταση της συνάρτησης$\ f.$
\monades{6}
\item Να λύσετε την εξίσωση $f(x) + f(x + 1) = 6.$
\monades{7}
\end{rlist}
\end{alist}
