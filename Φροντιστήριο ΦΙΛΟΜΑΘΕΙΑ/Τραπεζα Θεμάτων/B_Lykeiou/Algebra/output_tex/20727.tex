\begin{alist}
\item Να βρείτε τα πεδία ορισμού των συναρτήσεων f και g.
\monades{10}
\item Να λύσετε τις παρακάτω εξισώσεις:
\begin{rlist}
\item $logx = 3.$
\monades{7}
\item $\ln(x - 1) = 1.$
\monades{8}
\end{rlist}
\end{alist}
