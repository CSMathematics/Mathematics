\item 3.4 Οι τριγωνομετρικές συναρτήσεις
\end{ylh}
Στο παρακάτω σχήμα δίνονται οι γραφικές παραστάσεις των συναρτήσεων $f(x) = \hm  x$ και $g(x) = \syn  x$, $x \in \lbrack 0,\ 2\pi\rbrack$.

\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = {$x$}, ylabel = {$y$},
    xmin = -0.2, xmax = 6.8,
    ymin = -1.2, ymax = 1.2,
    xtick = {1.57, 3.14, 4.71, 6.28},
    xticklabels = {$\frac{\pi}{2}$, $\pi$, $\frac{3\pi}{2}$, $2\pi$},
    ytick = {-1, 1},
    grid = both, grid style = {gray!20},
    samples = 100, width = 12cm, height = 7cm,
    tick label style = {font=\small}
]
    % f(x) = sin(x) (maincolor)
    \addplot[very thick, maincolor, domain=0:6.28] {sin(deg(x))} node[above, pos=0.1] {$f(x)$};
    
    % g(x) = cos(x) (secondarycolor)
    \addplot[very thick, secondarycolor, domain=0:6.28] {cos(deg(x))} node[above, pos=0.9] {$g(x)$};
    
    % Intersection points A and B
    % A is at pi/4 = 0.785, sqrt(2)/2 = 0.707
    \filldraw (axis cs:0.785, 0.707) circle (2pt) node[above right] {$A$};
    \draw[dashed, thick] (axis cs:0.785, 0.707) -- (axis cs:0.785, 0);
    \draw (axis cs:0.785, 0.1) -- (axis cs:0.885, 0.1) -- (axis cs:0.885, 0); % Right angle
    
    % B is at 5pi/4 = 3.927, -sqrt(2)/2 = -0.707
    \filldraw (axis cs:3.927, -0.707) circle (2pt) node[below left] {$B$};
    \draw[dashed, thick] (axis cs:3.927, -0.707) -- (axis cs:3.927, 0);
    \draw (axis cs:3.927, -0.1) -- (axis cs:3.827, -0.1) -- (axis cs:3.827, 0); % Right angle
    
    \node[below left] at (axis cs:0,0) {$0$};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{alist}
\item Να βρείτε τις συντεταγμένες των σημείων Α και Β.
\monades{10}
\item Να βρείτε την μονοτονία της συνάρτησης $g(x)$ στο $\left\lbrack \frac{\pi}{2},\pi \right\rbrack$ και την μονοτονία της συνάρτησης $f(x)$ στο $\left\lbrack \frac{3\pi}{2},2\pi \right\rbrack$.
\monades{4}
\item Με την βοήθεια του ερωτήματος β) ή με όποιον άλλο τρόπο θέλετε, να συγκρίνετε, με δικαιολόγηση, τους αριθμούς:
\begin{rlist}
\item
  $\syn \left( \frac{2\pi}{3} \right)$ και $\syn \left( \frac{5\pi}{6} \right)$.
\monades{5}
\item
  $\hm \left( \frac{5\pi}{3} \right)$ και $\hm \left( \frac{11\pi}{6} \right)$.
\monades{6}
\end{rlist}
\end{alist}
