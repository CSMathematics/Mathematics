\begin{ylh}{1}
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
Δίνονται οι συναρτήσεις $f, g: \mathbb{R} \rightarrow \mathbb{R}$ με τύπους $f(x) = 4^x$ και $g(x) = 2^x - \frac{1}{4}$.
\begin{alist}
\item Να αποδείξετε ότι οι γραφικές παραστάσεις των συναρτήσεων $f$ και $g$ έχουν ακριβώς ένα κοινό σημείο $A$, του οποίου να βρείτε τις συντεταγμένες.
\monades{9}
\item Να αποδείξετε ότι η γραφική παράσταση της συνάρτησης $f$ βρίσκεται πάνω από τη γραφική παράσταση της $g$, με εξαίρεση το σημείο $A$.
\monades{9}
\item Να παραστήσετε γραφικά τις συναρτήσεις $f$ και $g$ στο ίδιο σύστημα αξόνων.
\monades{7}
\end{alist}
