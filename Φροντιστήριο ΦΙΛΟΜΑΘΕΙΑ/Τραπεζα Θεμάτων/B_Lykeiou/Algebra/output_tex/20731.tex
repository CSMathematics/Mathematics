\begin{alist}
\item Να δείξετε ότι το πολυώνυμο x-1 είναι παράγοντας του πολυωνύμου $P(x).$
\monades{5}
\item Να παραγοντοποιήσετε το πολυώνυμο $P(x)$ σε πολυώνυμα πρώτου ή δευτέρου βαθμού.
\monades{8}
\item
\begin{rlist}
\item Να λύσετε την εξίσωση $P(x) =$ 0. \monades{5}
\item Αν οι αριθμοί -1 και 1 είναι οι ρίζες της εξίσωσης P(x)=0, να λύσετε την εξίσωση
(2ημx -- 1) \textsuperscript{4} + 6 (2ημx -- 1) \textsuperscript{2} -- 7 = 0, για $x\epsilon\mathbb{R.}$
\monades{7}
\end{rlist}
\end{alist}
