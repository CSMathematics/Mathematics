\Askhsh{20640}{2}
\begin{ylh}{1}
\item 4.1 Πολυώνυμα
\end{ylh}
Δίνεται το πολυώνυμο $P(x) = 2x^3 - 8x^2 + 7x - 1$.
\begin{alist}
\item Να αποδείξετε ότι έχει ρίζα τον αριθμό 1.
\monades{9}
\item Έστω $Q(x)$ πολυώνυμο το οποίο δεν έχει ρίζα τον αριθμό 1.
\end{alist}
\begin{alist}
\item
  Να αποδείξετε ότι το πολυώνυμο $R_1(x) = P(x) + Q(x)$ δεν έχει ρίζα τον αριθμό 1.
\monades{8}
\end{alist}
\begin{alist}
\item
  Να αποδείξετε ότι το πολυώνυμο $R_2(x) = P(x) \cdot Q(x)$ έχει ρίζα τον αριθμό 1.
\monades{8}
\end{alist}
