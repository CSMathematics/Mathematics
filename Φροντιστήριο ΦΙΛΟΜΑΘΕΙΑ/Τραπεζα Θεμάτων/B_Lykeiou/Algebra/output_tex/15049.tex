Δίνεται η συνάρτηση $f(x) = \hm \left(\frac{\pi}{2} - x\right) + \hm (\pi + x)$, $x \in \mathbb{R}$
\begin{alist}
\item Να αποδείξετε ότι $f(x) = \syn  x - \hm  x$.
\monades{6}
\item Να αποδείξετε ότι $-2 \le f(x) \le 2$. Κατόπιν να εξετάσετε αν ο αριθμός 2 είναι η μέγιστη τιμή της συνάρτησης.
\monades{10}
\item Να βρείτε:
\end{alist}
\begin{alist}
\item
  Το σημείο τομής της γραφικής παράστασης $C_f$της f με τον άξονα $y'y$.
\monades{3}
\end{alist}
\begin{alist}
\item
  Δυο σημεία τομής της $C_f$ με τον $x'x$.
\monades{6}
\end{alist}
