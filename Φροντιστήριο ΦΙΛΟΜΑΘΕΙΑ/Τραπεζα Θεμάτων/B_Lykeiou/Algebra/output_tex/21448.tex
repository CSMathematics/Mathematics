\begin{ylh}{1}
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
Όταν ένας ασθενής παίρνει μια δόση ενός φαρμάκου τη χρονική στιγμή $t = 0$, τότε ο οργανισμός του το μεταβολίζει έτσι ώστε η ποσότητά του $f(t)$ (σε mg) να μειώνεται μετά από $t$ ημέρες σύμφωνα με τη συνάρτηση
$f(t) = q_0 \cdot \alpha^t$, $t \ge 0$,
όπου οι αριθμοί $\alpha, q_0$ είναι κατάλληλες θετικές σταθερές.
\begin{alist}
\item Να εξηγήσετε τι παριστάνει η σταθερά $q_0$ στο πλαίσιο του προβλήματος και να αιτιολογήσετε γιατί ισχύει $0 < \alpha < 1$.
\monades{6}
\item Υποθέτουμε τώρα ότι μία ημέρα μετά τη λήψη του φαρμάκου, η ποσότητά του στον οργανισμό του ασθενούς έχει υποδιπλασιαστεί.
\begin{rlist}
\item Να αποδείξετε ότι $\alpha = \frac{1}{2}$.
\monades{5}
\item Να μεταφέρετε στην κόλλα σας και να συμπληρώσετε τον παρακάτω πίνακα τιμών της συνάρτησης $f$, εκφράζοντας τις τιμές $f(t)$ ως συνάρτηση της αρχικής τιμής $q_0$.
\monades{4}
\begin{center}
\begin{tblr}{hlines = {1pt,white},
vlines = {0.5pt,white},
column{odd} = {maincolor!40,c},
column{even} = {maincolor!10,c},
column{1} = {maincolor,c,fg={white}}}
$t$ & 0 & 1 & 2 & 3 & 4 & 5 & 6 \\
$f(t)$ & $q_0$ & $\frac{q_0}{2}$ & & & & & \\
\end{tblr}
\end{center}
\end{rlist}
\item Υποθέτουμε τώρα ότι $\alpha = \frac{1}{2}$ και ότι η ποσότητα του φαρμάκου που παραμένει στον οργανισμό στο τέλος της 4\textsuperscript{ης} ημέρας είναι 25 mg.
\begin{rlist}
\item Να υπολογίσετε την ποσότητα της δόσης που πήρε ο ασθενής.
\monades{5}
\item Να σχεδιάσετε τη γραφική παράσταση της συνάρτησης $f$ στο διάστημα $[0, 6]$.
\monades{5}
\end{rlist}
\end{alist}
