ΘΕΜΑ 4

\begin{alist}
\item Να εξετάσετε αν υπάρχει γωνία $x$ τέτοια ώστε $\eta\mu x = \sigma\upsilon\nu x = 0$.

\monades{5}

\item Να αποδείξετε ότι εξίσωση $\sqrt{3} \cdot \eta\mu x = 3 \cdot \sigma\upsilon\nu x$ είναι ισοδύναμη με την εξίσωση $\varepsilon\phi x = \sqrt{3}$ και κατόπιν να τη λύσετε στο διάστημα $[0, 2\pi]$.

\monades{7}

\item Να σχεδιάσετε τις γραφικές παραστάσεις των συναρτήσεων $f(x) = \sqrt{3} \cdot \eta\mu x$ και $g(x) = 3 \cdot \sigma\upsilon\nu x$ στο ίδιο σύστημα αξόνων στο διάστημα $[0, 2\pi]$ και να ερμηνεύσετε γραφικά το συμπέρασμα του ερωτήματος β).

\monades{7}

\item Αξιοποιώντας το ερώτημα γ) να λύσετε γραφικά την ανίσωση $\sqrt{3} \cdot \eta\mu x < 3 \cdot \sigma\upsilon\nu x$ στο διάστημα $[0, 2\pi]$.

\monades{6}
\end{alist}
