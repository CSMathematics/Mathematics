Δίνονται οι συναρτήσεις $f(x) = x^4 - 3x^2 - 4$ και $g(x) = -x^2 + 4$ με πεδίο ορισμού το $\mathbb{R}$.
\begin{alist}
\item Να δείξετε ότι $f(-x) = f(x)$ και $g(-x) = g(x)$ για κάθε $x \in \mathbb{R}$. \monades{7}
\begin{center}
\begin{tikzpicture}[scale=0.6, >=latex]
    \draw[very thin, color=gray!30, step=2] (-5,-9) grid (5,7);
    \draw[->] (-5,0) -- (5.5,0) node[right] {$x$};
    \draw[->] (0,-9) -- (0,7.5) node[above] {$y$};
    \foreach \x in {-4,-2,2,4}
        \draw (\x,2pt) -- (\x,-2pt) node[below, font=\small] {\x};
    \foreach \y in {-8,-6,-4,-2,2,4,6}
        \draw (2pt,\y) -- (-2pt,\y) node[left, font=\small] {\y};
    \node[below left, font=\small] at (0,0) {$0$};
    % g(x) = -x^2 + 4 part (for x >= 0)
    \draw[very thick, color=secondarycolor, samples=100, smooth, domain=0:3.2] plot (\x, {-\x*\x + 4}) node[right] {$g$};
    % f(x) = x^4 - 3x^2 - 4 part (for x >= 0)
    % Roots of f: x^2 = 4 or x^2 = -1 => x = 2
    \draw[very thick, color=maincolor, samples=100, smooth, domain=0:2.4] plot (\x, {\x*\x*\x*\x - 3*\x*\x - 4}) node[above left] {$f$};
\end{tikzpicture}
\end{center}
\item Στο παραπάνω σχήμα δίνεται μέρος των γραφικών παραστάσεων των συναρτήσεων $f$ και $g$.
Αφού μεταφέρετε το σχήμα στην κόλλα σας, να συμπληρώσετε τις γραφικές παραστάσεις σε όλο το $\mathbb{R}$. Να αιτιολογήσετε την απάντησή σας. \monades{6}
\item Να λύσετε , αλγεβρικά ή γραφικά:
\begin{rlist}
\item την εξίσωση $f(x) = g(x)$. \monades{6}
\item την ανίσωση $f(x) < g(x)$. \monades{6}
\end{rlist}
\end{alist}
