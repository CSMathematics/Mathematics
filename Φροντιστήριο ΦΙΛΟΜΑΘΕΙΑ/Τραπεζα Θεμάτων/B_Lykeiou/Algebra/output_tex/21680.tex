\begin{ylh}{1}
\item 5.2 Λογάριθμοι
\item 5.3 Λογαριθμική συνάρτηση
\end{ylh}
Δίνεται η συνάρτηση $f(x) = (x - 1) \ln x, x > 0$ και η ευθεία $\epsilon: y = 2x - 2$.
\begin{alist}
\item Να αποδείξετε ότι $\frac{1}{3} f(2) + f(4) = f(8)$.
\monades{8}
\item Να αιτιολογήσετε γιατί η γραφική παράσταση $C_f$ της $f$ είναι από τον άξονα $x'x$ και πάνω.
\monades{8}
\item Να βρείτε:
\end{alist}
\begin{alist}
\item
  Τα κοινά σημεία της $C_f$ με την ευθεία.
\monades{4}
\end{alist}
\begin{alist}
\item
  Για ποιες τιμές του $x$ η $C_f$ είναι κάτω από την ευθεία.
\monades{5}
\end{alist}
