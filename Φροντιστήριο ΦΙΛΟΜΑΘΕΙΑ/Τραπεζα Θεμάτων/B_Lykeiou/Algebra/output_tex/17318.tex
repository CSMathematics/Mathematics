\item 5.2 Λογάριθμοι
\item 5.3 Λογαριθμική συνάρτηση
\end{ylh}
Δίνεται η συνάρτηση $f(x) = ln(x^{2} - 2x + 3)$, με $x\mathbb{\in R}$.
\begin{alist}
\item Να βρείτε το $f(3)$.
\monades{5}
\item Να δείξετε ότι $ln3 + 3ln2 - f(3) = ln4$.
\monades{7}
\item Να λύσετε την εξίσωση $f(x) = ln4$.
\monades{13}
\end{alist}
