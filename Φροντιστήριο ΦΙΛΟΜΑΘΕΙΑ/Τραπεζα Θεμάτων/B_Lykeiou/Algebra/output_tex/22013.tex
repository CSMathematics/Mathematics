\begin{ylh}{1}
\item 4.3 Πολυωνυμικές εξισώσεις και ανισώσεις
\end{ylh}
Δίδεται το πολυώνυμο $P(x) = x^{4} + 1$.
\begin{alist}
\item Να αποδείξετε ότι το πολυώνυμο $P(x)$ δεν έχει πραγματικές ρίζες.
\monades{5}
\item Να βρείτε δύο αριθμούς $\alpha,\ \beta$ τέτοιους ώστε:
\begin{equation*}x^{4} + 1 = \left( x^{2} + \alpha x + 1 \right) \cdot \left( x^{2} + \beta x + 1 \right)\end{equation*}
\monades{10}
\item Θεωρούμε την ακόλουθη πρόταση: «\emph{Κάθε πολυώνυμο που μπορεί να αναλυθεί σε γινόμενο πολυωνύμων μικρότερου μη μηδενικού βαθμού, έχει πραγματικές ρίζες}». Είναι η πρόταση αυτή Σωστή ή Λάθος; Αν η πρόταση είναι σωστή, να δώσετε απόδειξη. Αν η πρόταση είναι λάθος, να δώσετε αντιπαράδειγμα.
\monades{10}
\end{alist}
