Η γραφική παράσταση μιας συνάρτησης $f$ με πεδίο ορισμού το $[-4, 4]$ φαίνεται στο παρακάτω σχήμα.

\begin{center}
\begin{tikzpicture}[scale=0.7, >=latex]
    \draw[very thin, color=gray!30] (-4.5,-0.5) grid (4.5,6.5);
    \draw[->] (-4.5,0) -- (4.5,0) node[right] {$x$};
    \draw[->] (0,-0.5) -- (0,6.5) node[above] {$y$};
    
    \foreach \x in {-4,-3,-2,-1,1,2,3,4}
        \draw (\x,2pt) -- (\x,-2pt) node[below, font=\small] {\x};
    \foreach \y in {1,2,3,4,5,6}
        \draw (2pt,\y) -- (-2pt,\y) node[left, font=\small] {\y};
    \node[below left, font=\small] at (0,0) {$0$};
    
    % W-shaped function
    \draw[very thick, color=blue!70!black] (-4,6) -- (-2,3) -- (0,5) -- (2,3) -- (4,6);
    
    \filldraw[blue!70!black] (-4,6) circle (2pt);
    \filldraw[blue!70!black] (-2,3) circle (2pt);
    \filldraw[blue!70!black] (0,5) circle (2pt);
    \filldraw[blue!70!black] (2,3) circle (2pt);
    \filldraw[blue!70!black] (4,6) circle (2pt);
\end{tikzpicture}
\end{center}
