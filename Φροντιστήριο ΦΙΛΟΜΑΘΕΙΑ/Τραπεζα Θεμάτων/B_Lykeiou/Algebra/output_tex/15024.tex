\begin{ylh}{1}
\item 2.1 Μονοτονία-Ακρότατα-Συμμετρίες Συνάρτησης
\end{ylh}
Η γραφική παράσταση μιας συνάρτησης $f$ με πεδίο ορισμού το $[-4, 4]$ φαίνεται στο παρακάτω σχήμα.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = {$x$},
    ylabel = {$y$},
    xmin = -4.5, xmax = 4.5,
    ymin = -0.5, ymax = 6.5,
    xtick = {-4,-3,-2,-1,1,2,3,4},
    ytick = {1,2,3,4,5,6},
    grid = both,
    grid style = {gray!30},
    samples = 150,
    width = 10cm, height = 7cm,
    tick label style = {font=\small},
    every axis x label/.style = {at={(current axis.right of origin)}, anchor=north west},
    every axis y label/.style = {at={(current axis.above origin)}, anchor=south east},
]
    \addplot[very thick, maincolor, domain=-4:-2] {-1.5*x};
    \addplot[very thick, maincolor, domain=-2:0] {x+5};
    \addplot[very thick, maincolor, domain=0:2] {-x+5};
    \addplot[very thick, maincolor, domain=2:4] {1.5*x};
    \addplot[only marks, mark=*, mark size=2pt, maincolor] coordinates {(-4,6) (-2,3) (0,5) (2,3) (4,6)};
\end{axis}
\end{tikzpicture}
\end{center}
\begin{alist}
\item Να αιτιολογήσετε γιατί η συνάρτηση είναι άρτια.
\monades{8}
\item Να βρείτε τα διαστήματα μονοτονίας της $f$.
\monades{8}
\item Να βρείτε την ελάχιστη τιμή της $f$ καθώς και για ποιες τιμές του $x$ τις παρουσιάζει.
\monades{9}
\end{alist}
