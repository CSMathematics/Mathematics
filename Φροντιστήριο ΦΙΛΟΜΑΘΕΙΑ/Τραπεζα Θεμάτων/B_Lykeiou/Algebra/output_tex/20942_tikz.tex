\begin{tikzpicture}[scale=2.5]
    % Unit Circle
    \draw[thick] (0,0) circle (1);
    
    % Axes
    \draw[->] (-1.3,0) -- (1.5,0) node[right] {$x$};
    \draw[->] (0,-1.3) -- (0,1.3) node[above] {$y$};
    
    % Origin and ticks
    \node[below left] at (0,0) {$O$};
    \node[below right] at (1,0) {$1$};
    
    % cos(omega) = -1/3. 
    % acos(-1/3) approx 109.47 degrees. 
    % 3rd quadrant: 360 - 109.47 = 250.53 degrees.
    \def\angle{250.53}
    \coordinate (M) at ({cos(\angle)}, {sin(\angle)});
    
    % Terminal side
    \draw[thick, red] (0,0) -- (M);
    \filldraw[blue] (M) circle (1pt) node[below left, black] {$M$};
    
    % Projection
    \draw[dashed] (M) -- (-0.333, 0) node[above] {$-\frac{1}{3}$};
    
    % Angle omega
    \draw[->, thick, green!60!black] (0.25,0) arc (0:\angle:0.25);
    \node at (-0.3, -0.4) {$\omega$};

\end{tikzpicture}
