\begin{ylh}{1}
\item 2.1 Μονοτονία-Ακρότατα-Συμμετρίες Συνάρτησης
\item 5.1 Εκθετική συνάρτηση
\end{ylh}
Δίνεται το σημείο $Α\left( 1,\frac{1}{2} \right)$ το οποίο ανήκει στη γραφική παράσταση μίας συνάρτησης $f.$
\begin{alist}
\item Αν η συνάρτηση $f$ είναι η εκθετική συνάρτηση $a^{x},\ \ 0 < a < 1,$ να βρείτε το $\alpha.$
\monades{13}
\item Για $a = \frac{1}{2}$ ,
i) να βρείτε τη μονοτονία της συνάρτησης $f(x) = a^{x}.$
\monades{4}
ii) να συγκρίνετε τους αριθμούς $\alpha^{\sqrt{2}},{\ a}^{\sqrt{3}}.$
\monades{8}
\end{alist}
