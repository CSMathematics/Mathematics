\begin{ylh}{1}
\item 3.3 Αναγωγή στο 1o Τεταρτημόριο
\item 3.5 Βασικές τριγωνομετρικές εξισώσεις
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{\alpha+1}{2} \syn (\beta x)$, με $\alpha, \beta > 0$, η οποία έχει ελάχιστο $-2$ και περίοδο $\frac{\pi}{2}$.
\begin{alist}
\item Να δείξετε ότι $\alpha = 3$ και $\beta = 4$.
\monades{5}
\item Δίνεται η παράσταση $A = \dfrac{\hm \left(\frac{\pi}{2} - x\right) \cdot \ef (\pi - x) \cdot \hm (2\pi + x)}{\syn (3\pi - x) \cdot \syf \left(\frac{7\pi}{2} - x\right) \cdot \syn \left(\frac{\pi}{2} + x\right)}$. Να δείξετε ότι $A = -1$.
\monades{10}
\item Να λύσετε την εξίσωση $f(x) = 2A$, στο διάστημα $\left[\pi, \frac{3\pi}{2}\right]$.
\monades{10}
\end{alist}
