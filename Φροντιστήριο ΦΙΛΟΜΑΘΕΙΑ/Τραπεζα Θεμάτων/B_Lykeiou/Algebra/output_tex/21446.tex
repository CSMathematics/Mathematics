Δίνεται η συνάρτηση $f(x) = \ln(e^x - 2)$.

\begin{alist}
\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f$.

\monades{7}

\item Να λύσετε την εξίσωση $f(x) + x = 3\ln 2$.

\monades{9}

\item Να λύσετε την ανίσωση $f(x) + x \ge 3\ln 2$.

\monades{9}
\end{alist}
