Δίνεται το πολυώνυμο $P(x) = x^3 - x^2 - 2x$.
\begin{alist}
\item Να λύσετε την εξίσωση $P(x) = 0$.
\monades{7}
\item Να λύσετε την εξίσωση $\ln^3 x - \ln^2 x - 2\ln x = 0$.
\monades{8}
\item Να λύσετε την ανίσωση $\ln^3 x - \ln^2 x - 2\ln x > 0$.
\monades{10}
\end{alist}
