\begin{tikzpicture}[scale=2]
    % Grid
    \draw[help lines, step=0.5, gray!30] (-1.5,-1.5) grid (1.5,1.5);
    
    % Unit Circle
    \draw[thick] (0,0) circle (1);
    
    % Axes
    \draw[->, thick] (-1.5,0) -- (1.5,0) node[right] {$x$};
    \draw[->, thick] (0,-1.5) -- (0,1.5) node[above] {$y$};
    
    % Point M(x, 1/2) in 2nd quadrant
    % sin(omega) = 1/2 => omega = 150 degrees (5pi/6)
    \def\angle{150}
    \coordinate (O) at (0,0);
    \coordinate (M) at ({cos(\angle)}, {sin(\angle)});
    
    % Draw terminal side
    \draw[thick, red] (O) -- (M);
    \filldraw[blue] (M) circle (1.5pt) node[above left, black] {$M(x, \frac{1}{2})$};
    
    % Projection to y-axis
    \draw[dashed] (M) -- (0, 0.5) node[right] {$\frac{1}{2}$};
    \draw[dashed] (M) -- ({cos(\angle)}, 0) node[below] {$x$};

    % Angle omega
    \draw[->, thick, green!60!black] (0.3,0) arc (0:\angle:0.3);
    \node at (-0.2, 0.4) {$\omega$};
    
    % Origin
    \node[below right] at (0,0) {$O$};

\end{tikzpicture}
