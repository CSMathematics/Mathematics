Δίνεται το πολυώνυμο $P(x) = 2x^3 - x^2 - 8x + 4$.
\begin{alist}
\item
\begin{rlist}
\item Να βρείτε το πηλίκο και το υπόλοιπο της διαίρεσης $P(x) : (x - 2)$.
\monades{10}
\item Να γράψετε την ταυτότητα της διαίρεσης $P(x) : (x - 2)$.
\monades{9}
\end{rlist}
\item Αν $P(x) = (2x - 1)(x^2 - 4)$, να λύσετε την εξίσωση $P(x) = 0$.
\monades{6}
\end{alist}
