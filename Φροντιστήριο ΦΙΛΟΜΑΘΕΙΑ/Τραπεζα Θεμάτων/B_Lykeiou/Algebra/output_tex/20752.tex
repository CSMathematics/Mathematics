\begin{ylh}{1}
\item 3.2 Βασικές Τριγωνομετρικές Ταυτότητες
\item 4.3 Πολυωνυμικές εξισώσεις και ανισώσεις
\end{ylh}
Δύο συμμαθητές ο Αλέξανδρος και ο Φίλιππος που κάθονται στο ίδιο θρανίο σχεδιάζουν τον τριγωνομετρικό κύκλο σε μιλιμιτρέ χαρτί και στη συνέχεια προσπαθώντας να υπολογίσουν τις συντεταγμένες ενός δοσμένου σημείου Μ αυτού του κύκλου διαφωνούν στην απάντησή τους. Ο Αλέξανδρος εκτιμά ότι οι συντεταγμένες του σημείου Μ είναι $M(0.8, 0.6)$ ενώ ο Φίλιππος εκτιμά ότι οι συντεταγμένες του είναι $M(1, 1)$.
\begin{alist}
\item Ποιος από τους δύο έχει σίγουρα άδικο; Να αιτιολογήσετε την απάντησή σας.
\monades{08}
\item Αν υποθέσουμε ότι το σημείο του οποίου υπολογίστηκαν σωστά οι συντεταγμένες του είναι το $M(0.8, 0.6)$
\begin{rlist}
\item να αιτιολογήσετε ότι $\hm \omega = 0.6$ και $\syn \omega = 0.8$
\monades{03}
\item να υπολογίσετε την τιμή της παράστασης
$A = \hm (\pi - \omega) - 2\syn \left(\frac{\pi}{2} - \omega\right) + \ef (-\omega) + \syf (\pi + \omega)$.
\monades{05}
\end{rlist}
\item Δίνεται η πολυωνυμική συνάρτηση $f(x) = 5\syn \omega \cdot x^3 - 10\hm \omega \cdot x^2 + 5x - 3$, $x \in \mathbb{R}$ όπου ω η γωνία που βρήκατε στο προηγούμενο ερώτημα. Να βρείτε το διάστημα στο οποίο η γραφική παράσταση της συνάρτησης f βρίσκεται κάτω από τον άξονα x΄x.
\monades{09}
\end{alist}
