\begin{tikzpicture}[scale=0.8]
    % Grid
    \draw[very thin, lightgray!30] (-3,-4) grid (3,4);
    
    % Axes
    \draw[->, thick] (-3.5,0) -- (3.5,0) node[right] {$x$};
    \draw[->, thick] (0,-4.5) -- (0,4.5) node[above] {$y$};
    
    % Cubic function y = x^3 - 3x
    \draw[ultra thick, blue, domain=-2.3:2.3, samples=100] plot (\x, {\x^3 - 3*\x});
    
    % Extremes
    \filldraw[red] (-1,2) circle (2.5pt) node[above left, black] {$(-1,2)$};
    \filldraw[red] (1,-2) circle (2.5pt) node[below right, black] {$(1,-2)$};
    
    % Origin
    \node[below left] at (0,0) {$0$};
    
    % Common labels
    \foreach \x in {-2,-1,1,2} \node[below, font=\scriptsize] at (\x,0) {$\x$};
    \foreach \y in {-3,-1,1,3} \node[left, font=\scriptsize] at (0,\y) {$\y$};

\end{tikzpicture}
