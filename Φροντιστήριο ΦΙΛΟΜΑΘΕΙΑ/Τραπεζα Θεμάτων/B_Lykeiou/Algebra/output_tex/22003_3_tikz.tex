\begin{tikzpicture}[scale=1]
    % Grid
    \draw[help lines, gray!30] (0,-1) grid[xstep=0.5, ystep=1] (2,1);
    
    % Axes
    \draw[->, thick] (-0.5,0) -- (2.5,0) node[right] {$x$};
    \draw[->, thick] (0,-1.5) -- (0,1.5) node[above] {$y$};
    
    % Function y = sin(2*pi*x)
    \draw[ultra thick, blue, domain=0:2.1, samples=150] plot (\x, {sin(2*\x*180)});
    
    % Labels
    \node[below left] at (0,0) {$0$};
    \foreach \x in {0.5, 1, 1.5, 2} \draw (\x, 0.1) -- (\x, -0.1) node[below] {$\x$};
    \draw (0.1, 1) -- (-0.1, 1) node[left] {$1$};
    \draw (0.1, -1) -- (-0.1, -1) node[left] {$-1$};
