ΘΕΜΑ 4

\begin{alist}
\item Δίνεται η συνάρτηση $f(x) = \sqrt{x^2 + 1} - x$ με πεδίο ορισμού το $\mathbb{R}$.

\end{alist}
\begin{alist}
\item
  Να αποδείξετε ότι $\sqrt{x^2 + 1} - x > 0$, για κάθε $x \in (-\infty, 0)$.

\monades{03}


\end{alist}
\begin{alist}
\item
  Να βρείτε τα διαστήματα στα οποία η γραφική παράσταση της $f$ βρίσκεται πάνω από τον άξονα $x'x$.

\monades{09}

\item Δίνεται η συνάρτηση $g(x) = \ln(\sqrt{x^2 + 1} + x)$, με πεδίο ορισμού το $\mathbb{R}$.

\end{alist}
\begin{alist}
\item
  Να αποδείξετε ότι $g(-x) + g(x) = 0$, για κάθε $x \in \mathbb{R}$.

\monades{09}


\end{alist}
\begin{alist}
\item
  Να αποδείξετε ότι η γραφική παράσταση της συνάρτησης $g$ έχει κέντρο συμμετρίας την αρχή των αξόνων $O$.

\monades{04}
\end{alist}
