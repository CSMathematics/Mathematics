\begin{alist}
\item Με βάση το σχήμα:
\begin{rlist}
\item να βρείτε το συνημίτονο της γωνίας $\omega$.

\monades{4}

\item να αιτιολογήσετε γιατί $\eta\mu\omega > 0$.

\monades{8}
\end{rlist}
\item \monades{13} Να υπολογίσετε το ημίτονο της γωνίας $\omega$.
