Δίνεται η συνάρτηση $f(x) = \log\frac{4x - 1}{2x + 5}$ .

\begin{alist}
\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f$.

\monades{7}

\item Να λύσετε την εξίσωση $f(x) = \log 3 - \log 7$.

\monades{9}

\item Να λύσετε την ανίσωση $f(x) > \log 3 - \log 7$.

\monades{9}
\end{alist}
