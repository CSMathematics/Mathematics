ΘΕΜΑ 4

Δίνεται το πολυώνυμο $P(x) = -x^3 - 4x^2 - x + 6$.

\begin{alist}
\item Να λύσετε την ανίσωση $P(x) < 0$.

\monades{10}

\item Από τα παρακάτω σχήματα, ένα μόνο μπορεί να αντιστοιχεί στην γραφική παράσταση της πολυωνυμικής συνάρτησης $P(x)$. Να βρείτε ποιο αιτιολογώντας την απάντησή σας.

\monades{7}

\item Να αποδείξετε ότι η εξίσωση $P(x) = \ln x$ έχει μοναδική λύση την $x = 1$.

\monades{8}

\begin{longtable}[]{@{}cc@{}}
\begin{tikzpicture}[scale=0.5]
    \draw[gray!30, very thin] (-5,-2) grid (3,8);
    \draw[->] (-5,0) -- (3,0) node[right] {$x$};
    \draw[->] (0,-2) -- (0,8) node[above] {$y$};
    % P(x) = -(x-1)(x+2)(x+3) -> roots 1, -2, -3. y-int 6.
    \draw[very thick, blue, smooth, domain=-3.5:1.5] plot (\x, {-(\x-1)*(\x+2)*(\x+3)});
    \node[below] at (0,-2) {Σχήμα 1};
\end{tikzpicture}
 & 
\begin{tikzpicture}[scale=0.5]
    \draw[gray!30, very thin] (-5,-8) grid (3,2);
    \draw[->] (-5,0) -- (3,0) node[right] {$x$};
    \draw[->] (0,-8) -- (0,2) node[above] {$y$};
    % Variant: reflected P(x)
    \draw[very thick, blue, smooth, domain=-3.5:1.5] plot (\x, {(\x-1)*(\x+2)*(\x+3)});
    \node[below] at (0,-8) {Σχήμα 2};
\end{tikzpicture}
 \\
\begin{tikzpicture}[scale=0.5]
    \draw[gray!30, very thin] (-3,-8) grid (5,2);
    \draw[->] (-3,0) -- (5,0) node[right] {$x$};
    \draw[->] (0,-8) -- (0,2) node[above] {$y$};
    % Variant: shifted roots
    \draw[very thick, blue, smooth, domain=-1.5:4] plot (\x, {-(\x+1)*(\x-2)*(\x-3)});
    \node[below] at (1,-8) {Σχήμα 3};
\end{tikzpicture}
 & 
\begin{tikzpicture}[scale=0.5]
    \draw[gray!30, very thin] (-3,-2) grid (5,8);
    \draw[->] (-3,0) -- (5,0) node[right] {$x$};
    \draw[->] (0,-2) -- (0,8) node[above] {$y$};
    % Variant: shifted reflected
    \draw[very thick, blue, smooth, domain=-1.5:4] plot (\x, {(\x+1)*(\x-2)*(\x-3)});
    \node[below] at (1,-2) {Σχήμα 4};
\end{tikzpicture}
\end{longtable}
\end{alist}
