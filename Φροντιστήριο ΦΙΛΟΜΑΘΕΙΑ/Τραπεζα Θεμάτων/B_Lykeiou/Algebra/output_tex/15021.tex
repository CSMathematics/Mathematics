Δίνεται η συνάρτηση $f(x) = \frac{x^2 + 1}{x}$.
\begin{alist}
\item Να βρείτε το πεδίο ορισμού της.
\monades{5}
\item Να αποδείξετε ότι η γραφική της παράσταση έχει κέντρο συμμετρίας το $O(0,0)$.
\monades{6}
\item Να υπολογίσετε την παράσταση $f(\ln 2) + f\left(\ln \frac{1}{2}\right)$.
\monades{7}
\item Να αποδείξετε ότι $f(\hm \theta) + f(\hm (\pi + \theta)) = 0$, για κάθε $\theta \in \mathbb{R}$ με $\hm \theta \neq 0$.
\monades{7}
\end{alist}
