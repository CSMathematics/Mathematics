\begin{alist}
\item Έχουμε: $\eta\mu(\pi - x) = \eta\mu x$, $\eta\mu(\pi + x) = -\eta\mu x$, $\sigma\upsilon\nu(-x) = \sigma\upsilon\nu x$, επομένως:
\[A = \eta\mu^2 x + (-\eta\mu x)^2 + \sigma\upsilon\nu^2 x = \eta\mu^2 x + \eta\mu^2 x + \sigma\upsilon\nu^2 x = 1 + \eta\mu^2 x.\]

\item 
\begin{align*}
B &= \frac{\eta\mu x}{1 + \sigma\upsilon\nu x} + \frac{1 + \sigma\upsilon\nu x}{\eta\mu x} = \frac{\eta\mu^2 x + (1 + \sigma\upsilon\nu x)^2}{(1 + \sigma\upsilon\nu x)\eta\mu x} = \frac{\eta\mu^2 x + 1 + 2\sigma\upsilon\nu x + \sigma\upsilon\nu^2 x}{(1 + \sigma\upsilon\nu x)\eta\mu x} \\
&= \frac{2(1 + \sigma\upsilon\nu x)}{(1 + \sigma\upsilon\nu x)\eta\mu x} = \frac{2}{\eta\mu x}.
\end{align*}

\item Θα εξετάσουμε αν η εξίσωση $A = B$ δηλαδή $\eta\mu^2 x + 1 = \dfrac{2}{\eta\mu x} \quad (1)$ έχει λύση. \\
Θέτουμε $\eta\mu x = \omega$ και η εξίσωση $(1)$ μετασχηματίζεται στην $\omega^2 + 1 = \dfrac{2}{\omega} \quad (2)$. \\
Επειδή $\dfrac{3\pi}{2} < x < 2\pi$ είναι $-1 < \eta\mu x < 0$ δηλαδή $-1 < \omega < 0$. \\
Τότε: 
\begin{align*}
(2) &\Leftrightarrow \omega^3 + \omega = 2 \Leftrightarrow \omega^3 + \omega - 2 = 0 \Leftrightarrow \omega^3 - \omega + 2\omega - 2 = 0 \Leftrightarrow \\
&\omega(\omega^2 - 1) + 2(\omega - 1) = 0 \Leftrightarrow \omega(\omega - 1)(\omega + 1) + 2(\omega - 1) = 0 \Leftrightarrow \\
&(\omega - 1)[\omega(\omega + 1) + 2] = 0 \Leftrightarrow (\omega - 1)(\omega^2 + \omega + 2) = 0 \Leftrightarrow \\
&\omega - 1 = 0 \text{ ή } \omega^2 + \omega + 2 = 0 \Leftrightarrow \omega = 1 \text{ ή } \omega^2 + \omega + 2 = 0
\end{align*}
Η $\omega = 1$ απορρίπτεται λόγω του περιορισμού ενώ η εξίσωση $\omega^2 + \omega + 2 = 0$ έχει διακρίνουσα $\Delta = -7$, οπότε είναι αδύνατη. \\
Επομένως, δεν υπάρχουν τιμές του $x$ τέτοιες, ώστε $A = B$.
\end{alist}
