\lysh{17919}{4}
\begin{alist}
\item Η ευθεία $\text{AB}$ έχει εξίσωση της μορφής $(\varepsilon): y = \alpha x + \beta$. Το σημείο $\text{A}\left(1, -\dfrac{3}{4}\right)$ είναι σημείο της $(\varepsilon)$, άρα:
\[-\frac{3}{4} = \alpha\cdot 1 + \beta \quad (1).\]
Το σημείο $\text{B}(4, -3)$ είναι σημείο της $(\varepsilon)$, άρα:
\[-3 = \alpha\cdot 4 + \beta \quad (2).\]
Λύνουμε το σύστημα των εξισώσεων (1) και (2):
\[\begin{cases} -\dfrac{3}{4} = \alpha\cdot 1 + \beta \\ -3 = \alpha\cdot 4 + \beta \end{cases} \Leftrightarrow \begin{cases} 3\alpha = -3 + \dfrac{3}{4} \\ 4\alpha + \beta = -3 \end{cases} \Rightarrow \begin{cases} \alpha = -\dfrac{3}{4} \\ 4\cdot\left(-\dfrac{3}{4}\right) + \beta = -3 \end{cases} \Leftrightarrow \begin{cases} \alpha = -\dfrac{3}{4} \\ \beta = 0 \end{cases}.\]
Άρα η ευθεία $\text{AB}$ έχει εξίσωση $(\varepsilon): y = -\dfrac{3}{4}x$.
\item 
\begin{rlist}
\item Για κάθε $x \in \mathbb{R}$ ισχύει και $-x \in \mathbb{R}$. Έχουμε:
\[f(-x) = \frac{1}{4}(-x)^3 - (-x) = -\frac{1}{4}x^3 + x = -\left(\frac{1}{4}x^3 - x\right) = -f(x).\]
\item Στο βi) αποδείξαμε ότι $f(-x) = -f(x)$ για κάθε $x \in \mathbb{R}$, δηλαδή ότι η $f$ είναι περιττή. 
Συνεπώς η γραφική της παράσταση είναι συμμετρική ως προς την αρχή των αξόνων $(0,0)$.
\end{rlist}
\end{alist}
