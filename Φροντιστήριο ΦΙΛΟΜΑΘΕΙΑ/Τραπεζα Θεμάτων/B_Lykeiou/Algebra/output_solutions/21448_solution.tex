\lysh{21448}{4}
\begin{alist}
\item Στο πλαίσιο του προβλήματος η σταθερά $q_0$ παριστάνει τη δόση του φαρμάκου που πήρε ο ασθενής (την ποσότητα του φαρμάκου στον οργανισμό τη χρονική στιγμή $t = 0$). \\
Η ποσότητα του φαρμάκου που παραμένει στον οργανισμό του ασθενούς μειώνεται όσο περνούν οι ημέρες, δηλαδή η εκθετική συνάρτηση $f(t) = q_0\cdot\alpha^t$ είναι φθίνουσα, το οποίο συμβαίνει όταν $0 < \alpha < 1$.
\item 
\begin{rlist}
\item Μία ημέρα μετά τη λήψη του φαρμάκου, η ποσότητά του στον οργανισμό του ασθενούς έχει υποδιπλασιαστεί, δηλαδή
\begin{align*}
f(1) &= \frac{q_0}{2} \Leftrightarrow \\
q_0\cdot\alpha &= \frac{q_0}{2} \Leftrightarrow \alpha &= \frac{1}{2}.
\end{align*}
\item Παρατηρούμε ότι: 
\[f(0) = q_0\cdot\alpha^0 = q_0, \quad f(1) = q_0\cdot\frac{1}{2} = \frac{q_0}{2}, \quad f(2) = q_0\cdot\left(\frac{1}{2}\right)^2 = \frac{q_0}{4}, \quad f(3) = q_0\cdot\left(\frac{1}{2}\right)^3 = \frac{q_0}{8},\]
\[f(4) = q_0\cdot\left(\frac{1}{2}\right)^4 = \frac{q_0}{16}, \quad f(5) = q_0\cdot\left(\frac{1}{2}\right)^5 = \frac{q_0}{32}, \quad f(6) = q_0\cdot\left(\frac{1}{2}\right)^6 = \frac{q_0}{64}.\]
Οπότε:
\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{myhtblr}{}
$t$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
$f(t)$ & $q_0$ & $\dfrac{q_0}{2}$ & $\dfrac{q_0}{4}$ & $\dfrac{q_0}{8}$ & $\dfrac{q_0}{16}$ & $\dfrac{q_0}{32}$ & $\dfrac{q_0}{64}$ \\
\end{myhtblr}
\end{center}
\end{rlist}
\item 
\begin{rlist}
\item Έχουμε $f(4) = 25 \Leftrightarrow \dfrac{q_0}{16} = 25 \Leftrightarrow q_0 = 25\cdot 16 \Leftrightarrow q_0 = 400 \text{ mg}$.
\item Με τη βοήθεια του $\beta\text{ii)}$ ερωτήματος έχουμε τον παρακάτω πίνακα τιμών για τη συνάρτηση $f(t) = 400\cdot\left(\dfrac{1}{2}\right)^t$ στο διάστημα $[0, 6]$:
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{myhtblr}{}
$t$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
$f(t)$ & $400$ & $200$ & $100$ & $50$ & $25$ & $12{,}5$ & $6{,}25$ \\
\end{myhtblr}
\end{center}
\end{rlist}
\end{alist}
