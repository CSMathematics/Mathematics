\lysh{20731}{4}
\begin{alist}
\item Το $x-1$ είναι παράγοντας του πολυωνύμου $P(x)$, διότι το $1$ είναι ρίζα του πολυωνύμου, αφού $\text{P}(1) = 1 + 6 \cdot 1 - 7 = 0$.
\item Το πολυώνυμο παραγοντοποιείται:
\begin{align*}
P(x) &= x^4 + 6x^2 - 7 = x^4 + 7x^2 - x^2 - 7 = x^2(x^2 + 7) - (x^2 + 7) = \\
&= (x^2 + 7)(x^2 - 1) = (x^2 + 7)(x + 1)(x - 1).
\end{align*}
\item 
\begin{rlist}
\item Η εξίσωση $P(x) = 0$ γίνεται $(x^2 + 7)(x + 1)(x - 1) = 0$. \\
Επειδή $x^2 + 7 > 0$, οι λύσεις της εξίσωσης είναι $x = -1$ ή $x = 1$.
\item Στην εξίσωση $(2\hm  x - 1)^4 + 6(2\hm  x - 1)^2 - 7 = 0$ αν θέσουμε $2\hm  x - 1 = \omega$ προκύπτει η εξίσωση $P(\omega) = 0$, σύμφωνα με το $\text{i}$ ερώτημα θα έχουμε $\omega = -1$ ή $\omega = 1$. \\
Άρα, $2\hm  x - 1 = -1$ ή $2\hm  x - 1 = 1$, τότε $\hm  x = 0$ ή $\hm  x = 1$. \\
Επομένως, $x = 2\kappa\pi$ ή $2\kappa\pi + \pi$ ή $2\kappa\pi + \dfrac{\pi}{2}$ με $\kappa \in \mathbb{Z}$.
\end{rlist}
\end{alist}
