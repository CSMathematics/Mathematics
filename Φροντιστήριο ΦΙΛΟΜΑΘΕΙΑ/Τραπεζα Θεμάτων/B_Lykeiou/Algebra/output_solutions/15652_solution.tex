\lysh{15652}{2}
\begin{alist}
\item Έχουμε ισοδύναμα
\begin{align*}
\hm ^2\varphi + \syn ^2\varphi = 1 &\Leftrightarrow \\
\left(\frac{3}{5}\right)^2 + \syn ^2\varphi = 1 &\Leftrightarrow \\
\syn ^2\varphi = 1 - \frac{9}{25} &\Leftrightarrow \\
\syn ^2\varphi = \frac{16}{25} &\Leftrightarrow \\
\syn \varphi = \pm\frac{4}{5}.
\end{align*}
Επειδή η γωνία $\varphi$ είναι οξεία, έχουμε τελικά $\syn \varphi = \dfrac{4}{5}$.
\item Όπως φαίνεται από το σχήμα, η γωνία $\omega$ είναι παραπληρωματική της γωνίας $\varphi$, οπότε
\[\hm \omega = \hm \varphi = \frac{3}{5} \text{ και } \syn \omega = -\syn \varphi = -\frac{4}{5}.\]
\end{alist}
