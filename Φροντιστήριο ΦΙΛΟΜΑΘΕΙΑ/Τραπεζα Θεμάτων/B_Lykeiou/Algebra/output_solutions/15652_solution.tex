\begin{alist}
\item Έχουμε ισοδύναμα
\begin{align*}
\eta\mu^2\varphi + \sigma\upsilon\nu^2\varphi = 1 &\Leftrightarrow \\
\left(\frac{3}{5}\right)^2 + \sigma\upsilon\nu^2\varphi = 1 &\Leftrightarrow \\
\sigma\upsilon\nu^2\varphi = 1 - \frac{9}{25} &\Leftrightarrow \\
\sigma\upsilon\nu^2\varphi = \frac{16}{25} &\Leftrightarrow \\
\sigma\upsilon\nu\varphi = \pm\frac{4}{5}.
\end{align*}
Επειδή η γωνία $\varphi$ είναι οξεία, έχουμε τελικά $\sigma\upsilon\nu\varphi = \dfrac{4}{5}$.

\item Όπως φαίνεται από το σχήμα, η γωνία $\omega$ είναι παραπληρωματική της γωνίας $\varphi$, οπότε
\[\eta\mu\omega = \eta\mu\varphi = \frac{3}{5} \text{ και } \sigma\upsilon\nu\omega = -\sigma\upsilon\nu\varphi = -\frac{4}{5}.\]
\end{alist}
