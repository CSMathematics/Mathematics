\lysh{17241}{2}
\begin{alist}
\item 
\begin{rlist}
\item Επειδή $P(-1) = (-1)^3 + (-1) + 2 = -2 + 2 = 0$, το υπόλοιπο της διαίρεσης του $P(x)$ με το $(x + 1)$ είναι $0$. Άρα, το $(x + 1)$ είναι παράγοντας του $P(x)$.
\item Κάνουμε τη διαίρεση του $P(x)$ με το $(x + 1)$ με τη βοήθεια του σχήματος Horner:
\begin{center}
\Horner{1,0,1,2}{-1}
\end{center}
Άρα, $P(x) = (x + 1)(x^2 - x + 2)$.
\end{rlist}
\item Το τριώνυμο $x^2 - x + 2$ έχει διακρίνουσα $\Delta = (-1)^2 - 4\cdot 1\cdot 2 = 1 - 8 = -7 < 0$ οπότε $x^2 - x + 2 > 0$ για κάθε $x \in \mathbb{R}$. Άρα,
\[P(x) < 0 \Leftrightarrow (x + 1)(x^2 - x + 2) < 0 \Leftrightarrow x + 1 < 0 \Leftrightarrow x < -1.\]
\end{alist}
