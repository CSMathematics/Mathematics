\lysh{15436}{4}
\begin{alist}
\item Είναι $BA = 2$, $MA = x$, $MK = 4-x$. \\
Η σχέση μεταξύ του διαστήματος $s$ που διανύεται, της ταχύτητας $v$ και του αντίστοιχου χρόνου κίνησης $t$, είναι: $v = \dfrac{s}{t} \Leftrightarrow t = \dfrac{s}{v}$. \\
Το τρίγωνο $BAM$ είναι ορθογώνιο, οπότε εφαρμόζοντας το Πυθαγόρειο Θεώρημα προκύπτει ότι $BM = \sqrt{4 + x^2}$.
\item Η κίνηση γίνεται σε δύο μέσα - κολύμβηση στη θάλασσα και τρέξιμο στη ξηρά - με διαφορετικές (αλλά σταθερές) ταχύτητες $v_{\kappa} = 3 \dfrac{km}{h}$ και $v_{\tau} = 5 \dfrac{km}{h}$ αντίστοιχα. Έτσι, ο συνολικός χρόνος κίνησης θα προκύψει ως άθροισμα των δύο επιμέρους χρόνων.
\begin{itemize}
    \item Ο χρόνος κίνησης από το $B$ στο $M$: $t_1 = \dfrac{BM}{v_{\kappa}} = \dfrac{\sqrt{4+x^2}}{3}$
    \item Ο χρόνος κίνησης από το $M$ στο $K$: $t_2 = \dfrac{MK}{v_{\tau}} = \dfrac{4-x}{5}$
    \item Ο χρόνος της συνολικής κίνησης: $t_{o\lambda} = t_1 + t_2 = \dfrac{\sqrt{4+x^2}}{3} + \dfrac{4-x}{5}$
\end{itemize}
Επομένως, η συνάρτηση που εκφράζει τον χρόνο κίνησης $t$ (σε $h$) του κολυμβητή - δρομέα ως προς την απόσταση $x$ (σε $km$) είναι η
\[t(x) = \frac{\sqrt{4+x^2}}{3} + \frac{4-x}{5}, \quad x \in [0, 4].\]
\item Αρκεί να λυθεί η εξίσωση:
\begin{align*}
t(x) = \frac{4}{3} &\Leftrightarrow \frac{\sqrt{4+x^2}}{3} + \frac{4-x}{5} = \frac{4}{3} \\
&\Leftrightarrow 5\sqrt{4+x^2} + 3(4-x) = 20 \\
&\Leftrightarrow 5\sqrt{4+x^2} = 3x + 8
\end{align*}
Αφού $x \in [0,4]$ έπεται ότι $3x + 8 > 0$, επομένως ισοδύναμα είναι:
\begin{align*}
25(4+x^2) &= (3x+8)^2 \Leftrightarrow \\
100 + 25x^2 &= 9x^2 + 48x + 64 \Leftrightarrow \\
16x^2 - 48x + 36 &= 0 \Leftrightarrow \\
4x^2 - 12x + 9 &= 0 \Leftrightarrow \\
(2x - 3)^2 &= 0 \Leftrightarrow x = \frac{3}{2}.
\end{align*}
Η λύση είναι δεκτή, διότι $\dfrac{3}{2} \in [0,4]$. \\
Επομένως ο κολυμβητής θα βγει στην ακτή σε απόσταση $1,5 \, km$ από το σημείο $A$.
\end{alist}
