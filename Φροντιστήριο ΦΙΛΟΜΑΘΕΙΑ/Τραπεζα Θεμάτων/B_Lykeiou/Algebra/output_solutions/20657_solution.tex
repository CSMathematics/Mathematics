\lysh{20657}{4}
\begin{alist}
\item Με βάση τα δεδομένα του προβλήματος έχουμε
\[\theta_0 = 100^\circ\text{C}, \quad \text{T} = 30^\circ\text{C}, \quad \text{ενώ } \theta(5) = 80^\circ\text{C}.\]
Άρα $80 = 30 + (100 - 30)e^{5k}$, οπότε $e^{5k} = \dfrac{50}{70} = \dfrac{5}{7}$. \\
Έτσι, θα έχουμε $5k = \ln\left(\dfrac{5}{7}\right)$. \\
Τελικά, $k = \dfrac{-0,336}{5} = -0,0672$.
\item Στο προηγούμενο ερώτημα βρήκαμε ότι $e^{5k} = \dfrac{5}{7}$ άρα $(e^k)^5 = \dfrac{5}{7}$, οπότε $e^k = \left(\dfrac{5}{7}\right)^{1/5}$. \\
Άρα $\theta(t) = 30 + (100 - 30)(e^k)^t = 30 + 70 \cdot \left[\left(\dfrac{5}{7}\right)^{1/5}\right]^t = 30 + 70 \cdot \left(\dfrac{5}{7}\right)^{t/5}$. \\
Εναλλακτικά, $\theta(t) = 30 + (100 - 30)e^{5kt/5} = 30 + 70 \cdot (e^{5k})^{t/5} = 30 + 70 \cdot \left(\dfrac{5}{7}\right)^{t/5}$.
\item Ζητάμε τον αριθμό 
\begin{align*}
\theta(100) &= 30 + 70 \cdot \left(\frac{5}{7}\right)^{20} = 30 + 70 \cdot \left[\left(\frac{5}{7}\right)^{10}\right]^2 \cong 30 + 70 \cdot (0,034)^2 \\
&= 30 + 70 \cdot 0,001156 = 30 + 0,08092 \cong 30,08^\circ\text{C}.
\end{align*}
\end{alist}
