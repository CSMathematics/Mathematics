\lysh{20752}{4}
\begin{alist}
\item Επειδή για το σημείο $M$ του τριγωνομετρικού κύκλου οι συντεταγμένες του είναι $M(\syn \omega, \hm \omega)$ για την αντίστοιχη γωνία $\omega$ που έχει τελική πλευρά OM, είναι φανερό ότι έχει άδικο ο Φίλιππος καθώς δεν υπάρχει γωνία $\omega$ με $\hm \omega = 1$ και $\syn \omega = 1$ καθώς τότε θα είχαμε $\hm ^2\omega + \syn ^2\omega = 1^2 + 1^2 = 1+1 = 2$, που είναι αδύνατο καθώς γνωρίζουμε ότι για κάθε γωνία $\omega$, ισχύει $\hm ^2\omega + \syn ^2\omega = 1$.
\item 
\begin{rlist}
\item Από το πρώτο ερώτημα συμπεραίνουμε ότι οι σωστές συντεταγμένες του σημείου M είναι $M(0,8, 0,6)$, επομένως $\syn \omega = 0,8 = \dfrac{4}{5}$ διότι αποτελεί την τετμημένη του M και $\hm \omega = 0,6 = \dfrac{3}{5}$ διότι αποτελεί την τεταγμένη του M.
\item Επίσης, $\ef \omega = \dfrac{\hm \omega}{\syn \omega} = \dfrac{0,6}{0,8} = \dfrac{3}{4}$ και $\syf \omega = \dfrac{\syn \omega}{\hm \omega} = \dfrac{0,8}{0,6} = \dfrac{4}{3}$. \\
Για την παράσταση A με αναγωγή στο πρώτο τεταρτημόριο έχουμε
\begin{align*}
\text{A} &= \hm (\pi-\omega) - 2\syn \left(\frac{\pi}{2}-\omega\right) + \ef (-\omega) + \syf (\pi+\omega) = \\
&= \hm \omega - 2\hm \omega - \ef \omega + \syf \omega = \\
&= -\hm \omega - \ef \omega + \syf \omega = \\
&= -\frac{3}{5} - \frac{3}{4} + \frac{4}{3} = \\
&= -\frac{36}{60} - \frac{45}{60} + \frac{80}{60} = -\frac{1}{60}
\end{align*}
\end{rlist}
\item Για τις τιμές που βρήκαμε στο προηγούμενο ερώτημα η συνάρτηση $f$ γίνεται $f(x) = 5\cdot 0,8x^3 - 10\cdot 0,6x^2 + 5x - 3 = 4x^3 - 6x^2 + 5x - 3$. Ζητάμε τις λύσεις της ανίσωσης $f(x) < 0 \Leftrightarrow 4x^3 - 6x^2 + 5x - 3 < 0$. Με τη βοήθεια του σχήματος Horner βρίσκουμε ότι $4x^3 - 6x^2 + 5x - 3 = (x-1)(4x^2 - 2x + 3)$. \\
Επομένως, λύνουμε την ανίσωση $(x-1)(4x^2 - 2x + 3) < 0 \Leftrightarrow x-1 < 0 \Leftrightarrow x < 1$, αφού $4x^2 - 2x + 3 > 0$ για κάθε πραγματικό αριθμό $x$ καθώς εύκολα αποδεικνύεται ότι η διακρίνουσά του είναι αρνητική και δεν έχει ρίζες. Επομένως η γραφική παράσταση της συνάρτησης $f$ βρίσκεται κάτω από τον άξονα $x'x$ όταν $x<1$, δηλαδή στο διάστημα $(-\infty, 1)$.
\end{alist}
