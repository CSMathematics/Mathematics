\lysh{21091}{2}
\begin{alist}
\item 
\begin{rlist}
\item Με βάση την γραφική παράσταση της $f$ έχουμε τον παρακάτω πίνακα τιμών:
\begin{center}
\begin{myhtblr}{}
$x$ & $-1$ & $0$ & $1$ & $2$ & $3$ \\
$f(x)$ & $\frac{1}{2}$ & $1$ & $2$ & $4$ & $8$ \\
\end{myhtblr}
\end{center}
\item Από τον παραπάνω πίνακα τιμών και με δεδομένο ότι η συνάρτηση είναι εκθετική παρατηρούμε ότι $f(-1) = \dfrac{1}{2} = 2^{-1}$, $f(0) = 1 = 2^0$, $f(1) = 2 = 2^1$, $f(2) = 4 = 2^2$ και $f(3) = 8 = 2^3$. Άρα η εκθετική συνάρτηση έχει τύπο $f(x) = 2^x$.
\end{rlist}
\item Έχουμε $f(x) = 32$, δηλαδή $2^x = 32$, οπότε $2^x = 2^5$ και τελικά $x = 5$.
\end{alist}
