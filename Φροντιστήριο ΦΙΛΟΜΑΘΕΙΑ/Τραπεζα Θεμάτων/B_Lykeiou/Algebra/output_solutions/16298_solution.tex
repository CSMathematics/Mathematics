{\begin{alist}
\item Επειδή $\syn \omega = -\dfrac{1}{2}$ και $\hm \omega > 0$, η γωνία θα βρίσκεται στο $2^0$ τεταρτημόριο του τριγωνομετρικού κύκλου. \\
Το μέτρο της γωνίας είναι $\omega = \dfrac{2\pi}{3}$ διότι $\syn \omega = -\dfrac{1}{2} \Leftrightarrow \syn \omega = -\syn \left(\dfrac{\pi}{3}\right) = \syn \left(\pi - \dfrac{\pi}{3}\right) = \syn \dfrac{2\pi}{3}$.
\item Για τη τριγωνομετρική εξίσωση για $\varphi \in \mathbb{R}$ έχουμε:
\begin{align*}
\syn \varphi = -\frac{1}{2} &\Leftrightarrow \syn \varphi = -\syn \frac{\pi}{3} = \syn \left(\pi - \frac{\pi}{3}\right) = \syn \frac{2\pi}{3}\Leftrightarrow\\& \Leftrightarrow \syn \varphi = \syn \frac{2\pi}{3}
\end{align*}

τότε $\varphi = 2\kappa\pi + \dfrac{2\pi}{3}$ ή $\varphi = 2\kappa\pi - \dfrac{2\pi}{3}$ με $\kappa \in \mathbb{Z}$.
\end{alist}
