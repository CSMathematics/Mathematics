\begin{alist}
\item Στο σχήμα α η γραφική παράσταση έχει άξονα συμμετρίας τον $yy'$ οπότε είναι άρτια και όχι περιττή, οπότε δεν είναι. Στο σχήμα γ η συνάρτηση είναι γνησίως αύξουσα οπότε δεν είναι. Στο σχήμα δ η συνάρτηση είναι μεν περιττή και γνησίως φθίνουσα, αλλά έχει πεδίο ορισμού το $[-2, 2]$ και όχι το $[-1, 1]$, οπότε δεν είναι.
Συνεπώς σωστή απάντηση είναι το σχήμα β.

\item Η γραφική παράσταση της $g(x) = f(x) + 2$ προκύπτει με κατακόρυφη μετατόπιση της $f(x)$ κατά 2 μονάδες προς τα πάνω. Η γραφική της παράσταση φαίνεται παρακάτω.

\begin{center}
\begin{tikzpicture}[scale=1.5]
\draw[step=0.5,gray!30,very thin] (-1.2,0.8) grid (1.2,3.2);
\draw[->] (-1.3,0) -- (1.3,0);
\draw[->] (0,0.8) -- (0,3.3);
\foreach \y in {1, 1.5, 2, 2.5, 3} {
  \draw (1pt, \y) -- (-1pt, \y) node[anchor=east] {\scriptsize \y};
}
\draw[purple!60, thick, domain=-1:1, samples=100] plot (\x, {-\x*\x*\x + 2});
\end{tikzpicture}
\end{center}

\item Η γραφική παράσταση της συνάρτησης $h(x) = f(x - 1)$ προκύπτει με οριζόντια μετατόπιση της $f(x)$ κατά 1 μονάδα προς τα δεξιά. Η γραφική της παράσταση φαίνεται παρακάτω.

\begin{center}
\begin{tikzpicture}[x=1.5cm,y=1.5cm]
\draw[step=0.5,gray!30,very thin] (-0.2,-1.2) grid (2.2,1.2);
\draw[->] (-0.3,0) -- (2.3,0);
\draw[->] (0,-1.3) -- (0,1.3);
\foreach \x in {0.5, 1, 1.5, 2} {
  \draw (\x, 1pt) -- (\x, -1pt) node[anchor=north] {\scriptsize \x};
}
\foreach \xtext/\x in {-1/-1, -0.5/-0.5, 0/0} {
  \draw (\x, 1pt) -- (\x, -1pt) node[anchor=north] {\scriptsize \xtext};
}
\foreach \y in {-1, -0.5, 0.5, 1} {
  \draw (1pt, \y) -- (-1pt, \y) node[anchor=east] {\scriptsize \y};
}

\draw[purple!60, thick, domain=0:2, samples=100] plot (\x, {-(\x-1)*(\x-1)*(\x-1)});
\node[black] at (0.2, 0.5) {$h$};
\end{tikzpicture}
\end{center}
\end{alist}
