\lysh{15617}{2}
\begin{alist}
\item Ισχύει ότι
\[f(x) = \ln\frac{1}{|x|} = \ln 1 - \ln|x| = 0 - \ln|x| = -\ln|x|.\]
\item 
\begin{rlist}
\item Η γραφική παράσταση της $f(x) = -\ln|x|$ είναι η συμμετρική της γραφικής παράστασης της $h(x) = \ln|x|$ ως προς τον άξονα $x'x$ και φαίνεται στο επόμενο σχήμα.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-7, xmax=7,
    ymin=-3, ymax=3,
    xtick={-7,-6,...,7},
    ytick={-3,-2,-1,1,2,3},
    ticklabel style={font=\tiny},
    width=10cm, height=8cm,
    samples=100,
    grid=both,
    grid style={line width=.1pt, draw=gray!10},
    major grid style={line width=.2pt,draw=gray!30}
]
\addplot[domain=0.04:7,very thick, maincolor] {ln(x)};
\addplot[domain=-7:-0.04, thick, maincolor] {ln(-x)} node[pos=0.05, above] {$h$};
\addplot[domain=0.04:7, thick, secondarycolor] {-ln(x)};
\addplot[domain=-7:-0.04, thick, secondarycolor] {-ln(-x)} node[pos=0.05, above] {$f$};
\end{axis}
\end{tikzpicture}
\end{center}
\item Οι γραφικές παραστάσεις των $f, g$ έχουν μοναδικό κοινό σημείο, διότι, πρέπει να ισχύει:
\[f(x) = g(x) \Leftrightarrow -\ln|x| = \ln x, x > 0 \Leftrightarrow -\ln x = \ln x \Leftrightarrow 0 = 2\ln x \Leftrightarrow \ln x = 0 \Leftrightarrow x = 1.\]
Αυτό φαίνεται και στο επόμενο σχήμα:

\end{axis}
\end{tikzpicture}
\end{center}
\end{rlist}
\end{alist}
