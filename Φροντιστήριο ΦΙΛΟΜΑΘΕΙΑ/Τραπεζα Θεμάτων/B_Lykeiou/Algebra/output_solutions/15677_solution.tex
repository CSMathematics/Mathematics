\begin{alist}
\item Η διαίρεση $P(x) : Q(x)$ φαίνεται στο παρακάτω σχήμα.
\begin{center}
\begin{tabular}{r|l}
$x^4 - 2x^3 - x^2 + \alpha x + \beta$ & $x^2 - 2x + 1$ \\ \cline{2-2}
$-x^4 + 2x^3 - x^2 \phantom{{} + \alpha x + \beta}$ & $x^2 - 2$ \\ \cline{1-1}
$-2x^2 + \alpha x + \beta$ & \\
$2x^2 - 4x + 2$ & \\ \cline{1-1}
$(\alpha - 4)x + \beta + 2$ &
\end{tabular}
\end{center}
Αφού το $P(x)$ διαιρείται με το πολυώνυμο $Q(x) = x^2 - 2x + 1$, θα πρέπει το υπόλοιπο της παραπάνω διαίρεσης να είναι το μηδενικό πολυώνυμο, που συμβαίνει μόνο όταν $\alpha - 4 = 0$ και $\beta + 2 = 0$, δηλαδή $\alpha = 4$ και $\beta = -2$.

\item Για $\alpha = 4, \beta = -2$ έχουμε $P(x) = x^4 - 2x^3 - x^2 + 4x - 2$.
\begin{rlist}
\item Η διαίρεση $P(x) : (x^2 + 5)$ φαίνεται στο παρακάτω σχήμα
\begin{center}
\begin{tabular}{r|l}
$x^4 - 2x^3 - x^2 + 4x - 2$ & $x^2 + 5$ \\ \cline{2-2}
$-x^4 - 5x^2 \phantom{{} + 4x - 2}$ & $x^2 - 2x - 6$ \\ \cline{1-1}
$-2x^3 - 6x^2 + 4x - 2$ & \\
$2x^3 + 10x \phantom{{} - 2}$ & \\ \cline{1-1}
$-6x^2 + 14x - 2$ & \\
$6x^2 + 30 \phantom{{} - 2}$ & \\ \cline{1-1}
$14x + 28$ &
\end{tabular}
\end{center}
και η ταυτότητα της διαίρεσης είναι η εξής: $P(x) = (x^2 + 5)(x^2 - 2x - 6) + 14x + 28$.

\item Η εξίσωση $P(x) = 14(x + 2)$ με τη βοήθεια της παραπάνω ταυτότητας γίνεται
\begin{align*}
(x^2 + 5)(x^2 - 2x - 6) + 14x + 28 &= 14(x + 2) \Leftrightarrow \\
(x^2 + 5)(x^2 - 2x - 6) + 14x + 28 &= 14x + 28 \Leftrightarrow \\
(x^2 + 5)(x^2 - 2x - 6) &= 0
\end{align*}
και επειδή $x^2 + 5 \ne 0$ για κάθε $x \in \mathbb{R}$, έχουμε ότι $x^2 - 2x - 6 = 0$ δηλαδή $x = \dfrac{2 \pm \sqrt{28}}{2} = \dfrac{2 \pm 2\sqrt{7}}{2} = 1 \pm \sqrt{7}$.
\end{rlist}
\end{alist}
