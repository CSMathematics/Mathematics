\item 
\begin{rlist}
\item Έχουμε:
\begin{align*}
x(e^x - 1) = 0 &\Leftrightarrow x = 0 \text{ ή } e^x - 1 = 0 \\
&\Leftrightarrow x = 0 \text{ ή } e^x = 1 \\
&\Leftrightarrow x = 0 \text{ ή } e^x = e^0 \\
&\Leftrightarrow x = 0
\end{align*}
\item Από το ερώτημα (i) γνωρίζουμε ότι κάθε παράγοντας του γινομένου $x(e^x - 1)$ έχει ρίζα το $0$. Ακόμη είναι $e^x - 1 > 0 \Leftrightarrow e^x > 1 \Leftrightarrow e^x > e^0 \Leftrightarrow x > 0$ και $e^x - 1 < 0 \Leftrightarrow x < 0$. \\
Ο προσδιορισμός του προσήμου του γινομένου $x(e^x - 1)$ γίνεται με τη βοήθεια του παρακάτω πίνακα:

\end{rlist}
\item 
\begin{rlist}
\item Η συνάρτηση $f$ ορίζεται για εκείνες τις τιμές του $x$, για τις οποίες ισχύει $x(e^x - 1) \ge 0$. Από το ερώτημα (α) προκύπτει ότι $x(e^x - 1) \ge 0 \Leftrightarrow x \in \mathbb{R}$. Άρα το πεδίο ορισμού της συνάρτησης $f$ είναι το $\mathbb{R}$.
\item Είναι
\begin{align*}
f(0) &= \sqrt{0\cdot(e^0-1)} = 0 \\
f(\ln 2) &= \sqrt{\ln 2\cdot(e^{\ln 2}-1)} = \sqrt{\ln 2\cdot(2-1)} = \sqrt{\ln 2} \\
f(-\ln 2) &= \sqrt{-\ln 2\cdot(e^{-\ln 2}-1)} = \sqrt{-\ln 2\cdot\left(e^{\ln \frac{1}{2}}-1\right)} = \sqrt{-\ln 2\cdot\left(\frac{1}{2}-1\right)} = \sqrt{\frac{\ln 2}{2}}
\end{align*}
\item Επειδή ισχύει $-\ln 2 < 0$ και $f(-\ln 2) > f(0)$ συμπεραίνουμε ότι η συνάρτηση $f$ δεν είναι γνησίως αύξουσα στο $\mathbb{R}$.
\end{rlist}
\end{alist}
