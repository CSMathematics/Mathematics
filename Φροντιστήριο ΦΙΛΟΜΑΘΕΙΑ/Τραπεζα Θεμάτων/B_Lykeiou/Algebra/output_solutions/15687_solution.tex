\lysh{15687}{2}
\begin{alist}
\item Από τις ιδιότητες των λογαρίθμων, έχουμε:
\[A = \log_4 3 + \log_4 \alpha - \log_4 \beta = \log_4(3\alpha) - \log_4 \beta = \log_4\frac{3\alpha}{\beta}.\]
\item Η παράσταση $A$ με $3\alpha = 16\beta$, γράφεται:
\[A = \log_4\frac{16\beta}{\beta} = \log_4 16 = \log_4 4^2 = 2\log_4 4 = 2.\]
\end{alist}
