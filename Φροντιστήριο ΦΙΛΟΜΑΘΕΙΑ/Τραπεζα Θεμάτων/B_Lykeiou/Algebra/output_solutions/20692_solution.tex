\begin{alist}
\item $f(100) = \log 100 = 2$ διότι η βάση του λογαρίθμου είναι το $10$, άρα από τον ορισμό έχουμε $10^2 = 100$.
\[f(\sqrt{10}) = \log(\sqrt{10}) = \log\left(10^{1/2}\right) = \frac{1}{2}.\]

\item Η εξίσωση γράφεται 
\begin{align*}
\log(x + 1) + \log(x - 1) = \log 10 - \log 5 &\Leftrightarrow \\
\log[(x + 1)(x - 1)] = \log\left(\frac{10}{5}\right) &\Leftrightarrow \log(x^2 - 1^2) = \log 2.
\end{align*}
Ώστε $x^2 - 1 = 2 \Leftrightarrow x^2 = 3$. Αλλά $x > 1$, οπότε $x = \sqrt{3}$. (η λύση $x = -\sqrt{3}$ απορρίπτεται).
\end{alist}
