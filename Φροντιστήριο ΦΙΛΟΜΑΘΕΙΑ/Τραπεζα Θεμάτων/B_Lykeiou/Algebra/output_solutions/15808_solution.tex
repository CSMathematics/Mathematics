\lysh{15808}{2}
\begin{alist}
\item Η συνάρτηση $f$ ορίζεται για όλες τις πραγματικές τιμές του $x$ για τις οποίες ισχύει:
\[x + 2 > 0 \Leftrightarrow x > -2\]
Συνεπώς το πεδίο ορισμού της $f$ είναι το $(-2, +\infty)$.
\item Η τετμημένη του σημείου τομής της γραφικής παράστασης της $f$ με τον άξονα $x'x$, είναι η λύση της εξίσωσης:
\begin{align*}
f(x) &= 0 \Leftrightarrow \\
\ln(x + 2) &= 0 \overset{\ln x = y \Leftrightarrow e^y = x}{\Leftrightarrow} \\
x + 2 &= e^0 \Leftrightarrow \\
x + 2 &= 1 \Leftrightarrow \\
x &= -1.
\end{align*}
Συνεπώς το σημείο τομής της γραφικής παράστασης της $f$ με τον άξονα $x'x$ είναι το $(-1, 0)$.
\item Η γραφική παράσταση της $f(x) = g(x + 2) = \ln(x + 2)$ θα προκύψει από μετατόπιση της γραφικής παράστασης της $g$ κατά δυο μονάδες αριστερά, όπως φαίνεται στο παρακάτω σχήμα.

\end{center}
\end{alist}
