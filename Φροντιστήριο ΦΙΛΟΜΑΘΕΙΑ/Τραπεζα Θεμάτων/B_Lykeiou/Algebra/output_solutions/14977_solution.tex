\begin{alist}
\item Το ημίτονο μίας γωνίας με μία πλευρά την ημιευθεία Ox και δεύτερη πλευρά που τέμνει στο σημείο Α του κύκλου, βρίσκεται από την προβολή του σημείου Α στον άξονα y'y.

Άρα, οι γωνίες που έχουν ημίτονο $\dfrac{1}{2}$ θα έχουν σημεία τομής με τον κύκλο τέτοια, ώστε η προβολή τους να τέμνει τον άξονα y'y στο $\dfrac{1}{2}$.

Οπότε φέρουμε την ευθεία $y = \dfrac{1}{2}$ (βλ. σχήμα παρακάτω) και προκύπτουν δύο γωνίες στο διάστημα $[0, 2\pi)$ που έχουν αυτό το ημίτονο. Είναι οι γωνίες $\alpha = 30^\circ = \dfrac{\pi}{6}\,\text{rad}$ και $\beta = 180^\circ - 30^\circ = 150^\circ = \dfrac{5\pi}{6}\,\text{rad}$.

\begin{center}
\begin{tikzpicture}[scale=3]
% Draw circle
\draw[thick] (0,0) circle (1cm);
% Draw axes
\draw[->] (-1.2,0) -- (1.2,0);
\draw[->] (0,-1.2) -- (0,1.2);
% Draw grid-like markers
\foreach \x/\xtext in {-1/-1,-0.8/-0.8,-0.6/-0.6,-0.4/-0.4,-0.2/-0.2,0.2/0.2,0.4/0.4,0.6/0.6,0.8/0.8,1/1} {
  \draw (\x, 1pt) -- (\x, -1pt) node[below] {\scriptsize \xtext};
}
\foreach \y/\ytext in {-1/-1,-0.8/-0.8,-0.6/-0.6,-0.4/-0.4,-0.2/-0.2,0.4/0.4,0.6/0.6,0.8/0.8,1/1} {
  \draw (1pt, \y) -- (-1pt, \y) node[left] {\scriptsize \ytext};
}
\node[below left] at (0,0) {\scriptsize $0$};

% Draw y=1/2 line
\draw[thick] (-1.1,0.5) -- (1.1,0.5);
\node[above] at (0.1, 0.5) {$\frac{1}{2}$};

% Points and angles
\coordinate (A) at (0.866, 0.5);
\coordinate (B) at (-0.866, 0.5);
\draw[thick] (0,0) -- (A) node[above right] {$A$};
\draw[thick] (0,0) -- (B);

\filldraw[blue!70!white, draw=black] (A) circle (0.8pt);
\filldraw[blue!70!white, draw=black] (B) circle (0.8pt);
\filldraw[blue!70!white, draw=black] (0,0) circle (0.8pt) node[above right, text=blue] {O};

% Add points around the circle
\foreach \a in {0,30,60,90,120,150,180,210,240,270,300,330} {
  \filldraw[blue!70!white, draw=black] (\a:1cm) circle (0.8pt);
}

% Arcs
\draw[green!60!black, thick, domain=0:30] plot ({0.3*cos(\x)}, {0.3*sin(\x)});
\node[green!60!black] at (15:0.4) {$\alpha=30^\circ$};

\draw[green!60!black, thick, domain=0:150] plot ({0.2*cos(\x)}, {0.2*sin(\x)});
\node[green!60!black] at (75:0.3) {$\beta=150^\circ$};

\end{tikzpicture}
\end{center}
\end{alist}
