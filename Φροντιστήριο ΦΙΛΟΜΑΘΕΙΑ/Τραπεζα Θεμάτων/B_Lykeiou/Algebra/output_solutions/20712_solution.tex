\lysh{20712}{4}
\begin{alist}
\item Γραφικά η υψομετρική διαφορά ανάμεσα στην υψηλότερη πλημμυρίδα και τη χαμηλότερη άμπωτη, εκφράζεται με τη διαφορά του ελαχίστου $-3$ της συνάρτησης $f$, από το μέγιστό της $3$. Συνεπώς η ζητούμενη υψομετρική διαφορά είναι $6$ μέτρα.
\item Από το σχήμα παρατηρούμε ότι το μικρότερο διάστημα που απαιτείται για να αρχίσει να επαναλαμβάνεται η γραφική παράσταση είναι $12$ ώρες. Συνεπώς η ζητούμενη περίοδος είναι $12$.
\item Ο τύπος της ημιτονοειδούς συνάρτησης $f$ είναι της μορφής $f(t) = \rho \cdot \hm (\omega t)$ όπου $\rho > 0, \omega > 0$. Δεδομένου ότι η μέγιστη τιμή της $f$ είναι $3$ συμπεραίνουμε ότι $\rho = 3$. Επίσης η περίοδος είναι $12$ οπότε $\dfrac{2\pi}{\omega} = 12 \Leftrightarrow \omega = \dfrac{\pi}{6}$. Συνεπώς η συνάρτηση $f$ έχει τύπο $f(t) = 3 \cdot \hm \left(\dfrac{\pi}{6}\cdot t\right)$.
\item Αναζητούμε τις λύσεις της εξίσωσης $f(t) = \dfrac{3}{2}$, όπου $0 \le t \le 24$. Είναι
\begin{align*}
f(t) = \frac{3}{2} &\Leftrightarrow 3 \cdot \hm \left(\frac{\pi}{6} \cdot t\right) = \frac{3}{2} \Leftrightarrow \hm \left(\frac{\pi}{6} \cdot t\right) = \frac{1}{2} \\
&\Leftrightarrow \hm \left(\frac{\pi}{6} \cdot t\right) = \hm \left(\frac{\pi}{6}\right) \Leftrightarrow \begin{cases} \dfrac{\pi}{6} \cdot t = 2\kappa\pi + \dfrac{\pi}{6} \\ \hspace{1cm} \text{ή} \\ \dfrac{\pi}{6} \cdot t = 2\kappa\pi + \pi - \dfrac{\pi}{6} \end{cases} \\
&\Leftrightarrow \begin{cases} t = 12\kappa + 1 \\ \hspace{0.5cm} \text{ή} \\ t = 12\kappa + 5 \end{cases}, \kappa \in \mathbb{Z}
\end{align*}
Επειδή $0 \le t \le 24$, έχουμε τελικά ότι οι ζητούμενες ώρες είναι $1, 5, 13, 17$. \\
Σχόλιο: Αυτό επιβεβαιώνεται και γραφικά από τις τετμημένες των σημείων τομής της γραφικής παράστασης της $f$, με την ευθεία $y = \dfrac{3}{2}$.
\end{alist}
