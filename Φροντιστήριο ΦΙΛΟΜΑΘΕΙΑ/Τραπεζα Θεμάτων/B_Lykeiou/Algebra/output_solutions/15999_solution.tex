\begin{alist}
\item Είναι γνωστό ότι $\sigma\upsilon\nu\left(\dfrac{\pi}{2} - \theta\right) = \eta\mu\theta$ και $\eta\mu(-\theta) = -\eta\mu\theta$, οπότε έχουμε:
\[A = 2\sigma\upsilon\nu\left(\frac{\pi}{2} - \theta\right) + \eta\mu(-\theta) = 2\eta\mu\theta - \eta\mu\theta = \eta\mu\theta\]
που είναι το ζητούμενο.

\item Από την ταυτότητα $\eta\mu^2\theta + \sigma\upsilon\nu^2\theta = 1$, με $\sigma\upsilon\nu\theta = \dfrac{12}{13}$ έχουμε:
\[\eta\mu^2\theta = 1 - \sigma\upsilon\nu^2\theta = 1 - \frac{144}{169} = \frac{25}{169}\]
Αλλά $\theta \in \left(\dfrac{3\pi}{2}, 2\pi\right)$, οπότε $\eta\mu\theta < 0$, άρα:
\[\eta\mu\theta = -\sqrt{\frac{25}{169}} = -\frac{5}{13}\]
\end{alist}
