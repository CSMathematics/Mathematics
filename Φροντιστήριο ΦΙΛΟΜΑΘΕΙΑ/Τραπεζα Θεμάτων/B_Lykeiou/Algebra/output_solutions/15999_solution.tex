\lysh{15999}{2}
\begin{alist}
\item Είναι γνωστό ότι $\syn \left(\dfrac{\pi}{2} - \theta\right) = \hm \theta$ και $\hm (-\theta) = -\hm \theta$, οπότε έχουμε:
\[A = 2\syn \left(\frac{\pi}{2} - \theta\right) + \hm (-\theta) = 2\hm \theta - \hm \theta = \hm \theta\]
που είναι το ζητούμενο.
\item Από την ταυτότητα $\hm ^2\theta + \syn ^2\theta = 1$, με $\syn \theta = \dfrac{12}{13}$ έχουμε:
\[\hm ^2\theta = 1 - \syn ^2\theta = 1 - \frac{144}{169} = \frac{25}{169}\]
Αλλά $\theta \in \left(\dfrac{3\pi}{2}, 2\pi\right)$, οπότε $\hm \theta < 0$, άρα:
\[\hm \theta = -\sqrt{\frac{25}{169}} = -\frac{5}{13}\]
\end{alist}
