\lysh{21994}{2}
\begin{alist}
\item Η αρχική ποσότητα του ραδιενεργού υλικού είναι το υψόμετρο του σημείου της γραφικής παράστασης που αντιστοιχεί στο $t=0$, δηλαδή η τεταγμένη του σημείου $\text{A}$. Οπότε, η αρχική ποσότητα του υλικού είναι $6$ (γραμμάρια).
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    grid=both,
    xmin=0, xmax=7.5,
    ymin=0, ymax=6.5,
    xtick={0,1,2,3,4,5,6,7},
    ytick={0,1,2,3,4,5,6},
    grid style={line width=.1pt, draw=gray!30},
    major grid style={line width=.2pt,draw=gray!50},
    thick,
    xlabel={$t$},
    ylabel={gr},
    width=10cm, height=8cm,
    samples=100
]
\addplot[domain=0:6.5, maincolor, ultra thick] {6*exp(-0.3465*x)}; % c approx ln(2)/2
\fill[black] (axis cs:0, 6) circle (2pt) node[above right] {\small $\text{A}$};
\fill[black] (axis cs:2, 3) circle (2pt) node[above right] {\small $\text{B}$};
\fill[black] (axis cs:6, 0.75) circle (2pt) node[above right] {\small $\Gamma$};
\end{axis}
\end{tikzpicture}
\end{center}
\item Η ημιζωή του ραδιενεργού υλικού είναι η τιμή του χρόνου $t$ κατά την οποία η ποσότητα του υλικού μειώνεται στο ήμισυ της αρχικής, δηλαδή γίνεται $3$ γραμμάρια. Είναι συνεπώς, η τετμημένη του σημείου $\text{B}$, δηλαδή $t=2$.
\item Το υψόμετρο της καμπύλης γίνεται μικρότερο του $1$ στο χρονικό διάστημα $(5,6)$, δηλαδή κατά τη διάρκεια της $6^{\eta\varsigma}$ ημέρας. Άρα, κατά τη διάρκεια της $6^{\eta\varsigma}$ ημέρας θα έχει απομείνει ποσότητα ραδιενεργού υλικού μικρότερη από $1\text{gr}$.
\end{alist}
