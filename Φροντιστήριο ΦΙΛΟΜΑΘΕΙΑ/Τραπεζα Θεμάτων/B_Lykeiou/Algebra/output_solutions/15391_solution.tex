\begin{alist}
\item Τα σημεία Α και Β αντιστοιχούν στις λύσεις της εξίσωσης $\eta\mu x = \sigma\upsilon\nu x$ για $x \in [0, 2\pi]$. \\
Γνωρίζουμε ότι $\eta\mu\dfrac{\pi}{4} = \sigma\upsilon\nu\dfrac{\pi}{4} = \dfrac{\sqrt{2}}{2}$, ενώ $\eta\mu\dfrac{5\pi}{4} = \eta\mu\left(\pi + \dfrac{\pi}{4}\right) = -\eta\mu\dfrac{\pi}{4} = -\dfrac{\sqrt{2}}{2}$ και $\sigma\upsilon\nu\dfrac{5\pi}{4} = \sigma\upsilon\nu\left(\pi + \dfrac{\pi}{4}\right) = -\sigma\upsilon\nu\dfrac{\pi}{4} = -\dfrac{\sqrt{2}}{2}$, έχουμε ότι $A\left(\dfrac{\pi}{4}, \dfrac{\sqrt{2}}{2}\right)$ και $B\left(\dfrac{5\pi}{4}, -\dfrac{\sqrt{2}}{2}\right)$.

\item Η συνάρτηση $g(x) = \sigma\upsilon\nu x$ είναι γνησίως φθίνουσα στο διάστημα $\left[\dfrac{\pi}{2}, \pi\right]$ και η $f(x) = \eta\mu x$ είναι γνησίως αύξουσα στο διάστημα $\left[\dfrac{3\pi}{2}, 2\pi\right]$.

\item 
\begin{rlist}
\item Παρατηρούμε ότι $\dfrac{\pi}{2} < \dfrac{2\pi}{3} = \dfrac{4\pi}{6} < \dfrac{5\pi}{6} < \pi$ και αφού η $g(x) = \sigma\upsilon\nu x$ είναι γνησίως φθίνουσα στο διάστημα $\left[\dfrac{\pi}{2}, \pi\right]$, άρα $\sigma\upsilon\nu\dfrac{2\pi}{3} > \sigma\upsilon\nu\dfrac{5\pi}{6}$.
\item Ανάλογα, είναι $\dfrac{3\pi}{2} < \dfrac{5\pi}{3} = \dfrac{10\pi}{6} < \dfrac{11\pi}{6} < 2\pi$ και αφού η $f(x) = \eta\mu x$ είναι γνησίως αύξουσα στο διάστημα $\left[\dfrac{3\pi}{2}, 2\pi\right]$, άρα $\eta\mu\dfrac{5\pi}{3} < \eta\mu\dfrac{11\pi}{6}$.
\end{rlist}
\end{alist}
