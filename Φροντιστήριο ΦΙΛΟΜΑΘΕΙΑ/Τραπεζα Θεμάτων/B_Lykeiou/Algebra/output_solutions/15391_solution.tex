\lysh{15391}{3}
\begin{alist}
\item Τα σημεία Α και Β αντιστοιχούν στις λύσεις της εξίσωσης $\hm  x = \syn  x$ για $x \in [0, 2\pi]$. \\
Γνωρίζουμε ότι $\hm \dfrac{\pi}{4} = \syn \dfrac{\pi}{4} = \dfrac{\sqrt{2}}{2}$, ενώ $\hm \dfrac{5\pi}{4} = \hm \left(\pi + \dfrac{\pi}{4}\right) = -\hm \dfrac{\pi}{4} = -\dfrac{\sqrt{2}}{2}$ και $\syn \dfrac{5\pi}{4} = \syn \left(\pi + \dfrac{\pi}{4}\right) = -\syn \dfrac{\pi}{4} = -\dfrac{\sqrt{2}}{2}$, έχουμε ότι $A\left(\dfrac{\pi}{4}, \dfrac{\sqrt{2}}{2}\right)$ και $B\left(\dfrac{5\pi}{4}, -\dfrac{\sqrt{2}}{2}\right)$.
\item Η συνάρτηση $g(x) = \syn  x$ είναι γνησίως φθίνουσα στο διάστημα $\left[\dfrac{\pi}{2}, \pi\right]$ και η $f(x) = \hm  x$ είναι γνησίως αύξουσα στο διάστημα $\left[\dfrac{3\pi}{2}, 2\pi\right]$.
\item 
\begin{rlist}
\item Παρατηρούμε ότι $\dfrac{\pi}{2} < \dfrac{2\pi}{3} = \dfrac{4\pi}{6} < \dfrac{5\pi}{6} < \pi$ και αφού η $g(x) = \syn  x$ είναι γνησίως φθίνουσα στο διάστημα $\left[\dfrac{\pi}{2}, \pi\right]$, άρα $\syn \dfrac{2\pi}{3} > \syn \dfrac{5\pi}{6}$.
\item Ανάλογα, είναι $\dfrac{3\pi}{2} < \dfrac{5\pi}{3} = \dfrac{10\pi}{6} < \dfrac{11\pi}{6} < 2\pi$ και αφού η $f(x) = \hm  x$ είναι γνησίως αύξουσα στο διάστημα $\left[\dfrac{3\pi}{2}, 2\pi\right]$, άρα $\hm \dfrac{5\pi}{3} < \hm \dfrac{11\pi}{6}$.
\end{rlist}
\end{alist}
