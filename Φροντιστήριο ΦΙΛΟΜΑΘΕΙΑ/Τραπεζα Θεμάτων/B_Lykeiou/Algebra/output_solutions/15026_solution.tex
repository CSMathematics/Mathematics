\lysh{15026}{4}
\begin{alist}
\item Η περίοδος της συνάρτησης $f$ είναι $T = \dfrac{2\pi}{\dfrac{\pi}{2}} = 4$.
\item Για κάθε $x \in \mathbb{R}$ είναι
\[-1 \le \hm \left(\dfrac{\pi x}{2}\right) \le 1 \text{ οπότε } -2 \le 2\hm \left(\dfrac{\pi x}{2}\right) \le 2 \text{ οπότε } -2 + 1 \le 1 + 2\hm \left(\dfrac{\pi x}{2}\right) \le 2 + 1 \text{ και τελικά}\]
\[-1 \le f(x) \le 3.\]
Επίσης $f(1) = 1 + 2\hm \left(\dfrac{\pi}{2}\right) = 3$ και $f(3) = 1 + 2\hm \left(\dfrac{3\pi}{2}\right) = -1$.
Συνεπώς η μέγιστη τιμή της $f$ είναι το $3$ και η ελάχιστη το $-1$.
\item Οι τετμημένες των σημείων στα οποία η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ είναι οι λύσεις της εξίσωσης $f(x) = 0$. Είναι
\[f(x) = 0 \Leftrightarrow 1 + 2\hm \left(\dfrac{\pi x}{2}\right) = 0 \Leftrightarrow \hm \left(\dfrac{\pi x}{2}\right) = -\dfrac{1}{2} \Leftrightarrow \hm \left(\dfrac{\pi x}{2}\right) = \hm \left(-\dfrac{\pi}{6}\right) \Leftrightarrow\]
\[\begin{cases} \dfrac{\pi x}{2} = 2\kappa\pi + \left(-\dfrac{\pi}{6}\right) \\ \\ \quad\quad \eta' \\ \dfrac{\pi x}{2} = 2\kappa\pi + \pi - \left(-\dfrac{\pi}{6}\right) \end{cases} \Leftrightarrow \begin{cases} \dfrac{\pi x}{2} = 2\kappa\pi - \dfrac{\pi}{6} \\ \\ \quad\quad \eta' \\ \dfrac{\pi x}{2} = 2\kappa\pi + \dfrac{7\pi}{6} \end{cases} \Leftrightarrow \begin{cases} \dfrac{x}{2} = 2\kappa - \dfrac{1}{6} \\ \\ \quad\quad \eta' \\ \dfrac{x}{2} = 2\kappa + \dfrac{7}{6} \end{cases} \Leftrightarrow \begin{cases} x = 4\kappa - \dfrac{1}{3} \\ \\ \quad\quad \eta' \\ x = 4\kappa + \dfrac{7}{3} \end{cases}, \quad \kappa \in \mathbb{Z}\]
οπότε οι ζητούμενες τετμημένες είναι όλοι οι πραγματικοί αριθμοί της μορφής $x = 4\kappa - \dfrac{1}{3}$ ή $x = 4\kappa + \dfrac{7}{3}$, όπου $\kappa \in \mathbb{Z}$.
\item Για κάθε $x \in \mathbb{R}$ είναι
\[f(1 - x) = 1 + 2\hm \left(\dfrac{\pi(1 - x)}{2}\right) = 1 + 2\hm \left(\dfrac{\pi - \pi x}{2}\right) = 1 + 2\hm \left(\dfrac{\pi}{2} - \dfrac{\pi x}{2}\right) = 1 + 2\syn \left(\dfrac{\pi x}{2}\right), \text{ οπότε }\]
\[(f(x) - 1)^2 + (f(1 - x) - 1)^2 = \left(1 + 2\hm \left(\dfrac{\pi x}{2}\right) - 1\right)^2 + \left(1 + 2\syn \left(\dfrac{\pi x}{2}\right) - 1\right)^2 =\]
\[4\hm ^2\left(\dfrac{\pi x}{2}\right) + 4\syn ^2\left(\dfrac{\pi x}{2}\right) = 4\left(\hm ^2\left(\dfrac{\pi x}{2}\right) + \syn ^2\left(\dfrac{\pi x}{2}\right)\right) = 4 \cdot 1 = 4\]
δηλαδή $(f(x) - 1)^2 + (f(1 - x) - 1)^2 = 4$, για κάθε $x \in \mathbb{R}$.
\end{alist}
