\lysh{15191}{2}
\begin{alist}
\item Δύο γωνίες σχεδιασμένες στον τριγωνομετρικό κύκλο είναι αντίθετες όταν έχουν τα σημεία τομής της τελικής τους πλευράς με τον τριγωνομετρικό κύκλο συμμετρικά ως προς τον άξονα $x'x$. Άρα η γωνία $-\hat{\omega}$ είναι όπως φαίνεται με διακεκομμένη γραμμή στο παρακάτω σχήμα.
\begin{center}
\begin{tikzpicture}[scale=3]
\draw[step=0.2,gray!30,very thin] (-1.1,-1.1) grid (1.1,1.1);
\draw[->] (-1.2,0) -- (1.2,0) node[right, font=\scriptsize] {$x$};
\draw[->] (0,-1.2) -- (0,1.2) node[above, font=\scriptsize] {$y$};
\draw (0,0) circle (1);
\foreach \x in {-1, -0.8, -0.6, -0.4, -0.2, 0.2, 0.4, 0.6, 0.8, 1}
    \node[below, font=\tiny] at (\x,0) {\x};
\foreach \y in {-0.8, -0.6, -0.4, -0.2, 0.2, 0.4, 0.6, 0.8, 1}
    \node[left, font=\tiny] at (0,\y) {\y};
% The angle \omega is such that sin(\omega) = 0.4 and cos(\omega) < 0.
% So cos(\omega) = -sqrt(1 - 0.16) = -sqrt(0.84) \approx -0.9165
\def\ang{156.42} % angle whose sine is 0.4
\draw[thick] (0,0) -- ({cos(\ang)}, {sin(\ang)});
\filldraw ({cos(\ang)}, {sin(\ang)}) circle (0.5pt);
\draw[thick, dashed] (0,0) -- ({cos(-\ang)}, {sin(-\ang)});
\filldraw ({cos(-\ang)}, {sin(-\ang)}) circle (0.5pt);
% angle arc for \omega
\draw[maincolor!60!black, thick, ->] (0.2,0) arc (0:\ang:0.2);
\node[maincolor!60!black, font=\scriptsize, above right] at (0.1, 0.16) {$\omega$};
% angle arc for -\omega
\draw[maincolor!60!black, thick, ->] (0.25,0) arc (0:-\ang:0.25);
\node[maincolor!60!black, font=\scriptsize, right] at (0.25, -0.1) {$-\omega$};
\filldraw (1,0) circle (0.5pt);
\end{tikzpicture}
\end{center}
\item Το ημίτονο γωνίας σχεδιασμένης στον τριγωνομετρικό κύκλο είναι η τεταγμένη του σημείου τομής της τελικής πλευράς της γωνίας με το κύκλο, οπότε έχουμε, λόγω συμμετρίας $\hm (-\omega) = -0,4$.
\end{alist}
