\begin{alist}
\item Όπως φαίνεται στο σχήμα, η γραφική παράσταση της $f$ τέμνει τον $x'x$ άξονα στο $(-2,0)$, οπότε έχουμε ισοδύναμα:
\begin{align*}
f(-2) &= 0 \Leftrightarrow \\
\frac{1}{4}(-2)^4 + \alpha\cdot(-2)^2 &= 0 \Leftrightarrow \\
4 + 4\alpha &= 0 \Leftrightarrow \\
\alpha &= -1.
\end{align*}

\item 
\begin{rlist}
\item Έχουμε $f(x) = \dfrac{1}{4}x^4 - x^2$. Για κάθε $x \in \mathbb{R}$ ισχύει και $-x \in \mathbb{R}$. Οπότε:
\[f(-x) = \frac{1}{4}(-x)^4 - (-x)^2 = \frac{1}{4}x^4 - x^2 = f(x).\]
\item Στο βi) αποδείξαμε ότι $f(-x) = f(x)$, δηλαδή ότι η $f$ είναι άρτια. Συνεπώς η γραφική της παράσταση είναι συμμετρική ως προς τον $y'y$ άξονα.
\end{rlist}

\item Πραγματικά $f(-\sqrt{3}) = \dfrac{1}{4}(-\sqrt{3})^4 - (-\sqrt{3})^2 = \dfrac{9}{4} - 3 = -\dfrac{3}{4}$, οπότε το σημείο $\left(-\sqrt{3}, -\dfrac{3}{4}\right)$ ανήκει στη γραφική παράσταση της $f$.
\end{alist}
