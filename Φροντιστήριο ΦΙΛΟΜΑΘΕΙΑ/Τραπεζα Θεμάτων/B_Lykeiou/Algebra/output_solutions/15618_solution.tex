\lysh{15618}{2}
\begin{alist}
\item Στο πολυώνυμο μπορούμε να εξάγουμε κοινό παράγοντα το $x$ και έχουμε:
\[P(x) = 2x^3 + x^2 - x = x(2x^2 + x - 1)\] που είναι το ζητούμενο.
\item Η εξίσωση γράφεται ισοδύναμα 
\begin{align*}
P(x) = 2x^3 + x^2 - x = 0 &\Leftrightarrow x(2x^2 + x - 1) = 0 \\
&\Leftrightarrow x(2x - 1)(x + 1) = 0 \\
&\Leftrightarrow x = 0 \text{ ή } x = \frac{1}{2} \text{ ή } x = -1.
\end{align*}
\end{alist}
