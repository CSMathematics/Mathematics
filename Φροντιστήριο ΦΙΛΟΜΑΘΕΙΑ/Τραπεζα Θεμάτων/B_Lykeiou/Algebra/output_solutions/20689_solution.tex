\lysh{20689}{4}
\begin{alist}
\item Η ανίσωση ορίζεται για κάθε $x \in \mathbb{R}$ με $x \ne -1$. \\
Είναι $\dfrac{x-2}{x+1} > 0 \Leftrightarrow (x-2)(x+1) > 0 \Leftrightarrow x < -1 \text{ ή } x > 2$. \\
Τελικά $x \in (-\infty, -1) \cup (2, +\infty)$.
\item 
\begin{rlist}
\item Η συνάρτηση $f$ ορίζεται σε όλο το $\mathbb{R}$ αν και μόνο αν $\dfrac{\alpha-2}{\alpha+1} > 0$. \\
Από το ερώτημα ($\alpha$) έχουμε ότι: $\dfrac{\alpha-2}{\alpha+1} > 0 \Leftrightarrow \alpha \in (-\infty, -1) \cup (2, +\infty)$.
\item Η συνάρτηση $f$ είναι γνησίως φθίνουσα αν και μόνο αν $0 < \dfrac{\alpha-2}{\alpha+1} < 1$. \\
Από το ερώτημα ($\text{i}$) έχουμε ότι: $\dfrac{\alpha-2}{\alpha+1} > 0 \Leftrightarrow \alpha \in (-\infty, -1) \cup (2, +\infty)$. \\
Ακόμη είναι:
\[\frac{\alpha-2}{\alpha+1} < 1 \Leftrightarrow \frac{\alpha-2}{\alpha+1} - 1 < 0 \Leftrightarrow \frac{\alpha-2-1(\alpha+1)}{\alpha+1} < 0 \Leftrightarrow \frac{-3}{\alpha+1} < 0 \Leftrightarrow \alpha+1 > 0 \Leftrightarrow \alpha > -1.\]
Οι παραπάνω ανισώσεις συναληθεύουν για $\alpha \in (2, +\infty)$. Επομένως για $\alpha \in (2, +\infty)$ η συνάρτηση $f$ είναι γνησίως φθίνουσα.
\item Ισχύει $\dfrac{\alpha-2}{\alpha+1} > 0$.
\begin{itemize}
\item Αν $\dfrac{\alpha-2}{\alpha+1} > 1$ τότε η συνάρτηση $f$ είναι γνησίως αύξουσα.
\item Αν $0 < \dfrac{\alpha-2}{\alpha+1} < 1$ τότε η συνάρτηση $f$ είναι γνησίως φθίνουσα.
\item Αν ισχύει $\dfrac{\alpha-2}{\alpha+1} = 1$, η $f$ είναι σταθερή αφού $f(x) = 1^x = 1$. \\
Τότε $\dfrac{\alpha-2}{\alpha+1} = 1 \Rightarrow \alpha-2 = \alpha+1 \Rightarrow 0\alpha = 3$, αδύνατη. \\
Τελικά, δεν υπάρχουν τιμές του πραγματικού αριθμού $\alpha$ για τις οποίες η συνάρτηση $f$ είναι σταθερή.
\end{itemize}
\end{rlist}
\end{alist}
