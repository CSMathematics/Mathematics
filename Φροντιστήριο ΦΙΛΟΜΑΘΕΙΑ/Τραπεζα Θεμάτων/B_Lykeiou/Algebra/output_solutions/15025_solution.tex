\lysh{15025}{4}
\begin{alist}
\item Για τη γωνία $\theta = \text{A}\hat{\text{O}}\text{M}$ γνωρίζουμε ότι $\dfrac{\pi}{2} < \theta < \pi$ και $\hm \theta = \dfrac{4}{5}$. Από τη τριγωνομετρική ταυτότητα $\hm ^2\theta + \syn ^2\theta = 1$ έχουμε ότι:
\[\left(\dfrac{4}{5}\right)^2 + \syn ^2\theta = 1 \Leftrightarrow \dfrac{16}{25} + \syn ^2\theta = 1 \Leftrightarrow \syn ^2\theta = 1 - \dfrac{16}{25} \Leftrightarrow \syn ^2\theta = \dfrac{9}{25}.\]
Όμως $\dfrac{\pi}{2} < \theta < \pi$ οπότε $\syn \theta < 0$ και επομένως $\syn \theta = -\dfrac{3}{5}$.
Επίσης $\varepsilon\phi\theta = \dfrac{\hm \theta}{\syn \theta} = \dfrac{\dfrac{4}{5}}{-\dfrac{3}{5}} = -\dfrac{4}{3}$ και τέλος $\sigma\phi\theta = \dfrac{1}{\varepsilon\phi\theta} = \dfrac{1}{-\dfrac{4}{3}} = -\dfrac{3}{4}$.
\item Γενικά, για τα σημεία $M$ και $K$ που η τελική πλευρά μιας γωνίας $\theta$ τέμνει τον τριγωνομετρικό κύκλο και την ευθεία $x = 1$ αντίστοιχα, ισχύει ότι $M(\syn \theta, \hm \theta)$ και $K(1, \varepsilon\phi\theta)$. Συνεπώς $M\left(-\dfrac{3}{5}, \dfrac{4}{5}\right)$ και $K\left(1, -\dfrac{4}{3}\right)$.
\item
\begin{rlist}
\item Είναι $\hm \phi = \dfrac{3}{5} > 0$ και $\syn \phi < 0$, οπότε η τελική πλευρά της γωνίας $\phi$ είναι στο $2$ο τεταρτημόριο.
\item Είναι $\dfrac{3}{5} < \dfrac{4}{5}$, δηλαδή $\hm \theta > \hm \phi$. Όμως η συνάρτηση $\hm  x$ είναι γνησίως φθίνουσα στο $\left[\dfrac{\pi}{2}, \pi\right]$, οπότε για να είναι $\hm \theta > \hm \phi$ θα πρέπει $\theta < \phi$.
Εναλλακτικά, βρίσκουμε το σημείο $N$ του κύκλου με τεταγμένη $\hm \phi = \dfrac{3}{5}$ και τετμημένη $\syn \phi < 0$ και διαπιστώνουμε το σημείο $N$ είναι πιο αριστερά και κάτω από το σημείο $M$ δηλαδή $\text{A}\hat{\text{O}}\text{M} < \text{A}\hat{\text{O}}\text{N}$ οπότε $\theta < \phi$.
\end{rlist}
\end{alist}
