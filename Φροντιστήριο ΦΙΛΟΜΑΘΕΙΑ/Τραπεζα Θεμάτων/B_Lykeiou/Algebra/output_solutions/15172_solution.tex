\begin{alist}
\item 
\begin{rlist}
\item Στην παράσταση $\hm (11\pi - x)$ κάνουμε την αναγωγή στο 1ο τεταρτημόριο,
\[\hm (11\pi - x) = \hm (10\pi + \pi - x) = \hm (5 \cdot 2\pi + \pi - x) = \hm (\pi - x) = \hm  x\text{, το ζητούμενο.}\]
\item Στον τύπο της συνάρτησης αντικαθιστούμε το $\hm (11\pi - x)$ με $\hm  x$, άρα
\[f(x) = 4\hm  x.\]
\end{rlist}
\item Παρατηρούμε πως η $f(x) = 4\hm  x$ έχει την μορφή $\rho\hm \omega x$ με $\rho=4$ και $\omega=1$. Άρα η συνάρτηση έχει μέγιστη τιμή 4 και ελάχιστη τιμή -4 με περίοδο $2\pi$. Έχοντας τα παραπάνω χαρακτηριστικά και τον παρακάτω πίνακα τιμών στο διάστημα $[0,2\pi]$,
\begin{center}
\begin{mytblr}{}
$x$ & $0$ & $\dfrac{\pi}{2}$ & $\pi$ & $\dfrac{3\pi}{2}$ & $2\pi$ \\ 
$\hm{x}$ & $0$ & $1$ & $0$ & $-1$ & $0$ \\
$4\hm  x$ & $0$ & $4$ & $0$ & $-4$ & $0$ \\ 
\end{mytblr}
\end{center}
