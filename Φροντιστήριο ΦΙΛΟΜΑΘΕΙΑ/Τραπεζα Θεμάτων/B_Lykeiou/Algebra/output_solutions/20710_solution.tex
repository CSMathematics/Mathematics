\begin{alist}
\item Είναι $\beta + \alpha = \log 20 + \log 50 = \log(20 \cdot 50) = \log 1000 = \log 10^3 = 3$.
\item Είναι $\beta + \alpha = 3 > e \Rightarrow \ln(\alpha + \beta) > \ln e \Rightarrow \ln(\alpha + \beta) > 1$.
\item Είναι $10^\beta - 10^\alpha = 10^{\log 50} - 10^{\log 20} = 50 - 20 = 30 = 10 \cdot (\beta + \alpha)$.
\end{alist}
