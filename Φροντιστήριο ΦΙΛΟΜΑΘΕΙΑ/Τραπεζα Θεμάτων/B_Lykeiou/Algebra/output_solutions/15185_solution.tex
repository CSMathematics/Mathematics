\lysh{15185}{2}
\begin{alist}
\item Το συνημίτονο της γωνίας $\omega$ είναι όσο και η τετμημένη του σημείου $B$ του τριγωνομετρικού κύκλου. Άρα, $\syn \omega = -\dfrac{3}{5}$.
\item Ισχύει ότι:
\begin{align*}
\hm ^2\omega + \syn ^2\omega &= 1 \Leftrightarrow \hm ^2\omega + \left(-\frac{3}{5}\right)^2 = 1 \Leftrightarrow \\
\hm ^2\omega &= 1 - \frac{9}{25} \Leftrightarrow \hm ^2\omega = \frac{16}{25} \Leftrightarrow \\
\hm \omega &= \pm\sqrt{\frac{16}{25}} = \pm\frac{4}{5}.
\end{align*}
Από το σχήμα φαίνεται ότι $\dfrac{\pi}{2} < \omega < \pi$ οπότε $\hm \omega > 0$. Άρα, $\hm \omega = \dfrac{4}{5}$.
\end{alist}
