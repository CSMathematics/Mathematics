\begin{alist}
\item Το συνημίτονο της γωνίας $\omega$ είναι όσο και η τετμημένη του σημείου $B$ του τριγωνομετρικού κύκλου. Άρα, $\sigma\upsilon\nu\omega = -\dfrac{3}{5}$.

\item Ισχύει ότι:
\begin{align*}
\eta\mu^2\omega + \sigma\upsilon\nu^2\omega &= 1 \Leftrightarrow \eta\mu^2\omega + \left(-\frac{3}{5}\right)^2 = 1 \Leftrightarrow \\
\eta\mu^2\omega &= 1 - \frac{9}{25} \Leftrightarrow \eta\mu^2\omega = \frac{16}{25} \Leftrightarrow \\
\eta\mu\omega &= \pm\sqrt{\frac{16}{25}} = \pm\frac{4}{5}.
\end{align*}
Από το σχήμα φαίνεται ότι $\dfrac{\pi}{2} < \omega < \pi$ οπότε $\eta\mu\omega > 0$. Άρα, $\eta\mu\omega = \dfrac{4}{5}$.
\end{alist}
