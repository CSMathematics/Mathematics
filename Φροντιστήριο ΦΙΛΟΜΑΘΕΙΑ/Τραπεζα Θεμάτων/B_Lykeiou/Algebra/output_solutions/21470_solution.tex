\lysh{21470}{4}
\begin{alist}
\item Γνωρίζουμε ότι μετά από δύο χρόνια έχει απομείνει το $\dfrac{1}{3}$ της αρχικής ποσότητας, δηλαδή:
\[Q(2) = \frac{1}{3}\cdot Q_0 \Leftrightarrow Q_0\cdot e^{2c} = \frac{Q_0}{3} \Leftrightarrow (e^c)^2 = \frac{1}{3} \overset{e^c>0}{\iff} e^c = \frac{1}{\sqrt{3}}.\] 
Οπότε
\begin{align*}
Q(t) &= Q_0\cdot e^{ct} \Leftrightarrow \\
Q(t) &= Q_0\cdot(e^c)^t \Leftrightarrow \\
Q(t) &= Q_0\cdot\left(\frac{1}{\sqrt{3}}\right)^t.
\end{align*}
\item Γνωρίζουμε ότι μετά από τέσσερα χρόνια έχει απομείνει $1$ κιλό ραδιενεργού υλικού και από το α) ερώτημα γνωρίζουμε επίσης ότι $Q(t) = Q_0\cdot\left(\dfrac{1}{\sqrt{3}}\right)^t$. Ζητούμενη είναι η αρχική ποσότητα που θάφτηκε, δηλαδή το $Q_0$. \\
Επομένως: $Q(4) = 1 \Leftrightarrow Q_0\cdot\left(\dfrac{1}{\sqrt{3}}\right)^4 = 1 \Leftrightarrow Q_0\cdot\left(\dfrac{1}{9}\right) = 1 \Leftrightarrow Q_0 = 9 \text{ κιλά}$.
\item Στα ερωτήματα α) και β) δείξαμε ότι $Q(t) = 9\cdot\left(\dfrac{1}{\sqrt{3}}\right)^t$. Έχουμε
\begin{align*}
Q(t) &= \frac{1}{81} \Leftrightarrow \\
9\cdot\left(\frac{1}{\sqrt{3}}\right)^t &= \frac{1}{81} \Leftrightarrow \\
3^2\cdot\left(3^{-\frac{1}{2}}\right)^t &= 3^{-4} \Leftrightarrow \\
3^{2-\frac{t}{2}} &= 3^{-4} \Leftrightarrow \\
2-\frac{t}{2} &= -4 \Leftrightarrow \\
\frac{t}{2} &= 6 \Leftrightarrow t = 12.
\end{align*}
Συνεπώς μετά από $12$ χρόνια η ποσότητα που θα έχει απομείνει θα είναι $\dfrac{1}{81}$ κιλά.
\end{alist}
