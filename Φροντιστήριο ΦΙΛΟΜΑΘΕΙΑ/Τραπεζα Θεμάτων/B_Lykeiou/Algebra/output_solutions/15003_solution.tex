\begin{alist}
\item
\begin{rlist}
\item Ισχύει ότι $\sigma\upsilon\nu\left(\dfrac{\pi}{2} - \alpha x\right) = \eta\mu(\alpha x)$ και $\sigma\upsilon\nu(\pi - \alpha x) = -\sigma\upsilon\nu(\alpha x)$. Άρα, ο τύπος της συνάρτησης γίνεται:
\[f(x) = \eta\mu(\alpha x)(\eta\mu(\alpha x) + 2) - \sigma\upsilon\nu(\alpha x)(-\sigma\upsilon\nu(\alpha x)) - 1 =\]
\[\eta\mu^2(\alpha x) + 2\eta\mu(\alpha x) + \sigma\upsilon\nu^2(\alpha x) - 1 =\]
\[1 + 2\eta\mu(\alpha x) - 1 = 2\eta\mu(\alpha x).\]

\item Η $2\eta\mu(\alpha x)$ έχει περίοδο $T = \dfrac{2\pi}{\alpha}$. Από τη γραφική παράσταση προκύπτει ότι η συνάρτηση $f$ έχει περίοδο $T = \pi$. Άρα,
\[\dfrac{2\pi}{\alpha} = \pi \Leftrightarrow \alpha = 2.\]
\end{rlist}

\item Οι τετμημένες των κοινών σημείων της γραφικής παράστασης της $f$ με την ευθεία $y = 1$ είναι οι λύσεις της εξίσωσης
\[2\eta\mu 2x = 1 \Leftrightarrow \eta\mu 2x = \dfrac{1}{2} \Leftrightarrow \eta\mu 2x = \eta\mu\dfrac{\pi}{6}.\]
Οπότε,
\[\begin{cases} 2x = \dfrac{\pi}{6} + 2\kappa\pi \\ \text{ή} \\ 2x = \pi - \dfrac{\pi}{6} + 2\kappa\pi \end{cases} \Leftrightarrow \begin{cases} x = \dfrac{\pi}{12} + \kappa\pi \\ \text{ή} \\ x = \dfrac{5\pi}{12} + \kappa\pi \end{cases}, \quad \kappa \in \mathbb{Z}.\]

Επειδή $x \in [0, \pi]$ έχουμε ότι:
\[0 \le x \le \pi \Leftrightarrow 0 \le \dfrac{\pi}{12} + \kappa\pi \le \pi \Leftrightarrow\]
\[-\dfrac{1}{12} \le \kappa \le 1 - \dfrac{1}{12} \Leftrightarrow -\dfrac{1}{12} \le \kappa \le \dfrac{11}{12} \overset{\kappa \in \mathbb{Z}}{\implies} \kappa = 0\]
και
\[0 \le x \le \pi \Leftrightarrow 0 \le \dfrac{5\pi}{12} + \kappa\pi \le \pi \Leftrightarrow\]
\[-\dfrac{5}{12} \le \kappa \le 1 - \dfrac{5}{12} \Leftrightarrow -\dfrac{5}{12} \le \kappa \le \dfrac{7}{12} \overset{\kappa \in \mathbb{Z}}{\implies} \kappa = 0.\]
Άρα, $x = \dfrac{\pi}{12}$ ή $x = \dfrac{5\pi}{12}$.
\end{alist}
