\lysh{20817}{2}
\begin{alist}
\item Από την τριγωνομετρική ταυτότητα $\hm ^2\omega + \syn ^2\omega = 1$ έχουμε ισοδύναμα:
\begin{align*}
\hm ^2\omega + \syn ^2\omega &= 1 \Leftrightarrow \\
\hm ^2\omega + \left(-\frac{4}{5}\right)^2 &= 1 \Leftrightarrow \\
\hm ^2\omega &= 1 - \frac{16}{25} \Leftrightarrow \\
\hm ^2\omega &= \frac{9}{25} \Leftrightarrow \\
\hm \omega &= \pm \frac{3}{5}.
\end{align*}
Επειδή $\pi < \omega < \frac{3\pi}{2}$, $\hm \omega < 0$, οπότε $\hm \omega = -\frac{3}{5}$.
\item Έχουμε $\syn \omega = -\frac{4}{5}$ και $\hm \omega = -\frac{3}{5}$, οπότε $\ef \omega = \dfrac{-\frac{3}{5}}{-\frac{4}{5}} = \frac{3}{4}$. Συνεπώς η τιμή της παράστασης $\text{A}$ είναι:
\[\text{A} = \frac{\hm \omega + \syn \omega}{1 + \ef \omega} = \frac{-\frac{3}{5} - \frac{4}{5}}{1 + \frac{3}{4}} = \frac{-\frac{7}{5}}{\frac{7}{4}} = -\frac{4}{5}.\]
\end{alist}
