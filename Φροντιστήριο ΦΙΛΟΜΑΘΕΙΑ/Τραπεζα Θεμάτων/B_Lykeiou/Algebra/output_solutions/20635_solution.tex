\lysh{20635}{2}
\begin{alist}
\item Η συνάρτηση ορίζεται μόνο όταν ισχύει $x + 1 > 0$. Είναι: $x + 1 > 0 \Leftrightarrow x > -1$. \\
Άρα το πεδίο ορισμού της συνάρτησης είναι το $A = (-1, +\infty)$.
\item Με $x=0$ έχουμε:
\[f(0) = \ln(0 + 1) = \ln 1 = 0\]
Άρα η γραφική παράσταση της $f$ διέρχεται από την αρχή $\text{O}$.
\item Με $x > -1$ έχουμε:
\[f(x) = 2 \Leftrightarrow \ln(x + 1) = 2 \Leftrightarrow x + 1 = e^2 \Leftrightarrow x = e^2 - 1\]
που είναι αποδεκτή αφού περιέχεται στο διάστημα $\text{A} = (-1, +\infty)$.
\end{alist}
