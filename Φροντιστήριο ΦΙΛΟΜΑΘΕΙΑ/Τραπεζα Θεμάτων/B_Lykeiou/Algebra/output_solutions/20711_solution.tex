\lysh{20711}{2}
\begin{alist}
\item Είναι $1 < 3 < 4 \Leftrightarrow \log 1 < \log 3 < \log 4 \Leftrightarrow 0 < \alpha < \beta$.
\item 
\begin{rlist}
\item Είναι $\beta + \alpha = \log 4 + \log 3 = \log(4 \cdot 3) = \log 12 > \log 10 = 1$, οπότε πράγματι $\beta + \alpha > 1$.
\item Είναι $0 < \alpha < \beta \Rightarrow \dfrac{\alpha}{\beta} < 1 \Rightarrow \ln\dfrac{\alpha}{\beta} < \ln 1 \Rightarrow \ln\dfrac{\alpha}{\beta} < 0$.
\end{rlist}
\end{alist}
