\lysh{20761}{2}
\begin{alist}
\item Για κάθε γωνία $\omega$, αν είναι $\mu$ το μέτρο της σε μοίρες και $\alpha$ το μέτρο της σε ακτίνια, ισχύει ότι $\dfrac{\alpha}{\pi} = \dfrac{\mu}{180^\circ}$, επομένως για τη γωνία $\omega$ έχουμε
\[\frac{\alpha}{\pi} = \frac{-1125^\circ}{180^\circ} \Leftrightarrow \frac{\alpha}{\pi} = \frac{-25}{4} \Leftrightarrow \alpha = \frac{-25\pi}{4}\]
Άρα η γωνία $\omega$ σε ακτίνια είναι $\dfrac{-25\pi}{4}$.
\item Ισχύει ότι $-1125^\circ = -3 \cdot 360^\circ + (-45^\circ)$, επομένως
\begin{align*}
\hm (-1125^\circ) &= \hm [-3 \cdot 360^\circ + (-45^\circ)] = \hm (-45^\circ) = -\hm  45^\circ = -\frac{\sqrt{2}}{2} \text{ και ομοίως} \\
\syn (-1125^\circ) &= \syn (-45^\circ) = \syn  45^\circ = \frac{\sqrt{2}}{2} \\
\ef (-1125^\circ) &= \ef (-45^\circ) = -\ef  45^\circ = -1 \\
\syf (-1125^\circ) &= \syf (-45^\circ) = -\syf  45^\circ = -1
\end{align*}
\textbf{Β' τρόπος} \\
Ισχύει ότι $\dfrac{-25\pi}{4} = -\dfrac{24\pi + \pi}{4} = -6\pi - \dfrac{\pi}{4} = (-3)\cdot 2\pi + \left(-\dfrac{\pi}{4}\right)$, επομένως
\begin{align*}
\hm \left(\frac{-25\pi}{4}\right) &= \hm \left[(-3)\cdot 2\pi + \left(-\frac{\pi}{4}\right)\right] = \hm \left(-\frac{\pi}{4}\right) = -\hm \frac{\pi}{4} = -\frac{\sqrt{2}}{2} \text{ και ομοίως} \\
\syn \left(\frac{-25\pi}{4}\right) &= \syn \left(-\frac{\pi}{4}\right) = \syn \frac{\pi}{4} = \frac{\sqrt{2}}{2} \\
\ef \left(\frac{-25\pi}{4}\right) &= \ef \left(-\frac{\pi}{4}\right) = -\ef \frac{\pi}{4} = -1 \\
\syf \left(\frac{-25\pi}{4}\right) &= \syf \left(-\frac{\pi}{4}\right) = -\syf \frac{\pi}{4} = -1
\end{align*}
\end{alist}
