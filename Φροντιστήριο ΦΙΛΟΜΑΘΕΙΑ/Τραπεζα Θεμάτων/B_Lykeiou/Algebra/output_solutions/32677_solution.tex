\lysh{32677}{4}
\begin{alist}
\item Ο τύπος της συνάρτησης $f$ διαδοχικά γράφεται:
\begin{align*}
f(x) &= -x^2 + 2x + 1 = \\
&= -x^2 + 2x - 1 + 2 = \\
&= -(x^2 - 2x + 1) + 2 = \\
&= -(x - 1)^2 + 2.
\end{align*}
Εναλλακτικά, ξεκινώντας από το ζητούμενο έχουμε:
\begin{align*}
-(x - 1)^2 + 2 &= \\
&= -(x^2 - 2x + 1) + 2 = \\
&= -x^2 + 2x - 1 + 2 = \\
&= -x^2 + 2x + 1 = f(x).
\end{align*}
Παρατηρούμε ότι $f(x) = \varphi(x - 1) + 2$. Άρα, η γραφική παράσταση της $f$ προκύπτει από μετατόπιση της γραφικής παράστασης της $\varphi$ κατά μία μονάδα δεξιά και δύο μονάδες επάνω:
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    grid=both,
    xmin=-2.2, xmax=5.2,
    ymin=-1.2, ymax=2.5,
    xtick={-2,-1,0,1,2,3,4,5},
    ytick={-1,1,2},
    width=8cm, height=6cm,
    samples=100
]
\addplot[domain=-0.5:2.5, maincolor, ultra thick] {-(x-1)^2+2} node[pos=0.8, right] {\small $f(x) = -(x-1)^2+2$};
\end{axis}
\end{tikzpicture}
\end{center}
\item 
\begin{rlist}
\item Από τη γραφική της παράσταση, προκύπτει ότι η $f$ είναι γνησίως αύξουσα στο διάστημα $(-\infty, 1]$ και γνησίως φθίνουσα στο διάστημα $[1, +\infty)$.
\item Η $f$ παρουσιάζει ολικό μέγιστο στο $x_0 = 1$ το $f(1) = 2$.
\item Οι ρίζες της εξίσωσης $f(x) = \kappa$, $\kappa < 2$ είναι οι τετμημένες των σημείων τομής της γραφικής παράστασης της $f$ με την οριζόντια ευθεία $y = \kappa$. \\
Από τη γραφική παράσταση προκύπτει ότι για $\kappa < 2$, υπάρχουν δύο σημεία τομής. Άρα, η εξίσωση έχει δύο ρίζες.
\end{rlist}
\end{alist}
