\begin{alist}
\item Από τα δεδομένα έχουμε:
\begin{align*}
\begin{cases} P(1) = 2 \\ P(2) = 15 \end{cases} \Leftrightarrow \begin{cases} 2\cdot 1^3 - \alpha \cdot 1^2 + 2\cdot 1 + \beta = 2 \\ 2\cdot 2^3 - \alpha \cdot 2^2 + 2\cdot 2 + \beta = 15 \end{cases} &\Leftrightarrow \begin{cases} -\alpha + \beta = -2 \\ -4\alpha + \beta = -5 \end{cases} \\
&\xLeftrightarrow{(-)} \begin{cases} -3\alpha = -3 \\ -\alpha + \beta = -2 \end{cases} \Leftrightarrow \begin{cases} \alpha = 1 \\ \beta = -1. \end{cases}
\end{align*}
Άρα $P(x) = 2x^3 - x^2 + 2x - 1$.
\item 
\begin{rlist}
\item Από το α) ερώτημα, έχουμε $P(x) = 2x^3 - x^2 + 2x - 1$. Οπότε:
\[P(x) = 2x^3 - x^2 + 2x - 1 = x^2(2x-1) + (2x-1) = (2x-1)(x^2+1).\]
Άρα το πολυώνυμο $\pi(x) = x^2+1$ είναι παράγοντας του $P(x)$.
\item Έχουμε ισοδύναμα:
\[P(x) = 0 \text{, δηλαδή}\]
\[(2x-1)(x^2+1) = 0 \text{, οπότε}\]
\[(2x-1) = 0 \text{ (αφού } x^2+1 \ne 0 \text{ για κάθε } x) \text{ και τελικά}\]
\[x = \frac{1}{2}.\]
\end{rlist}
\item Έχουμε ισοδύναμα:
\begin{align*}
\syn ^3x + \syn  x &= 1 - \frac{1}{2}\hm ^2x \xLeftrightarrow[\quad (\cdot 2) \quad]{(\hm ^2x = 1 - \syn ^2x)} \\
2\syn ^3x + 2\syn  x &= 2 - (1 - \syn ^2x) \Leftrightarrow \\
2\syn ^3x + 2\syn  x &= 2 - 1 + \syn ^2x \Leftrightarrow \\
2\syn ^3x - \syn ^2x + 2\syn  x - 1 &= 0 \xLeftrightarrow{\quad (\alpha) \quad} \\
P(\syn  x) &= 0.
\end{align*}
Οπότε από το βii) ερώτημα έχουμε $\syn  x = \dfrac{1}{2}$ και επειδή $x \in (0, 2\pi)$,
\[x = \frac{\pi}{3}, \quad x = 2\pi - \frac{\pi}{3} = \frac{5\pi}{3}.\]
\end{alist}
