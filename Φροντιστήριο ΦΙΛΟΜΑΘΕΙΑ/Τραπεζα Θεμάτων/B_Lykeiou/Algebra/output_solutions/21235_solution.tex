\begin{alist}
\item Έχουμε, με αναγωγή στο $1^\circ$ τεταρτημόριο, ότι
\begin{align*}
\eta\mu(180^\circ - 20^\circ) &= \eta\mu 20^\circ \\
\sigma\upsilon\nu(90^\circ - 20^\circ) &= \eta\mu 20^\circ \\
\sigma\upsilon\nu(-3x) &= \sigma\upsilon\nu 3x
\end{align*}
Επομένως,
\[A = \frac{\eta\mu(180^\circ - 20^\circ) \cdot \sigma\upsilon\nu(-3x)}{\sigma\upsilon\nu(90^\circ - 20^\circ)} = \frac{\eta\mu 20^\circ \cdot \sigma\upsilon\nu(3x)}{\eta\mu 20^\circ} = \sigma\upsilon\nu(3x).\]

\item Η συνάρτηση $f(x) = \rho\sigma\upsilon\nu\omega x$ με $\rho, \omega > 0$ έχει μέγιστη τιμή $\rho$ και περίοδο $T = \dfrac{2\pi}{\omega}$. \\
Οπότε η $f(x) = \sigma\upsilon\nu 3x$ έχει μέγιστη τιμή $1$ και περίοδο $T = \dfrac{2\pi}{3}$.
\end{alist}
