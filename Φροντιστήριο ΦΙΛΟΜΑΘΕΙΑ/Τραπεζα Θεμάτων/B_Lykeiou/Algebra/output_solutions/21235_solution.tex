\lysh{21235}{2}
\begin{alist}
\item Έχουμε, με αναγωγή στο $1^\circ$ τεταρτημόριο, ότι
\begin{align*}
\hm (180^\circ - 20^\circ) &= \hm  20^\circ \\
\syn (90^\circ - 20^\circ) &= \hm  20^\circ \\
\syn (-3x) &= \syn  3x
\end{align*}
Επομένως,
\[A = \frac{\hm (180^\circ - 20^\circ) \cdot \syn (-3x)}{\syn (90^\circ - 20^\circ)} = \frac{\hm  20^\circ \cdot \syn (3x)}{\hm  20^\circ} = \syn (3x).\]
\item Η συνάρτηση $f(x) = \rho\syn \omega x$ με $\rho, \omega > 0$ έχει μέγιστη τιμή $\rho$ και περίοδο $T = \dfrac{2\pi}{\omega}$. \\
Οπότε η $f(x) = \syn  3x$ έχει μέγιστη τιμή $1$ και περίοδο $T = \dfrac{2\pi}{3}$.
\end{alist}
