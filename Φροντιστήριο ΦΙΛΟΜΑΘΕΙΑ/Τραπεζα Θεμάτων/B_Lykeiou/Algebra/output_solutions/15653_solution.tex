\lysh{15653}{2}
\begin{alist}
\item
\begin{rlist}
\item Η διαίρεση του $P(x) = x^3 + x^2 + 2x + 2$ με το $(x+1)$ είναι η παρακάτω:
\begin{center}
\begin{tabular}{r|l}
$x^3 + x^2 + 2x + 2$ & $x + 1$ \\ \cline{2-2}
$-x^3 - x^2 \phantom{{} + 2x + 2}$ & $x^2 + 2$ \\ \cline{1-1}
$2x + 2$ & \\
$-2x - 2$ & \\ \cline{1-1}
$0$ &
\end{tabular}
\end{center}
\item Η ταυτότητα της παραπάνω διαίρεσης είναι:
\begin{align*}
x^3 + x^2 + 2x + 2 &= (x + 1)(x^2 + 2) + 0 \Rightarrow \\
x^3 + x^2 + 2x + 2 &= (x + 1)(x^2 + 2).
\end{align*}
\end{rlist}
\item Έχουμε ισοδύναμα: 
\[P(x) < 0 \Leftrightarrow (x+1)(x^2 + 2) < 0 \overset{x^2+2>0}{\Leftrightarrow} x + 1 < 0 \Leftrightarrow x < -1.\]
\end{alist}
