\lysh{21444}{4}
\begin{alist}
\item Οι λύσεις τις εξίσωσης $f(x) = g(x)$ είναι οι τετμημένες των κοινών σημείων των γραφικών παραστάσεων των συναρτήσεων $f$ και $g$. Έχουμε:
\begin{align*}
f(x) = g(x) &\Leftrightarrow \\
4^x = 2^x - \frac{1}{4} &\Leftrightarrow \\
4 \cdot (2^x)^2 - 4 \cdot 2^x + 1 = 0 &\overset{y=2^x}{\iff} \\
4y^2 - 4y + 1 = 0 &\Leftrightarrow \\
(2y - 1)^2 = 0 &\Leftrightarrow y = \frac{1}{2}.
\end{align*}
Άρα

\[2^x &= \frac{1}{2} \Leftrightarrow 
2^x &= 2^{-1} \Leftrightarrow 
x &= -1.\]
Οπότε οι γραφικές παραστάσεις των συναρτήσεων $f$ και $g$ έχουν ακριβώς ένα κοινό σημείο το $\text{A}\left(-1, \dfrac{1}{4}\right)$, αφού $f(-1) = g(-1) = \dfrac{1}{4}$.
\item Θα δείξουμε ότι $f(x) > g(x)$ για κάθε $x \neq -1$. Έχουμε ισοδύναμα:
\begin{align*}
f(x) > g(x) &\Leftrightarrow \\
4^x > 2^x - \frac{1}{4} &\Leftrightarrow \\
4 \cdot (2^x)^2 - 4 \cdot 2^x + 1 > 0 &\overset{y=2^x}{\iff} \\
4y^2 - 4y + 1 > 0 &\Leftrightarrow \\
(2y - 1)^2 > 0.
\end{align*}
που ισχύει για κάθε για κάθε πραγματικό αριθμό $y \neq \dfrac{1}{2}$, δηλαδή για κάθε πραγματικό αριθμό $x$ για τον οποίο ισχύει: $2^x \neq \dfrac{1}{2} \Leftrightarrow 2^x \neq 2^{-1} \Leftrightarrow x \neq -1$.
\end{alist}
