\item Για κάθε $x \in \mathbb{R}$ έχουμε ότι:
\begin{itemize}
\item $-x \in \mathbb{R}$ και
\item $h(-x) = (-x)^3 - (-x) = -x^3 + x = -(x^3 - x) = -h(x)$
\end{itemize}
Επομένως η συνάρτηση $h$ είναι περιττή.
\item Συμπληρώνουμε το συμμετρικό τμήμα της δοθείσας γραφικής παράστασης ως προς την αρχή των αξόνων O.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-3, xmax=3,
    ymin=-5, ymax=5,
    xtick={-3,-2,...,3},
    ytick={-5,-4,...,5},
    grid=both,
    grid style={solid, gray!20},
	major grid style={line width=.2pt,draw=gray!30}
    thick,
    width=8cm, height=8cm,
    samples=100
]
\addplot[domain=-2.5:2.5, maincolor, very thick] {x^3 - x};
\end{axis}
\end{tikzpicture}
\end{center}
\item Τα σημεία τομής της γραφικής παράστασης της συνάρτησης $h$ με τον άξονα $x'x$ έχουν τεταγμένη $y = 0$, με $y = h(x)$. \\
Είναι $h(x) = 0 \Leftrightarrow x^3 - x = 0 \Leftrightarrow x(x^2 - 1) = 0 \Leftrightarrow x = 0 \text{ ή } x = 1 \text{ ή } x = -1$. \\
Άρα τα ζητούμενα σημεία είναι $(0, 0), (1, 0), (-1, 0)$.
\end{rlist}
\end{alist}
