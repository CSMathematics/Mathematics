\lysh{18229}{2}
\begin{alist}
\item Από την βασική τριγωνομετρική ταυτότητα $\hm ^2\theta + \syn ^2\theta = 1$ έχουμε
\[\hm ^2\theta = 1 - \syn ^2\theta = 1 - \left(-\frac{2}{3}\right)^2 = 1 - \frac{4}{9} = \frac{5}{9}\]
και $\theta \in \left(\pi, \dfrac{3\pi}{2}\right)$, οπότε $\hm \theta < 0$. Άρα, $\hm \theta = -\dfrac{\sqrt{5}}{3}$.
\item Επειδή $\pi - \theta + \theta = \pi$, έχουμε $\syn (\pi-\theta) = -\syn \theta$ και $\hm (\pi-\theta) = \hm \theta$. Επιπλέον,
\[\syn (-\theta) = \syn \theta \quad \text{και} \quad \hm (-\theta) = -\hm \theta\]
οπότε έχουμε:
\begin{align*}
\text{A} &= \syn (\pi-\theta)\syn (-\theta) - \hm (\pi-\theta)\hm (-\theta) \\
&= -\syn \theta\cdot\syn \theta - \hm \theta\cdot(-\hm \theta) \\
&= \hm ^2\theta - \syn ^2\theta \\
&= \frac{5}{9} - \frac{4}{9} = \frac{1}{9}.
\end{align*}
\end{alist}
