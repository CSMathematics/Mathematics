\begin{alist}
\item Το σημείο Α έχει τετμημένη μηδέν και τεταγμένη $f(0) = 2^0 = 1$, έτσι είναι $A(0,1)$. \\
Το σημείο Β έχει τεταγμένη $\dfrac{1}{5}$ και τετμημένη $x$ για την οποία ισχύει $g(x) = \dfrac{1}{5} \Leftrightarrow 5^{1-x} = 5^{-1}$. \\
Άρα $1 - x = -1$, έτσι $x = 2$. Ώστε $B\left(2, \dfrac{1}{5}\right)$.

\item Το σημείο $\Sigma$ έχει τετμημένη $x$ ώστε 
\begin{align*}
f(x) = g(x) &\Leftrightarrow 2^x = 5^{1-x} \\
&\Leftrightarrow 2^x = \frac{5}{5^x} \\
&\Leftrightarrow 2^x \cdot 5^x = 5
\end{align*}
Έτσι $10^x = 5$ άρα $x = \log 5$.

\item Είναι $x_B - x_{\Sigma} = 2 - \log 5 = \log(10^2) - \log 5 = \log\left(\dfrac{100}{5}\right) = \log 20$.
\end{alist}
