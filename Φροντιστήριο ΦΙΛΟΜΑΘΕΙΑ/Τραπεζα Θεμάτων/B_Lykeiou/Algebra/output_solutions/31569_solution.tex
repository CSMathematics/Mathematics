\begin{alist}
\item Η συνάρτηση $f(x) = -3\sigma\upsilon\nu 2x$ είναι της μορφής $f(x) = \rho\sigma\upsilon\nu(\omega x)$, με $\rho = -3$ και $\omega = 2$. Άρα η περίοδος της $f$ είναι $T = \dfrac{2\pi}{\omega} = \dfrac{2\pi}{2} = \pi$, η μέγιστη τιμή της είναι $3$ και η ελάχιστη είναι $-3$.

\item Σε διάστημα πλάτους μιας περιόδου, δηλαδή $\pi$, έχουμε τον παρακάτω πίνακα τιμών της συνάρτησης $f$:
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & $\frac{\pi}{4}$ & $\frac{\pi}{2}$ & $\frac{3\pi}{4}$ & $\pi$ \\ \hline
$2x$ & 0 & $\frac{\pi}{2}$ & $\pi$ & $\frac{3\pi}{2}$ & $2\pi$ \\ \hline
$\sigma\upsilon\nu 2x$ & 1 & 0 & $-1$ & 0 & 1 \\ \hline
$f(x) = -3\sigma\upsilon\nu 2x$ & $-3$ & 0 & 3 & 0 & $-3$ \\ \hline
\end{tabular}
\end{center}
