\begin{alist}
\item Η συνάρτηση $f(x) = \rho \cdot \sigma\upsilon\nu\omega x$, με $\rho, \omega > 0$: \\
Έχει ελάχιστο $-\rho$, οπότε έχουμε $-\dfrac{\alpha+1}{2} = -2 \Leftrightarrow \alpha + 1 = 4 \Leftrightarrow \alpha = 3$. \\
Ακόμα έχει περίοδο $T = \dfrac{2\pi}{\omega}$, οπότε έχουμε $\dfrac{2\pi}{\beta} = \dfrac{\pi}{2} \Leftrightarrow \beta = 4$. \\
Τελικά, η συνάρτηση είναι $f(x) = 2\sigma\upsilon\nu 4x \quad(1)$.

\item Έχουμε για τον αριθμητή: \\
$\eta\mu\left(\dfrac{\pi}{2} - x\right) = \sigma\upsilon\nu x$, $\varepsilon\varphi(\pi - x) = -\varepsilon\varphi x$, $\eta\mu(2\pi + x) = \eta\mu x$. \\
Έχουμε για τον παρονομαστή: \\
$\sigma\upsilon\nu(3\pi - x) = \sigma\upsilon\nu(\pi - x) = -\sigma\upsilon\nu x$, \\
$\sigma\varphi\left(\dfrac{7\pi}{2} - x\right) = \sigma\varphi\left(3\pi + \left(\dfrac{\pi}{2} - x\right)\right) = \sigma\varphi\left(\pi + \left(\dfrac{\pi}{2} - x\right)\right) = \sigma\varphi\left(\dfrac{\pi}{2} - x\right) = \varepsilon\varphi x$, \\
$\sigma\upsilon\nu\left(\dfrac{\pi}{2} + x\right) = \sigma\upsilon\nu\left(\dfrac{\pi}{2} - (-x)\right) = \eta\mu(-x) = -\eta\mu x$. \\
Αντικαθιστώντας έχουμε 
\[A = \frac{\sigma\upsilon\nu x \cdot (-\varepsilon\varphi x) \cdot \eta\mu x}{-\sigma\upsilon\nu x \cdot \varepsilon\varphi x \cdot (-\eta\mu x)} = -1.\]

\item Έχουμε, λόγω της $(1)$ και του β) ερωτήματος:
\begin{align*}
f(x) = 2A &\Leftrightarrow 2\sigma\upsilon\nu 4x = -2 \Leftrightarrow \sigma\upsilon\nu 4x = -1 \Leftrightarrow \sigma\upsilon\nu 4x = \sigma\upsilon\nu\pi \Leftrightarrow \\
4x = 2\kappa\pi \pm \pi &\Leftrightarrow x = \frac{\kappa\pi}{2} \pm \frac{\pi}{4}, \, \kappa \in \mathbb{Z}.
\end{align*}
Επειδή πρέπει $x \in \left[\pi, \dfrac{3\pi}{2}\right]$, δηλαδή $\pi \le x \le \dfrac{3\pi}{2}$, διακρίνουμε δύο περιπτώσεις:

\begin{minipage}[t]{0.48\textwidth}
\begin{align*}
\pi \le \frac{\kappa\pi}{2} + \frac{\pi}{4} &\le \frac{3\pi}{2} \Leftrightarrow \\
1 \le \frac{\kappa}{2} + \frac{1}{4} &\le \frac{3}{2} \Leftrightarrow \\
\frac{3}{4} \le \frac{\kappa}{2} &\le \frac{5}{4} \Leftrightarrow \\
\frac{3}{2} \le \kappa &\le \frac{5}{2} \overset{\kappa\in\mathbb{Z}}{\implies} \kappa = 2.
\end{align*}
Άρα $x = \pi + \dfrac{\pi}{4} = \dfrac{5\pi}{4}$.
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\textwidth}
\begin{align*}
\pi \le \frac{\kappa\pi}{2} - \frac{\pi}{4} &\le \frac{3\pi}{2} \Leftrightarrow \\
1 \le \frac{\kappa}{2} - \frac{1}{4} &\le \frac{3}{2} \Leftrightarrow \\
\frac{5}{4} \le \frac{\kappa}{2} &\le \frac{7}{4} \Leftrightarrow \\
\frac{5}{2} \le \kappa &\le \frac{7}{2} \overset{\kappa\in\mathbb{Z}}{\implies} \kappa = 3.
\end{align*}
Άρα, πάλι $x = \pi + \dfrac{\pi}{4} = \dfrac{5\pi}{4}$.
\end{minipage}

\end{alist}
