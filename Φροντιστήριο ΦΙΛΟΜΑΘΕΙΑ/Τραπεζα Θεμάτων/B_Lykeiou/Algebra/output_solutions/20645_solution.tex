\begin{alist}
\item Από την ισότητα $f(x) = g\left(x - \dfrac{\pi}{4}\right)$ συμπεραίνουμε ότι η γραφική παράσταση της $f$ προκύπτει από τη γραφική παράσταση της $g$, αν την μετατοπίσουμε κατά $\dfrac{\pi}{4}$ μονάδες προς τα δεξιά.
\item Στο παρακάτω σχήμα φαίνεται η γνωστή γραφική παράσταση της συνάρτησης $g(x) = \syn  x$ (με διακεκομμένη γραμμή) και η γραφική παράσταση της $f$ που προκύπτει από αυτή, αν την μεταφέρουμε δεξιά κατά $\dfrac{\pi}{4}$ μονάδες.

\item Είναι:
\[f\left(\frac{\pi}{2}\right) = \syn \left(\frac{\pi}{2} - \frac{\pi}{4}\right) = \syn \frac{\pi}{4} = \frac{\sqrt{2}}{2}\]
και
\[f(\pi) = \syn \left(\pi - \frac{\pi}{4}\right) = -\syn \frac{\pi}{4} = -\frac{\sqrt{2}}{2}\]
\item Με τη βοήθεια του προηγούμενου ερωτήματος, έχουμε:
\begin{align*}
\sqrt{2}f(x) + 1 = 0 &\Leftrightarrow f(x) = -\frac{1}{\sqrt{2}} \Leftrightarrow f(x) = -\frac{\sqrt{2}}{2} \\
&\Leftrightarrow \syn \left(x - \frac{\pi}{4}\right) = \syn \left(\pi - \frac{\pi}{4}\right) \Leftrightarrow \syn \left(x - \frac{\pi}{4}\right) = \syn \frac{3\pi}{4} \\
&\Leftrightarrow \begin{cases} x - \dfrac{\pi}{4} = 2\kappa\pi + \dfrac{3\pi}{4} \\ \text{ή} \\ x - \dfrac{\pi}{4} = 2\kappa\pi - \dfrac{3\pi}{4} \end{cases} \Leftrightarrow \begin{cases} x = 2\kappa\pi + \pi \\ \text{ή} \qquad \mu\varepsilon\ \kappa \in \mathbb{Z} \\ x = 2\kappa\pi - \dfrac{\pi}{2} \end{cases}
\end{align*}
\end{alist}
