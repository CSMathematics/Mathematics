\begin{alist}
\item Το σημείο $\text{M}$ είναι η τομή της τελικής πλευράς $\text{ΟZ}$ της γωνίας $\omega$ με τον τριγωνομετρικό κύκλο, επομένως $\text{M}(\sigma\upsilon\nu\omega, \eta\mu\omega)$ με $\sigma\upsilon\nu\omega = -\dfrac{1}{3}$.

\item Από τον τύπο $\eta\mu^2\omega + \sigma\upsilon\nu^2\omega = 1$ προκύπτει ότι $\eta\mu^2\omega = 1 - \sigma\upsilon\nu^2\omega$. \\
Αντικαθιστούμε το $\sigma\upsilon\nu\omega = -\dfrac{1}{3}$ και έχουμε:
\[\eta\mu^2\omega = 1 - \left(-\frac{1}{3}\right)^2 = 1 - \frac{1}{9} = \frac{8}{9}.\]
Επειδή $\pi < \omega < \dfrac{3\pi}{2}$, είναι $\eta\mu\omega < 0$, οπότε έχουμε $\eta\mu\omega = -\dfrac{2\sqrt{2}}{3}$. \\
Επομένως $\varepsilon\varphi\omega = \dfrac{\eta\mu\omega}{\sigma\upsilon\nu\omega} = 2\sqrt{2}$.

\item Έχουμε:
\[\eta\mu(\pi - \omega) = \eta\mu\omega = -\frac{2\sqrt{2}}{3}, \quad \sigma\upsilon\nu(\pi - \omega) = -\sigma\upsilon\nu\omega = \frac{1}{3}.\]
\end{alist}
