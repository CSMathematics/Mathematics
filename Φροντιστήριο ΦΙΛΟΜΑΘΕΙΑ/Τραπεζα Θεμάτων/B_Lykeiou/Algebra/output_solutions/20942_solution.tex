\lysh{20942}{2}
\begin{alist}
\item Το σημείο $\text{M}$ είναι η τομή της τελικής πλευράς $\text{ΟZ}$ της γωνίας $\omega$ με τον τριγωνομετρικό κύκλο, επομένως $\text{M}(\syn \omega, \hm \omega)$ με $\syn \omega = -\dfrac{1}{3}$.
\item Από τον τύπο $\hm ^2\omega + \syn ^2\omega = 1$ προκύπτει ότι $\hm ^2\omega = 1 - \syn ^2\omega$. \\
Αντικαθιστούμε το $\syn \omega = -\dfrac{1}{3}$ και έχουμε:
\[\hm ^2\omega = 1 - \left(-\frac{1}{3}\right)^2 = 1 - \frac{1}{9} = \frac{8}{9}.\]
Επειδή $\pi < \omega < \dfrac{3\pi}{2}$, είναι $\hm \omega < 0$, οπότε έχουμε $\hm \omega = -\dfrac{2\sqrt{2}}{3}$. \\
Επομένως $\ef \omega = \dfrac{\hm \omega}{\syn \omega} = 2\sqrt{2}$.
\item Έχουμε:
\[\hm (\pi - \omega) = \hm \omega = -\frac{2\sqrt{2}}{3}, \quad \syn (\pi - \omega) = -\syn \omega = \frac{1}{3}.\]
\end{alist}
