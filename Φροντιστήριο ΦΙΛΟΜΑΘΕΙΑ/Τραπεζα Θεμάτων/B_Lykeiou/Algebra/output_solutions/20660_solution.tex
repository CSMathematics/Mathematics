\[f(x) = \eta\mu(\pi-x) + \sigma\upsilon\nu\left(\frac{\pi}{2}-x\right) = \eta\mu x + \eta\mu x = 2\eta\mu x, \quad x \in \mathbb{R}.\]

\begin{rlist}

\[-1 \le \eta\mu x \le 1 \Leftrightarrow -2 \le 2\eta\mu x \le 2.\]
