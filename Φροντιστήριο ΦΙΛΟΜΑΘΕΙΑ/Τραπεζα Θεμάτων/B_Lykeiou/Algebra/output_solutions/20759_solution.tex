\lysh{20759}{4}
\begin{alist}
\item Από τα δεδομένα έχουμε ότι $(OAM) = \dfrac{4}{6} \Leftrightarrow \dfrac{(OA)\cdot(AM)}{2} = \dfrac{1\cdot(AM)}{2} = \dfrac{4}{6} \Leftrightarrow (AM) = \dfrac{4}{3}$ \\
άρα $y_M = \dfrac{4}{3}$, επομένως $\ef \omega = \dfrac{4}{3}$. Η γωνία $\hat{A}$ του τριγώνου $\text{AOM}$ είναι ορθή, επομένως για τη γωνία $\omega = \text{A}\hat{\text{O}}\text{M}$ ισχύει $0 < \omega < \dfrac{\pi}{2}$.
\item Έχουμε ότι
\[\hm ^2\omega + \syn ^2\omega = 1 \Leftrightarrow \ef ^2\omega + 1 = \frac{1}{\syn ^2\omega} \Leftrightarrow \left(\frac{4}{3}\right)^2 + 1 = \frac{1}{\syn ^2\omega} \Leftrightarrow \frac{25}{9} = \frac{1}{\syn ^2\omega}, \quad \omega \in \left(0, \frac{\pi}{2}\right)\]
άρα $\syn \omega = \dfrac{3}{5}$ και $\syf \omega = \dfrac{1}{\ef \omega} = \dfrac{1}{\dfrac{4}{3}} = \dfrac{3}{4}$. Επίσης ισχύει $0 < \omega < \dfrac{\pi}{2}$, άρα
\[\hm \omega = \sqrt{1 - \syn ^2\omega} = \sqrt{1 - \left(\frac{3}{5}\right)^2} = \sqrt{1 - \frac{9}{25}} = \sqrt{\frac{16}{25}} = \frac{4}{5}\]
\item Για τις τιμές που βρήκαμε στο προηγούμενο ερώτημα η συνάρτηση $f$ γίνεται $f(x) = \hm ^2x - 5\cdot\dfrac{4}{5}\hm  x + 5\cdot\dfrac{3}{5} = \hm ^2x - 4\hm  x + 3$. Ζητάμε τις λύσεις της εξίσωσης $f(x) = 0 \Leftrightarrow \hm ^2x - 4\hm  x + 3 = 0$ η οποία είναι δευτεροβάθμια εξίσωση με άγνωστο το $\hm  x$. \\
Θέτοντας $\hm  x = \omega$ παίρνει τη μορφή $\omega^2 - 4\omega + 3 = 0$ και έχει λύσεις $\omega = 1$ ή $\omega = 3$. Η δεύτερη από αυτές απορρίπτεται καθώς ισχύει ότι $-1 \le \hm  x \le 1$. Επομένως η εξίσωση $f(x) = 0$ είναι ισοδύναμη με την εξίσωση 
\[\hm  x = 1 \Leftrightarrow \hm  x = \hm \frac{\pi}{2} \Leftrightarrow x = 2\kappa\pi + \frac{\pi}{2}, \quad \kappa \in \mathbb{Z}\]
Επομένως τα ζητούμενα σημεία τομής είναι όλα τα σημεία της μορφής $\left(2\kappa\pi + \dfrac{\pi}{2}, 0\right), \, \kappa \in \mathbb{Z}$
\end{alist}
