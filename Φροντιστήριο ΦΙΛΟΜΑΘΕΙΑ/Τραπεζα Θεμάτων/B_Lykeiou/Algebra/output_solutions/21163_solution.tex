\lysh{21163}{2}
\begin{alist}
\item Εφόσον το σημείο $\text{A}$ ανήκει στη γραφική παράσταση της συνάρτησης $f(x) = \alpha^x$ οι συντεταγμένες του θα ικανοποιούν τον τύπο της και επειδή $0 < \alpha < 1$, θα ισχύει ότι
\[\alpha^1 = \frac{1}{2} \Leftrightarrow \alpha = \frac{1}{2}.\]
\item 
\begin{rlist}
\item Η συνάρτηση $f$ έχει τύπο $f(x) = \left(\dfrac{1}{2}\right)^x$ και είναι εκθετική με βάση $0 < \alpha < 1$ οπότε θα είναι μία γνησίως φθίνουσα συνάρτηση.
\item Η συνάρτηση $f$ είναι γνησίως φθίνουσα και ισχύει $\sqrt{2} < \sqrt{3}$, οπότε θα έχουμε ότι
\[f(\sqrt{2}) > f(\sqrt{3}) \Rightarrow \alpha^{\sqrt{2}} > \alpha^{\sqrt{3}}.\]
\end{rlist}
\end{alist}
