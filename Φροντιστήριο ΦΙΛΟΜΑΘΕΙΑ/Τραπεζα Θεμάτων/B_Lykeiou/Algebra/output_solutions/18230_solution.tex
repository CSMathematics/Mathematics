\lysh{18230}{2}
\begin{alist}
\item Το πολυώνυμο έχει παράγοντα το $(x-2)$, μόνο όταν $\text{P}(2) = 0$. \\
Πραγματικά,
\[\text{P}(2) = 2\cdot 8 + 4 - 8\cdot 2 - 4 = 16 + 4 - 16 - 4 = 0\]
οπότε το $(x-2)$ είναι παράγοντας του πολυωνύμου.
\item Είναι:
\begin{align*}
\text{P}(x) &= 2x^3 + x^2 - 8x - 4 \\
&= 2x^3 - 8x + x^2 - 4 \\
&= 2x(x^2 - 4) + (x^2 - 4) \\
&= (x^2 - 4)(2x + 1) \\
&= (x-2)(x+2)(2x+1).
\end{align*}
\item Ισχύει:
\[\text{P}(x) = 0 \Leftrightarrow (2x+1)(x-2)(x+2) = 0 \Leftrightarrow x = -\frac{1}{2} \text{ ή } x = 2 \text{ ή } x = -2.\]
Επομένως η εξίσωση έχει ρίζες τους αριθμούς $-\dfrac{1}{2}, 2$ και $-2$.
\end{alist}
