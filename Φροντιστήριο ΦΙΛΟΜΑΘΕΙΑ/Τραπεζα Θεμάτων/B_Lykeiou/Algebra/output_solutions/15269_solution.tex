\lysh{15269}{2}
\begin{alist}
\item Η γραφική παράσταση της $f$ αποτελείται από την $y = e^x$ για $x < 0$ και την $y = e^{-x}$ για $x \ge 0$ οπότε ο τύπος της είναι ο πρώτος από τους δοσμένους τύπους.
\begin{center}

\begin{axis}[

    xtick={-2.5, -2, -1.5, -1, -0.5, 0, 0.5, 1, 1.5, 2, 2.5},
    ytick={-0.5, 0, 0.5},
    extra y ticks={1},

    samples=100
]
\addplot[domain=-2.8:0, thick, maincolor] {exp(x)} node[pos=0.6, above left, font=\scriptsize] {$y=e^x$};
\addplot[domain=0:2.8, thick, maincolor] {exp(-x)} node[pos=0.6, above right, font=\scriptsize] {$y=e^{-x}$};
\end{axis}
\end{tikzpicture}
\end{center}
\item Από την παραπάνω γραφική παράσταση της $f$ συμπεραίνουμε ότι αυτή είναι:
\begin{itemize}
    \item γνησίως αύξουσα στο διάστημα $(-\infty, 0]$
    \item γνησίως φθίνουσα στο διάστημα $[0, +\infty)$ και
    \item παρουσιάζει ολικό μέγιστο για $x = 0$, το $f(0) = 1$
\end{itemize}
\item Διακρίνουμε τις παρακάτω δυνατές περιπτώσεις:
\begin{itemize}
    \item Αν $\alpha \le 0$, (ευθεία $\varepsilon_1$) τότε η $C_f$ και η ευθεία $y = \alpha$ δεν έχουν κοινό σημείο.
    \item Αν $0 < \alpha < 1$, (ευθεία $\varepsilon_2$) τότε η $C_f$ και η ευθεία $y = \alpha$ έχουν ακριβώς δυο κοινά σημεία.
    \item Αν $\alpha = 1$, (ευθεία $\varepsilon_3$) τότε η $C_f$ και η ευθεία $y = \alpha$ έχουν ένα μόνο κοινό σημείο.
    \item Αν $\alpha > 1$, (ευθεία $\varepsilon_4$) τότε η $C_f$ και η ευθεία $y = \alpha$ δεν έχουν κοινό σημείο.
\end{itemize}
