\begin{rlist}
\item Έχουμε $\hm  \dfrac{2\pi}{3} = \hm  \left( \pi - \dfrac{\pi}{3} \right) = \hm  \dfrac{\pi}{3} = \dfrac{\sqrt{3}}{2}$.
\item Έχουμε $\hm  \theta = \dfrac{\hm  \frac{2\pi}{3} - \syn  \frac{\pi}{3}}{\syn ^2 \frac{\pi}{4}} \overset{\text{i}}{=} \dfrac{ \frac{\sqrt{3}}{2} - \frac{1}{2}}{ \left(\frac{\sqrt{2}}{2}\right)^2 } = \dfrac{ \frac{\sqrt{3}}{2} - \frac{1}{2} }{ \frac{1}{2} } = \sqrt{3} - 1$.
\end{rlist}
\item Έχουμε $\syn  \theta = \pm \sqrt{1 - \hm ^2 \theta} = \pm \sqrt{1 - \left( \sqrt{3} - 1 \right)^2} = \pm \sqrt{2\sqrt{3} - 3} = \sqrt{2\sqrt{3} - 3}$, δεδομένου ότι αν μια γωνία ανήκει στο 1ο τεταρτημόριο το συνημίτονό της είναι θετικός αριθμός.
\end{alist}
