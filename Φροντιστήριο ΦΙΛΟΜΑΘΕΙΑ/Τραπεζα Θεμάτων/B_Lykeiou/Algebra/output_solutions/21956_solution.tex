\item Είναι
\[A = 2\log 5 + 3\log 2 - \log 20 = \log 5^2 + \log 2^3 - \log 20 = \log 25 + \log 8 - \log 20 = \log\frac{25 \cdot 8}{20} = \log 10 = 1\]

\item Για $e^x > 1 \Leftrightarrow x > 0$ είναι
\begin{align*}
\ln(e^x - 1) = A &\Leftrightarrow \\
\ln(e^x - 1) = 1 &\Leftrightarrow \\
e^x - 1 = e &\Leftrightarrow \\
e^x = e + 1 &\Leftrightarrow \\
x = \ln(e + 1)
\end{align*}
που είναι δεκτή αφού $\ln(e + 1) > \ln 1 \Leftrightarrow \ln(e + 1) > 0$.
\end{alist}
