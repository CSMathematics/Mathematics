\lysh{15266}{2}
\begin{alist}
\item Η τελική πλευρά της γωνίας $\theta$ είναι η $OP$ και το σημείο $P$ έχει τετμημένη $x = \dfrac{3}{5}$. Άρα \[\syn \theta = \frac{3}{5}.\]
\item Από την γνωστή ταυτότητα $\syn ^2\theta + \hm ^2\theta = 1$, έχουμε $\dfrac{9}{25} + \hm ^2\theta = 1$ απ' όπου προκύπτει ότι $\hm ^2\theta = 1 - \dfrac{9}{25} = \dfrac{16}{25}$. \\
Αλλά $0 < \theta < \dfrac{\pi}{2}$, οπότε $\hm \theta > 0$, άρα $\hm \theta = \dfrac{4}{5}$.
\item Επειδή η $O\Sigma$ είναι συμμετρική της $OP$ ως προς τον άξονα $x'x$, το σημείο $\Sigma$ έχει ίδια τετμημένη με το $P$ και αντίθετη τεταγμένη. \\
Άρα, $\syn (-\theta) = \syn \theta = \dfrac{3}{5}$ και $\hm (-\theta) = -\hm \theta = -\dfrac{4}{5}$
\end{alist}
