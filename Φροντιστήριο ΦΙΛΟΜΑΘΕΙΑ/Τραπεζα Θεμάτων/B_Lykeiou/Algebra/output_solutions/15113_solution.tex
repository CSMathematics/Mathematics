\begin{alist}
\item Έχουμε:
\begin{align*}
P(x) &= -2x^3 + 4x^2 + 2(x^3 - 1) + 9 \Leftrightarrow \\
P(x) &= -2x^3 + 4x^2 + 2x^3 - 2 + 9 \Leftrightarrow \\
P(x) &= 0x^3 + 4x^2 + 7 \Leftrightarrow \\
P(x) &= 4x^2 + 7.
\end{align*}
Συνεπώς το πολυώνυμο $P(x)$ είναι $2^{ou}$ βαθμού.
\item Για να είναι τα πολυώνυμα $P(x)$ και $Q(x)$ ίσα, πρέπει να είναι ίδιου βαθμού και να έχουν τους αντίστοιχους συντελεστές ίσους. Άρα πρέπει $\alpha = 4$.
\end{alist}
