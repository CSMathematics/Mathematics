\lysh{15288}{4}
\begin{alist}
\item Η περίοδος της συνάρτησης είναι $T = \dfrac{2\pi}{3}$, η μέγιστη τιμή της είναι $\max f(x) = 2 + 1 = 3$, και η ελάχιστη τιμή της είναι $\min f(x) = -2 + 1 = -1$.
\item
\begin{rlist}
\item Από την γραφική παράσταση της ημιτονοειδούς συνάρτησης $g$, παρατηρούμε ότι παρουσιάζει μέγιστο για $x = -\dfrac{\pi}{2}$ και το επόμενο μέγιστο για $x = \dfrac{3\pi}{2}$. Άρα η περίοδος της συνάρτησης είναι $T = \dfrac{3\pi}{2} - \left(-\dfrac{\pi}{2}\right) = 2\pi$, οπότε $\beta = 1$. Η καμπύλη προκύπτει από κατακόρυφη μετατόπιση της γραφικής παράστασης της $\alpha\hm  x$ κατά $1$ μονάδα προς τα πάνω, άρα $\gamma = 1$. Επίσης
\begin{align*}
g\left(\frac{3\pi}{2}\right) &= 3 \Leftrightarrow \\
\alpha \cdot \hm \frac{3\pi}{2} + 1 &= 3 \Leftrightarrow \\
\alpha \cdot (-1) &= 2 \Leftrightarrow \\
\alpha &= -2
\end{align*}
Τελικά $g(x) = -2\hm  x + 1$.
\item Η εξίσωση γίνεται:
\begin{align*}
f(x) &= g(x) \Leftrightarrow \\
2\hm  3x + 1 &= -2\hm  x + 1 \Leftrightarrow \\
\hm  3x &= -\hm  x \Leftrightarrow \\
\hm  3x &= \hm (-x) \Leftrightarrow \\
\begin{cases} 3x = 2\kappa\pi - x \\ \text{ή} \\ 3x = 2\kappa\pi + (\pi + x) \end{cases}&, \kappa \in \mathbb{Z}.
\end{align*}
Δηλαδή:
\[\begin{cases} x = \dfrac{\kappa\pi}{2} \\ \text{ή} \\ x = \kappa\pi + \dfrac{\pi}{2} \end{cases}, \kappa \in \mathbb{Z}.\]
Επειδή $x \in [0, \pi)$, για $\kappa = 0$ στον πρώτο τύπο λύσεων προκύπτει $x = 0$ και στον δεύτερο τύπο λύσεων προκύπτει $x = \dfrac{\pi}{2}$ (που προκύπτει και από τον πρώτο τύπο για $\kappa=1$).
\end{rlist}
\end{alist}
