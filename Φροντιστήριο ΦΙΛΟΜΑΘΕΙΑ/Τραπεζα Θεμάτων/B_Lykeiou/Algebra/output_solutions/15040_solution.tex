\lysh{15040}{2}
\begin{alist}
\item Με $x = 1$ έχουμε:
\[x^3 - 7x + 6 = 1^3 - 7 \cdot 1 + 6 = 7 - 7 = 0\]
οπότε ο αριθμός 1 είναι ρίζα της εξίσωσης.
\item Όπως φαίνεται στο διπλανό σχήμα Horner το πηλίκο της διαίρεσης $(x^3 - 7x + 6) : (x - 1)$ είναι $x^2 + x - 6$ και το υπόλοιπο $0$, οπότε η ταυτότητα της διαίρεσης είναι η:
\begin{center}
\Horner{1,0,-7,6}{1}
\end{center}
\[x^3 - 7x + 6 = (x - 1)(x^2 + x - 6)\]
\vspace{0.5cm}
\item Από το ερώτημα β) έχουμε:
\[x^3 - 7x + 6 = 0 \Leftrightarrow (x - 1)(x^2 + x - 6) = 0 \Leftrightarrow x = 1 \text{ ή } x = 2 \text{ ή } x = -3\]
\end{alist}
