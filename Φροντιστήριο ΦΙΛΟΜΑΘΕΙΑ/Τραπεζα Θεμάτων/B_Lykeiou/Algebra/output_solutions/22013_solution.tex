\lysh{22013}{4}
\begin{alist}
\item Η ποσότητα $x^4 + 1$ είναι αυστηρά θετική, μάλιστα $x^4 + 1 \ge 0 + 1 = 1$ για κάθε $x \in \mathbb{R}$. \\
Οπότε, το $x^4 + 1$ δεν μηδενίζεται για καμιά τιμή του $x$ και συνεπώς, το πολυώνυμο $P(x)$ δεν έχει πραγματικές ρίζες. \\
\textit{Εναλλακτικά:} Αν υπήρχε πραγματικός αριθμός $\rho$ τέτοιος ώστε $\rho^4 + 1 = 0$, τότε θα έπρεπε $\rho^4 = -1$, πράγμα άτοπο.
\item Θα εκτελέσουμε τις πράξεις στο δεύτερο μέλος και στη συνέχεια θα εξισώσουμε τους συντελεστές των δυνάμεων του $x$.
\begin{align*}
x^4 + 1 = (x^2 + \alpha x + 1) \cdot (x^2 + \beta x + 1) &\Leftrightarrow \\
x^4 + 1 = x^4 + (\alpha + \beta)x^3 + (\alpha\beta + 2)x^2 + (\alpha + \beta)x + 1 &\Leftrightarrow \\
\begin{cases} \alpha + \beta = 0 \\ \alpha \cdot \beta + 2 = 0 \\ \alpha + \beta = 0 \end{cases} \Leftrightarrow \begin{cases} \alpha = -\beta \\ \alpha \cdot \beta = -2 \end{cases} \Leftrightarrow \begin{cases} \alpha = -\beta \\ \alpha^2 = 2 \end{cases} \Leftrightarrow \begin{cases} \alpha = -\beta \\ \alpha = \pm \sqrt{2} \end{cases} &\Leftrightarrow \\
(\alpha = \sqrt{2} \text{ και } \beta = -\sqrt{2}) \text{ ή } (\alpha = -\sqrt{2} \text{ και } \beta = \sqrt{2}). &
\end{align*}
Άρα: $\alpha = \sqrt{2}$ και $\beta = -\sqrt{2}$ ή $\alpha = -\sqrt{2}$ και $\beta = \sqrt{2}$. \\
Συνεπώς, $x^4 + 1 = (x^2 + \sqrt{2}x + 1) \cdot (x^2 - \sqrt{2}x + 1)$.
\item Η πρόταση είναι λάθος, σύμφωνα με τα $\alpha$) και $\beta$) ερώτημα. Το αντιπαράδειγμα είναι το πολυώνυμο $P(x) = x^4 + 1$ το οποίο αναλύεται σε γινόμενο δύο πολυωνύμων $2^{ou}$ βαθμού:
\[x^4 + 1 = (x^2 + \sqrt{2}x + 1) \cdot (x^2 - \sqrt{2}x + 1),\]
αλλά δεν έχει πραγματικές ρίζες.
\end{alist}
