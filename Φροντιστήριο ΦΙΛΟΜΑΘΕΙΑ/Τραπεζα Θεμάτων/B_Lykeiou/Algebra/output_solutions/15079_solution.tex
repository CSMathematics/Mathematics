\begin{alist}
\item Το συνημίτονο μιας γωνίας σχεδιασμένης στον τριγωνομετρικό κύκλο είναι η τετμημένη του σημείου τομής της τελικής πλευράς της με τον κύκλο. Επειδή η τετμημένη του σημείου A είναι $0,6 = \dfrac{3}{5}$, έχουμε $\sigma\upsilon\nu\omega = \dfrac{3}{5}$.

\item
\begin{rlist}
\item Εφόσον η ΟΓ είναι προέκταση της ΟA έχουμε $\text{A}\hat{\text{O}}\Gamma = \pi \text{ rad}$. Επομένως $\text{B}\hat{\text{O}}\Gamma = \hat{\varphi} = \pi + \hat{\omega}$.
\item Το συνημίτονο της γωνίας $\hat{\varphi}$ είναι η τετμημένη του σημείου $\Gamma$, δηλαδή $\sigma\upsilon\nu\varphi = -\dfrac{3}{5}$.
\end{rlist}
\end{alist}
