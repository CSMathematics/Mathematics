\lysh{15642}{2}
\begin{alist}
\item Ένα πολυώνυμο $P(x)$ έχει παράγοντα το $x - \rho$ αν και μόνο αν $P(\rho) = 0$. Έχουμε
\[P(1) = 2(1-1)^{20} - 3(1-1)^{10} + 5 \cdot 1^2 - 3 \cdot 1 - 2 = 0,\]
άρα το $x-1$ είναι παράγοντας του $P(x)$.
\item
\begin{rlist}
\item Είναι $P(0) = 2(0-1)^{20} - 3(0-1)^{10} + 5 \cdot 0^2 - 3 \cdot 0 - 2 = 2 - 3 - 2 = -3$.
\item Αφού $P(0) = -3 \ne 0$, το $x$ δεν είναι παράγοντας του πολυωνύμου $P(x)$.
\end{rlist}
\end{alist}
