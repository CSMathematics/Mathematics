\begin{alist}
\item Το $\rho$ καθορίζει τη μέγιστη τιμή της συνάρτησης $f$, που είναι ίση με $\rho$ και την ελάχιστη τιμή της, που είναι ίση με $-\rho$. Άρα $\rho = 2$. \\
Το $\omega$ καθορίζει την περίοδο της συνάρτησης $g$, που είναι ίση με $\dfrac{2\pi}{\omega}$. \\
Άρα $\dfrac{2\pi}{\omega} = \pi \Leftrightarrow \omega = 2$.

\item 
\begin{rlist}
\item Για τη συνάρτηση $f(x) = 2\eta\mu x$, με $x \in [0, \pi]$, είναι:
\begin{center}
\renewcommand{\arraystretch}{2}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & $0$ & $\dfrac{\pi}{2}$ & $\pi$ \\ \hline
$f(x) = 2\eta\mu x$ & $0$ & $2$ & $0$ \\ \hline
\end{tabular}
\end{center}
Για τη συνάρτηση $g(x) = \eta\mu(2x)$, με $x \in [0, \pi]$, είναι:
\begin{center}
\renewcommand{\arraystretch}{2}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $0$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{2}$ & $\dfrac{3\pi}{4}$ & $\pi$ \\ \hline
$2x$ & $0$ & $\dfrac{\pi}{2}$ & $\pi$ & $\dfrac{3\pi}{2}$ & $2\pi$ \\ \hline
$g(x) = \eta\mu(2x)$ & $0$ & $1$ & $0$ & $-1$ & $0$ \\ \hline
\end{tabular}
\end{center}
