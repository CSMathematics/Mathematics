\begin{alist}
\item Αν μετατοπίσουμε τη $C_f$ μια μονάδα προς τα πάνω, τότε ο τύπος που αντιστοιχεί στη γραφική παράσταση που προκύπτει $\varphi(x) = x^2 + 1, x \in \mathbb{R}$.

\item Η γραφική παράσταση της συνάρτησης $\varphi(x)$ φαίνεται στο παρακάτω σχήμα.

\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-2.5, xmax=2.5,
    ymin=-0.5, ymax=5.5,
    xtick={-2,-1,0,1,2},
    ytick={0,1,2,3,4,5},
    thick,
    width=8cm, height=10cm,
    samples=100
]
\addplot[domain=-2.2:2.2, densely dotted, very thick] {x^2} node[pos=0.8, right, black] {\small $C_f$};
\addplot[domain=-2.1:2.1, maincolor, very thick] {x^2 + 1} node[pos=0.8, left, black] {\small $C_\varphi$};
\end{axis}
\end{tikzpicture}
\end{center}

\item Από το παραπάνω σχήμα προκύπτει ότι η $\varphi(x)$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, 0]$, γνησίως αύξουσα στο διάστημα $[0, +\infty)$ και παρουσιάζει ελάχιστο για $x = 0$. Η ελάχιστη τιμή της είναι ίση με $\varphi(0) = 1$.
\end{alist}
