\lysh{18693}{4}
\begin{alist}
\item Η συνάρτηση $f$ με $f(x) = \left(\dfrac{2-\lambda}{4}\right)^x$ είναι εκθετική και ορίζεται στο $\mathbb{R}$ όταν η βάση είναι θετική και διάφορη της μονάδας. Δηλαδή, (i) $\dfrac{2-\lambda}{4} > 0$ ή ισοδύναμα $\lambda < 2$ και (ii) $\dfrac{2-\lambda}{4} \neq 1$ ή ισοδύναμα $\lambda \neq -2$. Άρα, $\lambda \in (-\infty, -2) \cup (-2, 2)$.
\item Μία εκθετική συνάρτηση της μορφής $\alpha^x$ είναι γνησίως φθίνουσα όταν $0 < \alpha < 1$. \\
Επομένως, $0 < \dfrac{2-\lambda}{4} < 1$ ή ισοδύναμα $-2 < \lambda < 2$.
\item Για $\lambda = 0$ έχουμε την συνάρτηση $f(x) = \left(\dfrac{1}{2}\right)^x$ με $x \in \mathbb{R}$.
\begin{rlist}
\item Η $f$ είναι εκθετική με βάση μικρότερη της μονάδας, άρα είναι γνησίως φθίνουσα συνάρτηση. Η γραφική της παράσταση, με βάση τον πίνακα τιμών, φαίνεται στο παρακάτω σχήμα:

\item Η εξίσωση $f(x) + f(x+1) = 6$ μετά την αντικατάσταση γίνεται
\[\left(\frac{1}{2}\right)^x + \left(\frac{1}{2}\right)^{x+1} = 6 \Leftrightarrow \left(\frac{1}{2}\right)^x + \frac{1}{2}\left(\frac{1}{2}\right)^x = 6.\]
Άρα, $\dfrac{3}{2}\left(\dfrac{1}{2}\right)^x = 6 \Leftrightarrow \left(\dfrac{1}{2}\right)^x = 4 \Leftrightarrow 2^{-x} = 2^2$ συνεπώς $x = -2$.
\end{rlist}
