\lysh{15287}{4}
\begin{alist}
\item Η συνάρτηση $f(x) = \rho\hm (\omega x)$ με $\rho > 0$, έχει μέγιστη τιμή $\rho$. Με βάση το σχήμα η συνάρτηση έχει μέγιστη τιμή 3, άρα $\rho = 3$. Επίσης η περίοδος της συνάρτησης είναι $\pi$, οπότε $\pi = \dfrac{2\pi}{\omega} \Leftrightarrow \omega = 2$. Άρα $f(x) = 3\hm (2x)$.
\item Η ευθεία $y = \alpha x$ διέρχεται από το σημείο $E$ της γραφικής παράστασης της $f$ που έχει τεταγμένη 3, η οποία είναι η μέγιστη τιμή της συνάρτησης. Η συνάρτηση $f(x) = 3\hm (2x)$ παρουσιάζει μέγιστη τιμή σε διάστημα μιας περιόδου στο $\dfrac{1}{4}$ της περιόδου, δηλαδή στη θέση $x = \dfrac{\pi}{4}$. Άρα είναι $E\left(\dfrac{\pi}{4}, 3\right)$ και $3 = \alpha \cdot \dfrac{\pi}{4} \Leftrightarrow \alpha = \dfrac{12}{\pi}$. Οπότε η εξίσωση της ευθείας είναι:
\[y = \frac{12}{\pi}x.\]
\item Η εξίσωση $3\hm (2x) - \dfrac{12}{\pi}x = 0$ γράφεται ισοδύναμα $3\hm (2x) = \dfrac{12}{\pi}x$. Οι λύσεις της εξίσωσης είναι οι τετμημένες των σημείων τομής της γραφικής παράστασης της $f(x) = 3\hm (2x)$ με την ευθεία $y = \dfrac{12}{\pi}x$. Με βάση το σχήμα τα σημεία τομής είναι 3, οι τετμημένες των οποίων είναι
\begin{itemize}
    \item $x = 0$, δεδομένου ότι $f(0) = 3\hm  0 = 0$ και η ευθεία $y = \dfrac{12}{\pi}x$ διέρχεται από το σημείο $(0,0)$.
    \item $x = \dfrac{\pi}{4}$, δεδομένου ότι $f\left(\dfrac{\pi}{4}\right) = 3\hm \left(2 \cdot \dfrac{\pi}{4}\right) = 3\hm \left(\dfrac{\pi}{2}\right) = 3 \cdot 1 = 3$ και η ευθεία $y = \dfrac{12}{\pi}x$ διέρχεται από το σημείο $\left(\dfrac{\pi}{4}, \dfrac{12}{\pi} \cdot \dfrac{\pi}{4}\right) = \left(\dfrac{\pi}{4}, 3\right)$ και
    \item $x = -\dfrac{\pi}{4}$, δεδομένου ότι $f\left(-\dfrac{\pi}{4}\right) = 3\hm \left(-2 \cdot \dfrac{\pi}{4}\right) = 3\hm \left(-\dfrac{\pi}{2}\right) = 3 \cdot (-1) = -3$ και η ευθεία $y = \dfrac{12}{\pi}x$ διέρχεται από το σημείο $\left(-\dfrac{\pi}{4}, \dfrac{12}{\pi} \cdot \left(-\dfrac{\pi}{4}\right)\right) = \left(-\dfrac{\pi}{4}, -3\right)$.
\end{itemize}
Άρα η εξίσωση $3\hm (2x) - \dfrac{12}{\pi}x = 0$ έχει λύσεις τις $x = 0$, $x = \dfrac{\pi}{4}$ και $x = -\dfrac{\pi}{4}$.
\end{alist}
