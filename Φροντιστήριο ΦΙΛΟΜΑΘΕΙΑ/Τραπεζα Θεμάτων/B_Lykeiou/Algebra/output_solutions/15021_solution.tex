\lysh{15021}{4}
\begin{alist}
\item Η συνάρτηση $f$ ορίζεται για κάθε πραγματική τιμή του $x$ για την οποία ισχύει $x \ne 0$.
Συνεπώς το πεδίο ορισμού της $f$ είναι το $A = (-\infty, 0) \cup (0, +\infty)$.
\item Για κάθε $x \in A$ και $-x \in A$. Επίσης,
\[f(-x) = \dfrac{(-x)^2 + 1}{-x} = -\dfrac{x^2 + 1}{x} = -f(x), \quad \text{για κάθε } x \in A,\]
που σημαίνει ότι η συνάρτηση $f$ είναι περιττή, δηλαδή η γραφική της παράσταση έχει κέντρο συμμετρίας το $O(0, 0)$.
\item Είναι $\ln\dfrac{1}{2} = \ln 1 - \ln 2 = -\ln 2$ και αφού $f$ περιττή είναι $f\left(\ln\dfrac{1}{2}\right) = f(-\ln 2) = -f(\ln 2)$, δηλαδή $f(\ln 2) + f\left(\ln\dfrac{1}{2}\right) = 0$.
\item Για κάθε $\theta \in \mathbb{R}$ είναι $\hm (\pi + \theta) = -\hm \theta$ και αφού $f$ περιττή είναι $f(\hm (\pi + \theta)) = f(-\hm \theta) = -f(\hm \theta)$, δηλαδή $f(\hm \theta) + f(\hm (\pi + \theta)) = 0$.
\end{alist}
