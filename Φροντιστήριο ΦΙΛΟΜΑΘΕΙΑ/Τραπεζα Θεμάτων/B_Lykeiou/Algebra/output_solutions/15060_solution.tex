\lysh{15060}{2}
\begin{alist}
\item Η τεταγμένη του σημείου $M$ είναι ίση με $\dfrac{1}{2}$, οπότε $\hm \theta = \dfrac{1}{2}$.
\item Η γωνία $\theta$ περιέχεται στο δεύτερο τεταρτημόριο, όπου ισχύει $\syn \theta < 0$. Από την βασική ταυτότητα $\hm ^2\theta + \syn ^2\theta = 1$, έχουμε:
\[\syn ^2\theta = 1 - \hm ^2\theta = 1 - \dfrac{1}{4} = \dfrac{3}{4}\]
και επειδή ισχύει $\syn \theta < 0$, παίρνουμε $\syn \theta = -\dfrac{\sqrt{3}}{2}$.
\item Ισχύει $\hm \theta = \dfrac{1}{2} = \hm \dfrac{\pi}{6}$ και $\dfrac{\pi}{2} < \theta < \pi$, οπότε η γωνία $\theta$ είναι η παραπληρωματική της γωνίας $\dfrac{\pi}{6}$, δηλαδή
\[\theta = \pi - \dfrac{\pi}{6} = \dfrac{5\pi}{6}.\]
\end{alist}
