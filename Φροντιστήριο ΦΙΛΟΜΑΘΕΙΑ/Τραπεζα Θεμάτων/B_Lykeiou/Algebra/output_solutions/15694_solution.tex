\begin{alist}
\item Θέλουμε να είναι $m - M < 0 \Leftrightarrow 5 \cdot \log\left(\dfrac{d}{10}\right) < 0 \Leftrightarrow \log\left(\dfrac{d}{10}\right) < 0 \Leftrightarrow \log\left(\dfrac{d}{10}\right) < \log 1$ \\
επομένως $\dfrac{d}{10} < 1 \Leftrightarrow d < 10$, αφού η συνάρτηση $f(x) = \log x$ είναι γνησίως αύξουσα στο $(0, +\infty)$.

\item Έχουμε $M = m - 5 \cdot \log\left(\dfrac{d}{10}\right) = 1{,}157 - 5 \cdot \log\left(\dfrac{100}{10}\right) = 1{,}157 - 5 \cdot \log 10 = 1{,}157 - 5 = -3{,}843$.

\item Είναι $\log\left(\dfrac{d}{10}\right) = \dfrac{m-M}{5} \Leftrightarrow \dfrac{d}{10} = 10^{\frac{m-M}{5}}$, άρα $d = 10 \cdot 10^{\frac{m-M}{5}} = 10^{1 + \frac{m-M}{5}} = 10^{\frac{5+m-M}{5}}$.

\item Με χρήση της παραπάνω σχέσης, έχουμε
\[d = 10^{\frac{5+0{,}46+5{,}14}{5}} = 10^{\frac{10{,}6}{5}} = 10^{\frac{53}{25}} = \sqrt[25]{10^{53}} \cong 131 \text{ parsec}.\]
\end{alist}
