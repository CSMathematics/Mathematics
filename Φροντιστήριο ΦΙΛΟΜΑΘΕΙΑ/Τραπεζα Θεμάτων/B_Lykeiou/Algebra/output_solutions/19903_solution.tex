\begin{alist}
\item Ισχύει ότι:
\begin{align*}
\log 100 &= \log 10^2 = 2\log 10 = 2, \\
\log 2 + \log 5 &= \log(2 \cdot 5) = \log 10 = 1 \quad \text{και} \\
\log 1 &= 0.
\end{align*}
Άρα, $\alpha = 2 + 1 + 0 = 3$.

\item Για $\alpha = 3$ η εξίσωση γράφεται:
\begin{align*}
9 \cdot 2^x &= 4 \cdot 3^x \Leftrightarrow \\
\frac{2^x}{3^x} &= \frac{4}{9} \Leftrightarrow \\
\left(\frac{2}{3}\right)^x &= \left(\frac{2}{3}\right)^2 \Leftrightarrow x = 2.
\end{align*}
\end{alist}
