\lysh{20725}{2}
\begin{alist}
\item Οι συναρτήσεις $f$ και $g$ ως λογαριθμικές ορίζονται για $x > 0$ και $x + 2 > 0 \Leftrightarrow x > -2$ αντίστοιχα. Επομένως, πεδίο ορισμού της συνάρτησης $f$ είναι το $D_f = (0, +\infty)$ και πεδίο ορισμού της συνάρτησης $g$ είναι το $D_g = (-2, +\infty)$.
\item 
\begin{rlist}
\item Η εξίσωση $f(x) = 2$ είναι ισοδύναμη με την εξίσωση $\log x = 2$. Για να βρούμε το $x>0$ (περιορισμός από το $\alpha$) ερώτημα) τέτοιο ώστε $\log x = 2$ θα χρησιμοποιήσουμε τον ορισμό του λογαρίθμου με βάση $10$. Άρα $\log x = 2 \Leftrightarrow x = 10^2 \Leftrightarrow x = 100$.
\item Η εξίσωση $g(x) = 2f(x)$ είναι ισοδύναμη με την εξίσωση $\log(x + 2) = 2\log x$. \\
Η $\log(x + 2) = 2\log x$ ορίζεται για $x > 0$ και $x > -2$, άρα για $x > 0$. \\
Η εξίσωση $\log(x + 2) = 2\log x$ ισοδύναμα γράφεται $\log(x + 2) = \log x^2$. Συνεπώς, $x + 2 = x^2 \Leftrightarrow x^2 - x - 2 = 0$. Το τριώνυμο έχει λύσεις $-1$ και $2$. Μόνο η $x=2$ είναι δεκτή.
\end{rlist}
\end{alist}
