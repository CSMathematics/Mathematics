\begin{alist}
\item Το πολυώνυμο έχει παράγοντα το $x - 3$ αν και μόνο αν $P(3) = 0$. \\
Επίσης, αφού το υπόλοιπο της διαίρεσης $P(x) : (x + 1)$ είναι $v = -16$, τότε $P(-1) = -16$. \\
Είναι: 
\[\begin{cases} P(3) = 0 \\ P(-1) = -16 \end{cases} \Leftrightarrow \begin{cases} 9\alpha + \beta = 48 \\ \alpha + \beta = 8 \end{cases} \Leftrightarrow \begin{cases} 8\alpha = 40 \\ \alpha + \beta = 8 \end{cases} \Leftrightarrow \begin{cases} \alpha = 5 \\ \beta = 3 \end{cases}\]
Επομένως, το πολυώνυμο γίνεται: $P(x) = x^3 - 5x^2 + 7x - 3$.
\item Η εξίσωση: $P(x) = 0 \Leftrightarrow x^3 - 5x^2 + 7x - 3 = 0 \quad (1)$ \\
Εφαρμόζοντας το σχήμα Horner για $\rho = 3$, είναι:
\begin{center}
\Horner{1,-5,7,-3}{3}
\end{center}
Άρα $P(x) = (x - 3)(x^2 - 2x + 1) = (x - 3)(x - 1)^2$. \\
Η $(1)$ ισοδύναμα γράφεται: $P(x) = 0 \Leftrightarrow (x - 3)(x - 1)^2 = 0 \Leftrightarrow x = 3 \text{ ή } x = 1$.
\item Είναι
\begin{align*}
P(x) < 0 &\Leftrightarrow (x - 3)(x - 1)^2 < 0 \overset{(x - 1)^2 \ge 0}{\Longleftrightarrow} x - 3 < 0 \text{ και } x \ne 1 \Leftrightarrow \\
x &\in (-\infty, 1) \cup (1, 3).
\end{align*}
\item Έστω $P(\ln\kappa) < 0 \quad (2)$. \\
Καταρχάς πρέπει $\kappa > 0$. \\
Θέτοντας $\ln\kappa = \lambda, \lambda \in \mathbb{R}$ η ανίσωση $(2)$ γίνεται ισοδύναμα
\[P(\lambda) < 0 \overset{\gamma)}{\Longleftrightarrow} \begin{cases} \lambda < 3 \\ \lambda \ne 1 \end{cases} \Leftrightarrow \begin{cases} \ln\kappa < 3 \\ \ln\kappa \ne 1 \end{cases} \Leftrightarrow \begin{cases} \ln\kappa < \ln e^3 \\ \ln\kappa \ne \ln e \end{cases} \overset{\ln x \uparrow}{\Longleftrightarrow} \begin{cases} \kappa < e^3 \\ \kappa \ne e \end{cases}.\]
Επομένως, είναι: $\kappa \in (0, e) \cup (e, e^3)$.
\end{alist}
