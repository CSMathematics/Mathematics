\lysh{14459}{4}

\begin{alist}
\item Επειδή για κάθε $x \in \mathbb{R}$ έχουμε $x^2 + 1 > 0$, οπότε $\dfrac{1}{x^2 + 1} > 0$, δηλαδή $f(x) > 0$, η γραφική
παράσταση $C_f$ της $f$ είναι πάνω από τον άξονα $x'x$.

\item Αρκεί να αποδείξουμε ότι η εξίσωση $f(x) = \alpha$ έχει δυο λύσεις και να τις προσδιορίσουμε.
Είναι:
$f(x) = \alpha \stackrel{\alpha \ne 0}{\Leftrightarrow} \dfrac{1}{x^2 + 1} = \alpha \Leftrightarrow x^2 + 1 = \dfrac{1}{\alpha} \Leftrightarrow x^2 = \dfrac{1}{\alpha} - 1$, $(1)$
και επειδή $0 < \alpha < 1$, έχουμε $\dfrac{1}{\alpha} > 1$ οπότε $\dfrac{1}{\alpha} - 1 > 0$.

Επομένως η $(1)$ έχει δυο (αντίθετες) λύσεις τις $x_1 = \sqrt{\dfrac{1}{\alpha} - 1}$ και $x_2 = -\sqrt{\dfrac{1}{\alpha} - 1}$ που είναι και οι
τετμημένες των κοινών σημείων τους.

\item Αρκεί να αποδείξουμε ότι $\dfrac{|x|}{x^2 + 1} \le \dfrac{1}{2}$, ή αρκεί $2|x| \le x^2 + 1$,
που ισχύει, αφού από την προφανή ανισότητα $(|x| - 1)^2 \ge 0$ και την ισότητα $|x|^2 = x^2$ έπεται
ότι $x^2 + 1 \ge 2|x|$.
\end{alist}
