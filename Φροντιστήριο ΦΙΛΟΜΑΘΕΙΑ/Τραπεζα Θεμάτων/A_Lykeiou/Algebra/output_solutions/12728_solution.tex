\lysh{12728}{4}

\begin{alist}
\item Η ευθεία $y + \dfrac{1}{2}x = 4$ τέμνει τον $y'y$ άξονα στο σημείο $\Delta(0, 4)$ και τον $x'x$ άξονα στο σημείο $E(8, 0)$, αφού:
για $x = 0$, έχουμε: $y + \dfrac{1}{2} \cdot 0 = 4 \Leftrightarrow y = 4$ και
για $y = 0$, έχουμε: $0 + \dfrac{1}{2}x = 4 \Leftrightarrow x = 8$.

\item Το $\Gamma$ είναι σημείο του $\Delta E$ ευθύγραμμου τμήματος, άρα έχει τετμημένη που παίρνει τιμές μεταξύ των τιμών που έχουν οι τετμημένες των σημείων $\Delta(0, 4)$ και $E(8, 0)$. Δηλαδή:
$0 \le t \le 8$.
Οι συντεταγμένες του σημείου $\Gamma$ ικανοποιούν την εξίσωση $y + \dfrac{1}{2}x = 4$, οπότε:
\[y_\Gamma + \dfrac{1}{2}t = 4 \Leftrightarrow y_\Gamma = 4 - \dfrac{1}{2}t.\]

\item Το εμβαδόν του τραπεζίου δίνεται από τη σχέση:
$E = \dfrac{(O\Delta + B\Gamma)OB}{2}$,
όπου: $O\Delta = y_\Delta = 4$, $B\Gamma = 4 - \dfrac{1}{2}t$ και $OB = x_\Gamma = t$.
Οπότε:
$E(t) = \dfrac{\left(4 + 4 - \dfrac{1}{2}t\right)t}{2} = \dfrac{8t - \dfrac{1}{2}t^2}{2} = 4t - \dfrac{1}{4}t^2$, με $t \in [0, 8]$.

\item Το εμβαδόν γίνεται ίσο με $9,75$ όταν:
\begin{align*}
E(t) = 9,75 \ &\Leftrightarrow \\
4t - \dfrac{1}{4}t^2 = 9,75 \ &\Leftrightarrow \\
16t - t^2 = 39 \ &\Leftrightarrow \\
t^2 - 16t + 39 = 0 \ &\Leftrightarrow \\
t &= 3 \text{ ή } t = 13
\end{align*}

Επειδή $t \in [0, 8]$, δεκτή είναι η λύση $t = 3$. Συνεπώς το σημείο $\Gamma$ έχει συντεταγμένες:
\begin{gather*}
x_\Gamma = t = 3 \\
y_\Gamma = 4 - \dfrac{1}{2}t = 4 - \dfrac{1}{2} \cdot 3 = 4 - \dfrac{3}{2} = 2,5.
\end{gather*}
\end{alist}
