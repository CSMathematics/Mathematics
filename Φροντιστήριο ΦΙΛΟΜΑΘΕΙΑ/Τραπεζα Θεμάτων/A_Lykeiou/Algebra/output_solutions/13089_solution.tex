\lysh{13089}{4}

\begin{alist}
\item 
\begin{rlist}
\item Η ακολουθία $(\alpha_\nu)$ είναι αριθμητική πρόοδος αφού κάθε όρος της, πέραν του
$1$ου, προκύπτει από τον προηγούμενο με πρόσθεση του αριθμού $2$, με $\alpha_1 = 1$
και $\omega = 2$ οπότε ο γενικός της όρος είναι
\[\alpha_\nu = \alpha_1 + (\nu -1) \cdot \omega = 1 + (\nu -1) \cdot 2 = 1 + 2\nu - 2 = 2\nu - 1.\]
\item Η ακολουθία $(\beta_\nu)$ είναι αριθμητική πρόοδος αφού κάθε όρος της, πέραν του
$1$ου, προκύπτει από τον προηγούμενο με πρόσθεση του αριθμού $-2$, με $\beta_1 = 13$
και $\omega = -2$ οπότε ο γενικός της όρος είναι
\[\beta_\nu = \beta_1 + (\nu -1) \cdot \omega = 13 + (\nu -1) \cdot (-2) = 13 - 2\nu + 2 = 15 - 2\nu.\]
\end{rlist}

\item Έστω ότι διάβασε το βιβλίο σε $\nu$ μέρες. Τότε $\alpha_1 + \alpha_2 + \dots + \alpha_\nu = \beta_1 + \beta_2 + \dots + \beta_\nu$
δηλαδή:
\[\dfrac{[2 \cdot 1 + (\nu -1) \cdot 2] \cdot \nu}{2} = \dfrac{[2 \cdot 13 + (\nu -1) \cdot (-2)] \cdot \nu}{2}\]
και ισοδύναμα:
\[\dfrac{(2 + 2\nu - 2) \cdot \nu}{2} = \dfrac{(26 - 2\nu + 2) \cdot \nu}{2},\]
δηλαδή $\dfrac{2\nu \cdot \nu}{2} = \dfrac{(28 - 2\nu) \cdot \nu}{2}$.
Και αφού $\nu \ne 0$, θα είναι $2\nu = 28 - 2\nu$, δηλαδή $4\nu = 28$ και τελικά $\nu = 7$.

\item Προφανώς ανεξάρτητα από τον τρόπο που διάβασε το βιβλίο, το πλήθος των
σελίδων του βιβλίου είναι το $S_7 = \dfrac{[2 \cdot 1 + (7 -1) \cdot 2] \cdot 7}{2} = \dfrac{14 \cdot 7}{2} = 49$.

\item Για κάθε $\nu = 1, 2, \dots, 7$,
είναι $\beta_{8-\nu} = \beta_1 + (8 - \nu - 1) \cdot (-2) = 13 + (7 - \nu) \cdot (-2) = 13 - 14 + 2\nu = 2\nu - 1 = \alpha_\nu$.
\end{alist}
