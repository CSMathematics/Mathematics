\begin{alist}
\item Είναι:
\begin{align*}
f(\sqrt{3}) + f(-\sqrt{3}) &= (\sqrt{3} - 1)^2 + (-\sqrt{3} - 1)^2 = (\sqrt{3})^2 + 1 - 2\sqrt{3} + (\sqrt{3})^2 + 1 + 2\sqrt{3} =\\&= 3 + 1 - 2\sqrt{3} + 3 + 1 + 2\sqrt{3} = 4 + 4 = 8
\end{align*}

\item Ένα σημείο $M(x, f(x))$ της γραφικής παράστασης της $f$, βρίσκεται κάτω από την ευθεία $y = 4$ μόνο όταν ισχύει $f(x) < 4$.
Είναι:
\[f(x) < 4 \Leftrightarrow (x - 1)^2 < 4 \Leftrightarrow \sqrt{(x - 1)^2} < \sqrt{4} \Leftrightarrow |x - 1| < 2 \Leftrightarrow -2 < x - 1 < 2 \Leftrightarrow -1 < x < 3 \Leftrightarrow x \in (-1, 3)\]

Οι ακέραιοι που περιέχονται στο διάστημα $(-1, 3)$ είναι οι: $0, 1, 2$ και επειδή
$f(0) = 1, f(1) = 0, f(2) = 1$, τα ζητούμενα σημεία είναι τα $A(0, 1), B(1, 0), \Gamma(2, 1)$.

\item Αν $f(\alpha) = f(\beta)$, τότε:
\begin{gather*}
(\alpha - 1)^2 = (\beta - 1)^2\text{, οπότε} \\
(\alpha - 1)^2 - (\beta - 1)^2 = 0
\end{gather*}
απ’ όπου προκύπτει ότι:
\begin{gather*}
(\alpha - 1 + \beta - 1)(\alpha - 1 - \beta + 1) = 0\text{, δηλαδή} \\
(\alpha + \beta - 2)(\alpha - \beta) = 0
\end{gather*}

Η τελευταία ισότητα, με $\alpha \ne \beta$ δίνει:
$\alpha + \beta - 2 = 0$, οπότε $\alpha + \beta = 2$
που είναι το ζητούμενο.
\end{alist}
