\begin{alist}
\item Είναι $\alpha_{25} = \alpha_1 + 24\omega$ και $\alpha_{12} = \alpha_1 + 11\omega$.
Από τη σχέση της υπόθεσης $\alpha_{25} = \alpha_{12} + 39$ έχουμε:
\begin{align*}
\alpha_1 + 24\omega = \alpha_1 + 11\omega + 39 \ &\Leftrightarrow \\
24\omega - 11\omega = 39 \ &\Leftrightarrow \\
13\omega &= 39 \Leftrightarrow \omega = 3.
\end{align*}

\item Έστω $\alpha_\nu = 152$ ο ζητούμενος όρος. Ισχύει:
\begin{align*}
\alpha_\nu = 152 \Leftrightarrow \alpha_1 + (\nu - 1)\omega = 152 \ &\Leftrightarrow \\
2 + (\nu - 1)3 = 152 \ &\Leftrightarrow \\
3(\nu - 1) = 150 \ &\Leftrightarrow \\
\nu - 1 &= 50 \Leftrightarrow \nu = 51.
\end{align*}

Άρα ο $51^{\text{ος}}$ όρος της προόδου είναι ίσος με 152.
\end{alist}
