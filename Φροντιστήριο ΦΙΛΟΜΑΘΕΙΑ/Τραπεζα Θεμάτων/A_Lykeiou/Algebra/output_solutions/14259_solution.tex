\lysh{14259}{2}

\begin{alist}
\item Η αριθμητική πρόοδος έχει διαφορά $\omega = \alpha_2 - \alpha_1 = 7 - 2 = 5$.

\item Έχουμε $\alpha_{20} = \alpha_1 + 19\omega = 2 + 19 \cdot 5 = 97$.

\item Το άθροισμα $2 + 7 + 12 + \dots + 97$ είναι άθροισμα των $20$ πρώτων όρων της 
αριθμητικής προόδου με $\alpha_1 = 2$ και $\omega = 5$.
Άρα $S_{20} = \dfrac{20}{2} (\alpha_1 + \alpha_{20}) = 10 \cdot (2 + 97) = 990$.
\end{alist}
