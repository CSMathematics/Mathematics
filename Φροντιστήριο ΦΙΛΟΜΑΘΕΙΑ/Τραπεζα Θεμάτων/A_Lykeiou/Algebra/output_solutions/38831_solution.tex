\lysh{38831}{4}

\begin{alist}
\item
\begin{rlist}
\item Η πλευρά κάθε τετραγώνου είναι υποτείνουσα, έστω $u$, ενός ορθογωνίου τριγώνου το οποίο σχηματίζει με τα μισά των πλευρών του προηγούμενου, μεγαλύτερου, από αυτό. Άρα, αν το μεγαλύτερο τετράγωνο έχει πλευρά $4 \text{ cm}$, οι κάθετες πλευρές του ορθογωνίου τριγώνου θα είναι $\dfrac{4}{2} = 2 \text{ cm}$. Από το Πυθαγόρειο θεώρημα έχουμε ότι για την υποτείνουσα $u$ ισχύει:
$u^2 = 2^2 + 2^2 = 8$
οπότε $u = \sqrt{8} = 2\sqrt{2} \text{ cm}$.

\item Όπως και στο ερώτημα $\alpha)$i., η πλευρά $y$ του μικρού τετραγώνου είναι η υποτείνουσα ενός ορθογωνίου τριγώνου με κάθετες πλευρές $\dfrac{x}{2}$. Από το Πυθαγόρειο θεώρημα έχουμε ότι
\[y^2 = \left(\dfrac{x}{2}\right)^2 + \left(\dfrac{x}{2}\right)^2\]
ή
$y^2 = \dfrac{2x^2}{4}$
και επειδή $y > 0$, βρίσκουμε $y = \dfrac{x}{\sqrt{2}}$ ή $\dfrac{y}{x} = \dfrac{1}{\sqrt{2}}$.
\end{rlist}

\item Αν $\alpha_{\nu+1}$ και $\alpha_\nu$ οι πλευρές δύο διαδοχικών τετραγώνων, τότε από το προηγούμενο ερώτημα έχουμε ότι $\dfrac{\alpha_{\nu+1}}{\alpha_\nu} = \dfrac{1}{\sqrt{2}}$. Άρα ο λόγος των πλευρών δύο διαδοχικών τετραγώνων είναι σταθερός και ίσος με $\dfrac{1}{\sqrt{2}}$. Αυτό σημαίνει ότι οι πλευρές των τετραγώνων είναι όροι γεωμετρικής προόδου με λόγο $\lambda = \dfrac{1}{\sqrt{2}}$ και πρώτο όρο την πλευρά του αρχικού τετραγώνου, δηλαδή $\alpha_1 = \alpha$. Οπότε $\alpha_\nu = \alpha \left(\dfrac{1}{\sqrt{2}}\right)^{\nu-1}$.

\item Πρέπει
$\alpha_6 > 1$ ή $6 \left(\dfrac{1}{\sqrt{2}}\right)^5 > 1$ ή $\dfrac{6}{4\sqrt{2}} > 1$ ή $6 > 4\sqrt{2}$ ή
$6^2 > (4\sqrt{2})^2$ ή $36 > 32$
που ισχύει. Άρα, η επιλογή του είναι σωστή.
\end{alist}
