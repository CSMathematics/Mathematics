\lysh{37179}{2}

\begin{alist}
\item Ισχύουν:
\begin{align*}
K \ge \Lambda \ &\Leftrightarrow \\
K - \Lambda \ge 0 \ &\Leftrightarrow \\
2\alpha^2 + \beta^2 - 2\alpha\beta \ge 0 \ &\Leftrightarrow \\
\alpha^2 + \alpha^2 + \beta^2 - 2\alpha\beta \ge 0 \ &\Leftrightarrow \\
\alpha^2 + (\alpha - \beta)^2 &\ge 0,
\end{align*}

η οποία ισχύει για κάθε $\alpha, \beta \in \mathbb{R}$. Άρα $K \ge \Lambda$, για κάθε τιμή των $\alpha, \beta$.

\item Εκτελώντας τις πράξεις όπως στο $\alpha)$ ερώτημα έχουμε ότι:
\begin{align*}
K = \Lambda \ &\Leftrightarrow \\
\alpha^2 + (\alpha - \beta)^2 = 0 \ &\Leftrightarrow \\
\alpha^2 = 0 \text{ και } (\alpha - \beta)^2 = 0 \ &\Leftrightarrow \\
\alpha &= 0 \text{ και } \alpha = \beta.
\end{align*}

Οπότε $K = \Lambda$ αν και μόνο αν $\alpha = \beta = 0$.
\end{alist}
