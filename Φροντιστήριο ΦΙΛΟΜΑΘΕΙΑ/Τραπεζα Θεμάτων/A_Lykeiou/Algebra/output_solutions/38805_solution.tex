\lysh{38805}{4}

\begin{alist}
\item Η σχέση είναι της μορφής $R = \alpha T + \beta$.
Η αντίσταση είναι $12,56 \ \Omega$ σε θερμοκρασία $0^\circ\text{C}$, οπότε: $12,56 = \alpha \cdot 0 + \beta$, δηλαδή $\beta = 12,56$.
Άρα η εξίσωση γίνεται $R = \alpha T + 12,56$.

Όταν η θερμοκρασία αυξάνεται στους $27^\circ\text{C}$, η αντίσταση αυξάνεται στα $13,8 \ \Omega$, οπότε
$13,8 = \alpha \cdot 27 + 12,56$, δηλαδή $1,24 = 27\alpha$, οπότε $\alpha = 0,046$ (στρογγυλοποίηση στο χιλιοστό).
Άρα η ζητούμενη εξίσωση είναι $R = 0,046T + 12,56$.

\item Η κλίση της ευθείας $R = 0,046T + 12,56$ είναι $0,046$.
Η πρόταση που περιγράφει τι αντιπροσωπεύει η κλίση είναι η πρόταση \textbf{i}, διότι:
όταν η θερμοκρασία αυξηθεί κατά $1^\circ\text{C}$, η νέα αντίσταση του χάλκινου σύρματος θα είναι
$R^\prime = 0,046(T + 1) + 12,56 = (0,046T + 12,56) + 0,046 = R + 0,046$, δηλαδή η αντίσταση θα μεταβληθεί (για την ακρίβεια θα αυξηθεί) κατά $0,046 \ \Omega$.

\item
\begin{rlist}
\item Στην εξίσωση $R = 0,046T + 12,56$ για $R = 10,26 \ \Omega$, έχουμε:
\begin{gather*}
10,26 = 0,046T + 12,56\text{, ισοδύναμα} \\
-2,30 = 0,046T\text{και τελικά} \\
T = -50^\circ\text{C}.
\end{gather*}

\item Έχουμε
\begin{align*}
d(T, -50^\circ\text{C}) \le 50^\circ\text{C} \ &\Leftrightarrow \\
|T + 50| \le 50 \ &\Leftrightarrow \\
-50 \le T + 50 \le 50 \ &\Leftrightarrow \\
-100 &\le T \le 0\text{, οπότε}
\end{align*}

\begin{gather*}
-100 \cdot 0,046 + 12,56 \le 0,046T + 12,56 \le 0 \cdot 0,046 + 12,56\text{και τελικά} \\
7,96 \le R \le 12,56.
\end{gather*}

Άρα η αντίσταση του σύρματος θα είναι από $7,96 \ \Omega$ μέχρι $12,56 \ \Omega$.
\end{rlist}
\end{alist}
