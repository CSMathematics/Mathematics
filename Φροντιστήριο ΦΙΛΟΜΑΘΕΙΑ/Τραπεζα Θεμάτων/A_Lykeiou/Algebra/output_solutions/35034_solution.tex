\lysh{35034}{2}

\begin{alist}
\item Προβάλλουμε τη γραφική παράσταση της $f$ στον άξονα $x'x$ και βρίσκουμε ότι το πεδίο
ορισμού της είναι το διάστημα $[-3, 8,5]$.

\item Από τη γραφική παράσταση της $f$ συμπληρώνουμε τον πίνακα που ακολουθεί:

\begin{center}
\begin{tblr}{hlines = {1pt,white},
vlines = {0.5pt,white},
column{odd} = {maincolor!40,c},
column{even} = {maincolor!10,c},
column{1} = {maincolor,c,fg={white}}}
$x$ & $-3$ & $-1$ & $0$ & $3$ & $7,25$ & $8,5$ \\
$y$ & $0$ & $3$ & $3,75$ & $6$ & $-2$ & $-4$ \\
\end{tblr}
\end{center}

\item Η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ στα σημεία:
$A(-3, 0)$ και $B(6, 0)$
και τον άξονα $y'y$ στο σημείο:
$\Gamma(0, 3,75)$

\item Το διάστημα του πεδίου ορισμού στο οποίο η συνάρτηση παίρνει θετικές τιμές, δηλαδή
αυτό για το οποίο βρίσκεται "πάνω" από τον άξονα $x'x$, είναι το $(-3, 6)$.
\end{alist}
