\lysh{12909}{2}

\begin{alist}
\item $x - 3 < 5 \Leftrightarrow -5 < x - 3 < 5 \Leftrightarrow -2 < x < 8 \Leftrightarrow x \in (-2, 8)$.

\item Οι ακέραιες τιμές του $x$ για τις οποίες ισχύει $x - 3 < 5$ είναι οι ακέραιοι αριθμοί που
ανήκουν στο διάστημα $(-2, 8)$, δηλαδή οι : $-1, 0, 1, 2, 3, 4, 5, 6, 7$.

\item Είναι $A = \{-1,0,1,2,3,4,5,6,7\}$ και $B = \{-3, -2, -1,0,3,4\}$ οπότε:
$A \cup B = \{-3, -2, -1,0,1,2,3,4,5,6,7\}$ και $A \cap B = \{-1,0,3,4\}$.
\end{alist}
