\lysh{14681}{2}

\begin{alist}
\item Επειδή $3 > 0$, είναι: $f(3) = 2 \cdot 3 + 2 = 8$.
Αφού $-3 < 0$, είναι $f(-3) = (-3)^2 - 1 = 9 - 1 = 8$.

\item Για $x < 0$ είναι $f(x) = x^2 - 1$.
Άρα $f(x) = 8 \Leftrightarrow x^2 - 1 = 8 \Leftrightarrow x^2 = 9 \Leftrightarrow x = \pm 3$. Δεκτή τιμή $x = -3 < 0$.
Για $x \ge 0$ είναι $f(x) = 2x + 2$.
Άρα $f(x) = 8 \Leftrightarrow 2x + 2 = 8 \Leftrightarrow 2x = 6 \Leftrightarrow x = 3$. Δεκτή τιμή αφού $x = 3 \ge 0$.
Επομένως η $f(x) = 8$ ισχύει για $x = -3$ και $x = 3$.
\end{alist}
