\lysh{14490}{4}

\begin{alist}
\item Οι ενδείξεις ενός ζαριού είναι οι ακέραιες τιμές από το $1$ ως το $6$.
Άρα $\Omega = \{1, 2, 3, 4, 5, 6\}$.

\item 
\begin{rlist}
\item Η εξίσωση $x^2 - 2x + \lambda - 2 = 0$ ως δευτεροβάθμια δεν έχει πραγματικές ρίζες όταν $\Delta < 0$.
Επομένως,
\[\Delta = (-2)^2 - 4 \cdot 1 \cdot (\lambda - 2) = 4 - 4\lambda + 8 = 12 - 4\lambda.\]
Άρα $12 - 4\lambda < 0 \Leftrightarrow 12 < 4\lambda \Leftrightarrow \lambda > 3$.
Δηλαδή $\lambda \in (3, +\infty)$, επιπλέον $\lambda \in \Omega$, δηλαδή $\lambda \in \{1, 2, 3, 4, 5, 6\}$.
Προκύπτει $\lambda = 4$ ή $5$ ή $6$.
Άρα το ζητούμενο σύνολο $A = \{4, 5, 6\}$.

\item Έστω $x_1$ και $x_2$ οι ρίζες της εξίσωσης με $x_1 \cdot x_2 = 1$.
Σύμφωνα με τους τύπους Vieta $x_1 \cdot x_2 = \dfrac{\gamma}{\alpha} = \lambda - 2$.
Άρα $\lambda - 2 = 1 \Leftrightarrow \lambda = 3$.
\end{rlist}

\item Για $\lambda = 3$ η εξίσωση γίνεται $x^2 - 2x + 1 = 0 \Leftrightarrow (x - 1)^2 = 0 \Leftrightarrow x = 1$ διπλή ρίζα.
Άρα η ρίζα της εξίσωσης είναι η $x = 1$ διπλή.
\end{alist}
