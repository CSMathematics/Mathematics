\lysh{36895}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{gather*}
|2x + 4| = 10\text{, οπότε} \\
2x + 4 = 10 \text{ ή } 2x + 4 = -10\text{, δηλαδή} \\
2x = 6 \text{ ή } 2x = -14\text{και τελικά} \\
x = 3 \text{ ή } x = -7.
\end{gather*}

\item Έχουμε ισοδύναμα:
\begin{gather*}
|x - 5| > 1\text{, οπότε} \\
x - 5 < -1 \text{ ή } x - 5 > 1\text{και τελικά} \\
x < 4 \text{ ή } x > 6.
\end{gather*}

\item Οι λύσεις της εξίσωσης του $\alpha)$ ερωτήματος είναι και λύσεις της ανίσωσης του $\beta)$ ερωτήματος, διότι και οι δυο είναι αριθμοί μικρότεροι του $4$ (αφού $-7 < 4$ και $3 < 4$).
\end{alist}
