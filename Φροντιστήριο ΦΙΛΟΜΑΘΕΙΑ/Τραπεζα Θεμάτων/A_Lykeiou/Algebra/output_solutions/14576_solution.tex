\lysh{14576}{3}

\begin{alist}
\item Η συνάρτηση ορίζεται για $x \in \mathbb{R}$, για τους οποίους ισχύει:
\[x^2 - 9 \ne 0 \Leftrightarrow x^2 \ne 9 \Leftrightarrow x \ne 3 \text{ και } x \ne -3.\]
Κατά συνέπεια το πεδίο ορισμού της $f$ είναι το $A = (-\infty, -3) \cup (-3, 3) \cup (3, +\infty)$.

\item Έχουμε: $f(x) = \dfrac{x^3 - 9x}{x^2 - 9} = \dfrac{x(x^2 - 9)}{x^2 - 9} = x$, για κάθε $x \in A$.

\item Η συνάρτηση $f$ για $x > 0$ έχει τύπο $f(x) = x$ της οποίας η γραφική παράσταση είναι ευθεία
η οποία διέρχεται από την αρχή των αξόνων (όμως το $(0, 0)$ θα εξαιρεθεί από τη γραφική
παράσταση, αφού $x > 0$). Από την ευθεία θα εξαιρεθεί επίσης το σημείο $(3, 3)$, διότι $3 \notin A$.
\end{alist}
