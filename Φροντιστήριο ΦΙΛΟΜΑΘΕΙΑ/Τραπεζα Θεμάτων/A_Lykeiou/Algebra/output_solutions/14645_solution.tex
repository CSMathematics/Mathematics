\lysh{14645}{4}

\begin{alist}
\item
\begin{rlist}
\item Από το πυθαγόρειο θεώρημα προκύπτει ότι για τη διαγώνιο $\delta$ ενός
τετραγώνου πλευράς $x$ ισχύει:
\begin{gather*}
\delta^2 = x^2 + x^2\text{, δηλαδή} \\
\delta^2 = 2x^2\text{, άρα}
\end{gather*}
$\sqrt{\delta^2} = \sqrt{2x^2}$ και επειδή $\delta, x > 0$ προκύπτει ότι:
$\delta = \sqrt{2}x$.

\item Από το ερώτημα $\alpha)$i) προκύπτει ότι αν ένα από τα τετράγωνα της ακολουθίας
έχει πλευρά $x$ το επόμενό του έχει πλευρά $\sqrt{2}x$ και τα αντίστοιχα εμβαδά
είναι $x^2$ και $(\sqrt{2}x)^2 = 2x^2$. Άρα, ο λόγος $\lambda$ των εμβαδών δύο διαδοχικών
τετραγώνων δίνεται από τη σχέση
$\lambda = \dfrac{2x^2}{x^2} = 2$
και είναι σταθερός. Οπότε, τα εμβαδά των τετραγώνων είναι διαδοχικοί όροι
γεωμετρικής προόδου με λόγο $\lambda = 2$ και πρώτο όρο $\alpha_1 = x^2$.
Ο γενικός όρος της προόδου δίνεται από τη σχέση $\alpha_\nu = \alpha_1 \lambda^{\nu-1} = x^2 2^{\nu-1}$.
\end{rlist}

\item
\begin{rlist}
\item Ισχύει ότι
\begin{gather*}
\alpha_4 = 8\text{, άρα} \\
x^2 2^{4-1} = 8\text{, δηλαδή} \\
x^2 \cdot 8 = 8\text{, οπότε} \\
x^2 = 1
\end{gather*}
και επειδή $x > 0$ έχουμε ότι $x = 1$.

\item Το άθροισμα των $\nu$ πρώτων όρων της γεωμετρικής προόδου δίνεται από τη
σχέση:
\[S_\nu = \alpha_1 \dfrac{\lambda^\nu - 1}{\lambda - 1} = 1 \cdot \dfrac{2^\nu - 1}{2 - 1} = 2^\nu - 1.\]
Για να είναι το συνολικό εμβαδόν των αρχικών τετραγώνων ίσο με $255 \text{ τ.}\mu.$
πρέπει να ισχύει:
\begin{gather*}
S_\nu = 255\text{ή ισοδύναμα} \\
2^\nu - 1 = 255\text{, δηλαδή} \\
2^\nu = 256.
\end{gather*}
Αλλά $256 = 2^8$. Άρα, $2^\nu = 2^8$, οπότε $\nu = 8$. Άρα το πλήθος των τετραγώνων
που έχουν συνολικό εμβαδόν $255 \text{ τ.}\mu.$ είναι $8$.
\end{rlist}
\end{alist}
