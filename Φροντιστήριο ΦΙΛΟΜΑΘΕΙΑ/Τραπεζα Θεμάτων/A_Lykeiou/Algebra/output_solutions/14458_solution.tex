\lysh{14458}{2}

\begin{alist}
\item 
\begin{rlist}
\item Από τη δοσμένη ισότητα με πράξεις έχουμε:
$x^2 + 4xy + xy + 4y^2 - 9xy = 0$, οπότε 
$x^2 + 4y^2 - 4xy = 0$, δηλαδή $(x - 2y)^2 = 0$ ή $(2y - x)^2 = 0$ που είναι το ζητούμενο.
\item Είναι: $(2y - x)^2 = 0$, οπότε $2y - x = 0$, άρα $y = \dfrac{x}{2}$ που είναι το ζητούμενο.
\end{rlist}

\item Από το ερώτημα $(\alpha)$ έχουμε: $\dfrac{x}{2} = y$, $(1)$.
Με τη βοήθεια της $(1)$ έχουμε:
\[\left(2y - \dfrac{x}{2}\right)^2 + \left(2y + \dfrac{x}{2}\right)^2 = (2y - y)^2 + (2y + y)^2 = y^2 + (3y)^2 = y^2 + 9y^2 = 10y^2.\]
\end{alist}
