\lysh{13054}{2}

\begin{alist}
\item 
\begin{rlist}
\item Με $\alpha = 1$ οι ευθείες είναι οι:
$\varepsilon_1 : y = 7x - 4$ και $\varepsilon_2 : y = -x + 4$
\item Η ευθεία $\varepsilon_1$ έχει συντελεστή διεύθυνσης $\lambda_1 = 7 > 0$, οπότε σχηματίζει με τον άξονα $x'x$
οξεία γωνία, ενώ η $\varepsilon_2$ έχει συντελεστή διεύθυνσης $\lambda_2 = -1 < 0$, οπότε σχηματίζει με τον
άξονα $x'x$ αμβλεία γωνία.
\end{rlist}

\item Οι ευθείες είναι παράλληλες μόνο όταν έχουν τον ίδιο συντελεστή διεύθυνσης, δηλαδή
μόνο όταν ισχύει $3\alpha + 4 = 3 - 4\alpha$. Είναι:
\[3\alpha + 4 = 3 - 4\alpha \Leftrightarrow 7\alpha = -1 \Leftrightarrow \alpha = -\dfrac{1}{7}\]
Άρα οι ευθείες είναι παράλληλες μόνο όταν $\alpha = -\dfrac{1}{7}$.
\end{alist}
