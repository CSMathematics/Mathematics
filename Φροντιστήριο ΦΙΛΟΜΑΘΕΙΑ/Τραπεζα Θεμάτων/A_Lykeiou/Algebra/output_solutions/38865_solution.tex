\lysh{38865}{4}

\begin{alist}
\item Στην αρχή του δεύτερου έτους απασχόλησης ο Αντόνιο θα πάρει: $8000 + 450 = 8450\text{\euro}$. Στην αρχή του τρίτου χρόνου απασχόλησης θα πάρει: $8450 + 450 = 8900\text{\euro}$.
Αντίστοιχα, η Μπάρμπαρα θα πάρει:
$8000 + \dfrac{5}{100} \cdot 8000 = 8000 \cdot \left(1 + \dfrac{5}{100}\right) = 8000 \cdot \dfrac{105}{100} = 80 \cdot 105 = 8400\text{\euro}$ και στην αρχή του τρίτου χρόνου, $8400 + \dfrac{5}{100} \cdot 8400 = 8400 \cdot \dfrac{105}{100} = 84 \cdot 105 = 8820\text{\euro}$.

\item Τα χρήματα που παίρνει ο Αντόνιο στην αρχή κάθε έτους απασχόλησης είναι όροι αριθμητικής προόδου με πρώτο όρο $\alpha_1 = 8000$ και διαφορά $\omega = 450$. Άρα στην αρχή του $\nu$-οστού έτους θα πάρει $\alpha_\nu = 8000 + (\nu - 1) \cdot 450 = 7550 + 450\nu$.
Τα χρήματα που παίρνει η Μπάρμπαρα στην αρχή κάθε έτους απασχόλησης είναι όροι
γεωμετρικής προόδου με πρώτο όρο $\beta_1 = 8000$ και λόγο $\lambda = 1 + \dfrac{5}{100} = 1,05$. Άρα στην
αρχή του $\nu$-οστού έτους θα πάρει $\beta_\nu = 8000 \cdot 1,05^{\nu-1}$.

\item Μπορεί μετά τη συμπλήρωση του πρώτου κιόλας χρόνου ο Αντόνιο να παίρνει περισσότερα χρήματα από την Μπάρμπαρα (όπως βλέπουμε στο $\alpha$ ερώτημα), αλλά αυτό συμβαίνει μέχρι και τη συμπλήρωση του $6$ου χρόνου, αφού ο Αντόνιο θα παίρνει $\alpha_7 = 7550 + 450 \cdot 6 = 10250\text{\euro}$ και η Μπάρμπαρα $\beta_7 = 8000 \cdot 1,05^6 \simeq 8000 \cdot 1,28 = 10240\text{\euro}$.
Στην αρχή του έβδομου έτους ο Αντόνιο θα πάρει $\alpha_8 = 7550 + 450 \cdot 7 = 10700\text{\euro}$ και η
Μπάρμπαρα $\beta_8 = 8000 \cdot 1,05^7 \simeq 8000 \cdot 1,34 = 10720\text{\euro}$. Άρα έχει δίκιο η
Μπάρμπαρα.

\item Ο Αντόνιο θα εισπράξει συνολικά στα 15 χρόνια εργασίας:
\[S_{15} = \dfrac{15}{2} \cdot (2 \cdot 8000 + 14 \cdot 450) = 15 \cdot (8000 + 7 \cdot 450) = 15 \cdot (8000 + 3150) = 15 \cdot 11150 = 167250\text{\euro}\text{και}\]
η Μπάρμπαρα:
\[S_{15} = 8000 \cdot \dfrac{1,05^{15} - 1}{1,05 - 1} = 8000 \cdot \dfrac{2,08 - 1}{0,05} = 160000 \cdot (2,08 - 1) = 172800\text{\euro}.\]

Άρα η Μπάρμπαρα θα κερδίσει $172800 - 167250 = 5550\text{\euro}$ περισσότερα από τον Αντόνιο.
\end{alist}
