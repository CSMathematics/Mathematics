\lysh{35044}{2}

\begin{alist}
\item Είναι:
\begin{align*}
|y - 2| < 1 \Leftrightarrow -1 < y - 2 < 1 \ &\Leftrightarrow \\
-1 + 2 < y < 1 + 2 \ &\Leftrightarrow 1 &< y < 3 \Leftrightarrow y \in (1, 3).
\end{align*}

\item Ισχύει ότι $1 < y < 3$. Άρα:
\begin{align*}
y &> 1 \Leftrightarrow y - 1 > 0 \Rightarrow |y - 1| = y - 1\text{ και} \\
y &< 3 \Leftrightarrow y - 3 < 0 \Rightarrow |y - 3| = -(y - 3) = 3 - y.
\end{align*}

Τότε η παράσταση $K$ γράφεται:
\[K = \dfrac{|y - 1| + |y - 3|}{2} = \dfrac{y - 1 + 3 - y}{2} = \dfrac{2}{2} = 1.\]
\end{alist}
