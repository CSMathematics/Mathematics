\begin{alist}
\item Αντικαθιστούμε στην δοθείσα ισότητα $x = -5$ και βρίσκουμε:
\begin{gather*}
\lambda = (2 \cdot (-5) + 5)^2 - 8 \cdot (-5) = \\
= (-10 + 5)^2 + 40 = (-5)^2 + 40 = \\
= 25 + 40 = 65.
\end{gather*}

\item Αντικαθιστούμε στην δοθείσα ισότητα $\lambda = 20$ και έχουμε:
$20 = (2x + 5)^2 - 8x$
ή ισοδύναμα
$20 = 4x^2 + 20x + 25 - 8x$
και τελικά
$4x^2 + 12x + 5 = 0$.
Το τριώνυμο $4x^2 + 12x + 5$ έχει διακρίνουσα:
\[\Delta = 12^2 - 4 \cdot 4 \cdot 5 = 144 - 80 = 64 > 0\]
και ρίζες:
\[x_{1,2} = \dfrac{-12 \pm \sqrt{64}}{2 \cdot 4} = \dfrac{-12 \pm 8}{8} = \begin{cases} \dfrac{-12 + 8}{8} = -\dfrac{1}{2} \\ \dfrac{-12 - 8}{8} = -\dfrac{5}{2} \end{cases}\]

\item 
\begin{rlist}
\item Η σχέση (1) ισοδύναμα γράφεται:
\[\lambda = (2x + 5)^2 - 8x \Leftrightarrow \lambda = 4x^2 + 20x + 25 - 8x \Leftrightarrow\]
$\Leftrightarrow 4x^2 + 12x + (25 - \lambda) = 0$ (2)

\item Για να μπορεί ο εξαγόμενος αριθμός $\lambda$ να είναι $5$, με βάση τη σχέση (2) θα πρέπει να
υπάρχει $x$ τέτοιος ώστε:
\begin{align*}
4x^2 + 12x + (25 - 5) = 0 \ &\Leftrightarrow \\
\Leftrightarrow 4x^2 + 12x + 20 = 0 \ &\Leftrightarrow \\
\Leftrightarrow x^2 + 3x + 5 &= 0.
\end{align*}
Το τριώνυμο $x^2 + 3x + 5$ έχει διακρίνουσα $\Delta = 3^2 - 4 \cdot 1 \cdot 5 = 9 - 20 = -11 < 0$.
Άρα, η τελευταία εξίσωση είναι αδύνατη. Οπότε, για καμία τιμή του $x$ δεν μπορεί ο
εξαγόμενος αριθμός $\lambda$ να είναι $5$.

\item Οι δυνατές τιμές που μπορεί να έχει ο αριθμός $\lambda$, είναι αυτές για τις οποίες η
εξίσωση (2) έχει πραγματικές ρίζες. Αυτό ισχύει αν και μόνο αν $\Delta \ge 0$ όπου $\Delta$ η
διακρίνουσα του τριωνύμου $4x^2 + 12x + (25 - \lambda)$. Οπότε, ισοδύναμα έχουμε ότι:
\begin{align*}
\Delta \ge 0 \Leftrightarrow 12^2 - 4 \cdot 4 \cdot (25 - \lambda) \ge 0 \ &\Leftrightarrow \\
\Leftrightarrow 144 - 400 + 16\lambda \ge 0 \ &\Leftrightarrow \\
16\lambda &\ge 256 \Leftrightarrow \lambda \ge 16.
\end{align*}
\end{rlist}
\end{alist}
