\lysh{14713}{3}

\begin{alist}
\item Είναι:
\[\alpha^3 + 2\alpha^2 + 9\alpha + 18 = \alpha^2(\alpha + 2) + 9(\alpha + 2) = (\alpha + 2)(\alpha^2 + 9)\]
που είναι το ζητούμενο.

\item 
\begin{rlist}
\item Ισχύει: $\alpha^2 + 2\alpha = \alpha(\alpha + 2)$, οπότε με $\alpha > 0$ έχουμε:
\[A = \dfrac{\alpha^3 + 2\alpha^2 + 9\alpha + 18}{\alpha^2 + 2\alpha} = \dfrac{(\alpha^2 + 9)(\alpha + 2)}{\alpha(\alpha + 2)} = \dfrac{\alpha^2 + 9}{\alpha}.\]

\item Επειδή $\alpha > 0$ για την απόδειξη της $A \ge 6$, δηλαδή της $\dfrac{\alpha^2 + 9}{\alpha} \ge 6$, αρκεί να αποδείξουμε
ότι $\alpha^2 + 9 \ge 6\alpha$, ή αρκεί $\alpha^2 - 6\alpha + 9 \ge 0$, που ισχύει αφού προκύπτει από την προφανή
ανισότητα $(\alpha - 3)^2 \ge 0$.

Η ισότητα $A = 6$ ισχύει μόνο όταν $(\alpha - 3)^2 = 0$, δηλαδή μόνο όταν $\alpha = 3$.
\end{rlist}
\end{alist}
