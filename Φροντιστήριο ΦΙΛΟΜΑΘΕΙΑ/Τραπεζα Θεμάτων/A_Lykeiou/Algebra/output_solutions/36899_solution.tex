\lysh{36899}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{align*}
\alpha + \dfrac{4}{\alpha} &\ge 4 \overset{\alpha > 0}{\Leftrightarrow} 
\alpha^2 + \alpha \cdot \dfrac{4}{\alpha} \ge 4\alpha\text{, οπότε} \\
\alpha^2 + 4 &\ge 4\alpha\Leftrightarrow
\alpha^2 - 4\alpha + 4 \ge 0\text{ και τελικά }(\alpha - 2)^2 \ge 0
\end{align*}
που ισχύει.

\item Από το $\alpha)$ ερώτημα έχουμε ότι $\alpha + \dfrac{4}{\alpha} \ge 4$ και ομοίως $\beta + \dfrac{4}{\beta} \ge 4$.

Πολλαπλασιάζοντας κατά μέλη τις δυο ανισότητες προκύπτει:
\[\left( \alpha + \dfrac{4}{\alpha} \right) \left( \beta + \dfrac{4}{\beta} \right) \ge 16.\]
\end{alist}
