\lysh{14599}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
$|2x| < 2$, οπότε διαιρούμε τα μέλη της ανίσωσης με το $2$ και έχουμε
\begin{gather*}
|x| < 1\text{και τελικά} \\
-1 < x < 1.
\end{gather*}

\item Η ανίσωση
$x^2 < 1$ ισχύει, αν και μόνο αν ισχύει η
\begin{gather*}
\sqrt{x^2} < \sqrt{1}\text{, που ισχύει αν και μόνο αν ισχύει η} \\
|x| < 1\text{, που ισχύει αν και μόνο αν ισχύει}
\end{gather*}
$-1 < x < 1$, δηλαδή $x \in (-1, 1)$, που ισχύει.
Άρα για κάθε $x \in (-1, 1)$, ισχύει $x^2 < 1$.
\end{alist}
