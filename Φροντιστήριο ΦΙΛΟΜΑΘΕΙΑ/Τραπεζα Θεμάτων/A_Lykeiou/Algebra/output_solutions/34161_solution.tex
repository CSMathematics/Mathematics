\lysh{34161}{2}

\begin{alist}
\item Είναι:
\begin{align*}
|2x - 1| &= 3 \Leftrightarrow 2x - 1 = 3 \text{ ή } 2x - 1 = -3 \\
\Leftrightarrow 2x &= 4 \text{ ή } 2x = -2 \\
\Leftrightarrow x &= 2 \text{ ή } x = -1.
\end{align*}

\item Είναι $\alpha = -1$ και $\beta = 2$. Τότε η εξίσωση γράφεται:
$-x^2 + 2x + 3 = 0$.
Για $\alpha = -1, \beta = 2$ και $\gamma = 3$, βρίσκουμε:
\[\Delta = \beta^2 - 4\alpha\gamma = 2^2 - 4(-1)(3) = 4 + 12 = 16 > 0.\]

Οι ρίζες της εξίσωσης είναι:
\[x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{-2 \pm \sqrt{16}}{2(-1)} = \dfrac{-2 \pm 4}{-2} = \begin{cases} \dfrac{2}{-2} = -1 \\ \dfrac{-6}{-2} = 3 \end{cases}\]
\end{alist}
