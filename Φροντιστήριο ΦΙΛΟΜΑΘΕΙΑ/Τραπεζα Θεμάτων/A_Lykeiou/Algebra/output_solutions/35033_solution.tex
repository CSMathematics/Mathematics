\lysh{35033}{2}

\begin{alist}
\item Είναι $2 < x < 3$, οπότε $x > 2$, δηλαδή $2x > 4$, οπότε $2x - 4 > 0$.
Άρα:
$A = |2x - 4| = 2x - 4$.

\item Από το $\alpha)$ ερώτημα έχουμε: $A = 2x - 4$.
Επίσης επειδή $2 < x < 3$, είναι $x < 3$, δηλαδή $x - 3 < 0$. Άρα $B = |x - 3| = -(x - 3) = 3 - x$.
Επομένως:
$A + B = (2x - 4) + (3 - x) = x - 1$.

\item Είναι:
\[A + B = 2 \Leftrightarrow x - 1 = 2 \Leftrightarrow x = 3 \notin (2, 3),\]
Επομένως, δεν υπάρχει $x \in (2, 3)$ ώστε να ισχύει $A + B = 2$.
\end{alist}
