\lysh{38848}{4}

\begin{alist}
\item
\begin{rlist}
\item Το καθαριστικό περιέχει $\dfrac{20}{100} \cdot 500 = 100\text{ ml}$ αιθανόλης.
\item Εάν διπλασιάσουμε την ποσότητα αιθανόλης στο καθαριστικό δηλαδή έχουμε $2 \cdot 100 = 200\text{ ml}$ αιθανόλης, τότε θα έχουμε προσθέσει $100\text{ ml}$ αιθανόλης, οπότε θα αυξηθεί αντίστοιχα και η ποσότητα του καθαριστικού από $500$ σε $500 + 100 = 600\text{ ml}$.
Επομένως το ποσοστό της αιθανόλης στο νέο καθαριστικό θα είναι $\dfrac{200}{600} = \dfrac{1}{3}$, δηλαδή περίπου $33,3\%$ και όχι διπλάσιο, δηλαδή $40\%$.
\item Έστω ότι στο παραπάνω καθαριστικό πρέπει να προσθέσουμε $x\text{ ml}$ αιθανόλης. Τότε η αιθανόλη θα είναι $(100 + x)\text{ ml}$ και η ποσότητα του καθαριστικού θα γίνει $(500 + x)\text{ ml}$. Έτσι, αφού το καθαριστικό θα περιέχει αιθανόλη κατά $45\%$, θα ισχύει:
\[\dfrac{100+x}{500+x} = \dfrac{45}{100} \Leftrightarrow 100 \cdot (100 + x) = 45 \cdot (500 + x) \Leftrightarrow 10000 + 100x = 22500 + 45x \Leftrightarrow\]
$55x = 12500 \Leftrightarrow x \simeq 227,27$. Επομένως θα πρέπει να προσθέσουμε $227,27\text{ ml}$ αιθανόλης.
\item Έστω ότι στο παραπάνω καθαριστικό πρέπει να προσθέσουμε $y\text{ ml}$ αιθανόλης. Τότε, ομοίως με το προηγούμενο ερώτημα, θα έχουμε $\dfrac{100+y}{500+y} > \dfrac{30}{100} \Leftrightarrow 100 \cdot (100 + y) > 30 \cdot (500 + y) \Leftrightarrow 10000 + 100y > 15000 + 30y \Leftrightarrow 70y > 5000 \Leftrightarrow 7y > 500 \Leftrightarrow y > 71,43\text{ ml}$.
\end{rlist}

\item Η ποσότητα των αιθέριων ελαίων παραμένει σταθερή κατά την αραίωση του αρώματος.
Η ποσότητα αυτή στο αρχικό άρωμα ήταν $\dfrac{20}{100} \cdot 150 = 30\text{ ml}$. Αν η Σοφία αγοράσει $z\text{ ml}$ αιθανόλης, για να αραιώσει το άρωμα, τότε η ποσότητα του νέου αρώματος θα είναι $(150 + z)\text{ ml}$, η ποσότητα αιθέριων ελαίων θα παραμείνει $30\text{ ml}$, και θα ισχύει:
$\dfrac{30}{150+z} = \dfrac{15}{100} \Leftrightarrow 100 \cdot 30 = 15 \cdot (150 + z) \Leftrightarrow 3000 = 2250 + 15z \Leftrightarrow 15z = 750 \Leftrightarrow z = 50$. Επομένως η Σοφία θα πρέπει να αγοράσει $50\text{ ml}$ επιπλέον αιθανόλης.
\end{alist}
