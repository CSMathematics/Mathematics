\lysh{36675}{4}

\begin{alist}
\item Το τριώνυμο: $x^2 - 4x + 2 - \lambda^2$ έχει $\alpha = 1, \beta = -4, \gamma = 2 - \lambda^2$ και διακρίνουσα:
\[\Delta = \beta^2 - 4\alpha\gamma = (-4)^2 - 4 \cdot 1 \cdot (2 - \lambda^2) = 16 - 8 + 4\lambda^2 = 8 + 4\lambda^2\]

Είναι $\Delta = 8 + 4\lambda^2 > 0$ για κάθε $\lambda \in \mathbb{R}$ οπότε η εξίσωση $x^2 - 4x + 2 - \lambda^2 = 0$ έχει δύο ρίζες πραγματικές και άνισες για κάθε $\lambda \in \mathbb{R}$.

\item Από τους τύπους Vieta βρίσκουμε
\begin{rlist}
\item $S = x_1 + x_2 = -\dfrac{\beta}{\alpha} = -\dfrac{-4}{1} = 4$.
\item $P = x_1 \cdot x_2 = \dfrac{\gamma}{\alpha} = \dfrac{2 - \lambda^2}{1} = 2 - \lambda^2$.
\end{rlist}

\item Έστω $x_1 = 2 + \sqrt{3}$ και $x_2$ η ζητούμενη ρίζα. Τότε:
\begin{rlist}
\item Από το άθροισμα των ριζών έχουμε ότι:
\[x_1 + x_2 = 4 \Leftrightarrow 2 + \sqrt{3} + x_2 = 4 \Leftrightarrow x_2 = 4 - 2 - \sqrt{3} \Leftrightarrow x_2 = 2 - \sqrt{3}\]
\item Από το γινόμενο των ριζών έχουμε ότι:
\begin{align*}
x_1 \cdot x_2 = 2 - \lambda^2 \Leftrightarrow (2 + \sqrt{3})(2 - \sqrt{3}) = 2 - \lambda^2 \ &\Leftrightarrow \\
\Leftrightarrow 2^2 - (\sqrt{3})^2 = 2 - \lambda^2 \Leftrightarrow 4 - 3 = 2 - \lambda^2 \ &\Leftrightarrow \\
\Leftrightarrow \lambda^2 &= 1 \Leftrightarrow \lambda = \pm 1
\end{align*}
\end{rlist}
\end{alist}
