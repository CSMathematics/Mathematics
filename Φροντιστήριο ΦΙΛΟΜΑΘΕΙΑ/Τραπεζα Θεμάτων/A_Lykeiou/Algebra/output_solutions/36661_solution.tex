\lysh{36661}{4}

\begin{alist}
\item Η εξίσωση (1) είναι $2$ου βαθμού, αν και μόνο αν:
\[\lambda^2 - \lambda \ne 0 \Leftrightarrow \lambda(\lambda - 1) \ne 0 \Leftrightarrow \lambda \ne 0 \text{ και } \lambda \ne 1.\]

\item Με $\lambda \ne 0$ και $\lambda \ne 1$ έχουμε:
\begin{align*}
(\lambda^2 - \lambda)x^2 - (\lambda^2 - 1)x + \lambda - 1 = 0 \Leftrightarrow \lambda(\lambda - 1)x^2 - (\lambda - 1)(\lambda + 1)x + \lambda - 1 = 0 \ &\Leftrightarrow \\
\Leftrightarrow \lambda x^2 - (\lambda + 1)x + 1 &= 0
\end{align*}
που είναι το ζητούμενο.

\item Με $\lambda \ne 0$ και $\lambda \ne 1$ η εξίσωση $\lambda x^2 - (\lambda + 1)x + 1 = 0$ έχει διακρίνουσα:
\[\Delta = \beta^2 - 4\alpha\gamma = [-(\lambda + 1)]^2 - 4 \cdot \lambda \cdot 1 = \lambda^2 + 2\lambda + 1 - 4\lambda = (\lambda - 1)^2 > 0.\]
Επομένως η εξίσωση έχει ρίζες πραγματικές και άνισες.

\item Οι ρίζες της εξίσωσης (1) είναι:
\[x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{\lambda + 1 \pm \sqrt{(\lambda - 1)^2}}{2\lambda} = \dfrac{\lambda + 1 \pm (\lambda - 1)}{2\lambda} = \begin{cases} \dfrac{\lambda + 1 + \lambda - 1}{2\lambda} = 1 \\ \dfrac{\lambda + 1 - \lambda + 1}{2\lambda} = \dfrac{1}{\lambda} \end{cases}\]
\end{alist}
