\begin{alist}
\item
\begin{rlist}
\item Ισχύει $x = 2^3 - 3^2 = 8 - 9 = -1$, οπότε $A = |-1 - 2| + |-1 + 3| = |-3| + |2| = 3 + 2 = 5$.

\item Έχουμε:
\[A = 5 \Leftrightarrow |x - 2| + 3 = 5 \Leftrightarrow |x - 2| = 2\text{, οπότε}\]
$x - 2 = 2$ ή $x - 2 = -2$, και τελικά
$x = 4$ ή $x = 0$.
\end{rlist}

\item Υποθέτουμε ότι η παράσταση $A$ μπορεί να πάρει την τιμή $2$.
Τότε $A = 2 \Rightarrow |x - 2| + 3 = 2 \Rightarrow |x - 2| = -1$, άτοπο αφού ισχύει $|x - 2| \ge 0$. Άρα $A \ne 2$.
\end{alist}
