\lysh{36890}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{gather*}
|x - 2| = 3\text{, οπότε} \\
x - 2 = -3 \text{ ή } x - 2 = 3\text{και τελικά} \\
x = -1 \text{ και } x = 5.
\end{gather*}

\item Μια εξίσωση δευτέρου βαθμού με ρίζες $x_1 = -1$ και $x_2 = 5$ είναι η $x^2 - Sx + P = 0$, όπου
$S = x_1 + x_2 = -1 + 5 = 4$ και $P = x_1 x_2 = (-1) \cdot 5 = -5$. Άρα $x^2 - 4x - 5 = 0$.
\end{alist}
