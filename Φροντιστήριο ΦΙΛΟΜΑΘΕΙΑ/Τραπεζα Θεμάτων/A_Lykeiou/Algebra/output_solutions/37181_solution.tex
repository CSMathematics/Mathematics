\lysh{37181}{2}

\begin{alist}
\item Εφόσον η εξίσωση $x^2 - (\lambda - 1)x + 6 = 0$ $(1)$ έχει λύση το $1$, ισχύει ότι:
\begin{align*}
1^2 - (\lambda - 1) \cdot 1 + 6 = 0 \ &\Leftrightarrow \\
1 - \lambda + 1 + 6 = 0 \ &\Leftrightarrow \\
8 - \lambda = 0 &\Leftrightarrow \lambda = 8.
\end{align*}

\item Για $\lambda = 2$ η εξίσωση $(1)$ γράφεται:
\[x^2 - (2 - 1)x + 6 = 0 \Leftrightarrow x^2 - x + 6 = 0.\]

Η διακρίνουσα, με $\alpha = 1, \beta = -1, \gamma = 6$, γίνεται:
\[\Delta = \beta^2 - 4\alpha\gamma = (-1)^2 - 4 \cdot 1 \cdot 6 = 1 - 24 = -23 < 0\]

Άρα η εξίσωση δεν έχει πραγματικές ρίζες για $\lambda = 2$.
\end{alist}
