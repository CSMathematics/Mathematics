\lysh{37195}{2}

\begin{alist}
\item Πρέπει:
\[x - 4 \ge 0 \text{ και } 6 - x \ge 0 \Leftrightarrow x \ge 4 \text{ και } -x \ge -6 \Leftrightarrow x \ge 4 \text{ και } x \le 6 \Leftrightarrow 4 \le x \le 6 \Leftrightarrow x \in [4, 6].\]

\item Για $x = 5$ είναι:
\[A = \sqrt{5 - 4} + \sqrt{6 - 5} = \sqrt{1} + \sqrt{1} = 1 + 1 = 2\text{, επομένως}\]

\[A^2 + A - 6 = 2^2 + 2 - 6 = 4 + 2 - 6 = 0.\]
\end{alist}
