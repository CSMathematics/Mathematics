\lysh{34310}{4}

\begin{alist}
\item Το τριώνυμο $\lambda x^2 + (2\lambda - 1)x + \lambda - 1$ έχει $\alpha = \lambda, \beta = 2\lambda - 1, \gamma = \lambda - 1$ και
διακρίνουσα:
\begin{gather*}
\Delta = (2\lambda - 1)^2 - 4\lambda(\lambda - 1) = \\
= 4\lambda^2 - 4\lambda + 1 - 4\lambda^2 + 4\lambda = \\
= 1
\end{gather*}
Άρα, η διακρίνουσα $\Delta$ είναι σταθερή, ανεξάρτητη του $\lambda$.

\item Οι ρίζες της εξίσωσης είναι οι:
\[x_{1,2} = \dfrac{-(2\lambda - 1) \pm \sqrt{1}}{2\lambda} = \dfrac{1 - 2\lambda \pm 1}{2\lambda} = \begin{cases} \dfrac{2 - 2\lambda}{2\lambda} = \dfrac{1 - \lambda}{\lambda} \\ \dfrac{-2\lambda}{2\lambda} = -1 \end{cases}\]

\item Η απόσταση των ριζών $x_1, x_2$ στον άξονα των πραγματικών αριθμών είναι
$|x_2 - x_1| = \left| -1 - \dfrac{1 - \lambda}{\lambda} \right|$. Οπότε πρέπει:
\begin{gather*}
\left| -1 - \dfrac{1 - \lambda}{\lambda} \right| = 2\text{ή ισοδύναμα} \\
\left| \dfrac{-\lambda - 1 + \lambda}{\lambda} \right| = 2\text{δηλαδή} \\
\dfrac{1}{|\lambda|} = 2\text{που σημαίνει ότι}
\end{gather*}
$\dfrac{1}{\lambda} = 2$ ή $\dfrac{1}{\lambda} = -2$ και τελικά
$\lambda = \dfrac{1}{2}$ ή $\lambda = -\dfrac{1}{2}$.
\end{alist}
