\lysh{37169}{2}

\begin{alist}
\item Το τριώνυμο $-x^2 + (\sqrt{3} - 1)x + \sqrt{3}$ έχει συντελεστές $\alpha = -1, \beta = \sqrt{3} - 1, \gamma = \sqrt{3}$ και διακρίνουσα:
\begin{gather*}
\Delta = \beta^2 - 4\alpha\gamma = (\sqrt{3} - 1)^2 - 4(-1)\sqrt{3} = \sqrt{3}^2 - 2\sqrt{3} + 1 + 4\sqrt{3} \\
= \sqrt{3}^2 + 2\sqrt{3} + 1^2 = (\sqrt{3} + 1)^2 > 0.
\end{gather*}

\item Οι ρίζες του τριωνύμου είναι οι:
\begin{align*}
x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} &= \dfrac{-(\sqrt{3} - 1) \pm \sqrt{(\sqrt{3} + 1)^2}}{2(-1)} = \dfrac{1 - \sqrt{3} \pm (\sqrt{3} + 1)}{-2} =\\&= \begin{cases} \dfrac{1 - \sqrt{3} + \sqrt{3} + 1}{-2} = \dfrac{2}{-2} = -1 \\[2ex] \dfrac{1 - \sqrt{3} - \sqrt{3} - 1}{-2} = \dfrac{-2\sqrt{3}}{-2} = \sqrt{3} \end{cases}
\end{align*}

Επομένως η παραγοντοποίηση γίνεται:
\[-x^2 + (\sqrt{3} - 1)x + \sqrt{3} = -(x - (-1))(x - \sqrt{3}) = -(x + 1)(x - \sqrt{3}).\]
\end{alist}
