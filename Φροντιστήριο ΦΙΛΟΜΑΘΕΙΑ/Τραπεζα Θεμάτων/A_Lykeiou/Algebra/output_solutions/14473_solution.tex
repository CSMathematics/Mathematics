\lysh{14473}{2}

\begin{alist}
\item Έχουμε:
\begin{gather*}
\dfrac{4x + 5y}{x - 4y} = -2\text{, δηλαδή} \\
4x + 5y = -2(x - 4y)\text{, οπότε} \\
4x + 5y = -2x + 8y\text{, δηλαδή} \\
3y = 6x\text{και τελικά} \\
y = 2x.
\end{gather*}

\item Για $y = 2x$ η παράσταση $A$ γράφεται:
\[A = \dfrac{2x^2 + 3y^2 + xy}{xy} = \dfrac{2x^2 + 3(2x)^2 + x(2x)}{x(2x)} = \dfrac{2x^2 + 3(4x^2) + 2x^2}{2x^2} = \dfrac{2x^2 + 12x^2 + 2x^2}{2x^2} = \dfrac{16x^2}{2x^2} = 8.\]
\end{alist}
