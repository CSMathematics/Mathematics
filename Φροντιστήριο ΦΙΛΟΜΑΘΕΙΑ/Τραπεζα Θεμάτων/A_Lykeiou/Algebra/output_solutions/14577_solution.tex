\lysh{14577}{2}

\begin{alist}
\item Παρατηρούμε ότι $(-1)^2 - (-1) - 2 = 1 + 1 - 2 = 0$, οπότε ο αριθμός $-1$ είναι ρίζα της
εξίσωσης $(1)$.

\item Το γινόμενο των ριζών της εξίσωσης ισούται με $P = \dfrac{\gamma}{\alpha} = \dfrac{-2}{1} = -2$ και η μια ρίζα της είναι
$x_1 = -1$. Άρα για τη δεύτερη ρίζα της εξίσωσης έχουμε:
\[x_1 \cdot x_2 = -2 \Leftrightarrow -1 \cdot x_2 = -2 \Leftrightarrow x_2 = 2.\]

\item Από τα προηγούμενα ερωτήματα οι ρίζες του τριωνύμου $x^2 - x - 2$ είναι $x_1 = -1$ και
$x_2 = 2$. Οπότε παραγοντοποιείται ως εξής:
$x^2 - x - 2 = (x - 2)(x + 1)$.

Η παράσταση γίνεται:
\[A = \dfrac{x^2 - x - 2}{x^2 + x} = \dfrac{(x - 2)(x + 1)}{x(x + 1)} = \dfrac{x - 2}{x}.\]
\end{alist}
