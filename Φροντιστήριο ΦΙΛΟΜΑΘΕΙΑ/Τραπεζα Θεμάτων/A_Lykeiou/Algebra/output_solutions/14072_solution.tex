\lysh{14072}{2}

\begin{alist}
\item Η συνάρτηση $f$ ορίζεται για τους πραγματικούς αριθμούς $x$ για τους οποίους ισχύει:
\[x - 2 \ne 0 \Leftrightarrow x \ne 2.\]
Συνεπώς το πεδίο ορισμού της συνάρτησης είναι το $A = \mathbb{R} - \{2\}$.

\item Το σημείο $M(1, 3)$ ανήκει στη γραφική παράσταση της συνάρτησης $f$, γιατί
\[f(1) = \dfrac{1 - 4}{1 - 2} = \dfrac{-3}{-1} = 3.\]

\item Η γραφική παράσταση της $f$ τέμνει τον $y'y$ άξονα στο σημείο $(0, 2)$ και τον $x'x$ άξονα
στο σημείο $(-2, 0)$, αφού
για $x = 0$, έχουμε: $f(0) = \dfrac{0 - 4}{0 - 2} = \dfrac{-4}{-2} = 2$ και
για $y = f(x) = 0$, έχουμε: $0 = \dfrac{x^2 - 4}{x - 2}$, οπότε
$x^2 - 4 = 0$, δηλαδή $(x + 2)(x - 2) = 0$, οπότε $x + 2 = 0$ ή $x - 2 = 0$ και τελικά 
$x = -2$ (δεκτή) ή $x = 2$ (απορρίπτεται).
\end{alist}
