\lysh{14650}{4}

\begin{alist}
\item Είναι:
\[|x - 1| \le 3 \Leftrightarrow -3 \le x - 1 \le 3 \Leftrightarrow -2 \le x \le 4.\]

\item Το σύνολο λύσεων της ανίσωσης $(1)$ πάνω στον άξονα των πραγματικών αριθμών
απεικονίζεται ως εξής:
\begin{center}
\begin{tikzpicture}
\draw[->] (-4, 0) -- (6, 0);
\draw[thick] (-2, 0) -- (4, 0);
\fill (-2, 0) circle (2pt);
\fill (4, 0) circle (2pt);
\node[below] at (-2, -0.1) {$-2$};
\node[below] at (4, -0.1) {$4$};
\end{tikzpicture}
\end{center}
Στον άξονα των πραγματικών αριθμών βλέπουμε τους πραγματικούς αριθμούς $x$, οι οποίοι
απέχουν από το $1$ απόσταση μικρότερη ή ίση του $3$.

\item Οι ακέραιοι που ικανοποιούν την ανίσωση $(1)$ είναι οι: $-2, -1, 0, 1, 2, 3$ και $4$.

\item Θέτουμε $\omega = |x|$ και η ανίσωση γίνεται $|\omega - 1| \le 3$. Από το $\gamma)$ ερώτημα γνωρίζουμε ότι την
επαληθεύουν οι ακέραιοι $-2, -1, 0, 1, 2, 3$ και $4$. Δεδομένου ότι $\omega = |x| \ge 0$, δεκτοί είναι οι
ακέραιοι: $0, 1, 2, 3$ και $4$. Συνεπώς:
\begin{bulletlist}
\item $|x| = 0 \Leftrightarrow x = 0$
\item $|x| = 1 \Leftrightarrow x = \pm 1$
\item $|x| = 2 \Leftrightarrow x = \pm 2$
\item $|x| = 3 \Leftrightarrow x = \pm 3$
\item $|x| = 4 \Leftrightarrow x = \pm 4$
\end{bulletlist}
\end{alist}
