\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{gather*}
|x - 5| < 2\text{, οπότε} \\
-2 < x - 5 < 2\text{και τελικά} \\
3 < x < 7.
\end{gather*}

\item Έχουμε ισοδύναμα:
\begin{gather*}
|2 - 3x| > 5\text{, οπότε} \\
2 - 3x < -5 \text{ ή } 2 - 3x > 5\text{, δηλαδή} \\
3x > 7 \text{ ή } 3x < -3\text{και τελικά} \\
x > \dfrac{7}{3} \text{ ή } x < -1.
\end{gather*}

\item Παριστάνουμε τις λύσεις των παραπάνω δυο ανισώσεων στον ίδιο άξονα των πραγματικών αριθμών, όπως φαίνεται στο παρακάτω σχήμα:
Οι κοινές λύσεις των ανισώσεων είναι οι πραγματικοί αριθμοί $x \in (3, 7)$.
\end{alist}
