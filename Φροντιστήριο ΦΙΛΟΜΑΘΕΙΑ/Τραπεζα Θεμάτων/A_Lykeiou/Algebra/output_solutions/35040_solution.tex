\lysh{35040}{2}

\begin{alist}
\item Είναι:
\begin{gather*}
K - \Lambda = 2\alpha^2 + \beta^2 + 9 - 2\alpha(3 - \beta) = 2\alpha^2 + \beta^2 + 9 - 6\alpha + 2\alpha\beta = \\
= \alpha^2 + 2\alpha\beta + \beta^2 + \alpha^2 - 6\alpha + 9 = (\alpha^2 + 2\alpha\beta + \beta^2) + (\alpha^2 - 6\alpha + 9).
\end{gather*}

\item Ισοδύναμα και διαδοχικά ισχύει ότι:
\begin{align*}
K \ge \Lambda \Leftrightarrow K - \Lambda \ge 0 \Leftrightarrow (\alpha^2 + 2\alpha\beta + \beta^2) + (\alpha^2 - 6\alpha + 9) \ge 0 \ &\Leftrightarrow \\
\Leftrightarrow (\alpha + \beta)^2 + (\alpha - 3)^2 &\ge 0,
\end{align*}
το οποίο ισχύει για κάθε τιμή των $\alpha, \beta$.

\item Η ισότητα $K = \Lambda \Leftrightarrow K - \Lambda = 0$ ισχύει αν και μόνο αν:
\begin{align*}
(\alpha + \beta)^2 + (\alpha - 3)^2 = 0 \Leftrightarrow \alpha + \beta = 0 \text{ και } \alpha - 3 = 0 \ &\Leftrightarrow \\
\Leftrightarrow \alpha &= 3 \text{ και } 3 + \beta = 0 \Leftrightarrow \alpha = 3 \text{ και } \beta = -3.
\end{align*}
\end{alist}
