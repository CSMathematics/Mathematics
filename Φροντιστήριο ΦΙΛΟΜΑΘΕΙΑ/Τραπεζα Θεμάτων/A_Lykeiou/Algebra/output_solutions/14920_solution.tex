\lysh{14920}{2}

\begin{alist}
\item Έχουμε
$\dfrac{\alpha_3}{\alpha_1} = 4$, οπότε $\dfrac{\alpha_1 \cdot \lambda^2}{\alpha_1} = 4$, δηλαδή $\lambda^2 = 4$ και τελικά $\lambda = \pm 2$.
Επειδή $\lambda > 0$, $\lambda = 2$.

\item Έχουμε
\[\alpha_{10} = \alpha_1 \cdot \lambda^9 = 4 \cdot 2^9 = 4 \cdot 512 = 2048.\]

\item Έχουμε
\[S_{10} = \alpha_1 \cdot \dfrac{\lambda^{10} - 1}{\lambda - 1} = 4 \cdot \dfrac{2^{10} - 1}{2 - 1} = 4 \cdot 1023 = 4092.\]
\end{alist}
