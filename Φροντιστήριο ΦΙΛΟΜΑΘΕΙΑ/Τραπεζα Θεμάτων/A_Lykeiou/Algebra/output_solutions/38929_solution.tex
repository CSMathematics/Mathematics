\lysh{38929}{4}

\begin{alist}
\item Το αρχικό χαρτόνι έχει διαστάσεις $2 + 2x$ και $3 + 2x$, οπότε η επιφάνειά του δίνεται από τη σχέση $E(x) = (2 + 2x)(3 + 2x)$.

\item
\begin{rlist}
\item Η χωρητικότητα $V$ του κουτιού δίνεται από τη σχέση $V = 2 \cdot 3 \cdot x = 6x$. Πρέπει
$6x = 12$, οπότε $x = \dfrac{12}{6} = 2\text{ dm}$. Άρα, οι διαστάσεις του χαρτονιού πρέπει να είναι
$2 + 2 \cdot 2 = 6\text{ dm}$ και $3 + 2 \cdot 2 = 7\text{ dm}$.
\item Έχουμε ότι
$|x - 2| \le 0,01$
ή
$-0,01 \le x - 2 \le 0,01$
ή
$1,99 \le x \le 2,01$.
Οπότε, $6 \cdot 1,99 \le 6 \cdot x \le 6 \cdot 2,01$, δηλαδή $11,94 \le V \le 12,06$.
\end{rlist}

\item Επειδή το αρχικό χαρτόνι έχει επιφάνεια $E(x) = (2 + 2x)(3 + 2x)\text{ dm}^2$, το κόστος του
είναι $0,02(2 + 2x)(3 + 2x)\text{ \euro}$. Πρέπει
$0,02(2 + 2x)(3 + 2x) \le 2,2$
ή
$(2 + 2x)(3 + 2x) \le 110$
ή
$4x^2 + 10x - 104 \le 0$
ή
$2x^2 + 5x - 52 \le 0$.
Το τριώνυμο έχει διακρίνουσα $\Delta = 5^2 - 4 \cdot 2 \cdot (-52) = 25 + 416 = 441$ και ρίζες $x_{1,2} = \dfrac{-5 \pm \sqrt{441}}{2 \cdot 2} = \dfrac{-5 \pm 21}{4}$, οπότε $x_1 = \dfrac{-5 + 21}{4} = 4$ και $x_2 = \dfrac{-5 - 21}{4} = -6,5$. Επομένως, για να είναι αρνητικό,
πρέπει $-6,5 \le x \le 4$ και, επειδή $x > 0$, τελικά έχουμε ότι $0 < x \le 4$.
\end{alist}
