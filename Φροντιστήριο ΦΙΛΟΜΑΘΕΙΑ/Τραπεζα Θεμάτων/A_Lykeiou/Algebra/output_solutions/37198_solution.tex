\lysh{37198}{2}

\begin{alist}
\item Πρέπει:
\[(x - 2)^5 \ge 0 \Leftrightarrow x - 2 \ge 0 \Leftrightarrow x \ge 2 \Leftrightarrow x \in [2, +\infty).\]

\item Για $x = 4$, είναι:
\[B = \sqrt[5]{(4 - 2)^5} = \sqrt[5]{2^5} = 2.\]

Τότε:
\[B^2 + 6B = 2^2 + 6 \cdot 2 = 4 + 12 = 16 = 2^4 = B^4.\]
\end{alist}
