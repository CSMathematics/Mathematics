\lysh{37197}{2}

\begin{alist}
\item Πρέπει:
\[1 - x \ge 0 \text{ και } x^4 \ge 0 \Leftrightarrow -x \ge -1 \text{ και } x \in \mathbb{R} \Leftrightarrow x \le 1 \text{ και } x \in \mathbb{R} \Leftrightarrow x \le 1 \Leftrightarrow x \in (-\infty, 1].\]

\item Για $x = -3$, είναι:
\[A = \sqrt{1 - (-3)} - \sqrt[4]{(-3)^4} = \sqrt{1 + 3} - \sqrt[4]{3^4} = \sqrt{4} - 3 = 2 - 3 = -1.\]

Τότε:
\[A^3 + A^2 + A + 1 = (-1)^3 + (-1)^2 + (-1) + 1 = -1 + 1 - 1 + 1 = 0.\]
\end{alist}
