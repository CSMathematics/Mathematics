\lysh{13168}{4}

\begin{alist}
\item 
\begin{rlist}
\item Ισχύει ότι $f(0) = -1 \Leftrightarrow 0^2 + 2 \cdot \lambda \cdot 0 + \gamma = -1 \Leftrightarrow \gamma = -1$.
\item Για να μην είναι η γραφική παράσταση της $f$ κάτω από την ευθεία $y = -\lambda^2 - 1$ αρκεί:
\[f(x) \ge -\lambda^2 - 1 \Leftrightarrow x^2 + 2\lambda x - 1 \ge -\lambda^2 - 1 \Leftrightarrow x^2 + 2\lambda x + \lambda^2 \ge 0,\]
η οποία γράφεται $(x + \lambda)^2 \ge 0$ που ισχύει.
\end{rlist}

\item Οι τετμημένες των σημείων τομής της γραφικής παράστασης της $f$ με τον άξονα $x'x$ είναι οι λύσεις της εξίσωσης $f(x) = 0 \Leftrightarrow x^2 + 2\lambda x - 1 = 0$.
Η διακρίνουσα είναι:
\[\Delta = (2\lambda)^2 - 4(-1) = 4\lambda^2 + 4 = 4(\lambda^2 + 1) > 0,\]
άρα η εξίσωση έχει δύο άνισες πραγματικές ρίζες οπότε η γραφική παράσταση της $f$ τέμνει τον $x'x$ σε δύο σημεία.
Οι τετμημένες των σημείων είναι:
\[x_{1,2} = \dfrac{-2\lambda \pm \sqrt{4(\lambda^2 + 1)}}{2} = \dfrac{-2\lambda \pm 2\sqrt{\lambda^2 + 1}}{2} = -\lambda \pm \sqrt{\lambda^2 + 1} \Leftrightarrow \begin{cases} x_1 = -\lambda - \sqrt{\lambda^2 + 1} \\ x_2 = -\lambda + \sqrt{\lambda^2 + 1} \end{cases}.\]

Επίσης, $-\sqrt{\lambda^2 + 1} < \sqrt{\lambda^2 + 1}$ άρα $-\lambda - \sqrt{\lambda^2 + 1} < -\lambda + \sqrt{\lambda^2 + 1}$.

Οπότε τα σημεία είναι τα $A(-\lambda - \sqrt{\lambda^2 + 1}, 0)$ και $B(-\lambda + \sqrt{\lambda^2 + 1}, 0)$.

\item Η απόσταση των $A$ και $B$ είναι:
\[(AB) = |x_2 - x_1| = |-\lambda + \sqrt{\lambda^2 + 1} - (-\lambda - \sqrt{\lambda^2 + 1})| = |2\sqrt{\lambda^2 + 1}|.\]
Οπότε:
\begin{align*}
(AB) \ge 2 \Leftrightarrow |2\sqrt{\lambda^2 + 1}| \ge 2 \Leftrightarrow \sqrt{\lambda^2 + 1} \ge 1 \ &\Leftrightarrow \\
\lambda^2 + 1 &\ge 1 \Leftrightarrow \lambda^2 \ge 0
\end{align*}
που ισχύει.
\end{alist}
