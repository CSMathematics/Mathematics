\lysh{34148}{2}

\begin{alist}
\item 
\begin{rlist}
\item Είναι:
\[|2x - 3| \le 5 \Leftrightarrow -5 \le 2x - 3 \le 5 \Leftrightarrow -2 \le 2x \le 8 \Leftrightarrow -1 \le x \le 4.\]

\item Είναι:
\[|2x - 3| \ge 1 \Leftrightarrow 2x - 3 \le -1 \text{ ή } 2x - 3 \ge 1 \Leftrightarrow 2x \le 2 \text{ ή } 2x \ge 4 \Leftrightarrow x \le 1 \text{ ή } x \ge 2.\]
\end{rlist}

\item Επειδή η πρώτη ανίσωση αληθεύει για $-1 \le x \le 4$ και η δεύτερη για $x \le 1 \text{ ή } x \ge 2$, οι
ανισώσεις συναληθεύουν για κάθε πραγματικό αριθμό $x$ με $-1 \le x \le 1 \text{ ή } 2 \le x \le 4$,
δηλαδή οι ανισώσεις συναληθεύουν όταν $x \in [-1, 1] \cup [2, 4]$. Οι κοινές λύσεις φαίνονται
εποπτικά στο παρακάτω σχήμα.

(Το σχήμα παραλείπεται)
\end{alist}
