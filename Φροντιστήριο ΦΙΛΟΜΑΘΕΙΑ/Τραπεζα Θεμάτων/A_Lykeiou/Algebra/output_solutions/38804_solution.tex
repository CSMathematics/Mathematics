\lysh{38804}{4}

\begin{alist}
\item
\begin{rlist}
\item Παρατηρούμε ότι: 
το 1ο σχήμα αποτελείται από: $5^2 - 1^2 - 4 = 20$ τετραγωνάκια πλεγμένα,
το 2ο σχήμα αποτελείται από: $6^2 - 2^2 - 4 = 28$ τετραγωνάκια πλεγμένα,
το 3ο σχήμα αποτελείται από: $7^2 - 3^2 - 4 = 36$ τετραγωνάκια πλεγμένα.
Οπότε το 10ο σχήμα θα αποτελείται από $14^2 - 10^2 - 4 = 92$ τετραγωνάκια πλεγμένα.

\item Παρατηρούμε ότι:
στο 1ο σχήμα η «τρύπα» αντιστοιχεί σε: $1 = 1^2$ τετραγωνάκι,
στο 2ο σχήμα η «τρύπα» αντιστοιχεί σε: $4 = 2^2$ τετραγωνάκια,
στο 3ο σχήμα η «τρύπα» αντιστοιχεί σε: $9 = 3^2$ τετραγωνάκια.
Οπότε στο 10ο σχήμα η «τρύπα» αντιστοιχεί σε: $10^2 = 100$ τετραγωνάκια.
\end{rlist}

\item Παρατηρούμε ότι:
ο 1ος όρος του μοτίβου αποτελείται από: $(1 + 4)^2 - 1^2 - 4 = 20$ τετραγωνάκια,
ο 2ος όρος του μοτίβου αποτελείται από: $(2 + 4)^2 - 2^2 - 4 = 28$ τετραγωνάκια,
ο 3ος όρος του μοτίβου αποτελείται από: $(3 + 4)^2 - 3^2 - 4 = 36$ τετραγωνάκια.
Οπότε ο $\nu$-οστός όρος του μοτίβου θα αποτελείται από: $\alpha_\nu = (\nu + 4)^2 - \nu^2 - 4 = \nu^2 + 8\nu + 16 - \nu^2 - 4 = 8\nu + 12$ τετραγωνάκια.

Αυτός είναι ο γενικός όρος αριθμητικής προόδου διαφοράς $\omega = 8$, αφού
$\alpha_{\nu+1} - \alpha_\nu = 8(\nu + 1) + 12 - (8\nu + 12) = 8\nu + 8 + 12 - 8\nu - 12 = 8$, για οποιονδήποτε φυσικό αριθμό $\nu \ge 1$.

Εναλλακτικά, παρατηρούμε ότι το πρώτο σχήμα αποτελείται από $20$ πλεγμένα τετραγωνάκια, το δεύτερο από $28$, το τρίτο από $36$, δηλαδή τα τετραγωνάκια αυξάνονται κατά $8$. Άρα τα πλεγμένα τετραγωνάκια είναι όροι αριθμητικής προόδου με πρώτο όρο $\alpha_1 = 20$ και διαφορά $\omega = 8$. Οπότε το $\nu$-οστό σχήμα στη σειρά αποτελείται από $\alpha_\nu = 20 + (\nu - 1) \cdot 8 = 8\nu + 12$ πλεγμένα τετραγωνάκια.

\item
\begin{rlist}
\item Ζητάμε το πλήθος των $\nu$ πρώτων σχημάτων, ώστε $500 < S_\nu < 600$, δηλαδή
\begin{align*}
500 &< \dfrac{\nu}{2} \cdot [2 \cdot 20 + (\nu - 1) \cdot 8] < 600\text{, οπότε} \\
500 &< \nu[20 + (\nu - 1) \cdot 4] < 600\text{, ισοδύναμα} \\
500 < \nu[20 + 4\nu - 4] < 600 \ &\Leftrightarrow \\
500 &< 4\nu^2 + 16\nu < 600\text{, ισοδύναμα} \\
125 &< \nu^2 + 4\nu < 150,
\end{align*}
οπότε λύνοντας τις ανισώσεις προκύπτει ότι $\nu \in (9,4 \ ; \ 10,4)$. Άρα, επειδή ο $\nu$ είναι φυσικός αριθμός, θα πρέπει να φτιάξει $10$ σχήματα με τη σειρά που δίνονται.

\item Τα $10$ πρώτα σχήματα του μοτίβου αποτελούνται από
$S_{10} = \dfrac{10}{2} \cdot [2 \cdot 20 + (10 - 1) \cdot 8] = 5 \cdot (40 + 72) = 560$ πλεγμένα τετραγωνάκια.
Οπότε οι 7 δεκάδες αποτελούνται από $7 \cdot 560 = 3920$ τετραγωνάκια. Για να πλέξει ο Αντρέας το κάθε τετραγωνάκι, χρειάζεται $16\text{cm}$ μαλλί, άρα συνολικά $3920 \cdot 16 = 62720\text{cm} = 627,2\text{m}$ μαλλί, που είναι περίπου $5$ κουβάρια μαλλί ($627,20 \div 140 = 4,48$).
\end{rlist}
\end{alist}
