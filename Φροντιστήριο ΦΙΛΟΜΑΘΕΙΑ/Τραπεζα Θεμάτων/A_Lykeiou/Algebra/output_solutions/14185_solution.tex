\lysh{14185}{4}

\begin{alist}
\item Η συνάρτηση ορίζεται για τους πραγματικούς αριθμούς $x$ για τους οποίους ισχύει:
\[x^3 - 2x^2 + x \ne 0 \Leftrightarrow x(x^2 - 2x + 1) \ne 0 \Leftrightarrow x(x - 1)^2 \ne 0 \Leftrightarrow \begin{cases} x \ne 0 \\ x \ne 1 \end{cases}.\]
Άρα $A = \mathbb{R} - \{0, 1\}$.

\item Έχουμε $f(x) = \dfrac{x}{x^3 - 2x^2 + x} = \dfrac{x}{x(x - 1)^2} = \dfrac{1}{(x - 1)^2}$, για κάθε $x \in A$.

\item Για κάθε $x \in \mathbb{R} - \{0, 1\}$ η εξίσωση γίνεται:
\[f(x) = 1 \Leftrightarrow \dfrac{1}{(x - 1)^2} = 1 \Leftrightarrow (x - 1)^2 = 1 \Leftrightarrow \begin{cases} x - 1 = 1 \\ x - 1 = -1 \end{cases} \Leftrightarrow \begin{cases} x = 2 \\ x = 0 \text{ (απορ.)} \end{cases}\]
Άρα έχει μοναδική λύση την $x = 2$.

\item Αναζητούμε τις τιμές του $x \in A$ για τις οποίες ισχύει:
\[f(x) > 1 \Leftrightarrow \dfrac{1}{(x - 1)^2} > 1 \Leftrightarrow (x - 1)^2 < 1 \Leftrightarrow |x - 1| < 1 \Leftrightarrow -1 < x - 1 < 1 \Leftrightarrow 0 < x < 2\]
και επειδή $x \ne 1$, έχουμε τελικά ότι $x \in (0, 1) \cup (1, 2)$.
\end{alist}
