\lysh{32741}{4}

\begin{alist}
\item Στην πρώτη περίπτωση το πλήθος των μαθητών είναι $x(x - 1)$ ενώ στη δεύτερη
$(x + 3)(x - 3) - 1$. Άρα πρέπει:
$x(x - 1) = (x + 3)(x - 3) - 1$
ή ισοδύναμα
$x^2 - x = x^2 - 3^2 - 1$
οπότε
\[-x = -9 - 1 \Leftrightarrow -x = -10 \Leftrightarrow x = 10.\]

\item Θέτουμε στον τύπο $x(x - 1)$ που δίνει το πλήθος των μαθητών όπου $x = 10$ και
βρίσκουμε:
$10(10 - 1) = 10 \cdot 9 = 90$.
Άρα, οι μαθητές της Α τάξης είναι 90.

\item Το πλήθος των μαθητών στις $\nu$ ομάδες εργασίας είναι όροι αριθμητικής προόδου με πρώτο
όρο $\alpha_1 = 2$, διαφορά $\omega = 2$ και άθροισμα $S_\nu = 90$. Οπότε, από τον τύπο
\[S_\nu = \dfrac{\nu}{2}[2\alpha_1 + (\nu - 1)\omega]\]
έχουμε ότι:
\begin{align*}
90 = \dfrac{\nu}{2}[2 \cdot 2 + (\nu - 1) \cdot 2] \ &\Leftrightarrow \\
90 = \dfrac{\nu}{2}(4 + 2\nu - 2) \ &\Leftrightarrow \\
90 = 2\nu + \nu^2 - \nu \ &\Leftrightarrow \\
\nu^2 + \nu - 90 &= 0.
\end{align*}

Η τελευταία εξίσωση είναι δευτέρου βαθμού ως προς $\nu$ με $\alpha = 1, \beta = 1$, και $\gamma = -90$. Η
διακρίνουσα είναι
\[\Delta = \beta^2 - 4\alpha\gamma = 1^2 - 4 \cdot 1 \cdot (-90) = 1 + 360 = 361 > 0\]
και έχει ρίζες τις:
\[\nu_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{-1 \pm \sqrt{361}}{2 \cdot 1} = \dfrac{-1 \pm 19}{2} = \begin{cases} 9 \\ -10 \end{cases}.\]

Η τιμή $\nu = -10$ απορρίπτεται διότι $\nu \in \mathbb{N}$. Άρα θα δημιουργηθούν $\nu = 9$ ομάδες εργασίας.
\end{alist}
