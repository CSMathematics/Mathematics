\lysh{33893}{4}

\begin{alist}
\item Έχουμε ισοδύναμα:
$|x - 4| < 2$, δηλαδή

\begin{gather*}
-2 < x - 4 < 2\text{, οπότε} \\
4 - 2 < x < 2 + 4\text{και τελικά} \\
2 < x < 6.
\end{gather*}
Άρα $x \in (2, 6)$.

\item Πάνω στον άξονα των πραγματικών αριθμών η απόσταση του $x$ από το $4$ είναι μικρότερη
από $2$, δηλαδή

\[d(x, 4) < 2 \Leftrightarrow |x - 4| < 2.\]

Άρα $2 < x < 6$.
\begin{rlist}
\item Έχουμε $2 < x < 6$, οπότε
\begin{gather*}
2 \cdot 3 < 3x < 6 \cdot 3\text{, δηλαδή} \\
6 < 3x < 18\text{, οπότε} \\
6 - 4 < 3x - 4 < 18 - 4\text{και τελικά} \\
2 < 3x - 4 < 14
\end{gather*}
και συνεπώς $3x - 4 > 0$.

\item Θα δείξουμε ότι $2 < d(3x, 4) < 14$.

Ισχύει ότι $d(3x, 4) = |3x - 4| \stackrel{(i)}{=} 3x - 4$. Από το $\beta)\text{i.}$ ερώτημα $2 < 3x - 4 < 14$, οπότε
$2 < d(3x, 4) < 14$.

\item Η απόσταση του $3x$ από το $19$ συμβολίζεται $d(3x, 19) = |3x - 19|$.

Από το $\beta)\text{i.}$ ερώτημα έχουμε $2 < 3x - 4 < 14$ οπότε αφαιρούμε από τα μέλη της ανίσωσης
$15$ και έχουμε:
$-13 < 3x - 19 < -1$, δηλαδή $3x - 19 < 0$. Οπότε $d(3x, 19) = |3x - 19| = -3x + 19$.

Έχουμε $-13 < 3x - 19 < -1$, δηλαδή
$13 > -3x + 19 > 1$ οπότε $1 < d(3x, 19) < 13$.
\end{rlist}
\end{alist}
