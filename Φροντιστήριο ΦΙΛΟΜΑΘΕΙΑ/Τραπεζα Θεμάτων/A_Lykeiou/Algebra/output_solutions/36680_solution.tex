\begin{alist}
\item Αφού $f(2) = g(2)$ έχουμε ισοδύναμα
\begin{align*}
f(2) = g(2) \ &\Leftrightarrow \\
2^2 - 4 \cdot 2 + \alpha = \alpha \cdot 2 - 5 \ &\Leftrightarrow \\
4 - 8 + \alpha = 2\alpha - 5 \ &\Leftrightarrow \\
\alpha - 2\alpha = -4 + 8 - 5 \ &\Leftrightarrow \\
-\alpha = -1 \ &\Leftrightarrow \\
\alpha &= 1
\end{align*}

\item Για $\alpha = 1$ έχουμε $f(x) = x^2 - 4x + 1$ και $g(x) = x - 5$.
\begin{rlist}
\item Η εξίσωση: $f(x) = g(x)$ γίνεται ισοδύναμα
\[x^2 - 4x + 1 = x - 5 \Leftrightarrow x^2 - 4x + 1 - x + 5 = 0 \Leftrightarrow x^2 - 5x + 6 = 0\]
Το τριώνυμο $x^2 - 5x + 6$ έχει διακρίνουσα:
\[\Delta = (-5)^2 - 4 \cdot 1 \cdot 6 = 25 - 24 = 1 > 0\]
και ρίζες τις:
\[x_{1,2} = \dfrac{-(-5) \pm \sqrt{1}}{2 \cdot 1} = \dfrac{5 \pm 1}{2} = \begin{cases} \dfrac{5 + 1}{2} = 3 \\ \dfrac{5 - 1}{2} = 2 \end{cases}\]

\item Η ανίσωση: $f(x) \ge g(x)$ γίνεται ισοδύναμα
\[x^2 - 4x + 1 \ge x - 5 \Leftrightarrow x^2 - 4x + 1 - x + 5 \ge 0 \Leftrightarrow x^2 - 5x + 6 \ge 0\]
Το τριώνυμο $x^2 - 5x + 6$ έχει ρίζες τις $x_1 = 3$ και $x_2 = 2$ και το πρόσημό του φαίνεται στον παρακάτω πίνακα.

\begin{center}
\begin{tabular}{|c|ccccccc|}

$x$ & $-\infty$ & & $2$ & & $3$ & & $+\infty$ \\ 
$x^2 - 5x + 6$ & & $+$ & $0$ & $-$ & $0$ & $+$ & \\ 
\end{tabular}
\end{center}

Από τον παραπάνω πίνακα προσήμων συμπεραίνουμε ότι:
\[x^2 - 5x + 6 \ge 0 \Leftrightarrow x \in (-\infty, 2] \cup [3, +\infty)\]

Από την ιδιότητα $|\alpha| = \alpha \Leftrightarrow \alpha \ge 0$ έχουμε ότι:
\begin{align*}
|f(x) - g(x)| = f(x) - g(x) \ &\Leftrightarrow \\
f(x) - g(x) \ge 0 \ &\Leftrightarrow \\
&x \in (-\infty, 2] \cup [3, +\infty)
\end{align*}
\end{rlist}
\end{alist}
