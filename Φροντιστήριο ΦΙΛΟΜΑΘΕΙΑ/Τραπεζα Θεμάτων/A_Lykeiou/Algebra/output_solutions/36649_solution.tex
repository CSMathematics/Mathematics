\lysh{36649}{4}

\begin{alist}
\item Το πλήθος των βακτηρίων, στο τέλος κάθε ώρας, είναι διαδοχικοί όροι γεωμετρικής προόδου $(\alpha_\nu)$ με πρώτο όρο $\alpha_1 = 102400$ και λόγο $\lambda = \dfrac{1}{2}$.
Ο $\nu$-οστός όρος της προόδου δίνεται από τον τύπο $\alpha_\nu = \alpha_1\lambda^{\nu-1}$ και είναι:
\[\alpha_\nu = 102400 \cdot \left(\dfrac{1}{2}\right)^{\nu-1}\]
Μετά από $6$ ώρες ο αριθμός των βακτηρίων θα είναι:
\[\alpha_6 = 102400 \cdot \left(\dfrac{1}{2}\right)^5 = 102400 \cdot \dfrac{1}{32} = 3200\]

\item
\begin{rlist}
\item Μετά την ξαφνική επιδείνωση του οργανισμού ο αριθμός των βακτηρίων άρχισε να αυξάνεται ώστε κάθε μια ώρα να τριπλασιάζεται. Άρα η ακολουθία $(\beta_\nu)$ είναι γεωμετρική πρόοδος με λόγο $\lambda = 3$ και πρώτο όρο $\beta_1 = 3200 \cdot 3 = 9600$.

\item Είναι:
\[\beta_\nu = \beta_1\lambda^{\nu-1} = 9600 \cdot 3^{\nu-1}, \nu \le 5\]

\item Ο αριθμός των βακτηρίων που θα υπάρχουν στον οργανισμό $3$ ώρες μετά από την στιγμή της επιδείνωσης θα είναι:
\[\beta_3 = 9600 \cdot 3^{3-1} = 9600 \cdot 3^2 = 86400.\]
\end{rlist}
\end{alist}
