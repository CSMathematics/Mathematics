\lysh{14925}{4}

\begin{alist}
\item Τα σημεία $A, B$ είναι τα σημεία τομής της ευθείας $y = x$ με τη γραφική παράσταση της
$f(x) = \dfrac{1}{x}$, οπότε οι τετμημένες των $A, B$ είναι οι λύσεις της εξίσωσης
\[\dfrac{1}{x} = x \Leftrightarrow x^2 = 1 \Leftrightarrow x = 1 \text{ ή } x = -1.\]
Όμως από το σχήμα βλέπουμε ότι το σημείο $B$ είναι στο $3^\text{o}$ τεταρτημόριο οπότε έχει
αρνητική τετμημένη και το σημείο $A$ στο $1^\text{o}$ οπότε έχει θετική τετμημένη. Συνεπώς
$A(1, f(1))$ δηλαδή $A(1, 1)$ και $B(-1, f(-1))$ δηλαδή $B(-1, -1)$.

Παρατηρούμε ότι τα σημεία $A, B$ έχουν αντίθετες συντεταγμένες οπότε είναι συμμετρικά
ως προς το σημείο $O(0, 0)$, δηλαδή το $O(0, 0)$ είναι το μέσο του $AB$.

\item Αφού $M(x, y)$ τυχαίο σημείο της γραφικής παράστασης της $f$, είναι $x \ne 0$ και $f(x) = y$
δηλαδή $y = \dfrac{1}{x}$ (1).

Το συμμετρικό του $M$ ως προς το $O(0, 0)$ είναι το $M'(-x, -y)$ και για να ανήκει στη γραφική
παράσταση της $f$, πρέπει και αρκεί $f(-x) = -y$ ισοδύναμα $-y = -\dfrac{1}{x}$ ισοδύναμα $y = \dfrac{1}{x}$
που ισχύει από την (1).

\item Ισοδύναμα έχουμε
\begin{align*}
(AB) \le (MM') \ &\Leftrightarrow \\
\sqrt{(1 + 1)^2 + (1 + 1)^2} \le \sqrt{(x + x)^2 + \left(\dfrac{1}{x} + \dfrac{1}{x}\right)^2} \ &\Leftrightarrow \\
8 \le 4x^2 + \dfrac{4}{x^2} \ &\Leftrightarrow \\
2 \le x^2 + \dfrac{1}{x^2} \ &\Leftrightarrow \\
0 \le x^2 + \dfrac{1}{x^2} - 2 \ &\Leftrightarrow \\
0 &\le \left(x - \dfrac{1}{x}\right)^2
\end{align*}
που ισχύει για κάθε $x \ne 0$.

Τέλος, $(AB) = (MM') \Leftrightarrow 0 = \left(x - \dfrac{1}{x}\right)^2 \Leftrightarrow x = \dfrac{1}{x} \Leftrightarrow x^2 = 1 \Leftrightarrow x = \pm 1$
που σημαίνει ότι ισχύει όταν τα σημεία $M, M'$ ταυτίζονται με τα σημεία $A, B$.
\end{alist}
