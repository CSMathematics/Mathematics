\lysh{14578}{3}

\begin{alist}
\item Η παράσταση ορίζεται για τους πραγματικούς αριθμούς $x$ για τους οποίους ισχύει:
\begin{gather*}
x^2 - x \ne 0 \text{ και } 1 - x \ne 0\text{, δηλαδή} \\
x(x - 1) \ne 0 \text{ και } x \ne 1\text{, τελικά} \\
x \ne 0 \text{ και } x \ne 1.
\end{gather*}

\item Έχουμε ισοδύναμα:
\begin{align*}
\dfrac{2x^2 - 1}{x^2 - x} + \dfrac{1}{1 - x} = 0 \Leftrightarrow \dfrac{2x^2 - 1}{x(x - 1)} - \dfrac{1}{x - 1} = 0 \ &\Leftrightarrow \\
2x^2 - 1 - x &= 0 \Leftrightarrow 2x^2 - x - 1 = 0.
\end{align*}

Η εξίσωση $2x^2 - x - 1 = 0$ έχει διακρίνουσα $\Delta = (-1)^2 - 4 \cdot 2 \cdot (-1) = 9 > 0$ και ρίζες τις
\[x_1 = \dfrac{-(-1) - \sqrt{9}}{2 \cdot 2} = -\dfrac{1}{2}\text{και}\]
$x_2 = \dfrac{-(-1) + \sqrt{9}}{2 \cdot 2} = 1$, που δεν είναι δεκτή, αφού $x \ne 1$.
Άρα η εξίσωση έχει μία λύση την $x = -\dfrac{1}{2}$.
\end{alist}
