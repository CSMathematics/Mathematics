\lysh{38832}{4}

\begin{alist}
\item Από τις $\beta\text{ Wh}$ που παράγονται, μόνο το $80\%$ φτάνει στις συσκευές, άρα $\beta \cdot 0,8 = 0,8\beta\text{ Wh}$.

\item
\begin{rlist}
\item Στις συσκευές φτάνουν $0,8\beta\text{ Wh}$, οπότε η συνολική ενέργεια είναι $0,8\beta\text{ Wh}$ επί τον αριθμό των εικοσαλέπτων που ποδηλατούν οι επιβάτες. Άρα η συνάρτηση που εκφράζει την ενέργεια $E$ (σε $\text{Wh}$) που φτάνει στις συσκευές σε σχέση με το πλήθος $n$ των εικοσαλέπτων είναι $E = (0,8\beta)n$.
\item Για κάθε συσκευή απαιτείται ενέργεια $10\text{Wh}$. Άρα για $D$ συσκευές απαιτείται ενέργεια ίση με $10D$, οπότε $E = 10D$ και $D = \dfrac{E}{10} = \dfrac{1}{10}E$, που είναι της μορφής $D = \alpha E$ με $\alpha = \dfrac{1}{10}$.
\end{rlist}

\item
\begin{rlist}
\item Από το $\beta)$ ερώτημα έχουμε ότι $D = \dfrac{E}{10} = \dfrac{0,8\beta n}{10} = (0,08\beta)n$.
Για $D = 780000$ και για $n = 195000$, έχουμε $780000 = (0,08\beta) \cdot 195000$ και τελικά $\beta = 50\text{Wh}$. Άρα κάθε ποδήλατο για $20$ λεπτά συνεχούς χρήσης παρήγε $50\text{Wh}$.
\item Φορτίζονται $D = 780000$ συσκευές, που χρειάζονται $10\text{Wh}$ η καθεμία για να φορτίσουν πλήρως, δηλαδή συνολικά $7800000\text{ Wh} = 7800\text{ kWh}$. Οπότε το αεροδρόμιο εξοικονομεί $7800 \cdot 0,2 = 1560$ ευρώ κάθε μήνα.
\end{rlist}
\end{alist}
