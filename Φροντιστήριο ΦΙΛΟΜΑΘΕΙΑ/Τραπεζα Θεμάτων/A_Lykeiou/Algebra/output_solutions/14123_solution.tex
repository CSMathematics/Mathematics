\begin{alist}
\item Έχουμε $\Delta = (-\alpha)^2 - 4[-(\alpha + 1)] = \alpha^2 + 4\alpha + 4 = (\alpha + 2)^2$.
\begin{itemize}
\item Αν $\alpha = -2$ τότε $\Delta = 0$ άρα το τριώνυμο έχει μοναδική ρίζα.
\item Αν $\alpha \ne -2$ τότε $\Delta > 0$ άρα έχει δυο ρίζες άνισες.
\end{itemize}

\item 
\begin{rlist}
\item Αφού $\alpha > -2$ τότε το $f(x)$ έχει δυο ρίζες άνισες, τις $x = \dfrac{-(-\alpha) \pm \sqrt{(\alpha+2)^2}}{2} \Leftrightarrow x = \dfrac{\alpha \pm (\alpha+2)}{2}$, οπότε η μία ρίζα είναι 
\begin{gather*}
x = \dfrac{\alpha + \alpha + 2}{2} = \dfrac{2\alpha+2}{2} = \dfrac{2(\alpha+1)}{2} = \alpha + 1\text{και η άλλη} \\
x = \dfrac{\alpha - \alpha - 2}{2} = -1.
\end{gather*}
\item Έχει ρίζες τις $-1, \alpha + 1$ με $\alpha > -2 \Leftrightarrow \alpha + 1 > -1$ και το πρόσημο του τριωνύμου
φαίνεται στον παρακάτω πίνακα:
\begin{center}
\begin{tabular}{|c|ccccccc|}

$x$ & $-\infty$ & & $-1$ & & $\alpha + 1$ & & $+\infty$ \\ 
$x^2 - \alpha x - (\alpha + 1)$ & & $+$ & $0$ & $-$ & $0$ & $+$ & \\ 
\end{tabular}
\end{center}
Άρα $f(x) \le 0$ για κάθε $x \in [-1, \alpha + 1]$.
Οπότε πρέπει $(\alpha + 1) - (-1) = \alpha + 2 = 2024 \Leftrightarrow \alpha = 2022$.
\item Παρατηρούμε ότι $-1 < \dfrac{\alpha}{2} < \alpha + 1$ διότι:
\[-1 < \dfrac{\alpha}{2} \Leftrightarrow \alpha > -2\text{ισχύει και}\]
$\dfrac{\alpha}{2} < \alpha + 1 \Leftrightarrow \alpha < 2\alpha + 2 \Leftrightarrow \alpha > -2$, ισχύει.
Τότε σύμφωνα με το ερώτημα $\text{ii)}$ θα είναι $f\left(\dfrac{\alpha}{2}\right) < 0$.
\end{rlist}
\end{alist}
