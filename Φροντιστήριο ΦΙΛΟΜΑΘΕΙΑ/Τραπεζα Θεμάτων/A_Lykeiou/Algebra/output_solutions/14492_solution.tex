\lysh{14492}{2}

\begin{alist}
\item Η περίμετρος ενός ορθογώνιου παραλληλογράμμου είναι $P = 2x + 2y$. Τότε:
\[4 \le x \le 7 \Leftrightarrow 2 \cdot 4 \le 2x \le 2 \cdot 7 \Leftrightarrow 8 \le 2x \le 14, \quad (1)\]
και
\[2 \le y \le 3 \Leftrightarrow 2 \cdot 2 \le 2y \le 2 \cdot 3 \Leftrightarrow 4 \le 2y \le 6, \quad (2)\]

Προσθέτουμε κατά μέλη τις ανισώσεις $(1)$ και $(2)$ και βρίσκουμε:
\[8 + 4 \le 2x + 2y \le 14 + 6 \Leftrightarrow 12 \le P \le 20.\]

\item Αν το $x$ μειωθεί κατά $1$ και το $y$ τριπλασιαστεί η νέα περίμετρος θα είναι:
$P' = 2(x - 1) + 2(3y) = 2x + 6y - 2$. Τότε:
\begin{align*}
2 \le y \le 3 \Leftrightarrow 6 \cdot 2 \le 6y \le 6 \cdot 3 \Leftrightarrow 12 \le 6y \le 18 \ &\Leftrightarrow \\
12 - 2 &\le 6y - 2 \le 18 - 2 \Leftrightarrow 10 \le 6y - 2 \le 16, \quad (3)
\end{align*}

Προσθέτουμε κατά μέλη τις ανισώσεις $(1)$ και $(3)$ και βρίσκουμε:
\[8 + 10 \le 2x + 6y - 2 \le 14 + 16 \Leftrightarrow 18 \le P' \le 30.\]
\end{alist}
