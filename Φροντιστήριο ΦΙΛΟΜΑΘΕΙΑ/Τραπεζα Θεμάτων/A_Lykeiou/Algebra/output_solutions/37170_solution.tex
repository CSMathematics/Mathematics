\lysh{37170}{2}

\begin{alist}
\item Για να βρούμε τη θερμοκρασία ενός σημείου που βρίσκεται $30 \text{ km}$ κάτω από την επιφάνεια της Γης, θέτουμε στη δοθείσα σχέση $x = 30$ και βρίσκουμε:
\begin{align*}
T = 15 + 25 \cdot 30 \ &\Leftrightarrow \\
T = 15 + 750 \ &\Leftrightarrow T = 765^\circ\text{C}.
\end{align*}

\item Για να βρούμε σε ποιο βάθος η θερμοκρασία είναι ίση με $290^\circ\text{C}$, θέτουμε στη δοθείσα σχέση $T = 290$ και βρίσκουμε ισοδύναμα:
\begin{align*}
290 = 15 + 25x \ &\Leftrightarrow \\
275 = 25x \ &\Leftrightarrow x = 11 \text{ km}.
\end{align*}

\item Για να βρούμε σε ποιο βάθος η θερμοκρασία είναι μεγαλύτερη από $440^\circ\text{C}$ λύνουμε την ανίσωση:
\begin{align*}
T > 440 \ &\Leftrightarrow \\
15 + 25x > 440 \ &\Leftrightarrow \\
25x > 425 \ &\Leftrightarrow x > 17 \text{ km}.
\end{align*}

Επομένως η θερμοκρασία είναι μεγαλύτερη των $440^\circ\text{C}$ σε βάθος άνω των $17 \text{ km}$.
\end{alist}
