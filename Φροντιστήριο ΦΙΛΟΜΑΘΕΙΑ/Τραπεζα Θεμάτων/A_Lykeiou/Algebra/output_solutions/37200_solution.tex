\begin{alist}
\item Είναι:
\begin{align*}
|x + 1| < 2 \Leftrightarrow -2 < x + 1 < 2 \ &\Leftrightarrow \\
-2 - 1 &< x + 1 - 1 < 2 - 1 \Leftrightarrow -3 < x < 1 \Leftrightarrow x \in (-3, 1).
\end{align*}

\item Από το ερώτημα $\alpha)$ ισχύει ότι:
\begin{align*}
-3 < x < 1 \Leftrightarrow -3 < x \text{ και } x < 1 \ &\Leftrightarrow \\
x + 3 &> 0 \text{ και } x - 1 < 0.
\end{align*}

Άρα:
$|x + 3| = x + 3$ και $|x - 1| = -(x - 1) = 1 - x$.

Τότε:
\[K = \dfrac{|x + 3| + |x - 1|}{4} = \dfrac{x + 3 + 1 - x}{4} = \dfrac{4}{4} = 1.\]
\end{alist}
