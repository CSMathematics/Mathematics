\lysh{36897}{2}

\begin{alist}
\item Η ακολουθία των $\nu$ πρώτων διαδοχικών θετικών ακεραίων $1, 2, 3, \dots, \nu$ είναι αριθμητική πρόοδος με $\alpha_1 = 1$, $\omega = 1$ και $\alpha_\nu = \nu$. Άρα το άθροισμα των $\nu$ πρώτων όρων αυτής, είναι:
$S_\nu = \dfrac{\nu}{2}(\alpha_1 + \alpha_\nu)$, δηλαδή $S_\nu = \dfrac{\nu}{2}(1 + \nu)$.

\item Ψάχνουμε το πλήθος $\nu$ των όρων που έχουν άθροισμα $45$, δηλαδή το $\nu$ ώστε $S_\nu = 45$, δηλαδή
$\dfrac{\nu}{2}(1 + \nu) = 45$, οπότε $\nu \cdot (\nu + 1) = 90$.

Οι δυο διαδοχικοί φυσικοί αριθμοί που έχουν γινόμενο ίσο με $90$ είναι οι αριθμοί $9$ και $10$ (δηλαδή $\nu = 9$ και $\nu + 1 = 10$). Άρα το άθροισμα των $9$ πρώτων φυσικών αριθμών είναι ίσο με $45$.
\end{alist}
