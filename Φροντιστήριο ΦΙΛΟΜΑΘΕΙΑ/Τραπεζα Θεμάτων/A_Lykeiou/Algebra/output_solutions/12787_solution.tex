\lysh{12787}{2}

\begin{alist}
\item Η εξίσωση έχει συντελεστές $\alpha = 1, \beta = -1, \gamma = -6$, και διακρίνουσα $\Delta = (-1)^2 - 4 \cdot 1 \cdot (-6) = 25 > 0$ οπότε έχει άνισες ρίζες που είναι οι αριθμοί
\[x_{1,2} = \dfrac{1 \pm \sqrt{25}}{2} = \begin{cases} x_1 = \dfrac{1+5}{2} = 3 \\ x_2 = \dfrac{1-5}{2} = -2 \end{cases}\]

\item Οι αριθμοί $\kappa - 2, \kappa, 2\kappa + 3$, με $\kappa > 0$ είναι διαδοχικοί όροι γεωμετρικής προόδου, μόνο όταν κανένας από αυτούς δεν είναι $0$ και ισχύει:
\[\kappa^2 = (2\kappa + 3)(\kappa - 2) \Leftrightarrow \kappa^2 = 2\kappa^2 - 4\kappa - 6 + 3\kappa \Leftrightarrow \kappa^2 - \kappa - 6 = 0\]
$\Leftrightarrow \kappa = 3 \text{ ή } \kappa = -2$ (από το ερώτημα ($\alpha$))

Η τιμή $\kappa = -2$ απορρίπτεται αφού πρέπει $\kappa > 0$, ενώ η τιμή $\kappa = 3$ είναι δεκτή. Άρα $\kappa = 3$.
\end{alist}
