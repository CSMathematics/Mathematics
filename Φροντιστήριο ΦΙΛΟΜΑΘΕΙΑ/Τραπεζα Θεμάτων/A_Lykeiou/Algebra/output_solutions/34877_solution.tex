\lysh{34877}{2}

\begin{alist}
\item Το τριώνυμο $x^2 - 2x - 3$ έχει διακρίνουσα $\Delta = (-2)^2 - 4 \cdot 1 \cdot (-3) = 4 + 12 = 16 > 0$. Οι ρίζες
της εξίσωσης (1) είναι:
\begin{gather*}
x_1 = \dfrac{-(-2) - \sqrt{16}}{2 \cdot 1} = \dfrac{2 - 4}{2} = -1, \\
x_2 = \dfrac{-(-2) + \sqrt{16}}{2 \cdot 1} = \dfrac{2 + 4}{2} = 3.
\end{gather*}

\item Δεδομένου ότι $x_1 < x_2$ είναι οι ρίζες της εξίσωσης (1), οι αριθμοί $x_1, 1, x_2$, δηλαδή οι αριθμοί
$-1, 1, 3$, με τη σειρά που δίνονται είναι διαδοχικοί όροι αριθμητικής προόδου αν και μόνο αν
$2 \cdot 1 = -1 + 3$, που ισχύει.
\end{alist}
