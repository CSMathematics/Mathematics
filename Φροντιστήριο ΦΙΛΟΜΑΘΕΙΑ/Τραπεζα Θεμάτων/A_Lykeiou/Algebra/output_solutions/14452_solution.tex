\lysh{14452}{2}

\begin{alist}
\item Έχουμε:
\[\alpha^2 + \alpha\beta + \beta^2 = (\alpha + \beta)^2 - \alpha\beta = (\sqrt{3} - 1 + \sqrt{3} + 1)^2 - (\sqrt{3} - 1) \cdot (\sqrt{3} + 1) = (2\sqrt{3})^2 - 2 = 12 - 2 = 10.\]

\item Είναι $\alpha\beta = (\sqrt{3} - 1)(\sqrt{3} + 1) = (\sqrt{3})^2 - 1^2 = 3 - 1 = 2$ οπότε
\[\dfrac{\beta}{\alpha} + \dfrac{\alpha}{\beta} + 1 = \dfrac{\beta^2 + \alpha^2 + \alpha\beta}{\alpha\beta} = \dfrac{10}{2} = 5.\]
\end{alist}
