\lysh{38928}{4}

\begin{alist}
\item
\begin{rlist}
\item Το μήκος της πλευράς του τοίχου είναι ίσο με το μήκος του συρματοπλέγματος που
μένει, αν αφαιρέσουμε $2x$, που είναι το συνολικό μήκος των πλαϊνών πλευρών.
Επομένως, είναι ίσο με $60 - 2x$. Επίσης, πρέπει $0 < 2x < 60$, δηλαδή $0 < x < 30$.
Άρα, τελικά $l(x) = 60 - 2x$, με $0 < x < 30$.
\item Έχουμε ότι $E = (60 - 2x)x = -2x^2 + 60x$, με $0 < x < 30$.
\end{rlist}

\item
\begin{rlist}
\item Πρέπει $E \ge 400$ δηλαδή: $-2x^2 + 60x \ge 400$ ή $-x^2 + 30x - 200 \ge 0$.
Το τριώνυμο έχει διακρίνουσα $\Delta = 30^2 - 4(-1)(-200) = 100$ και ρίζες $x_1 = \dfrac{-30 + \sqrt{100}}{2(-1)} = \dfrac{-20}{-2} = 10$ και $x_2 = \dfrac{-30 - \sqrt{100}}{2(-1)} = \dfrac{-40}{-2} = 20$. Θέλουμε το τριώνυμο να είναι θετικό, δηλαδή ετερόσημο του $(-1)$, άρα πρέπει $10 \le x \le 20$.
\item Είναι $10 \le x \le 20$
άρα $-20 \ge -2x \ge -40$
οπότε $40 \ge 60 - 2x \ge 20$
δηλαδή $20 \le l(x) \le 40$.
\end{rlist}

\item Αν το συρματόπλεγμα έχει μήκος $\alpha\text{ m}$, τότε, όπως και στο ερώτημα $\alpha)$, βρίσκουμε ότι το
εμβαδόν του χώρου που θα περιφράξει δίνεται από τη σχέση $E(x) = -2x^2 + \alpha x$. Για να
μπορεί να περιφράξει έναν χώρο επιφάνειας $800\text{ m}^2$, πρέπει να έχει λύση η εξίσωση
$E(x) = 800$ ή $-2x^2 + \alpha x = 800$ ή $-2x^2 + \alpha x - 800 = 0$.

Για να έχει λύση η εξίσωση, πρέπει η διακρίνουσα $\Delta$ του τριωνύμου να μην είναι αρνητική,
δηλαδή πρέπει
$\Delta \ge 0$
ή
$\alpha^2 - 4(-2)(-800) \ge 0$
ή
$\alpha^2 - 6400 \ge 0$
ή
$\alpha^2 \ge 6400$
και επειδή $\alpha > 0$, βρίσκουμε $\alpha \ge 80$.
\end{alist}
