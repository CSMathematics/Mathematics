\lysh{35382}{2}

\begin{alist}
\item Το τριώνυμο $2x^2 - 3x - 2$ έχει $\alpha = 2, \beta = -3, \gamma = -2$ και διακρίνουσα:
\[\Delta = \beta^2 - 4\alpha\gamma = (-3)^2 - 4 \cdot 2 \cdot (-2) = 9 + 16 = 25 > 0\]

Οι ρίζες του τριωνύμου είναι οι:
\[x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{-(-3) \pm \sqrt{25}}{2 \cdot 2} = \dfrac{3 \pm 5}{4} = \begin{cases} \dfrac{3 + 5}{4} = 2 \\ \dfrac{3 - 5}{4} = -\dfrac{1}{2} \end{cases}\]

Τότε:
\[2x^2 - 3x - 2 = 2\left(x - \left(-\dfrac{1}{2}\right)\right)(x - 2) = (2x + 1)(x - 2)\]

\item Για να ορίζεται η παράσταση $K$, πρέπει ο παρονομαστής της να είναι διαφορετικός του μηδενός. Δηλαδή:
\begin{align*}
2x^2 - 3x - 2 \ne 0 &\stackrel{(\alpha)}{\Leftrightarrow} \\
\Leftrightarrow (2x + 1)(x - 2) \ne 0 \ &\Leftrightarrow \\
\Leftrightarrow (2x + 1 \ne 0 \text{ και } x - 2 \ne 0) \ &\Leftrightarrow \\
\Leftrightarrow &left(x \ne -\dfrac{1}{2} \text{ και } x \ne 2\right)&
\end{align*}

\item Για $x \ne -\dfrac{1}{2}$ και $x \ne 2$ ισχύει ότι:
\[K = \dfrac{x^2 - 4x + 4}{2x^2 - 3x - 2} = \dfrac{(x - 2)^2}{(2x + 1)(x - 2)} = \dfrac{x - 2}{2x + 1}\]
\end{alist}
