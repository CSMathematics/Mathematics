\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{gather*}
3x - 1 < x + 9\text{, οπότε} \\
3x - x < 1 + 9\text{, δηλαδή} \\
2x < 10\text{ και τελικά} \\
x < 5.
\end{gather*}

\item Έχουμε ισοδύναμα:
\begin{align*}
2 - \dfrac{x}{2} &\le x + \dfrac{1}{2} \ \Leftrightarrow \\
4 - x &\le 2x + 1\text{, δηλαδή} \\
3x &\ge 3\text{ και τελικά} \\
x &\ge 1.
\end{align*}

\item Με χρήση του άξονα των πραγματικών αριθμών, βλέπουμε ότι οι κοινές λύσεις των ανισώσεων των ερωτημάτων α) και β) είναι οι πραγματικοί αριθμοί $x$ για τους οποίους ισχύει
$x \in [1, 5)$.
\end{alist}
