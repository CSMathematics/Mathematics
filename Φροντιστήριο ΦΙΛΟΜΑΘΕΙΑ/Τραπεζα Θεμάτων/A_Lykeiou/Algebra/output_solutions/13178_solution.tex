\lysh{13178}{2}

\begin{alist}
\item Η κλίση της ευθείας $OM$ είναι $\lambda = \dfrac{4-0}{3-0} = \dfrac{4}{3}$ και η ζητούμενη ευθεία έχει εξίσωση $y = \dfrac{4}{3}x$.

\item 
\begin{rlist}
\item Το σημείο $N$ ανήκει στην ευθεία $OM$ οπότε οι συντεταγμένες του $N$ επαληθεύουν την εξίσωση της $OM$, δηλαδή ισχύει ότι: $\lambda = \dfrac{4}{3} \cdot (-3) \Leftrightarrow \lambda = -4$.
\item Τα σημεία $M(3, 4)$ και $N(-3, -4)$ έχουν αντίθετες συντεταγμένες, οπότε είναι συμμετρικά ως προς το $O$.
\end{rlist}
\end{alist}
