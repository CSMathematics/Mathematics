\begin{alist}
\item Έχουμε $\alpha_4 - \alpha_2 = 10$, δηλαδή
\[(\alpha_1 + 3\omega) - (\alpha_1 + \omega) = 10\text{, οπότε}\]
$\alpha_1 + 3\omega - \alpha_1 - \omega = 10$ και τελικά $2\omega = 10 \Leftrightarrow \omega = 5$.

\item Έχουμε ισοδύναμα:
\begin{gather*}
\alpha_1 + \alpha_2 + \alpha_3 = 33\text{, δηλαδή} \\
\alpha_1 + (\alpha_1 + \omega) + (\alpha_1 + 2\omega) = 33\text{, οπότε} \\
3\alpha_1 + 3\omega = 33\text{, δηλαδή} \\
\alpha_1 + \omega = 11\text{, οπότε}
\end{gather*}
$\alpha_1 + 5 = 11$ και τελικά $\alpha_1 = 6$.
\end{alist}
