\lysh{14543}{4}

\begin{alist}
\item Έχουμε διαδοχικά: $3 = 4 - 1 = 2^2 - 1^2$, $5 = 3^2 - 2^2$ και σκεπτόμενοι ότι
$7 = M^2 - N^2 = (M - N)(M + N)$, ένα γινόμενο ακεραίων που δίνει $7$ είναι οι $1, 7$.
Αν $M > N$ τότε $M - N = 1$, $M + N = 7$ και λύνοντας το σύστημα έχουμε με αντικατάσταση:
$M = N + 1$ και $N + 1 + N = 7$, άρα $2N = 6$, δηλαδή $N = 3$ και $M = 4$.
Οπότε $7 = 4^2 - 3^2$.

\item
\begin{rlist}
\item Έστω οι διαδοχικοί ακέραιοι $k, k+1$, $k \in \mathbb{Z}$.
Η διαφορά των τετραγώνων τους είναι $(k + 1)^2 - k^2 = k^2 + 2k + 1 - k^2 = 2k + 1$, $k \in \mathbb{Z}$.
Ο οποίος είναι περιττός αριθμός.
\item Έχουμε ότι $2021 = 2 \cdot 1010 + 1$.
Από την απόδειξη στο προηγούμενο ερώτημα έχουμε:
\[2021 = (1010 + 1)^2 - 1010^2 = 1011^2 - 1010^2.\]
\end{rlist}

\item Η γραμμοσκιασμένη περιοχή έχει εμβαδόν όσο η διαφορά των εμβαδών των τετραγώνων
$AB\Gamma\Delta$ και $\Gamma HPK$.
Το τετράγωνο $AB\Gamma\Delta$ έχει πλευρά $\nu+1$, ενώ το $\Gamma HPK$ έχει πλευρά $\nu$.
Ισχύει $(\nu + 1)^2 - \nu^2 = 45$, οπότε $2\nu + 1 = 45 \Leftrightarrow 2\nu = 44 \Leftrightarrow \nu = 22$.
\end{alist}
