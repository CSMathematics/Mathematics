\lysh{14071}{2}

\begin{alist}
\item Η αλγεβρική παράσταση $K$, που εκφράζει το άθροισμα των αποστάσεων του αριθμού $x$ από
τους αριθμούς $2$ και $-1$, πάνω στον άξονα είναι η $K = |x + 1| + |x - 2|$ διότι το $|x + 1|$ δηλώνει
την απόσταση του $x$ από το $-1$ και το $|x - 2|$ την απόσταση του $x$ από το $2$.

\item Είναι $K = |x + 1| + |x - 2|$ τότε:
\begin{rlist}
\item για $x = \dfrac{1}{2}$ γίνεται: $K = \left|\dfrac{1}{2} + 1\right| + \left|\dfrac{1}{2} - 2\right| = \left|\dfrac{3}{2}\right| + \left|-\dfrac{3}{2}\right| = \dfrac{3}{2} + \dfrac{3}{2} = \dfrac{6}{2} = 3$

\item Αφού είναι $x > 2$ τότε $x - 2 > 0$ και $x + 1 > 0$ οπότε η παράσταση $K$ γίνεται:
$K = x + 1 + x - 2 = 2x - 1$.
Ακόμη έχουμε: $x > 2 \Leftrightarrow 2x > 4 \Leftrightarrow 2x - 1 > 3 \Leftrightarrow K > 3$.
\end{rlist}
\end{alist}
