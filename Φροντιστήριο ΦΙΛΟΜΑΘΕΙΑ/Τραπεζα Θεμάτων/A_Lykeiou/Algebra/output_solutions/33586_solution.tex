\lysh{33586}{4}

\begin{alist}
\item Το μηνιαίο κόστος $K(\nu)$ της επιχείρησης, εάν γεμίζει $\nu$ τόνερ το μήνα, δίνεται από τη
σχέση $K(\nu) = 6500 + 15\nu$ ευρώ, με $\nu \in \mathbb{N}$.

\item Τα μηνιαία έσοδα $E(\nu)$ της επιχείρησης από την πώληση $\nu$ τόνερ το μήνα, είναι
$E(\nu) = 25\nu, \nu \in \mathbb{N}$.

\item 
\begin{rlist}
\item Η επιχείρηση δεν έχει ζημιά αν και μόνο αν $K(\nu) \le E(\nu)$, δηλαδή $6500 + 15\nu \le 25\nu$,
οπότε $10\nu \ge 6500$ και τελικά $\nu \ge 650$. Οπότε πρέπει να πωλούνται τουλάχιστον $650$
τόνερ κάθε μήνα ώστε η επιχείρηση να μην έχει ζημιά.

\item Το μηνιαίο κέρδος δίνεται από τη σχέση:
$E(\nu) - K(\nu) = 25\nu - (6500 + 15\nu) = 10\nu - 6500$. Η επιχείρηση έχει τουλάχιστον $500$ ευρώ
μηνιαίο κέρδος αν και μόνο αν, $10\nu - 6500 \ge 500 \Leftrightarrow 10\nu \ge 7000 \Leftrightarrow \nu \ge 700$. Επομένως η
επιχείρηση πρέπει να πουλήσει τουλάχιστον $700$ τόνερ το μήνα.
\end{rlist}
\end{alist}
