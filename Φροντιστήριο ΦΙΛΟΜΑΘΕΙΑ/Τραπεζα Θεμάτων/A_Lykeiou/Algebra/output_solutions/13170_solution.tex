\lysh{13170}{4}

\begin{alist}
\item Αν καταθέσουμε στην τράπεζα κεφάλαιο $x$ € με επιτόκιο $\varepsilon \%$, τότε στο τέλος του 1ου έτους,
το κεφάλαιο στην τράπεζα θα είναι $x + \dfrac{\varepsilon}{100} \cdot x = x \cdot \left(1 + \dfrac{\varepsilon}{100}\right)$.

Στο τέλος του 2ου έτους, το κεφάλαιο στην τράπεζα θα είναι
\[x \cdot \left(1 + \dfrac{\varepsilon}{100}\right) + \dfrac{\varepsilon}{100} \cdot x \cdot \left(1 + \dfrac{\varepsilon}{100}\right) = x \cdot \left(1 + \dfrac{\varepsilon}{100}\right) \cdot \left(1 + \dfrac{\varepsilon}{100}\right) = x \left(1 + \dfrac{\varepsilon}{100}\right)^2.\]

\item 
\begin{rlist}
\item Αφού ονομάζουμε $y$ το ποσό που κατατέθηκε στην τράπεζα Β, άρα $15000 - y$ θα είναι
το ποσό που κατατέθηκε στην τράπεζα Α. Σύμφωνα με το $\alpha)$ ερώτημα το ποσό που θα
υπάρχει στην τράπεζα Β μετά από δύο έτη θα είναι $y \left(1 + \dfrac{3}{100}\right)^2 = y \cdot 1,03^2$, ενώ το
αντίστοιχο ποσό στην τράπεζα Α θα είναι
\[(15000 - y) \left(1 + \dfrac{2}{100}\right)^2 = (15000 - y) \cdot 1,02^2.\]

Οπότε, θα πρέπει να ισχύει $y \cdot 1,03^2 + (15000 - y) \cdot 1,02^2 = 15811$. Έτσι, θα
έχουμε: $y \cdot 1,03^2 - y \cdot 1,02^2 + 15000 \cdot 1,02^2 = 15811 \Leftrightarrow$
\[y \cdot (1,03^2 - 1,02^2) = 15811 - 15000 \cdot 1,02^2\]

\item Η προηγούμενη εξίσωση γράφεται
\begin{align*}
y(1,03 - 1,02)(1,03 + 1,02) &= 15811 - 15000 \cdot 1,0404 \ \Leftrightarrow \\
y \cdot 0,01 \cdot 2,05 &= 15811 - 15606 \Leftrightarrow y = \dfrac{205}{0,01 \cdot 2,05} = \dfrac{2050000}{205} = 10000.
\end{align*}

Άρα το ποσό που κατατέθηκε στην τράπεζα Β ήταν $10000$ €, ενώ στην τράπεζα Α είναι
$5000$ €.
\end{rlist}
\end{alist}
