\lysh{34150}{2}

\begin{alist}
\item Έχουμε:
\[(\alpha + \beta)^2 = \alpha^2 + 2\alpha\beta + \beta^2 \Leftrightarrow 12^2 = 272 + 2\alpha\beta \Leftrightarrow 144 = 272 + 2\alpha\beta \Leftrightarrow 2\alpha\beta = -128 \Leftrightarrow \alpha\beta = -64.\]

\item Μια εξίσωση που έχει ρίζες τους αριθμούς $\alpha, \beta$ είναι η $x^2 - Sx + P = 0$, όπου
$S = \alpha + \beta = 12$ και $P = \alpha\beta = -64$.

Άρα μια εξίσωση που έχει ρίζες τους αριθμούς $\alpha, \beta$ είναι η: $x^2 - 12x - 64 = 0$.

\item Το τριώνυμο $x^2 - 12x - 64$ έχει διακρίνουσα $\Delta = (-12)^2 - 4 \cdot 1 \cdot (-64) = 144 + 256 = 400 > 0$.

Οι ρίζες της εξίσωσης $x^2 - 12x - 64 = 0$ είναι:
\[\alpha, \beta = \dfrac{-(-12) \pm \sqrt{400}}{2 \cdot 1} = \dfrac{12 \pm 20}{2} = \begin{cases} 16 \\ -4 \end{cases}\]

Άρα $\alpha = 16, \beta = -4$ ή $\alpha = -4, \beta = 16$.
\end{alist}
