\lysh{13169}{2}

\begin{alist}
\item Αφού $3 \le x \le 5$, θα έχουμε $x - 5 \le 5 - 5$, άρα $x - 5 \le 0$. Ακόμα $3 \le x$, οπότε $3 - 2 \le x - 2$, άρα $1 \le x - 2$. Ώστε $x - 2 > 0$.

\item Γνωρίζουμε ότι αν $y \ge 0$, τότε $|y| = y$ ενώ αν $y \le 0$, τότε $|y| = -y$. Έτσι η δοθείσα εξίσωση γράφεται $x - 2 - (5 - x) = 2$, άρα $x - 2 - 5 + x = 2$, οπότε $2x = 9$. Τελικά
$x = \dfrac{9}{2}$, λύση η οποία είναι δεκτή, αφού $\dfrac{9}{2} = 4,5 \in [3, 5]$.
\end{alist}
