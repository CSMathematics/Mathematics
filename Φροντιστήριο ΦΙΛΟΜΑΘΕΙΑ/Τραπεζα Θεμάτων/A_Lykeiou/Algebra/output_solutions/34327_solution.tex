\lysh{34327}{4}

\begin{alist}
\item Το τριώνυμο $x^2 - 3x - 4$ έχει $\alpha = 1, \beta = -3, \gamma = -4$ και διακρίνουσα:
\[\Delta = \beta^2 - 4\alpha\gamma = (-3)^2 - 4 \cdot 1 \cdot (-4) = 9 + 16 = 25 > 0\]
Οι ρίζες της εξίσωσης (1) είναι:
\[x_{1,2} = \dfrac{-(-3) \pm \sqrt{25}}{2 \cdot 1} = \dfrac{3 \pm 5}{2} = \begin{cases} \dfrac{3 + 5}{2} = 4 \\ \dfrac{3 - 5}{2} = -1 \end{cases}.\]

\item 
\begin{rlist}
\item Ο αριθμός $\dfrac{\alpha}{\beta}$ είναι λύση της εξίσωσης (1) αν και μόνο αν την επαληθεύει, δηλαδή αν
και μόνο αν ισχύει:
\begin{align*}
&\left(\dfrac{\alpha}{\beta}\right)^2 - 3\dfrac{\alpha}{\beta} - 4 = 0 \Leftrightarrow \dfrac{\alpha^2}{\beta^2} - 3\dfrac{\alpha}{\beta} - 4 = 0 \stackrel{\cdot \beta^2}{\Leftrightarrow} \\
\alpha^2 - 3\alpha\beta - 4\beta^2 &= 0,
\end{align*}
το οποίο ισχύει από την υπόθεση.

\item Στο ερώτημα $\alpha)$ βρήκαμε ότι οι λύσεις της εξίσωσης (1) είναι οι $x = 4$ και $x = -1$.
Επίσης, στο ερώτημα $\beta\text{)i.}$ δείξαμε ότι ο αριθμός $\dfrac{\alpha}{\beta}$ είναι λύση της εξίσωσης (1).
Οπότε, πρέπει:
$\dfrac{\alpha}{\beta} = 4$ ή $\dfrac{\alpha}{\beta} = -1$.
Επειδή οι $\alpha, \beta$ είναι ομόσημοι, η περίπτωση $\dfrac{\alpha}{\beta} = -1$ απορρίπτεται. Άρα, ισχύει:
\[\dfrac{\alpha}{\beta} = 4 \Leftrightarrow \alpha = 4\beta,\]
δηλαδή, ο $\alpha$ είναι τετραπλάσιος του $\beta$.
\end{rlist}
\end{alist}
