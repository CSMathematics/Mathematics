\lysh{12998}{4}

\begin{alist}
\item 
\begin{rlist}
\item Οι αριθμοί αυτοί δεν μπορούν να είναι διαδοχικοί όροι αριθμητικής προόδου, καθώς θα έπρεπε να ισχύει $2\beta = \alpha + \gamma$, δηλαδή:
$162 = 27\sqrt{3} + 81\sqrt{3} \Leftrightarrow 162 = 108\sqrt{3} \Leftrightarrow 81 = 54\sqrt{3} \Leftrightarrow \dfrac{81}{54} = \sqrt{3} \Leftrightarrow \dfrac{3}{2} = \sqrt{3}$, το οποίο δεν ισχύει αφού ο $\sqrt{3}$ είναι άρρητος, ενώ ο $\dfrac{3}{2}$ είναι ρητός, άρα δεν μπορούν να είναι ίσοι.
\item Οι δύο αριθμοί είναι θετικοί, οπότε για να είναι ίσοι, αρκεί τα τετράγωνά τους να είναι ίσα μεταξύ τους:
$\beta^2 = \alpha \cdot \gamma \Leftrightarrow 81^2 = 27\sqrt{3} \cdot 81\sqrt{3} \Leftrightarrow 6561 = 2187 \cdot 3 \Leftrightarrow 6561 = 6561$, το οποίο ισχύει.
\end{rlist}

\item Η γεωμετρική πρόοδος θα έχει σταθερό λόγο: $\lambda = \dfrac{81}{27\sqrt{3}} = \sqrt{3}$.
Ο $\nu$-οστός όρος της γεωμετρικής προόδου είναι $\alpha_\nu = \alpha_1 \cdot \lambda^{\nu-1} = \alpha_1 \cdot (\sqrt{3})^{\nu-1}$.
Όμως, εφόσον ισχύει ότι $\alpha_7 = 27\sqrt{3}$ έχουμε ότι $\alpha_1 \cdot (\sqrt{3})^6 = 27\sqrt{3}$, οπότε
\[\alpha_1 = \dfrac{27\sqrt{3}}{27} = \sqrt{3}.\]
Συνεπώς, ισχύει ότι $\alpha_1 = \sqrt{3}$, άρα ο γενικός όρος της γεωμετρικής προόδου είναι
\[\alpha_\nu = \sqrt{3} \cdot (\sqrt{3})^{\nu-1} = (\sqrt{3})^\nu.\]

\item Για το άθροισμα των $10$ πρώτων όρων της γεωμετρικής προόδου ισχύει ότι:
\[S_{10} = \alpha_1 \cdot \dfrac{\lambda^{10} - 1}{\lambda - 1} = \sqrt{3} \cdot \dfrac{(\sqrt{3})^{10} - 1}{\sqrt{3} - 1} = \sqrt{3} \cdot \dfrac{243 - 1}{\sqrt{3} - 1} = \dfrac{242\sqrt{3}}{\sqrt{3} - 1} = \dfrac{242\sqrt{3}(\sqrt{3} + 1)}{(\sqrt{3} - 1)(\sqrt{3} + 1)} = \dfrac{242(3 + \sqrt{3})}{2} = 363 + 121\sqrt{3}.\]
\end{alist}
