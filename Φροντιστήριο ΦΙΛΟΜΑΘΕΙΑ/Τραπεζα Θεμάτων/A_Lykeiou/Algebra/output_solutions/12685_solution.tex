\lysh{12685}{2}

\begin{alist}
\item Κάνοντας πράξεις στη δοθείσα σχέση έχουμε διαδοχικά:
\begin{gather*}
\alpha \cdot \dfrac{1}{\alpha} + \alpha \cdot \dfrac{1}{\beta} + \beta \cdot \dfrac{1}{\alpha} + \beta \cdot \dfrac{1}{\beta} = 4\text{, τότε} \\
1 + \dfrac{\alpha}{\beta} + \dfrac{\beta}{\alpha} + 1 = 4\text{, τότε} \\
\dfrac{\alpha}{\beta} + \dfrac{\beta}{\alpha} = 2,
\end{gather*}
το οποίο είναι το ζητούμενο.

\item Από το ερώτημα ($\alpha$) έχουμε:
\begin{align*}
\dfrac{\alpha}{\beta} + \dfrac{\beta}{\alpha} = 2 \ &\Leftrightarrow
\dfrac{\alpha^2 + \beta^2}{\alpha\beta} = 2 \ \Leftrightarrow \\
\alpha^2 + \beta^2 = 2\alpha\beta \ &\Leftrightarrow
\alpha^2 + \beta^2 - 2\alpha\beta = 0 \ \Leftrightarrow \\
(\alpha - \beta)^2 = 0 \ &\Leftrightarrow 
\alpha - \beta = 0 \Leftrightarrow \alpha = \beta.
\end{align*}
\end{alist}
