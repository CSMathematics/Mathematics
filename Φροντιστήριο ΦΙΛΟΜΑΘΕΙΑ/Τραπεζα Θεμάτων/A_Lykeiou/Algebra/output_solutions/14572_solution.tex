\lysh{14572}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\[|x + 2| < 1 \Leftrightarrow -1 < x + 2 < 1 \Leftrightarrow -1 - 2 < x + 2 - 2 < 1 - 2 \Leftrightarrow -3 < x < -1.\]

\item Έχουμε διαδοχικά:
$-3 < x < -1$, πολλαπλασιάζουμε τα μέλη της ανίσωσης με το $2$, οπότε
$-6 < 2x < -2$, προσθέτουμε στα μέλη της ανίσωσης το $4$ και έχουμε
\begin{gather*}
-2 < 2x + 4 < 2\text{, οπότε τελικά} \\
|2x + 4| < 2.
\end{gather*}

Εναλλακτικά, πολλαπλασιάζουμε τα μέλη της ανίσωσης $|x + 2| < 1$ με το $2$ και έχουμε
\begin{gather*}
2 \cdot |x + 2| < 2 \cdot 1\text{, οπότε} \\
|2 \cdot (x + 2)| < 2\text{και τελικά} \\
|2x + 4| < 2.
\end{gather*}
\end{alist}
