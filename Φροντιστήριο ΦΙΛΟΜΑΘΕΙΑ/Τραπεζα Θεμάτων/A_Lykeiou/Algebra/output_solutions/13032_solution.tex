\lysh{13032}{2}

\begin{alist}
\item Το πεδίο ορισμού της συνάρτησης $f$ είναι $A_f = \mathbb{R}$. Η συνάρτηση $g$ ορίζεται όταν
$x + 5 \ge 0$, δηλαδή όταν $x \ge -5$. Οπότε το πεδίο ορισμού της συνάρτησης $g$ είναι
$A_g = [-5, +\infty)$.

\item $f(-1) = 1 - 3 \cdot (-1) = 1 + 3 = 4$ και $g(11) = \sqrt{11 + 5} = \sqrt{16} = 4$.
Άρα $f(-1) = g(11)$.

\item Αναζητούμε την τιμή του $x$ για την οποία:
\begin{align*}
f(x) = g(4) \ &\Leftrightarrow 
1 - 3x = \sqrt{4 + 5} \ \Leftrightarrow \\
1 - 3x = 3 \ &\Leftrightarrow
-3x = 2 \ \Leftrightarrow x= -\dfrac{2}{3}.
\end{align*}
\end{alist}
