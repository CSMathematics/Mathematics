\lysh{38846}{4}

\begin{alist}
\item Θα πρέπει να αποδείξουμε ότι για οποιαδήποτε τιμή του $x$ ισχύει:
\[f(x) \ge -2 \Leftrightarrow x^6 + x^4 - 2 \ge -2 \Leftrightarrow x^6 + x^4 \ge 0 \Leftrightarrow x^4 (x^2 + 1) \ge 0,\]
που είναι προφανές, διότι $x^4 \ge 0$ και $x^2 + 1 > 0$.

\item
\begin{rlist}
\item Επειδή $f\left(\dfrac{-1}{2}\right) = \left(\dfrac{-1}{2}\right)^6 + \left(\dfrac{-1}{2}\right)^4 - 2 = \dfrac{1}{64} + \dfrac{1}{16} - 2 = \dfrac{1+4-128}{64} = -\dfrac{123}{64}$, η αθλήτρια περνά από τη θέση $\left(-\dfrac{1}{2}, -\dfrac{123}{64}\right)$. Επειδή από τη γραφική παράσταση της $f$ παρατηρούμε ότι $f\left(\dfrac{1}{4}\right) < 0$, δεν περνά από τη θέση $\left(\dfrac{1}{4}, \dfrac{251}{16}\right)$, όταν βουτά.

\item Η κίνηση της αθλήτριας έξω από το νερό περιορίζεται στο τμήμα της γραφικής παράστασης της συνάρτησης $f$ που καθορίζεται από το σημείο από το οποίο η αθλήτρια πέφτει στο νερό από ύψος $3\text{m}$ και από το σημείο $A$ όπου εισέρχεται στο νερό. Για τις τετμημένες $x$ των σημείων αυτού του τμήματος ισχύει: $x \in [-1,19, -1)$.

\item Η αθλήτρια κολύμπησε ευθύγραμμα την απόσταση $\Gamma B$ μεταξύ των σημείων $\Gamma(0, -2)$ και $B(1, 0)$, η οποία είναι:
\[d(\Gamma, B) = \sqrt{(0 - 1)^2 + (-2 - 0)^2} = \sqrt{1^2 + 2^2} = \sqrt{5}\text{ m}.\]
\end{rlist}
\end{alist}
