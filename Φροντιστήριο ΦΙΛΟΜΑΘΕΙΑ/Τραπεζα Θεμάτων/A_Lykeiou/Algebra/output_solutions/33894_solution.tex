\lysh{33894}{4}

\begin{alist}
\item Πρέπει $|2 - x| \ne 0 \Leftrightarrow 2 - x \ne 0 \Leftrightarrow x \ne 2$. Άρα $A_f = (-\infty, 2) \cup (2, +\infty)$.

\item Το τριώνυμο $x^2 - 5x + 6$ έχει άθροισμα ριζών $S = x_1 + x_2 = -\dfrac{\beta}{\alpha} = 5$ και γινόμενο ριζών
$P = x_1 x_2 = \dfrac{\gamma}{\alpha} = 6$, άρα οι ρίζες του είναι: $x_1 = 2, x_2 = 3$. Οπότε $x^2 - 5x + 6 = (x - 3)(x - 2)$.
Έχουμε $f(x) = \dfrac{x^2 - 5x + 6}{|2 - x|} = \dfrac{(x - 2)(x - 3)}{|2 - x|} = \begin{cases} \dfrac{(x - 2)(x - 3)}{-(2 - x)} = \dfrac{(x - 2)(x - 3)}{x - 2} = x - 3, & x > 2 \\ \dfrac{(x - 2)(x - 3)}{2 - x} = \dfrac{-(2 - x)(x - 3)}{2 - x} = -x + 3, & x < 2 \end{cases}$.

\item 
\begin{rlist}
\item Για $x > 2$ η γραφική παράσταση της $f$ είναι τμήμα ευθείας με εξίσωση $y = x - 3$. Δυο σημεία
από τα οποία διέρχεται η ευθεία είναι τα $(3, 0)$ και $(4, 1)$. Για $x < 2$ η γραφική παράσταση
της $f$ είναι τμήμα ευθείας με εξίσωση $y = -x + 3$. Δυο σημεία από τα οποία διέρχεται η ευθεία
είναι τα $(1, 2)$ και $(0, 3)$. Άρα η γραφική παράσταση της $f$ αποτελείται από δύο ημιευθείες:

(Η γραφική παράσταση παραλείπεται)

\item Από την γραφική παράσταση της $f$ βλέπουμε ότι τέμνει τον $x'x$
άξονα στο σημείο $(3, 0)$ και τον $y'y$ άξονα στο $(0, 3)$.
\end{rlist}

\item Από το $\gamma\text{)i.}$ ερώτημα, παρατηρούμε ότι η γραφική παράσταση της $f$ είναι κάτω από τον
άξονα $x'x$ για $x \in (2, 3)$ και τέμνει τον $x'x$ στο $(3, 0)$ (δηλαδή $f(3) = 0$). Άρα η ανίσωση
$f(x) \le 0$ αληθεύει για $x \in (2, 3]$.
\end{alist}
