\lysh{34871}{2}

\begin{alist}
\item Οι αριθμοί $x + 2, x + 1, 3x + 2$, με τη σειρά που δίνονται, είναι διαδοχικοί όροι
αριθμητικής προόδου αν και μόνο αν $2(x + 1) = (3x + 2) + (x + 2)$, οπότε ισοδύναμα έχουμε:

\begin{gather*}
2x + 2 = 4x + 4\text{, δηλαδή} \\
-2x = 2\text{και τελικά} \\
x = -1.
\end{gather*}

\item Για $x = -1$, οι διαδοχικοί όροι της προόδου είναι $-1 + 2, -1 + 1, 3 \cdot (-1) + 2$, δηλαδή
$1, 0, -1$. Η διαφορά της αριθμητικής προόδου είναι: $\omega = 0 - 1 = -1$.
\end{alist}
