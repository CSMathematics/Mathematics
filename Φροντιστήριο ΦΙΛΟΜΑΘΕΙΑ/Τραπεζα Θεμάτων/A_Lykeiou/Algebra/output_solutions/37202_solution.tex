\lysh{37202}{2}

\begin{alist}
\item Το τριώνυμο $x^2 - 5x + 6$ έχει $\alpha = 1, \beta = -5, \gamma = 6$ και διακρίνουσα
\[\Delta = \beta^2 - 4\alpha\gamma = (-5)^2 - 4 \cdot 1 \cdot 6 = 25 - 24 = 1 > 0.\]

Οι ρίζες του τριωνύμου είναι οι:
\[x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{-(-5) \pm \sqrt{1}}{2 \cdot 1} = \dfrac{5 \pm 1}{2} = \begin{cases} \dfrac{5 + 1}{2} = 3 \\[2ex] \dfrac{5 - 1}{2} = 2 \end{cases}\]

Οπότε:
$x^2 - 5x + 6 = (x - 2)(x - 3)$.

\item
\begin{rlist}
\item Πρέπει:
\begin{align*}
x^2 - 5x + 6 \ne 0 \Leftrightarrow (x - 2)(x - 3) \ne 0 \ &\Leftrightarrow \\
x - 2 &\ne 0 \text{ και } x - 3 \ne 0 \Leftrightarrow x \ne 2 \text{ και } x \ne 3.
\end{align*}

Άρα το πεδίο ορισμού της συνάρτησης $f$ είναι το $A = \mathbb{R} - \{2, 3\}$.

\item Ο τύπος της συνάρτησης $f$ γράφεται:
\[f(x) = \dfrac{x - 2}{x^2 - 5x + 6} = \dfrac{x - 2}{(x - 2)(x - 3)} = \dfrac{1}{x - 3}.\]
\end{rlist}
\end{alist}
