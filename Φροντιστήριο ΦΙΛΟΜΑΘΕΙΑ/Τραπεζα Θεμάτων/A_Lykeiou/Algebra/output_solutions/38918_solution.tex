\lysh{38918}{4}

\begin{alist}
\item Έχουμε $V = \pi r^2 h \Leftrightarrow 440 = \pi r^2 h \Leftrightarrow h = \dfrac{440}{\pi r^2}$. Οπότε η σχέση $E = 2\pi rh + 2\pi r^2$ γίνεται:
\[E = 2\pi r \cdot \dfrac{440}{\pi r^2} + 2\pi r^2 = \dfrac{880}{r} + 2\pi r^2.\]

\item Έχουμε $2 < r < 8$, οπότε
$4 < r^2 < 64$ και τελικά
$8\pi < 2\pi r^2 < 128\pi$. (1)
Επίσης $2 < r < 8$, οπότε
\begin{align*}
\dfrac{1}{2} &> \dfrac{1}{r} > \dfrac{1}{8} \Leftrightarrow \dfrac{1}{8} < \dfrac{1}{r} < \dfrac{1}{2}\text{, οπότε} \\
\dfrac{880}{8} &< \dfrac{880}{r} < \dfrac{880}{2}\text{και τελικά}
\end{align*}
$110 < \dfrac{880}{r} < 440$. (2)

Προσθέτουμε τις (1) και (2) κατά μέλη και προκύπτει $110 + 8\pi < E < 440 + 128\pi$.

\item
\begin{rlist}
\item Από το $\beta)$ ερώτημα έχουμε $110 + 8\pi < E < 440 + 128\pi$. Ισχύει
$110 + 8\pi < 320,2 < 440 + 128$, οπότε είναι μεταξύ των τιμών που βρήκαμε στο $\beta)$ ερώτημα.
\item Για $r = 4,1\text{ cm}$, η σχέση $h = \dfrac{440}{\pi r^2}$ γίνεται: $h = \dfrac{440}{\pi \cdot 4,1^2} = \dfrac{440}{16,81\pi} \simeq \dfrac{440}{53} \simeq 8,3$. Οπότε το ύψος είναι περίπου διπλάσιο της ακτίνας. Μπορούμε να πούμε λοιπόν ότι η ποσότητα του μετάλλου ελαχιστοποιείται, όταν η ακτίνα της κονσέρβας είναι $r = 4,1\text{ cm}$ και το ύψος της περίπου διπλάσιο αυτής.
\end{rlist}
\end{alist}
