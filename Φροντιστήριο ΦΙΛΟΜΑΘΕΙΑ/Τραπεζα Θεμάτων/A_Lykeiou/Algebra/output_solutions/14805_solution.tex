\lysh{14805}{3}

\begin{alist}
\item Με $\sqrt{2} \simeq 1,41$ έχουμε $3\sqrt{2} > 4$, οπότε $3\sqrt{2} - 4 > 0$, άρα $|3\sqrt{2} - 4| = 3\sqrt{2} - 4$.
Επίσης $\sqrt{2} - 2 < 0$, οπότε $|\sqrt{2} - 2| = -\sqrt{2} + 2$.
Έχουμε λοιπόν:
\[\alpha = 3\sqrt{2} - 4 + 2(2 - \sqrt{2}) = 3\sqrt{2} - 4 + 4 - 2\sqrt{2} = \sqrt{2}\]
που είναι το ζητούμενο.

\item Με τη βοήθεια του ερωτήματος $\alpha)$ έχουμε:
\[\alpha^3 = \alpha^2 \cdot \alpha = (\sqrt{2})^2 \cdot \alpha = 2\alpha\]

\item Αφού $\alpha^3 = 2\alpha$, έχουμε:
\[A = \alpha^3 + (\alpha - 1)^2 = 2\alpha + \alpha^2 - 2\alpha + 1 = \alpha^2 + 1 = (\sqrt{2})^2 + 1 = 2 + 1 = 3.\]
\end{alist}
