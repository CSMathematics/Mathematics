\lysh{14617}{2}

\begin{alist}
\item Έχουμε ισοδύναμα: $|x - 7| < 1$,
\begin{gather*}
-1 < x - 7 < 1\text{, δηλαδή} \\
7 - 1 < x < 7 + 1\text{και τελικά} \\
6 < x < 8.
\end{gather*}

\item Έχουμε ισοδύναμα: $6 < k < 8$,
\begin{gather*}
\dfrac{1}{6} > \dfrac{1}{k} > \dfrac{1}{8}\text{, δηλαδή} \\
\dfrac{24}{6} > \dfrac{24}{k} > \dfrac{24}{8}\text{, οπότε} \\
4 > \dfrac{24}{k} > 3\text{ και τελικά} \\
3 < \dfrac{24}{k} < 4.
\end{gather*}
\end{alist}
