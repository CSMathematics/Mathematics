\lysh{14052}{3}

\begin{alist}
\item Έχουμε ισοδύναμα: $x^2 - 1 = 0 \Leftrightarrow x^2 = 1 \Leftrightarrow x = \pm 1$.

\item Ισχύει ότι $|x| \ge -x$ για κάθε $x \in \mathbb{R}$. Η ισότητα ισχύει για $x \le 0$. Άρα,
\[|x| + x = 0 \Leftrightarrow |x| = -x \Leftrightarrow x \le 0.\]

\item Δεδομένου ότι $\sqrt{x^2 - 1} \ge 0$ και $|x| \ge -x \Leftrightarrow |x| + x \ge 0$ για κάθε $x$ στο πεδίο ορισμού, έχουμε ισοδύναμα:
\begin{align*}
|x| + \sqrt{x^2 - 1} + x = 0 \Leftrightarrow \sqrt{x^2 - 1} + (|x| + x) = 0 \ &\Leftrightarrow \\
\begin{cases} x^2 - 1 = 0 \\ |x| + x = 0 \end{cases} \stackrel{(\alpha),(\beta)}{\Leftrightarrow} \begin{cases} x = \pm 1 \\ x \le 0 \end{cases}.
\end{align*}
Άρα $x = -1$.
\end{alist}
