\lysh{14649}{2}

\begin{alist}
\item Για $x \ge -1$ είναι $x + 1 \ge 0$, οπότε $|x + 1| + 2 = x + 1 + 2 = x + 3$.
Για $x < -1$ είναι $x + 1 < 0$, οπότε $|x + 1| + 2 = -(x + 1) + 2 = -x - 1 + 2 = 1 - x$.

Άρα, τελικά $K = \begin{cases} x + 3, & \alpha\nu \ x \ge -1 \\ 1 - x, & \alpha\nu \ x < -1 \end{cases}$

\item 
\begin{rlist}
\item Είναι $|x - 2| = 4 \Leftrightarrow x - 2 = 4 \text{ ή } x - 2 = -4 \Leftrightarrow x = 6 \text{ ή } x = -2$.
\item Για $x = 6 > -1$ είναι $K = 6 + 3 = 9$.
Για $x = -2 < -1$ είναι $K = 1 - (-2) = 1 + 2 = 3$.
\end{rlist}
\end{alist}
