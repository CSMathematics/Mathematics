\lysh{33889}{4}

\begin{alist}
\item Το τριώνυμο $3x^2 - 14x + 8$ έχει διακρίνουσα
\[\Delta = (-14)^2 - 4 \cdot 3 \cdot 8 = 196 - 96 = 100 > 0,\]
οπότε η εξίσωση $3x^2 - 14x + 8 = 0$ έχει δυο ρίζες άνισες, τις
\begin{gather*}
x_1 = \dfrac{-(-14) + \sqrt{100}}{2 \cdot 3} = \dfrac{14 + 10}{6} = 4\text{ και} \\
x_2 = \dfrac{-(-14) - \sqrt{100}}{2 \cdot 3} = \dfrac{14 - 10}{6} = \dfrac{4}{6} = \dfrac{2}{3}.
\end{gather*}

Το τριώνυμο $8x^2 - 14x + 3$ έχει διακρίνουσα
\[\Delta = (-14)^2 - 4 \cdot 8 \cdot 3 = 196 - 96 = 100 > 0\text{, οπότε}\]
η εξίσωση $8x^2 - 14x + 3 = 0$ έχει δυο ρίζες άνισες, τις
\begin{gather*}
x_1 = \dfrac{-(-14) + \sqrt{100}}{2 \cdot 8} = \dfrac{14 + 10}{16} = \dfrac{24}{16} = \dfrac{3}{2}\text{ και} \\
x_2 = \dfrac{-(-14) - \sqrt{100}}{2 \cdot 8} = \dfrac{14 - 10}{16} = \dfrac{4}{16} = \dfrac{1}{4}.
\end{gather*}

\item Ο αριθμός $\rho$ είναι ρίζα της εξίσωσης $(3)$ αν και μόνο αν την επαληθεύει, δηλαδή αν και μόνο αν ισχύει: $\alpha\rho^2 + \beta\rho + \gamma = 0$. (5)
\begin{rlist}
\item Εάν $\rho = 0$, τότε από την σχέση (5) προκύπτει $\gamma = 0$, άτοπο, αφού $\alpha \cdot \gamma \ne 0$. Άρα $\rho \ne 0$.

\item Ο $\dfrac{1}{\rho}$ είναι ρίζα της εξίσωσης $(4)$ αν και μόνο αν
\[\gamma \left(\dfrac{1}{\rho}\right)^2 + \beta \left(\dfrac{1}{\rho}\right) + \alpha = 0 \Leftrightarrow \gamma \dfrac{1}{\rho^2} + \beta \dfrac{1}{\rho} + \alpha = 0 \stackrel{\cdot \rho^2}{\Leftrightarrow} \gamma + \beta\rho + \alpha\rho^2 = 0,\]
που ισχύει λόγω της σχέσης (5).
\end{rlist}
\end{alist}
