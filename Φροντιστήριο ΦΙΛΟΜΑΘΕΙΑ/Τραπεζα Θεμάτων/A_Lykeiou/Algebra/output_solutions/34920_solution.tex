\lysh{34920}{2}

\begin{alist}
\item Το τριώνυμο $3x^2 + 6x - 12$ έχει $\alpha = 3, \beta = 6$ και $\gamma = -12$. Οπότε το άθροισμα των
ριζών του είναι $S = x_1 + x_2 = -\dfrac{\beta}{\alpha} = -\dfrac{6}{3} = -2$ και το γινόμενό τους είναι
\[P = x_1 x_2 = \dfrac{\gamma}{\alpha} = \dfrac{-12}{3} = -4.\]

\item Μια εξίσωση που έχει ρίζες τους αριθμούς $4x_1$ και $4x_2$, είναι η $x^2 - Sx + P = 0$, με
\begin{gather*}
S = 4x_1 + 4x_2 = 4(x_1 + x_2) = 4 \cdot (-2) = -8\text{και} \\
P = 4x_1 \cdot 4x_2 = 16(x_1 \cdot x_2) = 16 \cdot (-4) = -64.
\end{gather*}
Άρα, μια εξίσωση είναι η $x^2 + 8x - 64 = 0$.
\end{alist}
