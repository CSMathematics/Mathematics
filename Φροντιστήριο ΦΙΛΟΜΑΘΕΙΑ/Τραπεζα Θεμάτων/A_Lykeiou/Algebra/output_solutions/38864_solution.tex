\lysh{38864}{4}

\begin{alist}
\item Πρέπει $|x - 200| \le 2$, οπότε $-2 \le x - 200 \le 2$ και τελικά
$198 \le x \le 202$. Άρα οι αποδεκτές τιμές της μέτρησης είναι από $198\text{ m}$ μέχρι $202\text{ m}$.

\item
\begin{rlist}
\item Έχουμε ότι
$|x_\mu - x_\pi| \le \dfrac{1}{100} x_\pi$,
άρα
\[-\dfrac{1}{100} x_\pi \le x_\mu - x_\pi \le \dfrac{1}{100} x_\pi,\]
οπότε
\[-\dfrac{1}{100} x_\pi + x_\pi \le x_\mu \le \dfrac{1}{100} x_\pi + x_\pi,\]
δηλαδή
\[\dfrac{99}{100} x_\pi \le x_\mu \le \dfrac{101}{100} x_\pi\]
και τελικά
$0,99x_\pi \le x_\mu \le 1,01x_\pi$.

\item Από τη σχέση $0,99x_\pi \le x_\mu \le 1,01x_\pi$ προκύπτει ότι η μεγαλύτερη διαφορά δύο μετρήσεων είναι αυτή των δύο ακραίων τιμών $0,99x_\pi$ και $1,01x_\pi$, δηλαδή
$1,01x_\pi - 0,99x_\pi = 0,02x_\pi$.

\item Πρέπει $0,02x_\pi \le 0,5$ ή $x_\pi \le \dfrac{0,5}{0,02}$, ή $x_\pi \le 25$. Άρα οι μετρήσεις μιας απόστασης $x_\pi$ δεν μπορεί να διαφέρουν μεταξύ τους περισσότερο από $0,5\text{ m}$, αν η απόσταση αυτή είναι μέχρι και $25\text{ m}$.
\end{rlist}

\item Έχουμε ότι $0,99x_\pi \le x_\mu \le 1,01x_\pi$, άρα για $x_\mu = 150$:
\begin{align*}
0,99x_\pi \le 150 \le 1,01x_\pi \ &\Leftrightarrow \\
0,99 \le \dfrac{150}{x_\pi} \le 1,01 \ &\Leftrightarrow \\
\dfrac{1}{1,01} \le \dfrac{x_\pi}{150} \le \dfrac{1}{0,99} \ &\Leftrightarrow \\
\dfrac{150}{1,01} \le x_\pi \le \dfrac{150}{0,99} \ &\Leftrightarrow \\
148,51 \le &x_\pi \le 151,51
\end{align*}

\textit{Άλλος τρόπος:}
Έχουμε $|150 - x_\pi| \le \dfrac{1}{100} x_\pi$, άρα
\[-\dfrac{x_\pi}{100} \le 150 - x_\pi \le \dfrac{x_\pi}{100}\]
Δηλαδή πρέπει $-\dfrac{x_\pi}{100} \le 150 - x_\pi$ και $150 - x_\pi \le \dfrac{x_\pi}{100}$, ισοδύναμα
\[x_\pi - \dfrac{x_\pi}{100} \le 150 \text{ και } 150 \le x_\pi + \dfrac{x_\pi}{100}\]
οπότε
\[\dfrac{99}{100}x_\pi \le 150 \text{ και } 150 \le \dfrac{101}{100}x_\pi\]
Άρα τελικά $\dfrac{150 \cdot 100}{101} \le x_\pi \le \dfrac{150 \cdot 100}{99}$ ή $148,51 \le x_\pi \le 151,51$.
\end{alist}
