\lysh{12834}{4}

\begin{alist}
\item Για το εμβαδόν $(A\Delta E)$ ισχύει: $(A\Delta E) = (AB\Gamma\Delta) - (ABE) - (\Gamma\Delta E) =$
\begin{gather*}
= \dfrac{AB + \Gamma\Delta}{2} \cdot A\Delta - \dfrac{AB \cdot EZ}{2} - \dfrac{\Delta\Gamma \cdot EH}{2} = \dfrac{2 + 6}{2} \cdot 4 - \dfrac{2(4 - x)}{2} - \dfrac{6x}{2} = 16 - (4 - x) - 3x \\
= 16 - 4 + x - 3x = -2x + 12.
\end{gather*}
Ώστε $f(x) = -2x + 12$.
Για την μεταβλητή $x$, πρέπει $x \in [0,4]$, αφού απαιτούμε τα μήκη $EH$ και $EZ$ να είναι μη
αρνητικοί αριθμοί, άρα πρέπει $x \ge 0$ και $4 - x \ge 0$, άρα $x \le 4$.

\item Η γραφική παράσταση της συνάρτησης $y = -2x + 12$ με $x \in \mathbb{R}$, γνωρίζουμε ότι είναι μια ευθεία, οπότε εδώ, η γραφική παράσταση της συνάρτησης $f(x)$ θα είναι ένα ευθύγραμμο τμήμα, με άκρα τα σημεία $A(0,12)$ και $B(4,4)$, αφού $f(0) = -2 \cdot 0 + 12 = 12$ και $f(4) = -2 \cdot 4 + 12 = -8 + 12 = 4$.

\item $S = f\left(\dfrac{1}{16}\right) + f\left(\dfrac{2}{16}\right) + f\left(\dfrac{3}{16}\right) + \dots + f\left(\dfrac{64}{16}\right) =$
\[= \left(-2 \cdot \dfrac{1}{16} + 12\right) + \left(-2 \cdot \dfrac{2}{16} + 12\right) + \dots + \left(-2 \cdot \dfrac{64}{16} + 12\right)\]
$S = -\dfrac{2}{16} \cdot (1 + 2 + 3 + \dots + 64) + 12 \cdot 64 = -\dfrac{1}{8} \cdot \dfrac{64(64+1)}{2} + 12 \cdot 64$. Τελικά:
\[S = -4 \cdot 65 + 12 \cdot 64 = (-4 + 12) \cdot 65 - 12 = 8 \cdot 65 - 12 = 520 - 12 = 508.\]

Χρησιμοποιήσαμε την ισότητα $1 + 2 + 3 + \dots + \nu = \dfrac{\nu(\nu+1)}{2}$ αφού πρόκειται για άθροισμα $\nu$ διαδοχικών όρων μιας αριθμητικής προόδου $(\alpha_\nu)$ με $\alpha_1 = 1$, $\omega = 1$, $\alpha_\nu = \nu$, οπότε από τον
γνωστό τύπο $S_\nu = \dfrac{\nu}{2}(\alpha_1 + \alpha_\nu)$ παίρνουμε $S_\nu = \dfrac{\nu}{2}(1 + \nu) = \dfrac{\nu(\nu+1)}{2}$.
\end{alist}
