\lysh{36660}{4}

\begin{alist}
\item Οι αποστάσεις $(\alpha_i), i = 1, 2, 3, \dots, 20$ των κυψελών $K_1, K_2, K_3, K_4, \dots, K_{20}$ από την αποθήκη $A$ διαφέρουν πάντα κατά τον σταθερό αριθμό $3$. Αποτελούν επομένως διαδοχικούς όρους αριθμητικής προόδου με πρώτο όρο $\alpha_1 = 1$ (η απόσταση της κυψέλης $K_1$ από την αποθήκη $A$) και διαφορά $\omega = 3$ (η απόσταση δύο διπλανών κυψελών).
Ο $\nu$-οστός όρος της προόδου είναι:
\[\alpha_\nu = \alpha_1 + (\nu - 1)\omega = 1 + (\nu - 1)3 = 3\nu - 2.\]

\item Αναζητούμε τον όρο $\alpha_{20}$. Είναι:
\[\alpha_{20} = 3 \cdot 20 - 2 = 58\text{ m}\]

\item
\begin{rlist}
\item Οι διαδρομές που θα κάνει ο μελισσοκόμος είναι:
\[A \to K_1 \to A, \quad A \to K_2 \to A, \quad A \to K_3 \to A\]
Άρα θα διανύσει απόσταση:
\[(1 + 1) + (4 + 4) + (7 + 7) = 24\text{ μέτρα}\]

\item Εφόσον πρέπει να πάει και να γυρίσει σε κάθε κυψέλη, η συνολική απόσταση για να συλλέξει το μέλι από όλες τις κυψέλες είναι
\begin{gather*}
(1 + 1) + (4 + 4) + (7 + 7) + \dots + (58 + 58) = \\
= (1 + 4 + 7 + \dots + 58) + (1 + 4 + 7 + \dots + 58) = 2S_{20}
\end{gather*}
Τελικά η ζητούμενη συνολική απόσταση είναι:
\[2S_{20} = 2 \cdot \dfrac{20}{2}(\alpha_1 + \alpha_{20}) = 20(1 + 58) = 20 \cdot 59 = 1180\text{ μέτρα}.\]
\end{rlist}
\end{alist}
