\lysh{12914}{4}

\begin{alist}
\item 
\begin{rlist}
\item Επειδή η ευθεία $\varepsilon: y = c$ είναι παράλληλη με τον άξονα $x'x$, από το σχήμα προκύπτει
ότι για να έχει κοινά σημεία με την γραφική παράσταση της $f$ πρέπει $c \ge 1$.
\item Για να έχει κοινά σημεία η ευθεία $\varepsilon: y = c$ με τη γραφική παράσταση της $f$, πρέπει η εξίσωση $|x - 2| + 1 = c \Leftrightarrow |x - 2| = c - 1$ να έχει λύση. Επειδή είναι $|x - 2| \ge 0$, αρκεί $c - 1 \ge 0 \Leftrightarrow c \ge 1$.
\end{rlist}

\item Για $c \ge 1$ η $|x - 2| = c - 1$ έχει λύσεις τις:
$x - 2 = c - 1 \Leftrightarrow x = c + 1$ ή $x - 2 = -(c - 1) \Leftrightarrow x = 3 - c$.
Άρα τα κοινά σημεία είναι τα $A(3 - c, c)$ και $B(c + 1, c)$.

\item 
\begin{rlist}
\item Επειδή τα σημεία $A$ και $B$ έχουν την ίδια τεταγμένη, για να είναι $(AB) \le 2$ πρέπει το μήκος του οριζόντιου τμήματος ανάμεσα στις δύο ημιευθείες της γραφικής παράστασης της $f$ να είναι μικρότερο ή ίσο του $2$. Από το σχήμα προκύπτει ότι αυτό ισχύει για τα σημεία με τεταγμένη από $1$ έως και $2$. Άρα $1 \le c \le 2$.
\item Επειδή τα σημεία $A$ και $B$ έχουν την ίδια τεταγμένη, το μήκος $(AB)$ είναι
$(AB) = |c + 1 - (3 - c)| = |2c - 2|$. Άρα:
$(AB) \le 2 \Leftrightarrow |2c - 2| \le 2 \Leftrightarrow -2 \le 2c - 2 \le 2$, η οποία γίνεται:
\[-1 \le c - 1 \le 1 \Leftrightarrow 0 \le c \le 2.\]
Αλλά από το ερώτημα $\alpha)$ πρέπει $c \ge 1$. Άρα τελικά $1 \le c \le 2$.
\end{rlist}
\end{alist}
