\lysh{14704}{2}

\begin{alist}
\item Προσθέτουμε κατά μέλη τις ανισώσεις $2 \le x \le 3$ και $1 \le y \le 2$ βρίσκουμε:
\[2 + 1 \le x + y \le 3 + 2 \Leftrightarrow 3 \le x + y \le 5.\]

\item Πολλαπλασιάζουμε την ανίσωση $2 \le x \le 3$ με $2$ και βρίσκουμε: $4 \le 2x \le 6 \quad (1)$.

Πολλαπλασιάζουμε την ανίσωση $1 \le y \le 2$ με $-3$ και βρίσκουμε:
\[-3 \ge -3y \ge -6 \Leftrightarrow -6 \le -3y \le -3 \quad (2).\]

Προσθέτουμε κατά μέλη τις ανισώσεις $(1)$ και $(2)$ και βρίσκουμε:
$-2 \le 2x - 3y \le 3$.

\item Ισχύει ότι: $2 \le x \le 3 \quad (3)$ και $1 \le y \le 2 \Leftrightarrow 1 \ge \dfrac{1}{y} \ge \dfrac{1}{2} \Leftrightarrow \dfrac{1}{2} \le \dfrac{1}{y} \le 1 \quad (4)$.

Πολλαπλασιάζουμε κατά μέλη τις ανισώσεις $(3)$ και $(4)$ και βρίσκουμε:
\[2 \cdot \dfrac{1}{2} \le x \cdot \dfrac{1}{y} \le 3 \cdot 1 \Leftrightarrow 1 \le \dfrac{x}{y} \le 3.\]
\end{alist}
