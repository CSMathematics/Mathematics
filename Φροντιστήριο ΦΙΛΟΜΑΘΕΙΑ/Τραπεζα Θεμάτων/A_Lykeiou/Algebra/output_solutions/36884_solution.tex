\item Έχουμε ισοδύναμα:
\begin{align*}
x^2 + y^2 - 2x + 6y + 10 &= 0 \overset{(\alpha)}{\Leftrightarrow} \\
(x - 1)^2 + (y + 3)^2 &= 0\text{, οπότε}
\end{align*}

$x - 1 = 0$ και $y + 3 = 0$, δηλαδή
$x = 1$ και $y = -3$.
\end{alist}
