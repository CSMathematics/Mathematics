\lysh{12856}{2}

\begin{alist}
\item 
\begin{rlist}
\item Ισχύει $\epsilon \parallel \delta$ τότε οι δύο ευθείες θα έχουν την ίδια κλίση.
Η κλίση της ευθείας $\delta$ είναι $-3$. Άρα $\alpha = -3$, δηλαδή η κλίση της ευθείας $\epsilon$ είναι $-3$.
\item Η γωνία που σχηματίζει η ευθεία $\epsilon$ με τον $x'x$ άξονα είναι αμβλεία διότι η κλίση της $\epsilon$ είναι $\alpha = -3 < 0$.
\end{rlist}

\item Η ευθεία $\epsilon$ έχει εξίσωση $y = -3x + 5$.
Η ευθεία τέμνει τον άξονα $x'x$ σε σημείο με τεταγμένη $y = 0$.
Για $y = 0$ έχουμε: $-3x + 5 = 0 \Leftrightarrow x = \dfrac{5}{3}$.
Άρα το σημείο τομής της ευθείας με τον άξονα $x'x$ είναι το $\left(\dfrac{5}{3}, 0\right)$.

Η ευθεία τέμνει τον άξονα $y'y$ σε σημείο με τετμημένη $x = 0$. Για $x = 0$ έχουμε: $y = 5$.
Άρα το σημείο τομής της ευθείας με τον άξονα $y'y$ είναι το $(0, 5)$.
\end{alist}
