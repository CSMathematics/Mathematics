\lysh{34446}{2}

\begin{alist}
\item Για να βρούμε την απόσταση του αυτοκινήτου μετά από 25 λεπτά θέτουμε στη
δοθείσα σχέση $x = 25$ και βρίσκουμε:
\begin{align*}
y = 35 + 0,8 \cdot 25 \ &\Leftrightarrow \\
\Leftrightarrow y = 35 + 20 \ &\Leftrightarrow \\
\Leftrightarrow y &= 55\text{χιλιόμετρα}
\end{align*}

\item Για να βρούμε τα λεπτά που θα έχει κινηθεί το αυτοκίνητο, όταν θα απέχει 75
χιλιόμετρα από την πόλη Α θέτουμε στη δοθείσα σχέση $y = 75$ και βρίσκουμε:
\begin{align*}
75 = 35 + 0,8x \Leftrightarrow 40 = 0,8x \ &\Leftrightarrow \\
\Leftrightarrow x &= 50\text{λεπτά}
\end{align*}
\end{alist}
