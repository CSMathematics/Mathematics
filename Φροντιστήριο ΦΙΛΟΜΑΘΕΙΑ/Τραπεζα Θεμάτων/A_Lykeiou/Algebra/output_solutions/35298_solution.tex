\lysh{35298}{2}

\begin{alist}
\item Είναι:
\begin{gather*}
f(-1) = 2 \cdot (-1) + 4 = -2 + 4 = 2\text{και} \\
f(3) = 3 - 1 = 2.
\end{gather*}
Άρα $f(-1) = f(3)$.

\item Για $x < 0$ είναι:
\[f(x) = 0 \Leftrightarrow 2x + 4 = 0 \Leftrightarrow 2x = -4 \Leftrightarrow x = -2.\]

Για $x \ge 0$ είναι:
\[f(x) = 0 \Leftrightarrow x - 1 = 0 \Leftrightarrow x = 1.\]
\end{alist}
