\lysh{36671}{4}

\begin{alist}
\item Η απόσταση των σημείων $M(x)$ του άξονα των πραγματικών αριθμών από το σημείο $A(5)$ είναι μικρότερη ή ίση του $9$.

\item Από τον άξονα των αριθμών βρίσκουμε ότι: $x \in [-4, 14]$.

\item Είναι:
\begin{align*}
d(x, 5) \le 9 \Leftrightarrow |x - 5| \le 9 \Leftrightarrow -9 \le x - 5 \le 9 \ &\Leftrightarrow \\
\Leftrightarrow -9 + 5 &\le x - 5 + 5 \le 9 + 5 \Leftrightarrow -4 \le x \le 14
\end{align*}

\item Αφού $x \in [-4, 14]$ έχουμε ότι:
\begin{align*}
x &\ge -4 \Leftrightarrow x + 4 \ge 0 \Rightarrow |x + 4| = x + 4\text{και} \\
x &\le 14 \Leftrightarrow x - 14 \le 0 \Rightarrow |x - 14| = -(x - 14) = -x + 14
\end{align*}

οπότε $|x + 4| + |x - 14| = x + 4 + (-x + 14) = 18$.
\end{alist}
