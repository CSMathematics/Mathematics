\lysh{36898}{2}

\begin{alist}
\item Έχουμε ισοδύναμα (αφού $\alpha, \beta$ είναι ομόσημοι, άρα $\alpha \cdot \beta > 0$):
\begin{gather*}
\dfrac{\alpha}{\beta} + \dfrac{\beta}{\alpha} \ge 2\text{, οπότε} \\
\dfrac{\alpha^2 + \beta^2}{\alpha \cdot \beta} \ge 2\text{, οπότε} \\
\alpha^2 + \beta^2 \ge 2\alpha \cdot \beta\text{, συνεπώς} \\
\alpha^2 + \beta^2 - 2\alpha \cdot \beta \ge 0\text{ και τελικά } (\alpha - \beta)^2 \ge 0
\end{gather*}
που ισχύει.
\item Η ισότητα ισχύει αν και μόνο αν $\alpha - \beta = 0 \Leftrightarrow \alpha = \beta$, δηλαδή όταν οι αριθμοί $\alpha, \beta$ είναι ίσοι.
\end{alist}
