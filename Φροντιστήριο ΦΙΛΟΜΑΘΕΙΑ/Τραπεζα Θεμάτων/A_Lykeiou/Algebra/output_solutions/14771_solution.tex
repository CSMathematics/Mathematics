\lysh{14771}{4}

\begin{alist}
\item Από το σχήμα προκύπτει ότι η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ στο $1$ άρα:
\[f(1) = 0 \Leftrightarrow \sqrt{1 - \alpha} = 0 \Leftrightarrow 1 - \alpha = 0 \Leftrightarrow \alpha = 1.\]

\item Για να ορίζεται η συνάρτηση $f$ πρέπει και αρκεί $x - 1 \ge 0 \Leftrightarrow x \ge 1$. Άρα το πεδίο ορισμού
της $f$ είναι το $A_f = [1, +\infty)$.

\item 
\begin{rlist}
\item Από το σχήμα βλέπουμε ότι η τετμημένη του σημείου $\Gamma$ είναι $2$, άρα η τεταγμένη του
είναι $f(2) = \sqrt{2 - 1} = \sqrt{1} = 1$. Αντίστοιχα η τετμημένη του σημείου $\Delta$ είναι $3$ και η
τεταγμένη του $f(3) = \sqrt{3 - 1} = \sqrt{2}$.

\item Το μήκος του τμήματος $B\Gamma$ είναι $(B\Gamma) = \sqrt{(2 - 3)^2 + (1 - 0)^2} = \sqrt{1 + 1} = \sqrt{2}$. 

\item Παρατηρούμε ότι $(\Delta B) = |\sqrt{2} - 0| = \sqrt{2} = (B\Gamma)$. Άρα το τρίγωνο $B\Gamma\Delta$ είναι ισοσκελές με
$(\Delta B) = (B\Gamma)$.
\end{rlist}
\end{alist}
