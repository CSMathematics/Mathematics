\lysh{35299}{2}

\begin{alist}
\item Από τα δεδομένα της άσκησης είναι $\alpha_1 = 120$ και $\omega = 20$. Τότε:
\begin{align*}
\alpha_\nu = \alpha_1 + (\nu - 1)\omega \ &\Leftrightarrow \\
\Leftrightarrow \alpha_\nu = 120 + (\nu - 1)20 \ &\Leftrightarrow \\
\Leftrightarrow \alpha_\nu = 100 + 20\nu&.
\end{align*}

\item Η τελευταία σειρά έχει:
\begin{align*}
\alpha_{10} = 100 + 20 \cdot 10 \ &\Leftrightarrow \\
\Leftrightarrow \alpha_{10} = 100 + 200 \ &\Leftrightarrow
\end{align*}
$\Leftrightarrow \alpha_{10} = 300$ καθίσματα.

\item Το γυμναστήριο έχει συνολικά:
$S_{10} = \dfrac{10}{2}(\alpha_1 + \alpha_{10}) = 5(120 + 300) = 5 \cdot 420 = 2100$ καθίσματα.
\end{alist}
