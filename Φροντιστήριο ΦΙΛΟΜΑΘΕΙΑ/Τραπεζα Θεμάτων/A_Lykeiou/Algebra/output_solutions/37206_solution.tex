\begin{alist}
\item Οι τετμημένες των σημείων τομής των $C_f, C_g$ αποτελούν λύσεις της εξίσωσης $f(x) = g(x)$. Τότε:
\begin{align*}
f(x) = g(x) \ &\Leftrightarrow \\
x^2 + 3x + 2 = x + 1 \ &\Leftrightarrow \\
x^2 + 2x + 1 = 0 \ &\Leftrightarrow \\
(x + 1)^2 = 0 \ &\Leftrightarrow \\
x + 1 &= 0 \Leftrightarrow x = -1
\end{align*}

Άρα οι $C_f, C_g$ έχουν μόνο ένα κοινό σημείο, το $A(-1, g(-1))$ δηλαδή το $A(-1, 0)$. (Η ευθεία εφάπτεται της παραβολής).

\item Οι τετμημένες των κοινών σημείων των $C_f, C_h$ είναι λύσεις της εξίσωσης $f(x) = h(x)$. Δηλαδή:
\begin{align*}
f(x) = h(x) \ &\Leftrightarrow \\
x^2 + 3x + 2 = x + \alpha \ &\Leftrightarrow
\end{align*}
$x^2 + 2x + (2 - \alpha) = 0, \text{ με } \alpha \in \mathbb{R}$ (1)

Το τριώνυμο έχει διακρίνουσα:
\[\Delta = 2^2 - 4 \cdot 1 \cdot (2 - \alpha) = 4 - 8 + 4\alpha = 4\alpha - 4 = 4(\alpha - 1)\]

\begin{rlist}
\item Αν $\alpha > 1$ τότε $\Delta > 0$ και η εξίσωση (1) έχει δύο ρίζες άνισες το οποίο σημαίνει ότι οι γραφικές παραστάσεις των $f, h$ έχουν δύο κοινά σημεία.
\item Αν $\alpha < 1$ τότε $\Delta < 0$ και η εξίσωση (1) δεν έχει πραγματικές ρίζες το οποίο σημαίνει ότι οι γραφικές παραστάσεις των $f, h$ δεν έχουν κοινά σημεία.
\end{rlist}
\end{alist}
