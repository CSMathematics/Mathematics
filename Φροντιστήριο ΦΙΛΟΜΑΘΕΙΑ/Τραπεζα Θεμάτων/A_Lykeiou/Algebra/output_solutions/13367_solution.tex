\lysh{13367}{4}

\begin{alist}
\item Για να βρούμε το είδος της γωνίας που σχηματίζει η ευθεία $\varepsilon$ με τον άξονα $x'x$, πρέπει να
βρούμε το πρόσημο του συντελεστή διεύθυνσης.
Ο συντελεστής διεύθυνσης της ευθείας $\varepsilon$ είναι: $\alpha = \omega^2 - 6\omega + 8$.
Η διακρίνουσα του τριωνύμου είναι: $\Delta = (-6)^2 - 4 \cdot 8 = 36 - 32 = 4$
και οι ρίζες:
\[\omega_{1,2} = \dfrac{6 \pm 2}{2} \Leftrightarrow \omega_1 = 4 \text{ και } \omega_2 = 2.\]

Το πρόσημο του τριωνύμου φαίνεται στον παρακάτω πίνακα:

\begin{itemize}
\item Αν $\omega \in (-\infty, 2) \cup (4, +\infty)$ ο συντελεστής διεύθυνσης $\alpha$ είναι θετικός και κατά
συνέπεια η γωνία που σχηματίζει η ευθεία $\varepsilon$ με τον άξονα $x'x$ είναι οξεία.
\item Αν $\omega \in (2, 4)$ ο συντελεστής διεύθυνσης $\alpha$ είναι αρνητικός και κατά συνέπεια η γωνία
που σχηματίζει η ευθεία $\varepsilon$ με τον άξονα $x'x$ είναι αμβλεία.
\item Αν $\omega = 2$ ή $\omega = 4$ τότε ο συντελεστής διεύθυνσης μηδενίζεται και η ευθεία $\varepsilon$
σχηματίζει μηδενική γωνία με τον άξονα $x'x$, είναι δηλαδή παράλληλη στον $x'x$.
\end{itemize}

\item 
\begin{rlist}
\item Σύμφωνα με το προηγούμενο ερώτημα για να σχηματίζει η ευθεία $\varepsilon$ αμβλεία γωνία
με τον άξονα $x'x$ πρέπει $\omega \in (2, 4)$. Επειδή όμως το $\omega$ είναι ακέραιος τότε $\omega = 3$.
\item Για $\omega = 3$, $\varepsilon : y = -x + 2$.
Για να βρούμε τα σημεία τομής της ευθείας $\varepsilon$ με τον άξονα $x'x$ θέτουμε στην εξίσωσή
της $y = 0$. Δηλαδή $-x + 2 = 0 \Leftrightarrow x = 2$. Άρα το σημείο τομής είναι το $A(2, 0)$.
Για να βρούμε το σημείο τομής της ευθείας $\varepsilon$ με τον άξονα $y'y$, θέτουμε στην
εξίσωσή της $x = 0$ και βρίσκουμε $y = 2$. Άρα το σημείο τομής είναι το $B(0, 2)$.
\end{rlist}

\item Σημειώνουμε πάνω στους άξονες τα σημεία $A$ και $B$ που βρήκαμε, τα ενώνουμε και
σχεδιάζουμε την ευθεία $\varepsilon$.
\end{alist}
