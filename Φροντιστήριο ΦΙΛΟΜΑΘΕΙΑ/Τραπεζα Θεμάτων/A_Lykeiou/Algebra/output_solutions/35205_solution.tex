\lysh{35205}{2}

\begin{alist}
\item Οι αριθμοί $x, 2x + 1, 5x + 4$ είναι διαδοχικοί όροι γεωμετρικής προόδου αν και
μόνο αν:

\begin{align*}
(2x + 1)^2 = x \cdot (5x + 4) \ &\Leftrightarrow \\
\Leftrightarrow 4x^2 + 4x + 1 = 5x^2 + 4x \ &\Leftrightarrow \\
\Leftrightarrow x^2 = 1 \ &\Leftrightarrow \\
\Leftrightarrow (x &= -1 \text{ ή } x = 1).
\end{align*}

\item
\begin{rlist}
\item Για $x = 1$ οι δοσμένοι αριθμοί γράφονται:
$1, 3, 9$.
Ο λόγος $\lambda$ είναι $\lambda = \dfrac{3}{1} = 3$.

\item Για $x = -1$ οι δοσμένοι αριθμοί γράφονται:
$-1, -1, -1$.
Ο λόγος $\lambda$ είναι $\lambda = \dfrac{-1}{-1} = 1$.
\end{rlist}
\end{alist}
