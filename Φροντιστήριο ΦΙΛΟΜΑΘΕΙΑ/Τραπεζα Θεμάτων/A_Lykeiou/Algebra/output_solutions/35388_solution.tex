\lysh{35388}{2}

\begin{alist}
\item Είναι:
\begin{align*}
\dfrac{\alpha + \beta}{\beta} &= 4 \Leftrightarrow \alpha + \beta = 4\beta \Leftrightarrow \alpha = 3\beta\text{ και} \\
\dfrac{\gamma}{\delta - \gamma} &= \dfrac{1}{4} \Leftrightarrow 4\gamma = \delta - \gamma \Leftrightarrow \delta = 5\gamma.
\end{align*}

\item Για $\alpha = 3\beta$ και $\delta = 5\gamma$ η παράσταση $\Pi$ γράφεται:
\[\Pi = \dfrac{\alpha\gamma + \beta\gamma}{\beta\delta - \beta\gamma} = \dfrac{(3\beta)\gamma + \beta\gamma}{\beta(5\gamma) - \beta\gamma} = \dfrac{3\beta\gamma + \beta\gamma}{5\beta\gamma - \beta\gamma} = \dfrac{4\beta\gamma}{4\beta\gamma} = 1.\]
\end{alist}
