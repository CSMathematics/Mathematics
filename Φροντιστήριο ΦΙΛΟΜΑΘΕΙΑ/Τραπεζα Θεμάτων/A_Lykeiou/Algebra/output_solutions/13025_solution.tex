\lysh{13025}{2}

\begin{alist}
\item H δοθείσα ανίσωση γράφεται $\dfrac{2x - 3}{7} \ge 5$ οπότε $2x - 3 \ge 35$, άρα
\[2x \ge 38 \Leftrightarrow x \ge 19.\]

\item Καθώς οι αντίθετοι αριθμοί έχουν ίσες απόλυτες τιμές, η δοθείσα ανίσωση γράφεται
ισοδύναμα $|x + 1| \le 23 \Leftrightarrow -23 \le x + 1 \le 23 \Leftrightarrow -23 - 1 \le x \le 23 - 1 \Leftrightarrow -24 \le x \le 22$.

\item Παριστάνουμε τις λύσεις των δύο παραπάνω ανισώσεων στον ίδιο άξονα, απ’ όπου
προκύπτει ότι $19 \le x \le 22$.
\end{alist}
