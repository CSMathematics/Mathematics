\lysh{37204}{4}

\begin{alist}
\item Επειδή το πλήθος των καθισμάτων της κάθε σειράς αυξάνει καθώς ανεβαίνουμε από σειρά σε σειρά κατά τον ίδιο πάντα αριθμό καθισμάτων $\omega$, οι αριθμοί που εκφράζουν το πλήθος των καθισμάτων κάθε σειράς, είναι διαδοχικοί όροι αριθμητικής προόδου $(\alpha_\nu)$ με $\alpha_1 = 16$ και διαφορά $\omega$.
Είναι:
\begin{align*}
\alpha_7 = 28 \Leftrightarrow \alpha_1 + (7 - 1)\omega = 28 \ &\Leftrightarrow \\
16 + 6\omega &= 28 \Leftrightarrow 6\omega = 12 \Leftrightarrow \omega = 2
\end{align*}
Άρα $\alpha_1 = 16$ και $\omega = 2$.

\item Έχουμε:
\begin{align*}
\alpha_\nu = \alpha_1 + (\nu - 1)\omega \Leftrightarrow \alpha_\nu = 16 + (\nu - 1) \cdot 2 \ &\Leftrightarrow \\
\alpha_\nu &= 16 + 2\nu - 2 \Leftrightarrow \alpha_\nu = 2\nu + 14 \text{ με } 1 \le \nu \le 20
\end{align*}

\item Το πλήθος των καθισμάτων του θεάτρου είναι:
\begin{align*}
S_{20} = \dfrac{20}{2}[2\alpha_1 + (20 - 1)\omega] \ &\Leftrightarrow \\
S_{20} = 10(2 \cdot 16 + 19 \cdot 2) \ &\Leftrightarrow \\
S_{20} = 10(32 + 38) \ &\Leftrightarrow \\
S_{20} &= 10 \cdot 70 \Leftrightarrow S_{20} = 700
\end{align*}

\item Ο αριθμός των κενών καθισμάτων σε κάθε σειρά είναι αριθμητική πρόοδος $(\beta_\nu)$ με
$\beta_1 = 6$ και $\omega = 3$. Ο $\nu$-οστός όρος που εκφράζει το πλήθος των κενών καθισμάτων είναι:
\begin{align*}
\beta_\nu = \beta_1 + (\nu - 1)\omega \ &\Leftrightarrow \\
\beta_\nu = 6 + (\nu - 1) \cdot 3 \ &\Leftrightarrow \\
\beta_\nu &= 6 + 3\nu - 3 \Leftrightarrow \beta_\nu = 3\nu + 3
\end{align*}

Άρα $\beta_\nu = 3\nu + 3$ με $1 \le \nu \le 11$ (διότι τα κενά καθίσματα δε μπορεί να είναι περισσότερα από τα καθίσματα της κάθε σειράς, δηλαδή πρέπει $\beta_\nu \le \alpha_\nu \Leftrightarrow \nu \le 11$)

\begin{rlist}
\item Όλα τα καθίσματα θα είναι κενά της $\nu$-οστής σειράς, όταν:
\[\beta_\nu = \alpha_\nu \Leftrightarrow 3\nu + 3 = 2\nu + 14 \Leftrightarrow \nu = 11\]
Άρα από την $11$η σειρά μέχρι την $20$η, όλα τα καθίσματα είναι κενά.

\item Το πλήθος των κενών καθισμάτων στις $10$ πρώτες σειρές είναι:
\begin{align*}
S_{10}^\prime = \dfrac{10}{2}[2\beta_1 + (10 - 1)\omega] \ &\Leftrightarrow \\
S_{10}^\prime &= 5(2 \cdot 6 + 9 \cdot 3) \Leftrightarrow S_{10}^\prime = 5 \cdot 39 \Leftrightarrow S_{10}^\prime = 195
\end{align*}

Το πλήθος των καθισμάτων στις πρώτες $10$ σειρές είναι:
\begin{align*}
S_{10} = \dfrac{10}{2}[2\alpha_1 + (10 - 1)\omega] \ &\Leftrightarrow \\
S_{10} &= 5(2 \cdot 16 + 9 \cdot 2) \Leftrightarrow S_{10} = 5 \cdot 50 \Leftrightarrow S_{10} = 250
\end{align*}

Ο αριθμός των θεατών που κάθονται στις πρώτες 10 θέσεις είναι:
$S_{10} - S_{10}^\prime = 250 - 195 = 55$

Αυτός είναι και ο συνολικός αριθμός θεατών, αφού από την $11$η σειρά και μετά όλα τα καθίσματα είναι κενά.
\end{rlist}
\end{alist}
