\lysh{13179}{4}

\begin{alist}
\item Η απόσταση $d(\alpha, \beta) = |\alpha - \beta|$ των αριθμών $\alpha$ και $\beta$ πάνω στον άξονα των πραγματικών αριθμών φαίνεται στο παρακάτω σχήμα.

\begin{rlist}
\item Από τον άξονα και αφού ο $\beta$ δεν μπορεί να πάρει τιμή μικρότερη (πιο αριστερά) του $1$ και ο $\alpha$ δεν μπορεί να πάρει τιμή μεγαλύτερη (πιο δεξιά) του $4$, συμπεραίνουμε ότι η μεγαλύτερη τιμή της $d(\alpha, \beta)$ είναι $3$ (μάλιστα $d(\alpha, \beta) = 3$ όταν $\beta = 1$ και $\alpha = 4$), δηλαδή $d(\alpha, \beta) \le 3$.
\item Είναι $1 \le \beta \le 2 \Leftrightarrow -1 \ge -\beta \ge -2 \Leftrightarrow -2 \le -\beta \le -1$.
Με πρόσθεση κατά μέλη των $2 \le \alpha \le 4$, $-2 \le -\beta \le -1$ έχουμε ότι
\[2 - 2 \le \alpha - \beta \le 4 - 1 \Leftrightarrow 0 \le \alpha - \beta \le 3.\]
Αφού $0 \le \alpha - \beta$ είναι $|\alpha - \beta| = \alpha - \beta \le 3$ οπότε $d(\alpha, \beta) \le 3$.
\end{rlist}

\item 
\begin{rlist}
\item Δεδομένου ότι $1 \le \beta \le 2$ και $2 \le \alpha \le 4$ έχουμε ότι οι αριθμοί $\alpha, \beta$ είναι θετικοί και αφού $\beta \le 2 \le \alpha$ είναι και $\beta \le \alpha$.
Έτσι αφού $\beta \le \alpha$ και $\beta > 0$ έχουμε ότι $\dfrac{\beta}{\beta} \le \dfrac{\alpha}{\beta}$ δηλαδή $1 \le \dfrac{\alpha}{\beta}$.
Επίσης αφού $\beta \le \alpha$ και $\alpha > 0$ έχουμε ότι $\dfrac{\beta}{\alpha} \le \dfrac{\alpha}{\alpha}$ δηλαδή $\dfrac{\beta}{\alpha} \le 1$.
Τελικά $\dfrac{\beta}{\alpha} \le 1 \le \dfrac{\alpha}{\beta}$.
\item Αφού δείξαμε ότι $\dfrac{\beta}{\alpha} \le 1 \le \dfrac{\alpha}{\beta}$ έχουμε ότι $1 - \dfrac{\beta}{\alpha} \ge 0$ οπότε $\left|1 - \dfrac{\beta}{\alpha}\right| = 1 - \dfrac{\beta}{\alpha}$ και
$\dfrac{\alpha}{\beta} - 1 \ge 0$ οπότε $\left|\dfrac{\alpha}{\beta} - 1\right| = \dfrac{\alpha}{\beta} - 1$.
Συνεπώς
\begin{align*}
\left|1 - \dfrac{\beta}{\alpha}\right| = \left|\dfrac{\alpha}{\beta} - 1\right| \ &\Leftrightarrow \\
1 - \dfrac{\beta}{\alpha} = \dfrac{\alpha}{\beta} - 1 \ &\Leftrightarrow \\
\alpha\beta - \alpha\beta \cdot \dfrac{\beta}{\alpha} = \alpha\beta \cdot \dfrac{\alpha}{\beta} - 1\alpha\beta \ &\Leftrightarrow \\
\alpha\beta - \beta^2 = \alpha^2 - \alpha\beta \ &\Leftrightarrow \\
0 = \alpha^2 - 2\alpha\beta + \beta^2 \ &\Leftrightarrow \\
0 &= (\alpha - \beta)^2 \Leftrightarrow \alpha = \beta.
\end{align*}
Όμως $1 \le \beta \le 2 \le \alpha \le 4$, οπότε για να είναι $\alpha = \beta$ θα πρέπει $\alpha = \beta = 2$.
\end{rlist}
\end{alist}
