\lysh{14412}{2}

\begin{alist}
\item Έχουμε $\alpha > \beta$ και ισοδύναμα $\alpha - \beta > 0$, οπότε $|\alpha - \beta| = \alpha - \beta$.
Επίσης $\alpha > 1$ ισοδύναμα $1 - \alpha < 0$, οπότε $|1 - \alpha| = \alpha - 1$.
Άρα $\dfrac{|\alpha - \beta|}{\alpha - \beta} - \dfrac{|1 - \alpha|}{1 - \alpha} = \dfrac{\alpha - \beta}{\alpha - \beta} - \dfrac{\alpha - 1}{1 - \alpha} = 1 + \dfrac{\alpha - 1}{\alpha - 1} = 1 + 1 = 2$.

\item Είναι $\alpha > 1$ και $\beta > 1$, οπότε $\alpha + \beta > 1 + 1$, δηλαδή $\alpha + \beta > 2$
και από το $\alpha)$ ερώτημα έχουμε ότι $\dfrac{|\alpha - \beta|}{\alpha - \beta} - \dfrac{|1 - \alpha|}{1 - \alpha} = 2$.
Επομένως $\alpha + \beta > \dfrac{|\alpha - \beta|}{\alpha - \beta} - \dfrac{|1 - \alpha|}{1 - \alpha}$.
\end{alist}
