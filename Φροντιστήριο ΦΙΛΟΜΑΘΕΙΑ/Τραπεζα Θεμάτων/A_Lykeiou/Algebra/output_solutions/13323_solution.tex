\lysh{13323}{2}

\begin{alist}
\item Για να αποδείξουμε την ζητούμενη ισότητα, θα χρησιμοποιήσουμε ευθεία απόδειξη.
Έχουμε λοιπόν:
\[(x - 1)^2 + (y + 4)^2 = (x^2 - 2x + 1) + (y^2 + 8y + 16) = x^2 + y^2 - 2x + 8y + 17.\]

\item Έχουμε διαδοχικά:
\begin{align*}
x^2 + y^2 - 2x + 8y + 17 &= 0 \Leftrightarrow (\alpha) \\
(x - 1)^2 + (y + 4)^2 = 0 \ &\Leftrightarrow \\
(x - 1)^2 &= 0 \text{ και } (y + 4)^2 = 0\text{, δηλαδή} \\
x - 1 &= 0 \text{ και } y + 4 = 0\text{, οπότε τελικά} \\
x &= 1 \text{ και } y = -4.
\end{align*}
\end{alist}
