\lysh{14295}{2}

\begin{alist}
\item Το ζητούμενο της ανίσωσης $|x - 1| \ge 6$ διατυπώνεται γεωμετρικά ως εξής: Να βρείτε τους πραγματικούς αριθμούς $x$ οι οποίοι απέχουν από το $1$ απόσταση μεγαλύτερη ή ίση του $6$.
Η σωστή αναπαράσταση είναι η $4$η, διότι:
\begin{center}
\begin{tikzpicture}
\draw[->] (-7, 0) -- (9, 0);
\draw[thick] (-7, 0) -- (-5, 0);
\draw[thick] (7, 0) -- (9, 0);
\fill (-5, 0) circle (2pt);
\fill (7, 0) circle (2pt);
\fill (1, 0) circle (2pt);
\node[below] at (-5, -0.1) {$-5$};
\node[below] at (7, -0.1) {$7$};
\node[below] at (1, -0.1) {$1$};
\draw[<->] (-5, 0.3) -- (1, 0.3) node[midway, above] {$6$};
\draw[<->] (1, 0.3) -- (7, 0.3) node[midway, above] {$6$};
\end{tikzpicture}
\end{center}

\item Θα λύσουμε την ανίσωση αλγεβρικά, οπότε έχουμε διαδοχικά:
$|x - 1| \ge 6$, δηλαδή
$x - 1 \le -6$ ή $x - 1 \ge 6$, οπότε
$x \le -5$ ή $x \ge 7$

Δηλαδή το σύνολο λύσεων της ανίσωσης είναι το $(-\infty, -5] \cup [7, +\infty)$ που είναι πράγματι η αναπαράσταση $4$.
\end{alist}
