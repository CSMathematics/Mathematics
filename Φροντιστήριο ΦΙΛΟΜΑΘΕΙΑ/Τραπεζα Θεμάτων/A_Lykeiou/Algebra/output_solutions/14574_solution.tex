\lysh{14574}{2}

\begin{alist}
\item Έχουμε $\alpha_1 = 2$ και $\alpha_3 = 8$, δηλαδή
$\alpha_1 + 2\omega = 8$, οπότε
$2 + 2\omega = 8$ και τελικά $\omega = 3$.

\item Θα πρέπει να βρούμε τον φυσικό αριθμό $\nu$, ώστε:
\begin{gather*}
\alpha_\nu = 35\text{, δηλαδή} \\
\alpha_1 + (\nu - 1)\omega = 35\text{, οπότε} \\
2 + (\nu - 1)3 = 35\text{, οπότε} \\
3(\nu - 1) = 33\text{, δηλαδή}
\end{gather*}
$\nu - 1 = 11$ και τελικά $\nu = 12$.
Επομένως ο $12$ος όρος της προόδου είναι $35$.
\end{alist}
