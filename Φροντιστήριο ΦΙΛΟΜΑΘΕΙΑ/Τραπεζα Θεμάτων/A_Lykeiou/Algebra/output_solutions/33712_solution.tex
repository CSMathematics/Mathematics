\lysh{33712}{4}

\begin{alist}
\item Το τριώνυμο έχει διακρίνουσα:
\[\Delta = \beta^2 - 4 \cdot 1 \cdot \beta^2 = -3\beta^2\]

\item 
\begin{rlist}
\item Για $\beta \ne 0$ ισχύει ότι:
$\Delta = -3\beta^2 < 0$.
Επειδή ο συντελεστής του $x^2$ είναι $1 > 0$, το τριώνυμο είναι θετικό για κάθε $x \in \mathbb{R}$.

\item Για $\beta = 0$ είναι $\Delta = 0$, οπότε το τριώνυμο είναι θετικό για κάθε $x \in \mathbb{R} - \{0\}$, αφού
για $x = 0$ μηδενίζεται.
\end{rlist}

\item Το τριώνυμο $\alpha^2 + \alpha\beta + \beta^2$ προκύπτει από το αρχικό τριώνυμο για $x = \alpha$. Διακρίνουμε τις
περιπτώσεις:

$1^\text{η}$ Περίπτωση:
Αν $\beta \ne 0$ τότε από το $(\beta \text{i})$ συμπεραίνουμε ότι:
$\alpha^2 + \alpha\beta + \beta^2 > 0$.

$2^\text{η}$ Περίπτωση:
Αν $\beta = 0$ (οπότε $\alpha \ne 0$), από το $(\beta \text{ii})$ συμπεραίνουμε ότι:
$\alpha^2 + \alpha\beta + \beta^2 > 0$.
\end{alist}
