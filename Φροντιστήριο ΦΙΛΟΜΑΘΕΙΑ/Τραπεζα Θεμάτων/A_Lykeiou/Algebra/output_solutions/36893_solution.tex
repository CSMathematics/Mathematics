\lysh{36893}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{align*}
|2x - 1| \le 7 \ &\Leftrightarrow \\
-7 &\le 2x - 1 \le 7\text{, οπότε} \\
-6 &\le 2x \le 8\text{και τελικά} \\
-3 &\le x \le 4.
\end{align*}

\item Έχουμε ισοδύναμα:
\begin{gather*}
|x - 1| > 2\text{, οπότε} \\
x - 1 < -2 \text{ ή } x - 1 > 2\text{και τελικά} \\
x < -1 \text{ ή } x > 3.
\end{gather*}

\item Στον άξονα των πραγματικών αριθμών βρίσκουμε τις κοινές λύσεις των ανισώσεων των ερωτημάτων $\alpha)$ και $\beta)$, όπως φαίνεται στο παρακάτω σχήμα:

Άρα οι κοινές λύσεις των ανισώσεων είναι οι πραγματικοί αριθμοί $x$ για τους οποίους ισχύει: $x \in [-3, -1) \cup (3, 4]$.
\end{alist}
