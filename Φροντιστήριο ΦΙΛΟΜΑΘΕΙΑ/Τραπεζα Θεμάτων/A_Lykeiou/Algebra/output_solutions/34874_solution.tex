\lysh{34874}{2}

\begin{alist}
\item Το τριώνυμο $2x^2 - 5x + 2$ έχει διακρίνουσα $\Delta = (-5)^2 - 4 \cdot 2 \cdot 2 = 25 - 16 = 9 > 0$. Οι ρίζες
της εξίσωσης (1) είναι:
\begin{gather*}
x_1 = \dfrac{-(-5) - \sqrt{9}}{2 \cdot 2} = \dfrac{5 - 3}{4} = \dfrac{2}{4} = \dfrac{1}{2}, \\
x_2 = \dfrac{-(-5) + \sqrt{9}}{2 \cdot 2} = \dfrac{5 + 3}{4} = \dfrac{8}{4} = 2.
\end{gather*}

\item Δεδομένου ότι $x_1 < x_2$ είναι οι ρίζες της εξίσωσης (1), οι αριθμοί $x_1, 1, x_2$, δηλαδή οι αριθμοί
$\dfrac{1}{2}, 1, 2$, με τη σειρά που δίνονται είναι διαδοχικοί όροι γεωμετρικής προόδου αν και μόνο αν
$1^2 = \dfrac{1}{2} \cdot 2$, που ισχύει.
\end{alist}
