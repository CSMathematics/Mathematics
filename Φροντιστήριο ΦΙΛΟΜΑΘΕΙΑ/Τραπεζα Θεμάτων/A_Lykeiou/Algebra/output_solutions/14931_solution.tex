\begin{alist}
\item Έχουμε $A = \alpha^2 - \beta^2 = (\alpha - \beta)(\alpha + \beta) = (1 + \sqrt{2} - 1 + \sqrt{2}) \cdot (1 + \sqrt{2} + 1 - \sqrt{2}) = 2\sqrt{2} \cdot 2 = 4\sqrt{2}$.

\item Έχουμε $B = \sqrt{\alpha^2} - \sqrt{\beta^2} = |\alpha| - |\beta| = |1 + \sqrt{2}| - |1 - \sqrt{2}|$ (ή $\sqrt{2}-1$ εξαρτάται πως ορίζονται τα $\alpha, \beta$, 
στο αρχείο του OCR γράφει $B = \sqrt{\dots}$.  Ας πούμε με βάση το (1) $\alpha=1+\sqrt{2}$, $\beta=1-\sqrt{2}$).
\[B = |1 + \sqrt{2}| - |1 - \sqrt{2}| = (1 + \sqrt{2}) - (\sqrt{2} - 1) = 1 + \sqrt{2} - \sqrt{2} + 1 = 2.\]

\item Έχουμε ισοδύναμα:
\[\sqrt{\alpha^2 - \beta^2} > \sqrt{\alpha^2} - \sqrt{\beta^2} \Leftrightarrow \sqrt{A} > B \Leftrightarrow A > B^2 \Leftrightarrow 4\sqrt{2} > 2^2 \Leftrightarrow 4\sqrt{2} > 4 \Leftrightarrow \sqrt{2} > 1,\]
που ισχύει.
\end{alist}
