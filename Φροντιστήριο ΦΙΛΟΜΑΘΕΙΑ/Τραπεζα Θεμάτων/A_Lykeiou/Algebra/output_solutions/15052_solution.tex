\lysh{15052}{4}

\begin{alist}
\item Με $x, y \in (0, 6)$ είναι:
\begin{gather*}
E_1 = xy, \\
E_2 = (6 - x)y = 6y - yx, \\
E_3 = (6 - x)(6 - y) = 36 - 6x - 6y + xy, \\
E_4 = x(6 - y) = 6x - xy.
\end{gather*}

\item Με $x = 4, y = 2$ έχουμε:
\begin{gather*}
E_1 = x \cdot y = 4 \cdot 2 = 8 \\
E_2 = (6 - x)y = 2 \cdot 2 = 4 \\
E_3 = (6 - x)(6 - y) = 2 \cdot 4 = 8 \\
E_4 = x(6 - y) = 4 \cdot 4 = 16
\end{gather*}

\item 
\begin{rlist}
\item Από την ισότητα $E_1 + E_3 = E_2 + E_4$ παίρνουμε
\[xy + 36 - 6x - 6y + xy = 2xy - 6x - 6y + 36\]
\begin{align*}
E_2+E_4 &= 6y - xy + 6x - xy = -2xy + 6x + 6y \\
2xy - 6x - 6y + 36 &= -2xy + 6x + 6y \Leftrightarrow 4xy - 12x - 12y + 36 = 0 \Leftrightarrow xy - 3x - 3y + 9 = 0.
\end{align*}
Οπότε: $4xy + 36 = 12x + 12y$, που γράφεται $xy + 9 = 3(x + y)$ και είναι η ζητούμενη ισότητα.

\item Από την ισότητα του ερωτήματος $(\gamma)$i. έχουμε:
$xy - 3x + 9 - 3y = 0$, οπότε $x(y - 3) - 3(y - 3) = 0$, δηλαδή $(y - 3)(x - 3) = 0$, άρα $y = 3$ ή $x = 3$.

\begin{itemize}
\item Αν $y = 3$, τότε $(KH) = 3$, οπότε το τμήμα $EZ$ διέρχεται από το $O$.
\item Αν $x = 3$, τότε $(KZ) = 3$, οπότε το τμήμα $\Theta H$ διέρχεται από το $O$.
\end{itemize}
Επομένως, σε κάθε περίπτωση, τουλάχιστον ένα από τα τμήματα $EZ$ και $H\Theta$ διέρχεται από το
κέντρο $O$ του τετραγώνου.
\end{rlist}
\end{alist}
