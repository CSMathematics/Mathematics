\lysh{33597}{4}

\begin{alist}
\item Από το σχήμα βλέπουμε ότι:
\begin{rlist}
\item τα σημεία τομής των $C_f$ και $C_g$ είναι τα $A(1, 1)$ και $B(3, 1)$.
\item η $C_f$ είναι κάτω από την $C_g$ για $x \in (1, 3)$.
\end{rlist}

\item
\begin{rlist}
\item Οι τετμημένες των σημείων τομής των $C_f$ και $C_g$ είναι οι λύσεις της εξίσωσης
\[f(x) = g(x) \Leftrightarrow |x - 2| = 1.\]

Έχουμε ισοδύναμα:
\begin{gather*}
|x - 2| = 1\text{, δηλαδή} \\
x - 2 = -1 \text{ ή } x - 2 = 1\text{και τελικά} \\
x = 1 \text{ ή } x = 3.
\end{gather*}

Έχουμε επίσης $f(1) = |1 - 2| = 1 = g(1)$ και $f(3) = |3 - 2| = 1 = g(3)$, οπότε τα κοινά σημεία
των δυο γραφικών παραστάσεων είναι τα $A(1, 1)$ και $B(3, 1)$.

\item Οι τιμές του $x \in \mathbb{R}$ για τις οποίες η $C_f$ είναι κάτω από την $C_g$ είναι λύσεις της
ανίσωσης $f(x) < g(x) \Leftrightarrow |x - 2| < 1$.

Έχουμε ισοδύναμα:
$|x - 2| < 1 \Leftrightarrow -1 < x - 2 < 1 \Leftrightarrow 2 - 1 < x < 2 + 1 \Leftrightarrow 1 < x < 3$, δηλαδή η $C_f$ είναι κάτω από
την $C_g$ για $x \in (1, 3)$.
\end{rlist}

\item Η παράσταση $A$ ορίζεται στους πραγματικούς αριθμούς αν και μόνο αν:
\[\begin{cases} 1 - f(x) \ge 0 \\ f(x) \ne 0 \end{cases} \Leftrightarrow \begin{cases} f(x) \le 1 \\ f(x) \ne 0 \end{cases} \Leftrightarrow \begin{cases} |x - 2| \le 1 \\ |x - 2| \ne 0 \end{cases} \stackrel{(\beta ii)}{\Leftrightarrow} \begin{cases} 1 \le x \le 3 \\ x \ne 2 \end{cases}.\]

Τελικά, η παράσταση $A$ ορίζεται στους πραγματικούς αριθμούς για $x \in [1, 2) \cup (2, 3]$.
\end{alist}
