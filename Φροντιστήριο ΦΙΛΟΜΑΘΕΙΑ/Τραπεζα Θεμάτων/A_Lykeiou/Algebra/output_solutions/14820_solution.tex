\lysh{14820}{4}

\begin{alist}
\item Ισοδύναμα έχουμε
\begin{rlist}
\item $x^2 + x + 1 \ge \dfrac{3}{4} \Leftrightarrow x^2 + x + 1 - \dfrac{3}{4} \ge 0 \Leftrightarrow x^2 + x + \dfrac{1}{4} \ge 0 \Leftrightarrow \left(x + \dfrac{1}{2}\right)^2 \ge 0$, που ισχύει.
Ως ισότητα ισχύει αν και μόνο αν $x + \dfrac{1}{2} = 0 \Leftrightarrow x = -\dfrac{1}{2}$.

\item $x^2 - x + 1 \ge \dfrac{3}{4} \Leftrightarrow x^2 - x + 1 - \dfrac{3}{4} \ge 0 \Leftrightarrow x^2 - x + \dfrac{1}{4} \ge 0 \Leftrightarrow \left(x - \dfrac{1}{2}\right)^2 \ge 0$, που ισχύει.
Ως ισότητα ισχύει αν και μόνο αν $x - \dfrac{1}{2} = 0 \Leftrightarrow x = \dfrac{1}{2}$.
\end{rlist}

\item Για $x \ne \dfrac{1}{2}$ και $x \ne -\dfrac{1}{2}$, όπως δείξαμε στο $\alpha)$ ερώτημα, είναι $x^2 + x + 1 > \dfrac{3}{4}$ και
\begin{gather*}
x^2 - x + 1 > \dfrac{3}{4}\text{, οπότε με πολλαπλασιασμό κατά μέλη έχουμε} \\
(x^2 + x + 1)(x^2 - x + 1) > \dfrac{9}{16}.
\end{gather*}

Επίσης, για $x = \dfrac{1}{2}$ είναι $x^2 + x + 1 > \dfrac{3}{4}$ και $x^2 - x + 1 = \dfrac{3}{4}$, οπότε με πολλαπλασιασμό
κατά μέλη έχουμε $(x^2 + x + 1)(x^2 - x + 1) > \dfrac{9}{16}$.

Τέλος, για $x = -\dfrac{1}{2}$ είναι $x^2 + x + 1 = \dfrac{3}{4}$ και $x^2 - x + 1 > \dfrac{3}{4}$, οπότε με πολλαπλασιασμό
κατά μέλη έχουμε $(x^2 + x + 1)(x^2 - x + 1) > \dfrac{9}{16}$.

Επομένως $(x^2 + x + 1)(x^2 - x + 1) > \dfrac{9}{16}$ για κάθε $x \in \mathbb{R}$.

\item 
\begin{rlist}
\item Η παράσταση $A$ ορίζεται για κάθε πραγματική τιμή του $x$ για την οποία
ισχύει $x^2 \ne 1 \Leftrightarrow x \ne \pm 1$.
\item Είναι
\[A = \dfrac{x^6 - 1}{x^2 - 1} = \dfrac{(x^3 - 1)(x^3 + 1)}{(x - 1)(x + 1)} = \dfrac{(x - 1)(x^2 + x + 1)(x + 1)(x^2 - x + 1)}{(x - 1)(x + 1)} = (x^2 + x + 1)(x^2 - x + 1)\]
για κάθε $x \in \mathbb{R} - \{1, -1\}$.

Όπως δείξαμε στο $\beta)$ είναι $(x^2 + x + 1)(x^2 - x + 1) > \dfrac{9}{16}$ για κάθε $x \in \mathbb{R}$, οπότε
$A > \dfrac{9}{16}$ για κάθε $x \in \mathbb{R} - \{1, -1\}$ και επομένως η παράσταση $A$ δεν μπορεί να
πάρει την τιμή $\dfrac{9}{16}$.

Εναλλακτικά, θα εξετάσουμε αν η εξίσωση
$\dfrac{x^6 - 1}{x^2 - 1} = \dfrac{9}{16}$ έχει λύση στο $\mathbb{R} - \{1, -1\}$. Είναι ισοδύναμα
\begin{align*}
\dfrac{x^6 - 1}{x^2 - 1} = \dfrac{9}{16} &\Leftrightarrow \dfrac{(x^2)^3 - 1}{x^2 - 1} = \dfrac{9}{16} \Leftrightarrow \dfrac{(x^2 - 1)(x^4 + x^2 + 1)}{x^2 - 1} = \dfrac{9}{16}\\& \Leftrightarrow x^4 + x^2 + 1 = \dfrac{9}{16} \Leftrightarrow 16(x^4 + x^2 + 1) = 9 \ \Leftrightarrow \\
&\Leftrightarrow 16x^4 + 16x^2 + 16 - 9 = 0 \Leftrightarrow 16x^4 + 16x^2 + 7 = 0
\end{align*}
και επειδή $16x^4 + 16x^2 + 7 \ge 7 > 0$, η εξίσωση είναι αδύνατη και επομένως η
παράσταση $A$ δεν μπορεί να πάρει την τιμή $\dfrac{9}{16}$.
\end{rlist}
\end{alist}
