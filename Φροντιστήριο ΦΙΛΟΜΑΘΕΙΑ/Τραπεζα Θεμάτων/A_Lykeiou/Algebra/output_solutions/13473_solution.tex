\item 
\begin{rlist}
\item Από $1 \le x \le 3$ έχουμε $|x - 3| = -x + 3$ και $|x - 1| = x - 1$,
τότε $A = -x + 3 + x - 1 = 2$.
\item $A = 2 \Leftrightarrow -x + 3 - x + 1 = 2 \Leftrightarrow -2x + 4 = 2 \Leftrightarrow x = 1$.
\end{rlist}
