\begin{alist}
\item Έχουμε:
\[A = (\sqrt[6]{3})^6 = \left( (\sqrt[6]{3})^2 \right)^3 = \left( \sqrt[3]{3} \right)^3 = 3^1 = 3\text{και προφανώς}\]
το $(\sqrt{3})^6 = ((\sqrt{3})^2)^3 = 3^3 = 27$.
Επίσης
\[B = (\sqrt[3]{3})^6 = ((\sqrt[3]{3})^3)^2 = 3^2 = 9.\]

Άρα: $A - B = 27 - 9 = 18$.
\begin{gather*}
A = (\sqrt{3})^6 = \left( (\sqrt{3})^2 \right)^3 = 3^3 = 27 \\
B = (\sqrt[3]{3})^6 = \left( (\sqrt[3]{3})^3 \right)^2 = 3^2 = 9
\end{gather*}
Άρα $A - B = 27 - 9 = 18$

\item Από το $\alpha)$ ερώτημα, $A - B = 18 > 0$, οπότε
\begin{gather*}
A > B\text{, δηλαδή} \\
(\sqrt{3})^6 > (\sqrt[3]{3})^6\text{και τελικά} \\
\sqrt{3} > \sqrt[3]{3}.
\end{gather*}
\end{alist}
