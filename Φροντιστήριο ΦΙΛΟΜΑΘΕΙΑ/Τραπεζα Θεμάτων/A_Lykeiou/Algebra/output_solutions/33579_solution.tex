\lysh{33579}{4}

\begin{alist}
\item Οι αριθμοί: $x^2 + 5, x^2 + x, 2x + 4$, με τη σειρά που δίνονται, είναι διαδοχικοί όροι
αριθμητικής προόδου, οπότε ισχύει η σχέση: $2(x^2 + x) = (2x + 4) + (x^2 + 5)$.

Έχουμε ισοδύναμα:
\begin{gather*}
2x^2 + 2x = x^2 + 2x + 9\text{, δηλαδή} \\
x^2 = 9\text{και τελικά} \\
x = 3 \text{ ή } x = -3
\end{gather*}

\item 
\begin{rlist}
\item Αν ο αριθμός $x^2 + 5$ είναι ο $4^\text{ος}$ όρος της προόδου, τότε ο $x^2 + x$ θα είναι ο $5^\text{ος}$ όρος
της. Άρα για $x = 3$, $\alpha_4 = 3^2 + 5 = 14$ και $\alpha_5 = 3^2 + 3 = 12$. Οπότε $\omega = \alpha_5 - \alpha_4 = 12 - 14 = -2$

\item Ισχύει $\alpha_4 = \alpha_1 + 3\omega$, δηλαδή $14 = \alpha_1 + 3 \cdot (-2)$, οπότε $\alpha_1 = 20$.

\item Το ζητούμενο άθροισμα είναι:
\begin{gather*}
S = \alpha_{15} + \alpha_{16} + \alpha_{17} + \dots + \alpha_{24} = (\alpha_1 + \alpha_2 + \dots + \alpha_{24}) - (\alpha_1 + \alpha_2 + \dots + \alpha_{14}) = \\
= S_{24} - S_{14} = \dfrac{24}{2} \cdot [2 \cdot 20 + 23 \cdot (-2)] - \dfrac{14}{2} \cdot [2 \cdot 20 + 13 \cdot (-2)] = \\
12 \cdot (40 - 46) - 7 \cdot (40 - 26) = -170.
\end{gather*}
\end{rlist}
\end{alist}
