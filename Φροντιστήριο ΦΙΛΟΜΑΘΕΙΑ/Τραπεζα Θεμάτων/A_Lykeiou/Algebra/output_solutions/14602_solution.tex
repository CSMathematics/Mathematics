\lysh{14602}{3}

\begin{alist}
\item Έχουμε:
\[0 < \alpha < 1 \stackrel{\cdot \alpha^2 > 0}{\Leftrightarrow} 0 < \alpha^3 < \alpha^2 \quad (1)\]
και
\[0 < \alpha < 1 \stackrel{\cdot \alpha > 0}{\Leftrightarrow} 0 < \alpha^2 < \alpha \quad (2)\]
Από $(1)$ και $(2)$ έχουμε $0 < \alpha^3 < \alpha$.

\item Από το δεδομένο και το $\alpha)$ ερώτημα έχουμε $0 < \alpha^3 < \alpha < 1$. Επιπλέον επειδή ο $\alpha$ 
είναι θετικός αριθμός, ομόσημος του $1$ δηλαδή, έχουμε ισοδύναμα:
\[\alpha < 1 \Leftrightarrow \dfrac{1}{\alpha} > 1.\]
Τελικά $0 < \alpha^3 < \alpha < 1 < \dfrac{1}{\alpha}$.
\end{alist}
