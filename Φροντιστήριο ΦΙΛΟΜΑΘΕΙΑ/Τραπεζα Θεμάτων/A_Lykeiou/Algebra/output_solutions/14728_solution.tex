\lysh{14728}{2}

\begin{alist}
\item Είναι: $-1 < 0$, άρα: $f(-1) = 2 \cdot (-1) - 1 = -2 - 1 = -3$.
Είναι: $1 > 0$, άρα: $f(1) = 1^2 + 1 = 2$.

\item Αφού $x \ge 0$ τότε: $f(x) \ge 2 \Leftrightarrow x^2 + 1 \ge 2 \Leftrightarrow x^2 \ge 1 \Leftrightarrow \sqrt{x^2} \ge 1 \Leftrightarrow |x| \ge 1$.
Επομένως: $x \le -1$ ή $x \ge 1$ και $x \ge 0$.
Άρα $x \ge 1$.
\end{alist}
