\lysh{36894}{2}

\begin{alist}
\item Έχουμε ισοδύναμα:
\begin{align*}
\alpha + \dfrac{1}{\alpha} &\le -2 \overset{\alpha < 0}{\Leftrightarrow}
\alpha^2 + \alpha \cdot \dfrac{1}{\alpha} \ge -2\alpha\text{, δηλαδή} \\
\alpha^2 + 1 + 2\alpha &\ge 0\text{, οπότε} (\alpha + 1)^2 \ge 0
\end{align*}
που ισχύει.

\item Έχουμε ισοδύναμα (επειδή $\alpha < 0 \Rightarrow |\alpha| = -\alpha$):
\begin{align*}
|\alpha| + \dfrac{1}{|\alpha|} \ge 2 \ &\Leftrightarrow \\
-\alpha - \dfrac{1}{\alpha} &\ge 2\text{, οπότε }\alpha + \dfrac{1}{\alpha} \le -2
\end{align*}
που από το $\alpha)$ ερώτημα ισχύει για κάθε $\alpha < 0$.
\end{alist}
