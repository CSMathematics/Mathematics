\lysh{12922}{2}

\begin{alist}
\item Για να είναι η παράσταση $A = 0$, πρέπει $\alpha^2 + \beta^2 = 0$ που ισχύει αν και μόνο αν $\alpha = 0$ και $\beta = 0$.

\item Είναι: $A - B = \alpha^2 + \beta^2 - 2\alpha\beta = (\alpha - \beta)^2 \ge 0$, ισχύει σαν τετράγωνο αριθμού.

\item $A - B = 0 \Leftrightarrow (\alpha - \beta)^2 = 0 \Leftrightarrow \alpha - \beta = 0 \Leftrightarrow \alpha = \beta$.
\end{alist}
