\lysh{38919}{4}

\begin{alist}
\item
\begin{rlist}
\item Από τη γραφική παράσταση της συνάρτησης παρατηρούμε ότι η συγκέντρωση είναι $0,1\text{ mg/L}$ μετά από $0,35$ ώρες, δηλαδή μετά από $21$ λεπτά ($0,35 \cdot 60 = 21$).
\item H ακριβής συγκέντρωση στο αίμα $4$ ώρες μετά τη λήψη είναι:
\[f(4) = \dfrac{7,8 \cdot 4^2 + 6 \cdot 4}{2 \cdot 4^4 + 30} = \dfrac{124,8 + 24}{512 + 30} = \dfrac{148,8}{542} \simeq 0,275\text{ mg/L}.\]
\end{rlist}

\item
\begin{rlist}
\item Έχουμε $f(10) = \dfrac{7,8 \cdot 10^2 + 6 \cdot 10}{2 \cdot 10^4 + 30} = \dfrac{780 + 60}{20030} = \dfrac{840}{20030} \simeq 0,04 < 0,1\text{ mg/L}$, άρα έχει δίκιο ο φαρμακοποιός.
Εναλλακτικά, από τη γραφική παράσταση βλέπουμε ότι το σημείο με τετμημένη $x = 10$ έχει τεταγμένη μικρότερη από $0,1$. Άρα έχει δίκιο ο φαρμακοποιός.
\item Από τη γραφική παράσταση της συνάρτησης, παρατηρούμε ότι η συγκέντρωση είναι από $0,1\text{ mg/L}$ και πάνω, σε διάστημα $6,22$ ωρών ($6,57 - 0,35 = 6,22$). Άρα το φάρμακο θα είναι αποτελεσματικό για $6$ ώρες και $13$ λεπτά περίπου.
\end{rlist}
\end{alist}
