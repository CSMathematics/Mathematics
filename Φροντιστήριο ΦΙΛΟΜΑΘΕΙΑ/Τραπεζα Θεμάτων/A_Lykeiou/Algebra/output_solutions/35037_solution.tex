\lysh{35037}{2}

\begin{alist}
\item Οι αριθμοί $\kappa - 2, 2\kappa, 7\kappa + 4$ είναι με τη σειρά που δίνονται, διαδοχικοί όροι μιας γεωμετρικής προόδου αν και μόνο αν:
\begin{align*}
(2\kappa)^2 = (\kappa - 2)(7\kappa + 4) \ &\Leftrightarrow \\
4\kappa^2 = 7\kappa^2 + 4\kappa - 14\kappa - 8 \ &\Leftrightarrow 3\kappa^2 - 10\kappa - 8 = 0\ ,\ (1)
\end{align*}
Η εξίσωση (1) έχει διακρίνουσα:
\[\Delta = (-10)^2 - 4 \cdot 3 \cdot (-8) = 100 + 96 = 196 > 0.\]
Άρα η εξίσωση (1) έχει ρίζες τις:
\[\kappa_{1,2} = \dfrac{10 \pm \sqrt{196}}{2 \cdot 3} = \dfrac{10 \pm 14}{6} = \begin{cases} \dfrac{24}{6} = 4 \\ -\dfrac{4}{6} = -\dfrac{2}{3} \end{cases}\]
Η τιμή $\kappa = -\dfrac{2}{3}$ απορρίπτεται, αφού $\kappa \in \mathbb{N}$. Άρα $\kappa = 4$. Για $\kappa = 4$ οι αριθμοί γράφονται $2, 8, 32$ οπότε ο λόγος είναι $\lambda = \dfrac{8}{2} = 4$.
\item
\begin{rlist}
\item Είναι:
\begin{gather*}
a_2 = a_1 \cdot \lambda = 4a_1 \\
a_4 = a_1 \cdot \lambda^3 = 4^3 a_1 = 64a_1 \\
a_5 = a_1 \cdot \lambda^4 = 4^4 a_1 = 256a_1
\end{gather*}

\item Ισχύει ότι:
\begin{gather*}
a_2 + a_5 = 4a_1 + 256a_1 = 260a_1\text{, ενώ} \\
4(a_1 + a_4) = 4(a_1 + 64a_1) = 4 \cdot 65a_1 = 260a_1.
\end{gather*}
Άρα $a_2 + a_5 = 4(a_1 + a_4)$.
\end{rlist}
\end{alist}
