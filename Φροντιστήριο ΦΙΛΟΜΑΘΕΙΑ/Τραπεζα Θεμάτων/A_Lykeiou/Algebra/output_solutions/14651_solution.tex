\lysh{14651}{4}

\begin{alist}
\item Από τους τύπους για το άθροισμα και γινόμενο των ριζών εξίσωσης 2ου βαθμού έχουμε:
\begin{gather*}
S = x_1 + x_2 = -\dfrac{\beta}{\alpha} = -\dfrac{-4\left(\lambda + \dfrac{1}{\lambda}\right)}{1} = 4\left(\lambda + \dfrac{1}{\lambda}\right)\text{και} \\
P = x_1 \cdot x_2 = \dfrac{\gamma}{\alpha} = \dfrac{16}{1} = 16.
\end{gather*}
Αφού $P = x_1 \cdot x_2 = 16 > 0$ και $S = x_1 + x_2 = 4\left(\lambda + \dfrac{1}{\lambda}\right) > 0$ για κάθε $\lambda > 0$, οι ρίζες $x_1, x_2$ είναι 
θετικές και άρα μπορούν να αποτελούν πλευρές ορθογωνίου.

\begin{rlist}
\item Η περίμετρος του ορθογωνίου είναι: 
\[P_0 = 2x_1 + 2x_2 = 2(x_1 + x_2) = 2 \cdot 4\left(\lambda + \dfrac{1}{\lambda}\right) = 8\left(\lambda + \dfrac{1}{\lambda}\right).\]
\item Το εμβαδόν είναι: $E = x_1 \cdot x_2 = 16$.
\end{rlist}

\item Έχουμε ισοδύναμα:
\[P_0 \ge 16 \Leftrightarrow 8\left(\lambda + \dfrac{1}{\lambda}\right) \ge 16 \Leftrightarrow \lambda + \dfrac{1}{\lambda} \ge 2 \Leftrightarrow \lambda + \dfrac{1}{\lambda} - 2 \ge 0 \Leftrightarrow \dfrac{\lambda^2 - 2\lambda + 1}{\lambda} \ge 0 \Leftrightarrow \dfrac{(\lambda - 1)^2}{\lambda} \ge 0,\]
που ισχύει για κάθε $\lambda > 0$.

\item Ισοδύναμα βρίσκουμε:
\[P_0 = 16 \Leftrightarrow \dfrac{(\lambda - 1)^2}{\lambda} = 0 \Leftrightarrow (\lambda - 1)^2 = 0 \Leftrightarrow \lambda = 1.\]
Για $\lambda = 1$ η περίμετρος είναι $16$ (δηλαδή $2(x_1 + x_2) = 16$ και τελικά $x_1 + x_2 = 8$) και το
εμβαδόν είναι $16$ (δηλαδή $x_1 \cdot x_2 = 16$), οπότε $x_1 = x_2 = 4$. Επομένως το ορθογώνιο είναι τετράγωνο.
\end{alist}
