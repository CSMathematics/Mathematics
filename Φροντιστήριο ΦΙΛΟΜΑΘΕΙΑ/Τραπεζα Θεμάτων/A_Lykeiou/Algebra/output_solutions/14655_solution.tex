\lysh{14655}{4}

\begin{alist}
\item Επειδή $x$ και $y$ είναι μήκη πλευρών έχουμε $x > 0$ και $y > 0$ με $x + y = 10 \Leftrightarrow y = 10 - x$.
Από τον τύπο του εμβαδού τριγώνου έχουμε:
$E(x) = \dfrac{1}{2} x \cdot y = \dfrac{1}{2} x \cdot (10 - x) = \dfrac{1}{2}(10x - x^2)$ με $x \in (0, 10)$.

\item 
\begin{rlist}
\item Αρκεί να αποδείξουμε την σχέση $E(x) \le \dfrac{25}{2}$ για κάθε $x \in (0, 10)$, αρκεί:
$\dfrac{1}{2}(10x - x^2) \le \dfrac{25}{2}$, αρκεί $10x - x^2 \le 25$ ή $x^2 - 10x + 25 \ge 0$ ή $(x - 5)^2 \ge 0$ που ισχύει
για $x \in (0, 10)$.

\item To εμβαδόν γίνεται μέγιστο, δηλαδή ίσο με $\dfrac{25}{2}$ για $x = 5$ γιατί
\[E(x) = \dfrac{25}{2} \Leftrightarrow \dfrac{1}{2}(10x - x^2) = \dfrac{25}{2} \Leftrightarrow (x - 5)^2 = 0 \Leftrightarrow x = 5.\]
\end{rlist}

\item Όταν $x = 5$ τότε $y = 5$, άρα το τρίγωνο ορθογώνιο ισοσκελές.
\end{alist}
