\lysh{15000}{2}

\begin{alist}
\item Από το σχήμα παρατηρούμε πως οι ζητούμενες τιμές είναι:
\[f(0) = 0, f(1) = -5, f(3) = -9 \text{ και } f(5) = -5.\]

\item Για να λύσουμε την εξίσωση $f(x) = 0$, βλέπουμε από το σχήμα τα σημεία που η γραφική
παράσταση $C_f$ τέμνει τον άξονα $x'x$, δηλαδή τα σημεία $(0, 0)$ και $(6, 0)$.

Άρα οι λύσεις της εξίσωσης είναι $x_1 = 0 \text{ και } x_2 = 6$.

\item Με τη βοήθεια του σχήματος, οι λύσεις της ανίσωσης $f(x) < 0$ είναι οι τετμημένες των
σημείων που έχουν τεταγμένες $y$ που βρίσκονται κάτω από τον $x'x$, δηλαδή $y < 0$ με $x$
ανάμεσα στο $0$ και το $6$. Προκύπτει $x \in (0, 6)$.
\end{alist}
