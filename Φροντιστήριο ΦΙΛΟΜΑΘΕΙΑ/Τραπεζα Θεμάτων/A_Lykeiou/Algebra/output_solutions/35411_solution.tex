\lysh{35411}{3}

\begin{alist}
\item Οι αριθμοί $4 - x, x, 2$ είναι διαδοχικοί αριθμοί αριθμητικής προόδου αν και μόνο αν:
\begin{align*}
x = \dfrac{4 - x + 2}{2} \ &\Leftrightarrow
\Leftrightarrow 2x = 6 - x \ \Leftrightarrow \\
\Leftrightarrow 3x = 6 &\Leftrightarrow x = 2
\end{align*}

\item Οι αριθμοί $4 - x, x, 2$ είναι διαδοχικοί αριθμοί γεωμετρικής προόδου αν και μόνο αν:
\begin{align*}
x^2 = (4 - x) \cdot 2 \ &\Leftrightarrow \\
\Leftrightarrow x^2 = 8 - 2x \ &\Leftrightarrow x^2 + 2x - 8 = 0
\end{align*}

Η εξίσωση έχει διακρίνουσα:
\[\Delta = \beta^2 - 4\alpha\gamma = 2^2 - 4 \cdot 1 \cdot (-8) = 4 + 32 = 36 > 0\]

Άρα η εξίσωση (1) έχει ρίζες τις:
\[x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2\alpha} = \dfrac{-2 \pm \sqrt{36}}{2 \cdot 1} = \dfrac{-2 \pm 6}{2} = \begin{cases} \dfrac{-2 + 6}{2} = 2 \\ \dfrac{-2 - 6}{2} = -4 \end{cases}\]

\item Από τα ερωτήματα $\alpha)$ και $\beta)$ βρίσκουμε $x = 2$.
\end{alist}
