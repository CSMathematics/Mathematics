\lysh{14745}{4}

\begin{alist}
\item Ζητάμε τις τετμημένες $x$ των σημείων της γραφικής παράστασης των οποίων οι τεταγμένες
$g(x)$ κυμαίνονται από $-2$ έως μηδέν. Παρατηρούμε ότι αυτό συμβαίνει για
$x \in [-3, -2] \cup [0, 2]$.

\item Καθώς $|g(x)| \le 2 \Leftrightarrow -2 \le g(x) \le 2$, ζητάμε τις τετμημένες $x$ των σημείων της γραφικής
παράστασης των οποίων οι τεταγμένες $g(x)$ κυμαίνονται από $-2$ έως $2$. Παρατηρούμε ότι
αυτό συμβαίνει για $x \in [-3, 3]$.

\item 
\begin{rlist}
\item Ελέγχουμε πόσα σημεία της γραφικής παράστασης έχουν τεταγμένη $\dfrac{4}{5}$ και
διαπιστώνουμε ότι αυτά είναι $3$. Βοηθητικά, σχεδιάζουμε την ευθεία με εξίσωση $y = \dfrac{4}{5}$, της
οποίας όλα τα σημεία έχουν τεταγμένη $\dfrac{4}{5}$ (αυτή η ευθεία είναι παράλληλη προς τον άξονα
$x'x$) και παρατηρούμε ότι έχει $3$ κοινά σημεία με την γραφική παράσταση.
Ανάλογα, η εξίσωση $g(x) = -1$ έχει $2$ λύσεις.

\item To πλήθος λύσεων της εξίσωσης $g(x) = k$ για τις διάφορες πραγματικές τιμές της
παραμέτρου $k$, ταυτίζεται με το πλήθος των κοινών σημείων της ευθείας $y = k$ (παράλληλη
προς τον άξονα $x'x$) με την γραφική παράσταση. Αποτυπώνουμε τα αποτελέσματα στον
παρακάτω πίνακα.

\begin{center}
\begin{mytblr}{}
\textbf{Τιμές του $k$} & {Πλήθος λύσεων\\της εξίσωσης $g(x) = k$} \\ 
$k < -1 \text{ ή } k > 1$ & $1$ \\
$k = -1 \text{ ή } k = 1$ & $2$ \\
$-1 < k < 1$ & $3$ \\
\end{mytblr}
\end{center}
\end{rlist}
\end{alist}
