\lysh{37817}{2}

\begin{alist}
\item Για κάθε $x \ne \pm \sqrt{2}$ έχουμε:
\[A = x^4 + \dfrac{x^4 - 4}{x^2 - 2} = x^4 + \dfrac{(x^2 - 2)(x^2 + 2)}{x^2 - 2} = x^4 + (x^2 + 2) = x^4 + x^2 + 2.\]

\item
\begin{rlist}
\item Παρατηρούμε ότι $x^4 \ge 0$, $x^2 \ge 0$ και $2 > 0$, άρα $x^4 + x^2 + 2 > 0$, δηλαδή $A > 0$.
\item Εφόσον $x^4 \ge 0$, $x^2 \ge 0$, για $x = 0$ η παράσταση $A$ παίρνει τη μικρότερη τιμή της που είναι ίση με $2$.
\end{rlist}
\end{alist}
