\lysh{13472}{2}

\begin{alist}
\item 
\begin{rlist}
\item Αν αντικαταστήσουμε τα $\alpha^2, \beta^2$ από τις δοσμένες ισότητες, βρίσκουμε ότι
\[\alpha^2 - \beta^2 = 2\alpha + \beta - (2\beta + \alpha) = \alpha - \beta.\]
\item H τελευταία ισότητα γράφεται $(\alpha - \beta)(\alpha + \beta) = \alpha - \beta$. Αλλά, $\alpha - \beta \ne 0$, αφού οι αριθμοί $\alpha, \beta$ είναι διαφορετικοί μεταξύ τους, οπότε παίρνουμε $\alpha + \beta = 1$, που είναι το ζητούμενο.
\end{rlist}

\item Όπως προηγουμένως, έχουμε:
\[\text{A} = \alpha^2 + \beta^2 = 2\alpha + \beta + 2\beta + \alpha = 3\alpha + 3\beta = 3(\alpha + \beta) = 3 \cdot 1 = 3\]
αφού $\alpha + \beta = 1$.
\end{alist}
