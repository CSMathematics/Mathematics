\lysh{34872}{2}

\begin{alist}
\item Θα λύσουμε την εξίσωση $\kappa x + 3 = 2x$ για $\kappa = 1$, οπότε: $x + 3 = 2x \Leftrightarrow x = 3$.
Για $\kappa = 3$ έχουμε $3x + 3 = 2x \Leftrightarrow x = -3$.

\item Για $\kappa = 2$ η εξίσωση γίνεται $2x + 3 = 2x \Leftrightarrow 0x = 3$, που είναι αδύνατη.
\end{alist}
