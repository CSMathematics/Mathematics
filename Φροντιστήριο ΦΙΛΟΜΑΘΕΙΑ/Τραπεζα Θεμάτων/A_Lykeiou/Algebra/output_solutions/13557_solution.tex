\lysh{13557}{4}

\begin{alist}
\item Οι τετμημένες των σημείων $A, B$ είναι οι λύσεις της εξίσωσης $f(x) = g(x)$, ισοδύναμα $x^2 = x$, δηλαδή $x^2 - x = 0$, οπότε $x(x - 1) = 0$ και τελικά $x = 0$ ή $x = 1$.
Επειδή το σημείο $A$ είναι πιο αριστερά από το σημείο $B$, η τετμημένη του $A$ είναι $0$ και η τετμημένη του $B$ είναι $1$.
Για $x = 0$ είναι $f(0) = 0$, οπότε $A(0,0)$.
Για $x = 1$ είναι $f(1) = 1$, οπότε $B(1,1)$.

\item 
\begin{rlist}
\item Από το σχήμα βλέπουμε ότι η γραφική παράσταση της $f$ είναι κάτω από τη
γραφική παράσταση της $g$, για τις τετμημένες των σημείων της γραφικής παράστασης της $g$ που είναι μεταξύ των $A, B$, δηλαδή για $0 < x < 1$.
\item Η γραφική παράσταση της $f$ είναι κάτω από τη γραφική παράσταση της $g$ για
τις τιμές του $x$ για τις οποίες ισχύει $x^2 < x$. Η ανίσωση $x^2 < x$ ισοδύναμα γίνεται $x^2 - x < 0$ που είναι μία ανίσωση $2$ου βαθμού και η λύση της βασίζεται
στο πρόσημο του τριωνύμου $x^2 - x$, που φαίνεται στον παρακάτω πίνακα.
\begin{center}
\begin{tabular}{|c|ccccccc|}

$x$ & $-\infty$ & & $0$ & & $1$ & & $+\infty$ \\ 
$x^2 - x$ & & $+$ & $0$ & $-$ & $0$ & $+$ & \\ 
\end{tabular}
\end{center}
Συνεπώς πράγματι $x^2 < x$ για $0 < x < 1$.
\end{rlist}

\item Αφού $\left(\dfrac{\alpha}{\beta}\right)^2 < \dfrac{\alpha}{\beta}$ με βάση τα παραπάνω συμπεραίνουμε ότι $0 < \dfrac{\alpha}{\beta} < 1$. Συνεπώς $\dfrac{\alpha}{\beta} < 1$ και ισοδύναμα $\alpha < \beta$.
\end{alist}
