\lysh{37196}{2}

\begin{alist}
\item Πρέπει:
\[x^2 + 4 \ge 0 \text{ και } x - 4 \ge 0 \Leftrightarrow x \in \mathbb{R} \text{ και } x \ge 4 \Leftrightarrow x \ge 4 \Leftrightarrow x \in [4, +\infty).\]

\item Για $x = 4$ είναι:
\[A = \sqrt{4^2 + 4} - \sqrt{4 - 4} = \sqrt{16 + 4} - 0 = \sqrt{20}.\]

Τότε:
\[A^2 - A = (\sqrt{20})^2 - \sqrt{20} = 20 - \sqrt{20} = 20 - \sqrt{4 \cdot 5} = 20 - 2\sqrt{5} = 2(10 - \sqrt{5}).\]
\end{alist}
