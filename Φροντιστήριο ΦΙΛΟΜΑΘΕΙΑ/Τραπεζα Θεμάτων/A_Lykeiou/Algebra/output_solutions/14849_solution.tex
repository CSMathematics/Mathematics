\lysh{14849}{2}

\begin{alist}
\item Καθώς είναι $2 = \sqrt{4}$, αρκεί να αποδείξουμε ότι $\sqrt{4} < \sqrt{5}$, το οποίο όμως είναι αληθές
αφού $4 < 5$.

\item Εφαρμόζοντας την ταυτότητα του τετραγώνου διαφοράς, παίρνουμε
\[(2 - \sqrt{5})^2 = 2^2 - 2 \cdot \sqrt{5} \cdot 2 + (\sqrt{5})^2 = 5 - 4\sqrt{5} + 4 = 9 - 4\sqrt{5}.\]

\item Με χρήση των ερωτημάτων $\alpha)$ και $\beta)$ έχουμε:
$\sqrt{9 - 4\sqrt{5}} = \sqrt{(2 - \sqrt{5})^2} = |2 - \sqrt{5}| = \sqrt{5} - 2$, αφού $2 < \sqrt{5} \Leftrightarrow 2 - \sqrt{5} < 0$.
\end{alist}
