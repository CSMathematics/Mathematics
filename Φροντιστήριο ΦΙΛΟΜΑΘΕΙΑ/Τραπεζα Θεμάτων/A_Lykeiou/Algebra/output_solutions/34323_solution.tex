\begin{alist}
\item Το τριώνυμο $x^2 - x + (\lambda - \lambda^2)$ έχει $\alpha = 1, \beta = -1, \gamma = \lambda - \lambda^2$ και διακρίνουσα
\begin{gather*}
\Delta = \beta^2 - 4\alpha\gamma = (-1)^2 - 4 \cdot 1 \cdot (\lambda - \lambda^2) = \\
= 1 - 4\lambda + 4\lambda^2 = (1 - 2\lambda)^2 \ge 0.
\end{gather*}
Επειδή $\Delta \ge 0$, για κάθε $\lambda \in \mathbb{R}$ το τριώνυμο έχει πραγματικές ρίζες.

\item Το τριώνυμο έχει δύο πραγματικές ρίζες ίσες αν και μόνο αν:
\begin{align*}
\Delta = 0 \Leftrightarrow (1 - 2\lambda)^2 = 0 \ &\Leftrightarrow \\
\Leftrightarrow 1 - 2\lambda &= 0 \Leftrightarrow \lambda = \dfrac{1}{2}.
\end{align*}

\item 
\begin{rlist}
\item Η σχέση $x_1 < x_0 < x_2$ ισοδύναμα γράφεται
$x_1 < \dfrac{x_1 + x_2}{2}$ και $\dfrac{x_1 + x_2}{2} < x_2 \Leftrightarrow$
\begin{align*}
(2x_1 < x_1 + x_2 \text{ και } x_1 + x_2 < 2x_2) \ &\Leftrightarrow \\
x_1 &< x_2,
\end{align*}
που ισχύει από υπόθεση.
