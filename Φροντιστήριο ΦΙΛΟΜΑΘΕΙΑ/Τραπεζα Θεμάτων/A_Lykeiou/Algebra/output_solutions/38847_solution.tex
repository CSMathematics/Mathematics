\lysh{38847}{4}

\begin{alist}
\item
\begin{rlist}
\item Η πραγματική τιμή $R$ της αντίστασης μπορεί να είναι μεγαλύτερη, μικρότερη ή ίση με $680\,\Omega$, αλλά σε κάθε περίπτωση η απόστασή της από τα $680\,\Omega$ του συγκεκριμένου αντιστάτη θα είναι μικρότερη ή ίση του $5\%$ των $680\,\Omega$ δηλαδή των $\dfrac{5}{100} \cdot 680 = 34\,\Omega$.
Η σχέση που περιγράφει τα παραπάνω είναι: $d(R, 680) \le 34$ ή $|R - 680| \le 34$.

\item Από τη σχέση του προηγούμενου ερωτήματος για $R = 715$ διαδοχικά παίρνουμε:
$|715 - 680| \le 34 \Leftrightarrow 35 \le 34$, που είναι αδύνατο. Άρα δεν μπορεί η πραγματική τιμή της αντίστασης να είναι $715\,\Omega$.

\item Από τη σχέση του ερωτήματος (i) διαδοχικά έχουμε:
\begin{align*}
|R - 680| \le 34 \Leftrightarrow -34 \le R - 680 \le 34 \Leftrightarrow 680 - 34 \le R \le 680 + 34 \ &\Leftrightarrow \\
646 &\le R \le 714.
\end{align*}
Άρα η ελάχιστη πραγματική τιμή της αντίστασης είναι $646\,\Omega$ και η μέγιστη $714\,\Omega$.
\end{rlist}

\item
\begin{rlist}
\item Αφού οι τιμές που παρατηρεί κυμαίνονται στο διάστημα $[650, 670]$ μία τιμή που μπορεί να δώσει για τον αντιστάτη μπορεί να είναι η τιμή $660\,\Omega$ που βρίσκεται στη μέση του διαστήματος των τιμών αυτών. Εναλλακτικά, μπορεί να καθοριστεί οποιαδήποτε τιμή στο παραπάνω διάστημα, όπως για παράδειγμα η τιμή $665\,\Omega$.

\item Για την τιμή των $660\,\Omega$: Ζητάμε το ποσοστό $x\%$, ώστε
\[|R - 660| \le \dfrac{x}{100} \cdot 660 \Leftrightarrow -6,6x \le R - 660 \le 6,6x \Leftrightarrow 660 - 6,6x \le R \le 660 + 6,6x.\]

Όμως οι τιμές της αντίστασης $R$ κυμαίνονται στο διάστημα $[650, 670]$.

Άρα $660 - 6,6x = 650 \Leftrightarrow x = \dfrac{10}{6,6} \simeq 1,5$. Επομένως ο κατασκευαστής μπορεί να εγγυηθεί ότι οι διαφοροποιήσεις από την προκαθορισμένη τιμή του αντιστάτη δεν θα υπερβαίνουν το $1,5\%$.

Εναλλακτικά: θέλουμε η πραγματική τιμή να μην απέχει περισσότερο από $10\,\Omega$ από το $660$. Επειδή $\dfrac{10}{660} \simeq 0,015$, το ζητούμενο ποσοστό είναι $1,5\%$.

Για την τιμή των $665\,\Omega$ εργαζόμαστε ομοίως: θέλουμε η πραγματική τιμή να μην απέχει περισσότερο από $5\,\Omega$ (ώστε να είμαστε εντός του διαστήματος). Επειδή $\dfrac{5}{665} \simeq 0,0075$, το ζητούμενο ποσοστό είναι $0,75\%$.
\end{rlist}
\end{alist}
