\lysh{14749}{3}

\begin{alist}
\item
\begin{rlist}
\item Η παράσταση $A = \frac{x}{x-|x|}$ ορίζεται όταν ο παρονομαστής είναι διάφορος του μηδενός, δηλαδή:
$x - |x| \ne 0 \Leftrightarrow |x| \ne x \Leftrightarrow x < 0$.
\item Για $x < 0$, έχουμε $|x| = -x$, οπότε:
$A = \frac{x}{x - (-x)} = \frac{x}{2x} = \frac{1}{2}$.
\end{rlist}
\item Για $x < 0$, η παράσταση $A = \frac{x}{x-|x|}$ είναι ίση με $\frac{1}{2}$.
Η εξίσωση γίνεται:
\begin{align*}
\frac{x^3}{x-|x|} = \frac{3}{2}x + 2 &\Leftrightarrow x^2 \cdot \frac{x}{x-|x|} = \frac{3}{2}x + 2 \\
&\Leftrightarrow \frac{1}{2}x^2 = \frac{3}{2}x + 2 \\
&\Leftrightarrow x^2 = 3x + 4 \\
&\Leftrightarrow x^2 - 3x - 4 = 0
\end{align*}
Το τριώνυμο έχει διακρίνουσα $\Delta = (-3)^2 - 4 \cdot 1 \cdot (-4) = 9 + 16 = 25$.
Οι ρίζες της εξίσωσης είναι:
$x = \frac{3 \pm \sqrt{25}}{2 \cdot 1} = \frac{3 \pm 5}{2}$ δηλαδή $x = 4$ ή $x = -1$.
Επειδή $x < 0$, δεκτή είναι μόνο η τιμή $x = -1$.
\end{alist}
