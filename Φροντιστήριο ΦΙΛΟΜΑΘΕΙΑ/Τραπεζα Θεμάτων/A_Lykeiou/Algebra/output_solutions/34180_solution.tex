\lysh{34180}{4}

\begin{alist}
\item Οι αριθμοί $2, x, 8$, με τη σειρά που δίνονται, αποτελούν διαδοχικούς όρους αριθμητικής προόδου αν και μόνο αν $2x = 8 + 2$, δηλαδή $2x = 10$ και τελικά $x = 5$. Η διαφορά $\omega$ της προόδου είναι $\omega = 5 - 2 = 3$.

\item Οι αριθμοί $2, x, 8$, με τη σειρά που δίνονται, αποτελούν διαδοχικούς όρους γεωμετρικής προόδου αν και μόνο αν $x^2 = 2 \cdot 8$, δηλαδή $x^2 = 16$ και τελικά $x = \pm 4$.
Για $x = -4$, ο λόγος της προόδου είναι $\lambda = \dfrac{-4}{2} = -2$, ενώ για $x = 4$ ο λόγος της προόδου είναι $\lambda = \dfrac{4}{2} = 2$.

\item Αν $(\alpha_\nu)$ είναι η αριθμητική πρόοδος $2, 5, 8, 11, \dots$ και $(\beta_\nu)$ η γεωμετρική πρόοδος $2, 4, 8, 16, \dots$, τότε:
\begin{rlist}
\item Το άθροισμα $S_\nu$ των $\nu$ πρώτων όρων της $(\alpha_\nu)$ με $\alpha_1 = 2$ και $\omega = 3$, είναι
\[S_\nu = \dfrac{\nu}{2} \left[2 \cdot 2 + (\nu - 1) \cdot 3\right] = \dfrac{\nu}{2} (1 + 3\nu) = \dfrac{3\nu^2 + \nu}{2}.\]

\item Ο $7$ος όρος της γεωμετρικής προόδου $(\beta_\nu)$ με $1$ο όρο $\beta_1 = 2$ και λόγο $\lambda = 2$ είναι
$\beta_7 = 2 \cdot 2^{7-1} = 2 \cdot 2^6 = 2^7 = 128$. Έχουμε ισοδύναμα:

\begin{gather*}
2 \cdot (S_\nu + 24) = \beta_7\text{, δηλαδή} \\
2 \cdot \left(\dfrac{3\nu^2 + \nu}{2} + 24\right) = 128\text{, οπότε} \\
3\nu^2 + \nu - 80 = 0.
\end{gather*}

Το τριώνυμο $3\nu^2 + \nu - 80$ έχει διακρίνουσα $\Delta = 1^2 - 4 \cdot 3 \cdot (-80) = 961 > 0$, οπότε η εξίσωση έχει δυο ρίζες διαφορετικές, τις:
$\nu_1 = \dfrac{-1 - \sqrt{961}}{2 \cdot 3} = \dfrac{-1 - 31}{6} = -\dfrac{16}{3}$, που απορρίπτεται γιατί $\nu \in \mathbb{N}^*$.
\[\nu_2 = \dfrac{-1 + \sqrt{961}}{2 \cdot 3} = \dfrac{-1 + 31}{6} = 5.\]
Τελικά $\nu = 5$.
\end{rlist}
\end{alist}
