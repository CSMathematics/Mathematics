\lysh{34152}{2}

\begin{alist}
\item Η παράσταση $A$ ορίζεται αν και μόνο αν $(x - 2)^2 \ge 0$, που ισχύει για κάθε πραγματικό
αριθμό $x$.

\item Η παράσταση $B$ ορίζεται αν και μόνο αν $(2 - x)^3 \ge 0 \Leftrightarrow 2 - x \ge 0 \Leftrightarrow x \le 2$.

\item Για κάθε $x \le 2$, έχουμε:
$A = \sqrt{(x - 2)^2} = |x - 2| = -(x - 2) = 2 - x$ και $B = \sqrt[3]{(2 - x)^3} = 2 - x$.

Άρα $A = B$.
\end{alist}
