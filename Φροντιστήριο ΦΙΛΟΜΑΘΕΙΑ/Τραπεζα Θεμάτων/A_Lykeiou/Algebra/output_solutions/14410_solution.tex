\lysh{14410}{2}

\begin{alist}
\item 
\begin{rlist}
\item Έχουμε $A + B = \alpha^2 + \alpha + \dfrac{1}{4} + (\beta - 3)^2 = \left(\alpha + \dfrac{1}{2}\right)^2 + (\beta - 3)^2 \ge 0$.
\item Έχουμε:
\[A + B = 0 \Leftrightarrow \left(\alpha + \dfrac{1}{2}\right)^2 + (\beta - 3)^2 = 0\text{, που ισχύει αν και μόνο αν}\]
$\alpha + \dfrac{1}{2} = 0 \text{ και } \beta - 3 = 0$, οπότε $\alpha = -\dfrac{1}{2} \text{ και } \beta = 3$.
\end{rlist}

\item Έχουμε ισοδύναμα:
\[A = -B \Leftrightarrow A + B = 0 \stackrel{(\alpha \text{ ii})}{\Leftrightarrow} \alpha = -\dfrac{1}{2}, \beta = 3.\]
\end{alist}
