\lysh{34184}{4}

\begin{alist}
\item Ισχύει ότι $x + y = 10 \Leftrightarrow y = 10 - x$. Εφόσον $x$ και $y$ είναι πλευρές τριγώνου, πρέπει:
$\begin{cases} x > 0 \\ y > 0 \end{cases} \Leftrightarrow \begin{cases} x > 0 \\ 10 - x > 0 \end{cases} \Leftrightarrow \begin{cases} x > 0 \\ x < 10 \end{cases} \Leftrightarrow 0 < x < 10$, δηλαδή $x \in (0, 10)$.

Οπότε το εμβαδόν του $A B \Gamma$ τριγώνου είναι:
$E = \dfrac{1}{2} \cdot AB \cdot A\Gamma$, δηλαδή
$E(x) = \dfrac{1}{2} x (10 - x) = \dfrac{1}{2} (-x^2 + 10x)$, με πεδίο ορισμού το διάστημα $(0, 10)$.

\item Έχουμε:
\begin{align*}
E(x) \le \dfrac{25}{2} \ &\Leftrightarrow \\
\dfrac{1}{2} (-x^2 + 10x) \le \dfrac{25}{2} \ &\Leftrightarrow \\
-x^2 + 10x - 25 \le 0 \ &\Leftrightarrow \\
x^2 - 10x + 25 \ge 0 \ &\Leftrightarrow
\end{align*}
$(x - 5)^2 \ge 0$, που ισχύει για κάθε $x \in (0, 10)$.

\item Το εμβαδόν $E(x)$ γίνεται μέγιστο αν και μόνο αν:
\begin{align*}
E(x) &= \dfrac{25}{2} \stackrel{(\beta)}{\Leftrightarrow} \\
(x - 5)^2 = 0 \ &\Leftrightarrow \\
x &= 5.
\end{align*}

Για $x = 5$, είναι και $y = 10 - x = 10 - 5 = 5$, οπότε $AB = A\Gamma = 5$ και το τρίγωνο $A B \Gamma$ είναι
ορθογώνιο και ισοσκελές.
\end{alist}
