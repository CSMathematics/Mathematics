\begin{alist}
\item Είναι:
\begin{gather*}
f(-5) = 8 - (-5) = 8 + 5 = 13\text{και} \\
f(4) = 2 \cdot 4 + 5 = 8 + 5 = 13.
\end{gather*}
Άρα $f(-5) = f(4)$.

\item Για $x < 0$ είναι:
$f(x) = 9 \Leftrightarrow 8 - x = 9 \Leftrightarrow -x = 1 \Leftrightarrow x = -1$, δεκτή.

Για $x \ge 0$ είναι:
$f(x) = 9 \Leftrightarrow 2x + 5 = 9 \Leftrightarrow 2x = 4 \Leftrightarrow x = 2$, δεκτή.
\end{alist}
