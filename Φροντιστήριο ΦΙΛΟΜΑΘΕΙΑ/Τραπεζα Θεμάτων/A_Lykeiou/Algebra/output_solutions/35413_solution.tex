\lysh{35413}{2}

\begin{alist}
\item Πρέπει:
\begin{align*}
x^2 - 1 \ne 0 \Leftrightarrow (x - 1)(x + 1) \ne 0 \ &\Leftrightarrow \\
\Leftrightarrow (x - 1 \ne 0 \text{ και } x + 1 \ne 0) \ &\Leftrightarrow \\
\Leftrightarrow (x &\ne 1 \text{ και } x \ne -1)
\end{align*}

Άρα το πεδίο ορισμού της $f$ είναι το $A = \mathbb{R} - \{-1, 1\}$.

\item Το σημείο $M\left(\alpha, \dfrac{1}{8}\right)$ ανήκει στη γραφική παράσταση της $f$ αν και μόνο αν:
\begin{align*}
f(\alpha) = \dfrac{1}{8} \Leftrightarrow \dfrac{1}{\alpha^2 - 1} = \dfrac{1}{8} \ &\Leftrightarrow \\
\Leftrightarrow \alpha^2 - 1 = 8 \ &\Leftrightarrow \\
\Leftrightarrow \alpha^2 - 9 = 0 \ &\Leftrightarrow \\
\Leftrightarrow (\alpha - 3)(\alpha + 3) = 0 \ &\Leftrightarrow \\
\Leftrightarrow (\alpha - 3 = 0 \text{ ή } \alpha + 3 = 0) \ &\Leftrightarrow \\
\Leftrightarrow (\alpha = 3 \text{ ή } \alpha = -3)&
\end{align*}
\end{alist}
