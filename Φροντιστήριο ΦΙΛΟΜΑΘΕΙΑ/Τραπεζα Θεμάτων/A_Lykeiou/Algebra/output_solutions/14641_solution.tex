\lysh{14641}{2}

\begin{alist}
\item Ο συντελεστής διεύθυνσης της ευθείας είναι $\alpha = \varepsilon\phi 45^\circ = 1$.

\item Η ευθεία $\varepsilon$ έχει εξίσωση της μορφής $y = \alpha x + \beta$. Από το ερώτημα $\alpha)$ γνωρίζουμε ότι
$\alpha = 1$. Άρα, $\varepsilon: y = x + \beta$. Επίσης, η ευθεία τέμνει τον άξονα $y'y$ στο σημείο $A(0, 3)$ οπότε,
$\beta = 3$. Άρα, η ευθεία $\varepsilon$ έχει εξίσωση $y = x + 3$.

\item Η ευθεία $\varepsilon$ τέμνει τον άξονα $x'x$ στο σημείο για το οποίο
\[y = 0 \Leftrightarrow x + 3 = 0 \Leftrightarrow x = -3.\]
Άρα, το σημείο τομής είναι το $(-3, 0)$.
\end{alist}
