\lysh{14224}{2}

\begin{alist}
\item Έχουμε: $A = \dfrac{x^2 - 1}{x^2 - x} = \dfrac{(x - 1)(x + 1)}{x(x - 1)} = \dfrac{x + 1}{x}$.

\item 
\begin{rlist}
\item Πρέπει να βρούμε την τιμή του $x$ για την οποία $A = 0$, δηλαδή $\dfrac{x + 1}{x} = 0$, που συμβαίνει όταν $x + 1 = 0$, δηλαδή όταν $x = -1$.
\item Για να πάρει η παράσταση $A$ την τιμή $2$, πρέπει να ισχύουν ισοδύναμα:
\[\dfrac{x + 1}{x} = 2 \Leftrightarrow x + 1 = 2x \Leftrightarrow x = 1,\]
που δεν είναι αποδεκτή τιμή για το $x$. 
Άρα η παράσταση $A$ δεν μπορεί να πάρει την τιμή $2$.
\end{rlist}
\end{alist}
