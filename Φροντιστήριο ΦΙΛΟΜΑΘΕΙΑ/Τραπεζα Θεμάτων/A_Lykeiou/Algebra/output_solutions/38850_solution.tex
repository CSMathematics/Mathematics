\lysh{38850}{4}

\begin{alist}
\item
\begin{rlist}
\item Η ακτίνα της κυλινδρικής τρύπας στον μικρό κρίκο είναι $\rho = \sqrt{2,5^2 - 2^2} = \sqrt{2,25} = 1,5\text{ cm}$, όπως προκύπτει από το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο $ABK$.
Η ακτίνα της κυλινδρικής τρύπας στον μεγάλο κρίκο είναι $\rho^\prime = \sqrt{4^2 - 2^2} = \sqrt{12} = 2\sqrt{3}\text{ cm}$, όπως προκύπτει από το Πυθαγόρειο θεώρημα στο ορθογώνιο τρίγωνο $A^\prime B^\prime N$.
\item Για την κυλινδρική τρύπα στον μικρό κρίκο έχουμε $V_A = \pi \rho^2 \cdot h = \pi \cdot 2,25 \cdot 4 = 9\pi \text{ cm}^3$.
Για την κυλινδρική τρύπα στον μεγάλο κρίκο έχουμε $V_B = \pi (\rho^\prime)^2 \cdot h = \pi \cdot 12 \cdot 4 = 48\pi \text{ cm}^3$.
\end{rlist}

\item
\begin{rlist}
\item Για τον μικρό κρίκο θα έχει μείνει:
\[\dfrac{4}{3}\pi (2,5)^3 - 9\pi - \dfrac{7\pi}{6} = \dfrac{62,5\pi}{3} - \dfrac{61\pi}{6} = \dfrac{125\pi - 61\pi}{6} = \dfrac{64\pi}{6} = \dfrac{32\pi}{3}\text{ cm}^3.\]
Για τον μεγάλο κρίκο θα έχει μείνει:
\[\dfrac{4}{3}\pi 4^3 - 48\pi - \dfrac{80\pi}{3} = \dfrac{256\pi}{3} - \dfrac{144\pi}{3} - \dfrac{80\pi}{3} = \dfrac{32\pi}{3}\text{ cm}^3.\]
\item Από το $\beta)$ ερώτημα προκύπτει ότι ο όγκος του ξύλου που μένει σε κάθε κρίκο είναι $\dfrac{32\pi}{3} \text{ cm}^3$, οπότε θα ελέγξουμε αν
\[V = \dfrac{4}{3}\pi R_1^3 - 4\pi(R_1^2 - 4) - \dfrac{4}{3}\pi(R_1 - 2)^2 (R_1 + 1) = \dfrac{32\pi}{3}.\]

Έχουμε:
\begin{gather*}
V = \dfrac{4}{3}\pi R_1^3 - 4\pi(R_1^2 - 4) - \dfrac{4}{3}\pi(R_1 - 2)^2 (R_1 + 1) = \\
\dfrac{4}{3}\pi R_1^3 - 4\pi R_1^2 + 16\pi - \dfrac{4}{3}\pi(R_1^2 - 4R_1 + 4)(R_1 + 1) = \\
\dfrac{4}{3}\pi R_1^3 - 4\pi R_1^2 + 16\pi - \dfrac{4}{3}\pi(R_1^3 - 4R_1^2 + 4R_1 + R_1^2 - 4R_1 + 4) = \\
\dfrac{4}{3}\pi R_1^3 - 4\pi R_1^2 + 16\pi - \dfrac{4}{3}\pi R_1^3 + \dfrac{12}{3}\pi R_1^2 - \dfrac{16}{3}\pi = 16\pi - \dfrac{16\pi}{3} = \dfrac{32\pi}{3}.
\end{gather*}

Άρα ο ισχυρισμός είναι σωστός, δηλαδή ανεξάρτητα από την ακτίνα της σφαίρας, ο όγκος του ξύλου που μένει είναι ίδιος. Οπότε απαντήσαμε και στην αρχική ερώτηση!
\end{rlist}
\end{alist}
