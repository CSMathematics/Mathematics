\lysh{33888}{4}

\begin{alist}
\item Αφού $(\alpha - 1)(1 - \beta) > 0$, οι $(\alpha - 1)$ και $(1 - \beta)$ είναι ομόσημοι, οπότε:
\[\begin{cases} \alpha - 1 > 0 \\ 1 - \beta > 0 \end{cases} \Leftrightarrow \begin{cases} \alpha > 1 \\ \beta < 1 \end{cases} \Leftrightarrow \beta < 1 < \alpha\]

ή

\[\begin{cases} \alpha - 1 < 0 \\ 1 - \beta < 0 \end{cases} \Leftrightarrow \begin{cases} \alpha < 1 \\ \beta > 1 \end{cases} \Leftrightarrow \alpha < 1 < \beta.\]

Σε κάθε περίπτωση το $1$ είναι μεταξύ των $\alpha$ και $\beta$.

\item Διακρίνουμε δυο περιπτώσεις:
\begin{rlist}
\item Αν $\beta < 1 < \alpha$, τότε:
\begin{gather*}
0 < \alpha - 1 \Rightarrow |\alpha - 1| = \alpha - 1, \\
0 < 1 - \beta \Rightarrow |1 - \beta| = 1 - \beta\text{και}
\end{gather*}
$\beta - \alpha < 0 \Rightarrow |\beta - \alpha| = \alpha - \beta$ άρα,
\[|\beta - \alpha| = 4 \Leftrightarrow \alpha - \beta = 4.\]

Οπότε: $|\alpha - 1| + |1 - \beta| = \alpha - 1 + 1 - \beta = \alpha - \beta = 4$.

\item Αν $\alpha < 1 < \beta$, τότε:
\begin{gather*}
\alpha - 1 < 0 \Rightarrow |\alpha - 1| = 1 - \alpha, \\
1 - \beta < 0 \Rightarrow |1 - \beta| = \beta - 1\text{και}
\end{gather*}
$\beta - \alpha > 0 \Rightarrow |\beta - \alpha| = \beta - \alpha$ άρα,
\[|\beta - \alpha| = 4 \Leftrightarrow \beta - \alpha = 4.\]

Οπότε: $|\alpha - 1| + |1 - \beta| = 1 - \alpha + \beta - 1 = \beta - \alpha = 4$.
\end{rlist}
\end{alist}
