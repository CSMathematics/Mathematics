\lysh{38843}{4}

\begin{alist}
\item
\begin{rlist}
\item Ο όγκος του υγρού είναι $V = 20 \cdot 5 + 100 = 200 \text{ cm}^3$.
\item Για να υπολογίσουμε τον όγκο $V_B$, πρέπει στα $100 \text{ cm}^3$ του κάτω μέρους του δοχείου να προσθέσουμε τα $20h \text{ cm}^3$ που περιέχονται στο κυλινδρικό πάνω μέρος του. Άρα, $V_B = 20h + 100$.
Ο όγκος του υγρού στο δοχείο Α δίνεται από τη σχέση $V_A = \alpha \cdot h$.
\end{rlist}

\item Για να περιέχουν τα δύο δοχεία τον ίδιο όγκο υγρού, πρέπει
$\alpha \cdot h = 20h + 100$
ή
$(\alpha - 20)h = 100$
Η εξίσωση για $\alpha - 20 = 0$, δηλαδή για $\alpha = 20$ είναι αδύνατη. Οπότε η επιλογή Β. απορρίπτεται.
Για $\alpha \neq 20$ έχουμε ότι:
$h = \dfrac{100}{\alpha - 20}$
Για $\alpha = 10 \text{ cm}^2$ βρίσκουμε ότι $h = \dfrac{100}{10-20} = -10 \text{ cm}$, που απορρίπτεται, καθώς πρέπει $h \ge 0$.
Για $\alpha = 40 \text{ cm}^2$ βρίσκουμε $h = \dfrac{100}{40-20} = 5 \text{ cm}$, που είναι αποδεκτό.

Άρα, αποδεκτή είναι η επιλογή Γ.

\item
\begin{rlist}
\item Πρέπει $|30h - (20h + 100)| \le 10$,
δηλαδή $|10h - 100| \le 10$,
οπότε $-10 \le 10h - 100 \le 10$,
άρα $90 \le 10h \le 110$,
και τελικά $9 \le h \le 11$.
\item Πρέπει $30h + 20h + 100 \le 600$ και $9 \le h \le 11$,
άρα $50h \le 500$ και $9 \le h \le 11$,
οπότε $h \le 10$ και $9 \le h \le 11$,
και τελικά $9 \le h \le 10$.
\end{rlist}
\end{alist}
