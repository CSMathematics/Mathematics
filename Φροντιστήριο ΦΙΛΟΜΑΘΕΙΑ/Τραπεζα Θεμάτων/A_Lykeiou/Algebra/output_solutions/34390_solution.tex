\lysh{34390}{4}

\begin{alist}
\item 
\begin{rlist}
\item Η περίμετρος του ορθογωνίου είναι $\Pi = 2\kappa + 2\lambda$, οπότε:
\[2\kappa + 2\lambda = 14 \Leftrightarrow \kappa + \lambda = 7.\]
Επίσης, εφαρμόζοντας το πυθαγόρειο θεώρημα στο τρίγωνο $A B \Delta$ βρίσκουμε ότι
\[\kappa^2 + \lambda^2 = \delta^2 \Leftrightarrow \kappa^2 + \lambda^2 = 25.\]
Από την ταυτότητα $(\kappa + \lambda)^2 = \kappa^2 + 2\kappa\lambda + \lambda^2$, έχουμε ότι:
\begin{align*}
7^2 = 25 + 2\kappa\lambda \ &\Leftrightarrow \\
2\kappa\lambda = 49 - 25 \ &\Leftrightarrow \\
\kappa\lambda &= 12.
\end{align*}
Άρα, το εμβαδόν του ορθογωνίου είναι $E = \kappa\lambda = 12 \text{ cm}^2$.

\item Δύο αριθμοί είναι ρίζες της εξίσωσης $x^2 - 7x + 12 = 0$ αν και μόνο αν έχουν
άθροισμα $S = -\dfrac{\beta}{\alpha} = -\dfrac{-7}{1} = 7$ και γινόμενο $P = \dfrac{\gamma}{\alpha} = \dfrac{12}{1} = 12$. Από το ερώτημα $\alpha\text{)i.}$
προκύπτει ότι οι διαστάσεις $\kappa$ και $\lambda$ ικανοποιούν τις συνθήκες αυτές, οπότε είναι ρίζες
της εξίσωσης.

\item Το τριώνυμο $x^2 - 7x + 12$ έχει διακρίνουσα
\[\Delta = (-7)^2 - 4 \cdot 1 \cdot 12 = 49 - 48 = 1\]
και ρίζες
\[x_{1,2} = \dfrac{-(-7) \pm \sqrt{1}}{2 \cdot 1} = \begin{cases} \dfrac{7 + 1}{2} = 4 \\ \dfrac{7 - 1}{2} = 3 \end{cases}.\]
Άρα, οι διαστάσεις του ορθογωνίου είναι $3\text{cm}$ και $4\text{cm}$.
\end{rlist}

\item Όπως και στο ερώτημα $\alpha)$, οι διαστάσεις ενός ορθογωνίου με περίμετρο $\Pi = 14$ και
εμβαδόν $E$ έχουν άθροισμα $S = 7$ και γινόμενο $P = E$. Άρα, είναι ρίζες της εξίσωσης
$x^2 - 7x + E = 0$.
Η εξίσωση έχει λύσεις, δηλαδή υπάρχει τέτοιο ορθογώνιο, αν και μόνο αν για τη διακρίνουσα
ισχύει
\begin{align*}
\Delta \ge 0 \Leftrightarrow (-7)^2 - 4 \cdot 1 \cdot E \ge 0 \ &\Leftrightarrow \\
\Leftrightarrow 49 - 4E \ge 0 \ &\Leftrightarrow \\
\Leftrightarrow E &\le \dfrac{49}{4}.
\end{align*}
\end{alist}
