\lysh{14753}{3}

\begin{alist}
\item Από την $(1)$ έχουμε ισοδύναμα:
\[2x - 2 \le 0 \Leftrightarrow 2x \le 2 \Leftrightarrow x \le 1 \Leftrightarrow -1 \le x \le 1.\]

\item Αρκεί να δείξουμε ότι: $d(x, -3) \le 4$.
Είναι: $d(x, -3) = |x + 3| \le |x| + |3| \le 1 + 3 = 4$, δηλαδή $d(x, -3) \le 4$.
\textit{Εναλλακτικά}, αξιοποιώντας το $\alpha$ ερώτημα, έχουμε:
\begin{align*}
-1 &\le x \le 1 \Leftrightarrow 2 \le x + 3 \le 4\text{, οπότε} \\
|x + 3| &\le 4 \Leftrightarrow d(x, -3) \le 4.
\end{align*}

\item Είναι:
\begin{align*}
-1 &\le x \le 1 \Leftrightarrow -2 \le 2x \le 2 \Leftrightarrow -5 \le 2x - 3 \le -1 < 0\text{, οπότε} \\
|2x - 3| &= -2x + 3. \\
-1 &\le x \le 1 \Leftrightarrow 3 \ge -3x \ge -3 \Leftrightarrow 7 \ge 4 - 3x \ge 1 > 0\text{, οπότε} \\
|4 - 3x| &= 4 - 3x.
\end{align*}
Τελικά: $A = |2x - 3| - |4 - 3x| = -2x + 3 - (4 - 3x) = -2x + 3 - 4 + 3x = x - 1$.
\end{alist}
