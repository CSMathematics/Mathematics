\begin{alist}
\item Είναι:
\[A \cdot B \cdot \Gamma = \sqrt[3]{5} \cdot \sqrt{3} \cdot \sqrt[6]{5} = 5^{\frac{1}{3}} \cdot 3^{\frac{1}{2}} \cdot 5^{\frac{1}{6}} = 5^{\frac{1}{3} + \dfrac{1}{6}} \cdot 3^{\frac{1}{2}} = 5^{\frac{3}{6}} \cdot 3^{\frac{1}{2}} = 5^{\frac{1}{2}} \cdot 3^{\frac{1}{2}} = \sqrt{5} \cdot \sqrt{3} = \sqrt{15}.\]

\item Είναι:
$A = \sqrt[3]{5} = 5^{\frac{1}{3}} = 5^{\frac{2}{6}} = \sqrt[6]{5^2} = \sqrt[6]{25}$ και $B = \sqrt{3} = 3^{\frac{1}{2}} = 3^{\frac{3}{6}} = \sqrt[6]{3^3} = \sqrt[6]{27}$.
Ισχύει ότι:
\[25 < 27 \Leftrightarrow \sqrt[6]{25} < \sqrt[6]{27} \Leftrightarrow A < B.\]
\end{alist}
