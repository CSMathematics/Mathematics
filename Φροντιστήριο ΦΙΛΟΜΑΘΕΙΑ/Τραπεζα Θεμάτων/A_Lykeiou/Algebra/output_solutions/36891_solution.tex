\lysh{36891}{2}

\begin{alist}
\item Έχουμε:
\[\begin{cases} \alpha_3 = 1 \\ \alpha_5 = 4 \end{cases} \Leftrightarrow \begin{cases} \alpha_1 \cdot \lambda^2 = 1 \\ \alpha_1 \cdot \lambda^4 = 4 \end{cases}\]

Με διαίρεση κατά μέλη προκύπτει:
\[\lambda^2 = 4 \overset{\lambda > 0}{\Leftrightarrow} \lambda = 2.\]
Για $\lambda = 2$ η πρώτη εξίσωση δίνει:
\[\alpha_1 \cdot 2^2 = 1 \Leftrightarrow \alpha_1 = \dfrac{1}{4}.\]

\item Ο $\nu$-οστός όρος μιας γεωμετρικής προόδου είναι: $\alpha_\nu = \alpha_1 \cdot \lambda^{\nu-1}$, οπότε
\[\alpha_\nu = \dfrac{1}{4} \cdot 2^{\nu-1} = \dfrac{1}{2^2} \cdot 2^{\nu-1} = 2^{-2} \cdot 2^{\nu-1} = 2^{\nu-3}.\]
\end{alist}
