\lysh{13031}{2}

\begin{alist}
\item Έχουμε:
\begin{gather*}
G(2) = \dfrac{2 \cdot 2 + 3}{2 - 4} = \dfrac{4 + 3}{-2} = -\dfrac{7}{2},\\
G(0) = \dfrac{2 \cdot 0 + 3}{0 - 4} = -\dfrac{3}{4}, \\
G\left(-\dfrac{1}{2}\right) = \dfrac{2 \cdot \left(-\dfrac{1}{2}\right) + 3}{-\dfrac{1}{2} - 4} = \dfrac{-1 + 3}{-\dfrac{9}{2}} = -\dfrac{2}{\dfrac{9}{2}} = -\dfrac{4}{9}.
\end{gather*}

\item Η συνάρτηση δεν ορίζεται για $x = 4$, διότι η τιμή αυτή του $x$ μηδενίζει τον παρονομαστή του κλάσματος στον τύπο της συνάρτησης.

\item Αναζητούμε την τιμή του $x \ne 4$ για την οποία:
\begin{align*}
G(x) = 3 \ &\Leftrightarrow 
\dfrac{2x + 3}{x - 4} = 3 \ \Leftrightarrow \\
2x + 3 = 3(x - 4) \ &\Leftrightarrow 
2x + 3 = 3x - 12 \ \Leftrightarrow \\
x &= 15.
\end{align*}
\end{alist}
