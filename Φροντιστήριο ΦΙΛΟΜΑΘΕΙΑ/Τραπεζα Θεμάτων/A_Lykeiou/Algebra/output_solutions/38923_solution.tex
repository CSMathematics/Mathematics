\lysh{38923}{4}

\begin{alist}
\item Έστω $x$ το μήκος του κήπου και $y$ το πλάτος του. Αφού η περίμετρος είναι $26\text{ m}$, θα πρέπει $2x + 2y = 26$, άρα $y = 13 - x$.
Οπότε το εμβαδόν του κήπου υπολογίζεται από την εξίσωση:
\[x(13 - x) = 40 \Leftrightarrow 13x - x^2 = 40 \Leftrightarrow x^2 - 13x + 40 = 0.\]

\textit{Εναλλακτική λύση:}
Αν $x_1$ και $x_2$ οι λύσεις της ζητούμενης εξίσωσης (μήκος και πλάτος), θα έχουμε:
$S = x_1 + x_2 = 13$, αφού η περίμετρος είναι $26$, άρα το $x_1 + x_2$ είναι η ημιπερίμετρος δηλαδή $13$.
$P = x_1 x_2 = 40$, αφού το εμβαδόν του ορθογώνιου κήπου είναι $40\text{ m}^2$.
Οπότε η ζητούμενη εξίσωση θα είναι της μορφής $x^2 - Sx + P = 0$, άρα
$x^2 - 13x + 40 = 0$.

\item Για να βρούμε τις διαστάσεις του κήπου θα λύσουμε την εξίσωση:
\begin{gather*}
x^2 - 13x + 40 = 0. \\
\Delta = (-13)^2 - 4 \cdot 1 \cdot 40 = 169 - 160 = 9 > 0. \\
x_{1,2} = \dfrac{-(-13) \pm \sqrt{9}}{2 \cdot 1} = \dfrac{13 \pm 3}{2}\text{, οπότε}
\end{gather*}
$x_1 = \dfrac{13 + 3}{2} = \dfrac{16}{2} = 8$ και $x_2 = \dfrac{13 - 3}{2} = \dfrac{10}{2} = 5$.

Άρα οι διαστάσεις του κήπου θα είναι μήκος $8\text{ m}$ και πλάτος $5\text{ m}$.

\item Αφού το εμβαδόν του κύκλου είναι $20\text{ m}^2$, θα υπολογίσουμε την ακτίνα του παρτεριού:
\[\pi \rho^2 = 20 \Leftrightarrow 3,14\rho^2 = 20 \Leftrightarrow \rho^2 = \dfrac{20}{3,14} \Leftrightarrow \rho^2 \simeq 6,37 \Leftrightarrow \rho = \sqrt{6,37} \Leftrightarrow \rho \simeq 2,52.\]

Άρα η διάμετρος του παρτεριού θα είναι περίπου $2,52 \cdot 2 = 5,04 > 5\text{ m}$, που είναι το πλάτος του κήπου. Δηλαδή, δεν θα χωρέσει το παρτέρι στον κήπο.

\item Το εμβαδόν του κήπου είναι $40\text{ m}^2$, άρα αν $y$ το νέο πλάτος του κήπου με $4 < y < 7$ και $x$ το νέο μήκος του, θα έχουμε: $xy = 40 \Leftrightarrow y = \dfrac{40}{x}$ και $4 < y < 7 \Leftrightarrow 4 < \dfrac{40}{x} < 7 \Leftrightarrow$
\[\dfrac{1}{4} > \dfrac{x}{40} > \dfrac{1}{7} \Leftrightarrow \dfrac{40}{4} > x > \dfrac{40}{7} \Leftrightarrow 10 > x > 5,7.\]

Άρα το νέο μήκος του κήπου $x \in (5,7, 10)$.
\end{alist}
