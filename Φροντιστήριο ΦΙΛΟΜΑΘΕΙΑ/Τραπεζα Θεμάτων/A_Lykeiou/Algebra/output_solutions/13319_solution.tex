\lysh{13319}{2}

\begin{alist}
\item Γνωρίζουμε ότι οι αριθμοί $\alpha, \beta, \gamma$, με αυτή τη σειρά, είναι διαδοχικοί όροι μιας αριθμητικής προόδου, αν και μόνον αν ισχύει $\beta - \alpha = \gamma - \beta$. Έτσι, θα πρέπει να ισχύει ότι
$\dfrac{x}{2} - (1 - x) = 2x - 1 - \dfrac{3x}{2}$, δηλαδή $\dfrac{x}{2} - 1 + x = \dfrac{x}{2} - 1$, άρα $-1 = -1$, που ισχύει.

\item Πρέπει να ισχύει $\dfrac{3x}{2} - 1 = 5 \Leftrightarrow \dfrac{3x}{2} = 6 \Leftrightarrow 3x = 12 \Leftrightarrow x = 4$.
\end{alist}
