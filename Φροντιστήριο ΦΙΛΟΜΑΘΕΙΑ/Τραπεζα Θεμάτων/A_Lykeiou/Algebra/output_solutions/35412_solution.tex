\lysh{35412}{2}

\begin{alist}
\item Ισχύει ότι:
\[5 < x < 10 \Leftrightarrow (5 < x \text{ και } x < 10) \Leftrightarrow (0 < x - 5 \text{ και } x - 10 < 0)\]

Τότε:
$|x - 5| = x - 5$ και $|x - 10| = -(x - 10)$

\item Είναι:
\[A = \dfrac{|x - 5|}{x - 5} + \dfrac{|x - 10|}{x - 10} = \dfrac{x - 5}{x - 5} + \dfrac{-(x - 10)}{x - 10} = 1 + (-1) = 0\]
\end{alist}
