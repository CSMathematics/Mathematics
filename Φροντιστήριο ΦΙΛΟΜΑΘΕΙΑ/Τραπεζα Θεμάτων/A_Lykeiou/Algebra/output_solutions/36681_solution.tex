\lysh{36681}{4}

\begin{alist}
\item Η γραφική παράσταση της $f$ διέρχεται από το σημείο $A(0, 2)$ για οποιαδήποτε τιμή του $\lambda$, αν και μόνο αν ισχύει ότι $f(0) = 2$ για κάθε $\lambda \in \mathbb{R}$.
Πράγματι είναι $f(0) = (\lambda + 1) \cdot 0^2 - (\lambda + 1) \cdot 0 + 2 = 2$ για κάθε $\lambda \in \mathbb{R}$.

\item Για $\lambda = -1$ ο τύπος της $f$ γράφεται:
$f(x) = (-1 + 1)x^2 - (-1 + 1)x + 2 = 2$
Επομένως η γραφική παράσταση της $f$ είναι μια ευθεία παράλληλη στον άξονα $x^\prime x$ που διέρχεται από το σημείο $A(0, 2)$.

\item Η γραφική παράσταση της $f$ τέμνει τον άξονα $x^\prime x$ στο σημείο $B(2, 0)$ και επομένως ισχύει ότι $f(2) = 0$. Είναι
\begin{align*}
f(2) = 0 \ &\Leftrightarrow \\
(\lambda + 1) \cdot 2^2 - (\lambda + 1) \cdot 2 + 2 = 0 \ &\Leftrightarrow \\
4\lambda + 4 - 2\lambda - 2 + 2 = 0 \ &\Leftrightarrow \\
2\lambda &= -4 \Leftrightarrow \lambda = -2
\end{align*}

Για $\lambda = -2$ έχουμε:
\[f(x) = (-2 + 1)x^2 - (-2 + 1)x + 2 = -x^2 + x + 2\]
Οι τετμημένες των σημείων τομής της γραφικής παράστασης της $f$ με τον $x^\prime x$ είναι οι λύσεις της εξίσωσης
\[f(x) = 0 \Leftrightarrow -x^2 + x + 2 = 0\]

Το τριώνυμο $-x^2 + x + 2$ έχει διακρίνουσα:
\[\Delta = 1^2 - 4 \cdot (-1) \cdot 2 = 1 + 8 = 9 > 0\]
και ρίζες τις:
\[x_{1,2} = \dfrac{-1 \pm \sqrt{9}}{2 \cdot (-1)} = \dfrac{-1 \pm 3}{-2} = \begin{cases} \dfrac{-1 + 3}{-2} = -1 \\ \dfrac{-1 - 3}{-2} = 2 \end{cases}\]

Επομένως η γραφική παράσταση της $f$ τέμνει τον $x^\prime x$ εκτός από το $B(2, 0)$ και στο σημείο $(-1, 0)$.

\item Για $\lambda = 1$ ο τύπος της $f$ γράφεται:
\[f(x) = (1 + 1)x^2 - (1 + 1)x + 2 = 2x^2 - 2x + 2 = 2(x^2 - x + 1)\]

Το τριώνυμο $x^2 - x + 1$ έχει διακρίνουσα:
\[\Delta = (-1)^2 - 4 \cdot 1 \cdot 1 = 1 - 4 = -3 < 0\]
οπότε για κάθε $x \in \mathbb{R}$ είναι ομόσημο του συντελεστή του $x^2$, δηλαδή του $\alpha = 1 > 0$.
Επομένως για κάθε $x \in \mathbb{R}$ ισχύει ότι:
\[x^2 - x + 1 > 0 \Leftrightarrow 2(x^2 - x + 1) > 0 \Leftrightarrow f(x) > 0\]
που σημαίνει ότι η γραφική παράσταση της $f$ βρίσκεται ολόκληρη πάνω από τον άξονα $x^\prime x$.
\end{alist}
