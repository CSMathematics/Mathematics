\lysh{12673}{2}

\begin{alist}
\item Με $0 < \alpha < \beta$ έχουμε $\dfrac{1}{\alpha} > \dfrac{1}{\beta}$, οπότε $\dfrac{3}{\alpha} > \dfrac{3}{\beta}$. Επομένως, $\dfrac{3}{\beta} < \dfrac{3}{\alpha}$.

\item Με $0 < \alpha < \beta$ έχουμε $\alpha^3 < \beta^3$. Επιπλέον, από το ερώτημα ($\alpha$) είναι $\dfrac{3}{\beta} < \dfrac{3}{\alpha}$, οπότε με πρόσθεση των δυο ανισοτήτων παίρνουμε:

\[\alpha^3 + \dfrac{3}{\beta} < \beta^3 + \dfrac{3}{\alpha}\]

που είναι το ζητούμενο.
\end{alist}
