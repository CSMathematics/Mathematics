\lysh{13026}{2}

\begin{alist}
\item $f(\sqrt{2}) = (\sqrt{2})^2 = 2$, $f\left(\dfrac{1}{2}\right) = 2 \cdot \dfrac{1}{2} = 1$.

\item Αν $x$ ρητός, τότε $f(x) = 2x$, οπότε η δοθείσα εξίσωση γράφεται ισοδύναμα:
$(2x)^2 = 4x - 1$, δηλαδή $4x^2 - 4x + 1 = 0 \Leftrightarrow (2x - 1)^2 = 0$. Έτσι, $2x - 1 = 0 \Leftrightarrow x = \dfrac{1}{2}$.
\end{alist}
