\lysh{36672}{4}

\begin{alist}
\item Αφού $-2 < x < 7$ το σημείο $M$ θα βρίσκεται μεταξύ των σημείων $A(-2)$ και $B(7)$.

\begin{rlist}
\item $|x + 2| = d(x, -2) = (MA)$, δηλαδή εκφράζει την απόσταση του σημείου $M$ από το σημείο $A$.
\item $|x - 7| = d(x, 7) = (MB)$, δηλαδή εκφράζει την απόσταση του σημείου $M$ από το σημείο $B$.
\end{rlist}

\item Είναι $|x + 2| + |x - 7| = (MA) + (MB) = (AB)$.

\item Είναι $A = |x + 2| + |x - 7| = (AB) = d(-2, 7) = |-2 - 7| = |-9| = 9$.

\item Αφού $-2 < x < 7$ είναι:
$x + 2 > 0$ οπότε $|x + 2| = x + 2$ και
$x - 7 < 0$ οπότε $|x - 7| = -(x - 7) = -x + 7$.

Επομένως $A = |x + 2| + |x - 7| = x + 2 - x + 7 = 9$.
\end{alist}
