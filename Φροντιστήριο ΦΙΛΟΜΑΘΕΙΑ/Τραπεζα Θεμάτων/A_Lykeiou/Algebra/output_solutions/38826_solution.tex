\lysh{38826}{4}

\begin{alist}
\item Αν συμβολίσουμε με $x$ το ποσό που ξοδεύουν οι πελάτες σε φαγητό και ποτά και $y = f(x)$ είναι το συνολικό ποσό που εισπράττει ο σερβιτόρος, τότε: αφού γνωρίζουμε ότι ο σερβιτόρος λαμβάνει $50\text{\euro}$ σταθερά την ημέρα και τα φιλοδωρήματα είναι το $10\%$ του ποσού $x$, η σχέση που εκφράζει το συνολικό ποσό είναι:
$y = f(x) = 50 + 0,1x$

\item Η γραφική παράσταση της $f(x) = 50 + 0,1x$ είναι ευθεία, οπότε αρκεί να βρούμε δύο σημεία. Βρίσκουμε τις τιμές για $x = 0$ και για $x = 100$, οπότε η ευθεία διέρχεται από τα σημεία $(0, 50)$ και $(100, 60)$. Άρα η γραφική παράσταση είναι:

\begin{tikzpicture}
\begin{axis}[
    axis lines = left,
    xlabel = $x$,
    ylabel = {$f(x)$},
    xmin=0, xmax=200,
    ymin=0, ymax=80,
    xtick={0, 100},
    ytick={0, 50, 60},
    clip=false
]
\addplot [
    domain=0:200, 
    samples=100, 
    color=maincolor,
    thick,
]
{50 + 0.1*x};
\addplot[mark=*] coordinates {(0, 50)};
\addplot[mark=*] coordinates {(100, 60)};
\draw[dashed] (100,0) -- (100,60) -- (0,60);
\end{axis}
\end{tikzpicture}

\item
\begin{rlist}
\item Υπολογίζουμε για $x = 312: f(312) = 50 + 0,1 \cdot 312 = 81,2\text{\euro}$. Όταν οι πελάτες ξοδεύουν $312\text{\euro}$, ο σερβιτόρος εισπράττει $81,2\text{\euro}$.

\item Θέλουμε να βρούμε τις τιμές του $x$ που ικανοποιούν τη σχέση:
$105 \le 50 + 0,1x \le 120$.
Θα είναι:
\begin{gather*}
105 - 50 \le 0,1x \le 120 - 50\text{, οπότε} \\
55 \le 0,1x \le 70\text{και τελικά} \\
550 \le x \le 700.
\end{gather*}
Άρα, το ποσό που πρέπει να ξοδέψουν οι πελάτες είναι μεταξύ $550\text{\euro}$ και $700\text{\euro}$.
\end{rlist}

\item Το καινούργιο ποσοστό που πρέπει να ζητήσει θα είναι $\alpha\%$ και τότε η σχέση που εκφράζει το ποσό που θα εισπράττει, όταν οι πελάτες ξοδέψουν $500\text{\euro}$ είναι:
\[50 + \alpha\% \cdot 500 = 50 + 0,1 \cdot 500 + 15,\]
οπότε $\alpha\% \cdot 500 = 0,1 \cdot 500 + 15$, δηλαδή
\begin{align*}
\alpha\% \cdot 500 &= 50 + 15 \Leftrightarrow \alpha\% = \dfrac{65}{500} \Leftrightarrow \alpha\% = 0,13\text{και τελικά} \\
\alpha\% &= 13\%.
\end{align*}
Πρέπει να ζητήσει το ποσοστό $13\%$.
\end{alist}
