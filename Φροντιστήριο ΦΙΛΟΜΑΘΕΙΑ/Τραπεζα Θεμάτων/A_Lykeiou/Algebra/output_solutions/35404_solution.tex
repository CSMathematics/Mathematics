\lysh{35404}{2}

\begin{alist}
\item Είναι:
\[|y - 3| < 1 \Leftrightarrow -1 < y - 3 < 1 \Leftrightarrow -1 + 3 < y < 1 + 3 \Leftrightarrow 2 < y < 4\]

\item Έχουμε τις σχέσεις:
\begin{gather*}
1 < x < 3 \\
2 < y < 4
\end{gather*}
τις οποίες προσθέτουμε κατά μέλη και βρίσκουμε:
\[1 + 2 < x + y < 3 + 4 \Leftrightarrow 3 < x + y < 7.\]
\end{alist}
