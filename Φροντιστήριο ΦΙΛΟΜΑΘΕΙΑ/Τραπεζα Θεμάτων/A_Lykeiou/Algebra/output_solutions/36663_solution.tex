\lysh{36663}{4}

\begin{alist}
\item Το εμβαδόν που καλύπτει κάθε πλακάκι τύπου Α είναι $E_A = d^2\text{ cm}^2$.
Το εμβαδόν που καλύπτει κάθε πλακάκι τύπου Β είναι $E_B = (d + 1)^2\text{ cm}^2$.

\item
\begin{rlist}
\item Αν το εμβαδόν της επιφάνειας είναι $E$, τότε ισχύει:
$E = 200d^2$ και $E = 128(d + 1)^2$
οπότε έχουμε:
\begin{align*}
200d^2 = 128(d + 1)^2 \Leftrightarrow 25d^2 = 16(d + 1)^2 \ &\Leftrightarrow \\
\Leftrightarrow 25d^2 = 16(d^2 + 2d + 1) \Leftrightarrow 25d^2 = 16d^2 + 32d + 16 \ &\Leftrightarrow \\
\Leftrightarrow 9d^2 - 32d - 16 &= 0.
\end{align*}

Η τελευταία εξίσωση έχει διακρίνουσα
\[\Delta = \beta^2 - 4\alpha\gamma = (-32)^2 - 4 \cdot 9 \cdot (-16) = 1024 + 576 = 1600 = 40^2\]
και ρίζες τους αριθμούς $d_1 = 4$ και $d_2 = -\dfrac{4}{9}$. Η λύση $d_2 = -\dfrac{4}{9}$ απορρίπτεται, αφού το μήκος της πλευράς είναι θετικός αριθμός. Άρα $d = 4$, οπότε κάθε πλακάκι τύπου Α έχει πλευρά $4$ και κάθε πλακάκι τύπου Β έχει πλευρά $5$.

\item Είναι:
\[E = 200d^2 = 200 \cdot 4^2 = 200 \cdot 16 = \SI{3200}{cm^2}.\]
\end{rlist}
\end{alist}
