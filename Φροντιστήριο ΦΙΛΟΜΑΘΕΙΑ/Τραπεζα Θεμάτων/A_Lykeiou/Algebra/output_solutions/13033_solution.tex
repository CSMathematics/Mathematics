\lysh{13033}{2}

\begin{alist}
\item 
\begin{rlist}
\item Η κλίση της ευθείας είναι $\alpha = -\dfrac{1}{2}$.
\item Επειδή $\alpha < 0$ ισχύει $\varepsilon\phi\omega < 0$, οπότε η γωνία $\omega$ είναι αμβλεία.
\end{rlist}

\item Θα εξετάσουμε αν οι συντεταγμένες των σημείων επαληθεύουν την εξίσωση της ευθείας.
Για το σημείο Α: $-\dfrac{1}{2} \cdot 6 + 4 = -3 + 4 = 1$, άρα το Α είναι σημείο της ευθείας.
Για το σημείο Β: $-\dfrac{1}{2} \cdot (-2) + 4 = 1 + 4 = 5 \ne 3$, άρα το Β δεν είναι σημείο της ευθείας.
Για το σημείο Γ: $-\dfrac{1}{2} \cdot 8 + 4 = -4 + 4 = 0$, άρα το Γ είναι σημείο της ευθείας.

\item Θα πρέπει:
\begin{align*}
5 = -\dfrac{1}{2} \cdot k + 4 \ &\Leftrightarrow 
1 = -\dfrac{1}{2} \cdot k \ \Leftrightarrow \\
2 = -k \ &\Leftrightarrow 
k = -2.
\end{align*}
\end{alist}
