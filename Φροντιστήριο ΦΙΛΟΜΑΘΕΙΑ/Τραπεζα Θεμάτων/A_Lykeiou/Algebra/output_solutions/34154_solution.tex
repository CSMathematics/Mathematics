\lysh{34154}{2}

\begin{alist}
\item Είναι:

\begin{gather*}
A + B = \dfrac{1}{3 - \sqrt{7}} + \dfrac{1}{3 + \sqrt{7}} = \dfrac{3 + \sqrt{7}}{(3 + \sqrt{7})(3 - \sqrt{7})} + \dfrac{3 - \sqrt{7}}{(3 + \sqrt{7})(3 - \sqrt{7})} = \\
= \dfrac{3 + \sqrt{7} + 3 - \sqrt{7}}{(3 + \sqrt{7})(3 - \sqrt{7})} = \dfrac{6}{3^2 - (\sqrt{7})^2} = \dfrac{6}{9 - 7} = \dfrac{6}{2} = 3.
\end{gather*}

Ισχύει ότι:

\[A \cdot B = \dfrac{1}{3 - \sqrt{7}} \cdot \dfrac{1}{3 + \sqrt{7}} = \dfrac{1}{(3 - \sqrt{7})(3 + \sqrt{7})} = \dfrac{1}{3^2 - (\sqrt{7})^2} = \dfrac{1}{9 - 7} = \dfrac{1}{2}.\]

\item Η ζητούμενη εξίσωση είναι της μορφής: $x^2 - Sx + P = 0$ με $S = A + B = 3$ και $P = A \cdot B = \dfrac{1}{2}$.Τελικά μια ζητούμενη εξίσωση είναι η:
$x^2 - 3x + \dfrac{1}{2} = 0$.
\end{alist}
