    \item 6.1. Η Έννοια της Συνάρτησης
\end{ylh}
Δίνεται η συνάρτηση: \[ f(x) = \begin{cases} 2x + 4, & x < 0 \\ x - 1, & x \ge 0 \end{cases} \]

\begin{alist}
\item Να δείξετε ότι $f(-1) = f(3)$.
\monades{13}
\item Να προσδιορίσετε τις τιμές του $x \in \mathbb{R}$, ώστε: $f(x) = 0$.
\monades{12}
\end{alist}
