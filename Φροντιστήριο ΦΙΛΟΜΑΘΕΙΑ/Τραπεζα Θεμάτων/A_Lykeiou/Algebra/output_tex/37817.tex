Δίνεται η παράσταση ${A} = x^4 + \dfrac{x^4 - 4}{x^2 - 2}$, $x \neq \pm \sqrt{2}$.

\begin{alist}
\item Να δείξετε ότι ${A} = x^4 + x^2 + 2$.
\monades{15}
\item

\begin{rlist}
\item Να αιτιολογήσετε γιατί ${A} > 0$ για κάθε $x \neq \pm \sqrt{2}$
\monades{5}
\item Για ποια τιμή του $x$ η παράσταση ${A}$ παίρνει τη μικρότερη τιμή της;
\monades{5}
\end{rlist}
\end{alist}
