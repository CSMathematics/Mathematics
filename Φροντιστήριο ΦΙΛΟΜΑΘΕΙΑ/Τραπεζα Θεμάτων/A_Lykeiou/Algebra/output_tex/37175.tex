\begin{ylh}{2}
    \item 6.1. Η Έννοια της Συνάρτησης
\end{ylh}
Δίνεται η συνάρτηση $f,$ με \(f(x) = \left\{ \begin{matrix}
8 - x, & \text{αν }x < 0 \\
2x + 5, & \text{αν }x \geq 0
\end{matrix} \right.\ .\)

\begin{alist}
\item Να δείξετε ότι $f( - 5) = f(4).$
\monades{13}
\item Να βρείτε τις τιμές του $x\mathbb{\in R,}$ώστε $f(x) = 9.$
\monades{12}
\end{alist}
