\Askhsh{14602}{3}
\begin{ylh}{2}
    \item 2.2. Διάταξη Πραγματικών Αριθμών
\end{ylh}
Αν $0 < a < 1$, τότε:

\begin{alist}
\item Να αποδείξετε ότι $0 < a^3 < a$.
\monades{13}
\item Να διατάξετε από τον μικρότερο προς τον μεγαλύτερο τους αριθμούς:

$0, a^3, 1, a, \dfrac{1}{a}$.
\monades{12}
\end{alist}
