    \item 5.2. Αριθμητική πρόοδος
\end{ylh}

\begin{alist}
\item

\begin{rlist}
\item Να γράψετε τον πρώτο όρο και τη διαφορά της προόδου.

\monades{4}

\monades{8}
\end{rlist}
