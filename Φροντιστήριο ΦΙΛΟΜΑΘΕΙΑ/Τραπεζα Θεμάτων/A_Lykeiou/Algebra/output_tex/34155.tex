\begin{ylh}{2}
    \item 2.4. Ρίζες Πραγματικών Αριθμών
\end{ylh}
Αν είναι $A = \sqrt[3]{5}, B = \sqrt{3}, \Gamma = \sqrt[6]{5}$, τότε:
\begin{alist}
\item Να αποδείξετε ότι $A \cdot B \cdot \Gamma = \sqrt{15}$. \monades{15}
\item Να συγκρίνετε τους αριθμούς $A, B$. \monades{10}
\end{alist}
