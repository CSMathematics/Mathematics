Δίνεται η συνάρτηση: \(f(x) = \left\{ \begin{array}{r}
2x - 1,\ \ x < 0\  \\
x^{2} + 1\ ,\ \ x \geq 0
\end{array} \right.\ \).

\begin{alist}
\item Να βρείτε τις τιμές της συνάρτησης $f( - 1)$ και $f(1)$.
\monades{12}
\item Για $\ \ x \geq 0$ να λύσετε την ανίσωση: $\ \ f(x) \geq 2$.
\monades{13}
\end{alist}
