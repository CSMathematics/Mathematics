\begin{alist}
\item Να δείξετε ότι:
\begin{rlist}
\item $f(-a) = 0$.
\monades{4}

\monades{4}
\end{rlist}

\item Αν ένας θετικός αριθμός $a$ είναι μεγαλύτερος από τον αντίστροφό του και η διαφορά τους ξεπερνάει τη μία μονάδα, να δείξετε ότι $a > \dfrac{1+\sqrt{5}}{2}$.\monades{6}
\item Να δείξετε ότι $|AB| > 1 + \sqrt{5}$.\monades{5}
\end{alist}
