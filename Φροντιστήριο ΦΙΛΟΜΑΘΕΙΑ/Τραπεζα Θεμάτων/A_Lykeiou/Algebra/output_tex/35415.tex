\begin{ylh}{2}
    \item 2.3. Απόλυτη Τιμή Πραγματικού Αριθμού
    \item 4.1. Ανισώσεις 1ου Βαθμού
\end{ylh}
Δίνεται η παράσταση: $\mathrm{A} = |x - 1| - |x - 2|$.

\begin{alist}
\item Για $1 < x < 2$, να δείξετε ότι: $\mathrm{A} = 2x - 3$.
\monades{13}
\item Για $x < 1$, να δείξετε ότι η παράσταση $\mathrm{A}$ έχει σταθερή τιμή (ανεξάρτητη του $x$), την οποία και να προσδιορίσετε.
\monades{12}
\end{alist}
