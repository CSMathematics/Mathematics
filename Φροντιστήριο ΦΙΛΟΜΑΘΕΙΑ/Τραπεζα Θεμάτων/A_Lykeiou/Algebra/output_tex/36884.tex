\begin{ylh}{2}
    \item 2.1. Οι Πράξεις και οι Ιδιότητές τους
\end{ylh}

\begin{alist}
\item Να δείξετε ότι για οποιουσδήποτε πραγματικούς αριθμούς $x, y$ ισχύει:
$(x - 1)^2 + (y + 3)^2 = x^2 + y^2 - 2x + 6y + 10$.
\monades{12}
\item Να βρείτε τους αριθμούς $x, y$, ώστε: $x^2 + y^2 - 2x + 6y + 10 = 0$.
\monades{13}
\end{alist}
