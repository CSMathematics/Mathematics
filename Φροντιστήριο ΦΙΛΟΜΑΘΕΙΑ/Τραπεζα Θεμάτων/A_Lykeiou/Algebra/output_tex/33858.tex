    \item 5.2. Αριθμητική πρόοδος

\begin{alist}

\monades{8}
\item Αν επιπλέον ο πρώτος όρος της είναι $a_{1} = 2$, τότε:
\begin{rlist}
\item
  Να βρείτε την τιμή του $\kappa$ και να αποδείξετε ότι$\ \omega = 7$.
\monades{8}
\item
  Να εξετάσετε αν ο αριθμός $72$ είναι όρος της προόδου.
\monades{9}
\end{rlist}
\end{alist}
