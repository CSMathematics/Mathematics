    \item 3.7. Κύκλος - Μεσοκάθετος - Διχοτόμος
    \item 5.3. Ορθογώνιο.
\end{ylh}
Έστω $\varepsilon_1, \varepsilon_2$ δυο κάθετες ευθείες που τέμνονται στο $\mathrm{O}$ και τυχαίο σημείο $\mathrm{M}$ του επιπέδου που δεν ανήκει στις ευθείες.
\begin{alist}
    \item Αν $\mathrm{M}_1$ είναι το συμμετρικό του $\mathrm{M}$ ως προς την $\varepsilon_1$ και $\mathrm{M}_2$ το συμμετρικό του $\mathrm{M}_1$ ως προς την $\varepsilon_2$, να αποδείξετε ότι:
    \begin{rlist}
        \item $\mathrm{O}\mathrm{M} = \mathrm{O}\mathrm{M}_1 = \mathrm{O}\mathrm{M}_2$ \monades{6}
        \item Τα σημεία $\mathrm{M}, \mathrm{O}$ και $\mathrm{M}_2$ είναι συνευθειακά. \monades{8}
        \item Το τρίγωνο $\mathrm{M}\mathrm{M}_1\mathrm{M}_2$ είναι ορθογώνιο. \monades{6}
    \end{rlist}
    \item Αν $\mathrm{M}_3$ είναι το συμμετρικό σημείο του $\mathrm{M}$ ως προς την $\varepsilon_2$, τι είδους παραλληλόγραμμο είναι το $\mathrm{M}\mathrm{M}_1\mathrm{M}_2\mathrm{M}_3$; Να αιτιολογήσετε την απάντησή σας. \monades{5}
\end{alist}
