\begin{ylh}{1}
    \item 3.2. 1ο Κριτήριο ισότητας τριγώνων
    \item 3.6. Κριτήρια ισότητας ορθογώνιων τριγώνων
    \item 4.6. Άθροισμα γωνιών τριγώνου.
\end{ylh}
Δίνεται ορθογώνιο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ με $(\hat{\mathrm{A}} = 90^\circ)$ και $\hat{\mathrm{B}} = 50^\circ$, το ύψος του $\mathrm{A}\Delta$ και σημείο $\mathrm{E}$ στην $\Delta\Gamma$ ώστε $\Delta\mathrm{E} = \mathrm{B}\Delta$. Το σημείο $\mathrm{Z}$ είναι η προβολή του $\Gamma$ στην $\mathrm{A}\mathrm{E}$.
\begin{alist}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item Το τρίγωνο $\mathrm{A}\mathrm{B}\mathrm{E}$ είναι ισοσκελές. \monades{6}
        \item $\Gamma\hat{\mathrm{A}}\mathrm{E} = 10^\circ$. \monades{10}
    \end{rlist}
    \item Να υπολογίσετε τις γωνίες του τριγώνου $\mathrm{Z}\Gamma\mathrm{E}$. \monades{9}
\end{alist}
