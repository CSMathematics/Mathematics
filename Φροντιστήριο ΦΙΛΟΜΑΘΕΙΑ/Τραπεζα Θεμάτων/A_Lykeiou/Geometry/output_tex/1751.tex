\begin{ylh}{1}
    \item 3.4. 3ο Κριτήριο ισότητας τριγώνων
    \item 3.6. Κριτήρια ισότητας ορθογώνιων τριγώνων
    \item 3.15. Εφαπτόμενα τμήματα.
\end{ylh}
Έστω ότι ο κύκλος $(\mathrm{O}, \rho)$ εφάπτεται των πλευρών του τριγώνου $\mathrm{P}\Gamma\mathrm{E}$ στα σημεία $\mathrm{A}, \Delta$ και $\mathrm{B}$.
\begin{alist}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item $\mathrm{P}\Gamma = \Gamma\Delta + \mathrm{A}\mathrm{P}$ \monades{6}
        \item $\mathrm{P}\Gamma - \Gamma\Delta = \mathrm{P}\mathrm{E} - \Delta\mathrm{E}$ \monades{8}
    \end{rlist}
    \item Αν $\mathrm{A}\Gamma = \mathrm{B}\mathrm{E}$, να αποδείξετε ότι
    \begin{rlist}
        \item Το τρίγωνο $\mathrm{P}\Gamma\mathrm{E}$ είναι ισοσκελές. \monades{6}
        \item Τα σημεία $\mathrm{P}, \mathrm{O}$ και $\Delta$ είναι συνευθειακά. \monades{5}
    \end{rlist}
\end{alist}
