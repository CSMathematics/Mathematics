\Askhsh{1819}{4}
\begin{ylh}{1}
    \item 3.4. 3ο Κριτήριο ισότητας τριγώνων
    \item 3.6. Κριτήρια ισότητας ορθογώνιων τριγώνων
    \item 4.6. Άθροισμα γωνιών τριγώνου.
\end{ylh}
Δίνεται ισόπλευρο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ και στην προέκταση της $\Gamma\mathrm{B}$ (προς το $\mathrm{B}$) θεωρούμε σημείο $\Delta$ τέτοιο ώστε $\mathrm{B}\Delta = \mathrm{B}\Gamma$, ενώ στην προέκταση της $\mathrm{B}\Gamma$ (προς το $\Gamma$) θεωρούμε σημείο $\mathrm{E}$ τέτοιο ώστε $\Gamma\mathrm{E} = \mathrm{B}\Gamma$. Φέρνουμε την κάθετη στην $\mathrm{E}\Delta$ στο σημείο $\mathrm{E}$, η οποία τέμνει την προέκταση της $\Delta\mathrm{A}$ στο $\mathrm{Z}$.
\begin{alist}
    \item Να υπολογίσετε τις γωνίες των τριγώνων $\Gamma\mathrm{A}\mathrm{E}$ και $\mathrm{B}\Delta\mathrm{A}$. \monades{8}
    \item Να αποδείξετε ότι η $\Gamma\mathrm{Z}$ είναι μεσοκάθετος του $\mathrm{A}\mathrm{E}$. \monades{12}
    \item Να αποδείξετε ότι $\mathrm{A}\mathrm{B} \parallel \Gamma\mathrm{Z}$. \monades{5}
\end{alist}
