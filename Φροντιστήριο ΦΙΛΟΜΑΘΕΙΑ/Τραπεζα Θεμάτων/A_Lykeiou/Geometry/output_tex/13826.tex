ΘΕΜΑ 2

Τα τρίγωνα ΑΒΚ και ΓΔΛ του σχήματος έχουν ΑΒ = ΓΔ = ΑΚ= ΓΛ και $\widehat{Α}$ = $\widehat{\Gamma}$.

\begin{alist}
\item Να αποδείξετε ότι τα τρίγωνα ΑΒΚ και ΓΔΛ είναι ίσα και ότι έχουν ΒΚ = ΔΛ.

\monades{12}

\item Έστω ότι Λ και Κ είναι τα μέσα των ΒΚ και ΔΛ αντίστοιχα:

\begin{alist}
\item
  Να εξετάσετε αν τα τμήματα ΒΛ, ΛΚ και ΚΔ είναι ίσα. Να αιτιολογήσετε την απάντησή σας. \monades{5}
\item
  Να αποδείξετε ότι οι ΑΛ και ΓΚ είναι κάθετες στην ευθεία ΚΛ.

\monades{8}


\begin{center}
\begin{tikzpicture}[scale=1.5]
\tkzDefPoint(0,0){L_line_start}
\tkzDefPoint(6,0){L_line_end}
\tkzDefPoint(2,0){K}
\tkzDefPoint(4,0){L}

% Define A, B such that ABK is a triangle. B is on line, A is above.
\tkzDefPoint(0.5,0){B}
\tkzDefPoint(1.5,2){A}

% Define \Gamma, \Delta. Since AB=CD, AK=CL, \widehat{A}=\widehat{\Gamma}.
% Congruent triangles ABK and \Gamma \Delta L
% \Gamma corresponds to A, \Delta corresponds to B, L corresponds to K.
% \Delta is on the line, \Gamma is above.
\tkzDefPoint(5.5,0){D} % \Delta
\tkzDefPoint(4.5,2){C} % \Gamma

\draw[l3] (B)--(L_line_end);
\draw[l3] (A)--(B);
\draw[l3] (A)--(K);
\draw[l3] (C)--(D);
\draw[l3] (C)--(L);

\draw[l3, dashed] (A)--(L);
\draw[l3, dashed] (C)--(K);

\tkzDrawPoints(A,B,C,D,K,L)
\tkzLabelPoint[above](A){$A$}
\tkzLabelPoint[below](B){$B$}
\tkzLabelPoint[above](C){$\Gamma$}
\tkzLabelPoint[below](D){$\Delta$}
\tkzLabelPoint[below](K){$K$}
\tkzLabelPoint[below](L){$\Lambda$}

\tkzMarkSegments[mark=s|](A,B C,D)
\tkzMarkSegments[mark=s||](A,K C,L)
\tkzMarkAngle[size=0.4,mark=none](B,A,K)
\tkzMarkAngle[size=0.4,mark=none](D,C,L)
\end{tikzpicture}
\end{center}
