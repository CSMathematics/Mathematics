\begin{ylh}{1}
    \item 5.5. Τετράγωνο
    \item 5.6. Εφαρμογές στα τρίγωνα
    \item 5.8. Το ορθόκεντρο τριγώνου.
\end{ylh}
Σε τετράγωνο $\mathrm{A}\mathrm{B}\Gamma\Delta$ προεκτείνουμε τη διαγώνιο $\mathrm{B}\Delta$ (προς το $\Delta$) κατά τμήμα $\Delta\mathrm{E} = \Delta\mathrm{B}$. Έστω $\mathrm{M}$ το μέσο της $\mathrm{A}\Delta$ και $\mathrm{N}$ το σημείο τομής των ευθειών $\mathrm{A}\mathrm{E}$ και $\Gamma\Delta$.
\begin{alist}
    \item Να αποδείξετε ότι $\Delta\mathrm{N} = \Delta\mathrm{M}$. \monades{6}
    \item Να υπολογίσετε τις γωνίες του τριγώνου $\mathrm{N}\mathrm{M}\Delta$. \monades{5}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item $\mathrm{M}\mathrm{N} \perp \mathrm{A}\Gamma$ \monades{7}
        \item $\Gamma\mathrm{M} \perp \mathrm{A}\mathrm{N}$ \monades{7}
    \end{rlist}
\end{alist}
