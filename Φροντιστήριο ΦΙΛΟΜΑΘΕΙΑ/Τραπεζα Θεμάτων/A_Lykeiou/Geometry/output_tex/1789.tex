\Askhsh{1789}{4}
\begin{ylh}{1}
    \item 5.2. Παραλληλόγραμμα
    \item 5.6. Εφαρμογές στα τρίγωνα
    \item 5.9. Μια ιδιότητα του ορθογώνιου τριγώνου
    \item 5.11. Ισοσκελές τραπέζιο.
\end{ylh}
Δίνεται ορθογώνιο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ με τη γωνία $\hat{\mathrm{A}}$ ορθή, και τυχαίο σημείο $\Delta$ της πλευράς $\mathrm{A}\mathrm{B}$. Έστω $\mathrm{K}, \mathrm{M}, \mathrm{N}$ τα μέσα των $\Gamma\Delta, \mathrm{B}\Gamma, \mathrm{B}\Delta$ αντίστοιχα.
Να αποδείξετε ότι:
\begin{alist}
    \item Το τετράπλευρο $\mathrm{K}\mathrm{M}\mathrm{N}\Delta$ είναι παραλληλόγραμμο. \monades{8}
    \item Το τετράπλευρο $\mathrm{A}\mathrm{K}\mathrm{M}\mathrm{N}$ είναι ισοσκελές τραπέζιο. \monades{9}
    \item Η διάμεσος του τραπεζίου $\mathrm{A}\mathrm{K}\mathrm{M}\mathrm{N}$ είναι ίση με $\frac{\mathrm{A}\mathrm{B}}{2}$. \monades{8}
\end{alist}
