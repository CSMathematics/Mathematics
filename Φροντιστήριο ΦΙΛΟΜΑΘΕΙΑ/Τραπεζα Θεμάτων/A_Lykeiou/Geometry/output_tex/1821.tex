    \item 5.4. Ρόμβος
    \item 5.9. Μια ιδιότητα του ορθογώνιου τριγώνου
    \item 5.10. Τραπέζιο.
\end{ylh}
Δίνεται ορθογώνιο τραπέζιο $\mathrm{A}\mathrm{B}\Gamma\Delta$ ($\hat{\mathrm{A}} = \hat{\Delta} = 90^\circ$) με $\mathrm{B}\Gamma = \Gamma\Delta = 2\mathrm{A}\mathrm{B}$ και $\mathrm{K}, \Lambda$ τα μέσα των $\mathrm{B}\Gamma$ και $\Gamma\Delta$. Η παράλληλη από το $\mathrm{K}$ προς την $\mathrm{A}\mathrm{B}$ τέμνει την $\mathrm{A}\Lambda$ στο $\mathrm{Z}$.
Να αποδείξετε ότι:
\begin{alist}
    \item Το $\mathrm{Z}$ είναι μέσο του $\mathrm{A}\Lambda$. \monades{6}
    \item $\mathrm{B}\Gamma = 2 \Delta\mathrm{Z}$. \monades{6}
    \item Το τετράπλευρο $\mathrm{Z}\mathrm{K}\Gamma\Lambda$ είναι ρόμβος. \monades{5}
    \item $\mathrm{A}\hat{\mathrm{K}}\Lambda = 90^\circ$. \monades{8}
\end{alist}
