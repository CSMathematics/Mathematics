\begin{alist}

\item Να αποδείξετε ότι ΒΖ = ΗΓ. \monades{09}

\item Αν η γωνία Α = 60\textsuperscript{ο} και Μ το μέσο ης ΑΔ, να αποδείξετε ότι ΗΜ = ΖΔ.
