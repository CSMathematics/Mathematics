\begin{ylh}{1}
    \item 3.2. 1ο Κριτήριο ισότητας τριγώνων
    \item 5.5. Τετράγωνο.
\end{ylh}
Δίνεται τετράγωνο $\mathrm{A}\mathrm{B}\Gamma\Delta$ και εντός αυτού ισόπλευρο τρίγωνο $\mathrm{M}\mathrm{B}\Gamma$. Αν η προέκταση της $\mathrm{A}\mathrm{M}$ τέμνει την $\mathrm{B}\Delta$ στο σημείο $\mathrm{E}$, να αποδείξετε ότι:
\begin{alist}
    \item $\Delta\hat{\mathrm{A}}\mathrm{E} = 15^\circ$. \monades{8}
    \item Τα τρίγωνα $\Delta\mathrm{A}\mathrm{E}$ και $\Delta\mathrm{E}\Gamma$ είναι ίσα. \monades{8}
    \item Η $\Gamma\mathrm{E}$ είναι διχοτόμος της γωνίας $\Delta\hat{\Gamma}\mathrm{M}$. \monades{9}
\end{alist}
