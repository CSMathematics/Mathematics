    \item 4.2. Τέμνουσα δύο ευθειών - Ευκλείδειο αίτημα
    \item 5.2. Παραλληλόγραμμα
    \item 5.3. Ορθογώνιο
    \item 5.6. Εφαρμογές στα τρίγωνα.
\end{ylh}
Δίνεται τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ και η διάμεσός του $\mathrm{A}\Delta$. Έστω $\mathrm{E}, \mathrm{H}$ και $\Theta$ τα μέσα των $\mathrm{B}\Delta, \mathrm{A}\Delta$ και $\mathrm{A}\Gamma$ αντίστοιχα.
\begin{alist}
    \item Να αποδείξετε ότι το τετράπλευρο $\Delta\mathrm{E}\mathrm{H}\Theta$ είναι παραλληλόγραμμο. \monades{10}
    \item Να βρείτε τη σχέση των πλευρών $\mathrm{A}\mathrm{B}$ και $\mathrm{B}\Gamma$ του τριγώνου $\mathrm{A}\mathrm{B}\Gamma$, ώστε το παραλληλόγραμμο $\Delta\mathrm{E}\mathrm{H}\Theta$ να είναι ρόμβος. \monades{10}
    \item Στην περίπτωση που το τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ είναι ορθογώνιο (η γωνία $\hat{\mathrm{B}}$ ορθή), να βρείτε το είδος του παραλληλογράμμου $\Delta\mathrm{E}\mathrm{H}\Theta$. \monades{5}
\end{alist}
