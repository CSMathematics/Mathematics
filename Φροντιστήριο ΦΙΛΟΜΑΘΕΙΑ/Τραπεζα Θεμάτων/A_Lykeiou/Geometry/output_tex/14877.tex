ΘΕΜΑ 2

Στο τρίγωνο ΑΒΓ του παρακάτω σχήματος τα σημεία Δ και Ε είναι τα μέσα των πλευρών ΑΒ και ΑΓ αντίστοιχα, ΑE=8 , ΕΔ=9 και ΔΒ=10 .

\begin{alist}
\item Να αποδείξετε ότι το τετράπλευρο ΔΕΓΒ είναι τραπέζιο. \monades{8}

\item Να υπολογίσετε το μήκος της πλευράς ΒΓ. \monades{8}

\item Να συγκρίνετε τις περιμέτρους του τριγώνου ΑΒΓ και του τετραπλεύρου ΔΕΓΒ. \monades{9}


\begin{center}
\begin{tikzpicture}[scale=0.3]
% Triangle with sides 20, 18, 16.
% Let A be at origin. AB = 20, BC = 18, AC = 16.
% But conventionally BC is base. Let C=(0,0), B=(18,0).
% Or just use standard construction:
\tkzDefPoint(0,0){B}
\tkzDefPoint(18,0){C} % B\Gamma = 18
% AB = 20, AC = 16
\tkzInterCC[R](B,20)(C,16) \tkzGetPoints{A_temp}{A}

\tkzDefMidPoint(A,B) \tkzGetPoint{D}
\tkzDefMidPoint(A,C) \tkzGetPoint{E}

\draw[l3] (A)--(B)--(C)--cycle;
\draw[l3] (D)--(E);

\tkzDrawPoints(A,B,C,D,E)
\tkzLabelPoint[above](A){$A$}
\tkzLabelPoint[below left](B){$B$}
\tkzLabelPoint[below right](C){$\Gamma$}
\tkzLabelPoint[above left](D){$\Delta$}
\tkzLabelPoint[above right](E){$E$}

\tkzLabelSegment[left](A,D){$10$}
\tkzLabelSegment[left](D,B){$10$}
\tkzLabelSegment[right](A,E){$8$}
\tkzLabelSegment[right](E,C){$8$}
\tkzLabelSegment[above](D,E){$9$}
\end{tikzpicture}
\end{center}
