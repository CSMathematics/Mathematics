\begin{alist}
    \item 
    \begin{rlist}
        \item Αφού για το σημείο $\mathrm{E}$ ισχύει $\Delta\mathrm{E} = \mathrm{B}\Delta$ (υπόθεση), άρα το $\Delta$ είναι μέσο του τμήματος $\mathrm{B}\mathrm{E}$. Στο τρίγωνο $\mathrm{A}\mathrm{B}\mathrm{E}$ το $\mathrm{A}\Delta$ είναι ύψος και διάμεσος. Οπότε το τρίγωνο $\mathrm{A}\mathrm{B}\mathrm{E}$ είναι ισοσκελές με ίσες πλευρές τις $\mathrm{A}\mathrm{B}$ και $\mathrm{A}\mathrm{E}$.
        
        \item Επειδή το τρίγωνο $\mathrm{A}\mathrm{B}\mathrm{E}$ είναι ισοσκελές με βάση τη $\mathrm{B}\mathrm{E}$ οπότε οι γωνίες της βάσης θα είναι ίσες, δηλαδή $\mathrm{A}\hat{\mathrm{E}}\mathrm{B} = \hat{\mathrm{B}} = 50^\circ$.
        Για τις γωνίες του τριγώνου $\mathrm{A}\mathrm{B}\mathrm{E}$ ισχύει $\mathrm{E}\hat{\mathrm{A}}\mathrm{B} + \mathrm{A}\hat{\mathrm{E}}\mathrm{B} + \hat{\mathrm{B}} = 180^\circ$ και αφού $\mathrm{A}\hat{\mathrm{E}}\mathrm{B} = \hat{\mathrm{B}} = 50^\circ$ άρα και $\mathrm{E}\hat{\mathrm{A}}\mathrm{B} + 2 \cdot 50^\circ = 180^\circ$, οπότε $\mathrm{E}\hat{\mathrm{A}}\mathrm{B} = 180^\circ - 100^\circ = 80^\circ$.
        
        Στο ορθογώνιο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ έχουμε:
        $\Gamma\hat{\mathrm{A}}\mathrm{E} + \mathrm{E}\hat{\mathrm{A}}\mathrm{B} = 90^\circ$ (αφού $\hat{\mathrm{A}} = 90^\circ$) και αφού $\mathrm{E}\hat{\mathrm{A}}\mathrm{B} = 80^\circ$ τότε θα ισχύει $\Gamma\hat{\mathrm{A}}\mathrm{E} + 80^\circ = 90^\circ$, άρα $\Gamma\hat{\mathrm{A}}\mathrm{E} = 10^\circ$.
    \end{rlist}
    
    \item Ισχύει ότι $\Gamma\hat{\mathrm{E}}\mathrm{Z} = \mathrm{A}\hat{\mathrm{E}}\mathrm{B} = 50^\circ$ ως κατακορυφήν γωνίες. Επειδή $\Gamma\hat{\mathrm{Z}}\mathrm{E} = 90^\circ$, γιατί $\Gamma\mathrm{Z} \perp \mathrm{A}\mathrm{Z}$ λόγω της προβολής του σημείου $\Gamma$ στην $\mathrm{A}\mathrm{E}$, το τρίγωνο $\Gamma\mathrm{Z}\mathrm{E}$ είναι ορθογώνιο οπότε οι οξείες γωνίες του είναι συμπληρωματικές, δηλαδή
    $\Gamma\hat{\mathrm{E}}\mathrm{Z} + \mathrm{E}\hat{\Gamma}\mathrm{Z} = 90^\circ \Leftrightarrow 50^\circ + \mathrm{E}\hat{\Gamma}\mathrm{Z} = 90^\circ$ οπότε $\mathrm{E}\hat{\Gamma}\mathrm{Z} = 40^\circ$.
    Άρα οι γωνίες του τριγώνου $\mathrm{Z}\Gamma\mathrm{E}$ είναι $90^\circ, 50^\circ, 40^\circ$.
\end{alist}
