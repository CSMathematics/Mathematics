\Askhsh{1757}{4}
\begin{ylh}{1}
    \item 5.2. Παραλληλόγραμμα
    \item 5.3. Ορθογώνιο
    \item 5.10. Τραπέζιο.
\end{ylh}
Θεωρούμε τραπέζιο $\mathrm{A}\mathrm{B}\Gamma\Delta$, τέτοιο ώστε $\hat{\mathrm{A}} = \hat{\Delta} = 90^\circ$, $\mathrm{A}\mathrm{B} = \frac{1}{4}\Delta\Gamma$ και $\mathrm{A}\mathrm{B} = \frac{1}{3}\mathrm{A}\Delta$. Επιπλέον, φέρνουμε $\mathrm{B}\mathrm{E} \perp \Delta\Gamma$.
\begin{alist}
    \item Να αποδείξετε ότι το τετράπλευρο $\mathrm{A}\mathrm{B}\mathrm{E}\Delta$ είναι ορθογώνιο. \monades{6}
    \item Να αποδείξετε ότι το τρίγωνο $\mathrm{B}\mathrm{E}\Gamma$ είναι ορθογώνιο και ισοσκελές. \monades{10}
    \item Αν $\mathrm{K}, \Lambda$ είναι τα μέσα των $\mathrm{B}\Delta$ και $\mathrm{A}\Gamma$ αντίστοιχα, να αποδείξετε ότι η $\mathrm{A}\Gamma$ διέρχεται από το μέσο του ευθυγράμμου τμήματος $\mathrm{B}\mathrm{K}$. \monades{9}
\end{alist}
