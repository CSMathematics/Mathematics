\begin{ylh}{1}
    \item 3.6. Κριτήρια ισότητας ορθογώνιων τριγώνων
    \item 4.6. Άθροισμα γωνιών τριγώνου.
\end{ylh}
Σε ορθογώνιο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ ($\hat{\mathrm{A}} = 90^\circ$) φέρνουμε τη διχοτόμο του $\mathrm{A}\Delta$. Έστω $\Delta\mathrm{K}$ και $\Delta\mathrm{P}$ οι προβολές του $\Delta$ στις $\mathrm{A}\mathrm{B}$ και $\mathrm{A}\Gamma$ αντίστοιχα. Η κάθετη της $\mathrm{B}\Gamma$ στο σημείο $\Delta$ τέμνει την πλευρά $\mathrm{A}\Gamma$ στο $\mathrm{E}$ και την προέκταση της πλευράς $\mathrm{A}\mathrm{B}$ (προς το $\mathrm{B}$) στο σημείο $\mathrm{Z}$.
\begin{alist}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item $\hat{\mathrm{B}} = \Delta\hat{\mathrm{E}}\Gamma$. \monades{8}
        \item $\Delta\mathrm{E} = \Delta\mathrm{B}$. \monades{8}
    \end{rlist}
    \item Να υπολογίσετε τη γωνία $\Delta\hat{\Gamma}\mathrm{Z}$. \monades{9}
\end{alist}
