\begin{ylh}{1}
    \item 3.3. 2ο Κριτήριο ισότητας τριγώνων
    \item 4.2. Τέμνουσα δύο ευθειών - Ευκλείδειο αίτημα.
\end{ylh}
Σε τραπέζιο $\mathrm{A}\mathrm{B}\Gamma\Delta$ ($\mathrm{A}\mathrm{B} \parallel \Gamma\Delta$) ισχύει $\mathrm{A}\mathrm{B} + \Gamma\Delta = \mathrm{A}\Delta$. Αν η διχοτόμος της γωνίας $\hat{\mathrm{A}}$ τέμνει την $\mathrm{B}\Gamma$ στο $\mathrm{E}$ και την προέκταση της $\Delta\Gamma$ στο $\mathrm{Z}$, να αποδείξετε ότι:
\begin{alist}
    \item Το τρίγωνο $\Delta\mathrm{A}\mathrm{Z}$ είναι ισοσκελές. \monades{7}
    \item Το $\mathrm{E}$ είναι το μέσο της $\mathrm{B}\Gamma$. \monades{10}
    \item Η $\Delta\mathrm{E}$ είναι διχοτόμος της γωνίας $\hat{\Delta}$ του τραπεζίου. \monades{8}
\end{alist}
