\begin{ylh}{1}
    \item 3.6. Κριτήρια ισότητας ορθογώνιων τριγώνων
    \item 4.6. Άθροισμα γωνιών τριγώνου
    \item 5.4. Ρόμβος
    \item 5.9. Μια ιδιότητα του ορθογώνιου τριγώνου.
\end{ylh}
Έστω ορθογώνιο τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$ με $\hat{\mathrm{A}} = 90^\circ$ και $\hat{\mathrm{B}} = 60^\circ$. Η διχοτόμος της γωνίας $\hat{\mathrm{B}}$ τέμνει την $\mathrm{A}\Gamma$ στο $\mathrm{Z}$. Τα σημεία $\mathrm{M}$ και $\mathrm{K}$ είναι τα μέσα των $\mathrm{B}\mathrm{Z}$ και $\mathrm{B}\Gamma$ αντίστοιχα. Αν το τμήμα $\Gamma\Lambda$ είναι κάθετο στη διχοτόμο $\mathrm{B}\delta$ να αποδείξετε:
\begin{alist}
    \item Το τρίγωνο $\mathrm{B}\mathrm{Z}\Gamma$ είναι ισοσκελές. \monades{6}
    \item Το τετράπλευρο $\mathrm{A}\mathrm{M}\mathrm{K}\mathrm{Z}$ είναι ρόμβος. \monades{6}
    \item $\Gamma\mathrm{Z} = 2\mathrm{Z}\mathrm{A}$. \monades{7}
    \item $\mathrm{B}\Lambda = \mathrm{A}\Gamma$. \monades{6}
\end{alist}
