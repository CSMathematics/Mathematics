\Askhsh{1802}{4}
\begin{ylh}{1}
    \item 4.6. Άθροισμα γωνιών τριγώνου
    \item 5.6. Εφαρμογές στα τρίγωνα.
\end{ylh}
Δίνεται τρίγωνο $\mathrm{A}\mathrm{B}\Gamma$, $\mathrm{A}\mathrm{M}$ διάμεσός του και $\mathrm{K}$ το μέσο του $\mathrm{A}\mathrm{M}$. Αν η προέκταση της $\mathrm{B}\mathrm{K}$ τέμνει την $\mathrm{A}\Gamma$ στο σημείο $\mathrm{N}$, και $\Lambda$ είναι το μέσο του $\Gamma\mathrm{N}$, να αποδείξετε ότι:
\begin{alist}
    \item Το σημείο $\mathrm{N}$ είναι μέσο του $\mathrm{A}\Lambda$. \monades{9}
    \item $\mathrm{K}\hat{\mathrm{M}}\Gamma = \mathrm{M}\hat{\mathrm{B}}\mathrm{K} + \mathrm{A}\hat{\mathrm{K}}\mathrm{N}$. \monades{9}
    \item $\mathrm{B}\mathrm{K} = 3\mathrm{K}\mathrm{N}$. \monades{7}
\end{alist}
