\begin{alist}
    \item Να αποδείξετε ότι το τετράπλευρο $\mathrm{E}\mathrm{Z}\mathrm{H}\Theta$ είναι ρόμβος. \monades{15}
    \item Αν το αρχικό τετράπλευρο $\mathrm{A}\mathrm{B}\Gamma\Delta$ είναι τετράγωνο, τότε το $\mathrm{E}\mathrm{Z}\mathrm{H}\Theta$ τι είδους παραλληλόγραμμο είναι; Δικαιολογήστε την απάντησή σας. \monades{10}
\end{alist}
