του τριγώνου ΑΒΓ κατασκευάζουμε το ισόπλευρο τρίγωνο ΑΓΔ.

\begin{alist}
\item Nα υπολογίσετε το μέτρο των γωνιών Β, Γ του τριγώνου ΑΒΓ. \monades{5}

\item Να αποδείξετε ότι το τρίγωνο ΑΒΔ είναι ισοσκελές. \monades{8}

\item Να υπολογίσετε το μέτρο της γωνίας ΑΒΔ. \monades{12}
