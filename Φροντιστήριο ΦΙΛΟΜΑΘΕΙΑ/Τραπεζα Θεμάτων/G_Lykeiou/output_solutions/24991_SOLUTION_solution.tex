\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ
\begin{alist}
    \item [α)] Έστω $x_1, x_2 \in (0, +\infty)$ τέτοια ώστε $f(x_1) = f(x_2)$. Τότε
    \[
    -2\ln x_1 + 1 = -2\ln x_2 + 1
    \]
    \[
    -2\ln x_1 = -2\ln x_2
    \]
    \[
    \ln x_1 = \ln x_2
    \]
    \[
    x_1 = x_2
    \]
    Άρα η συνάρτηση $f$ είναι συνάρτηση $1-1$ και επομένως αντιστρέφεται.

    \item [β)] Με $x > 0$ θέτουμε
    \[
    f(x) = y \Leftrightarrow -2\ln x + 1 = y \Leftrightarrow \ln x = \frac{1-y}{2} \Leftrightarrow x = e^{\frac{1-y}{2}} \Leftrightarrow f^{-1}(y) = e^{\frac{1-y}{2}}, y \in \mathbb{R},
    \]
    επομένως $f^{-1}(x) = e^{\frac{1-x}{2}}$, $x \in \mathbb{R}$.

    \item [γ)] Το πεδίο ορισμού της συνάρτησης $f$ είναι $D_f = (0, +\infty)$, ενώ το πεδίο ορισμού της συνάρτησης $g$ είναι $D_g = \mathbb{R}-\{0\}$ καθώς απαιτείται $x^2 > 0 \Leftrightarrow x \neq 0$. Επομένως δεν ισχύει γενικά $f = g$. Όμως για $x \in (0, +\infty)$ έχουμε $g(x) = 1 - \ln x^2 = 1 - 2\ln x = -2\ln x + 1 = f(x)$.
    Επομένως ισχύει $f = g$ για $x \in (0, +\infty)$.
\end{alist}
