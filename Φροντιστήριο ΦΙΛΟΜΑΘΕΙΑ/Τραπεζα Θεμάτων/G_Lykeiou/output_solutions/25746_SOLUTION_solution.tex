    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Η $g(x)=e^{-x}f(x)$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως γινόμενο παραγωγίσιμων με
\[ g'(x)=(e^{-x}f(x))' = e^{-x}f'(x)-e^{-x}f(x) = e^{-x}(f'(x)-f(x)) > 0 \]
για κάθε $x \in \mathbb{R}$, αφού $f'(x)>f(x)$, για κάθε $x\in\mathbb{R}$ και $e^{-x}>0$ για κάθε $x\in\mathbb{R}$.
Συνεπώς η $g$ είναι γνησίως αύξουσα στο $\mathbb{R}$. \monades{}
\item Για κάθε $x>0 \xrightarrow{g \uparrow} g(x) > g(0) \implies e^{-x}f(x) > e^0f(0) \implies e^{-x}f(x)>0 \xrightarrow{e^{-x}>0} f(x)>0$.
Για κάθε $x<0 \xrightarrow{g \uparrow} g(x) < g(0) \implies e^{-x}f(x) < e^0f(0) \implies e^{-x}f(x)<0 \xrightarrow{e^{-x}>0} f(x)<0$. \monades{}
\item Αφού $f(x)>0$ για κάθε $x>0$ από τη δοσμένη σχέση έχουμε ότι
\[ f'(x)>f(x)>0 \]
για κάθε $x>0$ που σημαίνει ότι η $f$ είναι γνησίως αύξουσα στο $(0,+\infty)$ άρα και 1-1 σε αυτό.
Είναι $|\eta\mu x|+1>0, |x|+1>0$ για κάθε $x\in\mathbb{R}$ οπότε η εξίσωση γίνεται
\[ f(|\eta\mu x|+1) = f(|x|+1) \xrightarrow{f:1-1} |\eta\mu x|+1 = |x|+1 \iff |\eta\mu x| = |x| \iff x=0. \] \monades{}
\item Το ζητούμενο εμβαδό είναι $E = \int_{0}^{1} |f(x)|dx$ και επειδή $f(x)>0$ για κάθε $x>0$
άρα και στο $[0,1]$ έχουμε $E = \int_{0}^{1} f(x)dx$. Από τη δοσμένη σχέση έχουμε:
\[ f'(x)>f(x) \implies f'(x)-f(x)>0 \implies \int_{0}^{1} (f'(x)-f(x))dx > 0 \]
\[ \implies \int_{0}^{1} f'(x)dx - \int_{0}^{1} f(x)dx > 0 \implies \left[f(x)\right]_0^1 > E \implies f(1)-f(0) > E \implies E < f(1). \] \monades{}
\end{alist}
