\noindent \textbf{Λύση}

\begin{alist}
\item Από τη σχέση $f'(x) \cdot f(x) = -x+1$ στο $(0,2)$ έχουμε
\[
    2f'(x) \cdot f(x) = -2x+2 \implies \PARENS{f^2(x)}' = \PARENS{-x^2+2x}'
\]
για κάθε $x \in (0,2)$. Όμως η συνάρτηση $f$ είναι συνεχής στο $[0,2]$, άρα και η $f^2$ είναι συνεχής στο $[0,2]$, οπότε $f^2(x) = -x^2+2x+c_1$ για κάθε $x \in [0,2]$.
Για $x=1$ έχουμε $f^2(1) = -1^2+2 \cdot 1+c_1 \implies 1 = -1+2+c_1 \implies c_1=0$.
Συνεπώς $f^2(x) = -x^2+2x$ για κάθε $x \in [0,2]$.

\item Εί ναι $-x^2+2x \ne 0$ για κάθε $x \in (0,2)$ άρα και $f(x) \ne 0$ για κάθε $x \in (0,2)$ και αφού η $f$ είναι συνεχής θα διατηρεί πρόσημο στο $(0,2)$. Όμως $f(1)>0$ οπότε $f(x)>0$ για κάθε $x \in (0,2)$ και άρα $f(x) = \sqrt{-x^2+2x}, x \in (0,2)$.
Επίσης $f^2(0) = -0^2+2 \cdot 0 = 0$ οπότε $f(0)=0$ και $f^2(2) = -2^2+2 \cdot 2 = 0$ οπότε $f(2)=0$. Τελικά $f(x) = \sqrt{-x^2+2x}, x \in [0,2]$.

\item Αν θέσουμε $y=f(x)$ με $y \ge 0$ έχουμε
\[
    y = \sqrt{-x^2+2x} \implies y^2 = -x^2+2x \implies x^2-2x+y^2 = 0 \implies (x^2-2x+1)+y^2 = 1 \implies (x-1)^2+y^2=1
\]
που σημαίνει ότι η γραφική παράσταση της $f$ αποτελείται από τα σημεία του κύκλου με κέντρο $\mathrm{K}(1,0)$ και ακτίνα $1$ που δεν είναι κάτω από το $xx'$, δηλαδή το ημικύκλιο που φαίνεται στο παρακάτω σχήμα.

\item Αφού $f(x) \ge 0$ για κάθε $x \in [0,2]$,το $\int_0^2 f(x)\mathrm{d}x$ παριστάνει το εμβαδόν μεταξύ της $C_f$, του $xx'$ και των κατακόρυφων ευθειών $x=0$ και $x=2$, δηλαδή το εμβαδόν του μισού κυκλικού δίσκου με κέντρο $\mathrm{K}(1,0)$ και ακτίνα $1$ (της σκιασμένης επιφάνειας του παραπάνω σχήματος) και άρα
\[
    \int_0^2 f(x)\mathrm{d}x = \frac{\pi \cdot \rho^2}{2} = \frac{\pi \cdot 1^2}{2} = \frac{\pi}{2}.
\]
\end{alist}
