\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\OnPar{ΛΥΣΗ}
\begin{alist}
\item Για κάθε $\alpha, \beta \in (0, +\infty)$ με $\alpha < \beta$ παίρνουμε $e^\alpha < e^\beta$ και $\ln \alpha < \ln \beta$ καθώς οι συναρτήσεις $e^x$ και $\ln x$ είναι γνησίως αύξουσες. Έτσι, $e^\alpha + \ln \alpha < e^\beta + \ln \beta$, άρα $f(\alpha) < f(\beta)$. Ώστε η $f$ είναι γνησίως αύξουσα στο $(0, +\infty)$.
\item Σύμφωνα με το α) ερώτημα, η εξίσωση $f(x) = 0$ έχει το πολύ μία λύση στο $(0, +\infty)$. Καθώς η $f$ είναι συνεχής (ως άθροισμα συνεχών) και γνησίως αύξουσα στο $(0, +\infty)$, για τιμές $x \in (0,1)$ το αντίστοιχο σύνολο τιμών της θα είναι το διάστημα
\[
f((0,1)) = \left(\lim_{x \to 0^+} f(x), \lim_{x \to 1^-} f(x)\right) = (-\infty, e)
\]
στο οποίο ανήκει ο αριθμός μηδέν, άρα υπάρχει τουλάχιστον ένα $x_0 \in (0,1)$ ώστε $f(x_0) = 0$, το οποίο $x_0$ είναι και μοναδικό λόγω της μονοτονίας της $f$.
Ώστε η $C_f$ τέμνει ακριβώς σε ένα σημείο $A(x_0, 0)$ τον άξονα $x'x$.
\item Για κάθε $x \in [x_0, 1]$ έχουμε $f(x_0) \le f(x) \le f(1)$, άρα $0 \le f(x) \le e$, καθώς η $f$ είναι γνησίως αύξουσα στο $(0, +\infty)$, άρα και στο $[x_0, 1]$.
\[
\begin{aligned}
\text{Άρα, } E &= \int_{x_0}^1 f(x)dx = \int_{x_0}^1 e^x dx + \int_{x_0}^1 \ln x dx \\
&= \left[e^x\right]_{x_0}^1 + \int_{x_0}^1 (x)'\ln x dx \\
&= e - e^{x_0} + \left[x\ln x\right]_{x_0}^1 - \int_{x_0}^1 x(\ln x)' dx \\
&= e - e^{x_0} + 1\ln 1 - x_0\ln x_0 - \int_{x_0}^1 x \frac{1}{x} dx \\
&= e - e^{x_0} + 0 - x_0\ln x_0 - \int_{x_0}^1 1 dx \\
&= e - e^{x_0} - x_0\ln x_0 - \left[x\right]_{x_0}^1 \\
&= e - e^{x_0} - x_0\ln x_0 - (1 - x_0) \\
&= e - e^{x_0} - x_0\ln x_0 - 1 + x_0 \\
&= e + (-e^{x_0} - x_0\ln x_0 - 1 + x_0) \\
&= e + (x_0 - 1) - \ln x_0 (x_0 - 1) \\
&= e + (x_0 - 1)(1 - \ln x_0).
\end{aligned}
\]
Χρησιμοποιήσαμε ότι $f(x_0) = 0$, άρα $\ln x_0 = -e^{x_0}$.
\end{alist}
