\begin{ylh}{1}
    \item [Συναρτήσεις - Ολοκληρωτικός Λογισμός]
\end{ylh}
\begin{alist}
    \item Η $f$ είναι συνεχής ως πηλίκο συνεχών. Επίσης από την υπόθεση είναι γνησίως αύξουσα.
    Επομένως το σύνολο τιμών της είναι $f((0,+\infty))=[f(0),\lim\limits_{x\to +\infty}f(x))$.
    Για να βρούμε το $\lim\limits_{x\to +\infty}f(x)$, θέτουμε $u=\sqrt{x}$. Τότε $\lim\limits_{x\to +\infty}u=\lim\limits_{x\to +\infty}\sqrt{x}=+\infty$ και
    \[ \lim_{u\to +\infty}\frac{u-1}{u+2} = \lim_{u\to +\infty}\frac{u\PARENS{1-\frac{1}{u}}}{u\PARENS{1+\frac{2}{u}}} = 1 . \]
    Επομένως $\lim\limits_{x\to +\infty}f(x) = \lim\limits_{u\to +\infty}\frac{\sqrt{x}-1}{\sqrt{x}+2} = 1$.
    Τελικά $f((0,+\infty))=[f(0),\lim\limits_{x\to +\infty}f(x))=[-\frac{1}{2},1)$. \monades{X}

    \item Για $x\in D_f$ και $y\in[-\frac{1}{2},1)$ έχουμε:
    \[
    \begin{aligned}
        f(x)=y & \Leftrightarrow \frac{\sqrt{x}-1}{\sqrt{x}+2}=y \\
        & \Leftrightarrow \sqrt{x}-1=y(\sqrt{x}+2) \\
        & \Leftrightarrow \sqrt{x}-1=y\sqrt{x}+2y \\
        & \Leftrightarrow \sqrt{x}-y\sqrt{x}=1+2y \\
        & \Leftrightarrow (1-y)\sqrt{x}=1+2y \\
        & \Leftrightarrow \sqrt{x}=\frac{1+2y}{1-y} \\
        & \Leftrightarrow x=\PARENS{\frac{1+2y}{1-y}}^2
    \end{aligned}
    \]
    όπου $\frac{1+2y}{1-y} \ge 0$ για $y\in[-\frac{1}{2},1)$.
    Επομένως ορίζεται η αντίστροφη της $f$ με πεδίο ορισμού το $[-\frac{1}{2},1)$ και τύπο
    \[ f^{-1}(x) = \PARENS{\frac{1+2x}{1-x}}^2 . \] \monades{X}

    \item
    \begin{rlist}
        \item Από τα πεδία ορισμού καθώς και τα σύνολα τιμών των δύο συναρτήσεων
        συμπεραίνουμε ότι η καμπύλη $C_1$ αντιστοιχεί στην $C_f$ και η $C_2$ στην $C_{f^{-1}}$. \monades{X}

        \item Επειδή $f^{-1}(x)\ge 0$ για $x\in[-\frac{1}{2},0]$ με την ισότητα να ισχύει μόνο για $x=-\frac{1}{2}$,
        το ολοκλήρωμα $\int_{-\frac{1}{2}}^{0}f^{-1}(x)dx$ παριστάνει το εμβαδόν του χωρίου $\Omega_1$ που
        περικλείεται μεταξύ της $C_{f^{-1}}$, τον άξονα $x'x$ και τις ευθείες $x=-\frac{1}{2}$, $x=0$.
        Επομένως $a>0$.
        Το ολοκλήρωμα $I=\int_{0}^{1}f(x)dx$ μπορούμε να το βρούμε συναρτήσει του $a$ με δύο
        τρόπους:

        1$^{\circ}$ τρόπος (Γεωμετρικά)
        Το συμμετρικό του χωρίου $\Omega_1$ ως προς την $y=x$ είναι το χωρίο $\Omega_2$ που
        περικλείεται μεταξύ της $C_f$, τον άξονα $x'x$ και τις ευθείες $x=0$, $x=1$. Επειδή
        $f(x)\le 0$, για $x\in[0,1]$ έχουμε:
        \[ E(\Omega_2) = E(\Omega_1) \Rightarrow -\int_{0}^{1}f(x)dx = \alpha \Rightarrow \int_{0}^{1}f(x)dx = -\alpha . \]

        2$^{\circ}$ τρόπος (Αλγεβρικά)
        Εφαρμόζοντας τον τύπο ολοκλήρωσης με αλλαγή μεταβλητής (από το 2$^{\circ}$ στο 1$^{\circ}$
        μέλος) και στην συνέχεια ολοκλήρωση κατά παραγόντες, έχουμε:
        \[ \int_{0}^{1}f(x)dx = \int_{f^{-1}(0)}^{0}f(f^{-1}(y))\cdot (f^{-1}(y))'dy = \int_{f^{-1}(-\frac{1}{2})}^{0}y\cdot (f^{-1}(y))'dy \]
        \[ [x \cdot f^{-1}(x)]_{-\frac{1}{2}}^{0} - \int_{-\frac{1}{2}}^{0} f^{-1}(x) dx = -\alpha . \] \monades{X}
    \end{rlist}
\end{alist}
