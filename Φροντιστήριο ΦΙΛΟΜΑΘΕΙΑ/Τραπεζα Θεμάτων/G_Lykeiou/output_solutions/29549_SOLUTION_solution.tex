\Askhsh{29549_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item[α)] Είναι
\[
\int_0^\pi f''(x) \eta\mu x \, dx = -\int_0^\pi f'(x) \sigma\upsilon\nu x \, dx
\]
\begin{align*}
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi - \int_0^\pi f'(x)\sigma\upsilon\nu x \, dx = -\int_0^\pi f'(x)\sigma\upsilon\nu x \, dx \\
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi = 0 \\
\Leftrightarrow &f'(\pi)\eta\mu\pi - f'(0)\eta\mu 0 = 0, \text{ που ισχύει.}
\end{align*}

\item[β)] Είναι $\int_0^\pi (f''(x) + f(x)) \eta\mu x \, dx = 0$ και έτσι έχουμε:
\[
\int_0^\pi f''(x)\eta\mu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0
\]
\begin{align*}
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi - \int_0^\pi f'(x)\sigma\upsilon\nu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &0 - \left( \left[ f(x)\sigma\upsilon\nu x \right]_0^\pi - \int_0^\pi f(x)(-\sigma\upsilon\nu x)' \, dx \right) + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &-\left[ f(x)\sigma\upsilon\nu x \right]_0^\pi + \int_0^\pi f(x)\eta\mu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &-\left[ f(x)\sigma\upsilon\nu x \right]_0^\pi + 2\int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &-\left( f(\pi)\sigma\upsilon\nu\pi - f(0)\sigma\upsilon\nu 0 \right) + 2\int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &-\left( f(\pi)(-1) - f(0)(1) \right) + 2\int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &f(\pi) + f(0) + 2\int_0^\pi f(x)\eta\mu x \, dx = 0
\end{align*}

\onoma{β τρόπος}
\[
\int_0^\pi (f''(x) + f(x)) \eta\mu x \, dx = 0
\]
\begin{align*}
\Leftrightarrow &\int_0^\pi f''(x)\eta\mu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi - \int_0^\pi f'(x)\sigma\upsilon\nu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi - \left( \left[ f(x)\sigma\upsilon\nu x \right]_0^\pi - \int_0^\pi f(x)(-\eta\mu x) \, dx \right) + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &\left[ f'(x)\eta\mu x \right]_0^\pi - \left[ f(x)\sigma\upsilon\nu x \right]_0^\pi - \int_0^\pi f(x)\eta\mu x \, dx + \int_0^\pi f(x)\eta\mu x \, dx = 0 \\
\Leftrightarrow &\left[ f'(x)\eta\mu x - f(x)\sigma\upsilon\nu x \right]_0^\pi = 0 \\
\Leftrightarrow &(f'(\pi)\eta\mu\pi - f(\pi)\sigma\upsilon\nu\pi) - (f'(0)\eta\mu 0 - f(0)\sigma\upsilon\nu 0) = 0 \\
\Leftrightarrow &(f'(\pi)\cdot 0 - f(\pi)\cdot(-1)) - (f'(0)\cdot 0 - f(0)\cdot 1) = 0 \\
\Leftrightarrow &f(\pi) + f(0) = 0
\end{align*}

\item[γ)] Η συνάρτηση $f$ είναι συνεχής στο $[0,\pi]$ και παραγωγίσιμη στο $(0,\pi)$ ως δύο φορές παραγωγίσιμη στο $\mathbb{R}$ και είναι $f(0)=f(\pi)=0$. Επειδή η συνάρτηση $f$ πληροί τις υποθέσεις του θεωρήματος Rolle στο $[0,\pi]$, υπάρχει τουλάχιστον ένα $x_0 \in (0,\pi)$ τέτοιο, ώστε $f'(x_0)=0$.

Ακόμη η $f'$ είναι συνεχής στο $[0,x_0]$ και παραγωγίσιμη στο $(0,x_0)$ ως παραγωγίσιμη στο $\mathbb{R}$ και είναι $f'(0)=f'(x_0)=0$. Επειδή η συνάρτηση $f'$ πληροί τις υποθέσεις του θεωρήματος Rolle στο $[0,x_0]$, υπάρχει ένα τουλάχιστον $\xi \in (0,x_0)$ (άρα $\xi \in (0,\pi)$) τέτοιο ώστε $f''(\xi)=0$ και έπεται το ζητούμενο.
\end{alist}
