\textbf{ΛΥΣΗ}
\begin{alist}
    \item α) Έχουμε $\left(\int_{7}^{9} f(x) dx\right)^2 = 16 \Leftrightarrow \int_{7}^{9} f(x) dx = 4 \text{ ή } \int_{7}^{9} f(x) dx = -4$.
    Η συνεχής συνάρτηση $f$ είναι μη αρνητική στο διάστημα $[7,9]$ και δεν είναι παντού μηδέν στο διάστημα αυτό, άρα $\int_{7}^{9} f(x) dx > 0$. Συνεπώς $\int_{7}^{9} f(x) dx = 4$.

    \item β) Το ζητούμενο εμβαδό είναι ίσο με:
    \[ \frac{4}{3} + e^2 - 5 + 1 + \int_{7}^{9} f(x) dx = \frac{4}{3} + e^2 - 4 + 4 = \frac{3e^2 + 4}{3} \text{ τ.μ.} \]

    \item γ) Γνωρίζουμε ότι το $\int_{1}^{9} f(x) dx$ είναι ίσο με το άθροισμα των εμβαδών των χωρίων που βρίσκονται πάνω από τον άξονα x'x μείον το άθροισμα των εμβαδών των χωρίων που βρίσκονται κάτω από τον άξονα x'x.
    Άρα
    \[ \int_{1}^{9} f(x) dx = \left(\frac{4}{3} + 4\right) - (e^2 - 5 + 1) = \frac{28}{3} - e^2. \]
\end{alist}
