    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με
\[ f'(x) = ((x-1)^3 - 3x)' = 3(x-1)^2 - 3 = 3x^2 - 6x + 3 - 3 = 3x(x-2) \]
Λύνουμε την εξίσωση
\[ f'(x) = 0 \Leftrightarrow 3x(x-2) = 0 \Leftrightarrow x = 0 \text{ ή } x = 2 \]
Τα πρόσημα της παραγώγου της $f$ φαίνονται στον ακόλουθο πίνακα:
\[
\begin{array}{|c|ccccccc|}
\hline
x & -\infty & & 0 & & 2 & & +\infty \\
\hline
f'(x) & & + & 0 & - & 0 & + & \\
\hline
f(x) & & \nearrow & & \searrow & & \nearrow & \\
\hline
\end{array}
\]
Άρα, η συνάρτηση $f$ είναι:
\begin{itemize}
    \item γνήσίως αύξουσα στο $(-\infty,0]$, αφού είναι συνεχής στο $(-\infty,0]$ και ισχύει $f'(x) > 0$ στο $(-\infty,0)$.
    \item γνήσίως φθίνουσα στο $[0,2]$, αφού είναι συνεχής στο $[0,2]$ και ισχύει $f'(x) < 0$ στο $(0,2)$.
    \item γνήσίως αύξουσα στο $[2,+\infty)$, αφού είναι συνεχής στο $[2,+\infty)$ και ισχύει $f'(x) > 0$ στο $(2,+\infty)$.
\end{itemize}
\item Η $f$ είναι γνήσίως αύξουσα και συνεχής στο $[2,+\infty)$, οπότε:
\[ f([2,+\infty)) = \left[f(2), \lim_{x\to+\infty} f(x)\right) = [-5,+\infty) \]
διότι $\lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} (x^3 + 3x - 1) = \lim_{x\to+\infty} x^3 = +\infty$.
\item Παρατηρούμε ότι $0 \in f([2,+\infty))$, οπότε η εξίσωση $f(x) = 0$ έχει μια ακριβώς πραγματική ρίζα στο διάστημα $[2,+\infty)$, αφού η $f$ είναι γνήσίως αύξουσα σε αυτό.
\end{alist}
