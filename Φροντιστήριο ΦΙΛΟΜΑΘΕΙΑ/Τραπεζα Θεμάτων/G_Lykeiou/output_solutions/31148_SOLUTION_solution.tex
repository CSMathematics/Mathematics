\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\ΛΥΣΗ
\begin{alist}
    \item [α)] Αρκεί να δείξουμε ότι για κάθε $x \in \mathbb{R}$ ισχύει
    \[ \frac{x^2+1}{e^x} \ge e^{-x} \Leftrightarrow x^2+1 \ge e^x \cdot e^{-x} \Leftrightarrow x^2+1 \ge 1 \Leftrightarrow x^2 \ge 0, \]
    ισχύει.
    Το ίσον αν και μόνον αν $x = 0$.

    \item [β)]
    \begin{rlist}
        \item [(i)] Από την εκφώνηση έχουμε τα σημεία $\Delta(0, f(x))$ και $Z(0, g(x))$.
        Από το α) ερώτημα προκύπτει ότι $E(x) = (B\Gamma Z\Delta) = (B\Gamma) \cdot (\Gamma Z) = (f(x) - g(x)) \cdot x$, άρα
        \[ E(x) = x\left(\frac{x^2+1}{e^x} - e^{-x}\right) = x \cdot \frac{x^2+1-e^x e^{-x}}{e^x} = \frac{x^3}{e^x}, x > 0. \]
        \item [(ii)] Για κάθε $x > 0$ έχουμε
        \[ E'(x) = \frac{(x^3)'e^x - x^3(e^x)'}{(e^x)^2} = \frac{3x^2e^x - x^3e^x}{(e^x)^2} = \frac{x^2e^x(3-x)}{(e^x)^2} = \frac{x^2}{e^x}(3-x). \]
        Άρα, έχουμε τον παρακάτω πίνακα μεταβολών.
        \[
        \begin{array}{|c||c|c|c|c|}
            \hline
            x & 0 & & 3 & +\infty \\
            \hline
            E'(x) & & + & 0 & - \\
            \hline
            E(x) & & \nearrow & \text{Ο.Μ.} & \searrow \\
            \hline
        \end{array}
        \]
        Έτσι για $x = 3$ το εμβαδόν του ορθογωνίου $B\Gamma Z\Delta$ μεγιστοποιείται.
    \end{rlist}

    \item [γ)] Αφού $h(x) > 0$ για κάθε $x \in [\ln2, 1]$ και $h(x)$ συνεχής, το ζητούμενο εμβαδόν θα είναι
    \begin{align*}
        E &= \int_{\ln2}^1 h(x)dx = \int_{\ln2}^1 xe^{-x}dx = \int_{\ln2}^1 x(-e^{-x})'dx \\
        &= [-xe^{-x}]_{\ln2}^1 + \int_{\ln2}^1 (x)'e^{-x}dx \\
        &= (-1e^{-1} - (-\ln2 e^{-\ln2})) + [-e^{-x}]_{\ln2}^1 \\
        &= -\frac{1}{e} + \frac{\ln2}{e^{\ln2}} + (-e^{-1} - (-e^{-\ln2})) \\
        &= -\frac{1}{e} + \frac{\ln2}{2} - \frac{1}{e} + \frac{1}{e^{\ln2}} \\
        &= -\frac{1}{e} + \frac{\ln2}{2} - \frac{1}{e} + \frac{1}{2} \\
        &= \frac{1+\ln2}{2} - \frac{2}{e} \\
        &= \frac{1}{2}\ln(2e) - \frac{2}{e} = \ln\sqrt{2e} - \frac{2}{e} \text{ τετραγωνικές μονάδες}.
    \end{align*}
\end{alist}
