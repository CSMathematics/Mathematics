\OnPar{ΛΥΣΗ}
\begin{alist}
    \item[α)]
    \begin{rlist}
        \item Στο διάστημα $[-2,-1]$ η γραφική παράσταση της $f$ βρίσκεται κάτω από τον άξονα $x'x$. Άρα το εμβαδόν του χωρίου που σχηματίζει η $C_f$ με τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x=-2$ και $x=-1$ είναι το ολοκλήρωμα $\int_{-2}^{-1} |f(x)|dx = -\int_{-2}^{-1} f(x)dx$. Το εμβαδόν όμως του χωρίου αυτού ισούται επίσης με το εμβαδόν του τριγώνου $\Gamma{\rm B}\Theta$.

        Για το τρίγωνο $\Gamma{\rm B}\Theta$ έχουμε ότι $\Gamma{\rm B} = |-1 - (-2)| = 1$ και ${\rm B}\Theta = |-2 - 0| = 2$.
        Άρα ${\rm E}(\Gamma{\rm B}\Theta) = \frac{1}{2} \cdot \Gamma{\rm B} \cdot {\rm B}\Theta = \frac{1}{2} \cdot 1 \cdot 2 = 1$. Επομένως:
        \[
            -\int_{-2}^{-1} f(x)dx = 1 \iff \int_{-2}^{-1} f(x)dx = -1
        \]

        \item Στο διάστημα $[-1,0]$ η γραφική παράσταση της $f$ βρίσκεται πάνω από τον άξονα $x'x$, οπότε το ολοκλήρωμα $\int_{-1}^{0} f(x)dx$ θα ισούται με το εμβαδόν του τριγώνου ${\rm O}\Gamma{\rm A}$.

        ${\rm O}\Gamma{\rm A}$. Είναι ${\rm O}\Gamma = |-1 - 0| = 1$ και ${\rm O}{\rm A} = 2$. Άρα
        \[
            \int_{-1}^{0} f(x)dx = \frac{1}{2} \cdot {\rm O}\Gamma \cdot {\rm O}{\rm A} = \frac{1}{2} \cdot 1 \cdot 1 = 1
        \]

        \item Στο διάστημα $[0,3]$ η γραφική παράσταση της $f$ βρίσκεται πάνω από τον άξονα $x'x$, οπότε το ολοκλήρωμα $\int_{0}^{3} f(x)dx$ θα ισούται με το εμβαδόν του τραπεζίου ${\rm O}{\rm A}\Lambda\Delta$.

        Είναι ${\rm O}{\rm A} = 2$, $\Lambda\Delta = 5 - 0 = 5$ και ${\rm O}\Delta = 3$. Άρα
        \[
            \int_{0}^{3} f(x)dx = \frac{{\rm O}{\rm A} + \Lambda\Delta}{2} \cdot {\rm O}\Delta = \frac{2+5}{2} \cdot 3 = \frac{21}{2}
        \]
    \end{rlist}
    \item[β)] Το εμβαδόν του χωρίου που περικλείεται από την γραφική παράσταση της $f$, τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x=-2$ και $x=3$ είναι το $\int_{-2}^{3} |f(x)|dx$. Σύμφωνα με το α) ερώτημα θα ισούται με το άθροισμα των εμβαδών των τριγώνων $\Gamma{\rm B}\Theta$, ${\rm O}\Gamma{\rm A}$ και του τραπεζίου ${\rm O}{\rm A}\Lambda\Delta$. Άρα:
    \[
        \int_{-2}^{3} |f(x)|dx = -\int_{-2}^{-1} f(x)dx + \int_{-1}^{0} f(x)dx + \int_{0}^{3} f(x)dx = 1+1+\frac{21}{2} = \frac{25}{2} \tau.\mu.
    \]
\end{alist}
