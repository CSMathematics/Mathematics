\Askhsh{29838_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item
    \begin{rlist}
    \item Είναι: \( \left| \frac{1-\sigma\upsilon\nu x}{x} \right| = \left| \frac{1-\sigma\upsilon\nu x}{x} \right| \leq \frac{2}{|x|} \)
    άρα \( \frac{2}{|x|} \leq \frac{2}{|x|} \) και \( \lim_{x \to \pm \infty} \left( - \frac{2}{|x|} \right) = 0 = \lim_{x \to \pm \infty} \frac{2}{|x|} \),
    οπότε από το κριτήριο παρεμβολής το \( \lim_{x \to \pm \infty} \frac{1-\sigma\upsilon\nu x}{x} = 0 \).

    \item Επίσης για \( x \neq 0 \) ισχύει: \( x f(x) + \sigma\upsilon\nu x = 1 - x^2 \eta \mu \frac{1}{x} \Leftrightarrow f(x) = \frac{1-\sigma\upsilon\nu x}{x} + x \eta \mu \frac{1}{x} \), οπότε:
    \[ \lim_{x \to 0} f(x) = \lim_{x \to 0} \left( \frac{1-\sigma\upsilon\nu x}{x} + x \eta \mu \frac{1}{x} \right) = \lim_{x \to 0} \frac{1-\sigma\upsilon\nu x}{x} + \lim_{x \to 0} x \eta \mu \frac{1}{x} = 0 + 0 = 0, \]
    γιατί
    \[ \lim_{x \to 0} \frac{1-\sigma\upsilon\nu x}{x} = \lim_{x \to 0} \frac{(1-\sigma\upsilon\nu x)(1+\sigma\upsilon\nu x)}{x(1+\sigma\upsilon\nu x)} = \lim_{x \to 0} \frac{1-\sigma\upsilon\nu^2 x}{x(1+\sigma\upsilon\nu x)} = \lim_{x \to 0} \left( \frac{\eta \mu x}{x} \cdot \frac{\eta \mu x}{1+\sigma\upsilon\nu x} \right) = 1 \cdot \frac{0}{2} = 0 \]
    και
    \[ \lim_{x \to 0} x \eta \mu \frac{1}{x} = 0. \]
    Λόγω του ερωτήματος (α, i) το \( \lim_{x \to + \infty} \frac{1-\sigma\upsilon\nu x}{x} = 0 \) και
    \[ \lim_{x \to + \infty} x \eta \mu \frac{1}{x} = \lim_{\upsilon \to 0^+} \frac{\eta \mu \upsilon}{\upsilon} = 1. \]
    Άρα \( \lim_{x \to + \infty} f(x) = 0+1=1 \).
    \end{rlist}

\item Από τα δεδομένα η $f$ είναι συνεχής στο $\mathbb{R}$, άρα το $f(0) = \lim_{x \to 0} f(x)$ (1).
Επίσης λόγω του ερωτήματος (α) για $x \neq 0$ ισχύει: \( f(x) = \frac{1-\sigma\upsilon\nu x}{x} + x \eta \mu \frac{1}{x} \) (2).
Λόγω της (2) είναι:
\[ f(0) = \lim_{x \to 0} f(x) = \lim_{x \to 0} \left( \frac{1-\sigma\upsilon\nu x}{x} + x \eta \mu \frac{1}{x} \right) = \lim_{x \to 0} \frac{1-\sigma\upsilon\nu x}{x} + \lim_{x \to 0} x \eta \mu \frac{1}{x} = 0+0=0, \]
εφόσον
\[ \lim_{x \to 0} \frac{1-\sigma\upsilon\nu x}{x} = 0 \quad (\text{βασικό όριο}) \quad \text{και} \quad \lim_{x \to 0} x \eta \mu \frac{1}{x} = 0, \]
γιατί \( \left| x \eta \mu \frac{1}{x} \right| = |x| \left| \eta \mu \frac{1}{x} \right| \leq |x| \cdot 1 = |x| \). Άρα: \( -|x| \leq x \eta \mu \frac{1}{x} \leq |x| \) και \( \lim_{x \to 0} (-|x|) = 0 = \lim_{x \to 0} |x| \), οπότε από το κριτήριο παρεμβολής θα είναι: \( \lim_{x \to 0} x \eta \mu \frac{1}{x} = 0 \). Επομένως λόγω της (1) το $f(0)=0$.

\item Εφόσον λόγω του ερωτήματος (α, ii) το \( \lim_{x \to + \infty} f(x) = -1 < 0 \), θα υπάρχει $k > 0$ με $k > \frac{1}{\pi}$ τέτοιος ώστε $f(k) < 0$ (3).
Λόγω της (2), θεωρούμε τη συνάρτηση \( f(x) = \frac{1-\sigma\upsilon\nu x}{x} + x \eta \mu \frac{1}{x} \) στο διάστημα \( \left[ \frac{1}{\pi}, k \right] \) όπου αυτή είναι συνεχής ως αποτέλεσμα πράξεων συνεχών συναρτήσεων.
Επίσης:
\[ f \left( \frac{1}{\pi} \right) = \frac{1-\sigma\upsilon\nu \frac{1}{\pi}}{\frac{1}{\pi}} + \frac{1}{\pi} \eta \mu \frac{1}{\frac{1}{\pi}} = \pi (1-\sigma\upsilon\nu \frac{1}{\pi}) + \pi \eta \mu \pi = \pi (1-\sigma\upsilon\nu \frac{1}{\pi}) + \pi \cdot 0 = \pi (1-\sigma\upsilon\nu \frac{1}{\pi}) > 0 \quad (4), \]
γιατί ισχύει: \( 0 < \frac{1}{\pi} < 2\pi \) οπότε \( \sigma\upsilon\nu \frac{1}{\pi} < 1 \Rightarrow 1-\sigma\upsilon\nu \frac{1}{\pi} > 0 \).
Λόγω των (3), (4) το \( f \left( \frac{1}{\pi} \right) \cdot f(k) < 0 \) και από θεώρημα Bolzano θα υπάρχει ένα τουλάχιστον \( x_0 \in \left( \frac{1}{\pi}, k \right) \subseteq \left( \frac{1}{\pi}, + \infty \right) \), έτσι ώστε να ισχύει \( f(x_0) = 0 \). Επομένως η εξίσωση \( f(x)=0 \) έχει μία τουλάχιστον ρίζα στο διάστημα \( \left( \frac{1}{\pi}, + \infty \right) \).
\end{alist}
