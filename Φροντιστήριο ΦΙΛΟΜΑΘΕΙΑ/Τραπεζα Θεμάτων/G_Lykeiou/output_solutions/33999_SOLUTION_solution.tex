\begin{ylh}{1}
    \item [Συναρτήσεις, Όρια, Παράγωγοι]
\end{ylh}
Δίνεται συνάρτηση $f:(0, +\infty) \to \mathbb{R}$.
\begin{alist}
    \item Να βρείτε το $\lim_{x \to 0^+} f(x)$, αν γνωρίζετε ότι $f(x) \geq \frac{1}{x}$ για κάθε $x$ κοντά στο $0$ από δεξιά. \monades{X}
    \item Δίνεται επιπλέον ότι για κάθε $x \in (0, +\infty)$ ισχύει $f'(x)\ln(x+1) + \frac{f(x)}{x+1} = 0$.
    \begin{rlist}
        \item Αν ορίσουμε τη συνάρτηση $g(x) = f(x)\ln(x+1)$, να δείξετε ότι η $g$ είναι σταθερή. \monades{X}
        \item Αν επιπλέον για κάθε $x \in (0, +\infty)$ ισχύει $\frac{1}{x} \leq f(x) \leq 1 + \frac{1}{x}$, να βρείτε τον τύπο της συνάρτησης $f(x)$. \monades{X}
    \end{rlist}
\end{alist}
