    \item [Ύλη από εικόνα]

\kefalaio{ΛΥΣΗ}
\begin{alist}
    \item Έχουμε $D_f = (0,+\infty)$ και $D_g = \mathbb{R}$. Για την $f \circ g$ είναι:
    \[ A_1 = \{x \in D_g : g(x) \in D_f\} = \{x \in \mathbb{R} : e^{x+2} \in (0,+\infty)\} = \mathbb{R} \neq \emptyset \]
    οπότε ορίζεται η σύνθεση με τύπο:
    \[ (f \circ g)(x) = f(g(x)) = \ln g(x) + 2g(x) = \ln e^{x+2} + 2e^{x+2} = x+2+2e^{x+2}. \]
    \item Η συνάρτηση $g$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $g'(x) = (e^{x+2})' = e^{x+2} > 0$ άρα η $g$ είναι γνησίως αύξουσα στο $\mathbb{R}$ οπότε και ‘1-1’.
    \item Για την εύρεση του τύπου της αντίστροφου συνάρτησης της $g$, για $x \in \mathbb{R}$ θέτουμε
    \[ g(x) = y \Leftrightarrow e^{x+2} = y, y>0 \Leftrightarrow x+2 = \ln y, y>0 \Leftrightarrow x = \ln y - 2, y>0 \]
    άρα
    \[ g^{-1}(y) = \ln y - 2, y>0 \text{ ή } g^{-1}(x) = \ln x-2, x>0. \]
\end{alist}
