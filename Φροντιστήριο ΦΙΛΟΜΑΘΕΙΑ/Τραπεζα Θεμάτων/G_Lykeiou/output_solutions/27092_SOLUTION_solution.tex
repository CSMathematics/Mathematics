\Askhsh{27092_SOLUTION}{1}
\begin{ylh}{1}
    \item [Διαφορικός Λογισμός]
\end{ylh}
\begin{alist}
    \item Από τη γραφική παράσταση της συνάρτησης $f'$ συμπεραίνουμε ότι:
    \[
    \begin{gathered}
    f'(x) = 0 \iff x = 0 \text{ ή } x = 3 \\
    f'(x) > 0 \iff x \in (0,3) \\
    f'(x) < 0 \iff x \in (-\infty,0) \cup (3,+\infty)
    \end{gathered}
    \]
    Τα πρόσημα της παραγώγου της $f'$ φαίνονται στον ακόλουθο πίνακα:
    \[
    \begin{array}{|c|c*{2}{c@{\vline}c@{\vline}}c|}
    \hline
    x & -\infty & \multicolumn{1}{c|}{0} & \multicolumn{1}{c|}{3} & +\infty \\
    \hline
    f'(x) & \multicolumn{1}{c}{-} & \multicolumn{1}{c|}{0} & \multicolumn{1}{c}{+} & \multicolumn{1}{c|}{0} & \multicolumn{1}{c}{-} \\
    \hline
    f(x) & \multicolumn{1}{c}{\searrow} & & \multicolumn{1}{c}{\nearrow} & & \multicolumn{1}{c}{\searrow} \\
    \hline
    \end{array}
    \]
    Άρα, η συνάρτηση $f$ είναι γνησίως φθίνουσα στα διαστήματα $(-\infty,0]$ και $[3,+\infty)$ και γνησίως αύξουσα στο διάστημα $[0,3]$, αφού είναι συνεχής ως πολυωνυμική. \monades{6}

    \item Αφού η γραφική παράσταση της συνάρτησης $f$ διέρχεται από τα σημεία $A(0,-1)$ και $B(3,2)$, τότε θα είναι:
    \[
    f(0) = -1 \quad \text{και} \quad f(3) = 2
    \]
    Από τον πίνακα του ερωτήματος (α) προκύπτει ότι η συνάρτηση $f$ παρουσιάζει τοπικό ελάχιστο για $x=0$, το $f(0)=-1$, και τοπικό μέγιστο για $x=3$, το $f(3)=2$. \monades{6}

    \item Αφού η $f'$ είναι πολυώνυμο δευτέρου βαθμού, τότε η συνάρτηση $f$ θα είναι πολυώνυμο τρίτου βαθμού. Έστω ότι:
    \[
    f(x) = \alpha x^3 + \beta x^2 + \gamma x + \delta, \quad \alpha \ne 0
    \]
    Τότε είναι:
    \[
    f'(x) = 3\alpha x^2 + 2\beta x + \gamma
    \]
    Από τα δεδομένα της άσκησης έχουμε ότι:
    \[
    f'(0) = 0 \implies \gamma = 0 \quad \text{και} \quad f(0) = -1 \implies \delta = -1
    \]
    Επίσης, είναι:
    \[
    \left\{
    \begin{array}{l}
    f'(3) = 0 \implies 27\alpha + 6\beta = 0 \\
    f(3) = 2 \implies 27\alpha + 9\beta - 1 = 2
    \end{array}
    \right\}
    \implies 3\beta = 3 \implies \beta = 1 \quad \text{και} \quad \alpha = -\frac{2}{9}
    \]
    Επομένως, ο τύπος της συνάρτησης $f$ είναι:
    \[
    f(x) = -\frac{2}{9}x^3 + x^2 - 1, \quad x \in \mathbb{R}
    \] \monades{6}

    \item Βρίσκουμε το σύνολο τιμών της $f$ στο $(0,3)$. Η $f$ είναι γνησίως αύξουσα και συνεχής στο $(0,3) = A_2$, οπότε:
    \[
    f(A_2) = f((0,3)) = (f(0), f(3)) = (-1,2)
    \]
    Διακρίνουμε τις περιπτώσεις:
    \begin{enumerate}
        \item Αν $\alpha \in (-\infty, -1] \cup [2, +\infty)$, τότε η εξίσωση $f(x) = \alpha$ δεν έχει λύση.
        \item Αν $\alpha \in (-1,2)$, τότε η εξίσωση $f(x) = \alpha$ έχει μία ακριβώς λύση, αφού η $f$ είναι γνησίως αύξουσα στο $(0,3)$.
    \end{enumerate} \monades{7}
\end{alist}
