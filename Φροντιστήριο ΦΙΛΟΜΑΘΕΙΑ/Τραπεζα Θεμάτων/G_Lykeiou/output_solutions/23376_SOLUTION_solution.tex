\Askhsh{23376_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
Έστω η γραφική παράσταση της συνάρτησης $f$ βρίσκεται πάνω από τον άξονα $x'x$ για κάθε $x \in \mathbb{R}$, είναι $f(x)>0$, για κάθε $x \in \mathbb{R}$.
Το πεδίο ορισμού της συνάρτησης $h = g \circ f$ αποτελείται από εκείνα τα $x \in \mathbb{R}$ για τα οποία
\[
\begin{aligned}
& x \in D_f \text{ και } f(x) \in D_g \\
\Leftrightarrow{} & x \in \mathbb{R} \text{ και } f(x) \in (0, +\infty) \\
\Leftrightarrow{} & x \in \mathbb{R} \text{ και } f(x) > 0 \\
\Leftrightarrow{} & x \in \mathbb{R}.
\end{aligned}
\]
Άρα $D_h = \mathbb{R}.$
Για κάθε $x \in \mathbb{R}$ είναι $h(x) = (g \circ f)(x) = g(f(x)) = \ln f(x) = \ln(\sqrt{x^2+1}-x).$

\begin{alist}
    \item [$\beta$)]
    \begin{rlist}
        \item [] Για κάθε $x \in \mathbb{R}$ είναι $h(-x) = -h(x)$, για κάθε $x \in \mathbb{R}$.
        Είναι $h(-x) = \ln(\sqrt{(-x)^2+1}-(-x)) = \ln(\sqrt{x^2+1}+x)$ και έτσι έχουμε:
        \[ h(-x) = -h(x) \Leftrightarrow \ln(\sqrt{x^2+1}+x) = -\ln(\sqrt{x^2+1}-x) \]
        \[ \Leftrightarrow \ln(\sqrt{x^2+1}+x) + \ln(\sqrt{x^2+1}-x) = 0 \]
        \[ \Leftrightarrow \ln\left((\sqrt{x^2+1}+x)(\sqrt{x^2+1}-x)\right) = 0 \]
        \[ \Leftrightarrow \ln\left((\sqrt{x^2+1})^2-x^2\right) = 0 \]
        \[ \Leftrightarrow \ln(x^2+1-x^2) = 0 \]
        \[ \Leftrightarrow \ln(1) = 0 \]
        \[ \text{που ισχύει.} \]
        Επομένως η συνάρτηση $h$ είναι περιττή.
        \item [] Η συνάρτηση $h$ είναι παραγωγίσιμη με:
        \[ h'(x) = \left(\ln\left(\sqrt{x^2+1}-x\right)\right)' = \frac{1}{\sqrt{x^2+1}-x} \cdot \left(\sqrt{x^2+1}-x\right)' \]
        \[ = \frac{1}{\sqrt{x^2+1}-x} \cdot \left(\frac{\cancel{2}x}{\cancel{2}\sqrt{x^2+1}}-1\right) \]
        \[ = \frac{1}{\sqrt{x^2+1}-x} \cdot \left(\frac{x}{\sqrt{x^2+1}}-1\right) \]
        \[ = \frac{1}{\sqrt{x^2+1}-x} \cdot \frac{x-\sqrt{x^2+1}}{\sqrt{x^2+1}} \]
        \[ = \frac{1}{\sqrt{x^2+1}-x} \cdot \frac{-\left( \sqrt{x^2+1} - x \right)}{\sqrt{x^2+1}} \]
        \[ = -\frac{1}{\sqrt{x^2+1}} \]
        Οπότε είναι $h'(x) < 0$ για κάθε $x \in \mathbb{R}$.
        Επομένως η συνάρτηση $h$ είναι γνησίως φθίνουσα αφού είναι συνεχής και ισχύει $h'(x) < 0$ για κάθε $x \in \mathbb{R}$. Επειδή η συνάρτηση $h$ είναι γνησίως μονότονη (γν. φθίνουσα) είναι και "1-1".
    \end{rlist}
    \item [$\gamma$)] Με $x>0$ έχουμε:
    \[ h(x-1)+h\left(\ln\frac{1}{x}\right)=0 \Leftrightarrow h(x-1)=-h(-\ln x) \stackrel{h\text{ περιττή}}{\Leftrightarrow} \]
    \[ h(x-1)=h(\ln x) \stackrel{h\text{ 1-1}}{\Leftrightarrow} x-1=\ln x \Leftrightarrow x=1. \]
\end{alist}
