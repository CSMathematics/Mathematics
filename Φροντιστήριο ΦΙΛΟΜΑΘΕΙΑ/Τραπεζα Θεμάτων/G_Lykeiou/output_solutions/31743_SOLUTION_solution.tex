\textbf{ΛΥΣΗ}
\begin{alist}
    \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $f'(x) = (x\eta\mu x + 4)' = \eta\mu x + x\sigma\upsilon\nu x$.
    οπότε: $f'(0) = 0$ και $f'\left(\frac{\pi}{2}\right) = \eta\mu\frac{\pi}{2} + \frac{\pi}{2}\sigma\upsilon\nu\frac{\pi}{2} = 1$.
    \item Λόγω του ερωτήματος (α) για τη συνάρτηση $\varphi$, με $\varphi(x) = f'(x) - \frac{1}{3}$, $x \in \mathbb{R}$
    έχουμε: $\varphi(0) = f'(0) - \frac{1}{3} = -\frac{1}{3} < 0$ και $\varphi\left(\frac{\pi}{2}\right) = f'\left(\frac{\pi}{2}\right) - \frac{1}{3} = 1 - \frac{1}{3} = \frac{2}{3} > 0$.
    \item Λόγω του ερωτήματος (α) η συνάρτηση $\varphi$, με $\varphi(x) = f'(x) - \frac{1}{3}$, $x \in \mathbb{R}$ είναι
    συνεχής στο $\left[0, \frac{\pi}{2}\right]$ ως άθροισμα συνεχών.
    
    Λόγω του ερωτήματος (β) είναι:
    \[ \left. \begin{array}{l} \varphi(0) < 0 \\ \varphi\left(\frac{\pi}{2}\right) > 0 \end{array} \right\} \Rightarrow \varphi(0) \cdot \varphi\left(\frac{\pi}{2}\right) < 0 \]
    Οπότε από θεώρημα Bolzano θα υπάρχει ένα τουλάχιστον $x_0 \in \left(0, \frac{\pi}{2}\right): \varphi(x_0) = 0$.
    
    Επομένως η εξίσωση $\varphi(x) = 0$, έχει μία τουλάχιστον ρίζα στο διάστημα $\left(0, \frac{\pi}{2}\right)$.
\end{alist}
