\Askhsh{24283_SOLUTION}{1}
\subsection*{ΛΥΣΗ}
\begin{alist}
    \item Η $f$ είναι συνεχής σε καθένα από τα διαστήματα $[-1,2)$, $(2,5]$ ως πολυωνυμική.
    \medskip
    Στο $x_0 = 2$ έχουμε:
    \[
    \lim_{x \to 2^-} f(x) = \lim_{x \to 2^-} (x^2 - 3) = 1 \quad \text{και} \quad \lim_{x \to 2^+} f(x) = \lim_{x \to 2^+} (x - 1) = 1.
    \]
    Επειδή $\lim_{x \to 2^-} f(x) = \lim_{x \to 2^+} f(x) = 1$ έπεται ότι $\lim_{x \to 2} f(x) = 1$.
    \medskip
    $f(2) = 2^2 - 3 = 1$, άρα $f(2) = \lim_{x \to 2} f(x)$ οπότε η $f$ είναι συνεχής στο $2$.
    \medskip
    Επομένως, η $f$ είναι συνεχής στο $[-1,5]$.

    \item Έχουμε ότι:
    \begin{align*}
    \lim_{x \to 2^-} \frac{f(x) - f(2)}{x - 2} &= \lim_{x \to 2^-} \frac{x^2 - 3 - 1}{x - 2} \\
    &= \lim_{x \to 2^-} \frac{x^2 - 4}{x - 2} \\
    &= \lim_{x \to 2^-} \frac{(x-2)(x+2)}{x - 2} \\
    &= \lim_{x \to 2^-} (x+2) = 4
    \end{align*}
    και
    \[
    \lim_{x \to 2^+} \frac{f(x) - f(2)}{x - 2} = \lim_{x \to 2^+} \frac{x - 1 - 1}{x - 2} = \lim_{x \to 2^+} \frac{x - 2}{x - 2} = 1.
    \]
    Είναι $\lim_{x \to 2^-} \frac{f(x) - f(2)}{x - 2} \neq \lim_{x \to 2^+} \frac{f(x) - f(2)}{x - 2}$, άρα η συνάρτηση $f$ δεν είναι παραγωγίσιμη στη θέση $x_0=2$.

    \item Το θεώρημα μέσης τιμής για τη συνάρτηση $f$ στο διάστημα $[-1,5]$ έχει τις εξής υποθέσεις:
    \begin{itemize}
        \item Η συνάρτηση $f$ είναι συνεχής στο $[-1,5]$.
        \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $(-1,5)$.
    \end{itemize}
    Η υπόθεση της συνέχειας στο $[-1,5]$ ικανοποιείται λόγω του ερωτήματος (α) ενώ η υπόθεση της παραγωγισιμότητας στο $(-1,5)$ δεν ικανοποιείται λόγω του ερωτήματος (β).
\end{alist}
