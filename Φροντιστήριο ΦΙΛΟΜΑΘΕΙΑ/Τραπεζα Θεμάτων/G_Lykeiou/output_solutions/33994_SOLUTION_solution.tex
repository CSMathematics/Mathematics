\begin{ylh}{1}
    \item [Συνέχεια συνάρτησης, Παράγωγος συνάρτησης, Εφαπτομένη γραφικής παράστασης]
\end{ylh}
Επειδή η $f$ είναι συνεχής στο $x_0 = 0$ ισχύει $f(0) = \lim_{x \to 0} f(x)$.
Για $x$ κοντά στο $0$ θέτουμε $\frac{f(x)}{\sin x} = h(x)$, με $\lim_{x \to 0} h(x) = H$.
Είναι $f(x) = h(x) \cdot \sin x$.
Άρα $\lim_{x \to 0} f(x) = \lim_{x \to 0} (h(x) \cdot \sin x) = \lim_{x \to 0} h(x) \cdot \lim_{x \to 0} \sin x = H \cdot 0 = 0$.

\textbf{β τρόπος}
Επειδή η $f$ είναι συνεχής στο $x_0 = 0$ ισχύει $f(0) = \lim_{x \to 0} f(x)$.
Για $x$ κοντά στο $0$ θέτουμε $\frac{f(x)}{\sin x} = h(x)$, με $\lim_{x \to 0} h(x) = H$.
Είναι $\frac{f(x)}{x} = \frac{f(x)}{\sin x} \cdot \frac{\sin x}{x} = h(x) \cdot \frac{\sin x}{x}$.
Άρα $\lim_{x \to 0} \frac{f(x)}{x} = \lim_{x \to 0} \left( \frac{f(x)}{\sin x} \cdot \frac{\sin x}{x} \right) = \lim_{x \to 0} h(x) \cdot \lim_{x \to 0} \frac{\sin x}{x} = H \cdot 1 = H$.

\begin{alist}
    \item[β)] Χρησιμοποιώντας τη συνάρτηση $h$ του ερωτήματος (α) έχουμε:
    \[ f'(0) = \lim_{x \to 0} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0} \frac{f(x)}{x} = \lim_{x \to 0} \left( \frac{f(x)}{\sin x} \cdot \frac{\sin x}{x} \right) = \lim_{x \to 0} h(x) \cdot \lim_{x \to 0} \frac{\sin x}{x} = H \cdot 1 = H \]
    Άρα η $f$ είναι παραγωγίσιμη στο $x_0 = 0$ με $f'(0) = H$.

    \textbf{β τρόπος}
    \[ f'(0) = \lim_{x \to 0} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0} \frac{f(x)}{x} \]
    Επειδή $f(0) = 0$, έχουμε $\frac{f(x)}{x} = \frac{f(x)}{\sin x} \cdot \frac{\sin x}{x} = h(x) \cdot \frac{\sin x}{x}$.
    \[ \lim_{x \to 0} \frac{f(x)}{x} = \lim_{x \to 0} \left( h(x) \cdot \frac{\sin x}{x} \right) = \lim_{x \to 0} h(x) \cdot \lim_{x \to 0} \frac{\sin x}{x} = H \cdot 1 = H \]
    Άρα $f'(0) = H$, οπότε η $f$ είναι παραγωγίσιμη στο $x_0 = 0$.

    \item[γ)] $g(x) = f(x) \cdot \sin x$, $x \in \mathbb{R}$.
    \begin{rlist}
        \item[i.] Η εξίσωση της εφαπτομένης της γραφικής παράστασης της συνάρτησης $g$ στο σημείο $(0, g(0))$ είναι $y - g(0) = g'(0)(x - 0)$. Έχουμε $g(0) = f(0) \cdot \sin 0 = 0$ και επειδή οι συναρτήσεις $f, \varphi$ (με $\varphi(x) = \sin x$) είναι παραγωγίσιμες στο $x_0 = 0$ προκύπτει ότι και η $g(x) = f(x) \cdot \sin x$ είναι παραγωγίσιμη στο $x_0 = 0$.
        Έτσι έχουμε $g'(0) = f'(0) \cdot \sin 0 + f(0) \cdot \cos 0 = H \cdot 0 + 0 \cdot 1 = 0$.
        Άρα η ζητούμενη εξίσωση εφαπτομένης γίνεται $y - 0 = 0x \Leftrightarrow y = 0$.

        \textbf{β τρόπος (για τον υπολογισμό της παραγώγου της $g$ στο $x_0 = 0$)} Είναι
        \[ g'(0) = \lim_{x \to 0} \frac{g(x) - g(0)}{x - 0} = \lim_{x \to 0} \frac{f(x) \cdot \sin x - 0}{x} = \lim_{x \to 0} \left( \frac{f(x)}{x} \cdot \sin x \right) = \lim_{x \to 0} \frac{f(x)}{x} \cdot \lim_{x \to 0} \sin x = H \cdot 0 = 0 \]
        Άρα $g'(0) = 0$.

        \item[ii.] Υποθέτουμε ότι η συνάρτηση $g$ είναι κυρτή. Τότε, η εφαπτομένη της γραφικής παράστασης της $g$ στο $x_0 = 0$ βρίσκεται "κάτω" από τη γραφική της παράσταση, με εξαίρεση το σημείο επαφής τους $(0,0)$. Άρα για κάθε $x \in \mathbb{R}$ ισχύει $g(x) \ge 0 \Rightarrow f(x) \cdot \sin x \ge 0$, με την ισότητα να ισχύει μόνο στο $x_0 = 0$. Αυτό όμως είναι άτοπο αφού η ισότητα ισχύει και για $x = \pi$ και έτσι έπεται το ζητούμενο.
    \end{rlist}
\end{alist}
