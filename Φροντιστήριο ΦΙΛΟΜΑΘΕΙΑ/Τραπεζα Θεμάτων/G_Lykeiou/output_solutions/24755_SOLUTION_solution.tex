ΛΥΣΗ
\begin{alist}
\item Η συνάρτηση $f$ είναι συνεχής στο $\mathbb{R}$ άρα και στο $0$ οπότε
\[
\alpha = f(0) = \lim_{x\to0} f(x) = \lim_{x\to0} \left(1 - \frac{\eta\mu x}{x}\right) = 1 - 1 = 0.
\]
Συνεπώς $\alpha = 0.$
\item Είναι
\[
\lim_{x\to0} \frac{f(x)-f(0)}{x-0} = \lim_{x\to0} \frac{\left(1 - \frac{\eta\mu x}{x}\right) - 0}{x} = \lim_{x\to0} \left(\frac{x - \eta\mu x}{x^2}\right) \overset{\frac{0}{0}}{=} \lim_{x\to0} \left(\frac{1 - \sigma\upsilon\nu x}{2x}\right) = 0,
\]
οπότε η συνάρτηση $f$ είναι παραγωγίσιμη στο $x_0 = 0$ με $f'(0) = 0.$
\item Η εφαπτομένη της $C_f$ στο σημείο $(0, f(0))$ έχει εξίσωση
\[
y - f(0) = f'(0)(x-0) \Leftrightarrow y=0,
\]
δηλαδή είναι ο άξονας $xx'.$
\end{alist}
