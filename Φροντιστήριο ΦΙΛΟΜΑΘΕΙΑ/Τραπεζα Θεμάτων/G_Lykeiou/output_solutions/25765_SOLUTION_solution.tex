\textbf{ΛΥΣΗ}

\begin{alist}
    \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $(0, +\infty)$ με $f'(x)=\dfrac{2}{x}+1$ και $f'(x)>0$ για κάθε $x>0$.
    Άρα η $f$ είναι γνησίως αύξουσα στο πεδίο ορισμού της, οπότε αντιστρέφεται. Το πεδίο ορισμού της $f^{-1}$ είναι το σύνολο τιμών της $f$. Η $f$ είναι συνεχής και γνησίως αύξουσα στο $(0, +\infty)$ και
    \[ \lim_{x \to 0^+} f(x) = -\infty, \lim_{x \to +\infty} f(x) = +\infty \]
    οπότε το σύνολο τιμών της $f$, δηλαδή το πεδίο ορισμού της $f^{-1}$ είναι το σύνολο $\mathbb{R}$.

    \item Η ανίσωση ορίζεται για κάθε $x$ που περιέχεται στο πεδίο ορισμού της $f^{-1}$, δηλαδή για $x \in \mathbb{R}$. Επειδή για κάθε $x \in \mathbb{R}$ ισχύει $f^{-1}(x) \in \mathbb{D}_f = (0, +\infty)$, διακρίνουμε τις παρακάτω περιπτώσεις:
    \begin{itemize}
        \item Αν $x \le 0$, τότε η ανίσωση αληθεύει (αφού το πρώτο μέλος της είναι θετικό για όλα τα $x$)
        \item Αν $x > 0$, τότε έχουμε:
        \[ f^{-1}(x) > x \iff f(f^{-1}(x)) > f(x) \iff x > f(x) \iff x > 2\ln x + x \iff 2\ln x < 0 \iff \ln x < 0 \iff x < 1 \]
    \end{itemize}
    Επομένως λύση της ανίσωσης είναι κάθε αριθμός $x$ με $x \in (-\infty, 1)$.

    \item
    \begin{rlist}
        \item Για κάθε $x \ne 0$ είναι $f(|x|) = 2\ln|x| + |x| = \ln|x|^2 + |x| = \ln x^2 + |x|$, οπότε
        \[ g(x) = e^{\ln x^2 + |x|} = e^{\ln x^2} e^{|x|} = x^2 e^{|x|} \]
        Επιπλέον, η $g$ είναι συνεχής στο $x_0 = 0$, οπότε $g(0) = \lim_{x \to 0} g(x) = \lim_{x \to 0} x^2 e^{|x|} = 0$.
        Άρα, $g(x) = x^2 e^{|x|}, x \in \mathbb{R}$.

        \item Η $g$ είναι συνεχής στο διάστημα $[-1, 1]$ και $g(x) \ge 0$ για κάθε $x \in [-1, 1]$, οπότε το ζητούμενο εμβαδόν $E$ είναι
        \begin{align*}
            E &= \int_{-1}^{1} x^2 e^{|x|} dx \\
            &= \int_{-1}^{0} x^2 e^{-x} dx + \int_{0}^{1} x^2 e^{x} dx \\
            &= \left[-e^{-x}x^2\right]_{-1}^{0} + 2\int_{-1}^{0} xe^{-x} dx + \left[e^{x}x^2\right]_{0}^{1} - 2\int_{0}^{1} xe^{x} dx \\
            &= e - 2\left[xe^{-x}\right]_{-1}^{0} + 2\int_{-1}^{0} e^{-x} dx + e - 2\left[xe^{x}\right]_{0}^{1} + 2\int_{0}^{1} e^{x} dx \\
            &= e - 2(1-e) + e - 2e + 2(e-1) \\
            &= 2e - 4
        \end{align*}
    \end{rlist}
\end{alist}
