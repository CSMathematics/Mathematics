\begin{alist}
\item Η συνάρτηση $f$ είναι συνεχής στο $\mathbb{R}$ άρα και στο $0$ οπότε
\[ \alpha = f(0) = \lim_{x \to 0} f(x) = \lim_{x \to 0} \left(2023 - \frac{\eta \mu x}{x}\right) = 2023 - 1 = 2022 . \]
Συνεπώς, $\alpha = 2022$. \monades{X}
\item \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \left(2023 - \frac{\eta \mu x}{x}\right) = 2023 - 0 = 2023 , \]
αφού για $x \ne 0$ είναι
\[ |\eta \mu x| \le 1 \implies \frac{|\eta \mu x|}{|x|} \le \frac{1}{|x|} \implies -\frac{1}{|x|} \le \frac{\eta \mu x}{x} \le \frac{1}{|x|} \]
και
\[ \lim_{x \to +\infty} \frac{1}{|x|} = \lim_{x \to +\infty} \left(-\frac{1}{|x|}\right) = 0 \]
οπότε από το κριτήριο παρεμβολής είναι και $\lim_{x \to +\infty} \frac{\eta \mu x}{x} = 0$. \monades{X}
\item Αφού $f(0) = 2022$ ο αριθμός $0$ είναι λύση της εξίσωσης $f(x) = 2022$.

Για $x \ne 0$ είναι
\[ f(x) = 2022 \iff 2023 - \frac{\eta \mu x}{x} = 2022 \iff \frac{\eta \mu x}{x} = 1 \iff \eta \mu x = x \]
που είναι αδύνατη για $x \ne 0$.

Συνεπώς η μοναδική λύση της εξίσωσης $f(x) = 2022$ είναι η $x = 0$. \monades{X}
\end{alist}
