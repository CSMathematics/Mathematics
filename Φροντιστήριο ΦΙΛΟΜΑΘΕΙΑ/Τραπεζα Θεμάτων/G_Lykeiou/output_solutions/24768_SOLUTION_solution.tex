\OnPar{ΛΥΣΗ}
\begin{alist}
\item Είναι:
\[
f(x) \ge \frac{3}{4} \Leftrightarrow 4(x^2 - x + 1) \ge 3 \Leftrightarrow 4x^2 - 4x + 1 \ge 0 \Leftrightarrow (2x-1)^2 \ge 0
\]
που ισχύει. Η ισότητα $f(x) = \frac{3}{4}$ ισχύει μόνο όταν $x = \frac{1}{2}$. \monades{0}
\item Οι συναρτήσεις έχουν αντίστοιχα πεδία ορισμού $\mathbb{D}_f = \mathbb{R}$, $\mathbb{D}_g = \left[ \frac{3}{4}, +\infty \right)$ και η σύνθεση $h = g \circ f$ ορίζεται μόνο όταν υπάρχουν αριθμοί $x$ ώστε
\[
x \in \mathbb{D}_f \text{ και } f(x) \in \mathbb{D}_g
\]
Είναι:
\[
\begin{cases}
x \in \mathbb{D}_f \\
f(x) \in \mathbb{D}_g
\end{cases}
\Leftrightarrow
\begin{cases}
x \in \mathbb{R} \\
f(x) \ge \frac{3}{4}
\end{cases}
\]
που ισχύει για κάθε $x \in \mathbb{R}$ από το ερώτημα (α).
Άρα $\mathbb{D}_h = \mathbb{R}$ και
\[
h(x) = (g \circ f)(x) = g(f(x)) = \sqrt{4(x^2 - x + 1) - 3} = \sqrt{4x^2 - 4x + 1} = \sqrt{(2x-1)^2} = |2x-1|
\] \monades{0}
\item Ισχύει: $\lim_{x \to 0}(2x-1) = -1 < 0$, οπότε κοντά στο $x_0=0$ είναι $2x-1 < 0$, άρα $|2x-1| = -2x+1$.
και
\begin{align*}
\lim_{x \to 0}\frac{h(x)-1}{\sqrt{x+1}-1} &= \lim_{x \to 0}\frac{-2x+1-1}{\sqrt{x+1}-1} \\
&= \lim_{x \to 0}\frac{-2x}{\sqrt{x+1}-1} \\
&= -2\lim_{x \to 0}\frac{x}{\sqrt{x+1}-1} \\
&= -2\lim_{x \to 0}\frac{x(\sqrt{x+1}+1)}{(\sqrt{x+1}-1)(\sqrt{x+1}+1)} \\
&= -2\lim_{x \to 0}\frac{x(\sqrt{x+1}+1)}{(x+1)-1} \\
&= -2\lim_{x \to 0}\frac{x(\sqrt{x+1}+1)}{x} \\
&= -2\lim_{x \to 0}(\sqrt{x+1}+1) \\
&= -2(\sqrt{0+1}+1) \\
&= -2(1+1) \\
&= -4
\end{align*} \monades{0}
\end{alist}
