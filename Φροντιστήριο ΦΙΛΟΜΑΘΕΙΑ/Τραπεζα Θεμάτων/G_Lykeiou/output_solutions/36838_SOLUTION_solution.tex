\textbf{ΛΥΣΗ}
\begin{alist}
    \item \textbf{Γνωρίζουμε ότι}
    \[ \int_1^3 f(x)dx = 6 \]
    \[ \int_1^8 f(x)dx = 29 \]
    \[ \int_3^5 f(x)dx = 8 \]
    \textbf{και}
    \[ \int_1^5 g(x)dx = -6 \]
    \begin{rlist}
        \item
        \[ \int_3^8 f(x)dx = \int_3^1 f(x)dx + \int_1^8 f(x)dx = -\int_1^3 f(x)dx + \int_1^8 f(x)dx = -6 + 29 = 23 \]
        \item \textbf{Χρησιμοποιώντας και το προηγούμενο ολοκλήρωμα έχουμε:}
        \[ \int_5^8 2f(x)dx = 2 \cdot \int_5^8 f(x)dx = 2 \left( \int_5^3 f(x)dx + \int_3^8 f(x)dx \right) = 2(-8 + 23) = 2 \cdot 15 = 30 \]
        \item
        \begin{align*}
            \int_1^5 (f(x) + g(x))dx &= \int_1^5 f(x)dx + \int_1^5 g(x)dx \\
            &= \int_1^3 f(x)dx + \int_3^5 f(x)dx + \int_1^5 g(x)dx \\
            &= 6 + 8 - 6 = 8
        \end{align*}
    \end{rlist}
    \item \textbf{Το ζητούμενο εμβαδόν Ε ισούται με το $\int_1^5 |g(x)|dx$ και αφού $g(x) \le 0$ για κάθε $x \in [1,5]$, έχουμε:}
    \[ E = \int_1^5 -g(x)dx = - \int_1^5 g(x)dx = -(-6) = 6 \text{ τ.μ.} \]
\end{alist}
