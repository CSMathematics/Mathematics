[Εκφώνηση ασκήσεως]
\begin{alist}
    \item α) Η $f$ είναι συνεχής στο $x_o = 0$, μόνο όταν ισχύει:
    \[ \lim_{x \to 0^-} f(x) = f(0) = \lim_{x \to 0^+} f(x) \quad (1) \]
    Επειδή,
    \[ \lim_{x \to 0^-} f(x) = \lim_{x \to 0^-} (-x^2 + x) = 0, \quad f(0) = 0 \quad \text{και} \quad \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \eta\mu x = 0 \]
    από την (1) προκύπτει ότι η $f$ είναι συνεχής στο $x_o = 0$.

    \item β) Με $x$ κοντά στο $x_o = 0$ και
    \textbullet \ $x < 0$, έχουμε $\frac{f(x) - f(0)}{x - 0} = \frac{-x^2 + x}{x} = 1 - x$, οπότε
    \[ \lim_{x \to 0^-} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^-} (1 - x) = 1, \]
    \textbullet \ $x > 0$, έχουμε $\frac{f(x) - f(0)}{x - 0} = \frac{\eta\mu x}{x}$, οπότε
    \[ \lim_{x \to 0^+} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^+} \frac{\eta\mu x}{x} = 1. \]
    Επομένως, $\lim_{x \to 0^-} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^+} \frac{f(x) - f(0)}{x - 0} = 1$, οπότε η $f$ είναι παραγωγίσιμη στο $x_o = 0$ και
    $f'(0) = 1$.

    \item γ) Η εφαπτομένη της $C_f$ στο $x_o = 0$ έχει εξίσωση που δίνεται από τον τύπο
    \[ y - f(0) = f'(0)(x - 0) . \]
    Εδώ έχουμε $f(0) = 0$ και $f'(0) = 1$ οπότε η εξίσωση της εφαπτομένης είναι η $y = x$.
\end{alist}
