ΛΥΣΗ

\begin{alist}
    \item[α)] Ισχύουν:
    \[ \lim_{x \to 0^-} f(x) = \lim_{x \to 0^-} (e^x) = 1 = f(0) \quad \text{και} \quad \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} (-x^2 + 1) = 1 \]
    οπότε $\lim_{x \to 0^-} f(x) = \lim_{x \to 0^+} f(x) = f(0)$, άρα η $f$ είναι συνεχής στο $x_o = 0$.
    Η γραφική παράσταση της $f$ φαίνεται στο σχήμα που ακολουθεί.

    \item[β)] Με $x$ κοντά στο $0$ και
    \begin{itemize}
        \item $x < 0$, έχουμε $\lim_{x \to 0^-} \frac{f(x)-f(0)}{x-0} = \lim_{x \to 0^-} \frac{e^x-1}{x} = \eval{\frac{de^x}{dx}}_{x=0} = 1$,
        \item $x > 0$, έχουμε $\lim_{x \to 0^+} \frac{f(x)-f(0)}{x-0} = \lim_{x \to 0^+} \frac{-x^2+1-1}{x} = \lim_{x \to 0^+} \frac{-x^2}{x} = \lim_{x \to 0^+} x = 0$.
    \end{itemize}
    οπότε η $f$ δεν είναι παραγωγίσιμη στο $x_o=0$. Επομένως δεν ορίζεται εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο $A(0, f(0))$.
\end{alist}
