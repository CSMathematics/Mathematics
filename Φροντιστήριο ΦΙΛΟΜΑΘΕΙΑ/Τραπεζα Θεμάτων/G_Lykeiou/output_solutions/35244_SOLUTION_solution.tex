\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
% Η παρεχόμενη εικόνα είναι μια λύση και όχι μια εκφώνηση άσκησης.
% Ακολουθώντας την οδηγία να μετατραπεί το παρεχόμενο κείμενο σε δομή LaTeX,
% ακόμη κι αν περιγράφει βήματα λύσης, αυτό γίνεται εντός της δομής του alist.
\begin{alist}
\item α) Θεωρούμε τη συνάρτηση $g(x)=\eta\mu x-(1+x)\sigma\upsilon x, x \in \left[\frac{\pi}{4}, \frac{\pi}{2}\right]$.
\begin{itemize}
    \item Η $g$ είναι συνεχής στο $\left[\frac{\pi}{4}, \frac{\pi}{2}\right]$ ως αποτέλεσμα πράξεων μεταξύ συνεχών συναρτήσεων και ισχύει
    \[
    \begin{cases}
    g\left(\frac{\pi}{4}\right) = \eta\mu\frac{\pi}{4}-\left(1+\frac{\pi}{4}\right)\sigma\upsilon\frac{\pi}{4} = \frac{\sqrt{2}}{2}-\left(1+\frac{\pi}{4}\right)\frac{\sqrt{2}}{2} = -\frac{\pi\sqrt{2}}{8} \\
    g\left(\frac{\pi}{2}\right) = \eta\mu\frac{\pi}{2}-\left(1+\frac{\pi}{2}\right)\sigma\upsilon\frac{\pi}{2} = 1-\left(1+\frac{\pi}{2}\right)\cdot 0 = 1
    \end{cases}
    \]
\end{itemize}
Επομένως σύμφωνα με το θεώρημα Bolzano η εξίσωση $g(x)=0 \Leftrightarrow \eta\mu x = (1+x)\sigma\upsilon x$ έχει μια, τουλάχιστον, λύση στο ανοικτό διάστημα $\left(\frac{\pi}{4}, \frac{\pi}{2}\right)$.

Ακόμη είναι $g'(x)=(\eta\mu x-(1+x)\sigma\upsilon x)' = \sigma\upsilon x - \sigma\upsilon x+(1+x)\eta\mu x = (1+x)\eta\mu x > 0$ για κάθε $x \in \left(\frac{\pi}{4}, \frac{\pi}{2}\right)$ και επειδή η $g$ είναι συνεχής στο $\left[\frac{\pi}{4}, \frac{\pi}{2}\right]$ έπεται ότι η $g$ είναι γνησίως αύξουσα. Οπότε η εξίσωση $g(x)=0 \Leftrightarrow \eta\mu x=(1+x)\sigma\upsilon x$ έχει μια ακριβώς λύση στο ανοικτό διάστημα $\left(\frac{\pi}{4}, \frac{\pi}{2}\right)$.

\item β) Είναι $f(x)=0 \Leftrightarrow \epsilon\phi x=1 \Leftrightarrow x=\frac{\pi}{4}$.
    \begin{center}
        \boxed{x \in \left(0, \frac{\pi}{2}\right)}
    \end{center}
    \textbf{Α' τρόπος}
    Για κάθε $x_1, x_2 \in \left[0, \frac{\pi}{2}\right]$ με $x_1 < x_2$ έχουμε:
    $x_1 < x_2 \Rightarrow \epsilon\phi x_1 < \epsilon\phi x_2 \Rightarrow \epsilon\phi x_1-1 < \epsilon\phi x_2-1 \Rightarrow f(x_1) < f(x_2)$.
    Άρα η συνάρτηση $f$ είναι γνησίως αύξουσα.
    Για κάθε $x \in \left[0, \frac{\pi}{4}\right]$ έχουμε: $x \le \frac{\pi}{4} \Rightarrow f(x) \le f\left(\frac{\pi}{4}\right) \Rightarrow f(x) \le 0$.
    Για κάθε $x \in \left(\frac{\pi}{4}, \frac{\pi}{2}\right]$ έχουμε: $x > \frac{\pi}{4} \Rightarrow f(x) > f\left(\frac{\pi}{4}\right) \Rightarrow f(x) > 0$.
    \textbf{Β' τρόπος}
    Η εξίσωση $f(x)=0$ έχει ακριβώς μια λύση, την $x=\frac{\pi}{4}$ η οποία χωρίζει το πεδίο ορισμού της $f$ στα διαστήματα $\left[0, \frac{\pi}{4}\right)$ και $\left(\frac{\pi}{4}, \frac{\pi}{2}\right]$, σε καθένα από τα οποία η $f$ διατηρεί σταθερό πρόσημο ως συνεχής.
    Στο διάστημα $\left[0, \frac{\pi}{4}\right)$ βρίσκουμε την τιμή $f(0)=\epsilon\phi 0-1=-1$ και επειδή $f(0)<0$ είναι $f(x)<0$ για κάθε $x \in \left[0, \frac{\pi}{4}\right)$.
    Στο διάστημα $\left(\frac{\pi}{4}, \frac{\pi}{2}\right]$ βρίσκουμε την τιμή $f(\rho)=\epsilon\phi\rho-1$ όπου $\rho \in \left(\frac{\pi}{4}, \frac{\pi}{2}\right)$ η λύση της εξίσωσης του ερωτήματος (α). Αφού η τιμή της $\eta\mu x=(1+x)\sigma\upsilon x$, την επαληθεύει και έτσι έχουμε:
    $\eta\mu\rho = (1+\rho)\sigma\upsilon\rho \Rightarrow \epsilon\phi\rho = 1+\rho \Rightarrow \epsilon\phi\rho-1 = \rho \Rightarrow f(\rho)=\rho > 0$.
\item γ) Το ζητούμενο εμβαδό $E$ είναι ίσο με $E=\int_0^{\pi/2}|f(x)|\text{d}x$.
    Έχουμε:
    \[
    \begin{aligned}
    E &= \int_0^{\pi/2}|f(x)|\text{d}x \\
    &= \int_0^{\pi/4}|f(x)|\text{d}x + \int_{\pi/4}^{\pi/2}|f(x)|\text{d}x \\
    &= \int_0^{\pi/4}(-f(x))\text{d}x + \int_{\pi/4}^{\pi/2}f(x)\text{d}x \\
    &= \int_0^{\pi/4}(-\epsilon\phi x+1)\text{d}x + \int_{\pi/4}^{\pi/2}(\epsilon\phi x-1)\text{d}x \\
    &= \int_0^{\pi/4}\left(\frac{\eta\mu x}{\sigma\upsilon x}+1\right)\text{d}x - \int_{\pi/4}^{\pi/2}\left(\frac{\eta\mu x}{\sigma\upsilon x}+1\right)\text{d}x \\
    &= [\ln(\sigma\upsilon x)+x]_0^{\pi/4} - [\ln(\sigma\upsilon x)+x]_{\pi/4}^{\pi/2} \\
    &= \left(\ln\frac{\sqrt{2}}{2}+\frac{\pi}{4}\right) - (\ln 1+0) - \left[\left(\ln\frac{\sqrt{2}}{2}+\frac{\pi}{2}\right) - \left(\ln\frac{\sqrt{2}}{2}+\frac{\pi}{4}\right)\right] \\
    &= 2\left(\ln\frac{\sqrt{2}}{2}+\frac{\pi}{4}\right) - \left(\ln 1+\frac{\pi}{2}\right) \\
    &= \frac{\pi}{6} \quad \text{τετραγωνικές μονάδες.}
    \end{aligned}
    \]
\end{alist}
