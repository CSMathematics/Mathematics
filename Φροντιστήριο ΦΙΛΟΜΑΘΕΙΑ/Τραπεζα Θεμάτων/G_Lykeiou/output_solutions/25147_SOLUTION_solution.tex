\Askhsh{25147_SOLUTION}{1}
\begin{ylh}{1}
    \item [Λύση]
\end{ylh}
\begin{alist}
\item [α)] Για να βρούμε τα κοινά σημεία των γραφικών παραστάσεων των δύο συναρτήσεων, λύνουμε το παρακάτω σύστημα για $x \in [0,2\pi]$, όπου ισχύει $e^{-x}>0$:
\[\left\{\begin{array}{l} y=f(x) \\ y=g(x) \end{array}\right. \Leftrightarrow \left\{\begin{array}{l} y=e^{-x} \\ y=e^{-x}\eta\mu x \end{array}\right. \Leftrightarrow e^{-x}=e^{-x}\eta\mu x \Leftrightarrow 1=\eta\mu x \Leftrightarrow x=2k\pi+\frac{\pi}{2}, k \in \mathbb{Z}\]
\[\Leftrightarrow x=\frac{\pi}{2}\]
Συνεπώς, μοναδικό κοινό σημείο των δύο γραφικών παραστάσεων είναι το $A\left(\frac{\pi}{2}, e^{-\frac{\pi}{2}}\right)$.

\item [β)] Η εφαπτομένη της γραφικής παράστασης της $f$ στο $A$ θα έχει εξίσωση:
$y-f\left(\frac{\pi}{2}\right)=f'\left(\frac{\pi}{2}\right)\left(x-\frac{\pi}{2}\right)$, με $f\left(\frac{\pi}{2}\right)=e^{-\frac{\pi}{2}}$ και εφόσον $f'(x)=e^{-x}(-1)=-e^{-x}$ θα είναι $f'\left(\frac{\pi}{2}\right)=-e^{-\frac{\pi}{2}}$.

Η εφαπτομένη της γραφικής παράστασης της $g$ στο $A$ θα έχει εξίσωση:
$y-g\left(\frac{\pi}{2}\right)=g'\left(\frac{\pi}{2}\right)\left(x-\frac{\pi}{2}\right)$, με $g\left(\frac{\pi}{2}\right)=e^{-\frac{\pi}{2}}$ και εφόσον $g'(x)=e^{-x}(-x)'\eta\mu x+e^{-x}(\eta\mu x)'=e^{-x}(-\eta\mu x+\sigma\upsilon\nu x)$ θα είναι $g'\left(\frac{\pi}{2}\right)=e^{-\frac{\pi}{2}}\left(\sigma\upsilon\nu\frac{\pi}{2}-\eta\mu\frac{\pi}{2}\right)=e^{-\frac{\pi}{2}}(0-1)=-e^{-\frac{\pi}{2}}$.

Συνεπώς, οι εφαπτομένες των γραφικών παραστάσεων των δύο συναρτήσεων είναι ίδιες, άρα ταυτίζονται, δηλαδή οι γραφικές παραστάσεις δέχονται κοινή εφαπτομένη στο $A$.

\item [γ)] Ισχύει ότι $f(x)-g(x)=f(x)-f(x)\eta\mu x=f(x)(1-\eta\mu x)\ge 0$, για κάθε $x \in [0,2\pi]$.
Συνεπώς, για το ζητούμενο εμβαδόν $E$ ισχύει:
\[ E=\int_{0}^{2\pi}\left|f(x)-g(x)\right|dx=\int_{0}^{2\pi}(f(x)-g(x))dx=\int_{0}^{2\pi}(e^{-x}-e^{-x}\eta\mu x)dx=\int_{0}^{2\pi}e^{-x}dx-\int_{0}^{2\pi}e^{-x}\eta\mu x\,dx. \]
Για τα δύο ολοκληρώματα ισχύει ότι:
\[I_1=\int_{0}^{2\pi}e^{-x}dx=\left[-e^{-x}\right]_{0}^{2\pi}=-e^{-2\pi}-(-e^0)=-e^{-2\pi}+1.\]
\[I_2=\int_{0}^{\frac{\pi}{2}}e^{-x}\eta\mu x\,dx=\int_{0}^{\frac{\pi}{2}}(-e^{-x})'\eta\mu x\,dx=\left[-e^{-x}\eta\mu x\right]_{0}^{\frac{\pi}{2}}+\int_{0}^{\frac{\pi}{2}}e^{-x}\sigma\upsilon\nu x\,dx\]
\[=-e^{-\frac{\pi}{2}}\eta\mu\frac{\pi}{2}-(-e^0\eta\mu 0)+\left[-e^{-x}\sigma\upsilon\nu x\right]_{0}^{\frac{\pi}{2}}-\int_{0}^{\frac{\pi}{2}}-e^{-x}(-\eta\mu x)\,dx\]
\[=-e^{-\frac{\pi}{2}}(1)-0-e^{-\frac{\pi}{2}}\sigma\upsilon\nu\frac{\pi}{2}-(-e^0\sigma\upsilon\nu 0)-I_2\]
\[=-e^{-\frac{\pi}{2}}+1-I_2.\]
Οπότε $2I_2=1-e^{-\frac{\pi}{2}} \Leftrightarrow I_2=\frac{1}{2}\left(1-e^{-\frac{\pi}{2}}\right)$.

Τελικά το εμβαδόν είναι ίσο με $E=e^{-\frac{\pi}{2}}+1-\frac{1}{2}\left(1-e^{-\frac{\pi}{2}}\right)=\frac{1-e^{-\frac{\pi}{2}}}{2}$.
\end{alist}
