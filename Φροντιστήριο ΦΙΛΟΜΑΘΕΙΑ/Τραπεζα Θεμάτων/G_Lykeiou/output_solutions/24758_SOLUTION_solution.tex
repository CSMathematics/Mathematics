\Askhsh{24758_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item α) Παρατηρούμε ότι $g(1)=g(-1)=0$ οπότε από τη σχέση $g(x) \geq 0$ για κάθε $x \in \mathbb{R}$, συμπεραίνουμε ότι η συνάρτηση $g$ παρουσιάζει και στο $1$ και στο $-1$ ελάχιστο το $0$.
Η συνάρτηση $g(x)=(x^2-1)f(x)$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως γινόμενο παραγωγίσιμων συναρτήσεων. Συνεπώς από το θεώρημα Fermat έχουμε ότι $g'(1)=g'(-1)=0$. Όμως $g'(x)=2xf(x)+(x^2-1)f'(x)$ οπότε $g'(1)=0 \Leftrightarrow 2 \cdot 1 \cdot f(1)+(1^2-1)f'(1)=0 \Leftrightarrow 2f(1)=0 \Leftrightarrow f(1)=0$ και $g'(-1)=0 \Leftrightarrow 2 \cdot (-1) \cdot f(-1)+((-1)^2-1)f'(-1)=0 \Leftrightarrow -2f(-1)=0 \Leftrightarrow f(-1)=0$.

\item β) Η $f$ είναι παραγωγίσιμη στο $1$ και στο $-1$ με $f'(1)=\lim_{x \to 1} \frac{f(x)-f(1)}{x-1}=\lim_{x \to 1} \frac{f(x)}{x-1}$ και $f'(-1)=\lim_{x \to -1} \frac{f(x)-f(-1)}{x+1}=\lim_{x \to -1} \frac{f(x)}{x+1}$.
Για κάθε $x \neq 1$ και κοντά στο $1$ έχουμε $\frac{f(x)}{x-1}=\frac{g(x)}{(x^2-1)(x-1)}=\frac{g(x)}{(x+1)(x-1)^2} \geq 0$ οπότε από τις ιδιότητες ορίου έχουμε ότι $\lim_{x \to 1} \frac{f(x)}{x-1} \geq 0$ δηλαδή $f'(1) \geq 0$.
Ομοίως για κάθε $x \neq -1$ και κοντά στο $-1$ έχουμε $\frac{f(x)}{x+1}=\frac{g(x)}{(x^2-1)(x+1)}=\frac{g(x)}{(x-1)(x+1)^2} \leq 0$ οπότε από τις ιδιότητες ορίου έχουμε ότι $\lim_{x \to -1} \frac{f(x)}{x+1} \leq 0$ δηλαδή $f'(-1) \leq 0$.

\item γ) Αφού $-1<1$ και $f'(-1) \leq 0 \leq f'(1)$ συμπεραίνουμε ότι η $f'$ δεν είναι γνησίως φθίνουσα και κατ' επέκταση η $f$ δεν είναι κοίλη.

\item δ) Εφαρμόζοντας παραγοντική ολοκλήρωση έχουμε :
\begin{align*}
\int_{-1}^{1} (x^3-3x)f'(x)dx &= [(x^3-3x)f(x)]_{-1}^{1} - \int_{-1}^{1} (3x^2-3)f(x)dx \\
&= (1^3-3 \cdot 1)f(1) - ((-1)^3-3 \cdot (-1))f(-1) - \int_{-1}^{1} 3(x^2-1)f(x)dx \\
&= (1-3)f(1) - (-1+3)f(-1) - 3\int_{-1}^{1} (x^2-1)f(x)dx \\
&= -2f(1) - 2f(-1) - 3\int_{-1}^{1} (x^2-1)f(x)dx \\
&= 0-0-3\int_{-1}^{1} (x^2-1)f(x)dx \leq 0
\end{align*}
αφού $(x^2-1)f(x)=g(x) \geq 0$ για κάθε $x \in \mathbb{R}$ οπότε $\int_{-1}^{1} (x^2-1)f(x)dx \geq 0$ και τελικά $-3\int_{-1}^{1} (x^2-1)f(x)dx \leq 0$. Συνεπώς $\int_{-1}^{1} (x^3-3x)f'(x)dx \leq 0$.
\end{alist}
