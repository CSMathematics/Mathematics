\Askhsh{36816_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{\eta \mu x}{x}$ για κάθε $x \in \PARENS{0, \frac{\pi}{2}}$.
\begin{alist}
    \item Αν η συνάρτηση $f$ είναι συνεχής στο $x_0 = 0$, να ορίσετε την $f(0)$. \monades{X}
    \item Να γράψετε τον τύπο της συνάρτησης $f$ (με τον ορισμό της στο $x=0$). \monades{X}
    \item Να αποδείξετε ότι η συνάρτηση $f$ είναι γνησίως φθίνουσα στο $\left[0, \frac{\pi}{2}\right]$. \monades{X}
    \item Για $x \in \left[\frac{\pi}{6}, \frac{\pi}{4}\right]$, να αποδείξετε ότι $\frac{\sqrt{2}}{6} \le \int_{\frac{\pi}{6}}^{\frac{\pi}{4}} f(x) dx \le \frac{1}{4}$. \monades{X}
\end{alist}
