\kefalaio{ΛΥΣΗ}
\begin{alist}
    \item [\textalpha)]
    \begin{rlist}
        \item Η γραφική παράσταση της $f$ εφάπτεται της $\varepsilon: y=\frac{1}{4}$ στο $x_0 = 0$ άρα οι συντεταγμένες του σημείου $(0,f(0))$ επαληθεύουν την εξίσωση της $\varepsilon$, οπότε $f(0)=\frac{1}{4}$ και ο συντελεστής διεύθυνσης $\lambda = 0$ ισούται με την παράγωγο στο $x_0 = 0$, δηλαδή $f'(0)=0$.
        \item Είνα\kern0pt\accent"7F
        \[ \lim_{x\to 0} \frac{4f(x)-1}{x \eta\mu x \cdot f(x)} = 4 \cdot \lim_{x\to 0} \frac{f(x)-\frac{1}{4}}{x \eta\mu x \cdot f(x)} \]
        \[ = 4 \cdot \lim_{x\to 0} \frac{f(x)-f(0)}{x \eta\mu x \cdot f(x)} = 4 \cdot \lim_{x\to 0} \frac{\frac{f(x)-f(0)}{x-0}}{\frac{\eta\mu x \cdot f(x)}{x}} \]
        \[ = 4 \cdot \lim_{x\to 0} \frac{\frac{f(x)-f(0)}{x}}{\frac{\eta\mu x}{x} \cdot f(x)} \stackrel{\substack{f \text{ συνεχής στο } 0 \\ \text{ως παραγωγίσιμη σε αυτό}}}{=} 4 \cdot \frac{f'(0)}{1 \cdot f(0)} = 4 \cdot \frac{0}{1 \cdot \frac{1}{4}} = 0. \]
    \end{rlist}

    \item [\textbeta)]
    \begin{rlist}
        \item Γνωρίζουμε ότι η συνάρτηση $f$ είναι κυρτή στο $\mathbb{R}$ οπότε η $f'$ είναι γνησίως αύξουσα.

        Είναι $f'(0)=0$.

        Για κάθε $x \in (0,1]$ ισχύει: $x>0 \xrightarrow{f' \text{ γν. αύξουσα}} f'(x)>f'(0) \Rightarrow f'(x)>0$.

        Άρα $f'(x) \ge 0$, για κάθε $x \in [0,1]$.
        \item Επειδή η $f'$ είναι γνησίως αύξουσα, είναι και συνάρτηση "$1-1$".

        $f'(x)=0 \Leftrightarrow f'(x)=f'(0) \xrightarrow{f' \text{ 1-1}} x=0$.

        Οπότε το ζητούμενο εμβαδό είναι ίσο με:
        \[ E = \int_{0}^{1} |f'(x)|dx \overset{(\beta i)}{=} \int_{0}^{1} f'(x)dx = f(1)-f(0)=1-\frac{1}{4}=\frac{3}{4} \text{ τετραγωνικές μονάδες}. \]
    \end{rlist}
\end{alist}
