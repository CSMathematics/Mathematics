\Askhsh{35602_SOLUTION}{1}
\begin{Lysh}
    \begin{alist}
        \item [α)] Η $f$ ορίζεται στο διάστημα $A = (-\infty, 1) \cup (1, +\infty)$ και
        \begin{align*}
            f(x)-y &= \frac{x^2-2x}{x-1}-(x-1) \\
            &= \frac{x^2-2x-(x-1)^2}{x-1} \\
            &= \frac{x^2-2x-(x^2-2x+1)}{x-1} \\
            &= \frac{x^2-2x-x^2+2x-1}{x-1} \\
            &= \frac{-1}{x-1} = \frac{1}{1-x}
        \end{align*}
        οπότε: $\lim_{x\to -\infty} (f(x)-y) = \lim_{x\to -\infty} \left(\frac{1}{1-x}\right) = 0$ και όμοια $\lim_{x\to +\infty} (f(x)-y) = \lim_{x\to +\infty} \left(\frac{1}{1-x}\right) = 0,$
        επομένως η ευθεία $(\varepsilon): y = x-1$ είναι πλάγια ασύμπτωτη της $C_f$.

        Εναλλακτικά: Αρκούσε ο υπολογισμός του ενός ορίου.

        \item [β)] Επίσης:
        \[
        \lim_{x\to 1^-} f(x) = \lim_{x\to 1^-} \frac{x^2-2x}{x-1} = \lim_{x\to 1^-} (x^2-2x) \frac{1}{x-1} = (-1) \cdot (-\infty) = +\infty \text{ και όμοια}
        \]
        \[
        \lim_{x\to 1^+} f(x) = \lim_{x\to 1^+} \frac{x^2-2x}{x-1} = \lim_{x\to 1^+} (x^2-2x) \frac{1}{x-1} = (-1) \cdot (+\infty) = -\infty,
        \]
        άρα η ευθεία $(\varepsilon'): x=1$ είναι κατακόρυφη ασύμπτωτη της $C_f$.

        Εναλλακτικά: Αρκούσε ο υπολογισμός του ενός ορίου.

        \item [γ)] Η $f$ είναι παραγωγίσιμη στο $A = (-\infty, 1) \cup (1, +\infty)$ με
        \begin{align*}
            f'(x)&=\left(\frac{x^2-2x}{x-1}\right)' \\
            &= \frac{(2x-2)\cdot(x-1)-(x^2-2x)\cdot 1}{(x-1)^2} \\
            &= \frac{2x^2-2x-2x+2-x^2+2x}{(x-1)^2} \\
            &= \frac{x^2-2x+2}{(x-1)^2} \\
            &= \frac{x^2-2x+1+1}{(x-1)^2} \\
            &= \frac{(x-1)^2+1}{(x-1)^2} > 0 \text{ στο } A=(-\infty, 1) \cup (1, +\infty)\text{,}
        \end{align*}
        οπότε η $f$ είναι γνησίως αύξουσα στα διαστήματα $(-\infty, 1)$ και $(1, +\infty)$.
    \end{alist}
\end{Lysh}
