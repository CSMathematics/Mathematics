\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\Askhseis{ΛΥΣΗ}

\begin{alist}
\item Η συνάρτηση $g$ ορίζεται, όταν και μόνο όταν, $x^2 > 0 \Leftrightarrow x \neq 0$.
Επομένως, το πεδίο ορισμού της $g$ είναι το σύνολο $\mathbb{R}^*$.
\item Η συνάρτηση $f$ έχει πεδίο ορισμού το $D_f = \mathbb{R}$, ενώ η συνάρτηση $g$ το $D_g = \mathbb{R}^*$.
Για να ορίζεται η $g(f(x))$ πρέπει και αρκεί:
\[ x \in D_f \quad \text{και} \quad f(x) \in D_g \]
ή ισοδύναμα
\[ \begin{cases} x \in \mathbb{R} \\ e^{3x+2} \neq 0 \end{cases} \Leftrightarrow \begin{cases} x \in \mathbb{R} \\ x \in \mathbb{R} \end{cases} \Leftrightarrow x \in \mathbb{R}. \]
Επομένως, ορίζεται η $g \circ f$ και είναι
\[ (g \circ f)(x) = g(f(x)) = g(e^{3x+2}) = \ln\left((e^{3x+2})^2\right) = \ln(e^{6x+4}) = (6x+4)\ln e = 6x + 4, \quad \text{για κάθε } x \in \mathbb{R}. \]
\item Είναι για $x \neq 0$:
\[ \frac{(g \circ f)(x) - \eta \mu^2 x - 4}{x} = \frac{6x + 4 - \eta \mu^2 x - 4}{x} = \frac{6x - \eta \mu^2 x}{x} = \frac{6x}{x} - \frac{\eta \mu^2 x}{x} = 6 - \eta \mu x \cdot \frac{\eta \mu x}{x} \]
Αλλά γνωρίζουμε ότι
\[ \lim_{x \to 0} 6 = 6, \quad \lim_{x \to 0} (\eta \mu x) = 0 \quad \text{και} \quad \lim_{x \to 0} \frac{\eta \mu x}{x} = 1 \]
Επομένως, έχουμε
\[ \lim_{x \to 0} \frac{(g \circ f)(x) - \eta \mu^2 x - 4}{x} = 6 - 0 \cdot 1 = 6. \]
\end{alist}
