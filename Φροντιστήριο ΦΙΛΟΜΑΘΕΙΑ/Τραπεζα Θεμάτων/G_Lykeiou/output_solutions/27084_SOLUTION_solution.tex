\section*{\color{maincolor} ΛΥΣΗ}

\begin{alist}
\item[α)] Η συνάρτηση $f$ είναι παραγωγίσιμη στο $(0, +\infty)$ με
\[ f'(x) = \left(x + \frac{1}{x}\right)' = 1 - \frac{1}{x^2} = \frac{x^2 - 1}{x^2} \]
Λύνουμε την εξίσωση
\[ f'(x) = 0 \iff \frac{x^2 - 1}{x^2} = 0 \overset{x>0}{\iff} x^2 - 1 = 0 \iff x = 1 \]
Τα πρόσημα της παραγώγου της $f$ φαίνονται στον ακόλουθο πίνακα:
\begin{center}
\begin{tabular}{|c|c|c|c|}
    \hline
    $x$ & $0$ & $1$ & $+\infty$ \\
    \hline
    $f'(x)$ & $-$ & $0$ & $+$ \\
    \hline
    $f(x)$ & $\searrow$ & & $\nearrow$ \\
    \hline
\end{tabular}
\end{center}
Άρα, η συνάρτηση $f$ είναι:
\begin{itemize}
    \item γνήσίως φθίνουσα στο $(0,1]$, αφού είναι συνεχής στο $(0,1)$ και ισχύει $f'(x) < 0$ στο $(0,1)$.
    \item γνήσίως αύξουσα στο $[1, +\infty)$, αφού είναι συνεχής στο $[1, +\infty)$ και ισχύει $f'(x) > 0$ στο $(1, +\infty)$.
\end{itemize}
Η $f$ είναι γνησίως φθίνουσα στο $(0,1]$, γνησίως αύξουσα στο $[1, +\infty)$ και είναι συνεχής στο $x_0 = 1$. Επομένως, η $f$ παρουσιάζει ολικό ελάχιστο στο $x_0 = 1$, το $f(1) = 2$.

\item[β)] Η συνάρτηση $f$ είναι δύο φορές παραγωγίσιμη στο $(0, +\infty)$ με
\[ f''(x) = \left(x + \frac{1}{x}\right)'' = \frac{2}{x^3} > 0 \text{ για κάθε } x \in (0, +\infty) \]
Συνεπώς, η $f$ είναι κυρτή, δηλαδή στρέφει τα κοίλα προς τα πάνω, στο $(0, +\infty)$.

\item[γ)] Είναι:
\[ \lim_{x \to +\infty} (f(x) - x) = \lim_{x \to +\infty} \left(x + \frac{1}{x} - x\right) = \lim_{x \to +\infty} \frac{1}{x} = 0 \]
Άρα, η ευθεία $y = x$ είναι ασύμπτωτη της γραφικής παράστασης της συνάρτησης $f$ στο $+\infty$.
\end{alist}
