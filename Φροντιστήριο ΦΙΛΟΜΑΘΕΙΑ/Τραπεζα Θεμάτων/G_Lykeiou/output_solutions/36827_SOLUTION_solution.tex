    \item [Λύση]

\begin{alist}
    \item Παρατηρούμε ότι $g'(x) = (-x)' \cdot e^{-x} = -e^{-x} < 0$ για κάθε $x \in \mathbb{R}$, άρα η $g$ είναι γνησίως φθίνουσα στο $\mathbb{R}$, άρα είναι και «1 -- 1», επομένως υπάρχει η συνάρτηση $g^{-1}$.
    
    Θέτουμε $y = e^{-x} > 0$ για κάθε $x \in \mathbb{R}$, άρα $\ln y = -x \Leftrightarrow x = -\ln y$ για κάθε $y \in (0, +\infty)$.
    
    Έτσι, $g^{-1}(y) = -\ln y$ για κάθε $y > 0$, δηλαδή $g^{-1}(x) = -\ln x$ για κάθε $x > 0$.
    
    Ώστε $g^{-1} = -f$.
    
    \item Το πεδίο ορισμού $D$ της συνάρτησης $g \circ f$, αποτελείται από τους αριθμούς $x$ για τους οποίους ισχύει $x \in A_f = (0, +\infty)$ και $f(x) = \ln x \in A_g = \mathbb{R}$.
    
    Ώστε $D = (0, +\infty)$.
    
    Για κάθε $x \in D$ είναι $(g \circ f)(x) = g(f(x)) = g(\ln x) = e^{-\ln x} = \frac{1}{e^{\ln x}} = \frac{1}{x}$.
    
    \item Παρατηρούμε ότι η συνάρτηση $h$ είναι παραγωγίσιμη, άρα και συνεχής στο $[2, 8]$, με $h'(x) = -\frac{1}{x^2}$.
    
    Επομένως, από το Θεώρημα Μέσης Τιμής, θα υπάρχει $\xi \in (2, 8)$ ώστε $h'(\xi) = \frac{h(8)-h(2)}{8-2}$, άρα
    \[ -\frac{1}{\xi^2} = \frac{\frac{1}{8} - \frac{1}{2}}{6} = \frac{\frac{1-4}{8}}{6} = \frac{-\frac{3}{8}}{6} = -\frac{3}{48} = -\frac{1}{16}. \]
    Ώστε $\xi^2 = 16$, άρα $\xi = 4$.
\end{alist}
