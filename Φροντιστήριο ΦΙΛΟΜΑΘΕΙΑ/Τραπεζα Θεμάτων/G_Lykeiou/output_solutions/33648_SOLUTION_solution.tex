    \begin{rlist}
        \item [i.] Η συνάρτηση $f$ είναι παραγωγίσιμη στο πεδίο ορισμού της $\mathbb{D}_f = (0, +\infty)$ και ισχύει:
        \[ f'(x) = 2 \ln x \cdot \frac{1}{x} - \frac{1}{x} \]
        Το πρόσημο της $f'(x)$ φαίνεται στον επόμενο πίνακα.
        \[
        \begin{tabular}{|c|c|c|c|}
        \hline
        $x$ & $(0, \sqrt{e})$ & $\sqrt{e}$ & $(\sqrt{e}, +\infty)$ \\
        \hline
        $f'(x)$ & $-$ & $0$ & $+$ \\
        \hline
        $f(x)$ & \makebox[0pt][c]{\Large$\searrow$} & \text{τοπ. ελάχ.} & \makebox[0pt][c]{\Large$\nearrow$} \\
        \hline
        \end{tabular}
        \]
        Από τον παραπάνω πίνακα συμπεραίνουμε ότι η $f$,
        \begin{itemize}
            \item είναι γνησίως φθίνουσα στο διάστημα $(0, 1]$ και γνησίως αύξουσα στο $[1, +\infty)$
            \item παρουσιάζει ολικό ελάχιστο για $x=1$, το $f(1) = 0$.
        \end{itemize}
        \item [ii.] Η $f'(x)$ είναι παραγωγίσιμη με
        \[
        f''(x) = 2 \frac{x \cdot \frac{1}{x} - \ln x \cdot 1}{x^2} - \frac{1}{x^2} = 2 \frac{1 - \ln x}{x^2} - \frac{1}{x^2} = \frac{2 - 2\ln x - 1}{x^2} = \frac{1 - 2\ln x}{x^2}
        \]
        Το πρόσημο της $f''(x)$ φαίνεται στον επόμενο πίνακα.
        \[
        \begin{tabular}{|c|c|c|c|}
        \hline
        $x$ & $(0, e)$ & $e$ & $(e, +\infty)$ \\
        \hline
        $f''(x)$ & $+$ & $0$ & $-$ \\
        \hline
        $f(x)$ & \makebox[0pt][c]{\Large$\smile$} & \text{Σ.Κ.} & \makebox[0pt][c]{\Large$\frown$} \\
        \hline
        \end{tabular}
        \]
        Από τον παραπάνω πίνακα συμπεραίνουμε ότι η $f$,
        \begin{itemize}
            \item είναι κυρτή στο διάστημα $(0, e]$ και κοίλη στο $[e, +\infty)$
            \item παρουσιάζει καμπή για $x=e$. Το σημείο καμπής είναι το $M(e, 1)$.
        \end{itemize}
    \end{rlist}
\item [β)] Είναι:
\[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} (\ln^2 x - \ln x) = +\infty \]
\[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} (\ln^2 x - \ln x) = \lim_{x \to +\infty} \ln x (\ln x - 1) = +\infty \]
οπότε η $C_f$ έχει κατακόρυφη ασύμπτωτη την $x=0$, ενώ δεν έχει οριζόντια ή πλάγια ασύμπτωτη στο $+\infty$. Η γραφική παράσταση της $f$ στο ίδιο σύστημα με εκείνη της $g$ φαίνονται στο σχήμα που ακολουθεί.

        \[
        f(x) = g(x) \Leftrightarrow \ln^2 x - \ln x = \ln x \Leftrightarrow \ln^2 x - 2 \ln x = 0 \Leftrightarrow \ln x (\ln x - 2) = 0 \Leftrightarrow x = 1 \text{ ή } x = e

        Επομένως τα κοινά σημεία των $C_f, C_g$ είναι τα $(1, 0)$ και $(e, 1)$.
        \item [ii.] Στο διάστημα $[1, e]$ ισχύει $g(x) \ge f(x)$, οπότε το μήκος του τμήματος AB είναι
        \[ (AB) = \ln \alpha - \ln^2 \alpha \]

        Θεωρούμε τη συνάρτηση $d(\alpha) = \ln \alpha - \ln^2 \alpha$, $\alpha \in (1, e)$. Η συνάρτηση είναι παραγωγίσιμη με
        \[ d'(\alpha) = \frac{1}{\alpha} - 2\ln \alpha \cdot \frac{1}{\alpha} = \frac{1}{\alpha} (1 - 2\ln \alpha) \]
        και ισχύει
        \[ d'(\alpha) \ge 0 \Leftrightarrow 1 - 2\ln \alpha \ge 0 \Leftrightarrow 1 \ge 2\ln \alpha \Leftrightarrow \frac{1}{2} \ge \ln \alpha \Leftrightarrow e^{1/2} \ge \alpha \Leftrightarrow \sqrt{e} \ge \alpha \]
        οπότε η $d$ είναι γνησίως αύξουσα στο διάστημα $(1, \sqrt{e}]$ και γνησίως φθίνουσα στο $[\sqrt{e}, e)$. Επομένως η συνάρτηση $d$ παίρνει την μέγιστη τιμή της, δηλαδή το μήκος του τμήματος AB γίνεται μέγιστο, όταν $\alpha = \sqrt{e}$.
