\textbf{ΛΥΣΗ}
\begin{alist}
\item Η $f$ είναι συνεχής στο διάστημα $(-\infty,0)$. Είναι $f'(x) = \frac{2}{x^3} < 0$ για κάθε $x \in (-\infty,0)$.
Επομένως η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty,0)$.
\item Επιπλέον, ισχύει ότι
\[ K = \lim_{x \to -\infty} f(x) = 1 \text{ και } \Lambda = \lim_{x \to 0^-} f(x) = -\infty, \text{ διότι} \]
\[ \lim_{x \to -\infty} \frac{1}{x^2} = 0 \text{ και } \lim_{x \to 0^-} \frac{1}{x^2} = +\infty \]
Επειδή όμως η συνάρτηση $f$ είναι γνησίως φθίνουσα και συνεχής στο διάστημα $(-\infty,0)$, τότε το σύνολο τιμών της στο διάστημα αυτό είναι το διάστημα $(\Lambda, K) = (-\infty, 1)$.

$\underline{2^{ος} \text{ τρόπος}:}$

Η $f$ είναι ορισμένη στο σύνολο $B = (-\infty,0)$.
Το σύνολο τιμών της $f$ είναι το $f(B) = \{y / y = f(x) \text{ για κάποιο } x \in B\}$.
Αρκεί λοιπόν να βρούμε τις τιμές του $y$ για τις οποίες η εξίσωση $y=f(x)$ έχει μία τουλάχιστον λύση, ως προς $x$, στο $B$.
\[ y = f(x) \Leftrightarrow y = 1 - \frac{1}{x^2} \Leftrightarrow \frac{1}{x^2} = 1 - y \quad (1) \]
Αν $1-y \leq 0$, τότε η (1) είναι αδύνατη,
Αν $1-y > 0 \Leftrightarrow y < 1$, έχουμε από την (1): $x^2 = \frac{1}{1-y} \Leftrightarrow x = \sqrt{\frac{1}{1-y}} \text{ ή } x = -\sqrt{\frac{1}{1-y}}$.
Επειδή όμως $x < 0$, είναι $x = -\sqrt{\frac{1}{1-y}}$, με $y < 1$.
Επομένως το σύνολο τιμών της $f$ είναι το $f(B) = (-\infty,1)$.
\item
\begin{rlist}
\item Αφού η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty,0)$ έπεται ότι είναι ``1-1''.

$\underline{2^{ος} \text{ τρόπος}:}$ Έστω $x_1, x_2 \in (-\infty,0)$ με $f(x_1) = f(x_2)$. Τότε είναι:
\[ 1 - \frac{1}{x_1^2} = 1 - \frac{1}{x_2^2} \Rightarrow -\frac{1}{x_1^2} = -\frac{1}{x_2^2} \Rightarrow \frac{1}{x_1^2} = \frac{1}{x_2^2} \Rightarrow x_1^2 = x_2^2 \Rightarrow x_1 = x_2. \]
Επομένως η συνάρτηση $f$ είναι ``1-1''.
\item Για να βρούμε την αντίστροφη της συνάρτησης $f$ θέτουμε $y = f(x)$ και λύνουμε ως προς $x$. Από το ερώτημα β) όμως, είναι $x = -\sqrt{\frac{1}{1-y}}$, με $y < 1$.
Επομένως, $f^{-1}(y) = -\sqrt{\frac{1}{1-y}}$, με $y < 1 \Leftrightarrow f^{-1}(y) = -\frac{1}{\sqrt{1-y}}$, με $y < 1$.
Οπότε η αντίστροφη της $f$ είναι η συνάρτηση $f^{-1}(x) = -\frac{1}{\sqrt{1-x}}$, με $x < 1$.
\end{rlist}
\end{alist}
