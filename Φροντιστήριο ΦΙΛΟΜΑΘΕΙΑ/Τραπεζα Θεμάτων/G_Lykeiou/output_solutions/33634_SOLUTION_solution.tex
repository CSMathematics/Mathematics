\kefalaio{ΑΥΣΗ}
\begin{alist}
    \item [α)] $I + J = \int_0^{\frac{\pi}{2}} \eta \mu^2 x \,dx + \int_0^{\frac{\pi}{2}} \sigma υ \nu^2 x \,dx = \int_0^{\frac{\pi}{2}} (\eta \mu^2 x + \sigma υ \nu^2 x) \,dx = \int_0^{\frac{\pi}{2}} 1 \,dx = \left[x\right]_0^{\frac{\pi}{2}} = \frac{\pi}{2} - 0 = \frac{\pi}{2}.$
    \item [β)] Αν θέσουμε στο ολοκλήρωμα $I = \int_0^{\frac{\pi}{2}} \eta \mu^2 x \,dx$ ως $u = \frac{\pi}{2} - x$ έχουμε ότι $du = -dx$, για $x=0$ είναι $u = \frac{\pi}{2}$ και για $x=\frac{\pi}{2}$ είναι $u=0$ οπότε το ολοκλήρωμα γίνεται
    \[ I = \int_{\frac{\pi}{2}}^0 \eta \mu^2 \left(\frac{\pi}{2} - u\right) du = \int_0^{\frac{\pi}{2}} \sigma υ \nu^2 u du = J. \]
    Αφού $I+J = \frac{\pi}{2}$ και $I=J$ έχουμε $I+I=\frac{\pi}{2} \Leftrightarrow 2I = \frac{\pi}{2} \Leftrightarrow I = \frac{\pi}{8}.$
    \item [γ)]
    \begin{rlist}
        \item Το εμβαδόν που περικλείεται μεταξύ της $C_f$, του άξονα $x'x$ και της ευθείας $x = \frac{\pi}{2}$ είναι το $I$. Το τετράγωνο $OBA\Gamma$ έχει εμβαδόν $\frac{\pi^2}{4}$, οπότε αφού $I+J=\frac{\pi^2}{4} \Rightarrow J = \frac{\pi}{4} - I \Rightarrow J = (\text{OBA\Gamma}) - I$ (Υπάρχει ανακολουθία στο αρχικό κείμενο, καθώς στο α) βρέθηκε $I+J=\frac{\pi}{2}$).\par
        συμπεραίνουμε ότι το εμβαδόν που περικλείεται μεταξύ της $C_f$, του άξονα $y'y$ και της ευθείας $y = \frac{\pi}{2}$ είναι το $J$.
        \item Η ευθεία OA έχει εξίσωση $y-0 = \frac{\frac{\pi}{2}-0}{\frac{\pi}{2}-0}(x-0) \Rightarrow y=x$ και αφού OA διαγώνιος του τετραγώνου OBA$\Gamma$, το χωρίζει σε δύο ίσα άρα και ισοδύναμα τρίγωνα, δηλαδή $(\text{ΟΑΒ}) = (\text{ΟΑ\Gamma})$. Είναι $I = (\text{ΟΑΒ}) - E(\Omega_2) + E(\Omega_1)$, $J = (\text{ΟΑ\Gamma}) + E(\Omega_2) - E(\Omega_1)$ και αφού $I=J$, $(\text{ΟΑΒ}) = (\text{ΟΑ\Gamma})$ συμπεραίνουμε ότι $E(\Omega_2) - E(\Omega_1) = -E(\Omega_2) + E(\Omega_1) \Leftrightarrow 2E(\Omega_2) = 2E(\Omega_1) \Leftrightarrow E(\Omega_2) = E(\Omega_1)$. Εναλλακτικά,
        \[ E(\Omega_1) = E(\Omega_2) \Leftrightarrow \int_0^{x_0} (x-f(x))dx = \int_{x_0}^{\frac{\pi}{2}} (f(x)-x)dx \]
        \[ \int_0^{x_0} x dx - \int_0^{x_0} f(x) dx = \int_{x_0}^{\frac{\pi}{2}} f(x) dx - \int_{x_0}^{\frac{\pi}{2}} x dx \]
        \[ \int_0^{\frac{\pi}{2}} x dx = \int_0^{\frac{\pi}{2}} f(x) dx \Leftrightarrow \left[\frac{x^2}{2}\right]_0^{\frac{\pi}{2}} = I \Leftrightarrow \frac{\pi^2}{8} = I \]
        που ισχύει.
    \end{rlist}
    \Askhseis{Σημείωση}: Το K είναι σημείο καμπής της $C_f$.
    \par Πράγματι, η συνάρτηση $f(x) = \frac{\pi}{2} \eta \mu^2 x$ είναι δύο φορές παραγωγίσιμη στο $\left[0, \frac{\pi}{2}\right]$ με
    $f'(x) = \pi \cdot \eta \mu x \cdot \sigma υ \nu x$ και $f''(x) = \pi \cdot \sigma υ \nu^2 x - \pi \cdot \eta \mu^2 x$. Είναι
    $f''(x) = 0 \Leftrightarrow \pi \cdot \sigma υ \nu^2 x - \pi \cdot \eta \mu^2 x = 0 \Leftrightarrow \sigma υ \nu^2 x = \eta \mu^2 x$.
    Για $x \in \left(0, \frac{\pi}{2}\right)$ είναι $\sigma υ \nu x > 0$ και $\eta \mu x > 0$ οπότε
    $\sigma υ \nu^2 x = \eta \mu^2 x \Leftrightarrow \sigma υ \nu x = \eta \mu x \Leftrightarrow \epsilon \phi x = 1 \Leftrightarrow x = \frac{\pi}{4}$. Επίσης
    $f''(x) > 0 \Leftrightarrow \pi \cdot \sigma υ \nu^2 x - \pi \cdot \eta \mu^2 x > 0 \Leftrightarrow \sigma υ \nu^2 x > \eta \mu^2 x \Leftrightarrow$
    $\sigma υ \nu x > \eta \mu x \Leftrightarrow \epsilon \phi x < 1 \Leftrightarrow 0 < x < \frac{\pi}{4}$.
    Συνεπώς η $f$ είναι κυρτή στο $\left[0, \frac{\pi}{4}\right]$, κοίλη στο $\left[\frac{\pi}{4}, \frac{\pi}{2}\right]$ και παρουσιάζει καμπή για $x = \frac{\pi}{4}$.
    Αφού $f\left(\frac{\pi}{4}\right) = \frac{\pi}{2} \eta \mu^2 \frac{\pi}{4} = \frac{\pi}{2} \left(\frac{\sqrt{2}}{2}\right)^2 = \frac{\pi}{2} \cdot \frac{2}{4} = \frac{\pi}{2} \cdot \frac{1}{2} = \frac{\pi}{4}$ το σημείο καμπής είναι το $\left(\frac{\pi}{4}, \frac{\pi}{4}\right)$ που ανήκει στην ευθεία $y=x$ και επομένως είναι το σημείο K.
\end{alist}
