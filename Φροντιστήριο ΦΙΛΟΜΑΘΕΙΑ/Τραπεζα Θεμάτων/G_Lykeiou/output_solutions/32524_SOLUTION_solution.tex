    \item [Ύλη από εικόνα]

\textbf{ΛΥΣΗ}

\begin{alist}
    \item [α)] Για κάθε $x \in (0, +\infty)$ ισχύει:
    \[ g'(x) = \left(\frac{e}{x} - \ln x\right)' = \left(\frac{e}{x}\right)' - (\ln x)' = e(x^{-1})' - \frac{1}{x} = -ex^{-2} - \frac{1}{x} = -\frac{e}{x^2} - \frac{1}{x} < 0. \]
    Επομένως η $g$ είναι γνησίως φθίνουσα.

    \item [β)] Η $g$ είναι συνάρτηση $1-1$ ως γνησίως μονότονη.
    Με $x \in (0, +\infty)$ έχουμε:
    \begin{align*}
        e(1-x) = x\ln x &\Leftrightarrow e - ex = x\ln x \\
        &\Leftrightarrow e - x\ln x = ex \\
        &\Leftrightarrow \frac{e}{x} - \ln x = e \\
        &\Leftrightarrow g(x) = g(1) \\
        &\Leftrightarrow x=1
    \end{align*}
    Επομένως η εξίσωση $e(1-x) = x\ln x$ έχει ακριβώς μία λύση την $x=1$.

    \item [γ)]
    \begin{rlist}
        \item Η συνάρτηση $f$ ορίζεται μόνο για εκείνους τους πραγματικούς αριθμούς $x$ που είναι τέτοιοι ώστε: $x>0$ και $e-x\ln x - ex \neq 0$.
        Είναι $e-x\ln x - ex \neq 0 \Leftrightarrow e(1-x) \neq x\ln x \overset{(\beta)}{\Leftrightarrow} x \neq 1$.
        Οπότε το πεδίο ορισμού της συνάρτησης $f$ είναι $D_f = (0,1)\cup(1,+\infty)$.

        \item Ισχύουν:
        \begin{itemize}
            \item $\lim_{x \to 1^+} (e-x\ln x - ex) = e - 1\cdot\ln 1 - e\cdot 1 = 0$ και
            \item επειδή η συνάρτηση $g$ είναι γνησίως φθίνουσα, για $x>1$ είναι $g(x) < g(1) \Rightarrow \frac{e}{x} - \ln x < e \Rightarrow e - x\ln x < ex \Rightarrow e - x\ln x - ex < 0$.
            \item Άρα $\lim_{x \to 1^+} \left(\frac{1}{e-x\ln x - ex}\right) = -\infty$.
            \item Επιπλέον είναι $\lim_{x \to 1^+} (x^2+x) = 1^2+1=2$.
            \item Έχουμε $\lim_{x \to 1^+} f(x) = \lim_{x \to 1^+} \left(\frac{x^2+x}{e-x\ln x - ex}\right) = \lim_{x \to 1^+} \left[ \frac{1}{e-x\ln x - ex} \cdot (x^2+x) \right] = -\infty$.
        \end{itemize}
    \end{rlist}
\end{alist}
