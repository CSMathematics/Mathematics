\noindent{\bfseries ΛΥΣΗ}

\begin{alist}
    \item Η συνάρτηση $f$ είναι συνεχής στο πεδίο ορισμού της, άρα και στο κλειστό διάστημα $[1,2]$ ως αθροίσματα γινομένων πολυωνυμικής με εκθετική και λογαριθμικής.
    \begin{itemize}
        \item $f'(1) = (1-2)e^1 + (1-1)\ln1 = -e < 0$
        \item $f'(2) = (2-2)e^2 + (2-1)\ln2 = \ln2 > 0$
    \end{itemize}
    Αφού $f'(1) < 0 < f'(2)$, επομένως από Θεώρημα Bolzano η εξίσωση $f'(x) = 0$ έχει μία τουλάχιστον ρίζα $x_0 \in (1,2)$, δηλαδή η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ σε ένα τουλάχιστον σημείο με τετμημένη $x_0 \in (1,2)$. \monades{X}
    \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο πεδίο ορισμού της, ως αθροίσματα γινομένων πολυωνυμικής με εκθετική και λογαριθμικής, με
    \[ f'(x) = e^x + (x-2)e^x + \ln x + \frac{x-1}{x} \]
    \[ = e^x (1 + x-2) + \ln x + \frac{x-1}{x} \]
    \[ = e^x (x-1) + \ln x + \frac{x-1}{x} \]
    \[ = (x-1) \left(e^x + \frac{1}{x}\right) + \ln x \]
    Για να υπάρχει μοναδικό σημείο της $C_f$ στο οποίο η εφαπτομένη ευθεία θα είναι οριζόντια, δηλαδή παράλληλη στον $x'x$, θα πρέπει η εξίσωση $f'(x)=0$ να έχει μοναδική ρίζα.
    Παρατηρούμε ότι το 1 είναι προφανής ρίζα της εξίσωσης $f'(x)=0$, αφού
    $f'(1) = (1-1)(e^1+1/1)+\ln1 = 0$.
    \begin{itemize}
        \item $f'(x) > 0$ για $x>1$ είναι $\ln x > 0$ και $(x-1) \left(e^x + \frac{1}{x}\right) > 0$, αφού $\left(e^x + \frac{1}{x}\right) > 0$ για κάθε $x>0$ και
        \item $f'(x) < 0$ για $x \in (0,1)$ είναι $\ln x < 0$ και $(x-1) \left(e^x + \frac{1}{x}\right) < 0$
    \end{itemize}
    Κάνοντας τον πίνακα προσήμου της $f'$ έχουμε:

    \[
    \begin{tabular}{|c|c|c|c|}
    \hline
    $x$ & 0 & 1 & $+\infty$ \\
    \hline
    $f'$ & $-$ & 0 & $+$ \\
    \hline
    $f$ & $\searrow$ & & $\nearrow$ \\
    \hline
    \end{tabular}
    \]

    Άρα η $f$ είναι γνησίως φθίνουσα συνάρτηση στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.
    Επομένως,
    Αν $\Delta_1 = (0,1]$, τότε $f(\Delta_1) = \left[f(1), \lim_{x \to 0^+} f(x)\right) = [-e,+\infty)$.

    Γιατί:
    \[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \left[ (x-2)e^x + (x-1)\ln x \right] = +\infty \]
    Αφού $\lim_{x \to 0^+} (x-2)e^x = (0-2)e^0 = -2$ και $\lim_{x \to 0^+} (x-1)\ln x = -1(-\infty) = +\infty$.
    Αν $\Delta_2=[1,+\infty)$, τότε $f(\Delta_2) = [f(1), \lim_{x \to +\infty} f(x)] = [-e, +\infty)$
    Γιατί:
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \left[ (x-2)e^x + (x-1)\ln x \right] = +\infty \]
    Αφού $\lim_{x \to +\infty} (x-2)e^x = (+\infty) \cdot (+\infty) = +\infty$ και $\lim_{x \to +\infty} (x-1)\ln x = (+\infty) \cdot (+\infty) = +\infty$.
    Το $0 \in [-e, +\infty)$, άρα υπάρχει ένα $x_1 \in (0,1]$ τέτοιο ώστε $f(x_1) = 0$, το οποίο είναι και μοναδικό γιατί η $f$ είναι γνησίως φθίνουσα στο $(0,1)$.
    Το $0 \in [-e, +\infty)$, άρα υπάρχει ένα $x_2 \in [1,+\infty)$ τέτοιο ώστε $f(x_2) = 0$, το οποίο είναι επίσης μοναδικό γιατί η $f$ είναι γνησίως αύξουσα στο $[1,+\infty)$.
    Επομένως, η γραφική παράσταση της $f$ τέμνει πράγματι τον άξονα $x'x$ σε δύο ακριβώς σημεία τα $x_1$ και $x_2$ είναι το $x_0$ του α) ερωτήματος.
\end{alist}
