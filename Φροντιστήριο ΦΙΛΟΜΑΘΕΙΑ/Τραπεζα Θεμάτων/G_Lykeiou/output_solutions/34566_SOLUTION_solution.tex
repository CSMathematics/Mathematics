    \item [Λύση από εικόνα]
\end{ylh}
ΛΥΣΗ
\begin{alist}
\item Αρκεί να αποδείξουμε ότι για την συνάρτηση $g(x) = xf^2(x)$ ικανοποιούνται οι προϋποθέσεις του θεωρήματος Rolle στο διάστημα $[\alpha, \beta]$, αφού γεωμετρικά το συμπέρασμα του θεωρήματος εξασφαλίζει ότι υπάρχει σημείο της γραφικής παράστασης με οριζόντια εφαπτομένη.
Πράγματι, η $g$ είναι παραγωγίσιμη, άρα και συνεχής, ως γινόμενο παραγωγίσιμων συναρτήσεων στο $[\alpha, \beta]$, ενώ από τα δεδομένα προκύπτει ότι
\[ g(\alpha) = \alpha f^2(\alpha) = \beta f^2(\beta) = g(\beta). \]
\item Αφού $f^2(x) > 0$ για κάθε $x \in [\alpha, \beta]$ και η συνάρτηση $f^2(x)$ είναι συνεχής στο $[\alpha, \beta]$, το ζητούμενο εμβαδό θα είναι:
\[ E = \int_{\alpha}^{\beta} f^2(x)dx = \int_{\alpha}^{\beta} x' \cdot f^2(x)dx = [xf^2(x)]_{\alpha}^{\beta} - \int_{\alpha}^{\beta} x(f^2(x))'dx = (\beta f^2(\beta) - \alpha f^2(\alpha)) - 2 \int_{\alpha}^{\beta} xf(x)f'(x)dx = 0 - 2(-\ln 2) = \ln 4 \text{ τετραγωνικές μονάδες}. \]
\item Αφού η συνάρτηση $f'(x)$ είναι συνεχής στο $[\alpha, \beta]$ με $f'(x) \neq 0$ για κάθε $x \in [\alpha, \beta]$, τότε θα διατηρεί σταθερό πρόσημο στο $[\alpha, \beta]$.
Αν ήταν $f'(x) > 0$ για κάθε $x \in [\alpha, \beta]$, τότε θα ήταν και $xf(x)f'(x) > 0$ για κάθε $x \in [\alpha, \beta]$, άρα και $\int_{\alpha}^{\beta} xf(x)f'(x)dx > 0$, δηλαδή $-\ln 2 > 0$, άτοπο.
Ώστε $f'(x) < 0$ για κάθε $x \in [\alpha, \beta]$, άρα η συνάρτηση $f$ είναι γνησίως φθίνουσα στο $[\alpha, \beta]$.
\item Η συνάρτηση $G$ είναι κοίλη στο $[\alpha, \beta]$, αφού έχουμε $G'(x) = f(x)$ και η $f$ είναι γνησίως φθίνουσα στο $[\alpha, \beta]$.
Σύμφωνα με σχόλιο του βιβλίου, η εφαπτομένη ευθεία της γραφικής παράστασης της $G$ σε κάθε σημείο, βρίσκεται «πάνω» από την γραφική παράσταση, με εξαίρεση το σημείο επαφής. Αφού η εφαπτομένη ευθεία της γραφικής παράστασης της $G$ στο σημείο $(\alpha, G(\alpha))$ έχει εξίσωση $y = G(\alpha) + G'(\alpha)(x - \alpha)$, άρα για κάθε $x \in (\alpha, \beta]$ θα ισχύει:
\[ G(x) < G(\alpha) + G'(\alpha)(x - \alpha), \text{ άρα } \frac{G(x)-G(\alpha)}{x-\alpha} < f(\alpha), \text{ καθώς } G'(\alpha) = f(\alpha) \text{ και } x - \alpha > 0. \]
\end{alist}
\textbf{Σημείωση:} Παράδειγμα τέτοιας συνάρτησης είναι η $f(x) = \frac{1}{\sqrt{x}}$ στο διάστημα $[1, 4]$.
