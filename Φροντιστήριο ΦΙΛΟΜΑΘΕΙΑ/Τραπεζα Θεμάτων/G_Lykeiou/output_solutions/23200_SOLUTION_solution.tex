\Askhsh{23200_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\textbf{ΛΥΣΗ}

\begin{alist}
\item Για τη συνάρτηση $f$ ισχύουν $f(0)=3$ και $f(1)=\ln2$, οπότε $f(0)=3>\ln2=f(1)$. Επιπλέον, η $f$ είναι γνησίως μονότονη και δεν είναι γνησίως αύξουσα, οπότε είναι γνησίως φθίνουσα.
\item Λόγω της μονοτονίας της $f$, αρκεί να αποδείξουμε ότι για οποιοδήποτε θετικό αριθμό $\alpha$ ισχύει $\alpha\ln\alpha \ge \ln\alpha$.
Πραγματικά έχουμε:
\begin{itemize}
    \item Αν $\alpha>1$, τότε $\alpha-1>0$ και $\ln\alpha>0$, οπότε
    \[ (\alpha-1)\ln\alpha > 0 \Rightarrow \alpha\ln\alpha - \ln\alpha > 0 \Rightarrow \alpha\ln\alpha > \ln\alpha. \]
    \item Αν $0<\alpha<1$, τότε $\alpha-1<0$ και $\ln\alpha<0$, οπότε
    \[ (\alpha-1)\ln\alpha > 0 \Rightarrow \alpha\ln\alpha - \ln\alpha > 0 \Rightarrow \alpha\ln\alpha > \ln\alpha. \]
    \item Αν $\alpha=1$, τότε το ζητούμενο γράφεται $0 \ge 0$ και προφανώς ισχύει.
\end{itemize}
\item Με $x>0$ έχουμε:
\[ f(e^{x-1}+\ln x)=\ln2 \Leftrightarrow f(e^{x-1}+\ln x)=f(1) \Leftrightarrow e^{x-1}+\ln x=1, (1) \]
αφού η $f$ ως γνησίως μονότονη είναι «1-1».
Η εξίσωση (1) έχει προφανή ρίζα τον αριθμό 1, αφού $e^{1-1}+\ln1=e^0+0=1$. Θα αποδείξουμε ότι είναι μοναδική.
Θεωρούμε τη συνάρτηση $h(x)=e^{x-1}+\ln x$, $x>0$. Η $h$ είναι παραγωγίσιμη στο πεδίο ορισμού της με $h'(x)=e^{x-1}+\frac{1}{x}>0$ για κάθε $x>0$. Άρα είναι γνησίως αύξουσα, οπότε και «1-1», οπότε ο αριθμός 1 είναι η μοναδική ρίζα της εξίσωσης.
\item Αρκεί να αποδείξουμε ότι υπάρχουν δύο διαφορετικοί αριθμοί $x_1, x_2$ ώστε $g(x_1)=g(x_2)$.
Με $x=0$ στον τύπο της συνάρτησης παίρνουμε:
\[ g(0)=f(0)+(3-\ln2)\cdot0-3=3-3=0 \]
ενώ με $x=1$ παίρνουμε:
\[ g(1)=f(1)+(3-\ln2)\cdot1-3=\ln2+3-\ln2-3=0 \]
δηλαδή για τους αριθμούς $x_1=0$ και $x_2=1$ ισχύει $g(x_1)=g(x_2)$.
Άρα η συνάρτηση $g$ δεν αντιστρέφεται.
\end{alist}
