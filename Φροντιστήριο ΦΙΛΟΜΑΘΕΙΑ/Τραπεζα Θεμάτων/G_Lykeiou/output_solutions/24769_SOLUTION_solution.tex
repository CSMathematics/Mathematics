\Askhsh{24769_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Η $f$ είναι παραγωγίσιμη στο πεδίο ορισμού της με
\[ f'(x) = \frac{1}{x+1} - \frac{x}{(x+1)^2} = \frac{x+1-x}{(x+1)^2} = \frac{x}{(x+1)^2} \]
που είναι το ζητούμενο.

Επειδή $(x+1)^2 > 0$ για κάθε $x > -1$, το πρόσημο της παραγώγου της $f'(x)$ είναι ίδιο με το πρόσημο του $x$. Επομένως η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-1, 0]$ και γνησίως αύξουσα στο διάστημα $[0, +\infty)$.
\item Η συνάρτηση $F$ είναι παραγωγίσιμη στο $[0, +\infty)$ με $F'(x)=f(x)$ και από το ερώτημα (α) η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$. Άρα η $F'$ είναι γνησίως αύξουσα στο $[0, +\infty)$, οπότε η $F$ είναι κυρτή στο διάστημα αυτό.
\item \begin{rlist}
\item Ισχύει $F(1)=\ln2$ και $F'(1)=f(1)=\ln2 - \frac{1}{2}$, οπότε η εξίσωση της εφαπτομένης της γραφικής παράστασης της $F$ στο $x_o=1$ είναι η:
\[ y - \ln2 = \left(\ln2 - \frac{1}{2}\right)(x-1) \Leftrightarrow y = \left(\ln2 - \frac{1}{2}\right)x - \ln2 + \frac{1}{2} + \ln2 \]
\[ \Leftrightarrow y = \left(\ln2 - \frac{1}{2}\right)x + \frac{1}{2}. \]
\item Η $F$ είναι κυρτή στο $[0, +\infty)$ οπότε η γραφική της παράσταση είναι από την εφαπτομένη σε οποιοδήποτε σημείο της και πάνω. Έτσι, για την εφαπτομένη στο $x_o=1$ έχουμε:
\[ F(x) \geq \left(\ln2 - \frac{1}{2}\right)x + \frac{1}{2} \Rightarrow 2F(x) \geq (2\ln2-1)x + 1 \Rightarrow 2F(x)-1 \geq (\ln4-1)x \]
\[ \Rightarrow \frac{2F(x)-1}{x} \geq \ln4-1. \]
που είναι το ζητούμενο.
\end{rlist}
\end{alist}
