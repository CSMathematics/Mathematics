\Askhsh{35245_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
Για κάθε $x \in \mathbb{R}$ είναι $f'(x) = \frac{1}{\left(\sqrt{x^2+1}\right)^3} > 0$ άρα είναι γνησίως αύξουσα.

\begin{alist}
    \item [α)] Για κάθε $x \in \mathbb{R}$ έχουμε:
    \[ f''(x) = \left( \frac{1}{\left(x^2+1\right)^{3/2}} \right)' = -\frac{3}{2}\left(x^2+1\right)^{-5/2} \cdot 2x = -\frac{3x}{\left(\sqrt{x^2+1}\right)^5}. \]
    \begin{itemize}
        \item $f''(x) > 0 \iff -3x > 0 \iff x < 0$
        \item $f''(x) < 0 \iff x > 0$
        \item $f''(x) = 0 \iff x = 0$
    \end{itemize}
    Από τα παραπάνω προκύπτει ότι η συνάρτηση $f$ είναι κυρτή στο διάστημα $(-\infty, 0]$, κοίλη στο διάστημα $[0, +\infty)$ και παρουσιάζει καμπή στη θέση $x_0 = 0$.

    \item [β)] Από το προηγούμενο ερώτημα προκύπτει ότι η $f'$ είναι:
    \begin{itemize}
        \item γνησίως αύξουσα στο $(-\infty, 0]$ αφού η $f'$ είναι συνεχής σε αυτό και ισχύει $f''(x) > 0$,
        \item γνησίως φθίνουσα στο $[0, +\infty)$ αφού η $f'$ είναι συνεχής σε αυτό και ισχύει $f''(x) < 0$ και
        \item παρουσιάζει μέγιστη τιμή στη θέση $x_0 = 0$, την $f'(0) = \frac{1}{\left(\sqrt{0^2+1}\right)^3} = 1$.
    \end{itemize}
    Άρα για κάθε $x \in \mathbb{R}$, ισχύει $f'(x) \leq f'(0) \iff f'(x) \leq 1$.

    Εναλλακτικά, για κάθε $x \in \mathbb{R}$ έχουμε:
    \[ f'(x) \leq 1 \iff \frac{1}{\left(\sqrt{x^2+1}\right)^3} \leq 1 \iff 1 \leq \left(\sqrt{x^2+1}\right)^3 \iff 1 \leq x^2+1 \iff x^2 \geq 0, \]
    που ισχύει.

    \item [γ)]
    \begin{rlist}
        \item Για κάθε $\alpha \in \mathbb{R}$ έχουμε:
        \[ \alpha < \alpha+1 \xrightarrow{\text{f γνησίως αύξουσα}} f(\alpha) < f(\alpha+1) \implies f(\alpha+1)-f(\alpha) > 0. \]
        Ακόμη για κάθε $\alpha \in \mathbb{R}$ είναι $f(\alpha+1)-f(\alpha) = \int_{\alpha}^{\alpha+1} f'(x)dx$.

        \item Για κάθε $x \in \mathbb{R}$ ισχύει $f'(x) \leq 1$ με την ισότητα να ισχύει μόνο όταν $x=0$ άρα $\int_{\alpha}^{\alpha+1} f'(x)dx < \int_{\alpha}^{\alpha+1} 1dx \implies f(\alpha+1)-f(\alpha) < \alpha+1-\alpha \implies f(\alpha+1)-f(\alpha) < 1$.
        Τελικά, για κάθε $\alpha \in \mathbb{R}$ ισχύει: $0 < f(\alpha+1)-f(\alpha) < 1$.
    \end{rlist}
\end{alist}
\textbf{β τρόπος (για τη δειάδειξη της ανισότητας)}

Για να αποδείξουμε ότι για κάθε $\alpha \in \mathbb{R}$ ισχύει: $f(\alpha+1)-f(\alpha)<1$ αρκεί να αποδείξουμε ότι $f(\alpha+1)-f(\alpha)<\alpha+1-\alpha$ δηλαδή αρκεί να αποδείξουμε ότι $f(\alpha+1)-(\alpha+1)<f(\alpha)-\alpha$.

Προς τούτο θεωρούμε την συνάρτηση $g(x)=f(x)-x$, $x \in \mathbb{R}$. Είναι $g'(x)=f'(x)-1 \leq 0$, $x \in \mathbb{R}$ με την ισότητα να ισχύει μόνο στο $x_0=0$, άρα η συνάρτηση $g$ είναι γνησίως φθίνουσα. Οπότε για κάθε $\alpha \in \mathbb{R}$ με $\alpha < \alpha+1$ ισχύει $g(\alpha)>g(\alpha+1) \implies f(\alpha)-\alpha > f(\alpha+1)-(\alpha+1)$ και έπεται το ζητούμενο.
