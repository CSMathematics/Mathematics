\Askhsh{34025_SOLUTION}{1}
\begin{ylh}{1}
    \item [Συνέχεια, Παραγώγιση, Μονοτονία, Όρια, Αντίστροφη Συνάρτηση]
\end{ylh}
\begin{alist}
    \item[α)]
    \begin{rlist}
        \item Για $x \in (1,+\infty)$ η συνάρτηση $f$ είναι συνεχής ως πηλίκο συνεχών συναρτήσεων και παραγωγίσιμη, με $f'(x)=\left(\frac{1}{\ln x}\right)' = -1 \frac{(\ln x)'}{\ln^2 x} = -\frac{1}{x \ln^2 x} < 0$.
        \item Είναι $f((1,+\infty)) \stackrel{f}{=} \left(\lim_{x \to +\infty} f(x), \lim_{x \to 1^+} f(x)\right)$. Όμως
        $\lim_{x \to 1^+} f(x) = \lim_{x \to 1^+} \frac{1}{\ln x} \stackrel{\ln x = u}{=} \lim_{u \to 0^+} \frac{1}{u} = +\infty$ και $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{1}{\ln x} \stackrel{\ln x = u}{=} \lim_{u \to +\infty} \frac{1}{u} = 0$, άρα
        $f((1,+\infty)) = (0,+\infty)$.
    \end{rlist}
    \item[β)]
    \begin{rlist}
        \item Από α) ερώτημα έχουμε για $x \in (1,+\infty)$: $f'(x)<0 \Rightarrow f \downarrow$, δηλαδή η συνάρτηση είναι γνησίως φθίνουσα, άρα αντιστρέφεται.
        \item Είναι $f^{-1} : (0,+\infty) \to (1,+\infty)$.
        Για να βρούμε την αντίστροφη της $f$ θέτουμε $y = f(x)$ και λύνουμε ως προς $x$, με $x>1$. Έχουμε:
        \[ y = f(x) \Leftrightarrow y = \frac{1}{\ln x} \Rightarrow \ln x = \frac{1}{y} \Rightarrow x = e^{\frac{1}{y}} \]
        Επομένως $f^{-1}(y) = e^{\frac{1}{y}}$, οπότε η αντίστροφη της $f$ είναι η συνάρτηση
        $f^{-1}(x) = e^{\frac{1}{x}}$, $x>1$.
    \end{rlist}
\end{alist}
