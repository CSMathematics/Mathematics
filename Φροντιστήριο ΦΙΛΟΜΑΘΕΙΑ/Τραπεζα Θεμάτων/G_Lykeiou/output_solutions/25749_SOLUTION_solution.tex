    \item [Ύλη από εικόνα]
\end{ylh}
\kefalaio{ΛΥΣΗ}
\begin{alist}
    \item [α)] Σημείο ασυνέχειας είναι μόνο το 1 διότι $1 \in D_f$ και δεν υπάρχει το όριο στο 1 αφού $\lim_{x \to 1^-} f(x)=1$ ενώ $\lim_{x \to 1^+} f(x)=2$.
    \item [β)] Η $f$ είναι συνεχής στο $[0,1]$ αφού είναι συνεχής στο $(0,1)$ και επιπλέον $\lim_{x \to 0^+} f(x) = 0 = f(0)$ και $\lim_{x \to 1^-} f(x) = 1 = f(1)$.

    Εναλλακτικά, η $f$ είναι συνεχής στο διάστημα $[0,1]$ του πεδίου ορισμού της, αφού το τμήμα της γραφικής της παράστασης που αντιστοιχεί στο διάστημα αυτό, είναι μια καμπύλη που δεν διακόπτεται για καμιά τιμή του $x$ που ανήκει στο $[0,1]$.
    \item [γ)]
    \begin{rlist}
        \item [i.] $\lim_{x \to 4} f(x) = 0$
        \item [iv.] $\lim_{x \to 4} \frac{1}{f(x)} = -\infty$ αφού $\lim_{x \to 4} f(x) = 0$ και $f(x) < 0$ για $x$ κοντά στο 4.
    \end{rlist}
\end{alist}
\OnPar{Σημείωση:} Το 2 και το 3 δεν είναι σημεία ασυνέχειας αφού δεν ανήκουν στο πεδίο ορισμού της $f$.
