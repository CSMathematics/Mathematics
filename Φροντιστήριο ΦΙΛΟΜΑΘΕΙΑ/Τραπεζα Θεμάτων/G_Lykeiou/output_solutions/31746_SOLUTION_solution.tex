\begin{ylh}{1}
    \item [Λύση από εικόνα]
\end{ylh}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με
\[
f'(x)=[(x^2-4x+6)e^x]' = (2x-4)e^x+(x^2-4x+6)e^x = (x^2-2x+2)e^x > 0,
\]
εφόσον $x^2-2x+2 > 0$, γιατί η διακρίνουσα του τριωνύμου είναι $\Delta = -4 < 0$.
Επομένως η συνάρτηση $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$.
Λόγω της μονοτονίας της $f$ και της συνέχειάς της, το
\[
f(A) = \left(\lim_{x \to -\infty} f(x), \lim_{x \to +\infty} f(x)\right) = (0, +\infty),
\]
γιατί:
\[
\lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} (x^2-4x+6)e^x = \lim_{x \to -\infty} \frac{x^2-4x+6}{e^{-x}} \overset{\frac{+\infty}{+\infty}}{=} \lim_{x \to -\infty} \frac{(x^2-4x+6)'}{(e^{-x})'}
\]
\[
= \lim_{x \to -\infty} \frac{2x-4}{-e^{-x}} \overset{\frac{-\infty}{-\infty}}{=} \lim_{x \to -\infty} \frac{(2x-4)'}{(-e^{-x})'} = \lim_{x \to -\infty} \frac{2}{e^{-x}} = \lim_{x \to -\infty} 2e^x = 2 \cdot 0 = 0
\]
και $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} (x^2-4x+6)e^x = (+\infty) \cdot (+\infty) = +\infty$.

\item Η εφαπτομένη της $C_f$ στο σημείο $M(0, f(0))$ είναι:
\[
(\varepsilon): y - f(0) = f'(0)(x-0) \text{ ή } (\varepsilon): y-6=2x \text{ ή } (\varepsilon): y=2x+6.
\]

\item Για κάθε $x \in \mathbb{R}$ η συνάρτηση $f$ είναι δύο φορές παραγωγίσιμη με
\[
f''(x)=[(x^2-2x+2)e^x]' = (2x-2)e^x+(x^2-2x+2)e^x = x^2e^x \ge 0,
\]
με το ίσον να ισχύει μόνο για $x=0$. Επομένως η συνάρτηση $f$ είναι κυρτή στο $\mathbb{R}$ και δεν εμφανίζει σημεία καμπής.

\item Λόγω του ερωτήματος (γ) η συνάρτηση $f$ είναι κυρτή στο $\mathbb{R}$, οπότε η εφαπτομένη της $(\varepsilon)$ θα βρίσκεται κάτω από τη γραφική παράσταση της $f$ με εξαίρεση το σημείο επαφής τους $M$. Επομένως θα ισχύει $f(x) \ge 2x+6$ για κάθε $x \in \mathbb{R}$.
\end{alist}
