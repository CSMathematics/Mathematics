\OnKef{Λύση}
\begin{alist}
\item Αφού η γραφική παράσταση της $f$ έχει την ευθεία $(\varepsilon): y=3x-2$ πλάγια ασύμπτωτη στο $+\infty$, το $\lim_{x \to +\infty} \frac{f(x)}{x}$ θα είναι ο συντελεστής διεύθυνσης της ευθείας $(\varepsilon)$ δηλαδή $\lim_{x \to +\infty} \frac{f(x)}{x}=3$ και το $\lim_{x \to +\infty} (f(x)-3x)$ θα είναι ο σταθερός όρος της ευθείας $(\varepsilon)$ δηλαδή $\lim_{x \to +\infty} (f(x)-3x)=-2$.

\item $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} (f(x)-3x+3x) = +\infty$ αφού $\lim_{x \to +\infty} (f(x)-3x)=-2$ και $\lim_{x \to +\infty} 3x = +\infty$.

\item $\lim_{x \to +\infty} \frac{f(x)-x}{xf(x)-3x^2} = \lim_{x \to +\infty} \frac{\frac{f(x)}{x}-\frac{x}{x}}{\frac{xf(x)}{x}-\frac{3x^2}{x}} = \lim_{x \to +\infty} \frac{\frac{f(x)}{x}-1}{f(x)-3x} = \frac{3-1}{-2} = -1$.
\end{alist}
