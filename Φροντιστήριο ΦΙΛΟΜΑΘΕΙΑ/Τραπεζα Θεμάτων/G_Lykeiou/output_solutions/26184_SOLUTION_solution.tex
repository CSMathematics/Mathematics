\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
    \item Καθώς η συνάρτηση $f$ είναι συνεχής στο $(0, +\infty)$, ως πηλίκο συνεχών συναρτήσεων, κατακόρυφη ασύμπτωτη μπορεί να είναι μόνο η ευθεία με εξίσωση $x = 0$.
    Πράγματι:
    \[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \frac{\ln x}{\sqrt{x}} = \lim_{x \to 0^+} \ln x \cdot \lim_{x \to 0^+} \frac{1}{\sqrt{x}} = -\infty \cdot (+\infty) = -\infty, \]
    διότι $\lim_{x \to 0^+} \ln x = -\infty$ και $\lim_{x \to 0^+} \frac{1}{\sqrt{x}} = +\infty$.
    Ελέγχουμε τώρα αν $\lim_{x \to +\infty} f(x) = l \in \mathbb{R}$.
    Παρατηρούμε ότι για το $\lim_{x \to +\infty} f(x)$, δηλαδή για το $\lim_{x \to +\infty} \frac{\ln x}{\sqrt{x}}$, έχουμε απροσδιοριστία $\frac{+\infty}{+\infty}$.
    Ελέγχουμε αν υπάρχει το $\lim_{x \to +\infty} \frac{(\ln x)'}{(\sqrt{x})'}$.
    Πράγματι: $\lim_{x \to +\infty} \frac{(\ln x)'}{(\sqrt{x})'} = \lim_{x \to +\infty} \frac{\frac{1}{x}}{\frac{1}{2\sqrt{x}}} = \lim_{x \to +\infty} \frac{2\sqrt{x}}{x} = \lim_{x \to +\infty} \frac{2}{\sqrt{x}} = 0$.
    Έτσι, με εφαρμογή του κανόνα L' Hospital, παίρνουμε
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{\ln x}{\sqrt{x}} = \lim_{x \to +\infty} \frac{(\ln x)'}{(\sqrt{x})'} = 0. \]
    Ώστε η ευθεία $y = 0$ είναι οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$.

    \item Για κάθε $x \in (0, +\infty)$ έχουμε ότι:
    \[ f'(x) = \frac{\sqrt{x}(\ln x)' - (\sqrt{x})' \ln x}{(\sqrt{x})^2} = \frac{\sqrt{x} \frac{1}{x} - \frac{1}{2\sqrt{x}} \ln x}{x} = \frac{\frac{1}{\sqrt{x}} - \frac{\ln x}{2\sqrt{x}}}{x} = \frac{\frac{2 - \ln x}{2\sqrt{x}}}{x} = \frac{2 - \ln x}{2x\sqrt{x}}. \]
    Έτσι, έχουμε: $f'(x) \ge 0 \iff 2 - \ln x \ge 0 \iff \ln x \le 2 \iff \ln x \le \ln e^2 \iff x \le e^2$.
    Έτσι, παίρνουμε τον παρακάτω πίνακα:
    \begin{center}
    \begin{tabular}{|c||c|c|c|}
    \hline
    $x$ & $0$ & $e^2$ & $+\infty$ \\
    \hline
    $f'(x)$ & & $+$ & $-$ \\
    \hline
    $f(x)$ & & $\nearrow \quad f(e^2) \quad \searrow$ & \\
    \hline
    \end{tabular}
    \end{center}
    Ώστε η γραφική παράσταση της $f$ έχει ολικό μέγιστο για $x = e^2$.

    \item Παρατηρούμε ότι:
    \[ I = \int_1^{e^2} \frac{1}{2\sqrt{x}} \ln x \, dx = \int_1^{e^2} (2\sqrt{x})' \ln x \, dx \]
    \[ = \left[ 2\sqrt{x} \ln x \right]_1^{e^2} - \int_1^{e^2} 2\sqrt{x} (\ln x)' \, dx \]
    \[ = (2\sqrt{e^2} \ln e^2 - 2\sqrt{1} \ln 1) - \int_1^{e^2} 2\sqrt{x} \frac{1}{x} \, dx \]
    \[ = (2e \cdot 2 - 2 \cdot 1 \cdot 0) - \int_1^{e^2} \frac{2}{\sqrt{x}} \, dx \]
    \[ = 4e - \int_1^{e^2} 2x^{-1/2} \, dx \]
    \[ = 4e - \left[ 2 \frac{x^{1/2}}{1/2} \right]_1^{e^2} \]
    \[ = 4e - \left[ 4\sqrt{x} \right]_1^{e^2} \]
    \[ = 4e - (4\sqrt{e^2} - 4\sqrt{1}) \]
    \[ = 4e - (4e - 4) = 4. \]
\end{alist}
