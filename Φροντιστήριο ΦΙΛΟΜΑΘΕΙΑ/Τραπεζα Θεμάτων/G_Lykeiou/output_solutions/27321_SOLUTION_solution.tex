\begin{alist}
    \item [α)] Για κάθε $x > 0$, έχουμε
    \[ f'(x) = a \left[(x)' \cdot e^{-\beta x} + x(e^{-\beta x})'\right] = a\left[e^{-\beta x} + x(-\beta x)' \cdot e^{-\beta x}\right] = a \cdot e^{-\beta x} \cdot (1 - \beta x). \]
    Έχουμε: $f'(x) \geq 0 \Leftrightarrow 1 - \beta x \geq 0 \Leftrightarrow x \leq \frac{1}{\beta}$, οπότε παίρνουμε τον παρακάτω πίνακα.
    \[
    \begin{tabular}{|c|ccc|}
        \hline
        $x$ & $0$ & $\frac{1}{\beta}$ & $+\infty$ \\
        \hline
        $f'(x)$ & & $+$ & $-$ \\
        \hline
        $f(x)$ & & \text{$\nearrow$} \quad \text{O.M.} \quad \text{$\searrow$} & \\
        \hline
    \end{tabular}
    \]
    Επομένως όταν ο τρέχων πληθυσμός είναι $x = \frac{1}{\beta}$ τότε ο πληθυσμός την αμέσως επόμενη χρονιά θα πάρει την μεγαλύτερη δυνατή τιμή, ίση με $f\left(\frac{1}{\beta}\right) = \frac{\alpha}{\beta e}$.

    \item [β)] Καθώς θέλουμε να προσεγγίσουμε τον πληθυσμό $y$ των Ψαριών, όταν την αμέσως προηγούμενη χρονιά, ο πληθυσμός είναι απεριόριστα μεγάλος, ζητάμε το
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \alpha \frac{x}{e^{\beta x}} \]
    και παρατηρούμε ότι έχουμε απροσδιοριστία $\frac{+\infty}{+\infty}$. Ελέγχουμε αν υπάρχει το
    \[ \lim_{x \to +\infty} \frac{(x)'}{(e^{\beta x})'} = \lim_{x \to +\infty} \frac{1}{\beta e^{\beta x}} \]
    το οποίο ισούται με μηδέν.
    Άρα $\lim_{x \to +\infty} f(x) = 0$ από τον κανόνα De L' Hospital, κάτι που σημαίνει ότι αν ο πληθυσμός κάποια χρονιά είναι απεριόριστα μεγάλος, την αμέσως επόμενη χρονιά ο πληθυσμός των ψαριών πρακτικά θα εξαφανιστεί.

    \item [γ)] Σύμφωνα με το θεμελιώδες θεώρημα του ολοκληρωτικού λογισμού, θα έχουμε:
    \begin{align*}
    F(\beta) - F(2\beta) &= \int_{2\beta}^{\beta} f(x)dx = a \int_{2\beta}^{\beta} xe^{-\beta x} dx = a \int_{2\beta}^{\beta} x \cdot \left(-\frac{1}{\beta} e^{-\beta x}\right)' dx \\
    &= a \left\{\left[-\frac{x}{\beta} e^{-\beta x}\right]_{2\beta}^{\beta} - \int_{2\beta}^{\beta}(x)' \cdot \left(-\frac{1}{\beta} e^{-\beta x}\right) dx\right\} \\
    &= a\left\{-\frac{\beta}{\beta} e^{-\beta^2} - \left(-\frac{2\beta}{\beta} e^{-2\beta^2}\right) + \frac{1}{\beta} \int_{2\beta}^{\beta} e^{-\beta x} dx\right\} \\
    &= a\left\{-e^{-\beta^2} + 2e^{-2\beta^2} + \frac{1}{\beta} \left[-\frac{1}{\beta} e^{-\beta x}\right]_{2\beta}^{\beta}\right\} \\
    &= a \left\{\frac{2-e^{\beta^2}}{e^{2\beta^2}} - \frac{1}{\beta^2} \left(e^{-\beta^2} - e^{-2\beta^2}\right)\right\} \\
    &= a \left(\frac{2-e^{\beta^2}}{e^{2\beta^2}} - \frac{e^{\beta^2}-1}{\beta^2 e^{2\beta^2}}\right) \\
    &= \frac{a}{\beta^2 e^{2\beta^2}} \left[\beta^2(2-e^{\beta^2}) - (e^{\beta^2}-1)\right] \\
    &= \frac{a}{\beta^2 e^{2\beta^2}} \left[2\beta^2 - \beta^2 e^{\beta^2} - e^{\beta^2} + 1\right] \\
    &= \frac{a}{\beta^2} \frac{2\beta^2+1-(1+\beta^2)e^{\beta^2}}{e^{2\beta^2}}.
    \end{align*}
\end{alist}
