\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item
    \begin{rlist}
        \item Είναι $D_f = \mathbb{R} - \{1\}$ και $D_g = \mathbb{R}$ οπότε:
        \[ A_1 = \{x \in D_g : g(x) \in D_f\} = \left\{x \in \mathbb{R} : \frac{1}{e^x} \neq 1\right\} = \{x \in \mathbb{R} : e^x \neq 1\} = \{x \in \mathbb{R} : x \neq 0\} = \mathbb{R}^*, \]
        είναι το πεδίο ορισμού της συνάρτησης $h(x) = (f \circ g)(x)$.
        \item Λόγω του ερωτήματος (α, i) το $A_1 \neq \emptyset$, οπότε ορίζεται η σύνθεση $f \circ g$ με
        τύπο $h(x) = (f \circ g)(x) = f(g(x)) = \frac{1}{g(x)-1} = \frac{1}{\frac{1}{e^x}-1} = \frac{e^x}{1-e^x}, \quad x \neq 0$.
    \end{rlist}
\item Για $x_1, x_2 \in \mathbb{R}^*$ με $h(x_1) = h(x_2) \Rightarrow \frac{e^{x_1}}{1-e^{x_1}} = \frac{e^{x_2}}{1-e^{x_2}} \Rightarrow$
    \[ e^{x_1} \cdot (1-e^{x_2}) = e^{x_2} \cdot (1-e^{x_1}) \Rightarrow e^{x_1} - e^{x_1} \cdot e^{x_2} = e^{x_2} - e^{x_1} \cdot e^{x_2} \Rightarrow e^{x_1} = e^{x_2} \Rightarrow x_1 = x_2, \]
    επομένως η συνάρτηση $h$ είναι '1-1'.
\item Είναι $\lim_{x \to +\infty} h(x) = \lim_{x \to +\infty} \frac{e^x}{1-e^x} \stackrel{\frac{+\infty}{-\infty}}{dL'H} \lim_{x \to +\infty} \frac{(e^x)'}{(1-e^x)'} = \lim_{x \to +\infty} \frac{e^x}{-e^x} = -1$.

    Εναλλακτικά: $\lim_{x \to +\infty} h(x) = \lim_{x \to +\infty} \frac{e^x}{1-e^x} = \lim_{x \to +\infty} \frac{e^x}{e^x\left(\frac{1}{e^x}-1\right)} = \lim_{x \to +\infty} \frac{1}{\frac{1}{e^x}-1} = \frac{1}{0-1} = -1$.
\end{alist}
