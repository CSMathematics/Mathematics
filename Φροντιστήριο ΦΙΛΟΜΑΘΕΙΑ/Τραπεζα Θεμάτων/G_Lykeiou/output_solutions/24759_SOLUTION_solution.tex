    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ
\begin{alist}
    \item [$\alpha$)]
        \begin{rlist}
            \item Για $x>0$ είναι $\frac{f(x)}{x} \geq x-1+\frac{1}{x}$ και αφού $\lim_{x\to +\infty} \left(x-1+\frac{1}{x}\right) = +\infty$ είναι και $\lim_{x\to +\infty} \frac{f(x)}{x} = +\infty$.
            \item Αφού $\lim_{x\to -\infty} \frac{f(x)}{x} = +\infty$ συμπεραίνουμε ότι η συνάρτηση $f$ δεν έχει (πλάγιες ή οριζόντιες) ασύμπτωτες στο $+\infty$.

                Επίσης για $x<0$ είναι $\frac{f(x)}{x} \leq x-1+\frac{1}{x}$ και αφού $\lim_{x\to -\infty} \left(x-1+\frac{1}{x}\right) = -\infty$ είναι και $\lim_{x\to -\infty} \frac{f(x)}{x} = -\infty$ οπότε η συνάρτηση $f$ δεν έχει (πλάγιες ή οριζόντιες) ασύμπτωτες στο $-\infty$.

                Τέλος η συνάρτηση $f$ είναι παραγωγίσιμη και άρα συνεχής στο $\mathbb{R}$ οπότε δεν έχει κατακόρυφες ασύμπτωτες. Τελικά η συνάρτηση $f$ δεν έχει ασύμπτωτες.
            \item Είναι $x^2-x+1 \geq \frac{3}{4} \Leftrightarrow x^2-x+\frac{1}{4} \geq 0 \Leftrightarrow \left(x-\frac{1}{2}\right)^2 \geq 0$ που ισχύει για κάθε $x \in \mathbb{R}$ οπότε αφού $f(x) \geq x^2-x+1$ για κάθε $x \in \mathbb{R}$ είναι και $f(x) \geq \frac{3}{4}$ για κάθε $x \in \mathbb{R}$.
        \end{rlist}
    \item [$\beta$)]
        \begin{rlist}
            \item Είναι $f'(x) \geq \frac{3}{4} \Leftrightarrow f(x) \geq f\left(\frac{1}{2}\right)$ για κάθε $x \in \mathbb{R}$ που σημαίνει ότι η συνάρτηση $f$ παρουσιάζει στο $\frac{1}{2}$ ελάχιστο το $f\left(\frac{1}{2}\right) = \frac{3}{4}$. Συνεπώς από το θεώρημα Fermat έχουμε ότι $f'\left(\frac{1}{2}\right) = 0$.
            \item Από Θεώρημα Μέσης Τιμής για την $f'$ στο $\left[\frac{1}{2},1\right]$ υπάρχει $\xi \in \left(\frac{1}{2},1\right)$ τέτοιο ώστε
            \[ f''(\xi) = \frac{f'(1)-f'\left(\frac{1}{2}\right)}{1-\frac{1}{2}} = \frac{1-0}{\frac{1}{2}} = \frac{1}{\frac{1}{2}} = 2 \]
            Αφού $\frac{1}{2} < \xi$ και $f'\left(\frac{1}{2}\right) < f'(\xi)$ συμπεραίνουμε ότι η $f'$ δεν είναι γνησίως φθίνουσα και κατ' επέκταση η $f$ δεν είναι κοίλη.

            Εναλλακτικά,

            θεωρούμε τη συνάρτηση $g(x) = f(x)-x^2+x-1$ η οποία είναι παραγωγίσιμη στο $\mathbb{R}$ ως πράξεις παραγωγίσιμων. Παρατηρούμε ότι $g(1) = f(1)-1^2+1-1 = 0$ οπότε από τη σχέση $f(x) \geq x^2-x+1 \Leftrightarrow g(x) \geq 0$ για κάθε $x \in \mathbb{R}$, συμπεραίνουμε ότι η συνάρτηση $g$ παρουσιάζει στο $1$ ελάχιστο το $0$. Συνεπώς από το θεώρημα Fermat έχουμε ότι $g'(1)=0$. Όμως $g'(x) = f'(x)-2x+1$ οπότε $g'(1)=0 \Leftrightarrow f'(1)-2\cdot 1+1=0 \Leftrightarrow f'(1)=1$.

            Αφού $\frac{1}{2} < 1$ και $f'\left(\frac{1}{2}\right) < f'(1)$ συμπεραίνουμε ότι η $f'$ δεν είναι γνησίως φθίνουσα και κατ' επέκταση η $f$ δεν είναι κοίλη.
        \end{rlist}
\end{alist}
