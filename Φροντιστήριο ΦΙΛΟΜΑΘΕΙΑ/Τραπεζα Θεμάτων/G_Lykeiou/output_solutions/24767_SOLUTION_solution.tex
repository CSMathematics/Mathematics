\Askhsh{24767_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ

\begin{alist}
\item [\textalpha)] Για κάθε $x_1, x_2 \in \mathbb{R}$ με $x_1 < x_2$ έχουμε:
\[
x_1 < x_2 \Rightarrow e^{x_1} < e^{x_2} \Rightarrow 0 < e^{x_1} + 1 < e^{x_2} + 1 \Rightarrow \frac{1}{e^{x_1} + 1} > \frac{1}{e^{x_2} + 1} \Rightarrow f(x_1) > f(x_2)
\]
οπότε η συνάρτηση $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$.

Επιπλέον η $f$ είναι συνεχής και
\begin{itemize}
    \item $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{1}{e^x + 1} = 0$, αφού $\lim_{x \to +\infty} (e^x + 1) = +\infty$
    \item $\lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} \frac{1}{e^x + 1} = 1$, αφού $\lim_{x \to -\infty} (e^x + 1) = 0+1=1$
\end{itemize}
οπότε $f(\mathbb{R}) = \left(\lim_{x \to +\infty} f(x), \lim_{x \to -\infty} f(x)\right) = (0, 1)$.

\item [\textbeta)] Η συνάρτηση είναι γνησίως μονότονη, οπότε είναι «1-1», άρα αντιστρέφεται.
Αν $f(x)=y$, $y \in (0, 1)$, τότε για οποιοδήποτε $x \in \mathbb{R}$ έχουμε:
\[
\frac{1}{e^x + 1} = y \Leftrightarrow e^x + 1 = \frac{1}{y} \Leftrightarrow e^x = \frac{1}{y} - 1 \Leftrightarrow e^x = \frac{1-y}{y} \Leftrightarrow x = \ln \frac{1-y}{y}
\]
Άρα, $f^{-1}(x) = \ln \frac{1-x}{x}$, $x \in (0, 1)$.
\end{alist}
