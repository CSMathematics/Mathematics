    \item [Ύλη από εικόνα]
\end{ylh}
\large\textbf{ΛΥΣΗ}

\begin{alist}
\item Η συνάρτηση ορίζεται μόνο όταν $1-\mathrm{e}^{-x} > 0$. Επειδή,
\[ 1-\mathrm{e}^{-x} > 0 \Leftrightarrow \mathrm{e}^{-x} < 1 \Leftrightarrow -x < 0 \Leftrightarrow x > 0 \]
έχουμε $\mathbb{D}_\mathrm{f} = (0, +\infty)$.

Για κάθε $x_1, x_2 \in (0, +\infty)$ με $x_1 < x_2$ έχουμε:
\[ -x_1 > -x_2 \Rightarrow \mathrm{e}^{-x_1} > \mathrm{e}^{-x_2} \Rightarrow -\mathrm{e}^{-x_1} < -\mathrm{e}^{-x_2} \]
\[ \Rightarrow 1-\mathrm{e}^{-x_1} < 1-\mathrm{e}^{-x_2} \Rightarrow \ln(1-\mathrm{e}^{-x_1}) < \ln(1-\mathrm{e}^{-x_2}) \Rightarrow f(x_1) < f(x_2) \]
Άρα η $f$ είναι γνησίως αύξουσα στο $(0, +\infty)$, οπότε αντιστρέφεται.

\item Έστω $f(x)=y, x>0$. Τότε έχουμε:
\[ \ln(1-\mathrm{e}^{-x}) = y \Leftrightarrow 1-\mathrm{e}^{-x} = \mathrm{e}^{y} \Leftrightarrow \mathrm{e}^{-x} = 1-\mathrm{e}^{y} \Leftrightarrow \]
\[ \begin{cases}
-x = \ln(1-\mathrm{e}^{y}) \\
1-\mathrm{e}^{y} > 0
\end{cases}
\Leftrightarrow
\begin{cases}
x = -\ln(1-\mathrm{e}^{y}) \\
y < 0
\end{cases} \]
όπου προφανώς $x>0$, αφού $\ln(1-\mathrm{e}^{y}) < 0$. Άρα $\mathrm{f}^{-1}(x)=-\ln(1-\mathrm{e}^{x})$, $x<0$.
\end{alist}
