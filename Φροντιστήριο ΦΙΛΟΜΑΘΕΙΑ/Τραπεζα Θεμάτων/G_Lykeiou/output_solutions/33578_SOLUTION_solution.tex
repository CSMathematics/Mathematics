    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ
\begin{alist}
    \item Αφού $x \in [0,\pi]$ θα έχουμε $\eta\mu x \ge 0$, καθώς η γωνία $x$ βρίσκεται στο 1ο ή 2ο τεταρτημόριο.
    Επίσης $x \ge 0 \Leftrightarrow e^x \ge e^0 = 1$.
    Προσθέτοντας κατά μέλη παίρνουμε $e^x + \eta\mu x \ge 1$.
    \item Παραγωγίζοντας την συνάρτηση $H(x)$ παίρνουμε:
    \[H'(x) = x' - \frac{(e^x+\eta\mu x)'}{e^x+\eta\mu x} = 1 - \frac{e^x+\sigma\upsilon\nu x}{e^x+\eta\mu x} = \frac{e^x+\eta\mu x - e^x-\sigma\upsilon\nu x}{e^x+\eta\mu x} = \frac{\eta\mu x-\sigma\upsilon\nu x}{e^x+\eta\mu x} = f(x).\]
    \item Έχουμε $I = \int_0^\pi xf'(x)dx = [xf(x)]_0^\pi - \int_0^\pi x'f(x)dx = \pi f(\pi) - 0f(0) - \int_0^\pi f(x)dx = \pi f(\pi) - [H(x)]_0^\pi = \pi f(\pi) - (H(\pi) - H(0)).$
    Αλλά $f(\pi) = \frac{\eta\mu\pi-\sigma\upsilon\nu\pi}{e^\pi+\eta\mu\pi} = \frac{0-(-1)}{e^\pi+0} = \frac{1}{e^\pi}$ και $H(\pi) = \pi - \ln(e^\pi + \eta\mu\pi) = \pi - \ln(e^\pi + 0) = \pi - \pi = 0$, ενώ
    $H(0) = 0 - \ln(e^0 + \eta\mu 0) = -\ln 1 = 0$.
    Ώστε $I = \pi f(\pi) = \frac{\pi}{e^\pi}$.
    \item Για κάθε $x \in (0,\pi]$ από α) ερώτημα παίρνουμε $x(e^x + \eta\mu x) \ge x \Leftrightarrow \frac{1}{x(e^x+\eta\mu x)} \le \frac{1}{x}$.
    Καθώς είναι $1 < e < \pi$, θα έχουμε $\int_1^e \frac{1}{x(e^x+\eta\mu x)}dx < \int_1^e \frac{1}{x}dx = [\ln x]_1^e = \ln e - \ln 1 = 1$.
    Το ίσον δεν ισχύει διότι οι συναρτήσεις $g(x) = \frac{1}{x(e^x+\eta\mu x)}$ και $t(x) = \frac{1}{x}$ δεν είναι ίσες στο διάστημα $[1,e]$, αφού $1 < \frac{\pi}{2} < e$ και $g\left(\frac{\pi}{2}\right) = \frac{2}{\pi(1+e^{\pi/2})} \ne t\left(\frac{\pi}{2}\right) = \frac{2}{\pi}$ κι' αυτό διότι $1+e^{\pi/2} > 1$.
\end{alist}
