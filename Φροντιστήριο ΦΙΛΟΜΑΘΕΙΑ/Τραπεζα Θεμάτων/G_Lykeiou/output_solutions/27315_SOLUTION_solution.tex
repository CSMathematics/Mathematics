\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
ΛΥΣΗ

\textbf{\textalpha)} Για $x<2$ έχουμε $f(x) = \frac{x^2-4}{x-2}$, οπότε:
\[ \lim_{x \to 2^-} f(x) = \lim_{x \to 2^-} \frac{x^2-4}{x-2} = \lim_{x \to 2^-} \frac{(x-2)(x+2)}{x-2} = \lim_{x \to 2^-} (x+2) = 4 \]
Για $x>2$ έχουμε $f(x) = \alpha x^2-4$, οπότε:
\[ \lim_{x \to 2^+} f(x) = \lim_{x \to 2^+} (\alpha x^2-4) = 4\alpha-4 \]

\textbf{\textbeta)} Η συνάρτηση $f$ είναι συνεχής στο $x_0=2$ αν και μόνο αν $\lim_{x \to 2^-} f(x) = \lim_{x \to 2^+} f(x) = f(2)$ και επειδή $f(2) = 4\alpha-4$ και λόγω του \textalpha) ερωτήματος έχουμε $4\alpha-4=4$ ή $4\alpha=8$. Άρα τελικά $\alpha=2$.

\textbf{\textgamma)} Για $\alpha=2$ η συνάρτηση $f$ γίνεται:
\[ f(x) = \begin{cases} \frac{x^2-4}{x-2}, & \text{αν } x < 2 \\ 2x^2-4, & \text{αν } x \ge 2 \end{cases} = \begin{cases} \frac{(x-2)(x+2)}{x-2}, & \text{αν } x < 2 \\ 2x^2-4, & \text{αν } x \ge 2 \end{cases} = \begin{cases} x+2, & \text{αν } x < 2 \\ 2x^2-4, & \text{αν } x \ge 2 \end{cases} \]
Θα εξετάσουμε αν η $f$ είναι παραγωγίσιμη στο $x_0=2$.
Για $x<2$: $\frac{f(x)-f(2)}{x-2} = \frac{x+2-4}{x-2} = \frac{x-2}{x-2} = 1$. Άρα $\lim_{x \to 2^-} \frac{f(x)-f(2)}{x-2} = 1$ (1)
Για $x>2$: $\frac{f(x)-f(2)}{x-2} = \frac{2x^2-4-4}{x-2} = \frac{2x^2-8}{x-2} = \frac{2(x^2-4)}{x-2} = \frac{2(x-2)(x+2)}{x-2} = 2x+4$.
Άρα $\lim_{x \to 2^+} \frac{f(x)-f(2)}{x-2} = \lim_{x \to 2^+} (2x+4) = 8$ (2)

Από τις σχέσεις (1) και (2) έχουμε ότι $\lim_{x \to 2^-} \frac{f(x)-f(2)}{x-2} \ne \lim_{x \to 2^+} \frac{f(x)-f(2)}{x-2}$, άρα η $f$ δεν είναι παραγωγίσιμη στο $x_0=2$.
Η $f$ είναι παραγωγίσιμη στο $(-\infty,2)$ και στο $(2,+\infty)$ ως πολυωνυμική με
\[ f'(x) = \begin{cases} 1, & \text{αν } x < 2 \\ 4x, & \text{αν } x > 2 \end{cases} \]
