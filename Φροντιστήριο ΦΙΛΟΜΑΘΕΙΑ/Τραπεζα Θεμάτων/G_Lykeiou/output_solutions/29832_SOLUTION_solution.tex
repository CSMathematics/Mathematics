    \item [Ύλη από εικόνα]
\end{ylh}
\textbf{ΛΥΣΗ}
\begin{alist}
    \item [α)] Για τη συνάρτηση $f$ είναι: $e^x - 1 \neq 0 \Leftrightarrow e^x \neq 1 \Leftrightarrow x \neq 0$, οπότε $D_f = \mathbb{R}^*$.

    Για τη συνάρτηση $g$ είναι:
    \[ \frac{1-x}{1+x} > 0 \Leftrightarrow (1-x) \cdot (1+x) > 0 \Leftrightarrow -1 < x < 1, \text{ οπότε } D_g = (-1,1). \]

    \item [β)] Λόγω του ερωτήματος (α) έχουμε $D_f = \mathbb{R}^* = (-\infty,0) \cup (0,+\infty)$ και $D_g = (-1,1)$.

    Για την $f \circ g$ το πεδίο ορισμού είναι το:
    \[ A_1 = \{x \in D_g : g(x) \in D_f \} = \{x \in (-1,1) : g(x) \in \mathbb{R}^* \} = \left\{ x \in (-1,1) : \ln \frac{1-x}{1+x} \neq 0 \right\} = \]
    \[ \left\{ x \in (-1,1) : \frac{1-x}{1+x} \neq 1 \right\} = \{x \in (-1,1) : 1-x \neq 1+x \} = \{x \in (-1,1) : x \neq 0 \} = (-1,0) \cup (0,1). \]

    \item [γ)] Για $x \in (-1,0) \cup (0,1)$ είναι: $(f \circ g)(x)=(f(g(x))=f\left(\ln \frac{1-x}{1+x}\right)=$
    \[ \frac{e^{\ln \frac{1-x}{1+x}}+1}{e^{\ln \frac{1-x}{1+x}}-1} = \frac{\frac{1-x}{1+x}+1}{\frac{1-x}{1+x}-1} = \frac{\frac{1-x+1+x}{1+x}}{\frac{1-x-(1+x)}{1+x}} = \frac{\frac{2}{1+x}}{\frac{-2x}{1+x}} = \frac{2}{-2x} = -\frac{1}{x}. \]
    Άρα $(f \circ g)(x) = -\frac{1}{x}$, $x \in (-1,0) \cup (0,1)$.
\end{alist}
