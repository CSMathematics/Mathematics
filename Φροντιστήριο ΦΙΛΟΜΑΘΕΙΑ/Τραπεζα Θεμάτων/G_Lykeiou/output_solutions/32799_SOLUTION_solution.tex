\begin{alist}
\item Το σύνολο των τεταγμένων των σημείων της $C_f$ είναι το σύνολο τιμών $f'$. Από τη γραφική παράσταση της $f'$ προκύπτει ότι $f'([-1,1])=[1,2]$. Άρα για κάθε $x \in [-1,1]$ ισχύει $1 \le f'(x) \le 2$.
\item Για κάθε $x \in (-1,1)$ είναι $f'(x) \ge 1 \Rightarrow f'(x) > 0$ και επειδή η συνάρτηση $f$ είναι συνεχής στο $[-1,1]$ (ως παραγωγίσιμη) έπεται ότι είναι γνησίως αύξουσα.
\item Η συνεχής συνάρτηση $f$ :
\begin{itemize}
    \item[-] είναι κυρτή στο διάστημα $[-1,0]$ αφού η $f'$ είναι γνησίως αύξουσα στο $(-1,0)$,
    \item[-] είναι κοίλη στο $[0,1]$ αφού η $f'$ είναι γνησίως φθίνουσα στο $(0,1)$ και
    \item[-] έχει σημείο καμπής το $\Gamma$.
\end{itemize}
\end{alist}
