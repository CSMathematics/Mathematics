\begin{alist}
\item Αφού η εφαπτομένη της $C_f$ στο σημείο της $A(0,1)$ σχηματίζει με τον $xx'$ γωνία $45^\circ$, έχουμε ότι $f'(0) = \varepsilon\phi45^\circ = 1$.
\item Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο της $A(0,1)$ είναι η
\[ y - f(0) = f'(0)(x-0) \Leftrightarrow y-1 = 1 \cdot x \Leftrightarrow y=x+1. \]
\item Είναι
\[ \lim_{x \to 0} \frac{f(x)-1}{x} = \lim_{x \to 0} \frac{f(x)-f(0)}{x-0} = f'(0) = 1. \]
\end{alist}
