    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Η $f$ ορίζεται στο $\mathbb{R}$, είναι συνεχής για $x<2$ ως σταθερή και για $x>2$ ως πράξεις συνεχών. Θα εξετάσουμε αν είναι συνεχής στο $x_0=2$.
\[ \lim_{x \to 2^-} f(x) = \lim_{x \to 2^-} (-1) = -1 \]
\[ \lim_{x \to 2^+} f(x) = \lim_{x \to 2^+} (e^{x-2}-2) = e^{2-2}-2 = e^0-2 = 1-2 = -1 \]
και $f(2) = -1$.
Άρα η $f$ είναι συνεχής στο $x_0=2$.

Η $f$ είναι παραγωγίσιμη για $x<2$ με $f'(x)=0$, ως σταθερή και για $x>2$ ως πράξεις παραγωγίσιμων με $f'(x) = e^{x-2}(x-2)' = e^{x-2} > 0$, για κάθε $x>2$.
Άρα η $f'$ είναι σταθερή με $f'(x)=0$ στο $(-\infty,2)$, γνησίως αύξουσα στο $[2,+\infty)$ και συνεχής στο $2$. Οπότε για $x \ge 2 \implies f(x) \ge f(2) \implies f(x) \ge -1$.
Οπότε ισχύει ότι $f(x) \ge -1$ για κάθε $x \in \mathbb{R}$.

Η $g$ είναι συνεχής και παραγωγίσιμη στο $\mathbb{R}-\{2\}$ ως πολυωνυμική με $g'(x) = -x+2$.
\begin{itemize}
    \item $g'(x)=0 \iff -x+2=0 \iff x=2$, απορρίπτεται γιατί $2 \notin A_g$.
    \item $g'(x)>0 \iff -x+2>0 \iff x<2$.
    \item $g'(x)<0 \iff x>2$.
\end{itemize}
Άρα η $g$ είναι γνησίως αύξουσα στο $(-\infty,2)$ και γνησίως φθίνουσα στο $(2,+\infty)$.

Av $\Delta_1 = (-\infty,2)$, τότε $g(\Delta_1) = \PARENS{\lim_{x \to -\infty} g(x), \lim_{x \to 2^-} g(x)} = (-\infty,-1)$.
γιατί η $g$ είναι γνησίως αύξουσα και συνεχής στο $(-\infty,2)$ με
\[ \lim_{x \to -\infty} g(x) = \lim_{x \to -\infty} \left(-\frac{1}{2} x^2+2x-3\right) = \lim_{x \to -\infty} \left(-\frac{1}{2} x^2\right) = -\infty \]
και
\[ \lim_{x \to 2^-} g(x) = \lim_{x \to 2^-} \left(-\frac{1}{2} x^2+2x-3\right) = -\frac{1}{2}(2)^2+2(2)-3 = -2+4-3 = -1. \]

Av $\Delta_2 = (2,+\infty)$, τότε $g(\Delta_2) = \PARENS{\lim_{x \to +\infty} g(x), \lim_{x \to 2^+} g(x)} = (-\infty,-1)$.
γιατί η $g$ είναι γνησίως φθίνουσα και συνεχής στο $(2,+\infty)$ με
\[ \lim_{x \to +\infty} g(x) = \lim_{x \to +\infty} \left(-\frac{1}{2} x^2+2x-3\right) = \lim_{x \to +\infty} \left(-\frac{1}{2} x^2\right) = -\infty \]
και
\[ \lim_{x \to 2^+} g(x) = \lim_{x \to 2^+} \left(-\frac{1}{2} x^2+2x-3\right) = -\frac{1}{2}(2)^2+2(2)-3 = -2+4-3 = -1. \]
Οπότε το σύνολο τιμών της $g$ είναι το $g(\Delta_1) \cup g(\Delta_2) = (-\infty,-1)$, δηλαδή οι τιμές της $g$ είναι μικρότερες του $-1$ για κάθε $x \neq 2$.

\item Η γραφική παράσταση της $f$ βρίσκεται πάνω από τη γραφική παράσταση της $g$ στο κοινό πεδίο ορισμού τους το $\mathbb{R}-\{2\}$, αφού $f(x) \ge -1$ για κάθε $x \in \mathbb{R}$, από το α)i. ερώτημα και $g(x) < -1$ για κάθε $x \neq 2$ από το α)ii. ερώτημα.

\item Παρατηρούμε ότι για τις συναρτήσεις του ερωτήματος α) και για $x_0=2$, ισχύει ότι
$f(x) > g(x)$ κοντά στο $2$,
ενώ
$\lim_{x \to 2^-} f(x) = -1$ από τη συνέχεια της $f$ που εξετάστηκε στο α) ερώτημα και
$\lim_{x \to 2^-} g(x) = -1$, από το β) ερώτημα.
Ο ισχυρισμός «Αν $\lim_{x \to x_0} f(x) > \lim_{x \to x_0} g(x)$ κοντά στο $x_0$, τότε $\lim_{x \to x_0} f(x) > \lim_{x \to x_0} g(x)$» είναι τελικά ψευδής.
\end{alist}
