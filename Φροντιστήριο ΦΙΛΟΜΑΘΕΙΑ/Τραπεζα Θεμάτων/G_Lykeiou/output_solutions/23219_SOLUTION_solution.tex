    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
    \item Η εφαπτομένη της $C_f$ στο σημείο της $(1,f(1))$ έχει εξίσωση $y - f(1) = f'(1)(x-1) \Leftrightarrow y-2=2x-2 \Leftrightarrow y=2x$.
    Η $f$ είναι κυρτή και επομένως η $C_f$ είναι πάνω από κάθε εφαπτομένη της με εξαίρεση το σημείο επαφής. Συνεπώς, $f(x) \ge 2x$ για κάθε $x \in \mathbb{R}$ και η ισότητα ισχύει μόνο για $x=1$.
    \item Είναι $\lim_{x \to +\infty} 2x = +\infty$ οπότε αφού $f(x) \ge 2x$ για κάθε $x \in \mathbb{R}$ είναι και $\lim_{x \to +\infty} f(x) = +\infty$.
    \item Αφού $f(x) \ge 2x$ για κάθε $x \in \mathbb{R}$ και η ισότητα ισχύει μόνο για $x=1$ έχουμε:
    \begin{rlist}
        \item $\int_{0}^{1} f(x)dx > \int_{0}^{1} 2xdx = [x^2]_{0}^{1} = 1-0=1$ οπότε $\int_{0}^{1} f(x)dx > 1$.
        \item $\int_{0}^{1} xf'(x)dx = [xf(x)]_{0}^{1} - \int_{0}^{1} f(x)dx = f(1)-0 - \int_{0}^{1} f(x)dx = 2 - \int_{0}^{1} f(x)dx < 1$ αφού

        όπως δείξαμε παραπάνω $\int_{0}^{1} f(x)dx > 1$ οπότε $-\int_{0}^{1} f(x)dx < -1$ και τελικά $2 - \int_{0}^{1} f(x)dx < 1$.

        Συνεπώς $\int_{0}^{1} xf'(x)dx < 1$.
    \end{rlist}
\end{alist}
