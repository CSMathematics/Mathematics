    \item [Διαφορικός Λογισμός, Ολοκληρωτικός Λογισμός, Γραφική Παράσταση Συνάρτησης, Εμβαδόν]

\begin{alist}
    \item [Α] Αρχικά, θα χρειαστούμε την παράγωγο συνάρτησης:
    \[ f'(x) = \frac{(1)'(1+x^2)^2 - (1+x^2)(2(1+x^2)2x)}{(1+x^2)^4} = \frac{-2x}{(1+x^2)^2} \]
    οπότε $f'(1) = \frac{-2}{(1+1)^2} = -\frac{1}{2}$. Η τεταγμένη του σημείου $\Delta$ είναι $f(1) = \frac{1}{1+1^2} = \frac{1}{2}$, οπότε η εφαπτομένη της γραφικής παράστασης της συνάρτησης $f(x)$ στο $\Delta$ έχει εξίσωση $y - f(1) = f'(1)(x - 1)$, ήτοι $y - \frac{1}{2} = -\frac{1}{2}(x - 1)$, άρα $y = -\frac{1}{2}x + 1$, εξίσωση η οποία επαληθεύεται από το ζεύγος $(x, y) = (0,1)$ δηλαδή από τις συντεταγμένες του σημείου $\Gamma(0,1)$, αφού η τεταγμένη του σημείου $\Gamma$ είναι $f'(0) = 1$. \monades{X}
    \item [Β] Αρκεί να αποδείξουμε ότι για κάθε $x \in [0,1]$ ισχύει $\frac{1}{1+x^2} \ge -\frac{x}{2} + 1$, σχέση που ισοδύναμα γράφεται:
    $2 - x(1+x^2)^2 + 2(1+x^2) \ge 0 \Leftrightarrow 2 + (x-2)(1+x^2)^2 \ge 0 \Leftrightarrow x(1+x^2)^2 - 2(1+x^2) - 2 \le 0$,
    που ισχύει. Η ισότητα ισχύει μόνο για $x=0$ και $x=2$. \monades{X}
    \item [Γ] Παρατηρούμε ότι η συνάρτηση $f(x)$ είναι άρτια, αφού για κάθε $x \in \mathbb{R}$ είναι
    $ -x \in \mathbb{R}$, ενώ $f(-x) = \frac{1}{1+(-x)^2} = \frac{1}{1+x^2} = f(x)$, άρα η γραφική παράσταση είναι συμμετρική ως προς τον άξονα $y'$.
    \vspace{0.5\baselineskip}

    Το ολοκλήρωμα $I = \int_{-1}^{1} f(x) \,dx$ εκφράζει το εμβαδόν του χωρίου που ορίζεται από την γραφική παράσταση της συνάρτησης $f$, τον άξονα $x'x$ και τις ευθείες με εξισώσεις $x = -1$ και $x = 1$, αφού $f(x) > 0$ για κάθε $x \in [-1,1]$.
    \vspace{0.5\baselineskip}

    Λόγω συμμετρίας μπορούμε να ισχυριστούμε ότι $I = 2 \int_{0}^{1} f(x) \,dx$.
    \vspace{0.5\baselineskip}

    Εναλλακτικά, παρατηρούμε ότι $I = \int_{-1}^{0} f(x) \,dx + \int_{0}^{1} f(x) \,dx$.
    \vspace{0.5\baselineskip}

    Έστω $J = \int_{-1}^{0} f(x) \,dx$. Θέτουμε $u = -x \Rightarrow du = -dx$, οπότε
    \[ J = \int_{1}^{0} f(-u) \,(-du) = \int_{0}^{1} f(-u) \,du = \int_{0}^{1} f(u) \,du. \]
    \vspace{0.5\baselineskip}

    Έτσι, έχουμε $I = 2 \int_{0}^{1} f(x) \,dx = \frac{\pi}{2}$.
    \vspace{0.5\baselineskip}

    Καθώς οι συναρτήσεις $f(x)$ και $g(x) = -\frac{x}{2} + 1$ είναι ίσες στο διάστημα $[0,1]$, αφού π.χ. $f\PARENS{\frac{1}{2}} = \frac{4}{5} \ne g\PARENS{\frac{1}{2}} = \frac{3}{4}$ από β) ερώτημα παίρνουμε:
    \[ \int_{0}^{1} \PARENS{f(x) - \PARENS{-\frac{x}{2} + 1}} \,dx = \int_{0}^{1} \left[ \frac{1}{1+x^2} - \left(-\frac{x}{2} + 1\right) \right] \,dx = \frac{3}{4}. \]
    \vspace{0.5\baselineskip}

    Άρα $2 \int_{0}^{1} f(x) \,dx > \frac{3}{2} \Leftrightarrow \int_{-1}^{1} f(x) \,dx > \frac{3}{2}$.
