    \item [Δυναμική θερμοκρασίας, Ολοκλήρωμα]
\end{ylh}
ΛΥΣΗ
\begin{alist}
    \item Είναι $\lim_{t\to+\infty} T(t) = E + (T_0 - E) \lim_{t\to+\infty} \frac{1}{e^{kt}} = E + (T_0 - E) \lim_{y\to+\infty} \frac{1}{e^y} = E$ διότι αν θέσουμε $y = kt$, τότε $\lim_{t\to+\infty} y = +\infty$ και $\lim_{y\to+\infty} \frac{1}{e^y} = 0$ αφού $\lim_{y\to+\infty} e^y = +\infty$.
    
    Πρακτικά, αυτό σημαίνει ότι μετά την πάροδο απεριόριστα μεγάλου χρονικού διαστήματος, η θερμοκρασία του σώματος θα εξισωθεί με την θερμοκρασία του περιβάλλοντος.
    \item Έχουμε $T'(t) = (E)' + (T_0 - E)(e^{-kt})' = (T_0 - E)(-kt)'e^{-kt} = -k(T_0 - E)e^{-kt}$
    
    άρα $T'(t) = -k[T(t) - E] = k[E - T(t)]$.
    \item Από το β) ερώτημα, έχουμε:
    \[ I = \frac{1}{k} \int_0^1 T'(t) \cdot \ln(T(t))dt = \frac{1}{k} \left\{ [T(t) \cdot \ln T(t)]_0^1 - \int_0^1 T(t)[\ln T(t)]'dt \right\} \]
    \[ = \frac{1}{k} \left\{ [T(t) \cdot \ln T(t)]_0^1 - \int_0^1 T(t) \frac{T'(t)}{T(t)} dt \right\} = \frac{1}{k} \left\{ [T(t) \cdot \ln T(t)]_0^1 - \int_0^1 T'(t)dt \right\} \]
    \[ = \frac{1}{k} \{ T(1) \ln T(1) - T(0) \ln T(0) - [T(1) - T(0)] \} = \frac{1}{k} (e^3 \cdot 3 - e^4 \cdot 4 - e^3 + e^4) . \]
    Τελικά $I = \frac{2e^3 - 3e^4}{k}$.
\end{alist}
