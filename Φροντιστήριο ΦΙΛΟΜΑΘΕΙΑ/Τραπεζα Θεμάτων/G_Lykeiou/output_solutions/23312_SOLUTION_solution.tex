\begin{ylh}{1}
    \item [Συναρτήσεις - Όρια - Συνέχεια - Διαφορικός Λογισμός]
\end{ylh}
Επειδή $f$ και $f'$ είναι δύο φορές παραγωγίσιμες συναρτήσεις στο $(-2,2)$, έχουμε:
\[ 2f(x)f''(x) - 2f'(x) + 2x = 0 \Rightarrow f(x)f''(x) - f'(x) + x = 0 \quad (1), \quad x \in (-2,2) \]
Επίσης $f$ και $f'$ είναι δύο φορές παραγωγίσιμες συναρτήσεις στο $(-2,2)$, οπότε:
\[ (1) \Rightarrow (f'(x))^2 + f(x)f''(x) - f''(x) + 1 = 0 \quad (2), \quad x \in (-2,2) \]
Αν η $f$ παρουσιάζει στο $x_0 \in (-2,2)$ σημείο καμπής τότε $f''(x_0) = 0$, οπότε:
\[ (2) \Rightarrow (f'(x_0))^2 = -1, \quad \text{άτοπο. Επομένως η } f \text{ δεν παρουσιάζει σημείο καμπής.} \]

\begin{alist}
    \item [Πληροφορίες από εικόνα]
    \item i. Για κάθε $x \in [-2,2]$ έχουμε $4-x^2 \ge 0$ και ισχύει ότι:
    \[ f^2(x) - 2f(x) + 1 = 4 - x^2 \Rightarrow (f(x)-1)^2 = 4-x^2 \quad (3) \]
    Θέτουμε $g(x) = f(x) - 1, x \in [-2,2]$. Τότε από την σχέση $(3)$ έχουμε:
    \[ g^2(x) = 4-x^2, \quad \text{για κάθε } x \in [-2,2] \]
    Έχουμε διαδοχικά:
    \[ g(x) = 0 \Leftrightarrow 4-x^2 = 0 \Leftrightarrow x = -2 \text{ ή } x = 2 \]
    Επομένως $g(x) \ne 0$ για κάθε $x \in (-2,2)$ και $g$ συνεχής στο $[-2,2]$, οπότε η $g$ διατηρεί πρόσημο στο $(-2,2)$, δηλαδή $g(x) > 0$ για κάθε $x \in (-2,2)$ ή $g(x) < 0$ για κάθε $x \in (-2,2)$.

    Επειδή $g(0) = f(0)-1 = 2 > 0$, έχουμε $g(x) > 0$ για κάθε $x \in (-2,2)$.
    Επίσης $g(-2) = g(2) = 0$.
    Άρα $g(x) = \sqrt{4-x^2}$ για κάθε $x \in [-2,2]$, δηλαδή $f(x) = 1 + \sqrt{4-x^2}$, για κάθε $x \in [-2,2]$.

    \item ii. Η $f$ είναι συνεχής στο $[-2,2]$ επομένως παρουσιάζει ολικά ακρότατα.
    Παρατηρούμε ότι: $0 \le \sqrt{4-x^2} \le 2 \Leftrightarrow 1 \le 1+\sqrt{4-x^2} \le 3 \Leftrightarrow$
    \[ 1 \le f(x) \le 3, \quad \text{για κάθε } x \in [-2,2] \]
    Επειδή $f(x) = 1 \Leftrightarrow x=-2$ ή $x=2$, η $f$ παρουσιάζει ολικό ελάχιστο το $1$, για $x_1 = 2$ και $x_2 = -2$.
    Επίσης $f(x) = 3 \Leftrightarrow x=0$, επομένως η $f$ παρουσιάζει ολικό μέγιστο το $3$, για $x_3 = 0$.
    Έχουμε $f(x) \ge 1$ και $\cos x \le 1$ για κάθε $x \in [-2,2]$. Η ισότητα στην δεύτερη ανίσωση ισχύει μόνο για $x = 0$.
    Όμως $f(0) = 3 > 1$, άρα $f(x) > \cos x$ για κάθε $x \in [-2,2]$.
    Τελικά η εξίσωση $f(x) = \cos x$ είναι αδύνατη στο $[-2,2]$.
\end{alist}
