\section*{\centering ΛΥΣΗ}

\begin{alist}
    \item Για $x=1$ στην $|f(x)-2x| \le (x-1)^2$ έχουμε
    \[
        |f(1)-2\cdot 1| \le (1-1)^2 \Rightarrow |f(1)-2| \le 0 \Rightarrow f(1)=2
    \]

    \item Είναι
    \[
        |f(x)-2x| \le (x-1)^2 \Rightarrow -(x-1)^2 \le f(x)-2x \le (x-1)^2 \Rightarrow 2x-(x-1)^2 \le f(x) \le 2x+(x-1)^2
    \]
    Όμως
    \[
        \lim_{x\to 1} (2x-(x-1)^2) = \lim_{x\to 1} (2x+(x-1)^2) = 2
    \]
    οπότε από κριτήριο παρεμβολής είναι και
    \[
        \lim_{x\to 1} f(x) = 2.
    \]

    \item Αφού $\lim_{x\to 1} f(x) = 2 = f(1)$ η $f$ είναι συνεχής στο $1$.
\end{alist}
