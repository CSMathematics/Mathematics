\begin{ylh}{1}
    \item [Συναρτήσεις - Πεδίο Ορισμού - Σύνθεση]
\end{ylh}
\textbf{ΛΥΣΗ}
\begin{alist}
\item Τα πεδία ορισμού των συναρτήσεων $f, g$ είναι αντίστοιχα:
    \[ D_f = \mathbb{R}, \text{ αφού } 1+e^x > 0 \text{ για κάθε } x \in \mathbb{R}. \]
    \[ D_g = (0, +\infty). \]
\item $D_{f \circ g} = \{x \in D_g \text{ και } g(x) \in D_f \}$ άρα
    \[ \{x \in (0, +\infty) \text{ και } 2\ln x \in \mathbb{R}\} = (0, +\infty). \]
    Και
    \[ (f \circ g)(x) = f(g(x)) = f(2\ln x) = \ln(1+e^{2\ln x}) = \ln(1+e^{\ln x^2}) = \ln(1+x^2). \]
\item Η συνάρτηση $h$ ορίζεται για $1+x^2>0 \Leftrightarrow x \in \mathbb{R}$ με $h(x) = \ln(1+x^2)$.
    Η συνάρτηση $f \circ g$ ορίζεται για $x \in (0, +\infty)$ με $(f \circ g)(x) = \ln(1+x^2)$.
    Οι δύο συναρτήσεις έχουν διαφορετικά πεδία ορισμού άρα δεν είναι ίσες.
\end{alist}
