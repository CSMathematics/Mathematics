\Askhsh{28684_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
% Η εκφώνηση της άσκησης δεν παρέχεται στην εικόνα, μόνο η λύση της.
\begin{alist}
\item Για $x \neq 0$ είναι $\frac{\eta\mu 2x}{x} = 2 \cdot \frac{\eta\mu 2x}{2x}$.
Έτσι, αν θέσουμε $u = 2x$, τότε
\[ \lim_{x \to 0} u = \lim_{x \to 0} 2x = 0, \]
οπότε
\[ \lim_{x \to 0} \frac{\eta\mu 2x}{x} = 2 \cdot \lim_{x \to 0} \frac{\eta\mu 2x}{2x} = 2 \cdot \lim_{u \to 0} \frac{\eta\mu u}{u} = 2 \cdot 1 = 2. \]
\item Είναι $xf(x) \le \eta\mu 2x$ για κάθε $x \in \mathbb{R}$.
Αν $x > 0$, τότε $f(x) \le \frac{\eta\mu 2x}{x}$.
Αν $x < 0$, τότε $f(x) \ge \frac{\eta\mu 2x}{x}$.
Επειδή η συνάρτηση $f$ (\text{εξ' υποθέσεως}) και η $g$ με $g(x) = \frac{\eta\mu 2x}{x}$ έχουν όριο κοντά στο $0$, τότε
\[ \lim_{x \to 0^+} f(x) \le \lim_{x \to 0^+} \frac{\eta\mu 2x}{x} \Rightarrow \kappa \le 2 \]
\[ \lim_{x \to 0^-} f(x) \ge \lim_{x \to 0^-} \frac{\eta\mu 2x}{x} \Rightarrow \kappa \ge 2 \]
Επομένως $\kappa = 2$.
\item Αφού η συνάρτηση $f$ είναι συνεχής στο $\mathbb{R}$, θα είναι συνεχής στο $0$. Άρα
\[ \lim_{x \to 0} f(x) = f(0) \Leftrightarrow \kappa = f(0) \Leftrightarrow f(0) = 2. \]
\item Γνωρίζουμε, από τα προηγούμενα ερωτήματα, ότι:
\[ \lim_{x \to 0} f(x) = 2 \quad (1) \]
Επίσης, έχουμε
\[ \lim_{x \to 0} \frac{\epsilon\phi x}{x} = \lim_{x \to 0} \frac{\eta\mu x}{x \sigma\upsilon\nu x} = \lim_{x \to 0} \left( \frac{\eta\mu x}{x} \cdot \frac{1}{\sigma\upsilon\nu x} \right) = \left( \lim_{x \to 0} \frac{\eta\mu x}{x} \right) \cdot \left( \lim_{x \to 0} \frac{1}{\sigma\upsilon\nu x} \right) = 1 \cdot 1 = 1 \quad (2) \]
Από τις σχέσεις (1) και (2) προκύπτει ότι
\[ \lim_{x \to 0} \left( f(x) \cdot \frac{\epsilon\phi x}{x} \right) = \left( \lim_{x \to 0} f(x) \right) \cdot \left( \lim_{x \to 0} \frac{\epsilon\phi x}{x} \right) = 2 \cdot 1 = 2 \]
Ο ισχυρισμός λοιπόν είναι λανθασμένος, διότι:
\[ \lim_{x \to 0} \left( f(x) \cdot \frac{\epsilon\phi x}{x} \right) = 2 > 0 \Rightarrow f(x) \cdot \frac{\epsilon\phi x}{x} > 0 \quad \text{κοντά στο } 0. \]
Επομένως $\left| f(x) \cdot \frac{\epsilon\phi x}{x} \right| = f(x) \cdot \frac{\epsilon\phi x}{x}$ κοντά στο $0$.
\end{alist}
