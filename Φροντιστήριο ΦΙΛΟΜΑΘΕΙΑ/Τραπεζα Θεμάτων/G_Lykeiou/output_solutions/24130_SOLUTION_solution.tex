\Askhsh{24130_SOLUTION}{1}
\begin{ylh}{1}
    \item [Συναρτήσεις]
\end{ylh}

Δίνεται η συνάρτηση $f(x) = \sqrt{x-1} + 3$ με πεδίο ορισμού $D_f = [1, \infty)$.
\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ είναι 1-1.
\item Να βρείτε το σύνολο τιμών της $f$ και την αντίστροφη συνάρτηση $f^{-1}$.
\item Να δείξετε ότι η γραφική παράσταση $C_f$ της $f$ είναι συμμετρική της γραφικής παράστασης $C_{f^{-1}}$ της $f^{-1}$ ως προς την ευθεία $y=x$.
\end{alist}
