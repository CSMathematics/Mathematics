\subsection*{\centering ΛΥΣΗ}
\begin{alist}
\item Η συνάρτηση $f$ είναι δυο φορές παραγωγίσιμη στο πεδίο ορισμού της με
    \[ f'(x) = -4 \frac{-1}{x^4} \cdot 2x = \frac{8}{x^3} \]
    και
    \[ f''(x) = 8 \frac{-1}{x^6} \cdot 3x^2 = -\frac{24}{x^4} \]
    απ' όπου προκύπτει ότι η συνάρτηση είναι:
    \begin{itemize}
        \item γνήσιως φθίνουσα στο διάστημα $(-\infty, 0)$
        \item γνήσιως αύξουσα στο διάστημα $(0, +\infty)$
        \item κοίλη σε καθένα από τα διαστήματα $(-\infty, 0)$ και $(0, +\infty)$
    \end{itemize}
    Επιπλέον,
    \[ \lim_{x \to +\infty} (f(x) - 4) = -\lim_{x \to +\infty} \frac{4}{x^2} = 0, \text{ οπότε } \lim_{x \to +\infty} f(x) = 4 \]
    άρα η $y=4$ είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$. Ομοίως βρίσκουμε ότι η ίδια ευθεία είναι οριζόντια ασύμπτωτη και στο $-\infty$. \monades{X}

\item Αν οι εφαπτόμενες της $C_f$ στα σημεία $A, B$ είναι κάθετες, τότε έχουμε $f'(x_1)f'(x_2) = -1$ και
    \[ f'(x_1)f'(x_2) = -1 \Rightarrow \frac{8}{x_1^3} \cdot \frac{8}{x_2^3} = -1 \Rightarrow (x_1x_2)^3 = -64 \Rightarrow x_1x_2 = -4 \] \monades{X}

\item \begin{rlist}
    \item Το εμβαδόν του ορθογωνίου ΑΒΓΔ είναι $(\text{ΑΒΓΔ})=4(\alpha-1)$, $\alpha > 1$ και
        \[ E(\Omega_2) = \int_1^\alpha \left(4 - \frac{4}{x^2}\right) dx = 4(\alpha-1) + 4 \left[\frac{1}{x}\right]_1^\alpha = 4\alpha - 4 + \frac{4}{\alpha} - 4 = \frac{4}{\alpha}(\alpha^2 - 2\alpha + 1) = \frac{4}{\alpha}(\alpha-1)^2 \]
        οπότε
        \begin{align*}
            E(\Omega_1) &= (\text{ΑΒΓΔ}) - E(\Omega_2) \\
            &= 4(\alpha-1) - \frac{4}{\alpha}(\alpha-1)^2 \\
            &= 4(\alpha-1) \left(1 - \frac{1}{\alpha}(\alpha-1)\right) \\
            &= 4(\alpha-1) \left(1 - 1 + \frac{1}{\alpha}\right) \\
            &= \frac{4}{\alpha}(\alpha-1)
        \end{align*}

    \item Είναι:
        \[ E(\Omega_1) = E(\Omega_2) \Leftrightarrow \frac{4}{\alpha}(\alpha-1) = \frac{4}{\alpha}(\alpha-1)^2 \overset{\alpha>1}{\Leftrightarrow} \alpha-1 = 1 \Leftrightarrow \alpha=2 \]
        Επομένως τα δυο χωρία είναι ισεμβαδικά μόνο όταν $\alpha=2$.
    \end{rlist} \monades{X}
\end{alist}
