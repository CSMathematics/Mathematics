\Askhsh{29150_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item \begin{rlist}
    \item Για κάθε $t \in [0,3]$, είναι:
    \[
    v(t) = x'(t) = (t-1)^2 + 2(t-2)(t-1)
    \]
    \[
    = (t-1)[(t-1)+2(t-2)]
    \]
    \[
    = (t-1)(3t-5)
    \]
    Το κινητό είχε ταχύτητα μηδέν μόνο όταν $v(t)=0$. Είναι $v(t)=0 \Leftrightarrow (t-1)(3t-5)=0 \Leftrightarrow t=1$ ή $t=\frac{5}{3}$. Οπότε το κινητό ήταν στιγμιαία ακίνητο (ταχύτητα μηδέν) τις χρονικές στιγμές $1 \text{ sec}$ και $\frac{5}{3} \text{ sec}$.

    \item Έχουμε:
    \[
    v(t) > 0 \Leftrightarrow (t-1)(3t-5)>0 \Leftrightarrow t \in \left[0,1\right) \cup \left(\frac{5}{3},3\right] \text{ και } v(t)<0 \Leftrightarrow t \in \left(1,\frac{5}{3}\right)
    \]
    Οπότε το κινητό Α κινήθηκε προς τα δεξιά στα χρονικά διαστήματα $\left[0,1\right)$ και $\left(\frac{5}{3},3\right]$ και προς τα αριστερά, στο χρονικό διάστημα $\left(1,\frac{5}{3}\right)$.
\end{rlist}

\item Από το χρονική στιγμή $0 \text{ sec}$ ως τη χρονική στιγμή $1 \text{ sec}$ το κινητό Α κινήθηκε προς τα δεξιά και διήνυσε διάστημα
\[
S_1 = |x(1)-x(0)| = |0-(0-2)(0-1)^2| = 2 \text{ m.}
\]
Από το χρονική στιγμή $1 \text{ sec}$ ως τη χρονική στιγμή $\frac{5}{3} \text{ sec}$ το κινητό Α κινήθηκε προς τα αριστερά και διήνυσε διάστημα
\[
S_2 = \left|x\left(\frac{5}{3}\right)-x(1)\right| = \left|\left(\frac{5}{3}-2\right)\left(\frac{5}{3}-1\right)^2-0\right| = \frac{4}{27} \text{ m.}
\]
Από το χρονική στιγμή $\frac{5}{3} \text{ sec}$ ως τη χρονική στιγμή $3 \text{ sec}$ το κινητό Α κινήθηκε προς τα δεξιά και διήνυσε διάστημα
\[
S_3 = \left|x(3)-x\left(\frac{5}{3}\right)\right| = \left|(3-2)(3-1)^2 - \left(\frac{5}{3}-2\right)\left(\frac{5}{3}-1\right)^2\right| = \frac{112}{27} \text{ m.}
\]
Άρα το συνολικό διάστημα που διήνυσε το κινητό Α από το χρονική στιγμή $0 \text{ sec}$ έως την χρονική στιγμή $3 \text{ sec}$ είναι:
\[
S = S_1 + S_2 + S_3 = 2 + \frac{4}{27} + \frac{112}{27} = \frac{54+4+112}{27} = \frac{170}{27} \text{ m.}
\]

\item Η μέση ταχύτητα $\bar{v}$ στο διάστημα $\left[1,\frac{5}{3}\right]$ είναι ίση με
\[
\bar{v}_{\left[1,\frac{5}{3}\right]} = \frac{x\left(\frac{5}{3}\right)-x(1)}{\frac{5}{3}-1} \text{ m/sec.}
\]
Όμως η συνάρτηση $x$ είναι συνεχής ως πολυωνυμική στο $\left[1,\frac{5}{3}\right]$ και παραγωγίσιμη στο $\left(1,\frac{5}{3}\right)$, επομένως από την εφαρμογή του θεωρήματος μέσης τιμής υπάρχει $t_0 \in \left(1,\frac{5}{3}\right)$ τέτοιο, ώστε
\[
x'(t_0) = \frac{x\left(\frac{5}{3}\right)-x(1)}{\frac{5}{3}-1} \Rightarrow v(t_0) = \bar{v}_{\left[1,\frac{5}{3}\right]}
\]
και έπεται το ζητούμενο.
\end{alist}
