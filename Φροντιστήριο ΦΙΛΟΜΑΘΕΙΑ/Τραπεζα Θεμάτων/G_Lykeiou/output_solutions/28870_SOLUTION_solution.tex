\Askhsh{28870_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
Δίνεται η συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ δύο φορές παραγωγίσιμη, με $f'(x) > 0$ για κάθε εσωτερικό σημείο $x$ του διαστήματος $(-3,-1)$, αφού η $C_f$ βρίσκεται πάνω από τον άξονα $x'x$.
Η $f'$ στο $[-1,0]$ είναι συνεχής και ισχύει $f'(x) > 0$ για κάθε εσωτερικό σημείο $x$ του διαστήματος $(-1,0)$, αφού η $C_f$ βρίσκεται πάνω από τον άξονα $x'x$.
Ομοίως η $f$ είναι συνεχής στο $[0,2]$ και $f'(x) < 0$ για κάθε $x \in (0,2)$, αφού η $C_f$ βρίσκεται κάτω από τον άξονα $x'x$.
Το πρόσημο της $f'$ και η μονοτονία της $f$ φαίνεται και στον παρακάτω πίνακα.

\[
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$     & $-3$  & $-1$  & $0$   & $2$   \\ \hline
$f'$    & $+$   &       & $-$   &       \\ \hline
$f$     & $\uparrow$ &       & $\downarrow$ &       \\ \hline
\end{tabular}
\]

\item [Α.] Να συμπληρώσετε τον παρακάτω πίνακα τιμών της $f'$ και να αιτιολογήσετε τις απαντήσεις σας. \monades{X}
\[

\item [Β.] ΕΞΕΤΑΣΤΕ ΑΝ ΟΙ ΠΑΡΑΚΑΤΩ ΠΡΟΤΑΣΕΙΣ ΕΙΝΑΙ ΣΩΣΤΕΣ Ή ΛΑΝΘΑΣΜΕΝΕΣ, δικαιολογώντας πλήρως την απάντησή σας.
\begin{rlist}
    \item Η $f'$ είναι γνησίως αύξουσα στο $[-3,0]$. \monades{X}
    \item Η $f$ είναι κοίλη στο $[-3,-1]$. \monades{X}
    \item Η $f$ παρουσιάζει σημείο καμπής στο $x=-1$. \monades{X}
    \item Το $\lim_{x \to 0} \frac{f'(x)-f'(0)}{x-0} = -4$. \monades{X}
    \item Η εφαπτομένη της γραφικής παράστασης της $f$ στο $x=2$ είναι οριζόντια. \monades{X}
\end{rlist}

\end{alist}
