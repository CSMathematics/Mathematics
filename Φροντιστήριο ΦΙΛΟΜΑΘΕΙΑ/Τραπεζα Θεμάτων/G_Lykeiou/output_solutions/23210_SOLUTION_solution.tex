\Askhsh{23210_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
    \item Καθώς η συνάρτηση $f'(x)$ είναι συνεχής ως παραγωγίσιμη και έχει μόνο δύο ρίζες, τους αριθμούς $a$ και $\beta$, από άμεση συνέπεια του Θεωρήματος Bolzano, θα διατηρεί σταθερό πρόσημο σε καθένα από τα διαστήματα $(-\infty, a)$, $(a, \beta)$, $(\beta, +\infty)$. Με χρήση της γραφικής παράστασης της $f'(x)$ έχουμε τον παρακάτω πίνακα, στον οποίο διακρίνονται τα διαστήματα μονοτονίας της $f$ και τα τοπικά ακρότατα, με την τιμή $f(a)$ να είναι τοπικό ελάχιστο και την τιμή $f(\beta)$ να είναι τοπικό μέγιστο.
    \[
    \begin{tabular}{|c|c|c|c|c|c|c|}
        \hline
        $x$ & $-\infty$ & & $a$ & & $\beta$ & $+\infty$ \\
        \hline
        $f'(x)$ & & $-$ & $0$ & $+$ & $0$ & $-$ \\
        \hline
        $f(x)$ & & $\searrow$ & Τ.Ε. & $\nearrow$ & Τ.Μ. & $\searrow$ \\
        \hline
    \end{tabular}
    \]
    \item Αφού η $f$ είναι συνεχής και γνησίως μονότονη σε καθένα από τα διαστήματα $(-\infty, a)$, $(a, \beta)$, $(\beta, +\infty)$, μπορούμε να βρούμε τα επιμέρους σύνολα τιμών.
    \[ f((-\infty, a]) = \left[f(a), \lim_{x \to -\infty} f(x)\right) = [f(a), +\infty) \]
    Στο οποίο διάστημα ανήκει ο αριθμός μηδέν, αφού $f(a) < 0$. Άρα, υπάρχει μοναδικός, λόγω μονοτονίας, αριθμός $\rho_1 \in (-\infty, a)$ ώστε $f(\rho_1) = 0$.

    Επειδή η $f$ είναι συνεχής στο $[a, \beta]$ με $f(a)f(\beta) < 0$, από την εφαρμογή του Θεωρήματος Bolzano, υπάρχει μοναδικό, λόγω μονοτονίας, $\rho_2 \in (a, \beta)$ ώστε $f(\rho_2) = 0$.
    \[ f([\beta, +\infty)) = \left(\lim_{x \to +\infty} f(x), f(\beta)\right] = (-\infty, f(\beta)] \]
    Στο οποίο διάστημα ανήκει ο αριθμός μηδέν, αφού $f(\beta) > 0$. Άρα, υπάρχει μοναδικός, λόγω μονοτονίας, αριθμός $\rho_3 \in (\beta, +\infty)$ ώστε $f(\rho_3) = 0$.

    Έτσι, υπάρχουν τρεις ακριβώς ρίζες της εξίσωσης $f(x) = 0$.
    \item Καθώς η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, με εφαρμογή του Θεωρήματος Μέσης Τιμής στο διάστημα $[x, x + 1]$ για κάθε $x \in \mathbb{R}$, θα υπάρχει $\xi \in (x, x + 1)$ ώστε
    \[ f'(\xi) = \frac{f(x+1)-f(x)}{x+1-x} = f(x+1) - f(x). \]
    Αλλά $f'(\xi) \le f'(x_0) = 2$, αφού η $f'(x)$ έχει ολικό μέγιστο στη θέση $x_0$.

    Ώστε $f(x + 1) - f(x) \le 2$, για κάθε $x \in \mathbb{R}$.
\end{alist}
