    \item [Ύλη από εικόνα]
\end{ylh}
\OnPar{ΛΥΣΗ}
\begin{alist}
\item [$\alpha$)]
    \begin{rlist}
    \item Η συνάρτηση $g$ είναι συνεχής στο κλειστό διάστημα $[-3,2]$ ως άθροισμα της συνεχούς στο $[-3,2]$ με την πολυωνυμική $x$.
    \item Η $C_f$ τέμνει τον άξονα $x'x$ στα σημεία $A$ και $B$ τα οποία είναι εκατέρωθεν του $O$, οπότε έστω $A(x_1, 0)$ και $B(x_2, 0)$ με $x_1 < 0 < x_2$ και $f(x_1) = f(x_2) = 0$.
    \end{rlist}
    Στο διάστημα $[x_1, x_2] \subseteq [-3,2]$ η συνάρτηση $g$ είναι συνεχής.
    \[ g(x_1) = f(x_1) + x_1 = 0 + x_1 = x_1 < 0 \]
    \[ g(x_2) = f(x_2) + x_2 = 0 + x_2 = x_2 > 0 \]
    \[ g(x_1) \cdot g(x_2) = x_1 \cdot x_2 < 0 \]
    Επομένως από το Θεώρημα Bolzano η εξίσωση $g(x) = 0$ έχει μία τουλάχιστον ρίζα στο $(x_1, x_2) \subseteq [-3,2]$. Δηλαδή η $g(x) = 0$ έχει μία τουλάχιστον ρίζα.
\item [$\beta$)]
    Αν η συνάρτηση $f$ είναι παραγωγίσιμη στο $(-1,2)$, τότε και η $g$ είναι παραγωγίσιμη στο $(-1,2)$ με $g'(x) = f'(x) + 1$. (1)
    Η $f$ παρουσιάζει μέγιστο στο $0$ το $f(0) = 3$ και αφού η $f$ είναι παραγωγίσιμη στο $0$, από Θεώρημα Fermat θα ισχύει ότι $f'(0) = 0$.
    Η εξίσωση της εφαπτομένης της $C_g$ σε ένα σημείο $x_0$ του πεδίου ορισμού της είναι:
    \[ y - g(x_0) = g'(x_0)(x - x_0) \]
    Για $x_0 = 0$ έχουμε:
    \[ y - g(0) = g'(0)(x - 0). \tag{2} \]
    Αλλά $g(0) = f(0) + 0 = 3$ και $g'(0) = f'(0) + 1$ λόγω της (1) ή $g'(0) = 1$.
    Οπότε η σχέση (2) γίνεται:
    \[ y - 3 = x \Leftrightarrow y = x + 3. \]
\end{alist}
