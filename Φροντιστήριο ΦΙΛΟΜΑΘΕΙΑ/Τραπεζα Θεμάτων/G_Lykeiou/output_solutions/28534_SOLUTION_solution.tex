\Askhsh{28534_SOLUTION}{1}
\begin{ylh}{1}
    \item [Μαθηματική Ανάλυση, Μεγιστοποίηση Συνάρτησης, Όρια]
\end{ylh}
\textbf{ΛΥΣΗ}

\begin{alist}
\item Αφού η περίμετρος του παραθύρου είναι $4 \text{ m}$, θα έχουμε $\pi \cdot x + 2x + 2y = 4$, άρα
\[
\frac{\pi}{2}x + x + y = 2
\]
οπότε
\[
\left(\frac{\pi}{2} + 1\right)x + y = 2
\]
έτσι θα είναι
\[
y = -\frac{\pi+2}{2} \cdot x + 2.
\]
Πρέπει $y > 0 \Leftrightarrow 2 > \frac{\pi+2}{2}x \Leftrightarrow x < \frac{4}{\pi+2}$.

Τότε $E(x) = \frac{\pi x^2}{2} + 2xy = \frac{\pi x^2}{2} + 2x \left(-\frac{\pi+2}{2} \cdot x + 2\right) = \frac{\pi x^2}{2} + 4x - (\pi+2)x^2 = $
\[
\left(\frac{\pi}{2} - \pi - 2\right)x^2 + 4x = -\left(2 + \frac{\pi}{2}\right)x^2 + 4x.
\]
Ώστε
\[
E(x) = -\frac{\pi+4}{2} \cdot x^2 + 4x, \text{ με } x \in \left(0, \frac{4}{\pi+2}\right).
\]

\item Η συνάρτηση $E(x)$ είναι παραγωγίσιμη ως πολυωνυμική με $E'(x) = -\frac{\pi+4}{2} \cdot 2x + 4$, άρα
\[
E'(x) = -(\pi+4) \cdot x + 4 \geq 0 \Leftrightarrow x \leq \frac{4}{\pi+4},
\]
οπότε παίρνουμε τον παρακάτω πίνακα μεταβολών.

\[
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & $0$ & $\frac{4}{\pi+4}$ & $\frac{4}{\pi+2}$ \\
\hline
$E'(x)$ & $+$ & $0$ & $-$ \\
\hline
$E(x)$ & $\nearrow$ & \boxed{\text{O.M.}} & $\searrow$ \\
\hline
\end{tabular}
\]

Άρα το συνολικό εμβαδόν του παραθύρου μεγιστοποιείται όταν $x = \frac{4}{\pi+4}$.

Αφού $x_0 = \frac{4}{\pi+4}$ θα είναι
\[
E(x_0) = -\frac{\pi+4}{2}\left(\frac{4}{\pi+4}\right)^2 + 4 \cdot \frac{4}{\pi+4} = -\frac{8}{\pi+4} + \frac{16}{\pi+4} = \frac{8}{\pi+4}.
\]

\item Από τα παραπάνω προκύπτει ότι $E(x) \leq E(x_0)$ για κάθε $x \in \left(0, \frac{4}{\pi+2}\right)$, άρα
\[
E(x) < E(x_0) \text{ κοντά στο } x_0.
\]
Ακόμα,
\[
\lim_{x \to x_0} [E(x) - E(x_0)] = 0, \text{ άρα } \lim_{x \to x_0} \frac{1}{E(x)-E(x_0)} = -\infty.
\]
Τέλος,
\[
\lim_{x \to x_0} \ln[E(x)] = \ln[E(x_0)] \text{ αφού η συνάρτηση } \ln[E(x)] \text{ είναι συνεχής στο } x_0.
\]
Αλλά $E(x_0) = \frac{8}{\pi+4} > 1$, άρα $\ln[E(x_0)] > 0$.
Έτσι, το ζητούμενο όριο ισούται με
\[
\lim_{x \to x_0} \frac{1}{E(x)-E(x_0)} \cdot \lim_{x \to x_0} \ln(E(x)) = (-\infty) \cdot \ln(E(x_0)) = -\infty.
\]
\end{alist}
