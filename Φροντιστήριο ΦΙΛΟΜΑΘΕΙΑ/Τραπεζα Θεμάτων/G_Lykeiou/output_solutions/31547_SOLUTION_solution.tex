\textbf{ΛΥΣΗ}
\begin{alist}
\item Για $x \neq 2$ είναι $f(x) = \frac{3-2x}{(x-2)^2}$ οπότε
\[
\lim_{x\to 2} f(x) = \lim_{x\to 2} \frac{3-2x}{(x-2)^2} = \lim_{x\to 2}(3-2x) \frac{1}{(x-2)^2} = -\infty
\]
αφού
\[
\lim_{x\to 2}(3-2x) = -1 \quad \text{και} \quad \lim_{x\to 2} \frac{1}{(x-2)^2} = +\infty.
\]
Επομένως, τα πλευρικά όρια της $f$ στο $x_o = 2$ είναι ίσα με το $-\infty$, άρα σύμφωνα με τον ορισμό, η $f$ έχει κατακόρυφη ασύμπτωτη την ευθεία $x=2$.

\item
\begin{rlist}
\item Η $f$ ορίζεται στο 2 και είναι $\lim_{x\to 2} f(x) \neq f(2)$ αφού $\lim_{x\to 2} f(x) = -\infty$ οπότε είναι ασυνεχής στο 2.
\item Αφού η $f$ δεν είναι συνεχής στο 2 δεν θα είναι ούτε παραγωγίσιμη στο 2.
\end{rlist}
\end{alist}
