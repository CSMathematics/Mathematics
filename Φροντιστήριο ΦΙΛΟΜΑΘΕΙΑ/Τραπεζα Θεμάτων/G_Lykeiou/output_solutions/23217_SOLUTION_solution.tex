\Askhsh{23217_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
    \item [\textbf{α)}] Για να ορίζεται η $f(x)=\ln(x-1)$ θα πρέπει $x>1$, ενώ για να ορίζεται η $g(x)=\frac{1}{x-1}$ θα πρέπει $x \neq 1$. Συνεπώς $D_f=(1,+\infty)$ και $D_g=(-\infty,1) \cup (1,+\infty)$.
    \begin{rlist}
        \item $\lim_{x \to 1^+} f(x) = \lim_{x \to 1^+} \ln(x-1) \overset{x-1=u}{=} \lim_{u \to 0^+} \ln u = -\infty$
        \item Η $g$ δεν έχει όριο στο $1$ διότι $\lim_{x \to 1^-} g(x) = \lim_{x \to 1^-} \frac{1}{x-1} = -\infty$
        αφού $\lim_{x \to 1^-}(x-1)=0$ και $x-1<0$ για $x<1$,
        ενώ $\lim_{x \to 1^+} g(x) = \lim_{x \to 1^+} \frac{1}{x-1} = +\infty$
        αφού $\lim_{x \to 1^+}(x-1)=0$ και $x-1>0$ για $x>1$.
    \end{rlist}
    \item [\textbf{β)}]
    \begin{rlist}
        \item Το πεδίο ορισμού της $f \cdot g$ είναι το $D_{f \cdot g} = D_f \cap D_g = (1,+\infty)$.
        \item Η συνάρτηση $f \cdot g$ ορίζεται για $x>1$, οπότε το όριο της στο $1$ ταυτίζεται με το δεξιό πλευρικό της όριο στο $1$.
        \[
        \text{Έχουμε ότι } \lim_{x \to 1^+} (f(x) \cdot g(x)) = \lim_{x \to 1^+} \left(\ln(x-1) \cdot \frac{1}{x-1}\right) = -\infty
        \]
        διότι $\lim_{x \to 1^+} \ln(x-1) = -\infty$ και $\lim_{x \to 1^+} \frac{1}{x-1} = +\infty$.
    \end{rlist}
\end{alist}
