    \item [Διαφορικός Λογισμός]
\end{ylh}
ΛΥΣΗ
\begin{alist}
    \item 
    \begin{rlist}
        \item Ισχύει:
        \[f(x)+2x = f'(x)+x^2 \Leftrightarrow f'(x)-2x = f(x)-x^2, \quad (1)\]
        Η $g$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $g'(x)=f'(x)-2x$, οπότε λόγω της (1), για κάθε $x \in \mathbb{R}$ ισχύει $g'(x)=g(x)$.
        \item Από το ερώτημα ($\alpha_{i}$) συμπεραίνουμε ότι $g(x)=c \cdot e^x, c \in \mathbb{R}$. \\
        Επίσης $g(0)=f(0)+0=1$, οπότε $c \cdot e^0 = 1 \Leftrightarrow c=1$. \\
        Έχουμε λοιπόν, $e^x = f(x)-x^2$, οπότε $f(x)=e^x+x^2, x \in \mathbb{R}$.
    \end{rlist}
    \item 
    \begin{rlist}
        \item Αρκεί να αποδείξουμε ότι υπάρχει μοναδικό $x_o \in (-1,0)$ ώστε $f'(x_o)=0$. \\
        Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο $f'(x)=e^x+2x$ η οποία είναι συνεχής στο διάστημα $[-1,0]$. \\
        Επιπλέον, $f'(-1)=\frac{1}{e}-2<0$ και $f'(0)=1>0$, οπότε $f'(-1)<0<f'(0)$. \\
        Επομένως, απ' το Θ. Ενδιαμέσων τιμών, υπάρχει $x_o \in (-1,0)$ ώστε $f'(x_o)=0$, που είναι το ζητούμενο. \\
        Η $f'$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο $f''(x)=e^x+2$ και $f''(x)>0$ για κάθε $x \in \mathbb{R}$. \\
        Άρα η $f'$ είναι γνησίως αύξουσα στο $\mathbb{R}$, οπότε η ρίζα $x_o$ είναι μοναδική.
        \item Από τη μονοτονία της $f'$ συμπεραίνουμε ότι:
        \begin{itemize}
            \item Για κάθε $x < x_o$ ισχύει $f'(x) < f'(x_o)$ δηλαδή $f'(x) < 0$, οπότε η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, x_o]$.
            \item Για κάθε $x > x_o$ ισχύει $f'(x) > f'(x_o)$ δηλαδή $f'(x) > 0$, οπότε η $f$ είναι γνησίως αύξουσα στο διάστημα $[x_o, +\infty)$.
        \end{itemize}
        Επομένως η $f$ παρουσιάζει ελάχιστο στο $x_o$. Η ελάχιστη τιμή $m$ της $f$ είναι
        \[m=f(x_o)=e^{x_o}+x_o^2\]
        Επιπλέον, $-1 < x_o < 0 \Rightarrow 0 < x_o^2 < 1$ και $-1 < x_o < 0 \Rightarrow e^{-1} < e^{x_o} < 1$, οπότε
        \[e^{-1} < x_o^2+e^{x_o} < 2 \Rightarrow e^{-1} < m < 2.\]
    \end{rlist}
\end{alist}
