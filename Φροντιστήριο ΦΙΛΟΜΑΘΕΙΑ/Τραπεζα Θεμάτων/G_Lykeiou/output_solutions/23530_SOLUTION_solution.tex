\Askhsh{23530_SOLUTION}{1}
\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\begin{alist}
\item Γνωρίζουμε ότι ο αριθμός $f'(-1)$ εκφράζει την κλίση της εφαπτομένης ευθείας $(\varepsilon)$ της γραφικής παράστασης της $f(x)$ στο σημείο $A(-1,f(-1))$ η οποία ισούται με
\[ \frac{y_A - y_O}{x_A - x_O} = \frac{f(-1) - 0}{-1 - 0} = \frac{e-1}{-1} = 1 - e. \]
Αφού η ευθεία $(\varepsilon)$ διέρχεται από την αρχή των αξόνων, η εξίσωση της θα είναι
\[ y = f'(-1) \cdot x, \text{ δηλαδή } (\varepsilon):y = (1 - e)x. \]
\item Αφού η ευθεία $y = 1 \cdot x + 0$ είναι ασύμπτωτη της γραφικής παράστασης της $f(x)$ στο $+\infty$, θα έχουμε ότι:
\[ \lim_{x \to +\infty} \left(\frac{f(x)}{x}\right) = 1 \]
\[ \lim_{x \to +\infty} (f(x) - x) = 0 \]
\item Έχουμε
\[ \lim_{x \to +\infty} \frac{xf(x) - x^2}{f(x)} = \lim_{x \to +\infty} \frac{x(f(x) - x)}{f(x)} = \lim_{x \to +\infty} \frac{f(x) - x}{\frac{f(x)}{x}} = \frac{\lim_{x \to +\infty} (f(x) - x)}{\lim_{x \to +\infty} \left(\frac{f(x)}{x}\right)} = \frac{0}{1} = 0 \]
\end{alist}
