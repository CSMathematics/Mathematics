\section*{ΛΥΣΗ}
\begin{alist}
\item Για κάθε $x > 0$ η $f$ είναι παραγωγίσιμη με $f'(x) = e^x - \frac{1}{x}$ και $f''(x) = e^x + \frac{1}{x^2} > 0$.
Άρα η $f$ είναι κυρτή στο $(0, +\infty)$.
\item Αφού η συνάρτηση $f'$ είναι γνησίως αύξουσα και συνεχής, θα έχουμε ότι για $x \in (0,1)$ οι αντίστοιχες τιμές $f'(x)$ θα ανήκουν στο διάστημα
\[ f'((0,1)) = \left(\lim_{x\to 0^+} f'(x), \lim_{x\to 1^-} f'(x)\right) = (-\infty, e-1) \]
στο οποίο ανήκει ο αριθμός μηδέν και καθώς η συνάρτηση $f'$ είναι γνησίως αύξουσα στο $(0,1)$ θα υπάρχει μοναδικό $x_0 \in (0,1)$ ώστε $f'(x_0) = 0$.
Τότε για $x > x_0$ θα είναι $f'(x) > f'(x_0) = 0$, άρα η $f$ είναι γνησίως αύξουσα στο διάστημα $[x_0, +\infty)$ και ανάλογα
για $x < x_0$ θα είναι $f'(x) < f'(x_0) = 0$, άρα η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, x_0]$. Έτσι, έχουμε τον παρακάτω πίνακα μεταβολών της $f$.
\begin{center}
\begin{tabular}{|c|c|c|c|}
    \hline
    $x$ & $0$ & $x_0$ & $+\infty$ \\
    \hline
    $f'(x)$ & \rule{0pt}{1.2em} & $-$ & $+$ \\
    \hline
    $f(x)$ & \rule{0pt}{1.2em} & $\searrow$ & $\nearrow$ \\
    \cline{3-3} % Line only under x_0 in this row
    \multicolumn{2}{|c|}{} & $f(x_0)$ & \\
    \hline
\end{tabular}
\end{center}
Ώστε η $f$ παρουσιάζει ελάχιστο για $x=x_0$ αφού για $x \le x_0 \Rightarrow f(x) \ge f(x_0)$ ενώ για $x \ge x_0 \Rightarrow f(x) \ge f(x_0)$.
Επίσης, από τον παραπάνω πίνακα έχουμε ότι $x_0 < 1 \Rightarrow f(x_0) < f(1) = e-3 < 0$.
\item Επειδή είναι $f(x) - f(x_0) > 0$ κοντά στο $x_0$ και $\lim_{x\to x_0} (f(x) - f(x_0)) = 0$, έχουμε ότι
\[ \lim_{x\to x_0} \frac{1}{f(x)-f(x_0)} = +\infty. \]
Ακόμα, $\lim_{x\to x_0} (f(x))^{2023} = (f(x_0))^{2023} < 0$, καθώς η $f(x)$ είναι συνεχής στο $(0, +\infty)$.
Ώστε το ζητούμενο όριο είναι $\lim_{x\to x_0} \frac{1}{f(x)-f(x_0)} \cdot \lim_{x\to x_0} (f(x))^{2023} = -\infty$.
\end{alist}
