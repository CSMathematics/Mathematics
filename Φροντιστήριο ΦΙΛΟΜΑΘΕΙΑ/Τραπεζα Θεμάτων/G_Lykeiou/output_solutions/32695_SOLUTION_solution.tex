\begin{ylh}{1}
    \item [Ύλη από εικόνα]
\end{ylh}
\textbf{ΛΥΣΗ}
\begin{alist}
\item Από τον τρόπο που ορίστηκε η $g$ προκύπτει ότι έχει πεδίο ορισμού το σύνολο
τιμών $\left[-\frac{1}{2},1\right)$ της $f$ και σύνολο τιμών το πεδίο ορισμού $[0,+\infty)$ της $f$.

Σύμφωνα με τον ορισμό της αντίστροφης, για να δείξουμε ότι η $g$ είναι η αντίστροφη
της $f$ αρκεί να δείξουμε την ισοδυναμία $f(x)=y \iff g(y)=x$.

Για $x \in [0,+\infty)$ και $y \in \left[-\frac{1}{2},1\right)$ έχουμε:
\[
f(x)=y \iff 1 - \frac{3}{\sqrt{x}+2} = y \iff \sqrt{x}+2-3 = y(\sqrt{x}+2) \iff
\]
\[
\iff \sqrt{x}-1 = y\sqrt{x}+2y \iff \sqrt{x}-y\sqrt{x} = 1+2y \iff
\]
\[
\iff (1-y)\sqrt{x} = 1+2y \iff \sqrt{x} = \frac{1+2y}{1-y} \iff x = \left(\frac{1+2y}{1-y}\right)^2
\]
\[
\iff g(y)=x,
\]
όπου $\frac{1+2y}{1-y} \ge 0$ για $y \in \left[-\frac{1}{2},1\right)$.

Επομένως η $g$ είναι η αντίστροφη της $f$.
\item Για $x \in [0,1)$ έχουμε: $f(x)=1 - \frac{3}{\sqrt{x}+2} = \frac{\sqrt{x}+2-3}{\sqrt{x}+2} = \frac{\sqrt{x}-1}{\sqrt{x}+2} < 0$
(γιατί $\sqrt{x}-1 < 0$ και $\sqrt{x}+2 > 0$)

ενώ $g(x)=\left(\frac{1+2x}{1-x}\right)^2 > 0$.
\item Για να δείξουμε ότι οι $C_f, C_g$ δεν έχουν κοινά σημεία αρκεί να δείξουμε ότι η εξίσωση
\[
f(x) = g(x), \quad x \in D_f \cap D_g = [0,1)
\]
είναι αδύνατη.

Από το β) ερώτημα έχουμε ότι $f(x) < 0$ και $g(x) > 0, x \in [0,1)$.
Επομένως η εξίσωση είναι αδύνατη και οι $C_f, C_g$ δεν έχουν κοινά σημεία.
\end{alist}
