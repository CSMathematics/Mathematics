    \item [Λύση]

\kefalaio{ΛΥΣΗ}
\begin{alist}
\item \textbf{α)} Είναι: $\lim_{x \to 0^-} f(x) = \lim_{x \to 0^-} e^x = 1$
\[
\lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \text{συν} x = 1
\]
\[
f(0) = 1
\]
οπότε ισχύει $\lim_{x \to 0^-} f(x) = \lim_{x \to 0^+} f(x) = f(0)$, επομένως η συνάρτηση $f$ είναι συνεχής στο $x_0 = 0$.

\item \textbf{β)} Είναι: $\lim_{x \to 0^-} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^-} \frac{e^x - 1}{x} \overset{\frac{0}{0}}{=} \lim_{\substack{\text{\tiny DL'H} \\ x \to 0^-}} \frac{(e^x - 1)'}{(x)'} = \lim_{x \to 0^-} \frac{e^x}{1} = 1$
\[
\lim_{x \to 0^+} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^+} \frac{\text{συν} x - 1}{x} = 0
\]
Εφόσον τα δύο όρια διαφέρουν, συμπεραίνουμε ότι η συνάρτηση $f$ δεν είναι παραγωγίσιμη στο $x_0 = 0$.

\item \textbf{γ)} Για $x > 0$ η συνάρτηση $f$ είναι παραγωγίσιμη με $f'(x) = (\text{συν} x)' = -\text{ημ} x$.
Η εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x = \frac{\pi}{2}$ είναι $(\varepsilon): y - f\left(\frac{\pi}{2}\right) = f'\left(\frac{\pi}{2}\right)\left(x - \frac{\pi}{2}\right)$.
\[
\text{Όμως } f\left(\frac{\pi}{2}\right) = \text{συν} \frac{\pi}{2} = 0 \text{ και } f'\left(\frac{\pi}{2}\right) = -\text{ημ} \frac{\pi}{2} = -1
\]
Οπότε $(\varepsilon): y = -\left(x - \frac{\pi}{2}\right)$ ή $(\varepsilon): y = -x + \frac{\pi}{2}$.
\end{alist}
