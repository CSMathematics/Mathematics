\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Έστω συνάρτηση $f$ γνησίως μονότονη στο $\mathbb{R}$ της οποίας η γραφική της παράσταση διέρχεται από τα σημεία $A(3,0)$ και $B(0,8)$.
\begin{alist}
\item Να αποδείξετε ότι η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$. \monades{7}
\item Να βρείτε για ποιες τιμές του $x$ η $C_f$ είναι κάτω από τον άξονα $xx'$ και για ποιες είναι πάνω από τον $xx'$. \monades{8}
\item Να λύσετε την ανίσωση $f(\ln x) > 0$. \monades{10}
\end{alist}
