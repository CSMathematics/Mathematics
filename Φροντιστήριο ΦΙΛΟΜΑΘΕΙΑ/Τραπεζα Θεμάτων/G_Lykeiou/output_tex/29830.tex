\Askhsh{29830}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις.
\end{ylh}
Δίνονται οι συναρτήσεις $f(x) = \sqrt{9-x^2}$ και $g(x) = \frac{\sqrt{4-x^2}}{x}$.
\begin{alist}
\item Να βρείτε τα πεδία ορισμού των συναρτήσεων $f$ και $g$. \monades{10}
\item Να ορίσετε τις συναρτήσεις:
    \begin{rlist}
    \item $f \cdot g$ \monades{7}
    \item $\frac{f}{g}$ \monades{8}
    \end{rlist}
\end{alist}
