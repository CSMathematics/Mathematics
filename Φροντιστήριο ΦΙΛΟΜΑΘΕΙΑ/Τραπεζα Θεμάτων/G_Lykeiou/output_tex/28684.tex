    \item 1.8 Συνέχεια συνάρτησης.
\end{ylh}
Δίνεται η συνεχής συνάρτηση $f: \mathbb{R} \to \mathbb{R}$, τέτοια ώστε
\[ \lim_{x \to 0} f(x) = \kappa, \text{ με } \kappa \in \mathbb{R}. \]
Αν επιπλέον ισχύει ότι $xf(x) \le \eta \mu 2x$ για κάθε $x \in \mathbb{R}$, τότε
\begin{alist}
\item Να αποδείξετε ότι
    \[ \lim_{x \to 0} \frac{\eta \mu 2x}{x} = 2 \]
    \monades{04}
\item Να αποδείξετε ότι $\kappa = 2$. \monades{09}
\item Να βρείτε το $f(0)$. \monades{04}
\item Να ελέγξετε την αλήθεια του παρακάτω ισχυρισμού:
    \[ \left| f(x) \cdot \frac{\varepsilon \varphi x}{x} \right| = -f(x) \cdot \frac{\varepsilon \varphi x}{x} \text{ κοντά στο } 0 \]
    Να δικαιολογήσετε τον ισχυρισμό σας. \monades{08}
\end{alist}
