\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 2.1 Η έννοια της παραγώγου
    \item 2.3 Κανόνες παραγώγισης.
\end{ylh}
Έστω μια συνάρτηση $f : (-\infty, 0) \to \mathbb{R}$ η οποία είναι παραγωγίσιμη στο $x_0 = -1$ και η συνάρτηση $g : \mathbb{R} \to \mathbb{R}$ με $g(x) = -x + 1$. Δίνεται ότι η εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο $A(-1, f(-1))$, έχει εξίσωση $y = g(x)$.

\begin{alist}
\item Να βρείτε το $f(-1)$ και το $f'(-1)$. \monades{5}
\item Να βρείτε:
    \begin{rlist}
    \item το πεδίο ορισμού των συναρτήσεων $f \circ g$ και $g \circ f$, \monades{6}
    \item τις παραγώγους $(f \circ g)'(2)$ και $(g \circ f)'(-1)$. \monades{8}
    \end{rlist}
\item Να αποδείξετε ότι η εφαπτομένη της $C_{f \circ g}$ στο σημείο της με τετμημένη $x_1 = 2$ και η εφαπτομένη της $C_{g \circ f}$ στο σημείο της με τετμημένη $x_0 = -1$, ταυτίζονται. \monades{6}
\end{alist}
