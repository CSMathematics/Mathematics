\Askhsh{29831}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις.
\end{ylh}
Δίνεται η συνάρτηση $f : \mathbb{R}^* \to \mathbb{R}$ και η συνάρτηση $g(x) = \ln \frac{1-x}{1+x}$.
\begin{alist}
\item Να αποδείξετε ότι το πεδίο ορισμού της συνάρτησης $g$ είναι το διάστημα $(-1, 1)$. \monades{8}
\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f \circ g$. \monades{9}
\item Δίνεται ότι η συνάρτηση $f(x)=\frac{e^x+1}{e^x-1}$, $x \in \mathbb{R}^*$. Να βρείτε τον τύπο της συνάρτησης $(f \circ g)(x)$. \monades{8}
\end{alist}
