Στο παρακάτω σχήμα δίνεται η γραφική παράσταση μιας συνάρτησης $f$, για την οποία γνωρίζουμε ότι είναι συνεχής και τέμνει τον άξονα $x'x$ σε ένα μόνο σημείο με τετμημένη $-2$ και τον άξονα $y'y$ σε ένα μόνο σημείο με τεταγμένη $2$.

\begin{alist}
\item Από την γραφική παράσταση ή με οποιονδήποτε άλλο τρόπο, να προσδιορίσετε τα όρια:
    \begin{rlist}
    \item $\lim_{x\to 0} f(x)$
    \item $\lim_{x\to -2^+} f(x)$
    \item $\lim_{x\to -2^-} f(x)$
    \end{rlist}
    \monades{12}
\item Να βρείτε τα όρια:
    \begin{rlist}
    \item $\lim_{x\to -2^+} \frac{1}{f(x)}$ \monades{6}
    \item $\lim_{x\to -2^-} \ln(f(x))$ \monades{7}
    \end{rlist}
    και να αιτιολογήσετε την απάντησή σας.
\end{alist}
