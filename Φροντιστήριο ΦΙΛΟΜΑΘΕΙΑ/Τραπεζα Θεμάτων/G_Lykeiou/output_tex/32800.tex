Δίνονται ακόμη ότι:
\begin{itemize}
    \item $\left(\int_7^9 f(x)dx\right)^2 = 16$ και
    \item η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ μόνο στα σημεία με τετμημένες $1, 3, 5, 7$.
\end{itemize}
\begin{alist}
    \item Να αποδείξετε ότι $\int_7^9 f(x)dx = 4$. \monades{10}
    \item Να υπολογίσετε το εμβαδό του χωρίου που περικλείεται από τη γραφική παράσταση της $f$ και τον άξονα $x'x$, όταν $x\in[1,9]$. \monades{07}
    \item Να υπολογίσετε το ολοκλήρωμα $\int_1^9 f(x)dx$. \monades{08}
\end{alist}
