\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνονται οι συναρτήσεις $g$ και $h$ ώστε:
\[ g(x) = 2\ln x, x>0 \text{ και } h(x) = \ln(1 + x^2), x \in \mathbb{R}. \]
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Η συνάρτηση $g$ είναι αντιστρέψιμη \monades{5}
    \item $g^{-1}(x) = e^{\frac{x}{2}}$, με $x \in \mathbb{R}$. \monades{10}
    \end{rlist}
\item Να ορίσετε τη συνάρτηση $h \circ g^{-1}$. \monades{10}
\end{alist}
