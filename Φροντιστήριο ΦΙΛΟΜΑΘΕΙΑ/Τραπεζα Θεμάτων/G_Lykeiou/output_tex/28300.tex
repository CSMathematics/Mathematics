\begin{alist}
    \item το πεδίο ορισμού και το σύνολο τιμών της $f$, \monades{6}
    \item τις τιμές $f(-1)$, $f(2)$ και $f(5)$, \monades{6}
    \item το ολικό μέγιστο και το ολικό ελάχιστο της $f$, εφόσον υπάρχουν, \monades{7}
    \item την τιμή της σύνθεσης $fof$ στο $-1$. \monades{6}
\end{alist}
