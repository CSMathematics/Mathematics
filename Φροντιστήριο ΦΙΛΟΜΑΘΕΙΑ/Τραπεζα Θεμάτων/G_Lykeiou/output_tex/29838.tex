    \item 1.8 Συνέχεια συνάρτησης.
\end{ylh}
Δίνεται η συνεχής συνάρτηση $f: \mathbb{R} \to \mathbb{R}$, για την οποία για κάθε $x \neq 0$ ισχύει:
\[ xf(x) + \sigma\upsilon\nu x = 1 - x^2 \eta\mu \frac{1}{x}. \]
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item $\lim_{x \to +\infty} \frac{1-\sigma\upsilon\nu x}{x} = 0$ \monades{4}
    \item $\lim_{x \to +\infty} f(x) = -1$. \monades{6}
    \end{rlist}
\item Να αποδείξετε ότι $f(0) = 0$. \monades{8}
\item Να αποδείξετε ότι η εξίσωση $f(x) = 0$ έχει μία τουλάχιστον ρίζα στο διάστημα $\left(\frac{1}{\pi}, +\infty\right)$. \monades{7}
\end{alist}
