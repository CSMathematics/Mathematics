Δίνεται η συνάρτηση $f(x)=x^3+5x-2, x \in \mathbb{R}$.
\begin{alist}
\item \begin{rlist}
    \item Να αποδείξετε ότι η γραφική παράσταση της $f$ τέμνει τον άξονα $x'x$ σε ένα μόνο σημείο με τετμημένη $x_0$ που περιέχεται στο διάστημα $(0, 1)$. \monades{5}
    \item Να εξετάσετε αν ο αριθμός $x_0$ είναι πιο κοντά στο 0 ή στο 1. \monades{4}
\end{rlist}
\item Να υπολογίσετε το όριο $\lim_{x\to+\infty} \frac{f(x_0+\theta)x^3+2x-5}{f(x_0-\theta)x-5}$, αν $x_0$ είναι ο αριθμός του ερωτήματος (α) και $\theta$ ένας θετικός αριθμός. \monades{9}
\item Να υπολογίσετε το εμβαδόν του χωρίου που ορίζεται από τη γραφική παράσταση $C_f$ της $f$, την εφαπτομένη της στο σημείο $A(1, 4)$ και τις ευθείες $x=1$ και $x=2$. \monades{7}
\end{alist}
