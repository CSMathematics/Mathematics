\begin{ylh}{1}
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση
    \item 3.5 Η συνάρτηση [ορισμένο ολοκλήρωμα της f από α έως χ].
\end{ylh}
Δίνεται η συνάρτηση $f(x)=\frac{1}{1-x}$, με $x \neq 1$.
\begin{alist}
\item Να αποδείξετε ότι η $f$ αντιστρέφεται και να βρείτε τον τύπο της αντιστρόφου. \monades{9}
\item Να ορίσετε τη συνάρτηση $f \circ f$. \monades{6}
\item Ένας μαθητής ισχυρίζεται ότι οι συναρτήσεις $f \circ f$ και $f^{-1}$ είναι ίσες. Συμφωνείτε με τον ισχυρισμό του μαθητή; Να αιτιολογήσετε την απάντησή σας. \monades{5}
\item Αν $\varphi(x) = (f \circ f)(x) = \frac{x-1}{x}$ με $x \in \mathbb{R}-\{0,1\}$ να υπολογίσετε το ολοκλήρωμα $\int_{2}^{3} \varphi(x)dx$. \monades{5}
\end{alist}
