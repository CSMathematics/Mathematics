\Askhsh{24757}{2}
\begin{ylh}{1}
    \item 2.1 Η έννοια της παραγώγου.
\end{ylh}
Έστω συνάρτηση $f$ παραγωγίσιμη στο $\mathbb{R}$. Η εφαπτομένη της $C_f$ στο σημείο της $A(0,1)$ σχηματίζει με τον $xx'$ γωνία $45^\circ$.
\begin{alist}
\item Να αποδείξετε ότι $f'(0)=1$. \monades{8}
\item Να γράψετε την εξίσωση της εφαπτομένης της $C_f$ στο σημείο της $A(0,1)$. \monades{8}
\item Να αποδείξετε ότι $\lim_{x\to0} \frac{f(x)-1}{x}=1$. \monades{9}
\end{alist}
