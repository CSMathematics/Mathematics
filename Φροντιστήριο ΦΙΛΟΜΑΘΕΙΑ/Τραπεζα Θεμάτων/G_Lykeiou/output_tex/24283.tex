\begin{ylh}{1}
    \item 1.5 Ιδιότητες των ορίων
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.1 Η έννοια της παραγώγου
    \item 2.5 Θεώρημα Μέσης Τιμής Διαφορικού Λογισμού.
\end{ylh}
Δίνεται η συνάρτηση
\[
f(x)=\begin{cases}
x^2-3, & \text{αν } x \in [-1,2] \\
x-1, & \text{αν } x \in (2,5]
\end{cases}
\]
\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ είναι συνεχής. \monades{10}
\item Να αποδείξετε ότι η συνάρτηση $f$ δεν είναι παραγωγίσιμη στη θέση $x_0 = 2$. \monades{09}
\item Να εξετάσετε ποιες από τις υποθέσεις του θεωρήματος μέσης τιμής, ικανοποιεί η συνάρτηση $f$ στο διάστημα $[-1,5]$. \monades{06}
\end{alist}
