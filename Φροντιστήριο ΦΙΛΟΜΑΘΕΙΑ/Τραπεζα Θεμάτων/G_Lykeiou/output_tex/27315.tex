Δίνεται η συνάρτηση $f(x)=\begin{cases}
    \frac{x^2 - 4}{x - 2}, & \text{αν } x < 2 \\
    ax^2 - 4, & \text{αν } x \ge 2
\end{cases}$, με $a \in \mathbb{R}$.

\begin{alist}
\item Να βρείτε τα πλευρικά όρια της $f$ στο $x_0=2$, δηλαδή τα $\lim_{x\to 2^-} f(x)$ και $\lim_{x\to 2^+} f(x)$. \monades{12}
\item Να βρείτε την τιμή του $a$, ώστε η συνάρτηση $f$ να είναι συνεχής στο $x_0=2$. \monades{07}
\item Αν $a=2$, να βρείτε όπου ορίζεται την παράγωγο της συνάρτησης $f$. \monades{06}
\end{alist}
