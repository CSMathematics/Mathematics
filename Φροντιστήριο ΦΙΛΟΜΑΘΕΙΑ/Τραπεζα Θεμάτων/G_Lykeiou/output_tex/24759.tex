    \item 2.3 Κανόνες παραγώγισης
    \item 2.7 Τοπικά ακρότατα συνάρτησης
    \item 2.8 Κυρτότητα - Σημεία καμπής συνάρτησης
    \item 2.9 Ασύμπτωτες - Κανόνες De l’ Hospital.

Έστω συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ παραγωγίσιμη, για την οποία ισχύει $f(x) \geq x^2 - x + 1$ για κάθε $x \in \mathbb{R}$.
\begin{alist}
    \item Να αποδείξετε ότι:
    \begin{rlist}
        \item Να υπολογίσετε το $\lim_{x \to +\infty} \frac{f(x)}{x}$. \monades{4}
        \item Να αποδείξετε ότι η συνάρτηση $f$ δεν έχει ασύμπτωτες. \monades{6}
        \item Να αποδείξετε ότι $f(x) \geq \frac{3}{4}$ για κάθε $x \in \mathbb{R}$. \monades{5}
    \end{rlist}
    \item Αν επιπλέον $f(1)=1$ και $f\left(\frac{1}{2}\right)=\frac{3}{4}$ να αποδείξετε ότι:
    \begin{rlist}
        \item $f'\left(\frac{1}{2}\right)=0$. \monades{5}
        \item η $f$ δεν είναι κοίλη. \monades{5}
    \end{rlist}
\end{alist}
