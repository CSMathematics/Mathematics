\Askhsh{29835}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνονται οι συναρτήσεις $f(x) = \sqrt{x+1}-1$ και $g(x) = 2-x$.
\begin{alist}
\item Να βρείτε το πεδίο ορισμού των συναρτήσεων $f$ και $g$. \monades{5}
\item Να αποδείξετε ότι για $x \in (-\infty,3]$ η $(f \circ g)(x) = \sqrt{3-x}-1$. \monades{10}
\item Να αποδείξετε ότι η συνάρτηση $\varphi(x)=(f \circ g)(x)$ είναι αντιστρέψιμη και να ορίσετε την αντίστροφό της. \monades{10}
\end{alist}
