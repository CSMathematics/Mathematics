\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.6 Συνέπειες του Θεωρήματος Μέσης Τιμής.
\end{ylh}
Δίνεται συνεχής συνάρτηση $f : \mathbb{R} \to \mathbb{R}$ για την οποία ισχύουν :
\begin{itemize}
    \item $f^2(x) - 5 = x^2$ για κάθε $x \in \mathbb{R}$
    \item $f(2) = 3$
\end{itemize}

\begin{alist}
    \item Να αποδείξετε ότι :
    \begin{rlist}
        \item $f(x) \neq 0$ για κάθε $x \in \mathbb{R}$. \monades{4}
        \item $f(x) = \sqrt{x^2 + 5}$ για κάθε $x \in \mathbb{R}$. \monades{5}
    \end{rlist}
    \item Δίνεται η συνάρτηση $g$ με $g(x) = x^2 - \cos x$, με $x \in \mathbb{R}$. Να αποδείξετε ότι:
    \begin{rlist}
        \item Η συνάρτηση $g$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, 0]$ και γνησίως αύξουσα στο διάστημα $[0, +\infty)$. \monades{7}
        \item Η εξίσωση $f^2(x) = 5 + \cos x$ έχει ακριβώς δύο ρίζες, αντίθετες μεταξύ τους, οι οποίες ανήκουν στο διάστημα $(-\pi,\pi)$. \monades{9}
    \end{rlist}
\end{alist}
