\begin{ylh}{2}
    \item 3.4 Ορισμένο ολοκλήρωμα
    \item 3.7 Εμβαδόν επιπέδου χωρίου.
\end{ylh}
Δίνονται οι συνεχείς στο $\mathbb{R}$ συναρτήσεις $f$ και $g$.
Αν $\int_{1}^{3} f(x)dx = 6$, $\int_{1}^{8} f(x)dx = 29$, $\int_{3}^{5} f(x)dx = 8$ και $\int_{1}^{5} g(x)dx = -6$, τότε:
\begin{alist}
\item Να βρείτε τα ολοκληρώματα:
\begin{rlist}
\item $\int_{3}^{8} f(x)dx$
\item $\int_{5}^{8} 2f(x)dx$
\item $\int_{1}^{5} (f(x) + g(x))dx$
\end{rlist} \monades{18}
\item Αν για τη συνάρτηση $g$ ισχύει ότι $g(x) \le 0$ για κάθε $x \in [1,5]$, τότε να βρείτε το εμβαδόν του χωρίου που σχηματίζεται από τη γραφική παράσταση της $g$, τον άξονα $x'x$ και τις ευθείες $x=1$ και $x=5$. \monades{07}
\end{alist}
