\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.2 Παραγωγίσιμες συναρτήσεις - Παράγωγος συνάρτηση
    \item 2.3 Κανόνες παραγώγισης.
\end{ylh}
Δίνεται η συνάρτηση $f : \mathbb{R} \to \mathbb{R}$ με τύπο $f(x) = x \eta \mu x + 4$ για κάθε $x \in \mathbb{R}$.
\begin{alist}
\item Να βρείτε την παράγωγο της $f$ και να υπολογίσετε τις τιμές $f'(0)$ και $f'\left(\frac{\pi}{2}\right)$. \monades{8}
\item Να αποδείξετε ότι για τη συνάρτηση $\varphi$, με $\varphi(x) = f'(x) - \frac{1}{3}$, $x \in \mathbb{R}$ ισχύουν $\varphi(0) < 0$ και $\varphi\left(\frac{\pi}{2}\right) > 0$. \monades{8}
\item Να αποδείξετε ότι η εξίσωση $\varphi(x)=0$, έχει μία τουλάχιστον ρίζα στο διάστημα $\left(0, \frac{\pi}{2}\right)$. \monades{9}
\end{alist}
