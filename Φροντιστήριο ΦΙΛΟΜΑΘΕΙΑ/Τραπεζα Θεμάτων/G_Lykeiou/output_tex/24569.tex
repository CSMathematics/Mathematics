\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \sqrt{1 - \sqrt{1 - x}}$.
\begin{alist}
\item Να αποδείξετε ότι το πεδίο ορισμού της συνάρτησης είναι το $D_f = [0,1]$. \monades{5}
\item
    \begin{rlist}
    \item Να αποδείξετε ότι η συνάρτηση $f$ είναι "1-1". \monades{10}
    \item Να λύσετε την εξίσωση $f(f(x)) = 0, x \in [0,1]$. \monades{10}
    \end{rlist}
\end{alist}
