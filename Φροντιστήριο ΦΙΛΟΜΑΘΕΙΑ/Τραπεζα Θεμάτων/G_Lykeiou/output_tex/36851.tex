\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.1 Η έννοια της παραγώγου
    \item 2.5 Θεώρημα Μέσης Τιμής Διαφορικού Λογισμού.
\end{ylh}
Δίνεται η συνάρτηση $f$ με
\[ f(x) = \begin{dcases}
    -5x^2 - 3x + 1, & \text{αν } x \leq 0 \\
    x^2 - 3x + 1, & \text{αν } x > 0
\end{dcases} \]
\begin{alist}
    \item Να εξετάσετε αν η συνάρτηση $f$ είναι συνεχής στο $0$. \monades{7}
    \item Να εξετάσετε αν η συνάρτηση $f$ είναι παραγωγίσιμη στο $0$. \monades{7}
    \item Να δικαιολογήσετε γιατί μπορούμε να εφαρμόσουμε το θεώρημα Rolle στο διάστημα $[-1,1]$ και να βρείτε ένα τουλάχιστον $x_0 \in (-1,1)$ για το οποίο ισχύει $f'(x_0) = 0$. \monades{11}
\end{alist}
