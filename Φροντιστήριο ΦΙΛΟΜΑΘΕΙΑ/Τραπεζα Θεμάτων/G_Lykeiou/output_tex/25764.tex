\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.6 Συνέπειες του Θεωρήματος Μέσης Τιμής
\end{ylh}
Δίνεται η συνάρτηση $f(x)$ που ορίζεται ως:
\[ f(x) = \begin{cases} \ln(x+1), & x \geq 0 \\ x^3, & x < 0 \end{cases} \]
\begin{alist}
\item Να εξετάσετε αν είναι συνεχής στο $x_0 = 0$. \monades{12}
\item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$. \monades{13}
\end{alist}
