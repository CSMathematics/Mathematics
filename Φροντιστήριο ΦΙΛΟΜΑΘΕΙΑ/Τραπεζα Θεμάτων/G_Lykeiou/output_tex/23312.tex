    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.8 Κυρτότητα - Σημεία καμπής συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f$ ορισμένη στο $[-2, 2]$ τέτοια ώστε:
$f$ συνεχής στο $[-2, 2]$, δύο φορές παραγωγίσιμη στο $(-2, 2)$ και
$f^2(x) - 2f(x) + x^2 - 3 = 0$, για κάθε $x \in [-2, 2]$.

\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ δεν έχει σημεία καμπής. \monades{08}
\item Αν $f(0) = 3$,
    \begin{rlist}
    \item Να αποδείξετε ότι $(f(x) - 1)^2 = 4 - x^2$, για κάθε $x \in [-2, 2]$ και κατόπιν ότι $f(x) = 1 + \sqrt{4 - x^2}$, $x \in [-2, 2]$. \monades{09}
    \item Να βρείτε τα ολικά ακρότατα της $f$ και στη συνέχεια να λύσετε την εξίσωση $f(x) = \sigma\upsilon\nu x$. \monades{08}
    \end{rlist}
\end{alist}
