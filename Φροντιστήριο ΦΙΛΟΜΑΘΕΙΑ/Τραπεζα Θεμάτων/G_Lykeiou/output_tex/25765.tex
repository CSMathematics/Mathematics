    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση
    \item 1.8 Συνέχεια συνάρτησης
    \item 3.5 Η συνάρτηση [ορισμένο ολοκλήρωμα της f από α έως χ]
    \item 3.7 Εμβαδόν επιπέδου χωρίου.
\end{ylh}
Δίνεται η συνάρτηση $f(x)=2\ln x+x, x>0$.
\begin{alist}
\item Να αποδείξετε ότι αντιστρέφεται και να βρείτε το πεδίο ορισμού της $f^{-1}$. \monades{7}
\item Να λύσετε την ανίσωση $f^{-1}(x)>x$. \monades{8}
\item Έστω $g: \mathbb{R} \to \mathbb{R}$ μια συνεχής συνάρτηση για την οποία ισχύει $g(x)=e^{f(|x|)}$ για κάθε $x \ne 0$.
    \begin{rlist}
    \item Να αποδείξετε ότι $g(x)=x^2 e^{|x|}, x \in \mathbb{R}$. \monades{4}
    \item Να βρείτε το εμβαδόν του χωρίου που ορίζεται από την $C_g$, τον $x'x$ και τις κατακόρυφες ευθείες $x=-1, x=1$. \monades{6}
    \end{rlist}
\end{alist}
