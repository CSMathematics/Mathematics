\OnPar{ΘΕΜΑ 4}

Για μια συνεχή συνάρτηση $f:[-1,+\infty) \to \mathbb{R}$ ισχύουν:
\begin{itemize}
    \item $(f(x)+x)^2 = x^2(x+1)$, για κάθε $x \in [-1,+\infty)$,
    \item $f(1)>-1$ και $f\left(-\frac{1}{2}\right)<\frac{1}{2}$.
\end{itemize}

\begin{alist}
    \item Αν $g(x)=f(x)+x$, $x \in [-1,+\infty)$ τότε
    \begin{rlist}
        \item Να βρείτε τις λύσεις της εξίσωσης $g(x)=0$. \monades{05}
        \item Να αποδείξετε ότι $g(x)<0$ για κάθε $x \in (-1,0)$ και $g(x)>0$ για κάθε $x \in (0,+\infty)$. \monades{07}
    \end{rlist}
    \item Να αποδείξετε ότι $f(x)=x(\sqrt{x+1}-1)$, $x \ge -1$. \monades{07}
    \item Αν η συνάρτηση $f$ είναι κυρτή τότε να αποδείξετε ότι η $h(x)=f(x+1)-f(x)$, $x \in (-1,+\infty)$ είναι γνησίως αύξουσα και έπειτα ότι
    \[\int_{2023}^{2024}(f(x+1)-f(x))dx < \int_{2023}^{2024}(f(x+2)-f(x+1))dx.\] \monades{06}
\end{alist}
