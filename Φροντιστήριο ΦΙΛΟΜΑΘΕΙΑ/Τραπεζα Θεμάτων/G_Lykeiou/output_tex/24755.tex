Δίνεται η συνάρτηση
\[
f(x) = \begin{cases}
    \dfrac{1 - \sin x}{x}, & x \neq 0 \\
    \alpha, & x = 0
\end{cases}
\]
η οποία είναι συνεχής στο $\mathbb{R}$.

\begin{alist}
    \item Να αποδείξετε ότι $\alpha = 0$. \monades{10}
    \item Να αποδείξετε ότι $f'(0) = 0$. \monades{10}
    \item Να γράψετε την εξίσωση της εφαπτομένης της $C_f$ στο σημείο $(0, f(0))$. \monades{5}
\end{alist}
