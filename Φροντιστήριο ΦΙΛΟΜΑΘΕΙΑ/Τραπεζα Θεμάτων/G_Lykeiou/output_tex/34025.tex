Δίνεται η συνάρτηση $f(x) = \frac{1}{\ln x}$, $x \in (1,+\infty)$.

\begin{alist}
\item α.
    \begin{rlist}
    \item Να δείξετε ότι $f'(x)<0$ με $x \in (1, +\infty)$. \monades{4}
    \item Να βρείτε το σύνολο τιμών της συνάρτησης. \monades{6}
    \end{rlist}
\item β.
    \begin{rlist}
    \item Να δείξετε ότι η $f$ αντιστρέφεται. \monades{6}
    \item Να βρείτε την αντίστροφη της $f$. \monades{9}
    \end{rlist}
\end{alist}
