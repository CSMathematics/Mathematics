\begin{ylh}{1}
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνεται η συνάρτηση $f$ με πεδίο ορισμού το $[0,+\infty)$, σύνολο τιμών το $[-\frac{1}{2},1)$ και τύπο
\[ f(x)=1-\frac{3}{\sqrt{x}+2}. \]
Δίνεται επίσης η συνάρτηση $g$ με πεδίο ορισμού το $[-\frac{1}{2},1)$, σύνολο τιμών το $[0,+\infty)$ και τύπο
\[ g(x)=\left(\frac{1+2x}{1-x}\right)^2. \]
Με δεδομένο ότι η συνάρτηση $f$ είναι 1-1,

\begin{alist}
    \item Να αποδείξετε ότι η συνάρτηση $g$ είναι η αντίστροφη της συνάρτησης $f$. \monades{12}
    \item Να αποδείξετε ότι $f(x)<0$ και $g(x)>0$ για κάθε $x$ που ανήκει στο $[0,1)$. \monades{06}
    \item Να αποδείξετε ότι οι γραφικές παραστάσεις $C_f$, $C_g$ των συναρτήσεων $f$, $g$ αντίστοιχα δεν έχουν κοινά σημεία. \monades{07}
\end{alist}
