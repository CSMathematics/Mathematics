    \item 2.1 Η έννοια της παραγώγου
    \item 2.2 Παραγωγίσιμες συναρτήσεις - Παράγωγος συνάρτηση
    \item 2.3 Κανόνες παραγώγισης
    \item 2.5 Θεώρημα Μέσης Τιμής Διαφορικού Λογισμού
    \item 3.5 Η συνάρτηση $\int_{\alpha}^{x} f(t) dt$.
\end{ylh}
Θεωρούμε τους αριθμούς $\alpha, \beta$ με $1 < \alpha < \beta$ και την παραγωγίσιμη στο $\mathbb{R}$ συνάρτηση $f$, με συνεχή παράγωγο, ώστε $f(x) > 0$, για κάθε $x \in [\alpha, \beta]$. Ας είναι $\lambda$ ο συντελεστής διεύθυνσης της ευθείας που διέρχεται από τα σημεία $A(\alpha, f(\alpha))$ και $B(\beta, f(\beta))$, με $f(\alpha) \neq f(\beta)$.

\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $g(x) = \frac{f(x)+\lambda x-f(\alpha)}{x}$ ικανοποιεί τις υποθέσεις του θεωρήματος Rolle στο διάστημα $[\alpha, \beta]$. \monades{5}
\item Να αποδείξετε ότι υπάρχει $c \in (\alpha, \beta)$ ώστε $cf'(c) - f(c) - \lambda \alpha + f(\alpha) = 0$. \monades{6}
\item Αν γνωρίζουμε ότι $f'(c) \neq \lambda$, να αποδείξετε ότι η εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο $M(c, f(c))$ και η ευθεία $AB$ τέμνονται σε σημείο του άξονα $y'y$. \monades{7}
\item Αν είναι $\frac{f(\alpha)}{f(\beta)} = e^2$, να αποδείξετε ότι το ολοκλήρωμα
\[ I = \int_{\sqrt{\alpha-1}}^{\sqrt{\beta-1}} \frac{x \cdot f'(x^2 + 1)}{f(x^2 + 1)} dx \]
ισούται με $-1$. \monades{7}
\end{alist}
