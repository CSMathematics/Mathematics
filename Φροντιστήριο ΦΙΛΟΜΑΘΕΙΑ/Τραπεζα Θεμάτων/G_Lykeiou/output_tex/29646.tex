\begin{ylh}{1}
    \item 2.7 Τοπικά ακρότατα συνάρτησης
    \item 2.8 Κυρτότητα - Σημεία καμπής συνάρτησης
    \item 3.7 Εμβαδόν επιπέδου χωρίου.
\end{ylh}
Έστω η συνάρτηση $f$ με $f(x) = -x^3 + 3x^2 + 1, x \geq 0$.
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Η $f$ παρουσιάζει στο $x_1 = 0$ τοπικό ελάχιστο, στο $x_2 = 2$ μέγιστο και το σημείο $\Gamma(1,f(1))$ είναι σημείο καμπής της $C_f$. \monades{09}
    \item Τα σημεία $A(x_1,f(x_1))$, $B(x_2,f(x_2))$ και $\Gamma(x_3,f(x_3))$ είναι συνευθειακά και το σημείο $\Gamma$ είναι το μέσο του τμήματος $AB$. \monades{03}
    \end{rlist}
\item Να αποδείξετε ότι η ευθεία $AB$ ορίζει με την γραφική παράσταση της $f$ δύο ισεμβαδικά χωρία. \monades{08}
\item Έστω $\varepsilon$ η εφαπτομένη της $C_f$ στο σημείο της $B$, η οποία τέμνει τον άξονα $y'y$ στο $\Delta$. Να αποδείξετε ότι το εμβαδόν του τριγώνου $AB\Delta$ ισούται με το εμβαδόν του χωρίου που περικλείεται μεταξύ της $C_f$, της ευθείας $\varepsilon$ και του άξονα $y'y$. \monades{05}
\end{alist}
