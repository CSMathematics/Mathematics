    \item 2.6 Συνέπειες του Θεωρήματος Μέσης Τιμής
    \item 2.7 Τοπικά ακρότατα συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x)=x(\ln x-1)+1$, $x>0$.
\begin{alist}
\item Να την μελετήσετε ως προς την μονοτονία και τα ακρότατα. \monades{13}
\item Να λύσετε την εξίσωση $x\ln x+1=x$. \monades{12}
\end{alist}
