\[ f(x) = \begin{cases} e^{\frac{1}{\lambda x+1}}, & x > 1, \lambda \in \mathbb{R} \\ 0, & x=1 \end{cases} \]
\begin{alist}
\item Να αποδείξετε ότι $\lambda = -1$. \monades{6}
\item Να βρείτε, όπου ορίζεται, την παράγωγο της $f$. \monades{8}
\item Να μελετήσετε την $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{6}
\item Να βρείτε το σύνολο τιμών της $f$. \monades{5}
\end{alist}
