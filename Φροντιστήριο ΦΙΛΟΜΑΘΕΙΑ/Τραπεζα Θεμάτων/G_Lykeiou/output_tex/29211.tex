Δίνεται η συνάρτηση $f$, με $f(x) = 1 - \frac{1}{x^2}$, $x < 0$.
\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ είναι γνησίως φθίνουσα στο πεδίο ορισμού της. \monades{05}
\item Να βρείτε το σύνολο τιμών της $f$. \monades{08}
\item
    \begin{rlist}
    \item Να αποδείξετε ότι η $f$ είναι ``$1-1$''. \monades{05}
    \item Να βρείτε την αντίστροφη της συνάρτησης $f$, την $f^{-1}$. \monades{07}
    \end{rlist}
\end{alist}
