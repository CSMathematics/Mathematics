\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνονται οι συναρτήσεις f και g, με $f(x) = \frac{x^2-4}{x+2}$ και $g(x) = x-2$.
\begin{alist}
    \item Να εξετάσετε αν οι συναρτήσεις f και g είναι ίσες και να δικαιολογήσετε την απάντησή σας. \monades{8}
    \item Να σχεδιάσετε τις γραφικές παραστάσεις των συναρτήσεων f και h, με $h(x) = |g(x)|$. \monades{7}
    \item Με τη βοήθεια των γραφικών παραστάσεων των συναρτήσεων ή με όποιο άλλο τρόπο θέλετε, να μελετήσετε ως προς τη μονοτονία και τα ακρότατα τις συναρτήσεις f και h. \monades{10}
\end{alist}
