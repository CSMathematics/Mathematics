\begin{ylh}{1}
    \item 1.2 Συναρτήσεις.
\end{ylh}
Δίνονται οι συναρτήσεις $f(x) = \sqrt{x}$ και $g(x) = \ln x$.
\begin{alist}
\item Να ορίσετε τη συνάρτηση $f \cdot g$. \monades{9}
\item Να ορίσετε τη συνάρτηση $\frac{f}{g}$. \monades{9}
\item Να βρεθούν οι τετμημένες των σημείων τομής των γραφικών παραστάσεων των συναρτήσεων $f \cdot g$ και $\frac{f}{g}$, που ορίσατε στα ερωτήματα (α) και (β). \monades{7}
\end{alist}
