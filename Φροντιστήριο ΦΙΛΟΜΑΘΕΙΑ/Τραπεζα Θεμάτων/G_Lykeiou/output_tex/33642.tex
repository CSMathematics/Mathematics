    \item 2.3 Κανόνες παραγώγισης

    \item 2.7 Τοπικά ακρότατα συνάρτησης.
\end{ylh}
Έστω $f: \mathbb{R} \to \mathbb{R}$ μια παραγωγίσιμη συνάρτηση για την οποία $f(0)=1$ και για κάθε $x \in \mathbb{R}$ ισχύει
\[ f(x) + 2x = f'(x) + x^2 \]
\begin{alist}
\item Να αποδείξετε ότι αν $g(x) = f(x) - x^2$, τότε ισχύει
    \begin{rlist}
    \item $g'(x)=g(x)$ για κάθε $x \in \mathbb{R}$ \monades{5}
    \item $f(x)=e^x + x^2, x \in \mathbb{R}$ \monades{6}
    \end{rlist}
\item Να αποδείξετε ότι
    \begin{rlist}
    \item Υπάρχει μοναδικό σημείο $M(x_o, f(x_o))$, $x_o \in (-1, 0)$ στο οποίο η εφαπτομένη της $C_f$ είναι οριζόντια. \monades{7}
    \item Η $f$ παρουσιάζει ελάχιστο στο $x_o$ και για την ελάχιστη τιμή $m$ της συνάρτησης ισχύει $e^{-1} < m < 2$. \monades{7}
    \end{rlist}
\end{alist}
