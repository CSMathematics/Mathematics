\[ xf(x) = \eta\mu x \quad \text{για κάθε } x \in \left[0, \frac{\pi}{2}\right) \]
\begin{alist}
\item Να βρείτε το $f(0)$. \monades{04}

\[ \frac{\sqrt{2}}{6} \leq \int_{\frac{\pi}{6}}^{\frac{\pi}{4}} f(x)dx \leq \frac{1}{4} \] \monades{08}
