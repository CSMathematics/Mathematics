\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 2.1 Η έννοια της παραγώγου
    \item 2.3 Κανόνες παραγώγισης.
\end{ylh}
Έστω $f : \mathbb{R} \to \mathbb{R}$ μια παραγωγίσιμη συνάρτηση με $f (0) = -2$ και $f'(0) = 0$. Έστω επίσης οι συναρτήσεις $g: \mathbb{R} \to \mathbb{R}$ με $g(x) = -x$ και $g \circ f : \mathbb{R} \to \mathbb{R}$.
\begin{alist}
    \item Να βρείτε την τιμή $(g \circ f)(0)$. \monades{6}
    \item Να βρείτε την παράγωγο $g'(-2)$. \monades{6}
    \item Να βρείτε την παράγωγο της $g \circ f$ στο $x_0 = 0$. \monades{6}
    \item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής παράστασης της $g \circ f$ στο σημείο με τετμημένη $x_0 = 0$. \monades{7}
\end{alist}
