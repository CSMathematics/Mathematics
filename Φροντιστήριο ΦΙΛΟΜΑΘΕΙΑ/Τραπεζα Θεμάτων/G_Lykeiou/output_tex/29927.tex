Δίνεται η συνάρτηση $f : (0, 1) \cup (1, +\infty) \to \mathbb{R}$ με $f(x) = \dfrac{x}{\ln x}$.
\begin{alist}
\item Να βρείτε τα όρια: $\lim_{x \to 0^+} f(x)$, $\lim_{x \to 1^-} f(x)$, $\lim_{x \to 1^+} f(x)$ και $\lim_{x \to +\infty} f(x)$. \monades{6}
\item \begin{rlist}
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{5}
    \item Να βρείτε το σύνολο τιμών της $f$. \monades{5}
\end{rlist}
\item Δίνεται η εξίσωση $e^x = x^a$ (1) με $x > 0$. Να αποδείξετε ότι η (1) είναι ισοδύναμη με την εξίσωση $f(x) = a$ και να βρείτε το πλήθος των λύσεων της εξίσωσης αυτής, για τις διάφορες τιμές του πραγματικού αριθμού $a$. \monades{9}
\end{alist}
