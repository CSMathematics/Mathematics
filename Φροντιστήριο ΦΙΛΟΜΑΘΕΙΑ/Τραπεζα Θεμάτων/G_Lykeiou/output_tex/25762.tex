\Askhsh{25762}{2}
\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.1 Η έννοια της παραγώγου.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \begin{cases}
    -x^2 + x, & x \le 0 \\
    \sin x, & x > 0
\end{cases}$.
\begin{alist}
\item Να αποδείξετε ότι είναι συνεχής στο $x_o = 0$. \monades{8}
\item Να αποδείξετε ότι η $f$ είναι παραγωγίσιμη στο $x_o = 0$ και $f'(0)=1$. \monades{10}
\item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της $O(0, 0)$. \monades{7}
\end{alist}
