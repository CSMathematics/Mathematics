\Askhsh{35170}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνονται οι συναρτήσεις $f$ και $g$ ώστε:
$f(x) = \ln(1+e^x)$ και $g(x) = 2\ln x$.
\begin{alist}
\item Να βρείτε τα πεδία ορισμού των συναρτήσεων $f$ και $g$. \monades{8}
\item Να ορίσετε τη συνάρτηση $f+g$. \monades{8}
\item Να μελετήσετε τη συνάρτηση $f+g$ ως προς τη μονοτονία. \monades{9}
\end{alist}
