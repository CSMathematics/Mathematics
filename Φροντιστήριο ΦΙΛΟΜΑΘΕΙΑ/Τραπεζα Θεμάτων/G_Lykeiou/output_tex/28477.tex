Δίνονται οι συναρτήσεις $f,g$ με $f(x)=e^{3x+2}, x \in \mathbb{R}$ και $g(x)=\ln x^2$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της $g$. \monades{04}
    \item Να βρείτε την συνάρτηση $g \circ f$. \monades{08}
    \item Αν $g(f(x)) = 6x + 4, x \in \mathbb{R}$ τότε να υπολογίσετε το
    \[ \lim_{x \to 0} \frac{(g \circ f)(x) - \sin^2 x - 4}{x} \] \monades{13}
\end{alist}
