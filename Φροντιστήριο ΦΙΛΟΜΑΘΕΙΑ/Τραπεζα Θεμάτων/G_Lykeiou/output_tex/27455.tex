    \item 1.2 Συναρτήσεις
    \item 1.4 Όριο συνάρτησης στο Χο
    \item 2.6 Συνέπειες του Θεωρήματος Μέσης Τιμής.

Δίνονται οι συναρτήσεις $f: \mathbb{R} \to \mathbb{R}$ με
\[ f(x) = \begin{cases} -1, & x < 2 \\ e^{x-2} - 2, & x \ge 2 \end{cases} \]
και $g: \mathbb{R} - \{2\} \to \mathbb{R}$ με $g(x) = -\frac{1}{2}x^2 + 2x - 3$.

\begin{alist}
\item Να μελετήσετε ως προς τη μονοτονία:
    \begin{rlist}
    \item τη συνάρτηση $f$ και να αποδείξετε ότι $f(x) \ge -1$ για κάθε $x \in \mathbb{R}$.
    \item τη συνάρτηση $g$ και να βρείτε το σύνολο τιμών της.
    \end{rlist}
\monades{14}
\item Να δικαιολογήσετε γιατί η γραφική παράσταση της $f$ βρίσκεται πάνω από τη γραφική παράσταση της $g$ για κάθε $x \neq 2$.
\monades{04}
\item Δίνεται ο ισχυρισμός:
    «Αν $f(x) > g(x)$ κοντά στο $x_0$, τότε και $\lim_{x \to x_0} f(x) > \lim_{x \to x_0} g(x)$.»
    Να εξετάσετε αν είναι αληθής ή ψευδής ο παραπάνω ισχυρισμός και να δικαιολογήσετε την απάντησή σας.
\monades{07}
\end{alist}
