    \item 1.2 Συναρτήσεις
    \item 1.5 Ιδιότητες των ορίων.

Θεωρούμε τις συναρτήσεις με τύπους $f(x)=x^2-x+1$ και $g(x)=\sqrt{4x-3}$.
\begin{alist}
\item Να αποδείξετε ότι για κάθε $x \in \mathbb{R}$ ισχύει $f(x) \ge \frac{3}{4}$. \monades{6}
\item Να βρείτε τη συνάρτηση $h=g \circ f$. \monades{9}
\item Αν $h(x)=|2x-1|$ είναι η σύνθεση του ερωτήματος β), να υπολογίσετε το όριο $\lim_{x \to 0} \frac{h(x)-1}{\sqrt{x+1}-1}$. \monades{10}
\end{alist}
