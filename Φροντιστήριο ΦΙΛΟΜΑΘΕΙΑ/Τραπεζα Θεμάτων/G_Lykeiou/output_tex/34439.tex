    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση
    \item 1.7 Όρια συνάρτησης στο άπειρο

Δίνονται οι συναρτήσεις $f(x)=\frac{1}{x-1}, x \neq 1$ και $g(x)=\frac{1}{e^{x}}, x \in \mathbb{R}$.
\begin{alist}
\item
    \begin{rlist}
    \item Να ορίσετε το πεδίο ορισμού της συνάρτησης $h(x)=(f \circ g)(x)$. \monades{6}
    \item Να βρείτε τον τύπο της συνάρτησης $h(x)=(f \circ g)(x)$. \monades{6}
    \end{rlist}
\item Αν $h(x)=\frac{e^{x}}{1-e^{x}}, x \in \mathbb{R}^{*}$ τότε: \par
    να αποδείξετε ότι η συνάρτηση $h$ είναι '1-1'. \monades{7}
\item Να υπολογίσετε το όριο: $\lim_{x \rightarrow+\infty} h(x)$. \monades{6}
\end{alist}
