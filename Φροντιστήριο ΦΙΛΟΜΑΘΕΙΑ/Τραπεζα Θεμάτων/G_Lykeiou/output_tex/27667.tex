Δίνεται η συνάρτηση $f(x)=e^x+\frac{x^2}{2}+2023,x \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item η συνάρτηση $f$ είναι κυρτή στο $\mathbb{R}$. \monades{05}
    \item το σύνολο τιμών της $f'$ είναι το $\mathbb{R}$. \monades{06}
    \end{rlist}
\item Να αποδείξετε ότι για τις διάφορες τιμές του πραγματικού αριθμού $α$, η εξίσωση $e^x+x=α$ έχει μοναδική ρίζα $ρ$. \monades{05}
\item Να αποδείξετε ότι για τις διάφορες τιμές του πραγματικού αριθμού $α$, η συνάρτηση $g(x)=αx-f(x)$ με $x \in \mathbb{R}$, έχει μέγιστη τιμή την $ρf'(\rho)-f(\rho)$. \monades{09}
\end{alist}
