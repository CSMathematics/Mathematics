    \item 1.8 Συνέχεια συνάρτησης
    \item 2.1 Η έννοια της παραγώγου
    \item 2.2 Παραγωγίσιμες συναρτήσεις - Παράγωγος συνάρτηση
    \item 2.3 Κανόνες παραγώγισης.
\end{ylh}
Δίνεται η συνεχής συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ τέτοια, ώστε:
\[
\lim_{x \to 0} \left( \frac{f(x)}{\eta \mu x} \right) = 0.
\]
\begin{alist}
\item Να αποδείξετε ότι $f(0)=0$. \monades{08}
\item Να αποδείξετε ότι η $f$ είναι παραγωγίσιμη στο $x_0 = 0$ με $f'(0)=0$. \monades{08}
\item Θεωρούμε τη συνάρτηση $g(x) = f(x) \cdot \eta \mu x, x \in \mathbb{R}$.
    \begin{rlist}
    \item Να προσδιορίσετε την εξίσωση της εφαπτομένης της γραφικής παράστασης της συνάρτησης $g$, στο σημείο $(0,g(0))$. \monades{04}
    \item Να αποδείξετε ότι η συνάρτηση $g$ δεν είναι κυρτή. \monades{05}
    \end{rlist}
\end{alist}
