\Askhsh{35168}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις.
\end{ylh}
Δίνονται οι συναρτήσεις $f$, $g$ και $h$ ώστε:
\[f(x) = \ln(1+e^x) , g(x) = 2\ln x \text{ και } h(x) = \ln(1+x^2).\]
\begin{alist}
\item Να βρείτε τα πεδία ορισμού των συναρτήσεων $f$ και $g$. \monades{8}
\item Να ορίσετε την συνάρτηση $f \circ g$. \monades{9}
\item Να εξετάσετε αν οι συναρτήσεις $f \circ g$ και $h$ είναι ίσες. \monades{8}
\end{alist}
