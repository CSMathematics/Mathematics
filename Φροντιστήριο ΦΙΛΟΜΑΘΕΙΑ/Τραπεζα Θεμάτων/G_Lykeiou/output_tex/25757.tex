    \item 1.8 Συνέχεια συνάρτησης
    \item 3.4 Ορισμένο ολοκλήρωμα
    \item 3.7 Εμβαδόν επιπέδου χωρίου.

Δίνεται η συνάρτηση
\[
f(x) = \begin{cases}
    (1-x)\eta\mu^2\left(\frac{1}{1-x}\right), & \text{αν } 0 \le x < 1 \\
    0, & \text{αν } x = 1
\end{cases}
\]
\begin{alist}
\item Να αποδειχθεί ότι η συνάρτηση $f$ είναι συνεχής. \monades{09}
\item Να αποδειχθεί ότι για κάθε $x \in [0,1]$, ισχύει $0 \le f(x) \le 1-x$. \monades{07}
\item Να αποδειχθεί ότι για το εμβαδό $E$ του χωρίου $\Omega$ που περικλείεται από την γραφική παράσταση της συνάρτησης $f$, τον άξονα $x'x$ και τις ευθείες $x=0, x=1$ ισχύει $E < \frac{1}{2}$ τετραγωνικές μονάδες. \monades{09}
\end{alist}
