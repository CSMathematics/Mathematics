Δίνεται η συνάρτηση $f(x)=\ln(\sqrt{x^2+1}-x)$, $x \in \mathbb{R}$.
\begin{alist}
\item Να αποδειχθεί ότι για κάθε $x \in \mathbb{R}$ είναι $f'(x)=-\frac{1}{\sqrt{x^2+1}}$. \monades{06}
\item Αφού πρώτα δικαιολογήσετε ότι η συνάρτηση $f$ αντιστρέφεται, να αποδειχθεί ότι το πεδίο ορισμού της αντίστροφης είναι το $\mathbb{R}$. \monades{13}
\item Να λυθεί η ανίσωση $f^{-1}(x+f(x))>x$, $x \in \mathbb{R}$. \monades{06}
\end{alist}
