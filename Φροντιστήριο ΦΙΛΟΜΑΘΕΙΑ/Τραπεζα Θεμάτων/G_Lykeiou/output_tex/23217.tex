Δίνονται οι συναρτήσεις $f(x) = \ln(x-1)$ και $g(x) = \frac{1}{x-1}$.
\begin{alist}
\item Να εξετάσετε αν υπάρχουν τα παρακάτω όρια αιτιολογώντας την απάντησή σας.
    \begin{rlist}
    \item $\lim_{x \to 1^+} f(x)$ \monades{7}
    \item $\lim_{x \to 1} g(x)$ \monades{8}
    \end{rlist}
\item Να βρείτε
    \begin{rlist}
    \item το πεδίο ορισμού της $f \cdot g$ \monades{4}
    \item το $\lim_{x \to 1} (f(x) \cdot g(x))$. \monades{6}
    \end{rlist}
\end{alist}
