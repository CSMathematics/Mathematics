    \item 1.8 Συνέχεια συνάρτησης
    \item 2.1 Η έννοια της παραγώγου.
\end{ylh}
\OnKef{ΘΕΜΑ 2}

Δίνεται η συνάρτηση
\[
f(x)=
\begin{cases}
e^x & , \text{αν } x < 0 \\
1 & , \text{αν } x = 0 \\
\cos x & , \text{αν } x > 0
\end{cases}
\]

\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ είναι συνεχής στο $x_0 = 0$. \monades{9}
\item Να εξετάσετε αν η συνάρτηση $f$ είναι παραγωγίσιμη στο $x_0 = 0$. \monades{9}
\item Να βρείτε την εξίσωση της εφαπτομένης, της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x = \frac{\pi}{2}$. \monades{7}
\end{alist}
