    \item 1.5 Ιδιότητες των ορίων
    \item 2.1 Η έννοια της παραγώγου

    \item 3.7 Εμβαδόν επιπέδου χωρίου.

Δίνεται η παραγωγίσιμη συνάρτηση $f: \mathbb{R} \rightarrow \mathbb{R}$, που είναι τέτοια, ώστε:
\begin{itemize}
    \item η γραφική παράσταση της $f$, να εφάπτεται της $\varepsilon: y = \frac{1}{4}$, στο $x_0 = 0$.
    \item είναι κυρτή και
    \item $f(1)=1$.
\end{itemize}
\begin{alist}
\item Να αποδειχθεί ότι:
    \begin{rlist}
        \item $f(0)=\frac{1}{4}$ και $f'(0)=0$. \monades{06}
        \item $\lim_{x \to 0} \frac{4f(x)-1}{\eta\mu x \cdot f(x)}=0$. \monades{07}
    \end{rlist}
\item Επιπλέον δίνεται ότι η πρώτη παράγωγος της $f$ είναι συνεχής.
    \begin{rlist}
        \item Να αποδείξετε ότι $f'(x) \ge 0$, για κάθε $x \in [0,1]$. \monades{06}
        \item Να υπολογίσετε το εμβαδό $E$ του χωρίου που περικλείεται από την γραφική παράσταση της $f'$, τον άξονα $x'x$ και την ευθεία $x=1$. \monades{06}
    \end{rlist}
\end{alist}
