    \item 1.8 Συνέχεια συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $g$ με $g(x) = \sqrt{1-x^2}, x \in [-1,1]$ και η συνεχής συνάρτηση $f$, ορισμένη στο $[0, \pi]$, με $f\left(\frac{\pi}{2}\right) = 1$, τέτοιες ώστε:
\[ (g \circ f)(x) = |\sigma\upsilon\nu x|, \text{ για κάθε } x \in [0, \pi]. \]
\begin{alist}
    \item \leavevmode
    \begin{rlist}
        \item Να αποδείξετε ότι $|f(x)| = |\eta\mu x|$. \monades{06}
        \item Να βρείτε τις ρίζες της εξίσωσης $f(x) = 0$. \monades{03}
    \end{rlist}
    \item Να βρείτε την συνάρτηση $f$. \monades{09}
    \item Δίνεται η συνάρτηση $h: (0, \pi] \to \mathbb{R}$ με $h(x) = \frac{1}{f(x) - x}$, όπου $f$ είναι η συνάρτηση του προηγούμενου ερωτήματος. Να υπολογίσετε το παρακάτω όριο:
    \[ \lim_{x \to 0} h(x) \] \monades{07}
\end{alist}
