    \item 1.8 Συνέχεια συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \sqrt{9x^2+16} - \frac{5}{2}\ln(8x+1)$.
\begin{alist}
\item Να βρείτε το πεδίο ορισμού της $f$ και να αποδείξετε ότι είναι συνεχής στο πεδίο ορισμού της. \monades{8}
\item Να αποδείξετε ότι $f(0) > 0$ και $f(1) < 0$. \monades{8}
\item Να αποδείξετε ότι η εξίσωση $f(x)=0$ έχει μία τουλάχιστον ρίζα στο διάστημα $(0,1)$. \monades{9}
\end{alist}
