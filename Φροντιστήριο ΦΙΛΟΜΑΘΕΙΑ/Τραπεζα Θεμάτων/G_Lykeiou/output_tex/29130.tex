ΘΕΜΑ 4

Δίνεται η συνάρτηση $f(x) = x \eta \mu x, x \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Η ευθεία $y=x$ εφάπτεται της $C_f$ στο σημείο $A\left(\frac{\pi}{2},f\left(\frac{\pi}{2}\right)\right)$. \monades{04}
    \item Η $C_f$ έχει άπειρα κοινά σημεία με την εφαπτομένη της $y=x$ τα οποία και να προσδιορίσετε. \monades{06}
    \end{rlist}
\item Για τη συνάρτηση $g: \mathbb{R} \rightarrow \mathbb{R}$ ισχύει $g(x)-x = \ln\left(1+\frac{1}{e^x}\right)$, για κάθε $x \in \mathbb{R}$. Να αποδείξετε ότι:
    \begin{rlist}
    \item Η $y=x$ είναι ασύμπτωτη της $C_g$ στο $+\infty$. \monades{05}
    \item Στο διάστημα $(0,+\infty)$, η $C_g$ βρίσκεται πάνω από την $y=x$. \monades{04}
    \end{rlist}
\item Να αποδείξετε ότι στο διάστημα $(0,+\infty)$ η γραφική παράσταση της συνάρτησης $g$ του ερωτήματος (β) βρίσκεται πάνω από τη γραφική παράσταση της $f$. \monades{06}
\end{alist}
