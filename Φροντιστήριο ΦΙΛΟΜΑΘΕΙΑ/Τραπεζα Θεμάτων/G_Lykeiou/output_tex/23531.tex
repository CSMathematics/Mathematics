ΘΕΜΑ 4

Δίνεται η συνάρτηση $f(x) = e^x - \ln x - 3$.

\begin{alist}
\item Να αποδείξετε ότι η $f$ είναι κυρτή στο $(0, +\infty)$. \monades{6}
\item Να αποδείξετε ότι η $f(x)$ παρουσιάζει θέση ολικού ελαχίστου σε κάποιο $x_0 \in (0, 1)$ με $f(x_0) < 0$. \monades{10}
\item Να υπολογίσετε το $\lim_{x \to x_0} \frac{(f(x))^{2023}}{f(x)-f(x_0)}$. \monades{9}
\end{alist}
