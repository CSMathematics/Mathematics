Έστω συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ με $f(0)=0$ και για την οποία ισχύει ότι $\lim_{x \to 0} \frac{f(x)}{x} = 2$.
\begin{alist}
\item Να αποδείξετε ότι $f'(0) = 2$. \monades{9}
\item Να βρείτε το $\lim_{x \to 0} f(x)$. \monades{8}
\item Να βρείτε το $\lim_{x \to 0} \frac{f(x)}{\eta\mu x}$. \monades{8}
\end{alist}
