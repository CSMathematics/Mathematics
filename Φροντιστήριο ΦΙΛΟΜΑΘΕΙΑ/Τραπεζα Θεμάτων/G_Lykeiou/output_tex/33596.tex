Δίνεται η συνάρτηση $f(x)=\ln x$ και το σημείο $A(0,2)$. Αν $K(x,\ln x)$ με $x>0$ τυχαίο σημείο της $C_f$ και $M(x_o,\ln x_o)$ με $x_o>0$ το σημείο εκείνο της $C_f$ που απέχει την ελάχιστη απόσταση από το σημείο $A$, να αποδείξετε ότι:
\begin{alist}
\item η απόσταση $AK$ συναρτήσει του $x>0$ είναι $d(x) = \sqrt{x^2 + \ln^2 x-4\ln x+4}.$ \monades{5}
\item $x_o^2+\ln x_o-2=0.$ \monades{7}
\item η εφαπτομένη της $C_f$ στο $M$
    \begin{rlist}
    \item είναι κάθετη στην $AM$. \monades{6}
    \item τέμνει τον άξονα $xx'$ στο σημείο $(x_o^3-x_o,0)$. \monades{7}
    \end{rlist}
\end{alist}
