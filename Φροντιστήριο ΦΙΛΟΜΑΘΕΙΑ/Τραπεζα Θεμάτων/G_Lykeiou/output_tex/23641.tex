Δίνεται η γνησίως αύξουσα συνάρτηση $f: \mathbb{R} \to \mathbb{R}$.
\begin{alist}
\item Να λύσετε την ανίσωση $f(x^2) < f(x)$. \monades{08}
\item Αν $\alpha^2 < \alpha$, τότε να αποδείξετε ότι $\lim_{x \to +\infty} ([f(\alpha^2 - \alpha) - f(0)] x) = -\infty$. \monades{09}
\item Να λύσετε την εξίσωση $f(e^x - 1) = f(0)$. \monades{08}
\end{alist}
