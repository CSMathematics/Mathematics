Στο παρακάτω σχήμα η τεθλασμένη γραμμή $\Theta A \Lambda$ αποτελεί γραφική παράσταση μιας συνεχούς συνάρτησης $f$ ορισμένης στο $\mathbb{R}$, που διέρχεται από το σημείο $A(0,2)$ και τέμνει τον άξονα $x'x$ στο $\Gamma(-1,0)$.

\begin{alist}
\item Να υπολογίσετε τα ολοκληρώματα:
    \begin{rlist}
    \item $\int_{-2}^{-1} f(x)dx$
    \item $\int_{-1}^{0} f(x)dx$
    \item $\int_{0}^{3} f(x)dx$
    \end{rlist}
    \monades{15}
\item Να βρείτε το εμβαδόν του χωρίου που περικλείεται από την γραφική παράσταση της $f$, τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x = -2$ και $x = 3$.
    \monades{10}
\end{alist}
