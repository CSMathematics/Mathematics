\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνεται η συνάρτηση $f : (0, +\infty) \to \mathbb{R}$ με $f(x) = -2\ln x+1, x > 0$.
\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ αντιστρέφεται. \monades{08}
\item Να βρείτε τη συνάρτηση $f^{-1}$. \monades{09}
\item Δίνεται επιπλέον η συνάρτηση $g$ με τύπο $g(x)=1-\ln x^2$. Να αποδείξετε ότι οι συναρτήσεις $f, g$ δεν είναι ίσες και στη συνέχεια να βρείτε το ευρύτερο υποσύνολο του $\mathbb{R}$ στο οποίο ισχύει $f = g$. \monades{08}
\end{alist}
