Δίνεται η συνάρτηση $f(x) = e^{\ln^2 x}$, $x > 0$.
\begin{alist}
\item Να αποδείξετε ότι η $f$ είναι παραγωγίσιμη στο $(0, +\infty)$ με $f'(x) = 2 \frac{\ln x}{x} f(x)$. \monades{8}
\item Να αποδείξετε ότι η $f$ έχει ολικό ελάχιστο ίσο με 1. \monades{7}
\item Να υπολογίσετε το ολοκλήρωμα $I = \int_1^e \frac{2 \cdot \ln x \cdot f(x) + x e^x}{x(f(x)+e^x)} dx$. \monades{10}
\end{alist}
