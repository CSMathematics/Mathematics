\Askhsh{29832}{2}
\begin{ylh}{1}
    \item 1.2 Συναρτήσεις.
\end{ylh}
Δίνεται οι συναρτήσεις $f(x) = \frac{e^x+1}{e^x-1}$ και $g(x) = \ln \frac{1-x}{1+x}$.
\begin{alist}
\item Να αποδείξετε ότι το πεδίο ορισμού της συνάρτησης $f$ είναι το $\mathbb{R}^*$ και της $g$ το διάστημα $(-1, 1)$. \monades{8}
\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f \circ g$. \monades{8}
\item Να βρείτε τον τύπο της συνάρτησης $(f \circ g)(x)$. \monades{9}
\end{alist}
