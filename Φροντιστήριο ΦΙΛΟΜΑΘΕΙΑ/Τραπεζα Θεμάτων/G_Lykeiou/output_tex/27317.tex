\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνεται η συνάρτηση $f$ με $f(x) = \sqrt{4-x^2}$, $x\in[0,2]$
\begin{alist}
\item Να μελετήσετε την $f$ ως προς τη μονοτονία στο $[0,2]$ \monades{10}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item Το σύνολο τιμών της $f$ είναι το $[0, 2]$. \monades{05}
    \item Ορίζεται η αντίστροφη συνάρτηση $f^{-1}$ της $f$. \monades{03}
    \item Οι συναρτήσεις $f$ και $f^{-1}$ είναι ίσες. \monades{07}
    \end{rlist}
\end{alist}
