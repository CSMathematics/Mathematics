Έστω η συνάρτηση $g:(0,+\infty) \to \mathbb{R}$ με $g(x)=\frac{e}{x}-\ln x$.
\begin{alist}
\item Να μελετήσετε τη συνάρτηση $g$ ως προς τη μονοτονία. \monades{06}
\item Να αποδείξετε ότι η εξίσωση $e(1-x)=x\ln x$ έχει ακριβώς μία λύση την $x=1$. \monades{06}
\item Δίνεται η συνάρτηση $f(x)=\frac{x^2+x}{e-x\ln x-ex}$.
\begin{rlist}
\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f$. \monades{06}
\item Να δείξετε ότι $\lim_{x \to 1^+} f(x)=-\infty$. \monades{07}
\end{rlist}
\end{alist}
