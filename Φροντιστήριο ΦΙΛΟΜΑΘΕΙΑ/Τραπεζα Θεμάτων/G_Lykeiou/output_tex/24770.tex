Δίνεται η συνάρτηση $f(x)=\ln(e^x-1)+x-1, x>0$.
\begin{alist}
    \item Να αποδείξετε ότι είναι γνησίως αύξουσα και κοίλη. \monades{8}
    \item \begin{rlist}
        \item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής της παράστασης στο $x_o=\ln2$. \monades{5}
        \item Να αποδείξετε ότι για κάθε $x>0$ ισχύει $\ln(e^x-1)\le 2x-\ln4$. \monades{4}
    \end{rlist}
    \item Να υπολογίσετε το ολοκλήρωμα $I=\int_{\ln2}^{\ln3} \frac{2-e^{-x}}{e^{-x}-1}dx$. \monades{8}
\end{alist}
