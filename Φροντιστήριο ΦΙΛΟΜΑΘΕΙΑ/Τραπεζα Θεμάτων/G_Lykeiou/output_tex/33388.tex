Δίνεται η συνάρτηση $f(x)=2x+\eta\mu x$ για κάθε $x \in \mathbb{R}$.
\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $f$ αντιστρέφεται. \monades{5}
\item
    \begin{rlist}
    \item Να βρείτε την εξίσωση εφαπτομένης της $C_f$ στο σημείο της $A\left(\frac{\pi}{2},\pi+1\right).$ \monades{7}
    \item Να δείξετε ότι η ευθεία $y=2x+1$ εφάπτεται της $C_f$ σε άπειρα σημεία. \monades{6}
    \end{rlist}
\item Να δείξετε ότι:
    \begin{rlist}
    \item $\left|f'(x)\right|\le 3$ για κάθε $x \in \mathbb{R}.$ \monades{3}
    \item $\left|f(\beta)-f(\alpha)\right|\le 3|\beta-\alpha|$, για κάθε $\alpha,\beta \in \mathbb{R}$ με $\alpha < \beta.$ \monades{4}
    \end{rlist}
\end{alist}
