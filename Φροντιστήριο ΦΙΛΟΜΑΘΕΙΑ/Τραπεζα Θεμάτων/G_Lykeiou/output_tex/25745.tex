\begin{ylh}{1}
    \item 1.8 Συνέχεια συνάρτησης
    \item 2.6 Συνέπειες του Θεωρήματος Μέσης Τιμής
    \item 2.7 Τοπικά ακρότατα συνάρτησης
    \item 2.8 Κυρτότητα - Σημεία καμπής συνάρτησης.
\end{ylh}
Δίνεται συνάρτηση $f : [0,2] \to \mathbb{R}$ η οποία είναι συνεχής στο $[0,2]$, δύο φορές παραγωγίσιμη στο $(0,2)$ και ισχύουν $f(1)=1$, $f'(1)=0$, $f(0)=f(2)$ και
\[ (f'(x))^2 + f(x) \cdot f''(x) < 0, \text{ για κάθε } x \in (0,2) \]
\begin{alist}
\item Να αποδείξετε ότι:
    \begin{rlist}
    \item $f(x) \neq 0$ για κάθε $x \in (0,2)$. \monades{5}
    \item $f(x) > 0$ για κάθε $x \in (0,2)$. \monades{5}
    \end{rlist}
\item Να μελετήσετε την $f$ ως προς την κυρτότητα και τα σημεία καμπής. \monades{7}
\item Να μελετήσετε την $f$ ως προς την μονοτονία και να βρείτε τις θέσεις των ακροτάτων. \monades{8}
\end{alist}
