Έστω $f:(1, +\infty) \to \mathbb{R}$ μια παραγωγίσιμη συνάρτηση ώστε για κάθε $x > 1$ να ισχύει
\[ xf(x)f'(x) = \frac{1}{2} \]
και $f(e)=1$.

\begin{alist}
\item Να αποδείξετε ότι η συνάρτηση $g(x) = f^2(x) - \ln x$, $x > 1$ είναι σταθερή και να βρείτε τον τύπο της f. \monades{9}
\end{alist}
Έστω $f(x) = \sqrt{\ln x}$, $x > 1$.
\begin{alist}
\item Να αποδείξετε ότι η ευθεία που διέρχεται από τα σημεία $A(-e, 0)$ και $B(e, 1)$ εφάπτεται στη γραφική παράσταση της f στο B. \monades{8}
\item Να αποδείξετε ότι για κάθε $x > 1$ ισχύει
\[ \frac{1}{x+1} < f^2(x+1) - f^2(x) < \frac{1}{x} \] \monades{8}
\end{alist}
