Έστω η συνεχής συνάρτηση $f:(0,+\infty) \to \mathbb{R}$ για την οποία ισχύει:
\[ \frac{1}{x} \le f(x) \le 1+\frac{1}{x}, \quad \text{για κάθε } x \in (0,+\infty). \]
\begin{alist}
    \item Να αποδείξετε ότι $\lim_{x \to 0^+} f(x) = +\infty$. \monades{07}
    \item Αν επιπλέον ισχύει $(x+1)f'(x)\cdot \ln(x+1) = -f(x)$, για κάθε $x \in (0,+\infty)$, τότε:
    \begin{rlist}
        \item Να αποδείξετε ότι η συνάρτηση $g(x)=f(x)\cdot\ln(x+1)$, $x>0$ είναι σταθερή. \monades{08}
        \item Να αποδείξετε ότι για κάθε $x \in (0,+\infty)$ ισχύει $\frac{\ln(x+1)}{x} \le g(x) \le \ln(x+1) + \frac{\ln(x+1)}{x}$ και έπειτα να βρείτε τον τύπο της $f$. \monades{10}
    \end{rlist}
\end{alist}
