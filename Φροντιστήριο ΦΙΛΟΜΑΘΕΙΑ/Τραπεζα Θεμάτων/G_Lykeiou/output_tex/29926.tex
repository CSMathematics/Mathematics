\begin{ylh}{1}
    \item 1.2 Συναρτήσεις
    \item 1.3 Μονότονες συναρτήσεις - Αντίστροφη συνάρτηση.
\end{ylh}
Δίνονται οι συναρτήσεις $f$ και $g$ με $f(x) = \ln(x-2) + 5$ για κάθε $x > 2$ και $g(x) = 2x-1$ με $x \in \mathbb{R}$.
\begin{alist}
\item
    \begin{rlist}
    \item Να αποδείξετε ότι η συνάρτηση $g$ είναι αντιστρέψιμη. \monades{6}
    \item Να βρείτε τη συνάρτηση $g^{-1}$. \monades{7}
    \end{rlist}
\item
    \begin{rlist}
    \item Να προσδιορίσετε το πεδίο ορισμού της συνάρτησης $f \circ g^{-1}$. \monades{6}
    \item Να βρείτε τον τύπο της συνάρτησης $f \circ g^{-1}$. \monades{6}
    \end{rlist}
\end{alist}
