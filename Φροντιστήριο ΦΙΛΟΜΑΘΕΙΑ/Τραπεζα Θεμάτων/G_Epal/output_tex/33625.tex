\begin{ylh}{1}
    \item 1.2 Η έννοια της παραγώγου.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x)=2x^2-7x$ με $x\in\mathbb{R}$. Αν η ευθεία $\varepsilon: y = \lambda x + \beta$ είναι εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο της $(2, -6)$, τότε:
\begin{alist}
    \item να δείξετε ότι $f'(x)=4x-7$. \monades{7}
    \item να δείξετε ότι $\lambda = 1$. \monades{9}
    \item να δείξετε ότι $\beta = -8$. \monades{9}
\end{alist}
