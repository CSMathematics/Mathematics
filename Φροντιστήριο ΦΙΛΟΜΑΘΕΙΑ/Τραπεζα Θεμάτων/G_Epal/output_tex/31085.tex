\Askhsh{31085}{2}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2-x}{x}$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού $A$ της $f$. \monades{9}
    \item Να αποδείξετε ότι για κάθε $x \in A$ ισχύει $f(x) = x - 1$. \monades{9}
    \item Να υπολογίσετε το $\displaystyle\lim\limits_{x\to-1} f(x)$. \monades{7}
\end{alist}
