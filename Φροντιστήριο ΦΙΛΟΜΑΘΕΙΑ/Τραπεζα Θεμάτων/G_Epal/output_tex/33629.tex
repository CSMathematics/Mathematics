    \item Όριο - συνέχεια συνάρτησης.
    \item Κανόνες παραγώγισης.

Δίνεται η συνάρτηση $f(x) = \frac{x^2}{x^2 + 1}$ με $x \in \mathbb{R}$.

\begin{alist}
    \item Να βρείτε το $\lim\limits_{x\to\sqrt{2}} f(x)$. \monades{5}

    \item Να δείξετε ότι για κάθε $x \in \mathbb{R}$ η $f'(x) = \frac{2x}{(x^2 + 1)^2}$. \monades{8}

    \item
    \begin{rlist}
        \item Να βρείτε τα διαστήματα στα οποία η συνάρτηση $f$ είναι γνησίως αύξουσα και αυτά στα οποία είναι γνησίως φθίνουσα. \monades{5}

        \item Να βρείτε τα ακρότατα της συνάρτησης $f$. \monades{3}
    \end{rlist}

    \item Να συγκρίνετε τις τιμές $f(2023)$ και $f(2301)$ της συνάρτησης $f$. \monades{4}
\end{alist}
