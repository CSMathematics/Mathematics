    \item 2.2 Παράγωγος Συνάρτησης
    \item 2.7 Τοπικά Ακρότατα Συνάρτησης

Δίνεται η συνάρτηση
\[ f(x) = x^3 + 3x + 6, \quad x \in \mathbb{R} \]
\begin{alist}
    \item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = 3x^2 + 3, \ x \in \mathbb{R}$.
    \monades{08}
    \item Να υπολογίσετε το $f'(0)$.
    \monades{07}
    \item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα.
    \monades{10}
\end{alist}
