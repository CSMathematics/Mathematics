\begin{ylh}{1}
    \item 2.2 Τοπικά ακρότατα συνάρτησης
    \item 2.7 Εφαρμογές των παραγώγων
\end{ylh}
Δίνεται συνάρτηση $f$ με πεδίο ορισμού το $\mathbb{R}$, τέτοια ώστε $f(2)=5$ και $f(0)=6$. Στο παρακάτω σχήμα δίνεται η γραφική παράσταση της παραγώγου $f'$ της συνάρτησης, η οποία είναι ευθεία που τέμνει τον άξονα $x'x$ στο σημείο με τετμημένη $2$ και τον άξονα $y'y$ στο σημείο με τεταγμένη $-1$.

\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xlabel=$x$,
    ylabel=$y$,
    xmin=-9.5, xmax=11.5,
    ymin=-5.5, ymax=5.5,
    xtick={-9,-8,...,11},
    ytick={-5,-4,...,5},
    grid=both,
    grid style={line width=.1pt, draw=gray!30},
    width=14cm, height=8cm,

]
    \addplot[thick, maincolor, domain=-9:11, samples=2] {0.5*x - 1};
    \node[anchor=south west] at (axis cs:7,3) {$y=f'(x)$};
    \addplot[only marks, mark=*, mark size=1.5pt, black] coordinates {(0,-1) (2,0)};
\end{axis}
\end{tikzpicture}
\end{center}

    \monades{8}

    \monades{8}
