Δίνεται η συνάρτηση
\[ f(x) = \sqrt{x^2 + 2x + 5}, x \in \mathbb{R} \]
\begin{alist}
\item Να αποδείξετε ότι
\[ f'(x) = \frac{x + 1}{\sqrt{x^2 + 2x + 5}}, x \in \mathbb{R} \]
\monades{05}

\item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία.
\monades{06}

\item Να αποδείξετε ότι $f(x) \geq 2$ για κάθε $x \in \mathbb{R}$.
\monades{06}

\item Να βρείτε το σημείο της γραφικής παράστασης της $f$ στο οποίο η εφαπτομένη της είναι παράλληλη στον άξονα $x'x$, καθώς και την εξίσωση της εφαπτομένης.
\monades{08}
\end{alist}
