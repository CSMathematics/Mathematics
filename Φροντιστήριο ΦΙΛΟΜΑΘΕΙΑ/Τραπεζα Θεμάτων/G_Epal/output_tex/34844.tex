\begin{ylh}{1}
	\item 2.2 Παρουσίαση Στατιστικών Δεδομένων
\end{ylh}

\begin{center}
\begin{mytblr}{}
{Όγκος $(cm^3)$ $[ - )$} & $\nu_i$ & $f_i$ & $f_i\%$ \\

50-60 & 20 & & \\

60-70 & 10 & & \\

70-80 & 40 & & \\

80-90 & 30 & & \\

Σύνολο & 100 & & \\
\end{mytblr}
\end{center}
\begin{alist}
	\item Να συμπληρώσετε τον παραπάνω πίνακα κατανομής σχετικών συχνοτήτων.
	\monades{12}
	\item Να σχεδιάσετε το ιστόγραμμα συχνοτήτων $\nu_i$ και το αντίστοιχο πολύγωνο συχνοτήτων της κατανομής.
	\monades{13}
\end{alist}
