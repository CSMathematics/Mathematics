\Askhsh{25876}{2}
\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση
\[ f(x) = x^2 - 2x, \ x \in \mathbb{R} \]
\begin{alist}
\item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = 2x - 2, \ x \in \mathbb{R}$.
\monades{09}
\end{alist}
Έστω $y = \lambda x + \beta$ η εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(1, -1)$. Να αποδείξετε ότι:
\begin{alist}\setcounter{enumi}{1}
\item $\lambda = 0$.
\monades{07}
\item Η εξίσωση της ευθείας $(\varepsilon)$ είναι $y = -1$.
\monades{09}
\end{alist}
