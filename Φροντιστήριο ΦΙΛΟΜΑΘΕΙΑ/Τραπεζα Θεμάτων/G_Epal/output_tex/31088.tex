\begin{ylh}{1}
    \item 2.1 Η έννοια της παραγώγου.
    \item 2.3 Κανόνες παραγώγισης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = -3 \cdot \frac{1}{x}$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού $A$ της $f$. \monades{9}
    \item Να αποδείξετε ότι για κάθε $x \in A$ ισχύει: $f'(x) = \frac{3}{x^2}$. \monades{9}
    \item Να υπολογίσετε το $f'(-3)$. \monades{7}
\end{alist}
