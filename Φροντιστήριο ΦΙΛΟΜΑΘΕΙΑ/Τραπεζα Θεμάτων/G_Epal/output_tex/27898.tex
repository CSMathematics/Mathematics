\begin{ylh}{1}
    \item 2.2 Παράγωγος συνάρτησης.
    \item 2.7 Τοπικά ακρότατα συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = -x^2+2x+4$ με $x \in \mathbb{R}$.
\begin{alist}
    \item Να δείξετε ότι $f'(x)=-2x+2$ με $x \in \mathbb{R}$. \monades{7}
    \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία της. \monades{10}
    \item Να δείξετε ότι για κάθε $x \in \mathbb{R}$ ισχύει $f(x) \le 5$. \monades{8}
\end{alist}
