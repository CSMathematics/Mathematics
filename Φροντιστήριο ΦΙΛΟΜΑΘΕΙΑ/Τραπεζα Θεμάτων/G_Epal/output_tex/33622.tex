\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = x^2 - 3x + 4$ με $x \in \R$.
\begin{alist}
    \item Να βρείτε την $f'(x)$. \monades{9}
    \item Να βρείτε την $f''(x)$. \monades{9}
    \item Να βρείτε την τιμή της παράστασης $\Pi = f(0) + f'(0) + f''(0)$. \monades{7}
\end{alist}
