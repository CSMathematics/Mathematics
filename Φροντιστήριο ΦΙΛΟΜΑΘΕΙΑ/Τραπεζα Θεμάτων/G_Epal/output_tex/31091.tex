Δίνεται η συνάρτηση $f(x) = \sqrt{x} - 1$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της συνάρτησης. \monades{8}
    \item Να δείξετε ότι $f'(x) = \frac{1}{2\sqrt{x}}$ για κάθε $x > 0$. \monades{10}
    \item Να υπολογίσετε το $f'(9)$. \monades{7}
\end{alist}
