\Askhsh{24361}{4}
\begin{ylh}{1}
	\item 2.5 Τοπικά ακρότατα συνάρτησης.
	\item 2.6 Μονοτονία συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = x^3 + \lambda x^2 - x, x \in \mathbb{R}$ και $\lambda \in \mathbb{R}$ σταθερά.
\begin{alist}
	\item Αν ισχύει $f'(1) = 0$, να βρείτε την τιμή του $\lambda \in \mathbb{R}$. \monades{7}

	Για $\lambda = -1$
	\item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και να βρείτε το είδος και την τιμή των τοπικών ακροτάτων της. \monades{10}
	\item Να βρείτε τα σημεία της γραφικής παράστασης της συνάρτησης $f$, στα οποία η εφαπτομένη είναι παράλληλη στην ευθεία $y = -x$. \monades{8}
\end{alist}
