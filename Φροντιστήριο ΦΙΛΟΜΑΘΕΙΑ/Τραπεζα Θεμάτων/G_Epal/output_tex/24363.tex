\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
    \item 1.4 Εφαρμογές των Παραγώγων
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{x}{x^2 + 1}$ με $x \in \mathbb{R}$.
\begin{alist}
    \item Να δείξετε ότι η παράγωγος της συνάρτησης $f$ είναι: $f'(x) = \frac{1 - x^2}{(x^2 + 1)^2}$ με $x \in \mathbb{R}$.
    \monades{8}

    \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και να βρείτε το είδος και την τιμή των τοπικών ακροτάτων της.
    \monades{10}

    \item Να δείξετε ότι η εφαπτομένη της γραφικής παράστασης της συνάρτησης $f$ στο σημείο $M(2, f(2))$, είναι η ευθεία $(\varepsilon): y = -\frac{3}{25}x + \frac{16}{25}$.
    \monades{7}
\end{alist}
