\Askhsh{31693}{2}
\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{1}{3} \cdot x^3 + 1$, με $x \in \mathbb{R}$.

\begin{alist}
    \item Να δείξετε ότι $f'(x) = x^2$ για κάθε $x \in \mathbb{R}$. \monades{09}

    Έστω $y = \lambda x + \beta$ η εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A\left(-1, \frac{2}{3}\right)$. Να αποδείξετε ότι:
    \item $\lambda = 1$. \monades{07}
    \item Η εξίσωση της ευθείας $(\varepsilon)$ είναι: $y = x + \frac{5}{3}$. \monades{09}
\end{alist}
