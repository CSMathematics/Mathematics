Δίνεται τετράγωνο ΑΒΓΔ με πλευρά $ΑΒ = 4$. Θεωρούμε τα εσωτερικά σημεία Κ,Λ,Μ και Ν των πλευρών ΑΒ,ΒΓ,ΓΔ και ΔΑ αντίστοιχα έτσι ώστε $ΑΚ = ΒΛ = ΓΜ = ΔΝ = x$.

\begin{alist}
\item Να δείξετε ότι
    \begin{rlist}
        \item το εμβαδόν του ΑΚΝ ως συνάρτηση του $x$ είναι $E_1(x) = \frac{1}{2}(4-x) \cdot x$, $x \in (0,4)$. \monades{5}
        \item το εμβαδόν του ΚΛΜΝ ως συνάρτηση του $x$ είναι $E(x) = 2 \cdot (x^2 - 4x + 8)$, $x \in (0,4)$. \monades{8}
    \end{rlist}
\item Να υπολογίσετε την πρώτη παράγωγο του εμβαδού $E(x)$. \monades{6}
\item Να βρείτε την τιμή του $x$, για την οποία το $E(x)$ γίνεται ελάχιστο. \monades{6}
\end{alist}
