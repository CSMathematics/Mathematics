    \item 2.3 Μέτρα Θέσης και Διασποράς
\end{ylh}
Μια μεταβλητή $X$ παίρνει τις τιμές $x_1=α$, $x_2=α+5$, $x_3=α+10$ και $x_4=α+35$, όπου $α$ πραγματικός αριθμός. Οι αθροιστικές σχετικές συχνότητες των τιμών δίνονται στον παρακάτω πίνακα:
\begin{center}
\begin{mytblr}{}
$x_i$ & $F_i$ \\

$a$ & $\frac{4}{\lambda}$ \\

$a+5$ & $\frac{11}{\lambda}$ \\

$a+10$ & $\frac{18}{\lambda}$ \\

$a+35$ & $\frac{25}{\lambda}$ \\

\end{mytblr}
\end{center}
όπου $\lambda$ θετικός ακέραιος.
\begin{alist}
\item Να αποδείξετε ότι $λ=25$. \monades{9}
\item Να βρείτε τις αθροιστικές σχετικές συχνότητες $F_1, F_2, F_3$ και $F_4$ αλλά και τις σχετικές συχνότητες $f_1, f_2, f_3$ και $f_4$. \monades{8}
\item Αν η μέση τιμή των παρατηρήσεων είναι $\bar{x} = 19$, να βρείτε την τιμή του $α$. \monades{8}
\end{alist}
