    \item 1.4 Εφαρμογές των Παραγώγων

    % Η ευθεία f'(x) = -x + 3
    \addplot[
        domain=-3.5:5,
        samples=2,
        color=maincolor,
        very thick,
    ] {-x+3};
    
    % Ετικέτα γραφικής παράστασης
    \node[above right, inner sep=2pt] at (axis cs:-3, 6) {$C_{f'}$};
    
    % Σημεία τομής με τους άξονες
    \addplot[only marks, mark=*, mark size=1.5pt, black] coordinates {(0,3) (3,0)};
\end{axis}
\end{tikzpicture}
\end{center}
Στο παρακάτω σχήμα δίνεται η γραφική παράσταση $C_{f'}$ της παραγώγου $f'$ μιας συνάρτησης $f$. Αν η $C_{f'}$ είναι ευθεία η οποία τέμνει τον άξονα $x'x$ στο σημείο $(3,0)$, τότε:
\begin{alist}
    \item Να αιτιολογήσετε ότι η $f$ είναι γνησίως αύξουσα στο διάστημα $(-\infty,3]$ και γνησίως φθίνουσα στο διάστημα $[3,+\infty)$. \monades{10}
    \item Να συγκρίνετε τους αριθμούς $f(3)$ και $f(4)$. \monades{06}
    \item Να βρείτε το είδος και την τιμή του ακροτάτου που παρουσιάζει η $f$ αν η γραφική της παράσταση διέρχεται από το σημείο $A(3,2)$. \monades{09}
\end{alist}
