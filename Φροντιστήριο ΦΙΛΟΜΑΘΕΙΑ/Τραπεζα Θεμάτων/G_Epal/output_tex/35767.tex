\begin{ylh}{1}
    \item 2.3 Μέτρα Θέσης και Διασποράς
\end{ylh}
Η μέση τιμή του βάρους έξι μαθητών της Γ' Λυκείου είναι 72 κιλά.
\begin{alist}
    \item Να δείξετε ότι το άθροισμα του βάρους των έξι μαθητών είναι 432 κιλά. \monades{8}
    \item Στην ομάδα των έξι μαθητών προστίθενται άλλοι δύο, με βάρος 77 και 83 κιλά αντίστοιχα. Να βρείτε:
    \begin{rlist}
        \item το άθροισμα των βαρών των οκτώ μαθητών. \monades{8}
        \item τη μέση τιμή του βάρους των οκτώ μαθητών. \monades{9}
    \end{rlist}
\end{alist}
