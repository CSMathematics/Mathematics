\Askhsh{32248}{2}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
\end{ylh}
Δίνονται οι συναρτήσεις $f(x) = \frac{(x+1)(x-1)}{x-2}$ και $g(x) = \frac{x+1}{x-2}$.
\begin{alist}
	\item Να βρείτε τα πεδία ορισμού των συναρτήσεων $f, g$. \monades{8}
	\item Να δείξετε ότι για $x \neq 2$ είναι: $f(x) - g(x) = x+1$. \monades{9}
	\item Να βρείτε το $\lim\limits_{x\to 2}[f(x)-g(x)]$. \monades{8}
\end{alist}
