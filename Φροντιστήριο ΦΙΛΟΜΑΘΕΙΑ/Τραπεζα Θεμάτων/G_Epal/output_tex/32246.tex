\Askhsh{32246}{2}
\begin{ylh}{1}
    \item 2.2 Παρουσίαση Στατιστικών Δεδομένων
\end{ylh}

Στον παρακάτω πίνακα, δίνεται ο αριθμός των μαθητών ενός τμήματος του σχολείου με 25 μαθητές, που απουσίασαν από 0 έως και 5 ημέρες στη διάρκεια του προηγούμενου χειμώνα.

\begin{center}
\begin{mytblr}{}
Ημέρες απουσίας & Αριθμός μαθητών που απουσίασε \\
0 & 10 \\
1 & 2 \\
2 & 2 \\
3 & $κ$ \\
4 & 2 \\
5 & 1 \\
\end{mytblr}
\end{center}

\begin{alist}
    \item Να βρείτε τον αριθμό των μαθητών $κ$, που απουσίασε από το σχολείο 3 ημέρες. \monades{8}
    \item Να βρείτε το ποσοστό των μαθητών που δεν απουσίασε από το σχολείο. \monades{9}
    \item Να βρείτε τον αριθμό των μαθητών που απουσίασε τουλάχιστον 1 ημέρα από το σχολείο. \monades{8}
\end{alist}
