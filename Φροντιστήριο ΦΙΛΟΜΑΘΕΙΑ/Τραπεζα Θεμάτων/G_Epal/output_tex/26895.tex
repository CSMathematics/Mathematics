Δύο θετικοί αριθμοί $x, y$ συνδέονται με τη σχέση $2x + y = 20$.
\begin{alist}
    \item 
    \begin{rlist}
        \item Να δείξετε ότι το γινόμενο των δύο αριθμών, ως συνάρτηση του $x$, δίνεται από τον τύπο $f(x) = -2x^2 + 20x$. \monades{5}
        \item Να βρείτε το πεδίο ορισμού της παραπάνω συνάρτησης. \monades{5}
    \end{rlist}
    \item Να μελετήσετε την συνάρτηση $f$ ως προς τη μονοτονία. \monades{8}
    \item Να βρείτε την τιμή του $x$ ώστε το γινόμενο των δύο αριθμών να γίνει μέγιστο. \monades{7}
\end{alist}
