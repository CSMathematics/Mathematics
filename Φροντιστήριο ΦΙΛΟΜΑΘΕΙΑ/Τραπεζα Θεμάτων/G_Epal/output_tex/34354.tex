\begin{ylh}{1}
    \item 2.2 Παρουσίαση Στατιστικών Δεδομένων
\end{ylh}
Στον επόμενο πίνακα σχετικών συχνοτήτων, φαίνεται η κλιμάκωση των γραπτών βαθμών του συνόλου των μαθητών της Γ' Λυκείου στο μάθημα των Μαθηματικών το προηγούμενο σχολικό έτος.

\begin{center}
    \begin{mytblr}{}
        Κλάσεις $[ - )$ & Σχετική Συχνότητα $f_i\%$ \\
        $0-4$ & $10$ \\
        $4-8$ & $10$ \\
        $8-12$ & \\
        $12-16$ & $35$ \\
        $16-20$ & $15$ \\
    \end{mytblr}
\end{center}

\begin{alist}
    \item Να βρεθεί το ποσοστό των μαθητών που έγραψαν βαθμό μεγαλύτερο ή ίσο του $8$ και μικρότερο του $12$. \monades{9}
    \item Αν οι μαθητές που έγραψαν βαθμό μεγαλύτερο ή ίσο του $12$ είναι $42400$ να υπολογίσετε το πλήθος του συνόλου των μαθητών. \monades{9}
    \item Πόσοι μαθητές έγραψαν βαθμό μεγαλύτερο ή ίσο του $16$; \monades{7}
\end{alist}
