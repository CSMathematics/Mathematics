\begin{ylh}{1}
    \item 2.5 Εφαπτομένη γραφικής παράστασης συνάρτησης.
    \item 2.6 Μονοτονία και ακρότατα συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \dfrac{1}{3}x^3 - x^2 + x + \dfrac{2}{3}, x \in \mathbb{R}$
\begin{alist}
    \item Να βρείτε την παράγωγο $f'$ της συνάρτησης $f$. \monades{06}
    \item Να εξετάσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{09}
    \item Να βρείτε την εξίσωση της εφαπτόμενης ευθείας στη γραφική παράσταση της συνάρτησης $f$ στο σημείο της $A(1,1)$. \monades{10}
\end{alist}
