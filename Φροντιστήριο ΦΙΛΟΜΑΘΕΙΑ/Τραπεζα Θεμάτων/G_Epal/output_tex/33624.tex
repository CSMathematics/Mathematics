	\item 1.1 Συναρτήσεις.

\begin{alist}
	\item Να βρείτε το πεδίο ορισμού της συνάρτησης $f$. \monades{7}
	\item Να βρείτε την τιμή της παράστασης $\Pi = f(0)+3f(2)$. \monades{9}
	\item Να υπολογίσετε το όριο $\lim\limits_{x\to2} f(x)$. \monades{9}
\end{alist}
