\Askhsh{34734}{4}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
    \item 1.3 Παράγωγος συνάρτησης.
    \item 1.4 Εφαπτομένη γραφικής παράστασης συνάρτησης.
\end{ylh}

Δίνεται η συνάρτηση $f(x)=\frac{a}{x}$, $x>0$.

\begin{alist}
    \item Να βρείτε τον αριθμό $a$, ώστε η γραφική παράσταση της συνάρτησης $f$ να διέρχεται από το σημείο $\mathrm{A}(1,4)$.
    \monades{5}
\end{alist}

Στη συνέχεια δίνεται ότι $a=4$.

\begin{alist}[resume]
    \item
    \begin{rlist}
        \item Να υπολογίσετε τον $f'(1)$.
        \monades{6}
        \item Να βρείτε την εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο $\mathrm{A}(1,4)$.
        \monades{7}
    \end{rlist}
    
    \item
    \begin{rlist}
        \item Να βρείτε τα σημεία τομής $\mathrm{K}, \Lambda$ της $(\varepsilon)$ με τους άξονες $x'x, y'y$ αντίστοιχα.
        \monades{3}
        \item Να βρείτε το εμβαδόν του τριγώνου που σχηματίζει η $(\varepsilon)$ με τους άξονες $x'x, y'y$.
        \monades{4}
    \end{rlist}
\end{alist}
