\begin{ylh}{1}
	\item 2.3 Μέτρα Θέσης και Διασποράς
\end{ylh}
Η βαθμολογία δέκα μαθητών σε ένα διαγώνισμα στη Στατιστική ήταν:
\[ 10, 10, 18, 8, 20, 7, 9, 17, 11, 10. \]

\begin{alist}
	\item Να βρείτε το εύρος και τη διάμεσο της βαθμολογίας. \monades{6}
	\item Να δείξετε ότι η μέση τιμή της βαθμολογίας είναι $\bar{x}=12$. \monades{6}
	\item Να δείξετε ότι η διασπορά της βαθμολογίας είναι $s^2=18,8$ και να δείξετε ότι το δείγμα των βαθμών δεν είναι ομοιογενές. Δίνεται ότι: $\sqrt{18,8} \approx 4,3$. \monades{8}
	\item Ένας μαθητής της τάξης ισχυριζόμενος ότι το δείγμα των βαθμών δεν είναι ομοιογενές, διατύπωσε την άποψη ότι αν σε κάθε βαθμό προσθέσουμε τον αριθμό 33, θα προκύψει ένα δείγμα αριθμών (όχι βαθμών) που θα είναι ομοιογενές. Συμφωνείτε με την άποψη του μαθητή; Να αιτιολογήσετε την απάντησή σας. \monades{5}
\end{alist}
