    \item 1.1 Συναρτήσεις.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = 2x + \sqrt{x}$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της συνάρτησης. \monades{8}
    \item Να δείξετε ότι $f'(x) = 2 + \frac{1}{2\sqrt{x}}$ για κάθε $x > 0$. \monades{10}
    \item Να υπολογίσετε το $f'(4)$. \monades{7}
\end{alist}
