\Askhsh{24356}{2}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
\end{ylh}
Δίνεται η συνάρτηση \[ g(x) = \frac{(x-1)\cdot(x-2)}{x^2 - 1} \].
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της συνάρτησης $g(x)$. \monades{9}
    \item Να αποδείξετε ότι η $g(x) = \dfrac{x-2}{x+1}$. \monades{9}
    \item Να υπολογίσετε το όριο $\lim\limits_{x\to1} g(x)$. \monades{7}
\end{alist}
