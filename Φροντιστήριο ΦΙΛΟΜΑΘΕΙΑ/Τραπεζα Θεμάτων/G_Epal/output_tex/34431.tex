    \item 2.2 Παρουσίαση Στατιστικών Δεδομένων

\centering
\begin{tikzpicture}[scale=1]

\node[maincolor, anchor=east] at (225:2.9) {Γάτα};

% Sector 3: Άλλα α3=45, [45, 90]

\node[color=maincolor, anchor=south] at (67.5:2.6) {Άλλα};

Στο παρακάτω κυκλικό διάγραμμα φαίνονται σε μοίρες τα μέτρα των γωνιών $\alpha_1, \alpha_2, \alpha_3, \alpha_4$ τα οποία αντιστοιχούν στους τέσσερις κυκλικούς τομείς, σχετικά με το αγαπημένο κατοικίδιο των μαθητών της Γ΄ λυκείου ενός ΕΠΑΛ.

\begin{alist}
\item Να υπολογίσετε τις σχετικές συχνότητες $f_1, f_3$, αυτών που προτιμούν πουλιά ή άλλα κατοικίδια αντίστοιχα. \monades{12}
\item Αν το σύνολο των μαθητών που ρωτήθηκαν ήταν $\nu = 120$, να βρείτε πόσοι μαθητές της τάξης αυτής προτιμούν τα πουλιά για κατοικίδιο. \monades{13}
\end{alist}
