\Askhsh{25875}{2}
\begin{ylh}{1}
    \item 1.2 Η έννοια της παραγώγου.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση
\[ f(x) = 2x^2 - 1, \ x \in \R \]
\begin{alist}
    \item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = 4x, \ x \in \R$.
    \monades{09}
\end{alist}
Έστω $y = \lambda x + \beta$ η εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(1,1)$. Να αποδείξετε ότι:
\begin{alist}[resume]
    \item $\lambda = 4$.
    \monades{07}
    \item Η εξίσωση της ευθείας $(\varepsilon)$ είναι $y = 4x - 3$.
    \monades{09}
\end{alist}
