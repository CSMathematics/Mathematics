Δίνεται ορθογώνιο ΑΒΓΔ με διαστάσεις $ΑΒ = 4$ και $ΒΓ = 2$. Θεωρούμε τα εσωτερικά σημεία Κ,Λ,Μ και Ν των πλευρών ΑΒ,ΒΓ,ΓΔ και ΔΑ αντίστοιχα έτσι ώστε $ΑΚ = ΒΛ = ΓΜ = ΔΝ = x$.

\begin{alist}
\item Να δείξετε ότι
    \begin{rlist}
        \item το εμβαδόν του ΑΚΝ ως συνάρτηση του x είναι $E_1(x) = \frac{1}{2}(2-x) \cdot x$, $x \in (0,2)$ και το εμβαδόν του ΚΒΛ ως συνάρτηση του x είναι $E_2(x) = \frac{1}{2}(4-x) \cdot x$, $x \in (0,2)$.
        \monades{6}
        \item το εμβαδόν του ΚΛΜΝ ως συνάρτηση του x είναι $E(x)=2 \cdot (x^2 - 3x + 4)$, $x \in (0,2)$.
        \monades{5}
    \end{rlist}

\item Να υπολογίσετε το ρυθμό μεταβολής του εμβαδού Ε ως προς το x όταν $x = \frac{3}{2}$.
\monades{7}

\item Να βρείτε την τιμή του x, για την οποία το E(x) γίνεται ελάχιστο.

\end{alist}
