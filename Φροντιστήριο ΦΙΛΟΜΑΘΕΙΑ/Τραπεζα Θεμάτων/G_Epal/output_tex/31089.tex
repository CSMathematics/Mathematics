\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = 4 \cdot \frac{1}{x} - 1$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού της συνάρτησης. \monades{8}
    \item Να δείξετε ότι $f'(x) = -\frac{4}{x^2}$ για κάθε $x \neq 0$. \monades{10}
    \item Να υπολογίσετε το $f'(-2)$. \monades{7}
\end{alist}
