\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{1}{3}x^3 - 2023, x \in \R$.

\begin{alist}
    \item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = x^2, x \in \R$. \monades{8}
    \item Να υπολογίσετε το $f'(-1)$. \monades{7}
    \item Να αποδείξετε ότι: $f'(-1) + f'(1) = 2$. \monades{10}
\end{alist}
