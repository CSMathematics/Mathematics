Δίνεται η συνάρτηση
\[ f(x) = 2x^3 - x^2 + x - 1, x \in \mathbb{R} \]
\begin{alist}
\item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{10}
\item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής παράστασης της συνάρτησης $g(x) = 6x^2 - 2x + 1, x \in \mathbb{R}$, στο σημείο της με τετμημένη $x_0 = 1$. \monades{08}
\item Να υπολογίσετε το όριο $\displaystyle \lim\limits_{x \to 1} \frac{f'(x) - 5}{g'(x) - 10}$. \monades{07}
\end{alist}
