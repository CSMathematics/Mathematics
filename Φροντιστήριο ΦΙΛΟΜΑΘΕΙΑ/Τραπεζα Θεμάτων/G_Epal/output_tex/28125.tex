\begin{ylh}{1}
	\item 1.3 Παράγωγος συνάρτησης.
	\item 1.4 Εφαρμογές των Παραγώγων
\end{ylh}
Δίνονται οι συναρτήσεις $f(x) = x^2 - 8x + 18$ και $g(x) = -(x-4)^4 + 2$.
\begin{alist}
	\item Να βρείτε τις παραγώγους των συναρτήσεων $f$ και $g$. \monades{8}
	\item Να βρείτε τα ακρότατα των συναρτήσεων αυτών. \monades{12}
	\item Να δείξετε ότι $f(x) \geq g(x), x \in \mathbb{R}$. \monades{5}
\end{alist}
