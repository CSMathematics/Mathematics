\begin{ylh}{1}
    \item 2.3 Μέτρα Θέσης και Διασποράς
\end{ylh}
Οι υπάλληλοι μιας μεγάλης εταιρείας πετρελαιοειδών, χρησιμοποιούν το καφέ και το εστιατόριο της εταιρείας καθημερινά. Αν τα χρήματα που ξοδεύουν ακολουθούν περίπου κανονική κατανομή με μέση τιμή $\bar{x} = 10$ ευρώ και συντελεστή μεταβολής $CV = 0,2$ τότε:
\begin{alist}
    \item να δείξετε ότι η τυπική απόκλιση της κατανομής των χρημάτων που ξοδεύουν οι υπάλληλοι της εταιρείας καθημερινά, είναι $s = 2$ ευρώ. \monades{9}
    \item 
    \begin{rlist}
        \item να βρείτε το ποσοστό των υπαλλήλων της εταιρείας που ξοδεύει καθημερινά το πολύ 10 ευρώ. \monades{8}
        \item να βρείτε το ποσοστό των υπαλλήλων της εταιρείας που ξοδεύει καθημερινά περισσότερα από 14 ευρώ. \monades{8}
    \end{rlist}
\end{alist}
