\Askhsh{34801}{4}
\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
    \item 1.7 Ακρότατα συνάρτησης.
\end{ylh}
Η θερμοκρασία σε βαθμούς Κελσίου ($^{\circ}$C) ενός δωματίου σε συνάρτηση με τον χρόνο $t$ σε ώρες δίνεται από τον τύπο $\theta(t) = t^3 - 3t^2 + 2t + c$, $0 \le t \le 4$, όπου $c$ ένας πραγματικός αριθμός.

\begin{alist}
\item Αν είναι γνωστό ότι η θερμοκρασία του δωματίου τη χρονική στιγμή $t=0$ είναι ίση με 8 βαθμούς Κελσίου, να βρείτε την τιμή του πραγματικού αριθμού $c$. \monades{06}
\end{alist}
Για $c=8$
\begin{alist}[resume]
\setcounter{enumi}{1}
\item Να βρείτε τον ρυθμό μεταβολής της θερμοκρασίας όταν $t=2$. \monades{09}
\item Να βρείτε τη χρονική στιγμή κατά την οποία ο ρυθμός μεταβολής της θερμοκρασίας γίνεται ελάχιστος και την τιμή του ελάχιστου ρυθμού μεταβολής. \monades{10}
\end{alist}
