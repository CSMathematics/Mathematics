	\item 1.1 Συναρτήσεις.
	\item 1.3 Παράγωγος συνάρτησης.
	\item 1.4 Εφαρμογές των Παραγώγων.
\end{ylh}
Ένα οικόπεδο σχήματος ορθογωνίου έχει μήκος $x$ μέτρα ($m$), πλάτος $y$ μέτρα ($m$) και εμβαδό 400 τετραγωνικά μέτρα ($m^2$).
\begin{alist}
	\item Να αποδείξετε ότι η περίμετρος του ορθογωνίου οικοπέδου ως συνάρτηση του $x$ δίνεται από τον τύπο $\Pi(x) = 2x + \dfrac{800}{x}$ με $x > 0$. \monades{8}
	\item Να μελετήσετε τη συνάρτηση $\Pi(x)$ ως προς τη μονοτονία της. \monades{9}
	\item Για ποια τιμή του $x$ η περίμετρος του οικοπέδου γίνεται ελάχιστη, και ποια είναι η ελάχιστη τιμή της; \monades{8}
\end{alist}
