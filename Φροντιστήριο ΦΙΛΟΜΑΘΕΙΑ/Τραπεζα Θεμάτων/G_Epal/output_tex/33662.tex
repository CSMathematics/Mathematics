	\item 1.3 Παράγωγος συνάρτησης.
	\item 1.4 Εφαρμογές των Παραγώγων
\end{ylh}
Δίνεται η συνάρτηση $f(x) = x^4 - 2x^2 + 2, x > 0$.
\begin{alist}
	\item Να βρείτε την παράγωγο συνάρτηση $f'(x)$ και το πρόσημο της $f'$. \monades{9}
	\item Να βρείτε τα ακρότατα της συνάρτησης f. \monades{9}
	\item Να αποδείξετε ότι $f(x) > 0$ για κάθε πραγματικό αριθμό $x > 0$. \monades{7}
\end{alist}
