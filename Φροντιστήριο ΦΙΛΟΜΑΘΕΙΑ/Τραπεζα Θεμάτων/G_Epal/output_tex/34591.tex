\Askhsh{34591}{2}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
    \item 2.1 Όριο συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση
\[ f(x) = \frac{x^2 - 3x + 2}{x-1} \]
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού $A$ της $f$. \monades{09}
    \item Να αποδείξετε ότι για κάθε $x \in A$ ισχύει:
    \[ f(x) = x-2 \]
    \monades{09}
    \item Να υπολογίσετε το $\lim\limits_{x \to 1} f(x)$. \monades{07}
\end{alist}
