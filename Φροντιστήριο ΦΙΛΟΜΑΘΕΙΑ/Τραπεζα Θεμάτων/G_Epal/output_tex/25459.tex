\begin{ylh}{2}
\item 1.3 Παράγωγος συνάρτησης
\item 1.4 Εφαρμογές των Παραγώγων
\end{ylh}
Δίνεται η συνάρτηση
\[ f(x) = 3x^2 - 6x + 2023, \ x \in \mathbb{R} \]
\begin{alist}
    \item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = 6x - 6$, $x \in \mathbb{R}$. \monades{08}
    \item Να υπολογίσετε το $f'(1)$. \monades{07}
    \item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα, για $x > 1$. \monades{10}
\end{alist}
