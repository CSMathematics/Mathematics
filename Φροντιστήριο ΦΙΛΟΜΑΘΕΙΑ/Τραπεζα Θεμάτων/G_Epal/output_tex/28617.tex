\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
    \item 1.3 Παράγωγος συνάρτησης.
    \item 1.4 Όριο συνάρτησης στο $x_0$.
\end{ylh}
Δίνεται συνάρτηση $f$ με τύπο $f(x)=ax^2-2x, x \in [0,+\infty)$, όπου $a$ ένας πραγματικός αριθμός.
\begin{alist}
    \item Αν ισχύει $f'(1)=0$ να αποδείξετε ότι $a=1$. \monades{06}
    \item Δίνεται επιπλέον η συνάρτηση $g$ με τύπο $g(x)=\sqrt{x}-\sqrt{2}$.
    \begin{rlist}
        \item Να βρείτε το πεδίο ορισμού της συνάρτησης $g$. \monades{03}
        \item Να λύσετε την εξίσωση $g(x)=0$. \monades{03}
    \end{rlist}
    \item Για $a=1$
    \begin{rlist}
        \item Να προσδιορίσετε τη συνάρτηση $h(x) = \frac{f}{g}(x)$. \monades{05}
        \item Να υπολογίσετε το όριο $\lim\limits_{x \to 2} h(x)$. \monades{08}
    \end{rlist}
\end{alist}
