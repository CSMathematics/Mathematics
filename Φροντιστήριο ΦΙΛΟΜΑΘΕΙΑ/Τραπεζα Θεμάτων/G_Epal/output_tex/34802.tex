\begin{ylh}{1}
	\item 1.3 Παράγωγος συνάρτησης.
	\item 1.4 Εφαρμογές των Παραγώγων.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = x^2 - 4x + c$, $x \in \mathbb{R}$, όπου $c$ ένας πραγματικός αριθμός.
\begin{alist}
	\item Να βρείτε το $c$ αν ισχύει $f(2) + f'(2) + f''(2023) = 0$. \monades{10}
	
	Για $c=2$
	
	\item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{09}
	
	\item Να συγκρίνετε τους αριθμούς $f(2023)$, $f(2024)$. \monades{06}
\end{alist}
