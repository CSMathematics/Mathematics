\Askhsh{31086}{2}
\begin{ylh}{1}
    \item 1.1 Συναρτήσεις.
\end{ylh}
Δίνεται η συνάρτηση: \[f(x) = \frac{x^2-2x}{x}.\]
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού $Α$ της $f$. \monades{9}
    \item Να αποδείξετε ότι για κάθε $x \in A$ ισχύει: $f(x) = x - 2$. \monades{9}
    \item Να υπολογίσετε το $\lim\limits_{x\to0} f(x)$. \monades{7}
\end{alist}
