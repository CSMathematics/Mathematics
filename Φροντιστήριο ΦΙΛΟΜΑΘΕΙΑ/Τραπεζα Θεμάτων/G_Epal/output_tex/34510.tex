\begin{ylh}{1}
    \item 1.2 Η έννοια της παραγώγου.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \frac{x+1}{x-2}, x \in \mathbb{R}-\{2\}$.

\begin{alist}
    \item Να δείξετε ότι η παράγωγος της συνάρτησης είναι $f'(x) = -\frac{3}{(x-2)^2}, x \in \mathbb{R}-\{2\}$. \monades{5}
    
    \item 
    \begin{rlist}
        \item Να βρείτε την εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της συνάρτησης $f$ στο σημείο της $A(3,4)$. \monades{8}
        
        \item Να δείξετε ότι η εφαπτομένη $(\varepsilon)$ διέρχεται από το σημείο $B(1,10)$. \monades{6}
    \end{rlist}
    
    \item Αν η εφαπτομένη $(\eta)$ της γραφικής παράστασης της $f$ στο σημείο της $\Gamma(x_0, f(x_0))$ έχει εξίσωση $y = f'(x_0)x$, να βρείτε το $x_0$. \monades{6}
\end{alist}
