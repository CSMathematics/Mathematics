    \item 2.2 Παράγωγος συνάρτησης.
    \item 2.7 Τοπικά ακρότατα συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση
\[ f(x) = 3x^2 - 6x + 2023, \ x \in \mathbb{R} \]
\begin{alist}
\item Να αποδείξετε ότι η παράγωγος της συνάρτησης $f$ είναι $f'(x) = 6x - 6, \ x \in \mathbb{R}$. \monades{05}
\item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία. \monades{10}
\item Να αποδείξετε ότι $f(x) \geq 2020$ για κάθε $x \in \mathbb{R}$. \monades{10}
\end{alist}
