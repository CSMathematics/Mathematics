\begin{ylh}{1}
    \item 1.3 Παράγωγος συνάρτησης.
    \item 1.4 Εφαρμογές των Παραγώγων.
\end{ylh}
Δίνεται η συνάρτηση $f(x) = x^3 - 2x^2 + x$, όπου $x \in \mathbb{R}$.
\begin{alist}
    \item 
    \begin{rlist}
        \item Για $x \in \mathbb{R}$, να βρείτε την παράγωγο της συνάρτησης $f$. \monades{5}
        \item Να βρείτε τον συντελεστή διεύθυνσης της εφαπτομένης της γραφικής παράστασης της συνάρτησης $f$ στο σημείο $M(2, f(2))$. \monades{5}
    \end{rlist}
    
    \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{9}
    
    \item Να βρείτε τα σημεία της γραφικής παράστασης της συνάρτησης $f$ στα οποία οι εφαπτόμενες είναι παράλληλες στον άξονα $x'x$. \monades{6}
\end{alist}
