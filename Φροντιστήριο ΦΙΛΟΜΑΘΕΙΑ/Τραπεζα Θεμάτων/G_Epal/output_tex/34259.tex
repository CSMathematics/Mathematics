    \item 2.2 Παράγωγος συνάρτησης.
    \item 2.4 Εφαπτομένη γραφικής παράστασης συνάρτησης.
    \item 2.7 Μονοτονία και ακρότατα συνάρτησης.
    \item 1.8 Όρια συνάρτησης στο $x_0 \in \mathbb{R}$.
\end{ylh}
Δίνεται συνάρτηση $f$ με τύπο $f(x) = ax^2 - 2x, x \in \mathbb{R}$, όπου $a$ ένας πραγματικός αριθμός.
\begin{alist}
    \item 
    \begin{rlist}
        \item Να βρείτε την παράγωγο $f'$ της συνάρτησης $f$. \monades{04}
        \item Αν η εφαπτομένη στη γραφική παράσταση της $f$ στο σημείο της $A(1, f(1))$ είναι παράλληλη στον άξονα $x'x$, να αποδείξετε ότι $a=1$. \monades{05}
    \end{rlist}
Για $a=1$
    \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα ακρότατα. \monades{09}
    \item Να υπολογίσετε το όριο $\displaystyle\lim\limits_{x\to1} \frac{f'(x)}{\sqrt{x} - 1}$. \monades{07}
\end{alist}
