\begin{ylh}{1}
	\item 2.3 Μέτρα Θέσης και Διασποράς
\end{ylh}
Οι προφορικοί βαθμοί έξι μαθητών στα μαθηματικά είναι $10, 13, 11, 15, 9, 14$.
\begin{alist}
	\item Να υπολογίσετε την μέση τιμή $\bar{x}$ του δείγματος των έξι μαθητών. \monades{10}
	\item 
	\begin{rlist}
		\item Να μεταφέρετε στην κόλλα σας και να συμπληρώσετε τον παρακάτω πίνακα \monades{5}
		      
		      \begin{center}
\begin{mytblr}{}
			      $t_i$ & $t_i - \bar{x}$ & $(t_i - \bar{x})^2$ \\
			      9     &                 &                    \\
			      10    &                 &                    \\
			      11    &                 &                    \\
			      13    &                 &                    \\
			      14    &                 &                    \\
			      15    &                 &                    \\
		      \end{mytblr}
\end{center}
		      
		\item Να υπολογίσετε τη διακύμανση $s^2$ του δείγματος. \monades{10}
	\end{rlist}
\end{alist}
