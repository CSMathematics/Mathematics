\begin{ylh}{2}
\item 1.1 Συναρτήσεις
\end{ylh}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2-4}{x-2}$.
\begin{alist}
    \item Να βρείτε το πεδίο ορισμού Α της συνάρτησης. \monades{7}
    \item Να δείξετε ότι $f(x) = x + 2$ για κάθε $x$ του πεδίου ορισμού Α. \monades{8}
    \item Να υπολογίσετε το όριο $\lim\limits_{x \to 2} f(x)$. \monades{10}
\end{alist}
