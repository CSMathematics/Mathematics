\Askhsh{27843}{2}
\begin{ylh}{1}
    \item 1.2 Η έννοια της παραγώγου.
    \item 1.3 Παράγωγος συνάρτησης.
\end{ylh}
Δίνεται η συνάρτηση $f(x)=-2x^3+x^2-4$.
\begin{alist}
    \item Να αποδείξετε ότι η παράγωγος της συνάρτησης είναι $f'(x)=-6x^2+2x$. \monades{8}
\end{alist}
Έστω $y=\lambda x+\beta$ η εξίσωση της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $Α(-1,-1)$. Να αποδείξετε ότι:
\begin{alist}
    \item $\lambda=-8$. \monades{7}
    \item Η εξίσωση της ευθείας $(\varepsilon)$ είναι: $y=-8x-9$. \monades{10}
\end{alist}
