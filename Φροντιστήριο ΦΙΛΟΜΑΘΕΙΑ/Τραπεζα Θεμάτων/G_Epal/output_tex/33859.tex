\Askhsh{33859}{4}
\begin{ylh}{2}
    \item 2.1 Συναρτήσεις.
    \item 2.7 Μονοτονία - Ακρότατα συνάρτησης.
\end{ylh}
Ένα οικόπεδο σχήματος ορθογωνίου έχει μήκος $x$ μέτρα, πλάτος $y$ μέτρα και περίμετρο 200 μέτρα.
\begin{alist}
    \item Να αποδείξετε ότι το εμβαδόν του οικοπέδου ως συνάρτηση του $x$ δίνεται από τον τύπο
    $E(x) = 100x - x^2$ και να βρείτε το πεδίο ορισμού της συνάρτησης $E(x)$.
    \monades{10}
    
    \item Να μελετήσετε τη συνάρτηση $E(x)$ ως προς τη μονοτονία της.
    \monades{9}
    
    \item Για ποια τιμή του $x$ το εμβαδόν του οικοπέδου γίνεται μέγιστο, και ποια είναι η μέγιστη τιμή του; Για την τιμή που βρήκατε τι σχήμα προκύπτει;
    \monades{6}
\end{alist}
