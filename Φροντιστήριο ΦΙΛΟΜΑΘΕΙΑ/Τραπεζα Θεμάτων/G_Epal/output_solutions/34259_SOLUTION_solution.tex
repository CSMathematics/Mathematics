\begin{alist}
    \item 
    \begin{rlist}
        \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $f'(x)=(ax^2-2x)'=2ax-2$.
        \item Η εφαπτομένη στη γραφική παράσταση της $f$ στο σημείο της $A(1, f(1))$ είναι παράλληλη στον άξονα $x'x$. Επομένως ισχύει $f'(1)=0 \Leftrightarrow 2a \cdot 1 - 2 = 0 \Leftrightarrow 2a = 2 \Leftrightarrow a=1$.
    \end{rlist}

    \item Για $a=1$ έχουμε $f(x) = x^2-2x$ και $f'(x)=2x-2$.

    Έχουμε $f'(x)=0 \Leftrightarrow 2x-2=0 \Leftrightarrow 2x=2 \Leftrightarrow x=1$ και επιπλέον
    \[ f'(x)>0 \Leftrightarrow 2x-2>0 \Leftrightarrow 2x>2 \Leftrightarrow x>1. \]
    Το πρόσημο της $f'$ και η μονοτονία της $f$ φαίνονται στον παρακάτω πίνακα. Επομένως η $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, 1]$ και γνησίως αύξουσα στο διάστημα $[1, +\infty)$, ενώ παρουσιάζει ελάχιστο για $x=1$ το $f(1)=-1$.

    \item Για $a=1$ έχουμε
    \[ \lim\limits_{x\to 1} \frac{f'(x)}{\sqrt{x}-1} = \lim\limits_{x\to 1} \frac{2x-2}{\sqrt{x}-1} = \lim\limits_{x\to 1} \frac{2(x-1)\cdot(\sqrt{x}+1)}{(\sqrt{x}-1)\cdot(\sqrt{x}+1)} = \lim\limits_{x\to 1} \frac{2(x-1)\cdot(\sqrt{x}+1)}{x-1} = \lim\limits_{x\to 1} 2(\sqrt{x}+1)=4 \]
\end{alist}
