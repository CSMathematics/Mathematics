\lysh{25876}{2}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη για κάθε $x \in \mathbb{R}$ με παράγωγο
\[ f'(x) = (x^2 - 2x)' = (x^2)' - (2x)' = 2x - 2 \cdot 1 = 2x - 2 \]

\item Ο συντελεστής διεύθυνσης $\lambda$ της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(1, -1)$ είναι ίσος με:
\[ \lambda = f'(1) = 2 \cdot 1 - 2 = 0 \]

\item Η εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x_0 = 1$ είναι:
\[ y = \lambda x + \beta \Leftrightarrow y = \beta \]
Επειδή το σημείο $A(1, -1)$ ανήκει στην εφαπτομένη, έχουμε:
\[ -1 = \beta \]
Άρα, η εξίσωση της εφαπτομένης είναι:
\[ y = -1 \]
\end{alist}
