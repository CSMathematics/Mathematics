\lysh{32248}{2}
\begin{alist}
    \item Για τις δύο συναρτήσεις πρέπει το $x-2 \neq 0 \Leftrightarrow x \neq 2$, άρα το $A_f = A_g = \mathbb{R} - \{2\}$.
    \item Για $x \neq 2$ είναι:
    \[ f(x)-g(x) = \frac{(x+1)(x-1)}{x-2} - \frac{x+1}{x-2} = \frac{(x+1)(x-1)-(x+1)}{x-2} = \frac{(x+1)(x-2)}{x-2} = x+1. \]
    \item Λόγω του ερωτήματος (β) είναι:
    \[ \lim\limits_{x \to 2}[f(x)-g(x)] = \lim\limits_{x \to 2}(x+1) = 3. \]
\end{alist}
