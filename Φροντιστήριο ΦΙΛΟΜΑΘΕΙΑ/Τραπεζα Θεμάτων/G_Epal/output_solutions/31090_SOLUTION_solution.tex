\lysh{31090}{2}
\begin{alist}
\item Το πεδίο ορισμού της συνάρτησης $f$ είναι το σύνολο που περιέχει τις τιμές για τις οποίες ορίζεται το κλάσμα, δηλαδή για $x \neq 0$. Συνεπώς $D_f = \mathbb{R} - \{0\}$.
\item Βρίσκουμε την πρώτη παράγωγο της συνάρτησης.
Παραγωγίζουμε τη συνάρτηση ως άθροισμα δύο συναρτήσεων:
\[ f'(x) = \left(2 \cdot \frac{1}{x}\right)' - (1)' = 2 \cdot \left(\frac{1}{x}\right)' - 0 = 2 \cdot \left(-\frac{1}{x^2}\right) = -\frac{2}{x^2} \]
Άρα, $f'(x) = -\frac{2}{x^2}$ για κάθε $x \in \mathbb{R} - \{0\}$.
\item Από το β) ερώτημα έχουμε $f'(x) = -\frac{2}{x^2}$.
Άρα, για $x=-2$ έχουμε
\[ f'(-2) = -\frac{2}{(-2)^2} = -\frac{2}{4} = -\frac{1}{2} \]
Επίσης, για $x=2$ έχουμε
\[ f(2) = \frac{2}{2} - 1 = 1 - 1 = 0 \]
Άρα: $f'(-2) + f(2) = -\frac{1}{2} + 0 = -\frac{1}{2}$.
\end{alist}
