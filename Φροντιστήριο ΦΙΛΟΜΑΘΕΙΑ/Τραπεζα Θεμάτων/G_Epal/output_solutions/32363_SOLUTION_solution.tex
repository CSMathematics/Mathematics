\lysh{32363}{3}
\begin{alist}
    \item Η ταχύτητα $v(t)$ του σημείου σε χρόνο $t$ είναι:
    \[ v(t) = x'(t) = (t^2 - 4t - 1)' = (t^2)' - (4t)' - (1)' = 2t - 4 \]
    \item Η ταχύτητα του σημείου τη χρονική στιγμή $t_1 = 1$ s είναι:
    \[ v(t_1) = v(1) = 2 \cdot 1 - 4 = -2 \text{ m/s} \]
    Η ταχύτητα του σημείου τη χρονική στιγμή $t_2 = 3$ s είναι:
    \[ v(t_2) = v(3) = 2 \cdot 3 - 4 = 2 \text{ m/s} \]
    \item Το σημείο είναι στιγμιαία ακίνητο όταν $v(t) = 0$, δηλαδή όταν
    \[ 2t - 4 = 0 \Leftrightarrow 2t = 4 \Leftrightarrow t = 2 \text{ s} \]
    Επομένως, το σημείο είναι στιγμιαία ακίνητο τη χρονική στιγμή $t = 2$ s.
    \item[δ)] Το σημείο κινείται στη θετική κατεύθυνση όταν $v(t) > 0$, δηλαδή όταν
    \[ 2t - 4 > 0 \Leftrightarrow 2t > 4 \Leftrightarrow t > 2 \]
    Άρα, το σημείο κινείται στη θετική κατεύθυνση στο χρονικό διάστημα $t > 2$ (και στην αρνητική κατεύθυνση όταν $0 \le t < 2$).
\end{alist}
