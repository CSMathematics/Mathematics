\lysh{34258}{2}
\begin{alist}
    \item  Για τη μέση τιμή $\bar{x}$ των βαρών ισχύει $\bar{x} = \frac{\sum t_i}{\nu}$, επομένως
    \[
        \bar{x} = \frac{76+74+75+75+78+72+70+80+75+75}{10} = \frac{750}{10} = 75 \text{ kg.}
    \]
    \item  Ισχύει ότι $CV = \frac{s}{\bar{x}} = \frac{3 \text{ kg}}{75 \text{ kg}} = 0,04 = 4\%$.
    \item  Το δείγμα είναι ομοιογενές γιατί ο συντελεστής μεταβλητότητας του δεν ξεπερνάει το 10\%.
\end{alist}
