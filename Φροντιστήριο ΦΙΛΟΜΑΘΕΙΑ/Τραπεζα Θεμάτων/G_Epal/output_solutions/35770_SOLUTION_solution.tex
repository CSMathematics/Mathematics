\lysh{35770}{2}

\begin{alist}
    \item Ο αριθμός των κρεβατιών για τα 15 δωμάτια σε αύξουσα σειρά είναι:
    $1, 2, 2, 3, 3, 3, 3, 3, \dots, 4$ οπότε η διάμεσος είναι η μεσαία παρατήρηση που είναι η 8\textsuperscript{η}.
    Δηλαδή $\delta = t_8 = 3$ κρεβάτια.

    \item Είναι:
    \begin{center}
    \begin{mytblr}{}
        \shortstack{\textbf{Αριθμός κρεβατιών} \\ \textbf{\boldmath{$x_i$}}} & \shortstack{\textbf{Αριθμός δωματίων} \\ \textbf{\boldmath{$\nu_i$}}} & \textbf{\boldmath{$x_i \cdot \nu_i$}} \\
        
        1 & 1 & 1 \\
        2 & 2 & 4 \\
        3 & 8 & 24 \\
        4 & 4 & 16 \\
        
        \textbf{ΣΥΝΟΛΟ} & \textbf{15} & \textbf{45} \\
    \end{mytblr}
    \end{center}
    Οπότε συνολικά τα κρεβάτια του ξενοδοχείου είναι 45.

    \item Η ζητούμενη μέση τιμή του αριθμού των κρεβατιών ανά δωμάτιο είναι:
    \[
    \bar{x} = \frac{\sum_{i=1}^{\nu} x_i \cdot \nu_i}{\nu} = \frac{45}{15} = 3 \text{ κρεβάτια.}
    \]
\end{alist}
