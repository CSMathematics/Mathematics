\lysh{35092}{4}

Δίνεται η λύση μιας άσκησης Στατιστικής.
\begin{alist}
    \item
	\begin{rlist}
\item Η κεντρική τιμή μιας κλάσης είναι το ημιάθροισμα των άκρων της, οπότε προκύπτει ο πίνακας:
    \begin{center}
\begin{mytblr}{}
      {Κλάσεις \\ $[ - )$} & {Κεντρικές \\ τιμές \\ $x_i$} & $\nu_i$ & $\nu_i x_i$ \\
      $30-40$ & $35$ & $\alpha$ & $35\alpha$ \\
      $40-50$ & $45$ & $2\alpha$ & $90\alpha$ \\
      $50-60$ & $55$ & $3\alpha$ & $165\alpha$ \\
      $60-70$ & $65$ & $4\alpha$ & $260\alpha$ \\
      Σύνολο & \SetCell{bg=gray!30} & $10\alpha$ & $550\alpha$ \\
    \end{mytblr}
\end{center}
    \item Η μέση τιμή είναι $\bar{x} = \dfrac{\sum x_i \nu_i}{\nu} = \dfrac{550\alpha}{10\alpha} = 55$.
\end{rlist}
    \item
\begin{rlist}
\item Με την βοήθεια του α) ερωτήματος συμπληρώνουμε τον πίνακα:
    
    \begin{center}
\begin{mytblr}{}
      {Κλάσεις \\ $[ - )$} & {Κεντρικές \\ τιμές \\ $x_i$} & $\nu_i$ & $x_i - \bar{x}$ & $(x_i - \bar{x})^2$ & $(x_i - \bar{x})^2 \nu_i$ \\
      $30-40$ & $35$ & $\alpha$ & $-20$ & $400$ & $400\alpha$ \\
      $40-50$ & $45$ & $2\alpha$ & $-10$ & $100$ & $200\alpha$ \\
      $50-60$ & $55$ & $3\alpha$ & $0$ & $0$ & $0$ \\
      $60-70$ & $65$ & $4\alpha$ & $10$ & $100$ & $400\alpha$ \\
      Σύνολο & \SetCell{bg=gray!30} & $10\alpha$ & \SetCell{bg=gray!30} & \SetCell{bg=gray!30} & $1000\alpha$ \\
    \end{mytblr}
\end{center}
    
    \item Έχουμε $s^2 = \dfrac{1}{\nu}\sum_{i=1}^{4}(x_i - \bar{x})^2 \nu_i = \dfrac{1}{10\alpha}1000\alpha = 100$.

    \item
    Παρατηρούμε ότι η καινούργια κατανομή προκύπτει από την κατανομή του ερωτήματος α) αν θέσουμε $\alpha=10$.
    \par
    Οπότε από τα προηγούμενα ερωτήματα και για την νέα κατανομή έχουμε μέση τιμή $\bar{x}=55$ και διακύμανση $s^2=100$.
\end{rlist}
\end{alist}
