\lysh{31091}{2}
\begin{alist}
    \item Το πεδίο ορισμού της συνάρτησης $f$ είναι το σύνολο που περιέχει τις τιμές για τις οποίες ορίζεται η ρίζα, δηλαδή για $x \ge 0$. Συνεπώς $D_f = [0, +\infty)$.
    
    \item Βρίσκουμε την πρώτη παράγωγο της συνάρτησης.
    
    Παραγωγίζουμε τη συνάρτηση ως άθροισμα δύο συναρτήσεων:
    \[
        f'(x) = (\sqrt{x})' - (1)' = \frac{1}{2\sqrt{x}} - 0 = \frac{1}{2\sqrt{x}}.
    \]
    Άρα, $f'(x) = \frac{1}{2\sqrt{x}}$ για κάθε $x > 0$.
    
    \item Από το β) ερώτημα έχουμε $f'(x) = \frac{1}{2\sqrt{x}}$.
    
    Άρα, για $x=9$ έχουμε $f'(9) = \frac{1}{2\sqrt{9}} = \frac{1}{2 \cdot 3} = \frac{1}{6}$.
\end{alist}
