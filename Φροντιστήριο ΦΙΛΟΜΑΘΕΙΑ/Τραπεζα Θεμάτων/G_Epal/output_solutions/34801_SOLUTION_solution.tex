\lysh{34801}{4}
\begin{alist}
    \item Ισχύει $\theta(0)=8 \Leftrightarrow 0^3-3 \cdot 0^2+2 \cdot 0+c=8 \Leftrightarrow c=8$.
    
    \item Για $c=8$ η συνάρτηση γίνεται $\theta(t)=t^3-3t^2+2t+8$ και ο ρυθμός μεταβολής της θερμοκρασίας $\theta$ τη χρονική στιγμή $t$ είναι $f(t)=\theta'(t)=(t^3-3t^2+2t+8)'=3t^2-6t+2$.
    Επομένως τη χρονική στιγμή $t=2$ ο ρυθμός μεταβολής είναι
    \[ f(2)=\theta'(2)=3 \cdot 2^2-6 \cdot 2+2=2 \, \text{°C/ώρα}. \]
    
    \item Για τον ρυθμό μεταβολής $f(t)=3t^2-6t+2$ έχουμε $f'(t)=(3t^2-6t+2)'=6t-6$, άρα
    \[ f'(t)=0 \Leftrightarrow 6t-6=0 \Leftrightarrow 6t=6 \Leftrightarrow t=1 \]
    και
    \[ f'(t)>0 \Leftrightarrow 6t-6>0 \Leftrightarrow 6t>6 \Leftrightarrow t>1. \]
    Το πρόσημο της $f'$ και η μονοτονία του ρυθμού μεταβολής $f$ φαίνονται στον παρακάτω πίνακα:
    \begin{center}
    \begin{tikzpicture}
    \tikzset{arrow style/.style={maincolor,>->,>=latex'}}
    \tikzset{z style/.style = {fill=white,inner sep=.2mm}}

    \tkzTabVar{-/,+/,-/}
    \end{tikzpicture}
    \end{center}
    Επομένως ο ρυθμός μεταβολής της θερμοκρασίας γίνεται ελάχιστος όταν $t=1$ και η ελάχιστη τιμή του είναι $f(1)=3 \cdot 1^2 - 6 \cdot 1 + 2 = -1 \, \text{°C/ώρα}$.
\end{alist}
