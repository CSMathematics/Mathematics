\lysh{35767}{2}

\begin{alist}
    \item Εφόσον η μέση τιμή του βάρους έξι μαθητών είναι 72 κιλά, το άθροισμα του βάρους των έξι μαθητών είναι $\sum_{i=1}^{6} t_i = 6 \cdot \bar{x} = 6 \cdot 72 = 432$ κιλά.
    \item 
    \begin{rlist}
        \item Λόγω του ερωτήματος (α), για να βρούμε το άθροισμα των βαρών των οκτώ μαθητών, θα προσθέσουμε στο συνολικό βάρος των έξι μαθητών, τα βάρη των δύο μαθητών που προστέθηκαν.

        Οπότε αυτό είναι: $432+77+83=592$ κιλά.
        \item Η μέση τιμή του βάρους των οκτώ μαθητών είναι:
        \[ \bar{x} = \frac{\sum_{i=1}^{8} t_i}{8} = \frac{592}{8} = 74 \text{ κιλά.} \]
    \end{rlist}
\end{alist}
