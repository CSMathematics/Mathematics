\lysh{25458}{4}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη για κάθε $x \in \mathbb{R}$ με
\[ f'(x) = \left(\sqrt{x^2 + 2x + 5}\right)' = \frac{1}{2\sqrt{x^2 + 2x + 5}}(x^2 + 2x + 5)' = \frac{2x + 2}{2\sqrt{x^2 + 2x + 5}} \]
\[ = \frac{2(x + 1)}{2\sqrt{x^2 + 2x + 5}} = \frac{x + 1}{\sqrt{x^2 + 2x + 5}} \]

\item Είναι:
\[ f'(x) = 0 \Leftrightarrow \frac{x + 1}{\sqrt{x^2 + 2x + 5}} = 0 \Leftrightarrow x + 1 = 0 \Leftrightarrow x = -1 \]
\[ f'(x) > 0 \Leftrightarrow \frac{x + 1}{\sqrt{x^2 + 2x + 5}} > 0 \Leftrightarrow x + 1 > 0 \Leftrightarrow x > -1 \]
\[ f'(x) < 0 \Leftrightarrow \frac{x + 1}{\sqrt{x^2 + 2x + 5}} < 0 \Leftrightarrow x + 1 < 0 \Leftrightarrow x < -1 \]
Άρα η συνάρτηση $f$ είναι γνησίως φθίνουσα στο διάστημα $(-\infty, -1]$ και γνησίως αύξουσα στο διάστημα $[-1, +\infty)$.

\item Αφού $f'(-1) = 0$, $f'(x) < 0$ στο $(-\infty, -1)$ και $f'(x) > 0$ στο $(-1, +\infty)$, τότε η $f$ παρουσιάζει ελάχιστο για $x = -1$. Επομένως, θα ισχύει:
\[ f(x) \ge f(-1) \Leftrightarrow f(x) \ge \sqrt{(-1)^2 + 2 \cdot (-1) + 5} \Leftrightarrow f(x) \ge 2 \quad \text{για κάθε } x \in \mathbb{R} \]

\item Τα ζητούμενα σημεία $(x, f(x))$ είναι εκείνα για τα οποία ισχύει $f'(x) = 0$. Σύμφωνα με το ερώτημα (β), η λύση της εξίσωσης $f'(x) = 0$ είναι $x = -1$. Άρα, στο σημείο $A(-1, f(-1)) = (-1, 2)$ η εφαπτομένη της γραφικής παράστασης της $f$ είναι παράλληλη στον άξονα $x'x$. Η εξίσωση της εφαπτομένης θα είναι:
\[ y = \lambda x + \beta \text{, όπου } \lambda = f'(-1) = 0 \]
Οπότε είναι:
\[ y = \beta \Leftrightarrow y = 2 \]
αφού οι συντεταγμένες του σημείου $A$ επαληθεύουν την εξίσωση της εφαπτομένης.
\end{alist}
