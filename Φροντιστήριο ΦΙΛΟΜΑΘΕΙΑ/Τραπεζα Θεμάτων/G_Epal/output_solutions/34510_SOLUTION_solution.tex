    \item Έχουμε
    \[ f'(x) = \left(\frac{x+1}{x-2}\right)' = \frac{(x+1)'(x-2)-(x+1)(x-2)'}{(x-2)^2} = \frac{x-2-(x+1)}{(x-2)^2} = -\frac{3}{(x-2)^2}, \; x \in \mathbb{R}-\{2\}. \]

    \item
    \begin{rlist}
        \item Είναι $f'(3)=-3$ οπότε $(\varepsilon): y=f'(3)x+\beta \Leftrightarrow y=-3x+\beta$. Ακόμα η $(\varepsilon)$ διέρχεται από το σημείο $A(3,4)$, άρα $4=-3 \cdot 3 + \beta \Leftrightarrow \beta = 13$. Τελικά $(\varepsilon): y=-3x+13$.
        
        \item Οι συντεταγμένες του $B(1,10)$ επαληθεύουν την εξίσωση $y=-3x+13$, αφού $10=(-3)1+13$, επομένως η $(\varepsilon)$ διέρχεται από το σημείο $B$.
    \end{rlist}
    
    \item Για το τυχαίο σημείο $\Gamma(x_0, f(x_0))$ της γραφικής παράστασης της συνάρτησης $f$ η εξίσωση της εφαπτομένης $(\eta)$ είναι $y=f'(x_0)x$, η οποία λόγω του α) είναι
    \[ y = -\frac{3}{(x_0-2)^2}x \quad (1). \]
    Η εφαπτομένη $(\eta)$, επιπλέον, διέρχεται από το σημείο $\Gamma(x_0, f(x_0))$, άρα η (1) γίνεται
    \[ \frac{x_0+1}{x_0-2} = -\frac{3x_0}{(x_0-2)^2} \Leftrightarrow (x_0-2)(x_0+1) = -3x_0 \Leftrightarrow x_0^2+2x_0-2=0 \Leftrightarrow x_0 = -1 \pm \sqrt{3}. \]
\end{alist}
