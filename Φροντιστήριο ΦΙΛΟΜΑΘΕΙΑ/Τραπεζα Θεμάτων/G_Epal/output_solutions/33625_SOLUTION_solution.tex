\lysh{33625}{2}
\begin{alist}
    \item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $f'(x) = (2x^2 - 7x)' = 4x - 7$.
    \item Η ευθεία $\varepsilon : y = \lambda x + \beta$ είναι εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο της $(2,-6)$, οπότε $\lambda = f'(2) = 4 \cdot 2 - 7 = 1$.
    \item Λόγω του ερωτήματος (β) η $\varepsilon$ γράφεται $\varepsilon : y = x + \beta$.
    
    Επειδή το σημείο $(2, -6)$ ανήκει στην εφαπτομένη, έχουμε:
    \[-6 = 2 + \beta \Leftrightarrow \beta = -8.\]
\end{alist}
