\lysh{27843}{2}
\begin{alist}
\item Είναι $f'(x)=\left(-2x^3+x^2-4\right)'=\left(-2x^3\right)'+\left(x^2\right)'-(4)'=-6x^2+2x$ με $x \in \mathbb{R}$.

\item Ο συντελεστής διεύθυνσης $\lambda$ της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(-1,-1)$ είναι ίσος με:
\[ \lambda=f'(-1)\overset{\alpha)}{=}-6(-1)^2+2(-1)=-6-2=-8 . \]

\item Η εξίσωση εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x_0=-1$ είναι:
\[ y=\lambda x+\beta \overset{\beta)}{\iff} y=-8x+\beta \]
Επειδή το σημείο $A(-1,-1)$ ανήκει στην εφαπτομένη, έχουμε:
\[ -1=-8(-1)+\beta \iff \beta=-9 \]
Άρα η εξίσωση της εφαπτομένης είναι: $y=-8x-9$.
\end{alist}
