\lysh{31088}{2}
\begin{alist}
    \item Η συνάρτηση $f$ ορίζεται όταν $x \neq 0$. Επομένως, έχει ως πεδίο ορισμού το σύνολο
    \[ A = (-\infty, 0) \cup (0, +\infty). \]

    \item Για κάθε $x \in A$ είναι:
    \[ f'(x) = \left(-3 \cdot \frac{1}{x}\right)' = -3 \cdot \left(\frac{1}{x}\right)' = -3 \left(-\frac{1}{x^2}\right) = \frac{3}{x^2}. \]

    \item Σύμφωνα με το προηγούμενο ερώτημα (β) έχουμε $f'(x) = \frac{3}{x^2}$.

    Άρα, για $x=-3$ έχουμε $f'(-3) = \frac{3}{(-3)^2} = \frac{3}{9} = \frac{1}{3}$.
\end{alist}
