\begin{alist}
    \item Ισχύει $N_1=\nu_1$ και $\nu_κ=N_{\kappa} - N_{\kappa-1}$ ή $N_{\kappa}=N_{\kappa-1}+\nu_{\kappa}$.
    Επίσης $f_i=\frac{\nu_i}{\nu}$ και $f_i\%=100 \cdot f_i$, οπότε ο πίνακας συμπληρωμένος είναι ο παρακάτω.
    \begin{center}
    \begin{mytblr}{}
        Ποσό σε ευρώ $x_i$  & $\nu_i$ & $N_i$ & $f_i$ & $f_i\%$ \\
        0            & 1     & 1     & 0,05  & 5       \\
        1            & 9     & 10    & 0,45  & 45      \\
        2            & 8     & 18    & 0,40  & 40      \\
        3            & 2     & 20    & 0,10  & 10      \\
        Σύνολο       & 20    & \SetCell{bg=gray!50}{} & 1     & 100     \\
    \end{mytblr}
    \end{center}

    \item Τη συγκεκριμένη ημέρα το πολύ 2 ευρώ ξόδεψαν:
    $\nu_1+\nu_2+\nu_3 = 1+9+8=18$ μαθητές.
    
    \textbf{Εναλλακτικά:} $N_3=18$ μαθητές.

    \item Το ποσοστό των μαθητών που ξόδεψαν τη συγκεκριμένη ημέρα, τουλάχιστον 2 ευρώ είναι: $f_3\% + f_4\% = 40+10 = 50\%$.
\end{alist}
