\lysh{24356}{2}
\begin{alist}
\item Είναι: $x^2 - 1 \neq 0 \Leftrightarrow x^2 \neq 1 \Leftrightarrow x \neq -1 \text{ και } x \neq 1$ οπότε πεδίο ορισμού της $g$ είναι το σύνολο $A = \mathbb{R} - \{-1, 1\}$.

\item Είναι $x^2 - 1 = (x - 1) \cdot (x + 1)$, ως διαφορά τετραγώνων. Για $x \in A = \mathbb{R} - \{-1, 1\}$, ισχύει: $\displaystyle g(x) = \frac{(x - 2) \cdot (x - 1)}{(x + 1) \cdot (x - 1)} = \frac{x - 2}{x + 1}$.

\item Λόγω του ερωτήματος (β) είναι: $\displaystyle \lim\limits_{x \to 1} g(x) = \lim\limits_{x \to 1} \frac{x - 2}{x + 1} = \frac{1 - 2}{1 + 1} = -\frac{1}{2}$.
\end{alist}
