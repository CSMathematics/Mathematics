\lysh{33624}{2}
\begin{alist}
\item Το $x-1 \neq 0 \Leftrightarrow x \neq 1$, επομένως το πεδίο ορισμού της συνάρτησης $f$ είναι το σύνολο $A = \mathbb{R} - \{1\} = (-\infty, 1) \cup (1, +\infty)$.
\item Είναι: $\Pi = f(0) + 3f(2) = \dfrac{2 \cdot 0}{0-1} + 3 \cdot \dfrac{2 \cdot 2}{2-1} = 0 + 3 \cdot 4 = 12$.
\item Είναι: $\lim\limits_{x \to 2} f(x) = \lim\limits_{x \to 2} \dfrac{2x}{x-1} = \dfrac{2 \cdot 2}{2-1} = \dfrac{4}{1} = 4$.
\end{alist}
