\lysh{31086}{2}
\begin{alist}
\item Η συνάρτηση $f$ ορίζεται όταν $x \neq 0$.

Επομένως, έχει ως πεδίο ορισμού το σύνολο $A = (-\infty, 0) \cup (0, +\infty)$.

\item Για κάθε $x \in A$ είναι: $f(x) = \frac{x^2 - 2x}{x} = \frac{x(x - 2)}{x} = x - 2$.

\item Σύμφωνα με το προηγούμενο ερώτημα (β) προκύπτει:
\[ \lim\limits_{x \to 0}f(x) = \lim\limits_{x \to 0}(x - 2) = -2. \]
\end{alist}
