\lysh{25460}{2}
\begin{alist}
\item Οι βαθμοί των 10 φοιτητών σε αύξουσα σειρά είναι:
\[ 3, 4, 5, 5, 6, 6, 6, 7, 8, 9 \]
Η μέση τιμή των βαθμών όλων των φοιτητών του δείγματος είναι:
\[ \bar{x} = \frac{\sum\limits_{i=1}^{10} t_i}{10} = \frac{59}{10} = 5,9 \]

\item Σύμφωνα με τα δεδομένα προκύπτει το ακόλουθο διάγραμμα συχνοτήτων:

\begin{center}
\begin{tikzpicture}[scale=1]
    % Άξονες
    \draw[very thick] (2,0) -- (10.5,0);
    \draw[very thick] (2,0) -- (2,4);
    
    % Ετικέτες αξόνων
    \node at (6,-0.8) {βαθμός};
    \node at (1.4,3.8) {$\nu_i$};
    
    % Διαγραμμίσεις y-άξονα
    \foreach \y in {1,2,3} {
        \filldraw (2,\y) circle (1.5pt);
        \node[left] at (1.8,\y) {\y};
    }
    
    % Διαγραμμίσεις x-άξονα και ράβδοι
    \foreach \x/\y in {3/1, 4/1, 5/2, 6/3, 7/1, 8/1, 9/1} {
        \filldraw (\x,0) circle (1.5pt);
        \node[below] at (\x,-0.1) {\x};
        \draw[very thick, black!70] (\x,0) -- (\x,\y);
    }
\end{tikzpicture}
\end{center}

\item Το πλήθος των φοιτητών με βαθμό μεγαλύτερο του 8 ισούται με 1, οπότε το αντίστοιχο ποσοστό είναι:
\[ \frac{1}{10} \cdot 100\% = 10\% \]
\end{alist}
