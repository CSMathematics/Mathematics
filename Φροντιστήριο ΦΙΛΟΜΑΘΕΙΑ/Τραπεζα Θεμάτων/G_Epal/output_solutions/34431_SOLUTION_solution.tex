\lysh{34431}{2}

\begin{alist}
    \item Για καθέναν από τους κυκλικούς τομείς του διαγράμματος ισχύει ότι το αντίστοιχο τόξο $a_i$ και η σχετική συχνότητα $f_i$ συνδέονται από τον τύπο:
    \[ f_i = \frac{a_i}{360^{\circ}} , \; i = 1,2,3,4. \]
    Οπότε για καθεμία από τις ζητούμενες σχετικές συχνότητες θα έχουμε:
    \par Πουλιά: $f_1 = \frac{90}{360} = \frac{1}{4} = 0,25$
    \par Άλλα: $f_3 = \frac{45}{360} = \frac{1}{8} = 0,125$
    
    \item Αν $\nu_i$ είναι η συχνότητα των απαντήσεων για καθένα από τα κατοικίδια, τότε ισχύει ότι $f_i = \frac{\nu_i}{\nu}$.
    \par Οπότε, αν $\nu_1$ είναι το πλήθος των μαθητών της τάξης αυτής που προτιμούν τα πουλιά για κατοικίδιο, τότε:
    \[ \nu_1 = f_1 \cdot \nu = 0,25 \cdot 120 = 30 \text{ μαθητές.} \]
\end{alist}
