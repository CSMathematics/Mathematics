\begin{alist}
\item Εφόσον το σημείο $\text{A}(1,4)$ ανήκει στη γραφική παράσταση της συνάρτησης $f$ θα
ισχύει ότι $f(1)=4 \Leftrightarrow \frac{a}{1}=4 \Leftrightarrow a=4$. Οπότε η συνάρτηση είναι $f(x)=\frac{4}{x}$.
\item 
\begin{rlist}
\item Ισχύει ότι
\[ f'(x)=\left(\frac{4}{x}\right)'=\frac{(4)'x-4(x)'}{x^2}=\frac{0 \cdot x-4 \cdot 1}{x^2}=\frac{-4}{x^2}. \]
Οπότε $f'(1)=\frac{-4}{1^2}=-4$.
\item Η εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο $\text{A}(1,4)$ έχει
συντελεστή διεύθυνσης $\lambda=f'(1)=-4$.
Οπότε η εξίσωσή της θα είναι $(\varepsilon): y=\lambda x+\beta=-4x+\beta$. Το σημείο $\text{A}(1,4)$ ανήκει στην
$(\varepsilon)$ οπότε θα ισχύει $4=-4 \cdot 1+\beta$, οπότε $\beta=8$.
Συνεπώς, η εφαπτομένη είναι $(\varepsilon): y=-4x+8$.
\end{rlist}
\item 

\begin{rlist}
\item Για το σημείο τομής $\text{K}$ με τον άξονα $x'x$ ισχύει ότι $y=0$, οπότε έχουμε
$0=-4x+8 \Leftrightarrow 4x=8 \Leftrightarrow x=2$. Άρα $\text{K}(2,0)$.
Αντίστοιχα, για το σημείο τομής $\Lambda$ με τον άξονα $y'y$ ισχύει ότι $x=0$, οπότε έχουμε
$y=-4 \cdot 0+8=8$. Άρα $\Lambda(0,8)$.
\item Αν $\text{O}$ η αρχή των αξόνων, τότε ισχύει ότι $E=\frac{1}{2}{OK} \cdot {O\Lambda}=\frac{1}{2} \cdot 2 \cdot 8=8$.
