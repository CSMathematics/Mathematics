\lysh{34591}{2}
\begin{alist}
	\item Η συνάρτηση $f$ ορίζεται όταν
	\[
		x-1 \neq 0 \Leftrightarrow x \neq 1
	\]
	Επομένως, έχει ως πεδίο ορισμού το σύνολο
	\[
		\mathrm{A}=(-\infty,1)\cup(1,+\infty)
	\]
	
	\item Η εξίσωση $x^2-3x+2=0$ έχει $\Delta = (-3)^2 - 4 \cdot 1 \cdot 2 = 9-8=1$ και ρίζες
	\[
		x_1 = \frac{-(-3)-\sqrt{1}}{2} = \frac{3-1}{2}=1 \text{ και } x_2 = \frac{-(-3)+\sqrt{1}}{2} = \frac{3+1}{2}=2
	\]
	Επομένως το τριώνυμο γράφεται
	\[
		x^2-3x+2=(x-1)(x-2)
	\]
	
	Για κάθε $x \in \mathrm{A}$ είναι:
	\[
		f(x)=\frac{x^2-3x+2}{x-1}=\frac{(x-1)(x-2)}{x-1}=x-2
	\]
	
	\item Σύμφωνα με το προηγούμενο ερώτημα (β) είναι:
	\[
		\lim\limits_{x\to1}f(x) = \lim\limits_{x\to1}(x-2)=-1
	\]
\end{alist}
