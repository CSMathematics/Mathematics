\lysh{28617}{4}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $[0, +\infty)$ με $f'(x) = (ax^2 - 2x)' = 2ax - 2$.

Επομένως έχουμε $f'(1) = 0 \Leftrightarrow 2a \cdot 1 - 2 = 0 \Leftrightarrow 2a = 2 \Leftrightarrow a = 1$.

\item 
\begin{rlist}
\item Για να ορίζεται η συνάρτηση $g$ πρέπει να ισχύει $x \ge 0$. Επομένως το πεδίο ορισμού της συνάρτησης $g$ είναι το $[0, +\infty)$.

\item Έχουμε $g(x) = 0 \Leftrightarrow \sqrt{x} - \sqrt{2} = 0 \Leftrightarrow \sqrt{x} = \sqrt{2} \Leftrightarrow x = 2$.
\end{rlist}

\item Για $a=1$ έχουμε
\begin{rlist}
\item $h(x) = \frac{f}{g}(x) = \frac{f(x)}{g(x)} = \frac{x^2 - 2x}{\sqrt{x} - \sqrt{2}}$, $x \in [0, 2) \cup (2, +\infty)$.

\item 
\[
\begin{aligned}
\lim\limits_{x \to 2} h(x) &= \lim\limits_{x \to 2} \frac{x^2 - 2x}{\sqrt{x} - \sqrt{2}} = \lim\limits_{x \to 2} \frac{x(x - 2)(\sqrt{x} + \sqrt{2})}{(\sqrt{x} - \sqrt{2})(\sqrt{x} + \sqrt{2})} = \lim\limits_{x \to 2} \frac{x(x - 2)(\sqrt{x} + \sqrt{2})}{x - 2} = \\[1ex]
&= \lim\limits_{x \to 2} \left[ x(\sqrt{x} + \sqrt{2}) \right] = 2 \cdot 2\sqrt{2} = 4\sqrt{2}.
\end{aligned}
\]
\end{rlist}
\end{alist}
