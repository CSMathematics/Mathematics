\lysh{34845}{2}
\begin{alist}
\item Έχουμε
\[
f'(x)=(1-x^3)'=(1)'-(x^3)'=0-3x^2=-3x^2, \quad x \in \mathbb{R}.
\]
\item Ο συντελεστής διεύθυνσης $\lambda$ της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της με $A(1,0)$ είναι ίσος με:
\[
\lambda = f'(1) \stackrel{\alpha)}{=} -3 \cdot 1^2 = -3.
\]
\item Η εξίσωση εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x_0=1$ είναι:
\[
y = \lambda x + \beta \stackrel{\beta)}{\iff} y = -3x + \beta
\]
Επειδή το σημείο $A(1,0)$ ανήκει στην εφαπτομένη, έχουμε:
\[
0 = -3 \cdot 1 + \beta \iff \beta = 3.
\]
Άρα η εξίσωση της εφαπτομένης είναι: $y = -3x+3$.
\end{alist}
