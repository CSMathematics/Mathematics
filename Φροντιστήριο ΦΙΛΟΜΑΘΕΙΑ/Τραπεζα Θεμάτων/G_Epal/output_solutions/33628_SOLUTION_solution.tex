\lysh{33628}{4}
\begin{alist}
    \item Στο ορθογώνιο τρίγωνο ΟΑΒ από το Πυθαγόρειο Θεώρημα είναι:
    \[(ΑΒ)^2 = (ΟΑ)^2 + (ΟΒ)^2 \Leftrightarrow (ΑΒ)^2 = 2^2 + x^2 \Leftrightarrow (ΑΒ)^2 = 4 + x^2, \text{ οπότε } (ΑΒ) = \sqrt{4+x^2}.\]
    Όμως $d(x) = (ΑΒ)$, επομένως $d(x) = \sqrt{x^2+4}$ με $x>0$.

    \item Λόγω του ερωτήματος (α) η απόσταση των σημείων Α και Β είναι
    \[d(x) = \sqrt{x^2+4} \text{ με } x>0.\]
    Η συνάρτηση είναι παραγωγίσιμη στο $(0, +\infty)$ με
    \[d'(x) = (\sqrt{x^2+4})' = \frac{1}{2\sqrt{x^2+4}}(x^2+4)' = \frac{1}{2\sqrt{x^2+4}} \cdot 2x = \frac{x}{\sqrt{x^2+4}}.\]
    Επομένως ο ρυθμός μεταβολής της απόστασης των σημείων Α και Β όταν $x=3$ είναι
    \[d'(3) = \frac{3}{\sqrt{3^2+4}} = \frac{3}{\sqrt{9+4}} = \frac{3}{\sqrt{13}}.\]

    \item Λόγω του ερωτήματος (β) για $x>0$ η $d'(x) = \frac{x}{\sqrt{x^2+4}}$ και είναι $d'(x)>0$ για κάθε $x \in (0, +\infty)$. Επομένως η συνάρτηση $d$ είναι γνησίως αύξουσα στο $(0, +\infty)$, οπότε καθώς το σημείο Β κινείται προς τα δεξιά στον ημιάξονα $Ox$, το μήκος του ΑΒ αυξάνεται.
\end{alist}
