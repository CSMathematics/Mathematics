\lysh{33573}{4}

\begin{alist}
\item Η μέση τιμή ενός δείγματος δίνεται από τον τύπο
\[ \overline{x} = \frac{\sum_{i=1}^{v} t_i}{v} \]
Για το δείγμα 1 έχουμε:
\[ \overline{x} = \frac{8+9+10+11+12}{5} = \frac{50}{5} = 10 \]
Για το δείγμα 2 έχουμε:
\[ \overline{x} = \frac{\sum t_i}{v} = \frac{8+10+10+10+12}{5} = \frac{50}{5} = 10 \]
Η διάμεσος ενός δείγματος με περιττό πλήθος παρατηρήσεων, οι οποίες έχουν διαταχθεί σε αύξουσα σειρά, είναι ίση με τη μεσαία παρατήρηση. Εφόσον, για το δείγμα 1 έχουμε πέντε παρατηρήσεις, τότε η τρίτη παρατήρηση είναι η διάμεσος, άρα $\delta = 10$.
Ομοίως, για το δείγμα 2 είναι $\delta = 10$.

\item Για τον υπολογισμό της διασποράς έχουμε:
Για το δείγμα 1 :
\[ s^2 = \frac{\sum_{i=1}^{v} (t_i - \overline{x})^2}{v} \]
\[ s^2 = \frac{(8-10)^2 + (9-10)^2 + (10-10)^2 + (11-10)^2 + (12-10)^2}{5} \]
\[ s^2 = \frac{4+1+0+1+4}{5} = \frac{10}{5} = 2 \]
Για το δείγμα 2 :
\[ s^2 = \frac{\sum_{i=1}^{v} (t_i - \overline{x})^2}{v} \]
\[ s^2 = \frac{(8-10)^2 + (10-10)^2 + (10-10)^2 + (10-10)^2 + (12-10)^2}{5} \]
\[ s^2 = \frac{4+0+0+0+4}{5} = \frac{8}{5} = 1,6 \]
Συνεπώς, η τυπική απόκλιση για τα δύο δείγματα είναι:
Για το δείγμα 1: $s = \sqrt{s^2} = \sqrt{2}$.
Για το δείγμα 2: $s = \sqrt{s^2} = \sqrt{1,6}$.

\item Ο συντελεστής μεταβολής δίνεται από τον τύπο $CV = \frac{s}{\overline{x}}$, οπότε :
Για το δείγμα 1: $CV = \frac{s}{\overline{x}} = \frac{\sqrt{2}}{10}$.
Για το δείγμα 2: $CV = \frac{s}{\overline{x}} = \frac{\sqrt{1,6}}{10}$.

\item Η άποψη του μαθητή δεν είναι σωστή. Τα δύο δείγματα που δόθηκαν έχουν την ίδια μέση τιμή και την ίδια διάμεσο, όμως ο συντελεστής μεταβολής του πρώτου δείγματος είναι μεγαλύτερος από το συντελεστή μεταβολής του δεύτερου δείγματος γιατί $\frac{\sqrt{2}}{10} > \frac{\sqrt{1,6}}{10}$. Επομένως τα δείγματα διαφέρουν ως προς την ομοιογένεια.
\end{alist}
