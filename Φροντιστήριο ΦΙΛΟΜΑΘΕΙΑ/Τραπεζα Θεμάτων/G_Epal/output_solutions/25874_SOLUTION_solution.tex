\lysh{25874}{2}
\begin{alist}
\item Η συνάρτηση $f$ ορίζεται όταν
\[ x - 1 \neq 0 \Leftrightarrow x \neq 1 \]
Επομένως, έχει ως πεδίο ορισμού το σύνολο
\[ A = (-\infty, 1) \cup (1, +\infty) \]
\item Για κάθε $x \in A$ είναι:
\[ f(x) = \frac{x^2 - 1}{x - 1} = \frac{(x - 1)(x + 1)}{x - 1} = x + 1 \]
\item Σύμφωνα με το προηγούμενο ερώτημα (β) είναι:
\[ \lim\limits_{x \to 1} f(x) = \lim\limits_{x \to 1} (x + 1) = 2 \]
\end{alist}
