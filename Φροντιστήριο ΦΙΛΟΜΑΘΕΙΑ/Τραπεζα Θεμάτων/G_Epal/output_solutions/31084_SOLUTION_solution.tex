\lysh{31084}{2}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη για κάθε $x \in \mathbb{R}$. Βρίσκουμε την πρώτη παράγωγο της συνάρτησης. Παραγωγίζουμε τη συνάρτηση ως άθροισμα δύο συναρτήσεων:
\[ f'(x) = \left(\frac{1}{3}x^3 - 2023\right)' = \left(\frac{1}{3}x^3\right)' - (2023)' = \frac{1}{3}(x^3)' = \frac{1}{3} \cdot 3 \cdot x^2 = x^2. \]
\item Από το α) ερώτημα έχουμε $f'(x) = x^2$. Άρα, για $x=-1$ έχουμε $f'(-1) = (-1)^2 = 1$.
\item Ομοίως για $x=1$ προκύπτει $f'(1) = 1^2 = 1$, άρα: $f'(-1) + f'(1) = 1 + 1 = 2$.
\end{alist}
