\lysh{35780}{2}

\begin{alist}
    \item Η ζητούμενη μέση τιμή του δείγματος των πέντε αριθμών που δόθηκαν είναι:
    \[ \bar{x} = \frac{\sum_{i=1}^{\nu} t_i}{\nu} = \frac{2+3+4+6+10}{5} = \frac{25}{5} = 5. \]
    \item Λόγω και του ερωτήματος (α) είναι:
    \begin{align*}
        s^2 &= \frac{1}{\nu}\sum_{i=1}^{\nu}(x_i - \bar{x})^2 = \frac{1}{5}\sum_{i=1}^{5}(x_i - 5)^2 \\
            &= \frac{1}{5}\left[ (2-5)^2 + (3-5)^2 + (4-5)^2 + (6-5)^2 + (10-5)^2 \right] \\
            &= \frac{1}{5}\left[ (-3)^2 + (-2)^2 + (-1)^2 + 1^2 + 5^2 \right] \\
            &= \frac{1}{5}(9+4+1+1+25) = \frac{40}{5} = 8.
    \end{align*}
    \item Λόγω του ερωτήματος (β) είναι: $s = \sqrt{s^2} = \sqrt{8}$ η ζητούμενη τυπική απόκλιση.
\end{alist}
