\lysh{32260}{4}
\begin{alist}
    \item Από τα δεδομένα έχουμε $E=(\text{ΑΒΓΔ}) = 4\ \text{m}^2$ και $\text{ΑΒ} = x\ \text{m}$ με $x>0$. \\
    Αν $\text{ΒΓ} = y$ με $y>0$ τότε έχουμε: $E = \text{ΑΒ} \cdot \text{ΒΓ} = x \cdot y$ ή $4=x \cdot y \Leftrightarrow y=\frac{4}{x}$, $x>0$ άρα $\text{ΒΓ} = \frac{4}{x}$, $x>0$.

    \item Η περίμετρος του ορθογωνίου είναι: $\Pi = 2 \cdot \text{ΑΒ} + 2 \cdot \text{ΒΓ} = 2x+2y$, η οποία λόγω του ερωτήματος (α) γράφεται: $\Pi = 2x+2\frac{4}{x} = 2x+\frac{8}{x}$, $x>0$. Άρα η συνάρτηση $f(x) = 2x+\frac{8}{x}$, $x>0$, δίνει την περίμετρο του ορθογωνίου ΑΒΓΔ.

    \item Για $x>0$ η συνάρτηση $f$ είναι παραγωγίσιμη με
    \[
        f'(x) = \left(2x+\frac{8}{x}\right)' = 2-\frac{8}{x^2} = \frac{2x^2-8}{x^2} = \frac{2(x^2-4)}{x^2},
    \]
    οπότε:
    $f'(x)=0 \Leftrightarrow x^2-4=0 \xrightarrow{x>0} x=2$ και $f'(x)>0 \Leftrightarrow x^2-4>0 \xrightarrow{x>0} x>2$.
    
    Και για την $f$ έχουμε τον παρακάτω πίνακα μεταβολών.
    \begin{center}
        \begin{tikzpicture}
        \tkzTabInit[color,lgt=1.5,espcl=2,colorC = maincolor!50,colorL = maincolor!30,colorV = maincolor!50]%
        {$x$ / .8,$f'(x)$ /.8,$f(x)$/0.8}%
        {$0$,$2$,$+\infty$}%
        \tkzTabLine{d,-,z,+}%
        \tkzTabVar{+/,-/,+/}%
        \end{tikzpicture}
    \end{center}
    Επομένως η $f$ είναι γνησίως φθίνουσα στο διάστημα $(0, 2]$ και γνησίως αύξουσα στο διάστημα $[2, +\infty)$. Στο $x_0 = 2$ παίρνει την ελάχιστη τιμή της $f(2) = 4+4=8$.

    \item[δ)] Λόγω του ερωτήματος (γ) η περίμετρος του ορθογωνίου γίνεται ελάχιστη όταν $x = 2$. Τότε όμως λόγω του ερωτήματος (α) το $y = \frac{4}{2} = 2$. Επομένως οι διαστάσεις του ορθογωνίου είναι ίσες, άρα αυτό είναι τετράγωνο. Οπότε έχει δίκιο ο μαθητής.
\end{alist}
