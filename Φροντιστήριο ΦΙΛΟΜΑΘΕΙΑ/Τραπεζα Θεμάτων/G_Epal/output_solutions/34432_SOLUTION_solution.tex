\lysh{34432}{2}

    \begin{alist}
        \item Από το ραβδόγραμμα παρατηρούμε ότι οι μαθητές επέλεξαν τα αθλήματα με τις παρακάτω συχνότητες:
        \\Καλαθοσφαίριση: $\nu_1=45$, Ποδόσφαιρο: $\nu_2=40$, Πετοσφαίριση: $\nu_3=20$ και Αντισφαίριση: $\nu_4=15$.
        \par Οπότε το σύνολο των μαθητών της Γ΄ λυκείου του συγκεκριμένου σχολείου ήταν:
        \[
            \nu=\nu_1+\nu_2+\nu_3+\nu_4=45+40+20+15=120 \text{ μαθητές.}
        \]
        \item Για καθένα από τα αθλήματα η συχνότητα $\nu_i$ και η σχετική συχνότητα $f_i$ συνδέονται από τον τύπο:
        \[
            f_i=\frac{\nu_i}{\nu}.
        \]
        Οπότε για καθεμία από τις ζητούμενες σχετικές συχνότητες θα έχουμε:
        \\Καλαθοσφαίριση: $f_1=\dfrac{\nu_1}{\nu}=\dfrac{45}{120}=\dfrac{3}{8}=0,375$
        \\Ποδόσφαιρο: $f_2=\dfrac{\nu_2}{\nu}=\dfrac{40}{120}=\dfrac{1}{3}=0,33...$
        \\Πετοσφαίριση: $f_3=\dfrac{\nu_3}{\nu}=\dfrac{20}{120}=\dfrac{1}{6}=0,16...$
        \\Αντισφαίριση: $f_4=\dfrac{\nu_4}{\nu}=\dfrac{15}{120}=\dfrac{1}{8}=0,125$
    \end{alist}
