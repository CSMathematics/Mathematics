\lysh{33622}{2}
\begin{alist}
    \item Για $x \in \mathbb{R}$ η συνάρτηση $f$ είναι παραγωγίσιμη με:
    \[
        f'(x) = (x^2 - 3x + 4)' = 2x - 3.
    \]
    \item Για $x \in \mathbb{R}$ η συνάρτηση $f$ είναι δυο φορές παραγωγίσιμη με:
    \[
        f''(x) = (2x - 3)' = 2.
    \]
    \item Είναι:
    \[
        \Pi = f(0) + f'(0) + f''(0) = (0^2 - 3 \cdot 0 + 4) + (2 \cdot 0 - 3) + 2 = 4 - 3 + 2 = 3.
    \]
\end{alist}
