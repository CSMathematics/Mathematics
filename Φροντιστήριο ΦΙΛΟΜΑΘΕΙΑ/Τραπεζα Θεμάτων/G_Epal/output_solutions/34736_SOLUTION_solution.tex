\begin{alist}
\item  Ισχύει ότι η συνάρτηση $f$ είναι γνησίως αύξουσα στο $[-1,0]$ και γνησίως φθίνουσα στο $[0,1]$, όπως φαίνεται από το σχήμα.
\item 
\begin{rlist}
\item Εφόσον το Β$(0,1)$ είναι σημείο της γραφικής παράστασης της συνάρτησης $f$ θα ισχύει ότι $f(0) = 1$. Η $(\varepsilon)$ είναι παράλληλη στον άξονα $x'x$, οπότε η γωνία που σχηματίζει με αυτόν είναι $0$ μοίρες.
\item Εφόσον η $(\varepsilon): y=1$ είναι εφαπτομένη της γραφικής παράστασης της $f$ στο σημείο Β$(0,1)$, η γωνία που σχηματίζει με τον άξονα $x'x$ είναι $0$ μοίρες και η κλίση της $(\varepsilon)$ είναι $\varepsilon\phi 0=0$ και ίση με την $f'(0)$, οπότε $f'(0) = 0$.
\end{rlist}
\end{alist}
