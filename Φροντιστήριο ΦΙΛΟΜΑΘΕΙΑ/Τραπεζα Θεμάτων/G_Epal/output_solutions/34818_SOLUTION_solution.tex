\lysh{34818}{2}

\begin{alist}
    \item Οι βαθμοί του μαθητή στα $ν = 5$ μαθήματα σε αύξουσα σειρά είναι:
    \[ 8, 12, 13, 14, 18 \]
    Αφού το πλήθος $ν$ των παρατηρήσεων είναι περιττός, η διάμεσος $δ$ θα είναι η μεσαία παρατήρηση ($3^{\text{ος}}$ βαθμός στη σειρά), δηλαδή:
    \[ δ = 13 \]
    
    \item Η μέση τιμή $\bar{x}$ των βαθμών του μαθητή είναι:
    \[ \bar{x} = \frac{\sum_{i=1}^{5} t_i}{5} = \frac{65}{5} = 13 \]
    
    \item Η μέση τιμή των βαθμών του μαθητή λαμβάνοντας υπόψη τους συντελεστές στάθμισης των πέντε μαθημάτων υπολογίζεται από τον τύπο:
    \[ \bar{x} = \frac{x_1 w_1 + x_2 w_2 + x_3 w_3 + x_4 w_4 + x_5 w_5}{w_1 + w_2 + w_3 + w_4 + w_5} \]
    όπου $x_1 = 8, x_2 = 18, x_3 = 12, x_4 = 14, x_5 = 13$ και $w_1 = 2, w_2 = 1, w_3 = 3, w_4 = 3, w_5 = 1$. Επομένως:
    \[ \bar{x} = \frac{8 \cdot 2 + 18 \cdot 1 + 12 \cdot 3 + 14 \cdot 3 + 13 \cdot 1}{2 + 1 + 3 + 3 + 1} = \frac{16 + 18 + 36 + 42 + 13}{10} = \frac{125}{10} = 12,5 \]
\end{alist}
