\lysh{25875}{2}
\begin{alist}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με
\[ f'(x) = (2x^2 - 1)' = (2x^2)' - (1)' = 2 \cdot 2x - 0 = 4x \]
\item Ο συντελεστής διεύθυνσης $\lambda$ της εφαπτομένης $(\varepsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(1,1)$ είναι ίσος με
\[ \lambda = f'(1) = 4 \cdot 1 = 4 \]
\item Η εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της $A(1,1)$ είναι:
\[ y = \lambda x + \beta \Leftrightarrow y = 4x + \beta \]
Επειδή το σημείο $A(1,1)$ ανήκει στην εφαπτομένη, έχουμε:
\[ 1 = 4 \cdot 1 + \beta \Leftrightarrow \beta = -3 \]
Άρα, η εξίσωση της εφαπτομένης είναι:
\[ y = 4x - 3 \]
\end{alist}
