\lysh{31695}{2}
\begin{alist}
\item Βρίσκουμε την πρώτη παράγωγο της δοσμένης συνάρτησης $f(x) = 5 \cdot \frac{1}{x}$ με $x \in \mathbb{R} - \{0\}$.
\[
f'(x) = \left(5 \cdot \frac{1}{x}\right)' = (5x^{-1})' = 5 \cdot (-1)x^{-2} = -\frac{5}{x^2}.
\]
Άρα, $f'(x) = -\frac{5}{x^2}$ για κάθε $x \neq 0$.
\item Ο συντελεστής διεύθυνσης $\lambda$ της εφαπτομένης $(\epsilon)$ της γραφικής παράστασης της $f$ στο σημείο της $A(1,5)$ είναι ίσος με:
\[
\lambda = f'(1) = -\frac{5}{1^2} = -5.
\]
Επομένως, $\lambda = -5$.
\item Η εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x_0 = 1$ είναι:
$y = \lambda x + \beta \Leftrightarrow y = -5 \cdot x + \beta$.
Επειδή το σημείο $A(1,5)$ ανήκει στην εφαπτομένη, έχουμε:
$5 = -5 \cdot 1 + \beta \Leftrightarrow \beta = 5 + 5 = 10$.
Άρα, η εξίσωση της εφαπτομένης είναι: $y = -5x + 10$.
\end{alist}
