\lysh{31085}{2}
\begin{alist}
\item Η συνάρτηση $f$ ορίζεται όταν $x \neq 0$.

Επομένως, έχει ως πεδίο ορισμού το σύνολο $A = (-\infty, 0) \cup (0, +\infty)$.

\item Για κάθε $x \in A$ είναι:
\[ f(x) = \frac{x^2-x}{x} = \frac{x(x-1)}{x} = x - 1. \]

\item Σύμφωνα με το προηγούμενο ερώτημα ($\beta$) προκύπτει:
\[ \lim\limits_{x \to -1} f(x) = \lim\limits_{x \to -1} (x - 1) = -2. \]
\end{alist}
