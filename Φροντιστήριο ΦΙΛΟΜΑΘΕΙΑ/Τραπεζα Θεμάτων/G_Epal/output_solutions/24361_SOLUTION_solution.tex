\lysh{24361}{4}
\begin{alist}
\item Έχουμε τη συνάρτηση $f(x) = x^3 + \lambda x^2 - x, x \in \mathbb{R}$ και $\lambda \in \mathbb{R}$ σταθερά. \\
Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $f'(x) = (x^3 + \lambda x^2 - x)' = 3x^2 + 2\lambda x - 1$ \\
Ισχύει $f'(1) = 0 \Leftrightarrow 3 \cdot 1^2 + 2\lambda \cdot 1 - 1 = 0 \Leftrightarrow 3 + 2\lambda - 1 = 0 \Leftrightarrow \lambda = -1$. \\
Για $\lambda = -1$ η συνάρτηση $f$ γράφεται: $f(x) = x^3 - x^2 - x$.

\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με $f'(x) = (x^3 - x^2 - x)' = 3x^2 - 2x - 1$ \\
Είναι $f'(x) = 0 \Leftrightarrow 3x^2 - 2x - 1 = 0 \Leftrightarrow x = -\frac{1}{3} \text{ ή } x = 1$ και $f'(x) > 0 \Leftrightarrow x < -\frac{1}{3} \text{ ή } x > 1$.

Επομένως η συνάρτηση $f$ είναι γνησίως αύξουσα στο διάστημα $\left(-\infty, -\frac{1}{3}\right]$, γνησίως φθίνουσα στο διάστημα $\left[-\frac{1}{3}, 1\right]$ και γνησίως αύξουσα στο διάστημα $[1, +\infty)$.

Για $x = -\frac{1}{3}$ η συνάρτηση $f$ παρουσιάζει τοπικό μέγιστο ίσο με
\[ f\left(-\frac{1}{3}\right) = \left(-\frac{1}{3}\right)^3 - \left(-\frac{1}{3}\right)^2 + \frac{1}{3} = -\frac{1}{27} - \frac{1}{9} + \frac{1}{3} = \frac{-1-3+9}{27} = \frac{5}{27} \]
και για $x = 1$
παρουσιάζει τοπικό ελάχιστο ίσο με $f(1) = 1^3 - 1^2 - 1 = -1$.

\item Επειδή ο συντελεστής διεύθυνσης της ευθείας $y = -x$ είναι ίσος με $-1$, τα ζητούμενα σημεία $(x, f(x))$ είναι εκείνα για τα οποία ισχύει
\[ f'(x) = -1 \Leftrightarrow 3x^2 - 2x - 1 = -1 \Leftrightarrow 3x^2 - 2x = 0 \Leftrightarrow x = 0 \text{ ή } x = \frac{2}{3}. \]
Για $x = 0$ το $f(0) = 0$ \\
και για $x = \frac{2}{3}$ το 
\[ f\left(\frac{2}{3}\right) = \left(\frac{2}{3}\right)^3 - \left(\frac{2}{3}\right)^2 - \frac{2}{3} = \frac{8}{27} - \frac{4}{9} - \frac{2}{3} = \frac{8-12-18}{27} = -\frac{22}{27}. \]

Άρα τα ζητούμενα σημεία είναι τα $(0,0)$ και $\left(\frac{2}{3}, f\left(\frac{2}{3}\right)\right) = \left(\frac{2}{3}, -\frac{22}{27}\right)$.
\end{alist}
