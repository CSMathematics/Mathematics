\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=1. Πρόσημα και Παρένθεση]
    \textbf{Λάθος:} $-(x - 3) = -x - 3$ \\
    \textbf{Σωστό:} $-(x - 3) = -x + 3$ \\
    \textbf{Γιατί:} Το "$-$" έξω από την παρένθεση αλλάζει τα πρόσημα **όλων** των όρων μέσα σε αυτήν.
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=2. Απαλοιφή Παρονομαστών στο "Σκέτο" Νούμερο]
    \textbf{Λάθος:} $\frac{x}{2} + 3 = 1$ πολλαπλασιάζουμε με 2: $x + 3 = 2$ \\
    \textbf{Σωστό:} $x + 6 = 2$ \\
    \textbf{Γιατί:} Πρέπει να πολλαπλασιάζουμε **κάθε** όρο της εξίσωσης με το Ε.Κ.Π., ακόμα και αυτούς που δεν έχουν παρονομαστή.
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=3. Μεταφορά Όρων]
    \textbf{Λάθος:} $2x + 5 = 10 \Rightarrow 2x = 10 + 5$ \\
    \textbf{Σωστό:} $2x = 10 - 5$ \\
    \textbf{Γιατί:} Όταν ένας όρος "πηδάει" το σύμβολο του ίσον ($=$), το πρόσημό του **πρέπει** να αλλάξει.
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=4. Διαίρεση με τον Συντελεστή]
    \textbf{Λάθος:} $2x = 10 \Rightarrow x = 10 - 2$ ή $x = 10 \cdot 2$ \\
    \textbf{Σωστό:} $x = \frac{10}{2} = 5$ \\
    \textbf{Γιατί:} Στο τελευταίο βήμα κάνουμε πάντα **διαίρεση** με τον αριθμό που βρίσκεται δίπλα στο $x$.
\end{tcolorbox}
