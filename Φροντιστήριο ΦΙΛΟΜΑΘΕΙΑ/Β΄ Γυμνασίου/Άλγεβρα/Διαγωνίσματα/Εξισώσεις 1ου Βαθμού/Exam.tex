\titlos{Μαθηματικά}{Β' Γυμνασίου}{Εξισώσεις 1ου Βαθμού}

\begin{thema}
\item\mbox{}\\
\begin{erwthma}
    \item Τι ονομάζουμε εξίσωση πρώτου βαθμού με έναν άγνωστο; \monades{5}
    \item Πότε μια εξίσωση ονομάζεται αδύνατη και πότε ταυτότητα (αόριστη); \monades{5}
    \item Να χαρακτηρίσετε τις παρακάτω προτάσεις ως Σωστές (Σ) ή Λανθασμένες (Λ):
    \begin{alist}
        \item Η εξίσωση $0x = 5$ είναι αόριστη.
        \item Η λύση της εξίσωσης $x + 3 = 10$ είναι ο αριθμός $x = 7$.
        \item Μια εξίσωση $0x = 0$ έχει άπειρες λύσεις.
        \item Αν $a = b$, τότε $a + c = b + c$ για κάθε αριθμό $c$.
        \item Η εξίσωση $x^2 = 9$ είναι πρώτου βαθμού.
    \end{alist} \monades{5}
\end{erwthma}

\item\mbox{}\\
\begin{erwthma}
    \item Να λύσετε τις παρακάτω εξισώσεις:
    \begin{alist}
        \item $2x - 5 = x + 3$
        \item $3(x - 2) = 2(x + 1) + 4$
    \end{alist} \monades{10}
    \item Να εξετάσετε αν ο αριθμός $x = 2$ είναι λύση της εξίσωσης: $5x - 4 = 3x + 1$. \monades{5}
\end{erwthma}

\item\mbox{}\\
\begin{erwthma}
    \item Να λύσετε την εξίσωση:
    \[ \frac{x - 1}{2} - \frac{x + 2}{3} = \frac{x + 1}{6} \] \monades{15}
    \item Να λύσετε την εξίσωση:
    \[ 2(x - 1) - (x - 4) = x + 2 \] και να χαρακτηρίσετε το αποτέλεσμα. \monades{10}
\end{erwthma}

\item\mbox{}\\
\begin{erwthma}
    \item Το τριπλάσιο ενός αριθμού αυξημένο κατά 5 ισούται με το διπλάσιο του ίδιου αριθμού αυξημένο κατά 12. Να βρείτε τον αριθμό. \monades{15}
    \item Να λύσετε τον τύπο της περιμέτρου ορθογωνίου $P = 2(a + b)$ ως προς $a$. \monades{10}
\end{erwthma}
\end{thema}
