\titlos{Μαθηματικά}{Β' Γυμνασίου}{Εξισώσεις 1ου Βαθμού - Λύσεις}

\section*{Θέμα Α}
\begin{enumerate}
    \item **Ορισμός:** Εξίσωση πρώτου βαθμού με έναν άγνωστο ονομάζεται κάθε ισότητα που περιέχει έναν άγνωστο και μπορεί να πάρει τη μορφή $ax + \beta = 0$.
    \item **Αδύνατη - Ταυτότητα:**
    \begin{itemize}
        \item **Αδύνατη:** Μια εξίσωση που δεν έχει καμία λύση (καταλήγει σε $0x = \beta, \beta \neq 0$).
        \item **Ταυτότητα:** Μια εξίσωση που αληθεύει για κάθε τιμή του αγνώστου (καταλήγει σε $0x = 0$).
    \end{itemize}
    \item **Σωστό/Λάθος:**
    \begin{enumerate}[label=\roman*)]
        \item Λάθος (είναι αδύνατη)
        \item Σωστό ($7+3=10$)
        \item Σωστό
        \item Σωστό
        \item Λάθος (είναι δευτέρου βαθμού λόγω του $x^2$)
    \end{enumerate}
\end{enumerate}

\section*{Θέμα Β}
\begin{enumerate}
    \item **Λύση εξισώσεων:**
    \begin{enumerate}[label=\roman*)]
        \item $2x - 5 = x + 3 \Rightarrow 2x - x = 3 + 5 \Rightarrow x = 8$
        \item $3(x - 2) = 2(x + 1) + 4 \Rightarrow 3x - 6 = 2x + 2 + 4 \Rightarrow 3x - 2x = 6 + 6 \Rightarrow x = 12$
    \end{enumerate}
    \item **Έλεγχος λύσης:** Για $x = 2$:
    \begin{itemize}
        \item Α' μέλος: $5 \cdot 2 - 4 = 10 - 4 = 6$
        \item Β' μέλος: $3 \cdot 2 + 1 = 6 + 1 = 7$
        \item Αφού $6 \neq 7$, ο αριθμός $x = 2$ **δεν** είναι λύση.
    \end{itemize}
\end{enumerate}

\section*{Θέμα Γ}
\begin{enumerate}
    \item **Εξίσωση με κλάσματα:**
    $\frac{x - 1}{2} - \frac{x + 2}{3} = \frac{x + 1}{6}$ (Ε.Κ.Π. = 6)
    $6 \cdot \frac{x - 1}{2} - 6 \cdot \frac{x + 2}{3} = 6 \cdot \frac{x + 1}{6} \Rightarrow$
    $3(x - 1) - 2(x + 2) = x + 1 \Rightarrow$
    $3x - 3 - 2x - 4 = x + 1 \Rightarrow 3x - 2x - x = 1 + 3 + 4 \Rightarrow 0x = 8$.
    Η εξίσωση είναι **αδύνατη**.
    \item **Ειδική περίπτωση:**
    $2(x - 1) - (x - 4) = x + 2 \Rightarrow 2x - 2 - x + 4 = x + 2 \Rightarrow x + 2 = x + 2 \Rightarrow 0x = 0$.
    Η εξίσωση είναι **ταυτότητα** (αόριστη).
\end{enumerate}

\section*{Θέμα Δ}
\begin{enumerate}
    \item **Πρόβλημα:** Έστω $x$ ο αριθμός.
    $3x + 5 = 2x + 12 \Rightarrow 3x - 2x = 12 - 5 \Rightarrow x = 7$.
    Ο αριθμός είναι το 7.
    \item **Επίλυση τύπου:** $P = 2(a + b) \Rightarrow P = 2a + 2b \Rightarrow P - 2b = 2a \Rightarrow a = \frac{P - 2b}{2}$.
\end{enumerate}
