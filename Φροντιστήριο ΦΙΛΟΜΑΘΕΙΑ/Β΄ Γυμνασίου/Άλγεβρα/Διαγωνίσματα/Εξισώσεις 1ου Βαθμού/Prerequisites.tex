\paragraph{Πράξεις με Ρητούς Αριθμούς}
\askhsh Να υπολογίσετε τις παρακάτω παραστάσεις:
\begin{enumerate}[label=\roman*)]
    \item $-5 + 8 - 3 + 12$
    \item $(-4) \cdot (-3) + (-10) : 2$
    \item $2 \cdot (3 - 7) - 4 \cdot (5 - 8)$
    \item $\frac{-12}{3} - \frac{-15}{-5}$
\end{enumerate}

\paragraph{Επιμεριστική Ιδιότητα - Απαλοιφή Παρενθέσεων}
\askhsh Να απλοποιήσετε τις παρακάτω αλγεβρικές παραστάσεις:
\begin{enumerate}[label=\roman*)]
    \item $3(x + 2) - 5(x - 1)$
    \item $-2(3x - 4) + 4(2x + 1)$
    \item $x - (2x - 3) + 4(x - 2)$
    \item $2(x - 3) - 3(2x + 4) + 10$
\end{enumerate}

\paragraph{Απλοποίηση Κλασμάτων}
\askhsh Να γράψετε σε απλούστερη μορφή τα παρακάτω κλάσματα:
\begin{enumerate}[label=\roman*)]
    \item $\frac{12x}{4}$
    \item $\frac{-15x + 10}{5}$
    \item $\frac{2(x + 3)}{6}$
\end{enumerate}
