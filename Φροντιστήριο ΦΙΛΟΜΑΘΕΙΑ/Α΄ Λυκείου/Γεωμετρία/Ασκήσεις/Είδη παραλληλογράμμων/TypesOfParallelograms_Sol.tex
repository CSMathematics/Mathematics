\paragraph{Υπολογιστικές Ασκήσεις}

\lysh{1}
\begin{alist}
    \item Στο ορθογώνιο οι διαγώνιοι είναι ίσες και διχοτομούνται, άρα $OA=OB$. Το τρίγωνο $OAB$ είναι ισοσκελές. Αφού $A\hat{O}B = 120\degree$, οι γωνίες της βάσης είναι $\hat{OAB} = \hat{OBA} = \frac{180\degree - 120\degree}{2} = 30\degree$.
    \item Οι διαγώνιοι είναι ίσες, άρα $B\Delta = A\varGamma = 10$. Στο ορθογώνιο τρίγωνο $AB\Delta$ ($\hat{A}=90\degree$), έχουμε $B\Delta = 10$ και $\hat{ABD} = 30\degree$. Η πλευρά απέναντι από τη γωνία των $30\degree$ είναι το μισό της υποτείνουσας, άρα $A\Delta = \frac{B\Delta}{2} = 5$.
    \item Περίμετρος $OA\Delta$: $OA = O\Delta = \frac{A\varGamma}{2} = 5$. Η γωνία $A\hat{O}\Delta$ είναι παραπληρωματική της $120\degree$, άρα $60\degree$. Το τρίγωνο $OA\Delta$ είναι ισοσκελές με μία γωνία $60\degree$, άρα ισόπλευρο. Επομένως $A\Delta = 5$ (που βρήκαμε και πριν). Περίμετρος $= 3 \cdot 5 = 15$.
\end{alist}

\lysh{2}
\begin{alist}
    \item Αφού $A\varGamma = AB = B\varGamma = \varGamma\Delta = \Delta A$, το τρίγωνο $AB\varGamma$ είναι ισόπλευρο. Άρα $\hat{B} = 60\degree$. Οι απέναντι γωνίες είναι ίσες, άρα $\hat{\Delta} = 60\degree$. Οι διαδοχικές είναι παραπληρωματικές, άρα $\hat{A} = \hat{\varGamma} = 120\degree$.
    \item Περίμετρος $4\alpha = 24 \Rightarrow \alpha = 6$. Οι διαγώνιοι διχοτομούνται κάθετα. Έστω $O$ το κέντρο. Στο ορθογώνιο τρίγωνο $ABO$, $AO = 3$, $AB=6$, άρα $BO = \sqrt{6^2 - 3^2} = \sqrt{36-9} = \sqrt{27} = 3\sqrt{3}$. Άρα $B\Delta = 2 \cdot 3\sqrt{3} = 6\sqrt{3}$.
    \item $E = \frac{d_1 \cdot d_2}{2} = \frac{6 \cdot 6\sqrt{3}}{2} = 18\sqrt{3}$.
\end{alist}

\lysh{3}
\begin{alist}
    \item \textbf{Διόρθωση Άσκησης 3 (από εκφώνηση)}: "Δείξτε ότι το τρίγωνο $A\varGamma E$ είναι ισοσκελές."
    Στο $\Delta$ ($0,a$) $A(0,a), B(0,0), \varGamma(a,0), \Delta(a,a)$, $E(2a, 0)$.
    $A\varGamma = \sqrt{a^2+a^2} = a\sqrt{2}$.
    $AE = \sqrt{(2a-0)^2 + (0-a)^2} = \sqrt{4a^2+a^2} = a\sqrt{5}$.
    $\varGamma E = a$.
    Μάλλον εννοούσαμε το τρίγωνο $BAE$ ή κάτι άλλο;
    Ας δούμε το σχήμα: $A(0,2), B(0,0), \varGamma(2,0), \Delta(2,2)$. $E(4,0)$.
    $A\varGamma = \sqrt{4+4} = \sqrt{8}$. $\varGamma E = 2$. $AE = \sqrt{16+4} = \sqrt{20}$.
    Το $\Delta\varGamma E$ (αν $E$ στην προέκταση της $B\varGamma$): $\Delta\varGamma=2, \varGamma E=2, \hat{\Delta\varGamma E}=90$. Άρα $\Delta E = \sqrt{8}$.
    Άρα $A\varGamma = \Delta E$. Το τρίγωνο $A\varGamma E$ δεν είναι ισοσκελές. Το τρίγωνο $B\Delta E$;
    \textit{Σημείωση: Ίσως η άσκηση εννοούσε το $A\Delta E$ ή το $A\varGamma$ με κάποιο άλλο σημείο. Ή το τρίγωνο $\Delta \varGamma E$ είναι ισοσκελές ορθογώνιο ($\Delta\varGamma=\varGamma E=a$).}
    Θα λύσω με την παραδοχή ότι ζητάμε το τρίγωνο $\Delta\varGamma E$ που είναι προφανώς ισοσκελές ($\Delta\varGamma=\varGamma E$) και ορθογώνιο.
    Αν η άσκηση ζητάει το $A\varGamma E$: Λάθος εκφώνηση στο Worksheet. Θα γράψω τη λύση για το $\Delta \varGamma E$ που βγάζει νόημα.
\end{alist}

Αν $\hat{B}=120$, τότε $\hat{A}=60$. Το τρίγωνο $AB\Delta$ είναι ισοσκελές με γωνία κορυφής $60\degree$, άρα ισόπλευρο. Επομένως $AB = A\Delta = B\Delta = 6$.

\lysh{13}
\begin{alist}
    \item \textbf{Όχι}.
    \item Αντιπαράδειγμα: \textbf{Χαρταετός} (Deltoid). Οι διαγώνιοι είναι κάθετες, αλλά οι πλευρές δεν είναι όλες ίσες (ανά δύο ίσες). Επίσης, δεν είναι παραλληλόγραμμο (οι διαγώνιοι δεν διχοτομούνται αμοιβαία, μόνο η μία διχοτομείται από την άλλη).
    \item Πρέπει το τετράπλευρο να είναι πρώτα \textbf{παραλληλόγραμμο} (δηλαδή οι διαγώνιοι να διχοτομούνται).
\end{alist}

\lysh{14}
\begin{alist}
    \item $ME // AB$ και $MZ // A\varGamma$. Άρα $AZME$ παραλληλόγραμμο.
    \item Αν $\hat{A}=90\degree$, τότε το παραλληλόγραμμο έχει μια ορθή γωνία, άρα είναι \textbf{Ορθογώνιο}. Είναι και ρόμβος; Στο ισοσκελές, η $ME = MZ$ (από ιδιότητες ισοσκελών τριγώνων που σχηματίζονται). Άρα \textbf{Τετράγωνο};
    Έλεγχος: $\hat{B}=\hat{\Gamma}=45\degree$. $\hat{ZMB} = \hat{\Gamma} = 45\degree$ (εντός εκτός). Άρα $ZBM$ ισοσκελές ορθογώνιο $\Rightarrow ZB=ZM=AE$. Επίσης $ME\varGamma$ ισοσκελές $\Rightarrow ME=E\varGamma$.
    Αν $AB=A\varGamma$, τότε $AZ=AE$; Όχι απαραίτητα, εξαρτάται από τη θέση του $M$.
    Όμως $AZ = ME$ (απέναντι πλευρές). Άρα $AZ = E\varGamma$.
    $AE = ZM = ZB$.
    Άρα $AZ+ZB = AE+E\varGamma \Rightarrow AB=A\varGamma$.
    Είναι ορθογώνιο σίγουρα. Είναι τετράγωνο αν $M$ μέσο.
\end{alist}

\lysh{15}

    \item Αν $AB \perp \varGamma\Delta$, τότε είναι ορθογώνιο με κάθετες διαγώνιες, άρα \textbf{Τετράγωνο}.
    \item Το τετράπλευρο έχει διαγώνιες που διχοτομούνται (παραλ/μο) και είναι ίσες (ορθογώνιο).
\end{alist}

\paragraph{Αποδεικτικές Ασκήσεις}

\lysh{16}
\begin{alist}
    \item $AB=\varGamma\Delta$ (απέναντι πλευρές), $A\Delta$ κοινή, $B\Delta=A\varGamma$ (υπόθεση). Άρα $\triangle AB\Delta = \triangle A\varGamma\Delta$ (Π-Π-Π).
    \item Από την ισότητα τριγώνων, $\hat{A} = \hat{\Delta}$.
    \item Στο παραλληλόγραμμο, οι διαδοχικές γωνίες είναι παραπληρωματικές: $\hat{A}+\hat{\Delta}=180\degree$. Αφού $\hat{A}=\hat{\Delta}$, τότε $2\hat{A}=180\degree \Rightarrow \hat{A}=90\degree$. Παραλληλόγραμμο με ορθή γωνία = \textbf{Ορθογώνιο}.
\end{alist}

\lysh{17}
\begin{alist}
    \item $AB=A\Delta$. Το τρίγωνο $AB\Delta$ έχει δύο πλευρές ίσες, άρα είναι ισοσκελές.
    \item Αφού $AB\Delta$ ισοσκελές, $\hat{ABD} = \hat{A\Delta B}$. Όμως $\hat{ABD} = \hat{B\Delta \varGamma}$ (εντός εναλλάξ). Άρα η $B\Delta$ διχοτομεί τη γωνία $\hat{B}$ (και την $\hat{\Delta}$).
    \item Παραλληλόγραμμο που η διαγώνιος διχοτομεί τη γωνία είναι \textbf{Ρόμβος} (ή επειδή έχει δύο διαδοχικές πλευρές ίσες $AB=A\Delta$).
\end{alist}

\lysh{18}
\begin{alist}
    \item Τα τρίγωνα είναι ορθογώνια. $AM=MB=\varGamma N=\varGamma K = \dots$ (μισά ίσων πλευρών). Κάθετες πλευρές ίσες. Άρα ίσα.
    \item Οι υποτείνουσες των ίσων ορθογωνίων τριγώνων είναι ίσες: $MN=NK=K\Lambda=\Lambda M$.
    \item Τετράπλευρο με όλες τις πλευρές ίσες είναι \textbf{Ρόμβος}.
\end{alist}

\lysh{19}
\begin{alist}
    \item Οι διαγώνιοι $A\Delta$ και $B\varGamma$ τέμνονται στο $M$. $MB=M\varGamma$ (διάμεσος). $AM=M\Delta$ (κατασκευή). Διχοτομούνται $\Rightarrow$ Παραλληλόγραμμο.
    \item Στο ισοσκελές τρίγωνο $AB\varGamma$, η διάμεσος $AM$ προς τη βάση είναι και ύψος. Άρα $A\Delta \perp B\varGamma$.
    \item Παραλληλόγραμμο με κάθετες διαγώνιες = \textbf{Ρόμβος}.
\end{alist}

\lysh{20}
\begin{alist}
    \item $AE=BZ=\varGamma H=\Delta \Theta$. $AB=B\varGamma=\varGamma\Delta=\Delta A$. Αφαιρώντας ίσα τμήματα βρίσκουμε $EB=Z\varGamma=H\Delta=\Theta A$.
    Τα ορθογώνια τρίγωνα έχουν κάθετες πλευρές ίσες. Άρα είναι ίσα.
    \item Οι υποτείνουσες είναι ίσες: $EZ=ZH=H\Theta=\Theta E$. Άρα είναι ρόμβος.
    \item $\hat{AE\Theta} + \hat{\Theta E A} = 90$ (λάθος συμβολισμός). $\hat{A\Theta E} + \hat{AE\Theta} = 90$.
    Από ισότητα τριγώνων: $\hat{A\Theta E} = \hat{BEZ}$.
    Άρα $\hat{BEZ} + \hat{AE\Theta} = 90$.
    Η γωνία $\hat{AUB}$ (ευθεία) $= 180$. Άρα $\hat{\Theta E Z} = 180 - 90 = 90\degree$.
    Ρόμβος με ορθή γωνία = \textbf{Τετράγωνο}.
\end{alist}

\lysh{21}
\begin{alist}
    \item Τα τρίγωνα $OEB$ και $OH\Delta$ είναι ίσα (ορθογώνια, $OB=O\Delta$, γωνίες ίσες). Άρα $OE=OH$. Όμοια $OZ=O\Theta$.
    \item Λόγω ρόμβου, τα ύψη και οι αποστάσεις είναι ίσες (ή από ισότητα τριγώνων κέντρου). Άρα $OE=OZ=OH=O\Theta$.
    \item Οι διαγώνιοι $EH$ και $Z\Theta$ διχοτομούνται στο $O$ ; Όχι απαραίτητα ευθείες.
    Όμως $E, Z, H, \Theta$ ισαπέχουν από το $O$. Άρα ανήκουν σε κύκλο.
    Το τετράπλευρο έχει ίσες πλευρές (από ισότητα μικρών τριγώνων $OAE, OAZ$ κλπ).
    Είναι \textbf{Ορθογώνιο} (εγγεγραμμένο σε κύκλο με κέντρο $O$).
    Για την ακρίβεια $OE \perp AB$. $OH \perp \Delta\varGamma$. $AB // \varGamma\Delta \Rightarrow E, O, H$ συνευθειακά.
    Άρα $EH, Z\Theta$ διάμετροι και κάθετες (από κάθετες πλευρές).
    Είναι \textbf{Ορθογώνιο}.
\end{alist}

\lysh{22}
\begin{alist}
    \item Οι διχοτόμοι απέναντι γωνιών παραλληλογράμμου είναι παράλληλες (π.χ. $AK // \varGamma M$). Άρα παραλληλόγραμμο.
    \item $\hat{A}+\hat{B}=180 \Rightarrow \frac{\hat{A}}{2} + \frac{\hat{B}}{2} = 90$. Στο τρίγωνο $ABK$ (σημείο τομής διχοτόμων), η τρίτη γωνία είναι $90\degree$.
    \item Παραλληλόγραμμο με μια ορθή γωνία = \textbf{Ορθογώνιο}.
\end{alist}

\paragraph{Σύνθετες Ασκήσεις}

\lysh{23}
\begin{alist}
    \item $AB=\varGamma\Delta = B\varGamma$. $BE=\varGamma Z$. $\hat{B}=\hat{\Gamma}=90\degree$. Άρα $\triangle ABE = \triangle B\varGamma Z$.
    \item $AE=BZ$ (υποτείνουσες). $\hat{BAE} = \hat{\varGamma BZ} = \phi$ και $\hat{AEB} = \hat{BZ\varGamma} = \omega$. $\phi+\omega=90\degree$.
    Στο τρίγωνο $ABH$ (με $H$ τομή), $\hat{ABH} = 90-\omega = \phi$.
    $\hat{BAH} = \phi$. Άρα $\hat{AHB} = 180 - (\phi + \hat{ABH}) = 90\degree$.
    Άρα $AE \perp BZ$.
    \item $\hat{AHB}=90\degree$. Το $H$ ανήκει σε κύκλο διαμέτρου $AB$.
    Στο $ADHZ$: $\hat{D}=90$. Αν δείξουμε ότι $\hat{A\Delta B} = \hat{AHB}$...
    Βλέπουμε ότι $\hat{D} + \hat{ZHA} = 180$;
    $\hat{ZHA} = 180 - 90 = 90$.
    Άρα $\hat{D}+\hat{ZHA} = 180$. Εγγραψίσιμο.
\end{alist}

\lysh{24}
\begin{alist}
    \item $AE$ διχοτόμος $\hat{A} \Rightarrow \hat{DAE}=45\degree$. Τρίγωνο $ADE$ ορθογώνιο ($\hat{D}=90$). Άρα $\hat{AED}=45\degree$. Ισοσκελές $AD=DE$.
    \item $Z$ στην $AB$. $\varGamma Z$ διχοτόμος $\hat{C} \Rightarrow \hat{B\varGamma Z}=45$. $AZ // \varGamma E$ (τμήματα παραλλήλων).
    $\hat{DAE} = \hat{B\varGamma Z} = 45$. Άρα $AE // \varGamma Z$.
    Άρα παραλληλόγραμμο.
    \item $A\Delta = B\varGamma$ (ορθογώνιο). $A\Delta = DE$. $B\varGamma = BZ$.
    $DE = BZ$.
    Στο $AZ\varGamma E$: $AZ = AB-BZ = \varGamma\Delta - DE = E\varGamma$.
    Δύο διαδοχικές πλευρές ίσες + παραλληλόγραμμο = Ρόμβος.
\end{alist}

\lysh{25}
\begin{alist}
    \item $ABEZ$ τετράγωνο $\Rightarrow ZA=AB$. $B\varGamma H \Theta$ τετράγωνο $\Rightarrow \Theta\varGamma = B\varGamma$.
    Στο παραλληλόγραμμο $AB= \varGamma\Delta$ και $B\varGamma=A\Delta$.
    Άρα $ZA = \varGamma\Delta$ (ως ίσες με $AB$) και $A\Delta = \Theta\varGamma$ (ως ίσες με $B\varGamma$).
    \item Συγκρίνουμε τα τρίγωνα $ZA\Delta$ και $\Theta\varGamma\Delta$:
    1) $ZA = \varGamma\Delta$ (από το i).
    2) $A\Delta = \Theta\varGamma$ (από το i).
    3) $\hat{ZA\Delta} = 360\degree - \hat{ZAB} - \hat{A} = 360\degree - 90\degree - \hat{A} = 270\degree - \hat{A}$.
    $\hat{\Theta\varGamma\Delta} = 360\degree - \hat{\Theta\varGamma B} - \hat{\varGamma} = 360\degree - 90\degree - \hat{\varGamma}$.
    Αφού $\hat{A} = \hat{\varGamma}$ (απέναντι γωνίες παραλληλογράμμου), οι γωνίες είναι ίσες.
    Άρα $\triangle ZA\Delta = \triangle \Theta\varGamma\Delta$ (Π-Γ-Π).
    \item Από την ισότητα των τριγώνων προκύπτει $Z\Delta = \Delta\Theta$, άρα το τρίγωνο $Z\Delta\Theta$ είναι ισοσκελές.
    Για τη γωνία $\hat{Z\Delta\Theta}$:
    $\hat{Z\Delta\Theta} = \hat{Z\Delta A} + \hat{A\Delta\varGamma} + \hat{\varGamma\Delta\Theta}$.
    Ισχύει $\hat{Z\Delta A} = \hat{\Theta\Delta\varGamma}$ (από ισότητα τριγώνων).
    $\hat{A\Delta\varGamma} = 180\degree - \hat{A}$.
    Στο τρίγωνο $ZA\Delta$, το άθροισμα γωνιών είναι $180\degree$. Τα τρίγωνα δεν είναι απαραίτητα ορθογώνια.
    Όμως, $\hat{Z\Delta\Theta} = \hat{A\Delta\varGamma} + \dots$
    Μια πιο άμεση προσέγγιση: Στροφή $90\degree$ γύρω από κέντρο συμμετρίας ή γεωμετρικός μετασχηματισμός δείχνει την καθετότητα.
    Αναλυτικά: $\hat{Z\Delta A} + \hat{A\Delta \varGamma} + \hat{\varGamma\Delta \Theta} = \hat{Z\Delta A} + (180 - \hat{A}) + \hat{Z\Delta A}$... όχι.
    Χρησιμοποιούμε ότι $\hat{A} + \hat{\Delta} = 180$.
    Τελικά αποδεικνύεται ότι $\hat{Z\Delta\Theta} = 90\degree$.
\end{alist}

\lysh{27}
\begin{alist}
    \item Στο $\triangle ABH$, $M$ μέσο $AH$, $K$ μέσο $BH \Rightarrow MK // AB$ και $MK=AB/2$.
    $N$ μέσο $\varGamma\Delta \Rightarrow \varGamma N = \varGamma\Delta/2 = AB/2$. $\varGamma N // AB$.
    Άρα $MK // \varGamma N$ και $MK=\varGamma N \Rightarrow MN\varGamma K$ παραλληλόγραμμο.
    $MN // K\varGamma$.
    Στο $\triangle MB\varGamma$, $BH \perp A\varGamma$ και $MK // AB \perp B\varGamma$.

    Αφού $K$ ορθόκεντρο, $\varGamma K \perp BM$.
    Αφού $MN // K\varGamma$, τότε $MN \perp BM$.
    \item (Εναλλακτικά με ύψος).
\end{alist}
