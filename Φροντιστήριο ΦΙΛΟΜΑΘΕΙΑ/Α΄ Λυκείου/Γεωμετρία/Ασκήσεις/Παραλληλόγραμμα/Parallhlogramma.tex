\begin{enumerate}[label=\roman*)]

    \item Υπολογίστε τις γωνίες $\hat{Β}$ και $\hat{\Delta}$.

    \item Βρείτε την τιμή του $x$.
    \item Υπολογίστε το μήκος των πλευρών $ΑΒ$ και $\varGamma\Delta$.
\end{enumerate}

\askhsh Η περίμετρος ενός παραλληλογράμμου $ΑΒ\varGamma\Delta$ είναι $44$ cm. Η πλευρά $ΑΒ$ είναι κατά $4$ cm μεγαλύτερη από την $Β\varGamma$.
\begin{enumerate}[label=\roman*)]
    \item Υπολογίστε τα μήκη των πλευρών $ΑΒ$ και $Β\varGamma$.
    \item Βρείτε τα μήκη των $\varGamma\Delta$ και $\Delta Α$.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, η γωνία $\hat{Β}$ είναι τριπλάσια της $\hat{Α}$.
\begin{enumerate}[label=\roman*)]
    \item Υπολογίστε τις γωνίες $\hat{Α}$ και $\hat{Β}$.
    \item Δείξτε ότι $\hat{Β} + \hat{\varGamma} = 180\degree$.
\end{enumerate}

    \item Δείξτε ότι $ΒΕ$ είναι ίση και παράλληλη προς την $\varGamma\Delta$.
    \item Τι είδους τετράπλευρο είναι το $Β\varGamma E\Delta$;
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, φέρνουμε τις διχοτόμους των γωνιών $\hat{Α}$ και $\hat{Β}$ που τέμνονται στο $Μ$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι $\frac{\hat{Α}}{2} + \frac{\hat{Β}}{2} = 90\degree$.
    \item Συμπεράνατε ότι το τρίγωνο $ΑΜΒ$ είναι ορθογώνιο.
\end{enumerate}

\askhsh Αν σε ένα τετράπλευρο $ΑΒ\varGamma\Delta$ ισχύει $ΑΒ = \Gamma\Delta$ και $Α\Delta = Β\Gamma$:
\begin{enumerate}[label=\roman*)]
    \item Συγκρίνετε τα τρίγωνα $ΑΒ\varGamma$ και $\varGamma\Delta Α$.
    \item Αποδείξτε ότι το $ΑΒ\varGamma\Delta$ είναι παραλληλόγραμμο.
\end{enumerate}

\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι $ΑΕ // \varGamma Ζ$ και $ΑΕ = \varGamma Ζ$.
    \item Αποδείξτε ότι το τετράπλευρο $ΑΕ\varGamma Ζ$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, οι διχοτόμοι των γωνιών $\hat{Α}$ και $\hat{\varGamma}$ τέμνουν τη διαγώνιο $Β\Delta$ στα $Ε$ και $Ζ$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι τα τρίγωνα $Α\Delta Ε$ και $\varGamma Β Ζ$ είναι ίσα.
    \item Αποδείξτε ότι το τετράπλευρο $ΑΕ\varGamma Ζ$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Δύο παραλληλόγραμμα $ΑΒ\varGamma\Delta$ και $ΑΒΕΖ$ έχουν κοινή την πλευρά $ΑΒ$ και βρίσκονται σε διαφορετικά ημιεπίπεδα ως προς αυτήν.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι τα τμήματα $\varGamma\Delta$ και $ΕΖ$ είναι παράλληλα και ίσα προς το $ΑΒ$.
    \item Αποδείξτε ότι το τετράπλευρο $\Delta\varGamma Ε Ζ$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Δύο παραλληλόγραμμα $ΑΒ\varGamma\Delta$ και $ΑΒΕΖ$ έχουν κοινή την πλευρά $ΑΒ$.
\begin{enumerate}[label=\roman*)]

    \item Αποδείξτε ότι το $\Delta\varGamma ΕΖ$ είναι παραλληλόγραμμο.

\paragraph{Προχωρημένες Ασκήσεις}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, οι διχοτόμοι των γωνιών $\hat{Α}$ και $\hat{Β}$ τέμνονται στο $Μ$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι $Α\hat{Μ}Β = 90^\circ$.
    \item Αν η προέκταση της $ΑΜ$ τέμνει την πλευρά $Β\varGamma$ στο $Ν$, δείξτε ότι $ΒΝ = ΑΒ$.
\end{enumerate}

\askhsh Δίνεται παραλληλόγραμμο $ΑΒ\varGamma\Delta$. Στην πλευρά $ΑΒ$ παίρνουμε τμήμα $ΑΕ = Α\Delta$ και στην $\varGamma\Delta$ τμήμα $\varGamma Ζ = \varGamma Β$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι τα τρίγωνα $Α\Delta Ε$ και $\varGamma Β Ζ$ είναι ισοσκελή και ίσα.
    \item Αποδείξτε ότι το $ΕΒΖ\Delta$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, οι διχοτόμοι των γωνιών $\hat{Α}, \hat{\Delta}$ τέμνονται στο $Κ$, και των $\hat{Β}, \hat{\varGamma}$ στο $Λ$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι $Α\hat{Κ}\Delta = 90^\circ$ και $Β\hat{Λ}\varGamma = 90^\circ$.
    \item Αποδείξτε ότι το τμήμα $ΚΛ$ είναι παράλληλο στην $ΑΒ$.
\end{enumerate}

\askhsh Στις πλευρές παραλληλογράμμου $ΑΒ\varGamma\Delta$ παίρνουμε τμήματα $ΑΕ = \varGamma Ζ$ στην $ΑΒ, \varGamma\Delta$ και $ΒΗ = \Delta \Theta$ στην $Β\varGamma, \Delta Α$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι τα τρίγωνα $ΕΒΗ$ και $Ζ\Delta\Theta$ είναι ίσα.
    \item Αποδείξτε ότι το $ΕΗΖ\Theta$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, η διχοτόμος της γωνίας $\hat{Α}$ τέμνει την πλευρά $\varGamma\Delta$ στο $Ε$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι το τρίγωνο $Α\Delta Ε$ είναι ισοσκελές.
    \item Αν η $ΒΕ$ είναι διχοτόμος της $\hat{Β}$, δείξτε ότι το $Ε$ είναι μέσο της $\varGamma\Delta$.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, $Μ$ είναι το μέσο της $Α\Delta$. Φέρνουμε $ΒΕ // Α\varGamma$ με το $Ε$ στην προέκταση της $\varGamma\Delta$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι το τετράπλευρο $ΑΒΕ\varGamma$ είναι παραλληλόγραμμο.
    \item Δείξτε ότι $\Delta\varGamma = \varGamma Ε$.
\end{enumerate}

\askhsh Δίνεται παραλληλόγραμμο $ΑΒ\varGamma\Delta$ με κέντρο $Ο$. Από το $Ο$ φέρνουμε δύο τυχαίες ευθείες που τέμνουν τις πλευρές στα $Κ, \Lambda, Μ, Ν$.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι το $Ο$ διχοτομεί τα τμήματα $Κ\Lambda$ και $ΜΝ$.
    \item Αποδείξτε ότι το $ΚΜ\Lambda Ν$ είναι παραλληλόγραμμο.
\end{enumerate}

\askhsh Αν σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$ ισχύει $ΑΒ = 2Α\Delta$ και $Μ$ είναι το μέσο της $ΑΒ$:
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι η $\Delta Μ$ είναι διχοτόμος της γωνίας $\hat{\Delta}$.
    \item Δείξτε ότι η γωνία $\Delta\hat{Μ}\varGamma$ είναι ορθή.
\end{enumerate}

\askhsh Σε παραλληλόγραμμο $ΑΒ\varGamma\Delta$, $Μ$ και $Ν$ είναι τα μέσα των $Β\varGamma$ και $\varGamma\Delta$ αντίστοιχα.
\begin{enumerate}[label=\roman*)]
    \item Δείξτε ότι οι $ΑΜ, ΑΝ$ τέμνουν τη διαγώνιο $Β\Delta$ σε τρία ίσα τμήματα.
\end{enumerate}
