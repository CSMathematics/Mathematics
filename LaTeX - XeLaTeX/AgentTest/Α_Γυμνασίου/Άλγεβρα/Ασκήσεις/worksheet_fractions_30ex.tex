\paragraph{Ενότητα 1: Η έννοια του κλάσματος}

    \askhsh Να γράψετε σε μορφή κλάσματος τα παρακάτω:
    \begin{alist}
        \item Το μισό του 1
        \item Τα δύο τρίτα της ώρας
        \item Το πηλίκο της διαίρεσης $3 : 4$
        \item Το πηλίκο της διαίρεσης $7 : 10$
    \end{alist}

    \askhsh Ποιο μέρος του σχήματος είναι χρωματισμένο αν:
    \begin{alist}
        \item Ένα τετράγωνο χωριστεί σε 4 ίσα μέρη και χρωματίσουμε τα 3.
        \item Ένας κύκλος χωριστεί σε 8 ίσα μέρη και χρωματίσουμε τα 2.
        \item Ένα ορθογώνιο χωριστεί σε 10 ίσα μέρη και χρωματίσουμε τα 5.
        \item Μια σοκολάτα έχει 12 κομμάτια και φάμε τα 4.
    \end{alist}

    \askhsh Να μετατρέψετε τους παρακάτω μεικτούς αριθμούς σε κλάσματα:
    \begin{alist}
        \item $2 \frac{1}{3}$
        \item $4 \frac{2}{5}$
        \item $1 \frac{7}{8}$
        \item $3 \frac{1}{2}$
    \end{alist}

    \askhsh Να μετατρέψετε τα παρακάτω καταχρηστικά κλάσματα σε μεικτούς αριθμούς:
    \begin{alist}
        \item $\frac{7}{2}$
        \item $\frac{13}{4}$
        \item $\frac{25}{3}$
        \item $\frac{10}{3}$
    \end{alist}

\paragraph{Ενότητα 2: Ισοδύναμα Κλάσματα - Απλοποίηση}


    \askhsh Να ελέγξετε αν τα παρακάτω ζεύγη κλασμάτων είναι ισοδύναμα:
    \begin{alist}
        \item $\frac{2}{3}$ και $\frac{4}{6}$
        \item $\frac{3}{5}$ και $\frac{9}{15}$
        \item $\frac{4}{7}$ και $\frac{5}{8}$
        \item $\frac{12}{20}$ και $\frac{3}{5}$
    \end{alist}

    \askhsh Να απλοποιήσετε τα παρακάτω κλάσματα ώστε να γίνουν ανάγωγα:
    \begin{alist}
        \item $\frac{8}{12}$
        \item $\frac{15}{20}$
        \item $\frac{24}{36}$
        \item $\frac{30}{45}$
    \end{alist}

    \askhsh Να βρείτε τον αριθμό $x$ ώστε να ισχύουν οι ισότητες:
    \begin{alist}
        \item $\frac{2}{5} = \frac{x}{15}$
        \item $\frac{3}{x} = \frac{12}{20}$
        \item $\frac{x}{7} = \frac{18}{21}$
        \item $\frac{5}{9} = \frac{20}{x}$
    \end{alist}

    \askhsh Να μετατρέψετε τα παρακάτω κλάσματα σε δενκαδικούς αριθμούς:
    \begin{alist}
        \item $\frac{3}{10}$
        \item $\frac{7}{100}$
        \item $\frac{1}{2}$
        \item $\frac{3}{4}$
    \end{alist}

    \askhsh Να μετατρέψετε τους παρακάτω δεκαδικούς αριθμούς σε κλάσματα (και να τα απλοποιήσετε αν χρειάζεται):
    \begin{alist}
        \item $0,4$
        \item $0,25$
        \item $1,5$
        \item $0,05$
    \end{alist}

\paragraph{Ενότητα 3: Σύγκριση Κλασμάτων}

    \askhsh Να συγκρίνετε τα κλάσματα (μεγαλύτερο, μικρότερο ή ίσο):
    \begin{alist}
        \item $\frac{3}{7}$ και $\frac{5}{7}$
        \item $\frac{9}{11}$ και $\frac{6}{11}$
        \item $\frac{1}{2}$ και $\frac{1}{3}$
        \item $\frac{4}{5}$ και $\frac{4}{9}$
    \end{alist}

    \askhsh Να συγκρίνετε τα κλάσματα μετατρέποντάς τα σε ομώνυμα:
    \begin{alist}
        \item $\frac{2}{3}$ και $\frac{3}{4}$
        \item $\frac{5}{6}$ και $\frac{7}{8}$
        \item $\frac{1}{2}$ και $\frac{3}{5}$
        \item $\frac{2}{9}$ και $\frac{1}{3}$
    \end{alist}

    \askhsh Να διατάξετε τα κλάσματα σε αύξουσα σειρά:
    \[ \frac{1}{2}, \quad \frac{2}{3}, \quad \frac{5}{6}, \quad \frac{1}{4} \]

\paragraph{Ενότητα 4: Πράξεις Κλασμάτων}

    \askhsh Να υπολογίσετε τα αθροίσματα (ομώνυμα κλάσματα):
    \begin{alist}
        \item $\frac{2}{7} + \frac{3}{7}$
        \item $\frac{5}{12} + \frac{1}{12} + \frac{5}{12}$
        \item $\frac{8}{15} - \frac{2}{15}$
        \item $\frac{9}{10} - \frac{3}{10} - \frac{1}{10}$
    \end{alist}

    \askhsh Να κάνετε τις παρακάτω πράξεις (ετερώνυμα κλάσματα):
    \begin{alist}
        \item $\frac{1}{2} + \frac{1}{3}$
        \item $\frac{2}{3} + \frac{1}{4}$
        \item $\frac{3}{5} - \frac{1}{2}$
        \item $\frac{5}{6} - \frac{1}{3}$
    \end{alist}

    \askhsh Να υπολογίσετε τα παρακάτω:
    \begin{alist}
        \item $2 + \frac{1}{3}$
        \item $1 - \frac{3}{5}$
        \item $3 - \frac{1}{4}$
        \item $5 + \frac{2}{7}$
    \end{alist}

    \askhsh Να κάνετε τις πράξεις:
    \begin{alist}
        \item $2 \frac{1}{2} + 1 \frac{1}{3}$
        \item $3 \frac{1}{4} - 1 \frac{1}{2}$
        \item $1 \frac{2}{3} + 2 \frac{1}{5}$
        \item $4 \frac{3}{5} - 2 \frac{1}{10}$
    \end{alist}

    \askhsh Να υπολογίσετε την τιμή των παραστάσεων:
    \begin{alist}
        \item $A = \frac{1}{2} + \frac{1}{3} + \frac{1}{6}$
        \item $B = \frac{2}{3} - (\frac{1}{2} - \frac{1}{4})$
        \item $\Gamma = (1 - \frac{1}{2}) + (1 - \frac{1}{3})$
    \end{alist}

    \askhsh Αν $x = \frac{1}{2}$ και $y = \frac{1}{3}$, να υπολογίσετε την τιμή της παράστασης:
    \[ A = x + y - x \cdot y \]
    (Υπόδειξη: Πρώτα κάντε τον πολλαπλασιασμό που ακολουθεί)

    \askhsh Να υπολογίσετε τα γινόμενα:
    \begin{alist}
        \item $\frac{2}{3} \cdot \frac{5}{7}$
        \item $\frac{3}{4} \cdot \frac{8}{9}$
        \item $\frac{1}{2} \cdot \frac{4}{5}$
        \item $\frac{5}{6} \cdot 12$
    \end{alist}

    \askhsh Να βρείτε τους αντίστροφους των αριθμών:
    \begin{alist}
        \item $\frac{2}{3}$
        \item $4$ (ή $\frac{4}{1}$)
        \item $\frac{1}{5}$
        \item $1 \frac{1}{2}$
    \end{alist}

    \askhsh Να κάνετε τους πολλαπλασιασμούς:
    \begin{alist}
        \item $3 \cdot \frac{2}{9}$
        \item $\frac{5}{8} \cdot 4$
        \item $10 \cdot \frac{3}{5}$
        \item $\frac{7}{10} \cdot 100$
    \end{alist}

    \askhsh Να υπολογίσετε:
    \begin{alist}
        \item Το $\frac{1}{2}$ του 100
        \item Τα $\frac{3}{4}$ του 20
        \item Τα $\frac{2}{3}$ του 24
        \item Τα $\frac{4}{5}$ του κιλού (σε γραμμάρια)
    \end{alist}

    \askhsh Να κάνετε τις διαιρέσεις πολλαπλασιάζοντας με τον αντίστροφο:
    \begin{alist}
        \item $\frac{2}{3} : \frac{5}{7}$
        \item $\frac{3}{4} : \frac{1}{2}$
        \item $5 : \frac{1}{3}$
        \item $\frac{4}{5} : 2$
    \end{alist}

    \askhsh Να απλοποιήσετε τα σύνθετα κλάσματα:
    \begin{alist}
        \item $\frac{\frac{2}{3}}{\frac{4}{5}}$
        \item $\frac{\frac{1}{2}}{3}$
        \item $\frac{4}{\frac{2}{5}}$
        \item $\frac{\frac{3}{4}}{\frac{3}{8}}$
    \end{alist}

    \askhsh Να κάνετε τις πράξεις:
    \begin{alist}
        \item $\left( \frac{1}{2} + \frac{1}{3} \right) \cdot \frac{6}{5}$
        \item $\left( 2 - \frac{1}{3} \right) : \frac{5}{3}$
        \item $\frac{3}{4} \cdot \frac{2}{3} + \frac{1}{2} : \frac{1}{4}$
    \end{alist}

    \askhsh Ένα κατάστημα πούλησε το πρωί το $\frac{1}{3}$ των εμπορευμάτων του και το απόγευμα το $\frac{1}{4}$.
    \begin{alist}
        \item Τι μέρος των εμπορευμάτων πούλησε συνολικά;
        \item Τι μέρος των εμπορευμάτων έμεινε απούλητο;
    \end{alist}

    \askhsh Σε μια τάξη τα αγόρια είναι τα $\frac{3}{5}$ των μαθητών. Αν τα κορίτσια είναι 10, πόσοι είναι όλοι οι μαθητές της τάξης;

    \askhsh Από ένα βαρέλι που χωράει 20 λίτρα λάδι, βγάλαμε τα $\frac{2}{5}$ της ποσότητας.
    \begin{alist}
        \item Πόσα λίτρα βγάλαμε;
        \item Πόσα λίτρα έμειναν στο βαρέλι;
    \end{alist}
