\newcommand{\Trigwno}[6]{
	\tkzDefPoint (#1){A}
	\tkzDefPoint (#2){B}
	\tkzDefPoint (#3){C}
	\tkzDrawSegments[line width=0.5mm](A,B B,C C,A)
	\tkzDrawPoints[size=7,fill=white](A,B,C)
	\node (M) at ($1/3*(#1)+1/3*(#2)+1/3*(#3)$){};
	\node (O) at ($(M)+(0.1,0)$){};
	\tkzFindAngle (O,M,A) \tkzGetAngle{a}
	\tkzFindAngle (O,M,B) \tkzGetAngle{b}
	\tkzFindAngle (O,M,C) \tkzGetAngle{c}
\node at ([shift={(\a:3mm)}]A.\a) {{\footnotesize #4}};
\node at ([shift={(\b:3mm)}]B.\b) {{\footnotesize #5}};
\node at ([shift={(\c:3mm)}]C.\c) {{\footnotesize #6}};}
\Trigwno{0,3}{0,0}{3,0}{$ A $}{$ B $}{$ \varGamma $}
\end{tikzpicture}
