\begin{thema}
\item \mbox{}\\\vspace{-5mm}
\begin{erwthma}
\item Να δώσετε τον ορισμό της τριγωνομετρικής ταυτότητας.
\item Γράψτε τις σχέσεις (τύπους) με τους οποίους συνδέονται οι τριγωνομετρικοί αριθμοί δύο αντίθετων γωνιών.
\item Πως ορίζεται η συνάρτηση του ημιτόνου? Ποιο είναι το πεδίο ορισμού και ποιο το σύνολο τιμών της?
\item Ποιες συνθήκες πρέπει να πληροί μια συνάρτηση ώστε να είναι περιοδική?
\item Να χαρακτηρίσετε τις παρακάτω προτάσεις ως σωστές (Σωστή) ή λανθασμένες (Λάθος).
\begin{alist}
\item Η συνάρτηση $ f(x)=\ef{x} $ έχει πεδίο ορισμού το $ \mathbb{R} $.
\item Η ισότητα $ \hm{30\degree}+x=x+\frac{1}{2} $ αποτελεί τριγωνομετρική ταυτότητα.
\item Η συνάρτηση $ f(x)=\syn{x} $ έχει περίοδο $ 4\pi $.
\item Ισχύει η σχέση $ \hm\left(\frac{\pi}{2}-x\right)=\syn{x} $
\item Το σύνολο τιμών της συνάρτησης $ f(x)=3\hm{2x} $ είναι το $ [-3,3] $.
\end{alist}
\end{erwthma}
\item\mbox{}\\\vspace{-5mm}
\begin{erwthma}
\item Δίνεται γωνία $ \omega\in[0,\frac{\pi}{2}] $ για την οποία ισχύει $ \hm{\omega}=\frac{2\sqrt{5}}{5} $. Να βρεθούν οι υπόλοιποι τριγωνομετρικοί αριθμοί της $ \omega $.
\item
\begin{alist}
\item Να αποδείξετε την παρακάτω τριγωνομετρική ταυτότητα
\[ \frac{\hm{x}}{\ef^2{x}}+\ef{x}\cdot\syn{x}=\dfrac{1}{\hm{x}} \]
\item Να υπολογίσετε την τιμή της παράστασης
\[ A=\frac{\hm{\frac{\pi}{4}}}{\ef^2{\frac{\pi}{4}}}+\ef{\frac{\pi}{4}}\cdot\syn{\frac{\pi}{4}} \]
\end{alist}
\end{erwthma}
\item\mbox{}\\\vspace{-5mm}
\begin{erwthma}
\item Να υπολογίσετε τους τριγωνομετρικούς αριθμούς των παρακάτω γωνιών
\begin{multicols}{3}
\begin{alist}
\item $ 135\degree $
\item $ 930\degree $
\item $ \dfrac{7\pi}{4} $
\end{alist}
\end{multicols}
\item Να υπολογίσετε την τιμή της παρακάτω παράστασης.
\[ A=\hm{40\degree}\cdot\hm{140\degree}\cdot\hm{20}\degree-\syn{40\degree}\cdot\syn{140\degree}\cdot\syn{70\degree}-\hm{160\degree} \]
\item Να αποδείξετε ότι 
\[ \frac{\syn^2{\frac{27\pi}{4}}+\hm{\frac{16\pi}{3}}\cdot\ef{\left(-\frac{\pi}{6}\right)}}{\hm{\frac{53\pi}{6}}\cdot\syf{\frac{21\pi}{4}}}=2 \]
\end{erwthma}
\item Η θερμοκρασία μιας περιοχής σε βαθμούς κελσίου ($ \degree C $) κατά τη διάρκεια ενός εικοσιτετράωρου δίνεται κατά προσέγγιση από τη συνάρτηση:
\[ f(t)=-8\syn{\frac{\pi t}{12}+4}\ \ ,\ \ 0\leq t\leq 24\ \ \text{ ο χρόνος $ t $ σε ώρες} \]
\begin{erwthma}
\item Να βρείτε τη μέγιστη και την ελάχιστη θερμοκρασία κατά τη διάρκεια του
εικοσιτετράωρου, καθώς και την περίοδο της συνάρτησης.
\item Να παραστήσετε γραφικά την $ f $ για $ t\in[0,24] $.
\item Να βρείτε με τη βοήθεια της γραφικής παράστασης, σε ποια διαστήματα μέσα στη μέρα η θερμοκρασία αυξάνεται και σε ποια μειώνεται.
\item Να βρείτε με τη βοήθεια της γραφικής παράστασης, πότε η θερμοκρασία είναι πάνω από $ 0\degree C $.
\end{erwthma}
\end{thema}
