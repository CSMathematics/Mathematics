from sympy import *
var("x, y")
f = x + sin(x*y)
step1 = Integral(f, x, y)
step2 = Integral(Integral(f, x).doit(), y)
step3 = step2.doit()
\end{sympycode}

\sympy{step1} &= \sympy{step2} \\
&= \sympy{step3}
