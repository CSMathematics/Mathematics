% Define the Riemann Sum command
\NewDocumentCommand{\drawRiemannSum}{mmmm}{
    % Calculate rectangle width
    \pgfmathsetmacro{\dx}{(#4 - #3)/#2}

    % Draw rectangles (left Riemann sum)
    \foreach \i in {0,...,\numexpr#2-1} {
      \pgfmathsetmacro{\xleft}{#3 + \i * \dx}
      \addplot [
           ybar,
           bar width=\dx,
           fill=red!10,
           draw=black,
         ] coordinates {(\xleft + \dx/2, {#1})};
    }
}

\drawRiemannSum{4*x/(x^2+1)}{25}{-3}{3}
