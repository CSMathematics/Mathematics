\NewDocumentCommand{\Horner}{ s O{} m m }
  {
    % Το \group_begin: εξασφαλίζει ότι οι επιλογές χρώματος/πλάτους 
    % ισχύουν ΜΟΝΟ για αυτό το σχήμα Horner και μετά μηδενίζονται.
    \group_begin: 
    \keys_set:nn { silke / horner } { #2 }
    
    \IfBooleanTF{#1}
      { \bool_set_false:N \l_silke_show_bool }
      { \bool_set_true:N \l_silke_show_bool }
      
    \silke_horner:nn { #3 } { #4 }
    \group_end:
  }

\bool_new:N \l_silke_show_bool
\bool_new:N \l_silke_first_term_bool
\seq_new:N \l_silke_top_seq
\seq_new:N \l_silke_middle_seq
\seq_new:N \l_silke_bottom_seq
\seq_new:N \l_silke_temp_seq
\int_new:N \l_silke_degree_int
\int_new:N \l_silke_power_int
\tl_new:N \l_silke_remainder_tl
\tl_new:N \l_silke_quotient_tl
\tl_new:N \l_silke_tmpa_tl
\tl_new:N \l_silke_tmpb_tl

\cs_new_protected:Npn \silke_horner:nn #1 #2
  {
    \seq_set_split:Nnn \l_silke_top_seq { , } { #1 }
    \int_set:Nn \l_silke_degree_int { \seq_count:N \l_silke_top_seq }
    
    \seq_clear:N \l_silke_middle_seq
    \seq_clear:N \l_silke_bottom_seq
    
    % Διαβάζει δυναμικά το χρώμα του κουτιού από τις επιλογές του χρήστη
    \seq_put_right:Nx \l_silke_middle_seq 
      { \exp_not:N \cellcolor { \l_silke_box_tl } \exp_not:N }
    
    \seq_put_right:Nx \l_silke_bottom_seq
      {
        \fp_eval:n { \seq_item:Nn \l_silke_top_seq { 1 } }
      }
    
    \int_step_inline:nnnn { 2 } { 1 } { \l_silke_degree_int }
      {
        \seq_put_right:Nx \l_silke_middle_seq
          {
            \fp_eval:n { \seq_item:Nn \l_silke_bottom_seq { ##1 - 1 } * (#2) }
          }
        \seq_put_right:Nx \l_silke_bottom_seq
          {
            \fp_eval:n
              {
                \seq_item:Nn \l_silke_top_seq { ##1 } + \seq_item:Nn \l_silke_middle_seq { ##1 }
              }
          }
      }
    \silke_print_scheme:n { #2 }
  }

\cs_new_protected:Npn \silke_print_scheme:n #1
  {
    \bool_if:NF \l_silke_show_bool
      {
        \silke_phantom:N \l_silke_middle_seq
        \silke_phantom:N \l_silke_bottom_seq
      }
    
    \seq_pop_right:NN \l_silke_bottom_seq \l_silke_remainder_tl
    
    \par\vspace{1ex}\noindent
    % Διαβάζει δυναμικά το πλάτος της στήλης
    \begin{tabular}{ *{\int_eval:n {\l_silke_degree_int-1}}{|C{\l_silke_width_tl}} || C{\l_silke_width_tl} | C{\l_silke_rwidth_tl} | }
        \hline \rule[-1ex]{0pt}{4ex}
        \seq_use:Nn \l_silke_top_seq { & } & \rho=#1 \tabularnewline
        \hline \rule[-1ex]{0pt}{4ex}
        \seq_use:Nn \l_silke_middle_seq { & } \tabularnewline
        \cline{1-\l_silke_degree_int} \rule[-1ex]{0pt}{4ex}
        \seq_use:Nn \l_silke_bottom_seq { & } & 
        % Διαβάζει δυναμικά το χρώμα του υπολοίπου
        \textcolor{\l_silke_rem_color_tl}{\mathbf{\l_silke_remainder_tl}} \tabularnewline
        \cline{1-\l_silke_degree_int}
    \end{tabular}
    
    % --- Εμφάνιση του Πηλίκου (Αν επιτρέπεται ΚΑΙ από τις δύο συνθήκες) ---
    \bool_lazy_and:nnT
      { \l_silke_show_bool }
      { \l_silke_show_quotient_bool }
      {
        \par\vspace{1.5ex}\noindent
        $\pi(\l_silke_var_tl) = \silke_format_quotient:N \l_silke_bottom_seq$, \quad $\upsilon = \l_silke_remainder_tl$
        \vspace{1ex}
      }
  }

\cs_new_protected:Npn \silke_format_quotient:N #1
  {
    \tl_clear:N \l_silke_quotient_tl
    \bool_set_true:N \l_silke_first_term_bool
    
    \int_step_inline:nnnn { 1 } { 1 } { \seq_count:N #1 }
      {
        \tl_set:Nx \l_silke_tmpa_tl { \seq_item:Nn #1 { ##1 } }
        \fp_compare:nF { \l_silke_tmpa_tl == 0 }
          {
            \bool_if:NTF \l_silke_first_term_bool
              {
                \fp_compare:nT { \l_silke_tmpa_tl < 0 } { \tl_put_right:Nn \l_silke_quotient_tl { - } }
                \bool_set_false:N \l_silke_first_term_bool
              }
              {
                \fp_compare:nTF { \l_silke_tmpa_tl < 0 }
                  { \tl_put_right:Nn \l_silke_quotient_tl { - } }
                  { \tl_put_right:Nn \l_silke_quotient_tl { + } }
              }
            
            \tl_set:Nx \l_silke_tmpb_tl { \fp_eval:n { abs(\l_silke_tmpa_tl) } }
            \int_set:Nn \l_silke_power_int { \seq_count:N #1 - ##1 }
            
            \bool_lazy_and:nnTF 
              { \fp_compare_p:n { \l_silke_tmpb_tl == 1 } }
              { \int_compare_p:n { \l_silke_power_int > 0 } }
              { }
              { \tl_put_right:Nx \l_silke_quotient_tl { \l_silke_tmpb_tl } }
            
            \int_compare:nT { \l_silke_power_int > 0 }
              {
                \tl_put_right:NV \l_silke_quotient_tl \l_silke_var_tl
                \int_compare:nT { \l_silke_power_int > 1 }
                  { \tl_put_right:Nx \l_silke_quotient_tl { ^{\int_use:N \l_silke_power_int} } }
              }
          }
      }
    \tl_if_empty:NTF \l_silke_quotient_tl 
      { 0 }
      { \tl_use:N \l_silke_quotient_tl }
  }

\cs_new_protected:Npn \silke_phantom:N #1

    \seq_set_eq:NN #1 \l_silke_temp_seq
  }

Η εντολή \texttt{\textbackslash Horner} δέχεται τα εξής ορίσματα: \\
%\texttt{\textbackslash Horner*[πλάτος\_στήλης]\{συντελεστές\}\{ρίζα $\rho$\}}

\vspace{1cm}

\section*{1. Βασική Χρήση (Πλήρως Αυτοματοποιημένη)}
Η πιο απλή μορφή της εντολής. Διαβάζει τους συντελεστές του πολυωνύμου, υπολογίζει τις πράξεις, χρωματίζει το υπόλοιπο και τυπώνει αυτόματα το πηλίκο $\Pi(x)$. \\
\textbf{Πολυώνυμο:} $P(x) = x^3 - 3x^2 + 2$ \\
\textbf{Διαίρεση με:} $x - 2$ (Άρα $\rho = 2$) \\
\textbf{Κώδικας:} \verb|\Horner{1, -3, 0, 2}{2}| \\
\Horner{1, -3, 0, 2}{2}

\vspace{0.5cm}

\section*{2. Απουσία Όρων (Μηδενικά)}
Αν το πολυώνυμο δεν έχει κάποιους ενδιάμεσους όρους, πρέπει να βάλουμε $0$ στη θέση τους. Η μακροεντολή είναι αρκετά "έξυπνη" ώστε στο τελικό πηλίκο να μην τυπώσει τους μηδενικούς όρους (π.χ. δεν θα γράψει $0x^2$). \\
\textbf{Πολυώνυμο:} $P(x) = x^4 + x^2 - 5$ \\
\textbf{Διαίρεση με:} $x + 1$ (Άρα $\rho = -1$) \\
\textbf{Κώδικας:} \verb|\Horner{1, 0, 1, 0, -5}{-1}| \\
\Horner{1, 0, 1, 0, -5}{-1}

\vspace{0.5cm}

\section*{3. Για Φυλλάδια Μαθητών (Η παράμετρος \texttt{*})}
Προσθέτοντας ένα αστεράκι (\texttt{*}) αμέσως μετά την εντολή, το σχήμα γίνεται "άλυτο". Οι ενδιάμεσες πράξεις και το υπόλοιπο κρύβονται, ενώ η αυτόματη εκτύπωση του $\Pi(x)$ απενεργοποιείται. Ιδανικό για ασκήσεις προς λύση. \\
\textbf{Κώδικας:} \verb|\Horner*{1, -5, 6, -2}{3}| \\
\Horner*{1, -5, 6, -2}{3}

\vspace{0.5cm}

\section*{4. Υποστήριξη Δεκαδικών Αριθμών}
Χάρη στη μηχανή FPU του \LaTeX3, η εντολή δέχεται ρίζες με δεκαδικούς (π.χ. $0.5$). Σε αυτή την περίπτωση, καλό είναι να αυξήσουμε το πλάτος των στηλών (με το προαιρετικό όρισμα \texttt{[]}), ώστε να χωράνε οι δεκαδικοί αριθμοί άνετα. \\
\textbf{Κώδικας:} \verb|\Horner[width=2.5em]{2, -1.5, 3, -1}{0.5}| \\
\Horner[width=2.5em]{2, -1.5, 3, -1}{0.5}

\vspace{0.5cm}

\section*{5. Μεγάλοι Αριθμοί \& Προσαρμογή Πλάτους}
Αν έχουμε μεγάλους συντελεστές (π.χ. εκατοντάδες), το προεπιλεγμένο πλάτος της στήλης (\texttt{2em}) ίσως να μην αρκεί. Το αυξάνουμε γράφοντας \texttt{[3.5em]} ή όσο χρειάζεται. \\
\textbf{Κώδικας:} \verb|\Horner[width=3.5em]{100, -250, 50, 0}{2}| \\
\Horner[width=3.5em]{100, -250, 50, 0}{2}

\section*{6. Έξυπνη Εμφάνιση Πηλίκου}
Η εντολή διαβάζει την κάτω γραμμή του σχήματος (αφαιρώντας το υπόλοιπο) και τυπώνει αυτόματα το πηλίκο $\Pi(x)$. Ο αλγόριθμος αναγνωρίζει πότε ο συντελεστής είναι $1$ ή $-1$ (και τυπώνει μόνο $x$ ή $-x$), καθώς και πότε ένας όρος είναι $0$ (οπότε τον παραλείπει τελείως). \\
\textbf{Παράδειγμα:} Διαίρεση του $2x^4 - x^3 - 3x^2 + x - 1$ με το $x - 1$. \\
Το πηλίκο θα βγει $2x^3 + x^2 - 2x - 1$ (πρόσεξε πώς το $1x^2$ γράφεται σωστά ως $x^2$). \\
\textbf{Κώδικας:} \verb|\Horner{2, -1, -3, 1, -1}{1}| \\
\Horner{2, -1, -3, 1, -1}{1}

\vspace{0.5cm}

\section*{7. Προσαρμογή Χρωμάτων \& Κουτιού}
Μπορείς εύκολα να αλλάξεις το χρώμα του τελικού υπολοίπου, καθώς και το στυλ του "κενού" κουτιού, επαναπροσδιορίζοντας τις αντίστοιχες εντολές πριν καλέσεις το \verb|\Horner|.

\textbf{Κώδικας:}
\begin{verbatim}
% Αλλάζουμε το χρώμα του υπολοίπου σε μπλε
\renewcommand{\HornerRemainderColor}{blue}

% Αλλάζουμε το κενό κουτί σε σκέτο μαύρο
\renewcommand{\HornerBox}{\cellcolor{black}}

\Horner{1, -3, 0, 2}{2}
\end{verbatim}

\renewcommand{\HornerRemainderColor}{blue}
\renewcommand{\HornerBox}{\cellcolor{black}}
\Horner[quotient=false,box=blue,remainder=cyan]{1, -3, 0, 2}{2}

% Επαναφορά στις αρχικές ρυθμίσεις για επόμενα σχήματα

\section*{8. Επιλογή Μεταβλητής}
Μπορείς να επιλέξεις το όνομα της μεταβλητής που θα εμφανίζεται στο πηλίκο, χρησιμοποιώντας την επιλογή \texttt{var}. Η προεπιλεγμένη τιμή είναι \texttt{x}. \\
\textbf{Κώδικας:} \verb|\Horner[var=y]{1, -3, 0, 2}{2}| \\
\Horner[var=y]{1, -3, 0, 2}{2}
