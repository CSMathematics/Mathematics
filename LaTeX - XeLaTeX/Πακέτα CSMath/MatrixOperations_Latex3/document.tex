\paragraph{Matrix definition and printing}
\csmDefinematrix{A}{2}{3}{1, -1, -2, 1, 0, 3}
\csmDefinematrix{B}{2}{3}{1, 2, 0, 1, 3, -2}
\csmDefinematrix{D}{3}{3}{8,4,0,4,-1,5,1,-1,1}
\csmDefinematrix{F}{3}{3}{1,2,-1,0,4,-2,3,1,0}
\csmMatrixIdentity{E}{3}
\[A=\printmatrix{A},\ B=\printmatrix{B},\ D=\printmatrix{D},\ I_3=\printmatrix{E}\]
\paragraph{Matrix addition - Linear combination}
\csmMatrixSum{C}{A,B}
$C=A+B=\printmatrix{A}+\printmatrix{B}=\printmatrix{C}$\\
\csmMatrixLinComb{J}[2]{A}[-3]{B}
$J=2A-3B=2\printmatrix{A}-3\printmatrix{B}=\printmatrix{J}$\\
\csmMatrixSub{K}{A}{B}
$K=A-B=\printmatrix{A}-\printmatrix{B}=\printmatrix{K}$
\paragraph{Matrix multiplication}
\csmMatrixMultiplication{H}{A}{D}
$H=A\cdot D=\printmatrix{A}\cdot\printmatrix{D}=\printmatrix{H}$
\paragraph{Transpose matrix}
\csmTransposeMatrix{H_T}{H}
$H_T=\printmatrix{H}^T=\printmatrix{H_T}$

\paragraph{Trace - Determinant}
Matrix $D$ is \csmIsMatrixSquare{D}. The trace is $\mathrm{tr}{D}=\csmMatrixTrace{D}$. The determinant is \[\det{D}=\printmatrix[style=vmatrix]{D}=\csmDeterminant{D}\]
\paragraph{Matrix power}
\csmMatrixPower{G}{F}{2}
$F=\printmatrix{F}\Rightarrow G=F^2=\printmatrix{G}$
