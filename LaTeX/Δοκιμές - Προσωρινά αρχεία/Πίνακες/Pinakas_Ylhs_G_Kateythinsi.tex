\tcbsubtitle{Τυπολόγιο - Συμβολισμοί\ \ \faFile*}
\begin{enumerate}[itemsep=0mm,leftmargin=5mm]
\begin{multicols}{2}
\item Συνάρτηση : $ f $
\item Πεδίο ορισμού της συνάρτησης $ f $ : $ D_f $
\item $ x : $ ανεξάρτητη μεταβλητή και $ y: $ εξαρτημένη μεταβλητή
\item Τύπος συνάρτησης $ y=f(x) $
\end{multicols}
\item Σύνολο τιμών : $ f(D_f)=\{y\in\mathbb{R}: y=f(x)\ ,\ x\in D_f\}$
\item Γραφική παράσταση συνάρτησης : $ C_f=\{ (x,y): y=f(x)\ ,\ x\in D_f \}$
\begin{multicols}{2}
\item Άρτια συνάρτηση
\begin{itemize}[leftmargin=4mm]
\item Για κάθε $ x\in D_f\Rightarrow -x\in D_f $
\item $ f(-x)=f(x) $ για κάθε $ x\in D_f $
\end{itemize}
\item Περιττή συνάρτηση
\begin{itemize}[leftmargin=3mm]
\item Για κάθε $ x\in D_f\Rightarrow -x\in D_f $
\item $ f(-x)=-f(x) $ για κάθε $ x\in D_f $
\end{itemize}
\end{multicols}
\item Περιοδική συνάρτηση
\begin{itemize}[leftmargin=3mm]
\item Για κάθε $ x\in D_f\Rightarrow x+T\in D_f $ και $ x-T\in D_f $
\item $ f(x)=f(x+T)=f(x-T) $ για κάθε $ x\in D_f $
\end{itemize}
\begin{multicols}{2}
\item Κατακόρυφη μετατόπιση $ g(x)=f(x)\pm c $
\item Οριζόντια μετατόπιση $ g(x)=f(x\pm c) $
\end{multicols}
\item Το σημείο $ A(a,\beta) $ ανήκει στη $ C_f \Leftrightarrow f(a)=\beta$
\item Σημεία τομής της $ C_f $ με τον άξονα
\begin{itemize}[leftmargin=4mm]
\item $ x'x : A(\rho,0)$ όπου $ \rho : $ ρίζα της εξίσωσης $ f(x)=0 $
\item $ y'y : A(0,f(0))$ αρκεί να ισχύει $ 0\in D_f $
\end{itemize}
\item Σχετική θέση της $ C_f $ με τον άξονα $ x'x $
\begin{itemize}[leftmargin=4mm]
\item Η $ C_f $ βρίσκεται πάνω από τον $ x'x \Leftrightarrow f(x)>0$
\item Η $ C_f $ βρίσκεται κάτω από τον $ x'x \Leftrightarrow f(x)<0$
\end{itemize}
\item Σημεία τομής των $ C_f,C_g $ :$ A(\rho,0) $ όπου $ \rho $ είναι ρίζα της εξίσωσης $ f(x)=g(x) $
\item Σχετική θέση της $ C_f $ με τη $ C_g $
\begin{itemize}[leftmargin=4mm]
\item Η $ C_f $ βρίσκεται πάνω από τη $ C_g \Leftrightarrow f(x)>g(x)$
\item Η $ C_f $ βρίσκεται κάτω από τη $ C_g \Leftrightarrow f(x)<g(x)$
\end{itemize}
\end{enumerate}
