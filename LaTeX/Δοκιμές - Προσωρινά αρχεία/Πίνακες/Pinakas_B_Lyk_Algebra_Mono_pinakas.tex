\begin{enumerate}[itemsep=0mm,leftmargin=3mm]
\item Λογαριθμική συνάρτηση:\\$ f(x)=\log_{a}{x},\ 0<a\neq 1 $
\item Με $ a=10,a=e $:\\$ f(x)=\log{x}\ ,\ f(x)=\ln{x} $
\item Πεδίο ορισμού $ D_f=(0,+\infty) $.
\item Σύνολο τιμών $ f(D_f)=\mathbb{R} $
\item Η $ f $ δεν έχει μέγιστη και ελάχιστη τιμή.
\item Η $ f $ έχει ρίζα τον αριθμό $ x=1 $. Η $ C_f $ τέμνει τον $ x'x $ στο $ A(1,0) $.
\item Για κάθε $ x_1,x_2\in\mathbb{R} $ ισχύει \[ x_1=x_2\Leftrightarrow \log_{a}{x_1}=\log_{a}{x_2} \]
\item Για τη συνάρτηση $ f(x)=\ln{(g(x))} $ πρέπει να ισχύει $ g(x)>0 $.
\begin{enumerate}[itemsep=0mm,label=\bf\Alph*.,leftmargin=0cm]
\item \textbf{Για {\boldmath$ a>1 $}}
\begin{itemize}[leftmargin=4mm]
\item $ f $ γνησίως αύξουσα στο $ (0,+\infty) $.
\item Η $ C_f $ έχει κατακόρυφη ασύμπτωτη τον ημιάξονα $ Oy' $ 
\item για κάθε $ x_1,x_2\in\mathbb{R} $ ισχύει \begin{gather*}
x_1<x_2\Leftrightarrow \log_{a}{x_1}<\log_{a}{x_2}
\end{gather*}
\item Για $ x>1 $ ισχύει $ \log_{a}x>0 $ ενώ για $ 0<x<1 $ έχουμε $ \log_{a}x<0 $.
\end{itemize}
\item \textbf{Για {\boldmath$ 0<a<1 $}}
\begin{itemize}[leftmargin=4mm]
\item $ f $ γνησίως αύξουσα στο $ (0,+\infty) $.
\item Η $ C_f $ έχει κατακόρυφη ασύμπτωτη τον ημιάξονα $ Oy' $ 
\item για κάθε $ x_1,x_2\in\mathbb{R} $ ισχύει \begin{gather*}
x_1<x_2\Leftrightarrow \log_{a}{x_1}>\log_{a}{x_2}
\end{gather*}
\item Για $ x>1 $ ισχύει $ \log_{a}x<0 $ ενώ για $ 0<x<1 $ έχουμε $ \log_{a}x>0 $.
\end{itemize}
\end{enumerate}
\end{enumerate}
\end{multicols}
