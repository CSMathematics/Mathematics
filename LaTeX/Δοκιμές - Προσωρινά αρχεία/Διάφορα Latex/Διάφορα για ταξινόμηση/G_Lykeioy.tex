\item\mbox{}\\\vspace{-7mm}
\begin{erwthma}
\item Να δώσετε τον ορισμό της απόλυτης τιμής ενός αριθμού $ a\in\mathbb{R} $.
\item Τι ονομάζεται $ \nu- $οστή ρίζα ενός μη αρνητικού αριθμού $ a $?
\item Να αποδείξετε ότι 
\[ |a\cdot \beta|=|a|\cdot|\beta| \]
\item Να χαρακτηρίσετε τις παρακάτω προτάσεις ως σωστές (Σωστή) ή λανθασμένες (Λάθος).
\begin{alist}
\item Αν $ a\in\mathbb{R} $ τότε $ \sqrt{a^2}=a $.
\item Ισχύει ότι $ \big||x|-x\big|=|x|-x $ για κάθε $ x\in\mathbb{R} $.
\item Αν $ \nu\in\mathbb{N}^* $ τότε ισχύει πάντα ότι $ a^{\nu}<\beta^{\nu}\Leftrightarrow a<\beta $.
\item Αν $ \mu,\nu\in\mathbb{N}^* $ τότε ισχύει ότι $ a^{\frac{\mu}{\nu}}=\sqrt[\nu]{a^{\mu}} $ για κάθε $ a\geq 0 $.
\item Αν $ \mu\in\mathbb{Z},\nu\in\mathbb{N}^* $ τότε ισχύει ότι $ a^{\frac{\mu}{\nu}}=\sqrt[\nu]{a^{\mu}} $ για κάθε $ a\geq 0 $.
\item Για οποιουσδήποτε πραγματικούς αριθμούς $ a,\beta\in\mathbb{R} $ ισχύει $ a^2+\beta^2=0\Leftrightarrow a=0\ \text{ και }\ \beta=0 $
\item Αν $ \rho>0 $ τότε $ |x-x_0|>\rho\Leftrightarrow x>x_0+\rho $ και $ x<x_0-\rho $
\end{alist}
\end{erwthma}
\item Αν $ 2\leq x\leq 3 $ και $ 1\leq y\leq 2 $, να βρείτε μεταξύ ποιων ορίων βρίσκεται η τιμή καθεμιάς από τις παρακάτω παραστάσεις.
\begin{erwthma}
\begin{multicols}{3}
\item $ 2x+3y $
\item $ 4xy $
\item $ \frac{x}{y} $
\item $ x^2+y^2 $
\item $ x-y $
\item $ \sqrt{x-1}+\sqrt{y} $
\end{multicols}
\end{erwthma}
\item Δίνεται η παράσταση 
\[ A=|x+1|+|2-x|+x-3 \]
\begin{erwthma}
\item Να γραφτεί η παράσταση $ A $ χωρίς απόλυτες τιμές.
\item Να λύσετε την εξίσωση $ A=0 $
\item Δίνεται το διάστημα $ [-\lambda+2,4] $ με κέντρο $ x_0=1 $.
\begin{alist}
\item Να βρεθεί η τιμή της παραμέτρου $ \lambda $.
\item Για $ \lambda=4 $ να βρεθεί η ακτίνα και το μήκος του παραπάνω διαστήματος.
\end{alist}
\end{erwthma}
\item Δίνεται οι παραστάσεις 
\[ A=\frac{\sqrt{7}\cdot\sqrt[12]{7^5}}{\sqrt[8]{7^5}}\cdot\sqrt[24]{7^{17}}\ \text{ και }\ B=\frac{\sqrt{x^2-4x+4}}{x-2} \]
\begin{erwthma}
\item Να δείξετε ότι $ A=7 $ και $ B=-1 $ όταν $ x<2 $.
\item Να βρεθούν οι τιμές του $ x $ για τις οποίες ισχύει $ d(x,B)<A $.
\item Να μετατραπεί το κλάσμα $ \dfrac{1}{Β-\sqrt{Α}} $ σε ισοδύναμο με ρητό παρονομαστή.
\end{erwthma}
