{\Large \textbf{Διαγώνισμα}}\\
\textbf{Μαθηματικά Προσανατολισμού Β΄ Λυκείου}\\
\textbf{Έννοια διανύσματος - Πρόσθεση - Πολλαπλασιασμός αριθμού με διάνυσμα}
\end{center}
\begin{thema}
\item\mbox{}\\\vspace{-7mm}
\begin{erwthma}
\item Να αναφέρετε με ποιους τρόπους γίνεται η πρόσθεση δύο διανυσμάτων και να σχεδιάσετε και το αντίστοιχο σχήμα για κάθε τρόπο.\\\monades{0.5}
\item Να γράψετε τρεις από τις ιδιότητες του πολλαπλασιασμού ανάμεσα σε πραγματικό αριθμό και διάνυσμα.\monades{1}
\item Ποιά είναι τα στοιχεία ενός διανύσματος;\monades{0.5}
\item Ποια διανύσματα λέγονται ομόρροπα και ποια αντίθετα;\monades{1}
\item \swstolathos
\begin{alist}
\item Αν ισχύει $ \overrightarrow{AB}=\overrightarrow{\varGamma\varDelta} $ τότε τα διανύσματα $ \overrightarrow{AB} $ και $ \overrightarrow{\varGamma\varDelta} $ είναι παράλληλα.
\item Αν ισχύει $ \overrightarrow{A\varDelta}=\overrightarrow{B\varGamma} $ τότε το τετράπλευρο $ AB\varGamma\varDelta $ είναι παραλληλόγραμμο.
\item Αν ισχύει $ \vec{a}=-3\vec{\beta} $ τότε τα διανύσματα $ \vec{a} $ και $ \vec{\beta} $ είναι ομόρροπα.
\item Ο γεωμετρικός τόπος των σημείων $ M $ για τα οποία ισχύει η ισότητα $ |\overrightarrow{MA}|=2 $ είναι κύκλος με κέντρο το σημείο $ A $ με ακτίνα $ 2 $.
\item Από την ισότητα $ |\overrightarrow{AB}|=|\overrightarrow{A\varGamma}| $ προκύπτει ότι τα σημεία $ A,B $ και $ \varGamma $ είναι συνευθειακά.
\end{alist}\monades{2}
\end{erwthma}
\item\mbox{}\\\vspace{-7mm}
\begin{erwthma}
\item Αν ισχύει $ \overrightarrow{A\varDelta}-\overrightarrow{Z\varGamma}=\overrightarrow{BE}+\overrightarrow{B \varDelta}-\overrightarrow{ZE} $ να αποδείξετε ότι το $ \varGamma $ είναι μέσο του ευθυγράμμου τμήματος $ AB $.\monades{1.5}
\item Να αποδείξετε ότι για οποιαδήποτε σημεία $ A,B,\varGamma,\varDelta $ ισχύει ότι : $ 4 \overrightarrow{AB}+3 \overrightarrow{A\varGamma}=4 \overrightarrow{\varDelta B}+3 \overrightarrow{\varDelta \varGamma}-7 \overrightarrow{\varDelta A} $.\monades{2}
\item Αν ισχύει ότι: $ 3 \overrightarrow{B\varDelta}-\overrightarrow{\varGamma A}=3 \overrightarrow{B \varGamma}-\overrightarrow{\varDelta B} $ να αποδείξετε ότι $ \overrightarrow{AB}\updownarrows\overrightarrow{\varGamma\varDelta} $.\monades{1.5}
\end{erwthma}
\item\mbox{}\\\vspace{-7mm}
\begin{erwthma}
\item Αν ισχύει $ 2 \overrightarrow{A\varLambda}+3 \overrightarrow{B\varLambda}+2 \overrightarrow{MB}=\overrightarrow{AK}+\overrightarrow{AM}+\overrightarrow{BK} $ να αποδείξετε ότι τα διανύσματα $ \overrightarrow{K\varLambda} $ και $ \overrightarrow{M\varLambda} $ είναι αντίρροπα.\monades{2}
\item Δίνεται τρίγωνο $ AB\varGamma $ και τα σημεία $ \varDelta,E $ του επιπέδου του τέτοια ώστε : $ \overrightarrow{A\varDelta}=5 \overrightarrow{A B}+8 \overrightarrow{A \varGamma} $ και $ \overrightarrow{A E}=3 \overrightarrow{AB}+10 \overrightarrow{A \varGamma} $. Να αποδειχθεί ότι : $ \overrightarrow{\varDelta E}\parallel \overleftrightarrow{B\varGamma} $.\monades{2}
\item Αν ισχύουν $ |\vec{a}|=3,|\vec{\beta}|=2 $ και $ |\vec{a}+\vec{\beta}|\geq 5 $ ,να αποδείξετε ότι τα διανύσματα $ \vec{a},\vec{\beta} $ είναι ομόρροπα.\monades{1}
\end{erwthma}
\item\mbox{}\\\vspace{-7mm}
\begin{erwthma}
\item Δίνονται σημεία $ A,B,\varGamma,\varDelta,E $ για τα οποία ισχύει ότι : $ 5 \overrightarrow{A \varDelta}-3 \overrightarrow{AE}-4 \overrightarrow{\varDelta \varGamma}=4 \overrightarrow{EB}-\overrightarrow{E \varGamma}+9 \overrightarrow{BA} $. Να αποδείξετε ότι τα σημεία $ A,B,\varGamma $ είναι συνευθειακά.\monades{1.5}
\item Να αποδείξετε ότι αν: $ (a+4) \overrightarrow{KA}+5 \overrightarrow{KB}=(a+9) \overrightarrow{K \varGamma} $, όπου $ a\in\mathbb{R} $, τότε τα σημεία $ A,B,\varGamma $ είναι συνευθειακά.\monades{1.5}
\item Έστω παραλληλόγραμμο $ AB\varGamma\varDelta $ και $ K $ το κέντρο του. Αν $ M $ είναι το μέσο του $ K\varGamma $, να αποδείξετε ότι $ \overrightarrow{\mathrm{AB}}+\overrightarrow{A\varDelta}=4 \overrightarrow{AM}-2 \overrightarrow{\mathrm{A\varGamma}} $.\monades{2}
\end{erwthma}
\end{thema}
