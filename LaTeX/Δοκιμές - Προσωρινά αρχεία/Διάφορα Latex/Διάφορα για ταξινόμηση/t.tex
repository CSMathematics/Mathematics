\begin{pycode}
from sympy import *
var("x, y")
binomials = []
for m in range(2, 6):
	binomials.append((x + y)**m)
print(r"\begin{align*}")
for expr in binomials:
	print(r"%s &= %s\\" % (latex(expr), latex(expand(expr))))
print(r"\end{align*}")
\end{pycode}
