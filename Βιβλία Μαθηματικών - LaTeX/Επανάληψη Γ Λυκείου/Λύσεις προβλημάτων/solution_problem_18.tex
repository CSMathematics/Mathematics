\subsection*{Λύση Προβλήματος 18 --- Κυρτότητα της αεροτομής \en{NACA 0012}}

\begin{alist}
  \item Για κάθε $x\in(0,1)$ έχουμε
        \begin{align*}
          f'(x)
          &=0{,}6\left(
            \frac{0{,}14845}{\sqrt{x}}
            -0{,}1260-0{,}7032x
            +0{,}8529x^2-0{,}406x^3
          \right)
        \end{align*}
        και επομένως
        \begin{align*}
          f''(x)
          &=0{,}6\left(
            -\frac{0{,}074225}{x^{3/2}}
            -0{,}7032+1{,}7058x-1{,}218x^2
          \right)\\
          &=-0{,}6\left(
            \frac{0{,}074225}{x^{3/2}}
            +1{,}218x^2-1{,}7058x+0{,}7032
          \right).
        \end{align*}

        Εξετάζουμε το τριώνυμο
        \[
          P(x)=1{,}218x^2-1{,}7058x+0{,}7032.
        \]
        Η διακρίνουσά του είναι
        \begin{align*}
          \Delta
          &=(-1{,}7058)^2
            -4\cdot1{,}218\cdot0{,}7032\\
          &=2{,}90975364-3{,}42599040\\
          &=-0{,}51623676<0.
        \end{align*}
        Επειδή ο συντελεστής του $x^2$ είναι θετικός, ισχύει
        \[
          P(x)>0
          \qquad\text{για κάθε }x\in\mathbb{R}.
        \]
        Επιπλέον, για $x\in(0,1)$,
        \[
          \frac{0{,}074225}{x^{3/2}}>0.
        \]
        Άρα η ποσότητα μέσα στην παρένθεση είναι θετική και τελικά
        \[
          {f''(x)<0\quad\text{για κάθε }x\in(0,1)}.
        \]

  \item Από το προηγούμενο ερώτημα προκύπτει ότι η $f$ είναι κοίλη στο
        $(0,1)$.

        Για την κάτω επιφάνεια ισχύει $g(x)=-f(x)$, οπότε
        \[
          g''(x)=-f''(x)>0
          \qquad\text{για κάθε }x\in(0,1).
        \]
        Επομένως, η $g$ είναι κυρτή στο $(0,1)$. Συνολικά,
        \[
          {\text{η άνω επιφάνεια είναι κοίλη και η κάτω επιφάνεια
          είναι κυρτή στο }(0,1)}.
        \]

  \item Η δεύτερη παράγωγος της $f$ είναι πάντοτε αρνητική στο $(0,1)$,
        ενώ η δεύτερη παράγωγος της $g$ είναι πάντοτε θετική στο ίδιο
        διάστημα. Άρα καμία από τις δύο συναρτήσεις δεν αλλάζει
        κυρτότητα στο εσωτερικό της χορδής και συνεπώς δεν έχει σημείο
        καμπής εκεί.

        Το άκρο $x=0$ δεν είναι εσωτερικό σημείο του πεδίου ορισμού και,
        επιπλέον, εξαιτίας του όρου $\sqrt{x}$ η $f'$ δεν ορίζεται στο
        $0$. Επομένως, ούτε το χείλος προσβολής μπορεί να χαρακτηριστεί
        σημείο καμπής με το κριτήριο της δεύτερης παραγώγου.

        Γεωμετρικά, καθώς κινούμαστε από το χείλος προσβολής προς το
        χείλος εκφυγής, η άνω επιφάνεια στρέφει συνεχώς τα κοίλα προς τα
        κάτω και η κάτω επιφάνεια, συμμετρικά, στρέφει συνεχώς τα κοίλα
        προς τα πάνω. Το περίγραμμα δεν παρουσιάζει αντιστροφή της
        καμπυλότητάς του στο εσωτερικό της χορδής.
\end{alist}
