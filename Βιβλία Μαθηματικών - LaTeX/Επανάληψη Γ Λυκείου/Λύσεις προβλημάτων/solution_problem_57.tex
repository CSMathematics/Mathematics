% Λύση του 57ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 57 --- Απόσταση δορυφόρου από σταθμό εδάφους}

\begin{alist}
  \item Είναι
        \[
          d(\theta)
          =
          \sqrt{R_E^2+r^2-2R_Er\syn\theta}.
        \]
        Θέτουμε
        \[
          u(\theta)=R_E^2+r^2-2R_Er\syn\theta.
        \]
        Τότε
        \[
          d(\theta)=\sqrt{u(\theta)}.
        \]
        Επειδή \((\syn\theta)'=-\hm\theta\), έχουμε
        \[
          u'(\theta)=2R_Er\hm\theta.
        \]
        Άρα
        \[
          d'(\theta)
          =
          \frac{u'(\theta)}{2\sqrt{u(\theta)}}
          =
          \frac{2R_Er\hm\theta}{2d(\theta)}.
        \]
        Επομένως
        \[
          {
          d'(\theta)=\frac{R_Er\hm\theta}{d(\theta)}
          }.
        \]

  \item Από τον κανόνα αλυσίδας,
        \[
          \frac{dd}{dt}
          =
          \frac{dd}{d\theta}\cdot\frac{d\theta}{dt}.
        \]
        Άρα
        \[
          d'(t)
          =
          \frac{R_Er\hm\theta}{d(\theta)}\theta'(t).
        \]

        Έχουμε
        \[
          R_E=6378,\qquad r=6378+400=6778.
        \]
        Για \(\theta=\dfrac{\pi}{4}\),
        \[
          d\left(\frac{\pi}{4}\right)
          =
          \sqrt{
          6378^2+6778^2
          -2\cdot6378\cdot6778\cdot\syn\frac{\pi}{4}
          }.
        \]
        Επειδή
        \[
          \syn\frac{\pi}{4}=\hm\frac{\pi}{4}=\frac{\sqrt2}{2},
        \]
        βρίσκουμε προσεγγιστικά
        \[
          d\left(\frac{\pi}{4}\right)\approx5048{,}13\si{km}.
        \]
        Επομένως
        \[
          d'(t)
          \approx
          \frac{6378\cdot6778\cdot\frac{\sqrt2}{2}}
               {5048{,}13}
          \cdot
          \frac{2\pi}{90}.
        \]
        Άρα
        \[
          {
          d'(t)\approx422{,}74\si{km/min}
          }.
        \]
        Το θετικό πρόσημο σημαίνει ότι εκείνη τη στιγμή η απόσταση
        δορυφόρου--σταθμού αυξάνεται.

  \item Η απόσταση είναι
        \[
          d(\theta)=\sqrt{R_E^2+r^2-2R_Er\syn\theta}.
        \]
        Επειδή η τετραγωνική ρίζα είναι γνησίως αύξουσα, η \(d(\theta)\)
        γίνεται ελάχιστη όταν γίνεται ελάχιστη η ποσότητα
        \[
          R_E^2+r^2-2R_Er\syn\theta.
        \]
        Αυτό συμβαίνει όταν το \(\syn\theta\) είναι μέγιστο. Στο διάστημα
        \(0\leq\theta\leq\pi\), το μέγιστο του \(\syn\theta\) είναι \(1\)
        και επιτυγχάνεται για
        \[
          \theta=0.
        \]
        Άρα η απόσταση είναι ελάχιστη όταν ο δορυφόρος βρίσκεται ακριβώς
        πάνω από τον σταθμό εδάφους. Τότε
        \[
          d(0)=r-R_E=400\si{km}.
        \]
\end{alist}
