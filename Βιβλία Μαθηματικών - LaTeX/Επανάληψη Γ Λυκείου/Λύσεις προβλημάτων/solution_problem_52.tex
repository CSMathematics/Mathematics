\subsection*{Λύση Προβλήματος 52 --- Θερμική ακτινοβολία}

\begin{alist}
  \item Έχουμε
        \[
          P(T)=\sigma T^4.
        \]
        Από τον ορισμό της παραγώγου,
        \[
          P'(T_0)
          =
          \lim_{h\to0}\frac{P(T_0+h)-P(T_0)}{h}.
        \]
        Άρα
        \[
          P'(T_0)
          =
          \lim_{h\to0}
          \frac{\sigma(T_0+h)^4-\sigma T_0^4}{h}.
        \]
        Βγάζουμε κοινό παράγοντα το \(\sigma\):
        \[
          P'(T_0)
          =
          \sigma
          \lim_{h\to0}
          \frac{(T_0+h)^4-T_0^4}{h}.
        \]
        Αναπτύσσουμε:
        \[
          (T_0+h)^4
          =
          T_0^4+4T_0^3h+6T_0^2h^2+4T_0h^3+h^4.
        \]
        Επομένως
        \[
          (T_0+h)^4-T_0^4
          =
          4T_0^3h+6T_0^2h^2+4T_0h^3+h^4.
        \]
        Άρα
        \[
          \frac{(T_0+h)^4-T_0^4}{h}
          =
          4T_0^3+6T_0^2h+4T_0h^2+h^3.
        \]
        Παίρνοντας το όριο για \(h\to0\), παίρνουμε
        \[
          P'(T_0)
          =
          \sigma\cdot4T_0^3.
        \]
        Δηλαδή
        \[
          {P'(T_0)=4\sigma T_0^3}.
        \]

  \item Για \(T=300\si{K}\), με
        \[
          \sigma\approx5{,}67\cdot10^{-8},
        \]
        έχουμε
        \[
          P(300)
          =
          5{,}67\cdot10^{-8}\cdot300^4.
        \]
        Επειδή
        \[
          300^4=8{,}1\cdot10^9,
        \]
        προκύπτει
        \[
          P(300)
          \approx
          459{,}27.
        \]
        Άρα
        \[
          {P(300)\approx459{,}27\si{W/m^2}}.
        \]

        Επίσης
        \[
          P'(300)=4\sigma\cdot300^3.
        \]
        Επειδή
        \[
          300^3=27\,000\,000,
        \]
        έχουμε
        \[
          P'(300)
          \approx
          4\cdot5{,}67\cdot10^{-8}\cdot27\,000\,000
          =
          6{,}1236.
        \]
        Άρα
        \[
          {P'(300)\approx6{,}12\ \si{W/(m^2K)}}.
        \]

  \item Το αποτέλεσμα
        \[
          P'(300)\approx6{,}12\ \si{W/(m^2K)}
        \]
        σημαίνει ότι, κοντά στους \(300\si{K}\), αύξηση της θερμοκρασίας
        κατά \(1\si{K}\) αυξάνει την ακτινοβολούμενη ισχύ ανά τετραγωνικό
        μέτρο περίπου κατά
        \[
          6{,}12\si{W/m^2}.
        \]

        Η αύξηση γίνεται έντονη επειδή η ισχύς δεν είναι ανάλογη της
        θερμοκρασίας, αλλά της τέταρτης δύναμής της:
        \[
          P(T)=\sigma T^4.
        \]
        Γι' αυτό και ο ρυθμός μεταβολής
        \[
          P'(T)=4\sigma T^3
        \]
        μεγαλώνει καθώς αυξάνεται η θερμοκρασία.
\end{alist}
