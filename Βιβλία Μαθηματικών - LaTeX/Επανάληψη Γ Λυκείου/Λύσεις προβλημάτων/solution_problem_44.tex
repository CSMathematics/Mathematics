% Λύση του 44ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 44 --- Ατμοσφαιρική πίεση}

\begin{alist}
  \item Έχουμε
        \[
          P(x)=101{,}29
          \left(\frac{288{,}14-6{,}49x}{288{,}08}\right)^{5{,}256}.
        \]
        Θέτουμε
        \[
          u(x)=\frac{288{,}14-6{,}49x}{288{,}08}.
        \]
        Τότε
        \[
          u'(x)=-\frac{6{,}49}{288{,}08}.
        \]
        Άρα, με τον κανόνα της αλυσίδας,
        \[
          P'(x)
          =
          101{,}29\cdot 5{,}256
          \left(\frac{288{,}14-6{,}49x}{288{,}08}\right)^{4{,}256}
          \left(-\frac{6{,}49}{288{,}08}\right).
        \]
        Επομένως
        \[
          {
          P'(x)
          =
          -\frac{101{,}29\cdot 5{,}256\cdot 6{,}49}{288{,}08}
          \left(\frac{288{,}14-6{,}49x}{288{,}08}\right)^{4{,}256}
          }.
        \]

  \item Για $x=2$ έχουμε
        \[
          P(2)
          =
          101{,}29
          \left(\frac{288{,}14-6{,}49\cdot2}{288{,}08}\right)^{5{,}256}.
        \]
        Δηλαδή
        \[
          P(2)
          =
          101{,}29
          \left(\frac{275{,}16}{288{,}08}\right)^{5{,}256}
          \approx
          79{,}58.
        \]
        Άρα
        \[
          {P(2)\approx 79{,}58\si{kPa}}.
        \]

        Επίσης
        \[
          P'(2)\approx -9{,}87.
        \]
        Άρα
        \[
          {P'(2)\approx -9{,}87\ \si{kPa}\text{ ανά }\si{km}}.
        \]
        Το αρνητικό πρόσημο δείχνει ότι, κοντά στα $2\si{km}$, όσο
        αυξάνεται το ύψος τόσο μειώνεται η ατμοσφαιρική πίεση. Πιο
        συγκεκριμένα, γύρω από τα $2\si{km}$, η πίεση μειώνεται περίπου
        κατά $9{,}87\si{kPa}$ για κάθε επιπλέον \si{km}.

  \item Η εφαπτομένη της γραφικής παράστασης της $P$ στο σημείο με
        τετμημένη $x=2$ έχει εξίσωση
        \[
          y-P(2)=P'(2)(x-2).
        \]
        Με τις προηγούμενες προσεγγίσεις παίρνουμε
        \[
          y-79{,}58=-9{,}87(x-2).
        \]
        Επομένως
        \[
          {y=79{,}58-9{,}87(x-2)}.
        \]

        Για $x=2{,}1$ η γραμμική προσέγγιση δίνει
        \[
          y
          =
          79{,}58-9{,}87(2{,}1-2)
          =
          79{,}58-0{,}987
          =
          78{,}593.
        \]
        Άρα η πίεση στο ύψος $2{,}1\si{km}$ εκτιμάται περίπου
        \[
          {78{,}59\si{kPa}}.
        \]
\end{alist}
