\subsection*{Λύση Προβλήματος 49 --- Αποσβεννυμένη ταλάντωση}

\begin{alist}
  \item Για κάθε πραγματικό αριθμό $\theta$ ισχύει
        \[
          -1\leq \syn\theta\leq1.
        \]
        Εφαρμόζουμε αυτή την ανισότητα για
        \[
          \theta=\omega t+\varphi.
        \]
        Άρα
        \[
          -1\leq \syn(\omega t+\varphi)\leq1.
        \]
        Επειδή
        \[
          A>0
          \quad\text{και}\quad
          e^{-\lambda t}>0
        \]
        για κάθε $t\geq0$, μπορούμε να πολλαπλασιάσουμε όλα τα μέλη της
        ανισότητας με το θετικό γινόμενο \(Ae^{-\lambda t}\). Παίρνουμε
        \[
          -Ae^{-\lambda t}
          \leq
          Ae^{-\lambda t}\syn(\omega t+\varphi)
          \leq
          Ae^{-\lambda t}.
        \]
        Επειδή
        \[
          x(t)=Ae^{-\lambda t}\syn(\omega t+\varphi),
        \]
        καταλήγουμε στο ζητούμενο:
        \[
          {
          -Ae^{-\lambda t}\leq x(t)\leq Ae^{-\lambda t}
          }.
        \]

  \item Επειδή \(\lambda>0\), έχουμε
        \[
          \lim_{t\to+\infty}e^{-\lambda t}=0.
        \]
        Άρα
        \[
          \lim_{t\to+\infty}Ae^{-\lambda t}=0
        \]
        και
        \[
          \lim_{t\to+\infty}\bigl(-Ae^{-\lambda t}\bigr)=0.
        \]
        Από το προηγούμενο ερώτημα ισχύει
        \[
          -Ae^{-\lambda t}\leq x(t)\leq Ae^{-\lambda t}.
        \]
        Τα δύο άκρα της ανισότητας έχουν το ίδιο όριο, δηλαδή το $0$.
        Επομένως, από το κριτήριο παρεμβολής,
        \[
          {
          \lim_{t\to+\infty}x(t)=0
          }.
        \]

  \item Το αποτέλεσμα σημαίνει ότι η απομάκρυνση του ταλαντωτή από τη
        θέση ισορροπίας τείνει στο μηδέν καθώς περνά ο χρόνος.

        Η συνάρτηση \(\syn(\omega t+\varphi)\) συνεχίζει να ταλαντώνεται
        ανάμεσα στο \(-1\) και στο \(1\), αλλά ο παράγοντας
        \[
          Ae^{-\lambda t}
        \]
        γίνεται όλο και μικρότερος και τείνει στο μηδέν. Άρα η
        ταλάντωση σβήνει: το σώμα πλησιάζει τελικά τη θέση ισορροπίας.
\end{alist}
