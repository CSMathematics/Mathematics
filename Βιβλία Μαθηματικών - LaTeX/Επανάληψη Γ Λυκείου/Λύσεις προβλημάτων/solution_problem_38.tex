\subsection*{Λύση Προβλήματος 38 --- Εκκένωση κωνικής δεξαμενής}

\begin{alist}
  \item Η δεξαμενή είναι ανεστραμμένος κώνος με ολικό ύψος
        $10\si{m}$ και ακτίνα βάσης $5\si{m}$. Από την ομοιότητα των
        τριγώνων έχουμε
        \[
          \frac{r(t)}{h(t)}=\frac{5}{10}=\frac12.
        \]
        Άρα
        \[
          {r(t)=\frac{h(t)}{2}}.
        \]

        Ο όγκος του υγρού είναι
        \[
          V(t)=\frac13\pi r^2(t)h(t).
        \]
        Αντικαθιστούμε $r(t)=\frac{h(t)}2$:
        \[
          V(t)
          =
          \frac13\pi
          \left(\frac{h(t)}2\right)^2h(t)
          =
          \frac{\pi}{12}h^3(t).
        \]
        Επομένως
        \[
          {V(t)=\frac{\pi}{12}h^3(t)}.
        \]

  \item Επειδή η δεξαμενή αδειάζει, ο όγκος μειώνεται. Άρα
        \[
          V'(t)=-2\si{m^3/min}.
        \]
        Από τη σχέση
        \[
          V(t)=\frac{\pi}{12}h^3(t)
        \]
        παραγωγίζουμε ως προς τον χρόνο:
        \[
          V'(t)
          =
          \frac{\pi}{12}\cdot3h^2(t)h'(t)
          =
          \frac{\pi}{4}h^2(t)h'(t).
        \]
        Τη στιγμή που $h(t)=6\si{m}$, έχουμε
        \[
          -2
          =
          \frac{\pi}{4}\cdot6^2\cdot h'(t).
        \]
        Δηλαδή
        \[
          -2=9\pi h'(t).
        \]
        Επομένως
        \[
          {h'(t)=-\frac{2}{9\pi}\si{m/min}}.
        \]
        Το αρνητικό πρόσημο δείχνει ότι η στάθμη του υγρού κατεβαίνει.
        Άρα ο ρυθμός καθόδου της στάθμης έχει μέτρο
        \[
          {\frac{2}{9\pi}\si{m/min}}.
        \]
\end{alist}
