% Λύση του 40ού προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 40 --- Νόμος του \en{Boyle}}

\begin{alist}
  \item Σύμφωνα με τον νόμο του \en{Boyle}, έχουμε
        \[
          P(t)V(t)=C.
        \]
        Τη στιγμή $t_0$ δίνεται ότι
        \[
          P(t_0)=150\si{kPa},
          \qquad
          V(t_0)=400\si{cm^3}.
        \]
        Άρα
        \[
          C=P(t_0)V(t_0)=150\cdot400=60000.
        \]
        Επομένως
        \[
          {C=60000\ \si{kPa\cdot cm^3}}.
        \]

        Παραγωγίζουμε τη σχέση
        \[
          P(t)V(t)=C
        \]
        ως προς τον χρόνο:
        \[
          P'(t)V(t)+P(t)V'(t)=0.
        \]
        Άρα
        \[
          P'(t)=-\frac{P(t)V'(t)}{V(t)}.
        \]
        Τη στιγμή $t_0$ ο όγκος μειώνεται με ρυθμό
        \[
          V'(t_0)=-20\si{cm^3/min}.
        \]
        Επομένως
        \[
          P'(t_0)
          =
          -\frac{150\cdot(-20)}{400}
          =
          \frac{3000}{400}
          =
          7{,}5.
        \]
        Άρα
        \[
          {P'(t_0)=7{,}5\si{kPa/min}}.
        \]
        Το θετικό πρόσημο δείχνει ότι η πίεση αυξάνεται.

  \item Αν ο ρυθμός μείωσης του όγκου παραμένει σταθερός, τότε
        \[
          V'(t)=-20
        \]
        και άρα
        \[
          V''(t)=0.
        \]
        Από τη σχέση
        \[
          P(t)=\frac{C}{V(t)}
        \]
        έχουμε
        \[
          P'(t)
          =
          -C\frac{V'(t)}{V^2(t)}.
        \]
        Παραγωγίζουμε ξανά:
        \[
          P''(t)
          =
          -C\left[
            \frac{V''(t)}{V^2(t)}
            -2\frac{(V'(t))^2}{V^3(t)}
          \right].
        \]
        Επειδή $V''(t)=0$, παίρνουμε
        \[
          P''(t)
          =
          \frac{2C(V'(t))^2}{V^3(t)}.
        \]
        Εδώ $C>0$, $V(t)>0$ και $(V'(t))^2>0$. Άρα
        \[
          {P''(t)>0}.
        \]

        Επομένως η $P'(t)$ είναι αύξουσα. Δηλαδή, καθώς το έμβολο
        συμπιέζει το αέριο και ο όγκος μειώνεται, η πίεση αυξάνεται με
        ολοένα και πιο γρήγορο ρυθμό.
\end{alist}
