% Λύση του 62ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 62 --- Μέγιστη ισχύς σε ηλεκτρικό φορτίο}

\begin{alist}
  \item Επειδή
        \[
          \mathcal E=12
          \quad\text{και}\quad
          r=3,
        \]
        έχουμε
        \[
          P(R)=\frac{\mathcal E^2R}{(R+r)^2}
          =
          \frac{144R}{(R+3)^2},
          \qquad R>0.
        \]
        Παραγωγίζουμε:
        \[
          P'(R)
          =
          144\left(\frac{R}{(R+3)^2}\right)'.
        \]
        Άρα
        \[
          P'(R)
          =
          144\cdot
          \frac{(R+3)^2-2R(R+3)}{(R+3)^4}.
        \]
        Επομένως
        \[
          P'(R)
          =
          144\cdot
          \frac{(R+3)-2R}{(R+3)^3}
          =
          {
          \frac{144(3-R)}{(R+3)^3}
          }.
        \]

  \item Για κάθε \(R>0\) ισχύει
        \[
          (R+3)^3>0.
        \]
        Άρα το πρόσημο της \(P'(R)\) είναι το πρόσημο του \(3-R\).

        Επομένως:
        \[
          P'(R)>0
          \quad\text{για}\quad
          0<R<3,
        \]
        και
        \[
          P'(R)<0
          \quad\text{για}\quad
          R>3.
        \]
        Συνεπώς η \(P\) είναι γνησίως αύξουσα στο \((0,3)\) και
        γνησίως φθίνουσα στο \((3,+\infty)\).

  \item Από τη μονοτονία προκύπτει ότι η \(P\) γίνεται μέγιστη για
        \[
          {R=3\si{\ohm}}.
        \]
        Η μέγιστη ισχύς είναι
        \[
          P(3)=\frac{144\cdot3}{(3+3)^2}
          =
          \frac{432}{36}
          =
          12.
        \]
        Άρα
        \[
          {P_{\max}=12\si{W}}.
        \]

  \item Η εσωτερική αντίσταση της πηγής είναι
        \[
          r=3\si{\ohm}.
        \]
        Βρήκαμε ότι η μέγιστη ισχύς στο φορτίο επιτυγχάνεται για
        \[
          R=3\si{\ohm}.
        \]
        Δηλαδή
        \[
          {R=r}.
        \]
        Αυτό συμφωνεί ακριβώς με το θεώρημα μέγιστης μεταφοράς ισχύος.
\end{alist}

\paragraph{Πηγές και έλεγχος ρεαλιστικότητας}
Το θεώρημα μέγιστης μεταφοράς ισχύος είναι πραγματικό θεώρημα της
ηλεκτρολογίας και χρησιμοποιείται σε κυκλώματα με ισοδύναμο
\en{Thévenin}: πηγή τάσης σε σειρά με εσωτερική αντίσταση
\cite{maximumPowerTransferTheorem}. Ο τύπος
\[
  P(R)=\frac{\mathcal E^2R}{(R+r)^2}
\]
προκύπτει από τον νόμο του \en{Ohm}. Πράγματι, το ρεύμα είναι
\[
  I=\frac{\mathcal E}{R+r},
\]
και η ισχύς στο φορτίο είναι
\[
  P=I^2R.
\]

Για \(R=r=3\si{\ohm}\), το ρεύμα είναι
\[
  I=\frac{12}{3+3}=2\si{A}.
\]
Άρα η ισχύς στο φορτίο είναι
\[
  P=I^2R=2^2\cdot3=12\si{W},
\]
όπως βρέθηκε παραπάνω. Η ίδια ισχύς χάνεται και στην εσωτερική
αντίσταση της πηγής, οπότε η απόδοση τότε είναι \(50\%\). Αυτό είναι
σημαντικό ρεαλιστικό σχόλιο: η συνθήκη \(R=r\) δίνει μέγιστη ισχύ στο
φορτίο, όχι μέγιστη ενεργειακή απόδοση.
