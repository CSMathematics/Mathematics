\subsection*{Λύση Προβλήματος 15 --- Ο κυβικός νόμος του Murray}

\begin{alist}
  \item Για $r>0$ έχουμε
        \[
          P(r)
          =\frac{8\mu\ell Q^2}{\pi}r^{-4}
          +\kappa\pi\ell r^2.
        \]
        Επομένως,
        \[
          \boxed{
          P'(r)
          =-\frac{32\mu\ell Q^2}{\pi r^5}
          +2\kappa\pi\ell r}.
        \]

  \item Παραγωγίζουμε ξανά:
        \[
          P''(r)
          =\frac{160\mu\ell Q^2}{\pi r^6}
          +2\kappa\pi\ell>0,
          \qquad r>0.
        \]
        Άρα η $P'$ είναι γνησίως αύξουσα στο $(0,+\infty)$.

        Επιπλέον,
        \[
          \lim_{r\to0^+}P'(r)=-\infty
          \qquad\text{και}\qquad
          \lim_{r\to+\infty}P'(r)=+\infty.
        \]
        Επομένως, η εξίσωση $P'(r)=0$ έχει μοναδική θετική λύση
        $r_0$. Η $P$ είναι γνησίως φθίνουσα στο $(0,r_0)$ και γνησίως
        αύξουσα στο $(r_0,+\infty)$. Συνεπώς, στο $r_0$ παρουσιάζει
        μοναδικό ολικό ελάχιστο.

  \item Από τη συνθήκη $P'(r_0)=0$ έχουμε
        \begin{align*}
          -\frac{32\mu\ell Q^2}{\pi r_0^5}
          +2\kappa\pi\ell r_0=0
          &\Longleftrightarrow
          \kappa\pi^2r_0^6=16\mu Q^2\\
          &\Longleftrightarrow
          Q^2=\frac{\kappa\pi^2}{16\mu}r_0^6.
        \end{align*}
        Επειδή $Q,r_0>0$, προκύπτει
        \[
          Q=\frac{\pi}{4}\sqrt{\frac{\kappa}{\mu}}\,r_0^3.
        \]
        Άρα
        \[
          \boxed{Q=Cr_0^3},
          \qquad
          C=\frac{\pi}{4}\sqrt{\frac{\kappa}{\mu}}>0.
        \]
        Η σταθερά $C$ δεν εξαρτάται από το μήκος $\ell$, το οποίο
        απαλείφθηκε κατά την επίλυση της εξίσωσης $P'(r_0)=0$.

  \item Για το μητρικό αγγείο και τα δύο θυγατρικά αγγεία ισχύουν
        αντίστοιχα
        \[
          Q=CR^3,\qquad
          Q_1=Cr_1^3,\qquad
          Q_2=Cr_2^3.
        \]
        Από τη διατήρηση της παροχής,
        \[
          Q=Q_1+Q_2.
        \]
        Επομένως,
        \[
          CR^3=Cr_1^3+Cr_2^3.
        \]
        Επειδή $C>0$, διαιρούμε με $C$ και βρίσκουμε
        \[
          \boxed{R^3=r_1^3+r_2^3}.
        \]
        Αυτή είναι η κλασική κυβική σχέση του νόμου του Murray.
\end{alist}
