\subsection*{Λύση Προβλήματος 19 --- Όγκος νερού στον ποταμό Ποτόμακ}

\begin{alist}
  \item Επειδή η $Q$ έχει διαφορετικό τύπο στα δύο διαστήματα,
        \begin{align*}
          \int_0^2 Q(t)\,dt
          &=\int_0^1(12+12t)\,dt
            +\int_1^2(26t-2)\,dt\\
          &=\left[12t+6t^2\right]_0^1
            +\left[13t^2-2t\right]_1^2\\
          &=18+37=55.
        \end{align*}
        Άρα
        \[
          {\int_0^2 Q(t)\,dt=55}.
        \]
        Το ίδιο αποτέλεσμα προκύπτει ως άθροισμα των εμβαδών δύο
        τραπεζίων:
        \[
          \frac{12+24}{2}\cdot1
          +\frac{24+50}{2}\cdot1=18+37=55.
        \]

  \item Η $Q$ μετριέται σε χιλιάδες \si{ft^3/s}, ενώ το $t$ σε ημέρες.
        Επομένως, μετατρέπουμε τις μονάδες:
        \begin{align*}
          V
          &=55\cdot1\,000\cdot86\,400\\
          &=4\,752\,000\,000\si{ft^3}.
        \end{align*}
        Σε κυβικά μέτρα,
        \begin{align*}
          V
          &=4\,752\,000\,000\cdot0{,}0283168\\
          &\approx134\,561\,434\si{m^3}.
        \end{align*}
        Συνεπώς, η εκτιμώμενη ποσότητα νερού είναι
        \[
          {V\approx1{,}35\cdot10^8\si{m^3}}.
        \]
        Η τιμή είναι εκτίμηση, επειδή μεταξύ των ημερήσιων μετρήσεων
        θεωρήσαμε γραμμική μεταβολή της παροχής.
\end{alist}
