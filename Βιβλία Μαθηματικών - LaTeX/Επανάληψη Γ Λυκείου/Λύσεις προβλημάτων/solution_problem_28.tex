% Λύση του 28ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 28 --- Πραγματική απόσταση ακινητοποίησης}

\begin{alist}
  \item Κατά τον χρόνο αντίδρασης η ταχύτητα παραμένει σταθερή. Άρα
        \[
          21=v_0t_r=\frac{280}{9}t_r
        \]
        και επομένως
        \[
          {t_r=\frac{189}{280}\si{s}=0{,}675\si{s}}.
        \]

  \item Κατά την πέδηση ισχύει
        \[
          v(\tau)=v_0+a\tau.
        \]
        Αν $t_b$ είναι ο χρόνος πέδησης, τότε $v(t_b)=0$, οπότε
        \[
          t_b=-\frac{v_0}{a}.
        \]
        Η απόσταση πέδησης είναι
        \begin{align*}
          75
          &=\int_0^{t_b}(v_0+a\tau)\,d\tau\\
          &=v_0t_b+\frac{a}{2}t_b^2.
        \end{align*}
        Χρησιμοποιώντας τη σχέση $a t_b=-v_0$, παίρνουμε
        \[
          75=\frac{v_0t_b}{2}.
        \]
        Επομένως,
        \begin{align*}
          t_b
          &=\frac{150}{v_0}
           =\frac{150}{280/9}
           =\frac{135}{28}\si{s}
           \approx4{,}82\si{s},\\
          a
          &=-\frac{v_0}{t_b}
           =-\frac{1568}{243}\si{m/s^2}
           \approx-6{,}45\si{m/s^2}.
        \end{align*}
        Άρα
        \[
          {t_b\approx4{,}82\si{s}},
          \qquad
          {a\approx-6{,}45\si{m/s^2}}.
        \]

  \item Ο συνολικός χρόνος μέχρι την ακινητοποίηση είναι
        \[
          t_{\mathrm{stop}}=t_r+t_b
          \approx0{,}675+4{,}821
          \approx5{,}50\si{s}.
        \]
        Η συνάρτηση της ταχύτητας είναι
        \[
          v(t)=
          \begin{cases}
          \dfrac{280}{9},
            &0\leq t\leq\dfrac{189}{280},\\[2mm]
          \dfrac{280}{9}
          -\dfrac{1568}{243}
          \left(t-\dfrac{189}{280}\right),
            &\dfrac{189}{280}<t\leq
             \dfrac{189}{280}+\dfrac{135}{28}.
          \end{cases}
        \]
        Τέλος,
        \begin{align*}
          s
          &=\int_0^{t_r}v_0\,dt
            +\int_{t_r}^{t_r+t_b}v(t)\,dt\\
          &=21+75=96.
        \end{align*}
        Συνεπώς,
        \[
          {s=96\si{m}},
        \]
        όπως ακριβώς αναφέρει ο \en{Highway Code}.

        Οι αποστάσεις αυτές αποτελούν τυπικό οδηγό και όχι εγγύηση:
        στην πράξη επηρεάζονται από τον οδηγό, την κατάσταση του
        οχήματος, τα ελαστικά, το οδόστρωμα και τον καιρό.
\end{alist}
