\subsection*{Λύση Προβλήματος 47 --- Ρυθμός χημικής αντίδρασης}

\begin{alist}
  \item Έχουμε
        \[
          C(t)=e^{-\frac{\ln2}{6}t}.
        \]
        Επειδή
        \[
          C(6)=e^{-\frac{\ln2}{6}\cdot6}=e^{-\ln2}=\frac12,
        \]
        η μέση μεταβολή της συγκέντρωσης στο διάστημα $[6,\,6+h]$ είναι
        \[
          \frac{C(6+h)-C(6)}{h}.
        \]
        Υπολογίζουμε
        \[
          C(6+h)
          =
          e^{-\frac{\ln2}{6}(6+h)}
          =
          e^{-\ln2}\cdot e^{-\frac{\ln2}{6}h}
          =
          \frac12 e^{-\frac{\ln2}{6}h}.
        \]
        Άρα
        \[
          \frac{C(6+h)-C(6)}{h}
          =
          \frac{\frac12 e^{-\frac{\ln2}{6}h}-\frac12}{h}
          =
          \frac12\cdot
          \frac{e^{-\frac{\ln2}{6}h}-1}{h}.
        \]

  \item Θέτουμε
        \[
          u=-\frac{\ln2}{6}h.
        \]
        Όταν $h\to0^+$, τότε και $u\to0$. Επίσης
        \[
          h=-\frac{6}{\ln2}u.
        \]
        Επομένως
        \[
          \frac{e^{-\frac{\ln2}{6}h}-1}{h}
          =
          \frac{e^u-1}{-\frac{6}{\ln2}u}
          =
          -\frac{\ln2}{6}\cdot\frac{e^u-1}{u}.
        \]
        Επειδή
        \[
          \lim_{u\to0}\frac{e^u-1}{u}=1,
        \]
        παίρνουμε
        \[
          \lim_{h\to0^+}
          \frac{C(6+h)-C(6)}{h}
          =
          \frac12\left(-\frac{\ln2}{6}\right)
          =
          -\frac{\ln2}{12}.
        \]
        Άρα
        \[
          {
          \lim_{h\to0^+}
          \frac{C(6+h)-C(6)}{h}
          =
          -\frac{\ln2}{12}
          \approx
          -0{,}0578
          }.
        \]
        Η μονάδα είναι \(\si{mol/L}\) ανά ώρα.

        Το αρνητικό πρόσημο σημαίνει ότι η συγκέντρωση του
        \(\mathrm{H_2O_2}\) μειώνεται τη χρονική στιγμή $t=6$ ώρες.
        Δηλαδή, εκείνη τη στιγμή η συγκέντρωση ελαττώνεται περίπου κατά
        \[
          0{,}0578\si{mol/L}
        \]
        ανά ώρα.

  \item Παραγωγίζουμε τη συνάρτηση
        \[
          C(t)=e^{-\frac{\ln2}{6}t}.
        \]
        Έχουμε
        \[
          C'(t)
          =
          -\frac{\ln2}{6}e^{-\frac{\ln2}{6}t}.
        \]
        Επειδή
        \[
          C(t)=e^{-\frac{\ln2}{6}t},
        \]
        προκύπτει
        \[
          C'(t)=-\frac{\ln2}{6}C(t).
        \]
        Άρα ο θετικός ρυθμός κατανάλωσης είναι
        \[
          -C'(t)=\frac{\ln2}{6}C(t).
        \]
        Επομένως
        \[
          {
          -C'(t)=\frac{\ln2}{6}C(t)
          }.
        \]
        Αυτό δείχνει ότι ο ρυθμός κατανάλωσης είναι ανάλογος της
        τρέχουσας συγκέντρωσης. Όσο λιγότερο \(\mathrm{H_2O_2}\) έχει
        απομείνει, τόσο μικρότερος γίνεται και ο ρυθμός κατανάλωσής του.
\end{alist}
