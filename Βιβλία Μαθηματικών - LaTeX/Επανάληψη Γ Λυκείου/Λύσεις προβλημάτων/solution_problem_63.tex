\subsection*{Λύση Προβλήματος 63 --- Κουτί από φύλλο Α4}

\begin{alist}
  \item Μετά την κοπή τετραγώνων πλευράς \(x\si{cm}\), η βάση του
        κουτιού έχει διαστάσεις
        \[
          21-2x
          \quad\text{και}\quad
          29{,}7-2x.
        \]
        Το ύψος του κουτιού είναι \(x\). Άρα
        \[
          {
          V(x)=x(21-2x)(29{,}7-2x)
          }.
        \]

        Για να σχηματίζεται κουτί, πρέπει
        \[
          x>0,\qquad 21-2x>0,\qquad 29{,}7-2x>0.
        \]
        Επομένως
        \[
          {0<x<10{,}5}.
        \]

  \item Αναπτύσσουμε:
        \[
          V(x)=x(21-2x)(29{,}7-2x).
        \]
        Επειδή
        \[
          (21-2x)(29{,}7-2x)
          =
          623{,}7-101{,}4x+4x^2,
        \]
        έχουμε
        \[
          V(x)=623{,}7x-101{,}4x^2+4x^3.
        \]
        Άρα
        \[
          V'(x)=623{,}7-202{,}8x+12x^2.
        \]

        Λύνουμε την εξίσωση \(V'(x)=0\):
        \[
          12x^2-202{,}8x+623{,}7=0.
        \]
        Οι ρίζες είναι περίπου
        \[
          x_1\approx4{,}04,
          \qquad
          x_2\approx12{,}86.
        \]
        Η δεύτερη ρίζα δεν ανήκει στο πεδίο ορισμού
        \((0,10{,}5)\). Άρα εξετάζουμε μόνο το
        \[
          x_1\approx4{,}04.
        \]

        Επειδή η παράγωγος \(V'\) είναι τριώνυμο με θετικό συντελεστή
        στο \(x^2\), έχουμε
        \[
          V'(x)>0
          \quad\text{για}\quad
          0<x<4{,}04
        \]
        και
        \[
          V'(x)<0
          \quad\text{για}\quad
          4{,}04<x<10{,}5.
        \]
        Επομένως ο όγκος γίνεται μέγιστος για
        \[
          {x\approx4{,}04\si{cm}}.
        \]

  \item Για \(x\approx4{,}04\), οι διαστάσεις της βάσης είναι
        \[
          21-2x\approx21-8{,}08=12{,}92\si{cm}
        \]
        και
        \[
          29{,}7-2x\approx29{,}7-8{,}08=21{,}62\si{cm}.
        \]
        Το ύψος είναι
        \[
          x\approx4{,}04\si{cm}.
        \]
        Άρα οι διαστάσεις του κουτιού είναι περίπου
        \[
          {
          12{,}92\si{cm}\times21{,}62\si{cm}\times4{,}04\si{cm}
          }.
        \]

        Ο μέγιστος όγκος είναι
        \[
          V(4{,}04)\approx1128{,}5\si{cm^3}.
        \]
        Δηλαδή
        \[
          {V_{\max}\approx1128{,}5\si{cm^3}}.
        \]
\end{alist}

\paragraph{Πηγές και έλεγχος ρεαλιστικότητας}
Οι διαστάσεις \(210\si{mm}\times297\si{mm}\), δηλαδή
\(21\si{cm}\times29{,}7\si{cm}\), είναι οι πραγματικές διαστάσεις
φύλλου Α4 στο πρότυπο ISO 216 \cite{iso216PaperSizes}. Το αποτέλεσμα
είναι ρεαλιστικό: το κόψιμο \(x\approx4{,}04\si{cm}\) αφήνει θετικές
διαστάσεις βάσης, περίπου \(12{,}92\si{cm}\) και \(21{,}62\si{cm}\), και
ύψος περίπου \(4{,}04\si{cm}\). Άρα το κουτί μπορεί πράγματι να
κατασκευαστεί από το φύλλο.
