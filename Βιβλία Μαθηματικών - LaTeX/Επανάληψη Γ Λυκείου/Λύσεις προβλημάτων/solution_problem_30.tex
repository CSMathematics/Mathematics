\subsection*{Λύση Προβλήματος 30 --- Το \en{Intelligent Driver Model}}

Θέτουμε
\[
  v_0=\frac{100}{3}\si{m/s},
  \qquad
  a=0{,}73\si{m/s^2}.
\]

\begin{alist}
  \item Για $0<v<v_0$ έχουμε
        \[
          A(v)=a\left[1-\left(\frac{v}{v_0}\right)^4\right]>0.
        \]
        Επομένως, $v'(t)=A(v(t))>0$ και η ταχύτητα αυξάνεται.

        Ως συνάρτηση της ταχύτητας, η επιτάχυνση ικανοποιεί
        \[
          A'(v)=-\frac{4av^3}{v_0^4}<0,
        \]
        άρα μειώνεται όσο αυξάνεται η ταχύτητα. Από τον κανόνα
        αλυσίδας,
        \[
          v''(t)
          =A'(v(t))v'(t)
          =A'(v(t))A(v(t))<0.
        \]
        Συνεπώς, η $v(t)$ είναι γνησίως αύξουσα και κοίλη μέχρι να
        προσεγγίσει την επιθυμητή ταχύτητα $v_0$.

  \item Η αρχική επιτάχυνση είναι $A(0)=a$. Ζητούμε
        \[
          A(v)=\frac{a}{2}.
        \]
        Άρα
        \begin{align*}
          a\left[1-\left(\frac{v}{v_0}\right)^4\right]
          &=\frac{a}{2},\\
          \left(\frac{v}{v_0}\right)^4&=\frac12,\\
          v&=\frac{v_0}{\sqrt[4]{2}}.
        \end{align*}
        Επομένως,
        \[
          {
            v=\frac{120}{\sqrt[4]{2}}\si{km/h}
            \approx100{,}9\si{km/h}
          }.
        \]

  \item Από τη διαφορική εξίσωση
        \[
          \frac{dv}{dt}=A(v)
        \]
        παίρνουμε
        \[
          dt=\frac{dv}{A(v)}.
        \]
        Εφόσον $100\si{km/h}=250/9\si{m/s}$,
        \[
          T=\int_0^{250/9}
          \frac{du}{0{,}73\left[1-(3u/100)^4\right]}.
        \]
        Θέτουμε $z=u/v_0=3u/100$. Τότε, όταν $u=250/9$,
        έχουμε $z=5/6$, οπότε
        \begin{align*}
          T
          &=\frac{v_0}{a}\int_0^{5/6}\frac{dz}{1-z^4}\\
          &=\frac{10000}{219}
            \left[
              \frac14\ln 11
              +\frac12\arctan\left(\frac56\right)
            \right].
        \end{align*}
        Άρα
        \[
          {T\approx43{,}2\si{s}}.
        \]
        Η τιμή συμφωνεί με την περιγραφή της εργασίας ότι το μοντέλο
        χρειάζεται περίπου $45\si{s}$ για την επιτάχυνση από $0$ σε
        $100\si{km/h}$.

  \item Επειδή $dx/dt=v$ και $dv/dt=A(v)$,
        \[
          \frac{dx}{dv}=\frac{v}{A(v)}.
        \]
        Επομένως,
        \[
          S=\int_0^{250/9}
          \frac{u}{a\left[1-(u/v_0)^4\right]}\,du.
        \]
        Θέτουμε $z=u/v_0$. Τότε
        \begin{align*}
          S
          &=\frac{v_0^2}{a}\int_0^{5/6}
            \frac{z}{1-z^4}\,dz\\
          &=\frac{v_0^2}{4a}
            \ln\left(
              \frac{1+(5/6)^2}{1-(5/6)^2}
            \right)\\
          &=\frac{250000}{657}\ln\left(\frac{61}{11}\right).
        \end{align*}
        Άρα
        \[
          {S\approx651{,}8\si{m}}.
        \]
\end{alist}
