\subsection*{Λύση Προβλήματος 27 --- Μεταβαλλόμενη μάζα του \en{Saturn V}}

\begin{alist}
  \item Από τη σχέση
        \[
          m(t)a(t)=T-m(t)g
        \]
        παίρνουμε
        \begin{align*}
          a(t)
          &=\frac{T}{m(t)}-g\\
          &=\frac{34\,000\,000}
          {2\,720\,000-13\,500t}-9{,}8.
        \end{align*}
        Επομένως,
        \begin{align*}
          a(0)
          &=\frac{34\,000\,000}{2\,720\,000}-9{,}8
           =2{,}7\;\si{m/s^2},\\
          a(100)
          &=\frac{34\,000\,000}{1\,370\,000}-9{,}8
           \approx15{,}02\;\si{m/s^2}.
        \end{align*}
        Άρα
        \[
          {a(0)=2{,}7\;\si{m/s^2}},
          \qquad
          {a(100)\approx15{,}0\;\si{m/s^2}}.
        \]

  \item Παραγωγίζοντας τη συνάρτηση της επιτάχυνσης, βρίσκουμε
        \[
          j(t)=a'(t)
          =\frac{34\,000\,000\cdot13\,500}
          {(2\,720\,000-13\,500t)^2}.
        \]
        Για κάθε $t\in[0,100]$ ο παρονομαστής είναι θετικός, επομένως
        \[
          {j(t)>0}.
        \]
        Άρα η επιτάχυνση είναι γνησίως αύξουσα σε όλο το διάστημα. Η
        αύξηση δεν οφείλεται σε αύξηση της ώσης, αλλά στη συνεχή μείωση
        της μάζας καθώς καταναλώνονται τα προωθητικά.

  \item Επειδή $v'(t)=a(t)$ και $v(0)=0$, έχουμε
        \begin{align*}
          v(t)
          &=\int_0^t
          \left(
          \frac{34\,000\,000}{2\,720\,000-13\,500s}
          -9{,}8
          \right)\,ds\\
          &=\frac{34\,000\,000}{13\,500}
          \ln\left(
          \frac{2\,720\,000}{2\,720\,000-13\,500t}
          \right)-9{,}8t.
        \end{align*}
        Για $t=100\si{s}$,
        \begin{align*}
          v(100)
          &=\frac{34\,000\,000}{13\,500}
          \ln\left(\frac{2\,720\,000}{1\,370\,000}\right)-980\\
          &\approx747{,}3\;\si{m/s}.
        \end{align*}
        Συνεπώς,
        \[
          {v(100)\approx747\;\si{m/s}}.
        \]
        Η τιμή είναι αποτέλεσμα του συγκεκριμένου απλοποιημένου
        μοντέλου. Η πραγματική κίνηση επηρεάζεται επιπλέον από την
        αεροδυναμική αντίσταση, τη μεταβολή της ώσης με το ύψος, τη
        στροφή της τροχιάς και τη μεταβολή του $g$.
\end{alist}
