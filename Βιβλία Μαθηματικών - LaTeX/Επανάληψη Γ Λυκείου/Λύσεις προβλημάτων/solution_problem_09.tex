% Λύση του 9ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 9 --- Χρόνος υλοτομίας μιας συστάδας}

\begin{alist}
  \item Για $t=0$ έχουμε
        \[
          V(0)=V_{\max}(1-e^0)^3=0.
        \]
        Επίσης, επειδή
        \[
          \lim_{t\to+\infty}e^{-kt}=0,
        \]
        προκύπτει
        \[
          \lim_{t\to+\infty}V(t)
          =V_{\max}(1-0)^3
          =V_{\max}.
        \]
        Άρα το μοντέλο θεωρεί μηδενικό αρχικό όγκο ξυλείας και προβλέπει
        ότι, όσο περνούν τα χρόνια, ο όγκος πλησιάζει την οριακή τιμή
        $V_{\max}$.

  \item Η σημερινή αξία είναι
        \[
          P(t)=pV_{\max}(1-e^{-kt})^3e^{-rt}.
        \]
        Παραγωγίζουμε με τον κανόνα του γινομένου και τον κανόνα της
        αλυσίδας:
        \begin{align*}
          P'(t)
          &=pV_{\max}\left[
             3(1-e^{-kt})^2ke^{-kt}e^{-rt}
             -r(1-e^{-kt})^3e^{-rt}
             \right]\\
          &=pV_{\max}e^{-rt}(1-e^{-kt})^2
             \left[3ke^{-kt}-r(1-e^{-kt})\right]\\
          &=pV_{\max}e^{-rt}(1-e^{-kt})^2
             \left[(3k+r)e^{-kt}-r\right].
        \end{align*}
        Επομένως,
        \[
          \boxed{
          P'(t)=pV_{\max}e^{-rt}(1-e^{-kt})^2
          \left[(3k+r)e^{-kt}-r\right]}.
        \]

  \item Για $t>0$ οι παράγοντες
        \[
          pV_{\max},\qquad e^{-rt},
          \qquad (1-e^{-kt})^2
        \]
        είναι θετικοί. Άρα το πρόσημο της $P'(t)$ καθορίζεται από τη
        συνάρτηση
        \[
          q(t)=(3k+r)e^{-kt}-r.
        \]
        Η $q$ είναι γνησίως φθίνουσα, επειδή
        \[
          q'(t)=-k(3k+r)e^{-kt}<0.
        \]
        Η εξίσωση $q(t)=0$ δίνει
        \begin{align*}
          (3k+r)e^{-kt}=r
          &\Longleftrightarrow
          e^{-kt}=\frac{r}{3k+r}\\
          &\Longleftrightarrow
          -kt=\ln\left(\frac{r}{3k+r}\right)\\
          &\Longleftrightarrow
          t=\frac{1}{k}
          \ln\left(\frac{3k+r}{r}\right).
        \end{align*}
        Επομένως,
        \[
          \boxed{
          t^*=\frac{1}{k}\ln\left(1+\frac{3k}{r}\right)}.
        \]

        Επειδή η $q$ είναι θετική πριν από το $t^*$ και αρνητική μετά
        από αυτό, έχουμε
        \[
        \begin{array}{c|ccc}
          t & (0,t^*) & t^* & (t^*,+\infty)\\ \hline
          P'(t) & + & 0 & -\\
          P(t) & \nearrow & \text{μέγιστο} & \searrow
        \end{array}
        \]
        Άρα η σημερινή αξία γίνεται ολικά μέγιστη στην ηλικία $t^*$.

  \item Για $k=0{,}03$ και $r=0{,}02$ βρίσκουμε
        \begin{align*}
          t^*
          &=\frac{1}{0{,}03}
            \ln\left(1+\frac{3\cdot0{,}03}{0{,}02}\right)\\
          &=\frac{100}{3}\ln(5{,}5)
          \simeq56{,}8.
        \end{align*}
        Συνεπώς, σύμφωνα με το συγκεκριμένο μοντέλο, η υλοτομία πρέπει
        να γίνει περίπου στα
        \[
          \boxed{57\text{ έτη}}.
        \]

        Οι σταθερές $p$ και $V_{\max}$ δεν επηρεάζουν το $t^*$, επειδή
        εμφανίζονται στην παράγωγο μόνο ως θετικός πολλαπλασιαστικός
        παράγοντας. Επηρεάζουν το ύψος της αξίας, όχι τη χρονική στιγμή
        στην οποία αυτή μεγιστοποιείται.
\end{alist}
