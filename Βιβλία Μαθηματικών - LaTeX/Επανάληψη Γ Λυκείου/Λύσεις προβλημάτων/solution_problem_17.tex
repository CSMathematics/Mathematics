\subsection*{Λύση Προβλήματος 17 --- Έργο της δύναμης Coulomb}

\begin{alist}
  \item Μετατρέπουμε το φορτίο σε Coulomb:
        \[
          q=10\,\si{\micro C}=10\cdot10^{-6}\,\si{C}=10^{-5}\,\si{C}.
        \]
        Από τον νόμο του Coulomb,
        \begin{align*}
          F(r)
          &=9\cdot10^9\frac{(10^{-5})^2}{r^2}\\
          &=\frac{9\cdot10^{-1}}{r^2}.
        \end{align*}
        Άρα
        \[
          \boxed{F(r)=\frac{0{,}9}{r^2}\;\si{N}}.
        \]

  \item Η δύναμη και η μετακίνηση έχουν την ίδια κατεύθυνση. Επομένως,
        το έργο της ηλεκτρικής δύναμης είναι
        \begin{align*}
          W
          &=\int_1^3\frac{0{,}9}{r^2}\,dr\\
          &=0{,}9\int_1^3r^{-2}\,dr\\
          &=0{,}9\left[-\frac{1}{r}\right]_1^3\\
          &=0{,}9\left(1-\frac{1}{3}\right)
           =0{,}6.
        \end{align*}
        Συνεπώς,
        \[
          \boxed{W=0{,}6\;\si{J}}.
        \]

  \item Η δύναμη είναι αντιστρόφως ανάλογη του τετραγώνου της
        απόστασης:
        \[
          F(r)=\frac{0{,}9}{r^2}.
        \]
        Καθώς το $r$ αυξάνεται, το $r^2$ αυξάνεται και επομένως η
        δύναμη μειώνεται. Για παράδειγμα,
        \[
          F(1)=0{,}9\;\si{N},
          \qquad
          F(3)=0{,}1\;\si{N}.
        \]
        Άρα, όταν η απόσταση τριπλασιάζεται, η δύναμη γίνεται εννέα
        φορές μικρότερη.
\end{alist}
