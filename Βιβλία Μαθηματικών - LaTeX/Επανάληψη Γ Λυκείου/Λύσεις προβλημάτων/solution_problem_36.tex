\subsection*{Λύση Προβλήματος 36 --- Κάμερα ασφαλείας και γωνιακή ταχύτητα}

\begin{alist}
  \item Η κάμερα βρίσκεται σε ύψος $5\si{m}$ και ο ύποπτος απέχει
        οριζόντια απόσταση $x(t)$ από τη βάση του τοίχου. Σχηματίζεται
        ορθογώνιο τρίγωνο με κατακόρυφη πλευρά $5$ και οριζόντια πλευρά
        $x(t)$.

        Επειδή η γωνία $\theta(t)$ μετριέται ως προς την κατακόρυφο,
        έχουμε
        \[
          \tan\theta(t)=\frac{x(t)}{5}.
        \]
        Ισοδύναμα,
        \[
          {\theta(t)=\arctan\left(\frac{x(t)}{5}\right)}.
        \]

  \item Παραγωγίζουμε τη σχέση
        \[
          \tan\theta(t)=\frac{x(t)}{5}
        \]
        ως προς τον χρόνο $t$:
        \[
          \frac{1}{\cos^2\theta(t)}\theta'(t)
          =
          \frac{x'(t)}{5}.
        \]
        Άρα
        \[
          \theta'(t)
          =
          \frac{x'(t)}{5}\cos^2\theta(t).
        \]

        Από το ορθογώνιο τρίγωνο, όταν $x=12$, η υποτείνουσα είναι
        \[
          d=\sqrt{5^2+12^2}=13.
        \]
        Επομένως
        \[
          \cos\theta=\frac{5}{13}
          \quad\Rightarrow\quad
          \cos^2\theta=\frac{25}{169}.
        \]
        Επειδή ο ύποπτος απομακρύνεται με ταχύτητα
        \[
          x'(t)=6\si{m/s},
        \]
        παίρνουμε
        \[
          \theta'(t)
          =
          \frac{6}{5}\cdot\frac{25}{169}
          =
          \frac{30}{169}.
        \]

        Άρα, τη στιγμή που ο ύποπτος απέχει $12\si{m}$ από τη βάση του
        τοίχου, η γωνιακή ταχύτητα της κάμερας είναι
        \[
          {\theta'(t)=\frac{30}{169}\si{rad/s}
          \approx0{,}178\si{rad/s}}.
        \]
\end{alist}
