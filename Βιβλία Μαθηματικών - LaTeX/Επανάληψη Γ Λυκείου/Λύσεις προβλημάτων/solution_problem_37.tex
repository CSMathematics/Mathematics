\subsection*{Λύση Προβλήματος 37 --- Φούσκωμα μετεωρολογικού μπαλονιού}

\begin{alist}
  \item Ο όγκος του σφαιρικού μπαλονιού είναι
        \[
          V=\frac{4}{3}\pi r^3.
        \]
        Παραγωγίζουμε ως προς τον χρόνο $t$:
        \[
          V'(t)=4\pi r^2(t)\,r'(t).
        \]
        Δίνεται ότι
        \[
          V'(t)=100\pi\si{cm^3/s}.
        \]
        Τη στιγμή $t_0$ έχουμε $r(t_0)=10\si{cm}$. Άρα
        \[
          100\pi
          =
          4\pi\cdot 10^2\cdot r'(t_0).
        \]
        Επομένως
        \[
          100\pi=400\pi r'(t_0)
        \]
        και άρα
        \[
          {r'(t_0)=\frac14\si{cm/s}}.
        \]

  \item Το εμβαδόν της επιφάνειας του μπαλονιού είναι
        \[
          E=4\pi r^2.
        \]
        Παραγωγίζουμε ως προς τον χρόνο:
        \[
          E'(t)=8\pi r(t)\,r'(t).
        \]
        Τη στιγμή $t_0$ έχουμε
        \[
          r(t_0)=10\si{cm},
          \qquad
          r'(t_0)=\frac14\si{cm/s}.
        \]
        Άρα
        \[
          E'(t_0)
          =
          8\pi\cdot10\cdot\frac14
          =
          20\pi.
        \]
        Επομένως, εκείνη τη στιγμή το εμβαδόν της επιφάνειας αυξάνεται
        με ρυθμό
        \[
          {E'(t_0)=20\pi\si{cm^2/s}}.
        \]
\end{alist}
