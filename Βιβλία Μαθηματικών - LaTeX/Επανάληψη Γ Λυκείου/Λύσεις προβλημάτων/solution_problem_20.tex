% Λύση του 20ού προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 20 --- Υδροστατική δύναμη σε φράγμα}

\begin{alist}
  \item Σε βάθος $y$, η πίεση του νερού είναι
        \[
          p(y)=\rho gy=1000\cdot9{,}8\,y=9800y\;\si{Pa}.
        \]
        Η οριζόντια λωρίδα έχει εμβαδόν
        \[
          dA=b\,dy=500\,dy.
        \]
        Επομένως,
        \[
          dF=p(y)\,dA
             =9800y\cdot500\,dy
             ={4\,900\,000y\,dy}.
        \]

  \item Η συνολική υδροστατική δύναμη είναι
        \begin{align*}
          F
          &=\int_0^{80}4\,900\,000y\,dy\\
          &=4\,900\,000
            \left[\frac{y^2}{2}\right]_0^{80}\\
          &=4\,900\,000\cdot3200\\
          &=15\,680\,000\,000\si{N}.
        \end{align*}
        Άρα
        \[
          {F=1{,}568\cdot10^{10}\si{N}}.
        \]

  \item Για το κέντρο πίεσης έχουμε
        \begin{align*}
          \bar y
          &=\frac{\displaystyle
              \int_0^{80}y\cdot4\,900\,000y\,dy}
             {\displaystyle
              \int_0^{80}4\,900\,000y\,dy}\\
          &=\frac{\displaystyle\int_0^{80}y^2\,dy}
                  {\displaystyle\int_0^{80}y\,dy}\\
          &=\frac{\dfrac{80^3}{3}}{\dfrac{80^2}{2}}
           =\frac{160}{3}.
        \end{align*}
        Συνεπώς,
        \[
          {\bar y=\frac{160}{3}\si{m}
          \approx53{,}3\si{m}}
        \]
        κάτω από την ελεύθερη επιφάνεια. Το σημείο βρίσκεται χαμηλότερα
        από το μέσο βάθος των $40\si{m}$, επειδή η πίεση αυξάνεται
        γραμμικά όσο πλησιάζουμε στον πυθμένα.
\end{alist}
