% Λύση του 53ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 53 --- Μοντέλο EOQ}

\begin{alist}
  \item Έχουμε
        \[
          C(Q)=\frac{SD}{Q}+\frac{HQ}{2},
          \qquad Q>0.
        \]
        Από τον ορισμό της παραγώγου,
        \[
          C'(Q)
          =
          \lim_{u\to0}
          \frac{C(Q+u)-C(Q)}{u}.
        \]
        Υπολογίζουμε:
        \[
          C(Q+u)
          =
          \frac{SD}{Q+u}+\frac{H(Q+u)}{2}.
        \]
        Άρα
        \[
          C(Q+u)-C(Q)
          =
          SD\left(\frac{1}{Q+u}-\frac{1}{Q}\right)
          +\frac{Hu}{2}.
        \]
        Επομένως
        \[
          \frac{C(Q+u)-C(Q)}{u}
          =
          SD\cdot
          \frac{\frac{1}{Q+u}-\frac{1}{Q}}{u}
          +\frac{H}{2}.
        \]
        Όμως
        \[
          \frac{1}{Q+u}-\frac{1}{Q}
          =
          \frac{Q-(Q+u)}{Q(Q+u)}
          =
          -\frac{u}{Q(Q+u)}.
        \]
        Άρα
        \[
          \frac{\frac{1}{Q+u}-\frac{1}{Q}}{u}
          =
          -\frac{1}{Q(Q+u)}.
        \]
        Συνεπώς
        \[
          \frac{C(Q+u)-C(Q)}{u}
          =
          -\frac{SD}{Q(Q+u)}+\frac{H}{2}.
        \]
        Παίρνοντας όριο για \(u\to0\), έχουμε
        \[
          C'(Q)
          =
          -\frac{SD}{Q^2}+\frac{H}{2}.
        \]
        Άρα
        \[
          {
          C'(Q)=-\frac{SD}{Q^2}+\frac{H}{2}
          }.
        \]

  \item Για ελάχιστο λύνουμε την εξίσωση
        \[
          C'(Q)=0.
        \]
        Δηλαδή
        \[
          -\frac{SD}{Q^2}+\frac{H}{2}=0.
        \]
        Άρα
        \[
          \frac{H}{2}=\frac{SD}{Q^2}.
        \]
        Επομένως
        \[
          HQ^2=2SD.
        \]
        Επειδή \(Q>0\), παίρνουμε
        \[
          {
          Q^*=\sqrt{\frac{2SD}{H}}
          }.
        \]
        Αυτό είναι το κλασικό αποτέλεσμα του μοντέλου EOQ.

        Για \(Q<Q^*\) το οριακό κόστος είναι αρνητικό, άρα το συνολικό
        κόστος μειώνεται όταν αυξάνεται το \(Q\). Για \(Q>Q^*\) το
        οριακό κόστος είναι θετικό, άρα το συνολικό κόστος αυξάνεται.
        Επομένως το \(Q^*\) δίνει ελάχιστο.

  \item Με
        \[
          D=10000,\qquad S=40,\qquad H=5
        \]
        έχουμε
        \[
          Q^*
          =
          \sqrt{\frac{2\cdot40\cdot10000}{5}}.
        \]
        Άρα
        \[
          Q^*
          =
          \sqrt{160000}
          =
          400.
        \]
        Επομένως
        \[
          {Q^*=400\ \text{μονάδες ανά παραγγελία}}.
        \]

        Στο \(Q=400\), το ετήσιο κόστος παραγγελιών είναι
        \[
          \frac{SD}{Q}
          =
          \frac{40\cdot10000}{400}
          =
          1000.
        \]
        Το ετήσιο κόστος διατήρησης αποθέματος είναι
        \[
          \frac{HQ}{2}
          =
          \frac{5\cdot400}{2}
          =
          1000.
        \]
        Άρα στο βέλτιστο σημείο τα δύο κόστη είναι ίσα:
        \[
          {
          \frac{SD}{Q^*}=\frac{HQ^*}{2}=1000
          }.
        \]
        Το συνολικό μεταβλητό ετήσιο κόστος είναι
        \[
          C(400)=1000+1000=2000.
        \]
\end{alist}
