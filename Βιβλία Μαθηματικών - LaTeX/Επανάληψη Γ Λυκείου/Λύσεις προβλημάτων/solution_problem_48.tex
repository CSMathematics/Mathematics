% Λύση του 48ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 48 --- Αισθητήρας Pt100}

\begin{alist}
  \item Έχουμε
        \[
          R(T)=100(1+AT+BT^2),
        \]
        όπου
        \[
          A=3{,}9083\cdot10^{-3},
          \qquad
          B=-5{,}775\cdot10^{-7}.
        \]
        Άρα
        \[
          R(T)-100
          =
          100(AT+BT^2)
          =
          100T(A+BT).
        \]
        Επομένως
        \[
          \frac{R(T)-100}{T}
          =
          100(A+BT).
        \]
        Παίρνοντας όριο για $T\to0^+$, έχουμε
        \[
          \lim_{T\to0^+}\frac{R(T)-100}{T}
          =
          100A.
        \]
        Άρα
        \[
          {
          \lim_{T\to0^+}\frac{R(T)-100}{T}
          =
          100\cdot3{,}9083\cdot10^{-3}
          =
          0{,}39083
          }.
        \]
        Η μονάδα είναι \(\si{\ohm}\) ανά \({}^\circ C\). Δηλαδή, πολύ
        κοντά στους \(0^\circ C\), η αντίσταση αυξάνεται αρχικά με ρυθμό
        περίπου
        \[
          0{,}39083\si{\ohm}
        \]
        για κάθε αύξηση θερμοκρασίας κατά \(1^\circ C\).

  \item Παραγωγίζουμε:
        \[
          R'(T)=100(A+2BT).
        \]
        Με τους συγκεκριμένους συντελεστές,
        \[
          R'(T)
          =
          100\left(3{,}9083\cdot10^{-3}
          -2\cdot5{,}775\cdot10^{-7}T\right).
        \]
        Δηλαδή
        \[
          R'(T)
          =
          100\left(3{,}9083\cdot10^{-3}
          -1{,}155\cdot10^{-6}T\right).
        \]
        Στο άκρο του διαστήματος εγκυρότητας, για $T=850$, έχουμε
        \[
          3{,}9083\cdot10^{-3}
          -1{,}155\cdot10^{-6}\cdot850
          =
          0{,}00292655>0.
        \]
        Άρα για κάθε
        \[
          0\leq T\leq850
        \]
        ισχύει
        \[
          R'(T)>0.
        \]
        Επομένως η αντίσταση είναι αύξουσα στο διάστημα εγκυρότητας
        του μοντέλου.

  \item Για \(T=100^\circ C\) έχουμε
        \[
          R(100)
          =
          100\left(1+A\cdot100+B\cdot100^2\right).
        \]
        Άρα
        \[
          R(100)
          =
          100\left(
          1+3{,}9083\cdot10^{-3}\cdot100
          -5{,}775\cdot10^{-7}\cdot10000
          \right).
        \]
        Επομένως
        \[
          R(100)
          =
          100(1+0{,}39083-0{,}005775)
          =
          138{,}5055.
        \]
        Άρα
        \[
          {R(100)\approx138{,}51\si{\ohm}}.
        \]

  \item Λύνουμε αλγεβρικά την εξίσωση
        \[
          R'(T)=0.
        \]
        Επειδή
        \[
          R'(T)=100(A+2BT),
        \]
        έχουμε
        \[
          A+2BT=0.
        \]
        Άρα
        \[
          T=-\frac{A}{2B}.
        \]
        Με αντικατάσταση:
        \[
          T
          =
          -\frac{3{,}9083\cdot10^{-3}}
          {2(-5{,}775\cdot10^{-7})}
          \approx
          3383{,}81.
        \]
        Δηλαδή το αλγεβρικό κρίσιμο σημείο είναι περίπου
        \[
          {3384^\circ C}.
        \]
        Όμως το μοντέλο του προβλήματος δίνεται μόνο για
        \[
          0\leq T\leq850.
        \]
        Άρα το \(3384^\circ C\) δεν ανήκει στο διάστημα εγκυρότητας και
        δεν πρέπει να ερμηνευτεί ως πραγματική θερμοκρασία μέγιστης
        αντίστασης του αισθητήρα.
\end{alist}
