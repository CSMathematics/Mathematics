\subsection*{Λύση Προβλήματος 2 --- Ιδανικό μοντέλο εκθετικής διαστολής}

\begin{alist}
  \item Μετατρέπουμε πρώτα τη σταθερά του \en{Hubble} σε
        \si{Gyr^{-1}}:
        \begin{align*}
          H_0
          &=67{,}4\;
            \frac{\si{km}}{\si{s}\,\si{Mpc}}
            \cdot
            \frac{1\;\si{Mpc}}{3{,}0857\cdot10^{19}\;\si{km}}
            \cdot
            \frac{3{,}15576\cdot10^{16}\;\si{s}}{1\;\si{Gyr}}\\
          &\simeq 0{,}06893\;\si{Gyr^{-1}}.
        \end{align*}
        Άρα $k\simeq0{,}06893$. Επιπλέον,
        \[
          d(0)=Ae^{k\cdot0}=A=100.
        \]
        Επομένως,
        \[
          {d(t)=100e^{0{,}06893t}},
          \qquad t\geq0.
        \]

  \item Παραγωγίζοντας τη συνάρτηση βρίσκουμε
        \[
          d'(t)
          =100\cdot0{,}06893e^{0{,}06893t}
          =0{,}06893\,d(t).
        \]
        Δηλαδή,
        \[
          {d'(t)=k\,d(t)}.
        \]
        Η παράγωγος $d'(t)$ εκφράζει την ταχύτητα με την οποία αυξάνεται
        η απόσταση. Η ισότητα δηλώνει ότι η ταχύτητα απομάκρυνσης είναι
        ανάλογη της απόστασης, δηλαδή αποτελεί τη μορφή του νόμου του
        \en{Hubble} μέσα στο συγκεκριμένο μοντέλο.

        Για $t=0$:
        \[
          d'(0)=0{,}06893\cdot100
          =6{,}893\;\si{Mpc.Gyr^{-1}}.
        \]
        Ισοδύναμα, από τον νόμο $v=H_0d$,
        \[
          v(0)=67{,}4\cdot100
          ={6740\;\si{km.s^{-1}}}.
        \]

  \item Αν $T$ είναι ο χρόνος διπλασιασμού, τότε
        \begin{align*}
          d(T)=2d(0)
          &\Longleftrightarrow
          100e^{0{,}06893T}=200\\
          &\Longleftrightarrow
          e^{0{,}06893T}=2\\
          &\Longleftrightarrow
          T=\frac{\ln2}{0{,}06893}.
        \end{align*}
        Συνεπώς,
        \[
          {T\simeq10{,}06\;\si{Gyr}}.
        \]

  \item Για κάθε $t\geq0$ ισχύει
        \[
          d'(t)=100k e^{kt}>0.
        \]
        Επομένως, η $d$ είναι γνησίως αύξουσα στο $[0,+\infty)$.
        Επίσης,
        \[
          d''(t)=100k^2e^{kt}>0,
        \]
        άρα η $d$ είναι κυρτή στο ίδιο διάστημα. Τέλος,
        \[
          {\lim_{t\to+\infty}d(t)=+\infty}.
        \]

        Στο ιδανικό μοντέλο, η απόσταση και η ταχύτητα απομάκρυνσης
        αυξάνονται συνεχώς. Η κυρτότητα εκφράζει την αύξηση της απόλυτης
        ταχύτητας $d'(t)$, ενώ ο σχετικός ρυθμός
        \[
          \frac{d'(t)}{d(t)}=k
        \]
        παραμένει σταθερός. Τα συμπεράσματα αυτά αφορούν το μοντέλο
        σταθερού ρυθμού διαστολής και δεν επεκτείνονται αυτούσια σε όλες
        τις εποχές της πραγματικής κοσμικής εξέλιξης.
\end{alist}

\paragraph{Έλεγχος ρεαλιστικότητας}
Η τιμή του $H_0$ προέρχεται από τα δεδομένα της αποστολής \en{Planck}
\cite{planck2020cosmologicalParameters}. Η εκθετική συνάρτηση δεν είναι
αυθαίρετη: προκύπτει όταν ο ρυθμός διαστολής θεωρείται σταθερός, όπως
στο ιδανικό μοντέλο τύπου \en{de Sitter}
\cite{frieman2008darkEnergy}. Η υπόθεση σταθερού $H_0$ αποτελεί
ελεγχόμενη επιστημονική εξιδανίκευση και δηλώνεται ρητά στην εκφώνηση.
