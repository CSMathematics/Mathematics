% Λύση του 50ού προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 50 --- Ταχύτητα του ήχου}

\begin{alist}
  \item Θεωρούμε τη συνάρτηση
        \[
          f(T)=v(T)-340.
        \]
        Δηλαδή
        \[
          f(T)=331\sqrt{1+\frac{T}{273}}-340.
        \]
        Η \(f\) είναι συνεχής στο διάστημα \([10,20]\), γιατί προκύπτει
        από πράξεις συνεχών συναρτήσεων και το υπόρριζο είναι θετικό στο
        διάστημα αυτό.

        Υπολογίζουμε:
        \[
          f(10)
          =
          331\sqrt{1+\frac{10}{273}}-340
          \approx
          337{,}01-340
          =
          -2{,}99<0.
        \]
        Επίσης
        \[
          f(20)
          =
          331\sqrt{1+\frac{20}{273}}-340
          \approx
          342{,}91-340
          =
          2{,}91>0.
        \]
        Άρα
        \[
          f(10)\cdot f(20)<0.
        \]
        Επομένως, από το θεώρημα Bolzano, υπάρχει τουλάχιστον ένα
        \[
          T^*\in(10,20)
        \]
        τέτοιο ώστε
        \[
          f(T^*)=0.
        \]
        Δηλαδή
        \[
          v(T^*)=340\si{m/s}.
        \]

  \item Για να βρούμε προσεγγιστικά τη θερμοκρασία, λύνουμε την εξίσωση
        \[
          331\sqrt{1+\frac{T}{273}}=340.
        \]
        Διαιρούμε με το \(331\):
        \[
          \sqrt{1+\frac{T}{273}}=\frac{340}{331}.
        \]
        Υψώνουμε στο τετράγωνο:
        \[
          1+\frac{T}{273}=\left(\frac{340}{331}\right)^2.
        \]
        Άρα
        \[
          \frac{T}{273}=\left(\frac{340}{331}\right)^2-1.
        \]
        Επομένως
        \[
          T
          =
          273\left[\left(\frac{340}{331}\right)^2-1\right].
        \]
        Υπολογίζουμε
        \[
          T\approx 15{,}05.
        \]
        Άρα
        \[
          {T^*\approx15{,}05^\circ C}.
        \]

  \item Από τους υπολογισμούς του πρώτου ερωτήματος,
        \[
          v(10)\approx337{,}01\si{m/s}<340\si{m/s}
        \]
        ενώ
        \[
          v(20)\approx342{,}91\si{m/s}>340\si{m/s}.
        \]
        Η ταχύτητα του ήχου αυξάνεται με τη θερμοκρασία στο διάστημα που
        εξετάζουμε. Άρα, πρακτικά, για θερμοκρασίες μικρότερες από
        περίπου
        \[
          15{,}05^\circ C
        \]
        η ταχύτητα του ήχου είναι μικρότερη από \(340\si{m/s}\), ενώ για
        θερμοκρασίες μεγαλύτερες από περίπου \(15{,}05^\circ C\) είναι
        μεγαλύτερη από \(340\si{m/s}\).
\end{alist}
