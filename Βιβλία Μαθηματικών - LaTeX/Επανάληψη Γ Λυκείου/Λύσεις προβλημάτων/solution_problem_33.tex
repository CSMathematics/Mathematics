% Λύση του 33ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 33 --- Ζαχαρούχα ποτά και ελαστικότητα}

\begin{alist}
  \item Για $t=1$ έχουμε
        \[
          D(1)=100\cdot0{,}76=76.
        \]
        Άρα το μοντέλο δίνει μείωση από $100$ σε $76$ μονάδες δείκτη,
        δηλαδή μείωση $24\%$.

        Επίσης,
        \[
          D'(t)=100\cdot0{,}76^t\ln(0{,}76).
        \]
        Επειδή $0{,}76<1$, ισχύει $\ln(0{,}76)<0$. Άρα
        \[
          D'(t)<0,\qquad t\geq0,
        \]
        οπότε η ζήτηση είναι γνησίως φθίνουσα.

  \item Η ελαστικότητα ως προς τον φόρο είναι
        \[
          E(t)=\frac{tD'(t)}{D(t)}.
        \]
        Αντικαθιστώντας,
        \[
          E(t)
          =
          \frac{
            t\cdot100\cdot0{,}76^t\ln(0{,}76)
          }{
            100\cdot0{,}76^t
          }.
        \]
        Επομένως,
        \[
          {E(t)=t\ln(0{,}76)}.
        \]
        Για $t=1$ παίρνουμε
        \[
          {E(1)=\ln(0{,}76)\approx-0{,}274}.
        \]

  \item Ο δείκτης φορολογικών εσόδων είναι
        \[
          R(t)=tD(t)=100t\cdot0{,}76^t.
        \]
        Παραγωγίζουμε:
        \[
          R'(t)=100\cdot0{,}76^t
          +100t\cdot0{,}76^t\ln(0{,}76).
        \]
        Άρα
        \[
          R'(t)=100\cdot0{,}76^t
          \bigl(1+t\ln(0{,}76)\bigr).
        \]
        Επειδή $100\cdot0{,}76^t>0$, το πρόσημο του $R'(t)$
        καθορίζεται από τον παράγοντα
        \[
          1+t\ln(0{,}76).
        \]
        Λύνουμε
        \[
          1+t\ln(0{,}76)=0.
        \]
        Έτσι,
        \[
          t_0=-\frac{1}{\ln(0{,}76)}
          \approx3{,}64.
        \]
        Επειδή ο παράγοντας $1+t\ln(0{,}76)$ είναι φθίνουσα γραμμική
        συνάρτηση του $t$, το $R(t)$ αυξάνεται για
        $0<t<t_0$ και μειώνεται για $t>t_0$.

        Άρα ο δείκτης εσόδων μεγιστοποιείται για
        \[
          {t_0\approx3{,}64\ \en{cent/oz}}.
        \]
\end{alist}
