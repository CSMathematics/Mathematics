\subsection*{Λύση Προβλήματος 7 --- Μέγιστη ευαισθησία στο μοντέλο Hill}

\begin{alist}
  \item Για $x=K$ έχουμε
        \[
          E(K)
          =E_{\max}\frac{K^2}{K^2+K^2}
          =\frac{E_{\max}}{2}.
        \]
        Επομένως, η παράμετρος $K=EC_{50}$ είναι πράγματι η συγκέντρωση
        που προκαλεί το μισό της μέγιστης απόκρισης.

        Επιπλέον,
        \begin{align*}
          \lim_{x\to+\infty}E(x)
          &=E_{\max}
            \lim_{x\to+\infty}
            \frac{x^2}{K^2+x^2}\\
          &=E_{\max}
            \lim_{x\to+\infty}
            \frac{1}{\frac{K^2}{x^2}+1}
            =E_{\max}.
        \end{align*}
        Άρα η απόκριση τείνει ασυμπτωτικά στη μέγιστη τιμή
        $E_{\max}$.

  \item Εφαρμόζοντας τον κανόνα παραγώγισης πηλίκου, βρίσκουμε
        \begin{align*}
          S(x)=E'(x)
          &=E_{\max}
            \frac{2x(K^2+x^2)-x^2(2x)}
                 {(K^2+x^2)^2}\\
          &=E_{\max}
            \frac{2K^2x}{(K^2+x^2)^2}.
        \end{align*}
        Επομένως,
        \[
          {
          S(x)=\frac{2E_{\max}K^2x}{(K^2+x^2)^2}}.
        \]

  \item Για να βρούμε πότε η ευαισθησία γίνεται μέγιστη,
        παραγωγίζουμε τη $S$. Γράφουμε
        \[
          S(x)=2E_{\max}K^2x(K^2+x^2)^{-2}.
        \]
        Με τον κανόνα του γινομένου και τον κανόνα της αλυσίδας,
        \begin{align*}
          S'(x)
          &=2E_{\max}K^2
            \left[
              (K^2+x^2)^{-2}
              -4x^2(K^2+x^2)^{-3}
            \right]\\
          &=2E_{\max}K^2
            \frac{K^2-3x^2}{(K^2+x^2)^3}.
        \end{align*}

        Για $x\geq0$ οι παράγοντες
        $2E_{\max}K^2$ και $(K^2+x^2)^3$ είναι θετικοί. Συνεπώς, το
        πρόσημο της $S'(x)$ καθορίζεται από το $K^2-3x^2$. Έχουμε
        \[
          S'(x)=0
          \Longleftrightarrow
          K^2-3x^2=0
          \Longleftrightarrow
          x=\frac{K}{\sqrt3}.
        \]
        Άρα,
        \[
        \begin{array}{c|ccc}
          x & \left[0,\frac{K}{\sqrt3}\right)
            & \frac{K}{\sqrt3}
            & \left(\frac{K}{\sqrt3},+\infty\right)\\ \hline
          S'(x) & + & 0 & -\\
          S(x) & \nearrow & \text{μέγιστο} & \searrow
        \end{array}
        \]
        και η ευαισθησία παρουσιάζει ολικό μέγιστο για
        \[
          {x=\frac{K}{\sqrt3}
          =\frac{EC_{50}}{\sqrt3}}.
        \]

  \item Στη συγκέντρωση μέγιστης ευαισθησίας ισχύει
        $x^2=\dfrac{K^2}{3}$. Επομένως,
        \begin{align*}
          E\left(\frac{K}{\sqrt3}\right)
          &=E_{\max}
            \frac{\frac{K^2}{3}}
                 {K^2+\frac{K^2}{3}}\\
          &=E_{\max}
            \frac{\frac13}{\frac43}
            =\frac{E_{\max}}{4}.
        \end{align*}
        Άρα, όταν η απόκριση μεταβάλλεται με τον μεγαλύτερο ρυθμό, έχει
        επιτευχθεί το
        \[
          {25\%}
        \]
        της μέγιστης απόκρισης.
\end{alist}
