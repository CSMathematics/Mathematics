\subsection*{Λύση Προβλήματος 41 --- Καμπύλη \en{Keeling} και εφαπτομένη}

\begin{alist}
  \item Το μοντέλο είναι
        \[
          C(t)=-0{,}395t^2+3{,}925t+421{,}08.
        \]
        Για $t=0$ έχουμε
        \[
          C(0)=421{,}08.
        \]
        Για $t=1$ έχουμε
        \[
          C(1)=-0{,}395+3{,}925+421{,}08=424{,}61.
        \]
        Για $t=2$ έχουμε
        \[
          C(2)=-0{,}395\cdot4+3{,}925\cdot2+421{,}08=427{,}35.
        \]
        Άρα το μοντέλο περνά από τις τρεις πραγματικές ετήσιες τιμές.

  \item Παραγωγίζουμε:
        \[
          C'(t)=-0{,}79t+3{,}925.
        \]
        Για το έτος $2024$ έχουμε $t=1$. Άρα
        \[
          C'(1)=-0{,}79+3{,}925=3{,}135.
        \]
        Επίσης
        \[
          C(1)=424{,}61.
        \]
        Η εφαπτομένη στο σημείο $(1,424{,}61)$ έχει εξίσωση
        \[
          y-C(1)=C'(1)(t-1).
        \]
        Επομένως
        \[
          {y-424{,}61=3{,}135(t-1)}.
        \]
        Ισοδύναμα,
        \[
          {y=3{,}135t+421{,}475}.
        \]

  \item Για το έτος $2025$ έχουμε $t=2$. Η εφαπτομένη δίνει
        \[
          y(2)=3{,}135\cdot2+421{,}475=427{,}745.
        \]
        Άρα η εκτίμηση μέσω της εφαπτομένης είναι
        \[
          427{,}745\ \en{ppm}.
        \]
        Η πραγματική ετήσια τιμή για το $2025$ είναι
        \[
          427{,}35\ \en{ppm}.
        \]
        Η διαφορά είναι
        \[
          427{,}745-427{,}35=0{,}395.
        \]
        Επομένως η εφαπτομένη υπερεκτιμά την τιμή του $2025$ κατά
        περίπου
        \[
          {0{,}395\ \en{ppm}}.
        \]
\end{alist}
