% Λύση του 26ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 26 --- Πραγματικό μοντέλο αγώνα $100$ μέτρων}

\begin{alist}
  \item Παραγωγίζοντας την ταχύτητα, βρίσκουμε
        \begin{align*}
          a(t)=v'(t)
          &=12{,}74\left(
            -0{,}0113e^{-0{,}0113t}
            +0{,}668e^{-0{,}668t}
          \right).
        \end{align*}
        Επομένως,
        \begin{align*}
          a(0)
          &=12{,}74(0{,}668-0{,}0113)\\
          &\approx8{,}37\;\si{m/s^2}.
        \end{align*}
        Άρα
        \[
          {a(0)\approx8{,}37\;\si{m/s^2}}.
        \]

  \item Αναζητούμε το πρόσημο της $a(t)$. Έχουμε
        \begin{align*}
          a(t)>0
          &\Leftrightarrow
          0{,}668e^{-0{,}668t}
          >0{,}0113e^{-0{,}0113t}\\
          &\Leftrightarrow
          e^{-0{,}6567t}>
          \frac{0{,}0113}{0{,}668}.
        \end{align*}
        Η εξίσωση $a(t)=0$ έχει τη μοναδική λύση
        \begin{align*}
          t_0
          &=\frac{\ln(0{,}668/0{,}0113)}{0{,}668-0{,}0113}\\
          &\approx6{,}21\;\si{s}.
        \end{align*}
        Επειδή η συνάρτηση $e^{-0{,}6567t}$ είναι γνησίως φθίνουσα,
        ισχύει
        \[
        \begin{array}{c|ccc}
          t &(0,t_0)&t_0&(t_0,9{,}88)\\ \hline
          a(t)&+&0&-
        \end{array}
        \]
        και επομένως η $v$ αυξάνεται μέχρι το $t_0$ και στη συνέχεια
        μειώνεται.

        Η μέγιστη ταχύτητα είναι
        \begin{align*}
          v(t_0)
          &=12{,}74\left(
            e^{-0{,}0113t_0}-e^{-0{,}668t_0}
          \right)\\
          &\approx11{,}68\;\si{m/s}.
        \end{align*}
        Συνεπώς,
        \[
          {t_0\approx6{,}21\;\si{s},
          \qquad v_{\max}\approx11{,}68\;\si{m/s}}.
        \]

  \item Επειδή $x'(t)=v(t)$ και $x(0)=0$, έχουμε
        \begin{align*}
          x(t)
          &=\int_0^t12{,}74
            \left(e^{-0{,}0113s}-e^{-0{,}668s}\right)\,ds\\
          &=12{,}74\left(
            \frac{1-e^{-0{,}0113t}}{0{,}0113}
            -\frac{1-e^{-0{,}668t}}{0{,}668}
          \right).
        \end{align*}
        Άρα
        \[
          {x(t)=12{,}74\left(
          \frac{1-e^{-0{,}0113t}}{0{,}0113}
          -\frac{1-e^{-0{,}668t}}{0{,}668}
          \right)}.
        \]
        Για $t=9{,}88\si{s}$ προκύπτει
        \[
          x(9{,}88)\approx100{,}05\;\si{m}.
        \]
        Η απόκλιση των περίπου $5\si{cm}$ από τα $100\si{m}$ είναι πολύ
        μικρή και οφείλεται στη στρογγυλοποίηση των τριών παραμέτρων του
        εμπειρικού μοντέλου. Επομένως, το μοντέλο αναπαράγει με πολύ
        καλή ακρίβεια την απόσταση και τον χρόνο του πραγματικού αγώνα.
\end{alist}
