% Λύση του 34ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 34 --- Άνθρακας-14 και ασύμπτωτη}

\begin{alist}
  \item Για $t=5730$ έχουμε
        \[
          C(5730)
          =
          100\left(\frac12\right)^{\frac{5730}{5730}}
          =
          100\cdot\frac12
          =
          50.
        \]
        Άρα μετά από μία ημιζωή απομένει το $50\%$ της αρχικής
        ποσότητας.

        Επίσης, επειδή $11460=2\cdot5730$,
        \[
          C(11460)
          =
          100\left(\frac12\right)^2
          =
          25.
        \]
        Άρα μετά από δύο ημιζωές απομένει το $25\%$.

  \item Γράφουμε
        \[
          C(t)=100e^{\frac{t}{5730}\ln(1/2)}.
        \]
        Επομένως
        \[
          C'(t)
          =
          \frac{100\ln(1/2)}{5730}
          \left(\frac12\right)^{\frac{t}{5730}}.
        \]
        Επειδή $\ln(1/2)<0$ και
        $\left(\frac12\right)^{t/5730}>0$, ισχύει
        \[
          C'(t)<0,\qquad t\geq0.
        \]
        Άρα το ποσοστό $^{14}\mathrm{C}$ μειώνεται συνεχώς.

        Για να δούμε πώς αλλάζει ο ρυθμός μείωσης, παραγωγίζουμε ξανά:
        \[
          C''(t)
          =
          \frac{100\bigl(\ln(1/2)\bigr)^2}{5730^2}
          \left(\frac12\right)^{\frac{t}{5730}}>0.
        \]
        Άρα η $C'$ είναι αύξουσα. Επειδή η $C'$ είναι αρνητική, αυτό
        σημαίνει ότι γίνεται όλο και λιγότερο αρνητική. Δηλαδή η
        ποσότητα μειώνεται, αλλά η μείωση γίνεται όλο και πιο αργή.

  \item Θέλουμε να βρούμε το $t$ για το οποίο
        \[
          C(t)=10.
        \]
        Έχουμε
        \[
          100\left(\frac12\right)^{\frac{t}{5730}}=10
        \]
        και άρα
        \[
          \left(\frac12\right)^{\frac{t}{5730}}=0{,}1.
        \]
        Παίρνουμε λογαρίθμους:
        \[
          \frac{t}{5730}\ln\left(\frac12\right)=\ln(0{,}1).
        \]
        Επομένως
        \[
          t=
          5730\cdot
          \frac{\ln(0{,}1)}{\ln(1/2)}
          \approx19035.
        \]
        Άρα το $10\%$ της αρχικής ποσότητας απομένει μετά από περίπου
        \[
          {19035\ \text{έτη}}.
        \]

  \item Επειδή $0<\frac12<1$, έχουμε
        \[
          \lim_{t\to+\infty}
          \left(\frac12\right)^{\frac{t}{5730}}
          =
          0.
        \]
        Άρα
        \[
          {\lim_{t\to+\infty}C(t)=0}.
        \]
        Η οριζόντια ασύμπτωτη είναι η ευθεία
        \[
          {y=0}.
        \]
        Στο μοντέλο αυτό, όσο περνούν τα χρόνια, η ποσότητα
        $^{14}\mathrm{C}$ πλησιάζει το $0\%$ χωρίς να γίνεται αρνητική.
\end{alist}
