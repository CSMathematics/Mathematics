\subsection*{Λύση Προβλήματος 8 --- Οικονομική ταχύτητα πτήσης}

Η συνάρτηση κόστους είναι
\[
  K(v)=d\left(\frac{A}{v}+Bv+\frac{C}{v^3}\right),
  \qquad v>0,
\]
όπου $A,B,C,d>0$.

\begin{alist}
  \item Όταν $v\to0^+$, οι όροι $\dfrac{A}{v}$ και
        $\dfrac{C}{v^3}$ τείνουν στο $+\infty$. Επομένως,
        \[
          {\lim_{v\to0^+}K(v)=+\infty}.
        \]

        Όταν $v\to+\infty$, ο όρος $Bv$ τείνει στο $+\infty$, ενώ οι
        άλλοι δύο όροι τείνουν στο μηδέν. Άρα,
        \[
          {\lim_{v\to+\infty}K(v)=+\infty}.
        \]

  \item Παραγωγίζουμε:
        \begin{align*}
          K'(v)
          &=d\left(-\frac{A}{v^2}+B-\frac{3C}{v^4}\right)\\
          &=\frac{d}{v^4}
            \left(Bv^4-Av^2-3C\right).
        \end{align*}
        Επομένως,
        \[
          {
          K'(v)=\frac{d}{v^4}
          \left(Bv^4-Av^2-3C\right)}.
        \]

  \item Επειδή $\dfrac{d}{v^4}>0$, έχουμε
        \[
          K'(v)=0
          \Longleftrightarrow
          Bv^4-Av^2-3C=0.
        \]
        Θέτουμε $y=v^2$, οπότε $y>0$. Η εξίσωση γίνεται
        \[
          By^2-Ay-3C=0.
        \]
        Η διακρίνουσα είναι
        \[
          \Delta=A^2+12BC
        \]
        και οι ρίζες είναι
        \[
          y_1=\frac{A-\sqrt{A^2+12BC}}{2B},
          \qquad
          y_2=\frac{A+\sqrt{A^2+12BC}}{2B}.
        \]
        Επειδή $\sqrt{A^2+12BC}>A$, ισχύει $y_1<0$. Άρα δε γίνεται
        δεκτή. Η μοναδική θετική λύση είναι
        \[
          v^2=y_2
          =\frac{A+\sqrt{A^2+12BC}}{2B}.
        \]
        Επομένως,
        \[
          {
          v_0=\sqrt{\frac{A+\sqrt{A^2+12BC}}{2B}}}.
        \]

  \item Το τριώνυμο
        \[
          By^2-Ay-3C
        \]
        είναι αρνητικό για $0<y<y_2$ και θετικό για $y>y_2$.
        Επειδή $y=v^2$, προκύπτει ότι
        \[
        \begin{array}{c|ccc}
          v & (0,v_0) & v_0 & (v_0,+\infty)\\ \hline
          K'(v) & - & 0 & +\\
          K(v) & \searrow & \text{ελάχιστο} & \nearrow
        \end{array}
        \]
        Άρα η $K$ είναι γνησίως φθίνουσα στο $(0,v_0]$ και γνησίως
        αύξουσα στο $[v_0,+\infty)$. Συνεπώς, το συνολικό κόστος της
        πτήσης γίνεται ολικά ελάχιστο για
        \[
          {
          v=\sqrt{\frac{A+\sqrt{A^2+12BC}}{2B}}}.
        \]
\end{alist}
