% Λύση του 10ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 10 --- Δοκάρι από κυκλικό κορμό}

\begin{alist}
  \item Η διαγώνιος της ορθογωνικής διατομής είναι διάμετρος του
        κύκλου και επομένως έχει μήκος $40\;\si{cm}$. Από το
        Πυθαγόρειο θεώρημα,
        \[
          w^2+h^2=40^2.
        \]
        Άρα,
        \[
          {w^2+h^2=1600}.
        \]

  \item Επειδή $w,h>0$, από τον γεωμετρικό περιορισμό προκύπτει
        \[
          w=\sqrt{1600-h^2},
          \qquad 0<h<40.
        \]
        Επομένως,
        \[
          Z(h)=\frac{h^2\sqrt{1600-h^2}}{6},
          \qquad 0<h<40.
        \]

  \item Επειδή $Z(h)>0$, η μεγιστοποίηση της $Z$ είναι ισοδύναμη με
        τη μεγιστοποίηση του τετραγώνου της:
        \[
          Z^2(h)=\frac{1}{36}h^4(1600-h^2).
        \]
        Ο θετικός παράγοντας $\dfrac1{36}$ δεν επηρεάζει τη θέση του
        μεγίστου. Μελετούμε λοιπόν τη συνάρτηση
        \[
          f(h)=h^4(1600-h^2),
          \qquad 0<h<40.
        \]
        Παραγωγίζουμε:
        \begin{align*}
          f'(h)
          &=4h^3(1600-h^2)-2h^5\\
          &=2h^3(3200-3h^2).
        \end{align*}
        Επειδή $h>0$,
        \[
          f'(h)=0
          \Longleftrightarrow
          3200-3h^2=0
          \Longleftrightarrow
          h^2=\frac{3200}{3}.
        \]
        Άρα
        \[
          h=40\sqrt{\frac23}.
        \]

        Η $f'$ είναι θετική πριν από αυτή την τιμή και αρνητική μετά
        από αυτή. Συνεπώς, εκεί παρουσιάζεται ολικό μέγιστο. Από τον
        περιορισμό βρίσκουμε
        \begin{align*}
          w^2
          &=1600-\frac{3200}{3}
            =\frac{1600}{3},\\
          w&=\frac{40}{\sqrt3}.
        \end{align*}

        Οι ζητούμενες διαστάσεις είναι
        \[
          {
          w=\frac{40}{\sqrt3}\simeq23{,}1\;\si{cm}}
        \]
        και
        \[
          {
          h=40\sqrt{\frac23}\simeq32{,}7\;\si{cm}}.
        \]

  \item Για τις βέλτιστες διαστάσεις έχουμε
        \[
          \frac{h}{w}
          =
          \frac{40\sqrt{\frac23}}{\frac{40}{\sqrt3}}
          =\sqrt2.
        \]
        Άρα,
        \[
          {h=\sqrt2\,w}.
        \]
        Το μεγαλύτερο ύψος τοποθετεί περισσότερο υλικό μακριά από τον
        ουδέτερο άξονα και αυξάνει το μέτρο αντίστασης στην εξεταζόμενη
        διεύθυνση κάμψης.
\end{alist}
