% Λύση του 35ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 35 --- Βενζίνη και μεταβολή πλεονάσματος}

\begin{alist}
  \item Για το 2023 έχουμε $p=3{,}519$. Άρα
        \[
          D(3{,}519)
          =
          148{,}697-3{,}288\cdot3{,}519
          \approx137{,}126.
        \]
        Η πραγματική τιμή που χρησιμοποιήθηκε για το 2023 είναι περίπου
        $137{,}127$ δισεκατομμύρια γαλόνια, οπότε η μικρή διαφορά
        οφείλεται στη στρογγυλοποίηση των συντελεστών.

        Για το 2024 έχουμε $p=3{,}304$. Άρα
        \[
          D(3{,}304)
          =
          148{,}697-3{,}288\cdot3{,}304
          \approx137{,}834.
        \]
        Επομένως το γραμμικό μοντέλο αναπαράγει τα δύο πραγματικά
        σημεία που δόθηκαν.

  \item Η παράγωγος είναι
        \[
          D'(p)=-3{,}288<0.
        \]
        Άρα, στο διάστημα τιμών που εξετάζουμε, όταν αυξάνεται η τιμή
        της βενζίνης, η ζητούμενη ποσότητα μειώνεται.

        Η ελαστικότητα είναι
        \[
          E(p)=\frac{pD'(p)}{D(p)}.
        \]
        Για $p=3{,}519$ παίρνουμε
        \[
          E(3{,}519)
          =
          \frac{3{,}519\cdot(-3{,}288)}{137{,}127}
          \approx -0{,}084.
        \]
        Η αρνητική τιμή δείχνει ότι η ποσότητα κινείται αντίθετα από
        την τιμή. Επειδή η απόλυτη τιμή είναι μικρότερη από $1$, η
        ζήτηση στο συγκεκριμένο τοπικό μοντέλο είναι ανελαστική.

  \item Η αύξηση του πλεονάσματος των καταναλωτών είναι
        \[
          \Delta CS
          =
          \int_{3{,}304}^{3{,}519}
          \left(148{,}697-3{,}288p\right)\,dp.
        \]
        Υπολογίζουμε:
        \[
          \Delta CS
          =
          \left[
            148{,}697p-\frac{3{,}288}{2}p^2
          \right]_{3{,}304}^{3{,}519}.
        \]
        Επομένως
        \[
          \Delta CS\approx29{,}56.
        \]
        Επειδή η ποσότητα μετριέται σε δισεκατομμύρια γαλόνια και η
        τιμή σε δολάρια ανά γαλόνι, η μονάδα του αποτελέσματος είναι
        δισεκατομμύρια δολάρια ανά έτος.

        Άρα η μείωση της μέσης τιμής από το 2023 στο 2024 αντιστοιχεί,
        στο τοπικό αυτό μοντέλο, σε αύξηση πλεονάσματος περίπου
        \[
          {29{,}56\ \text{δισ. δολάρια ανά έτος}}.
        \]
\end{alist}
