% Λύση του 5ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 5 --- Αρμονικό μοντέλο θερμοκρασίας}

\begin{alist}
  \item Η μέση τιμή των οκτώ μετρήσεων είναι
        \[
          D=\frac{26+28+30+32+35+33+32+30}{8}
          =\frac{246}{8}
          =30{,}75.
        \]

        Επειδή οι γωνίες $\dfrac{\pi t_i}{12}$ είναι διαδοχικά
        \[
          0,\ \frac{\pi}{4},\ \frac{\pi}{2},\
          \frac{3\pi}{4},\ \pi,\ \frac{5\pi}{4},\
          \frac{3\pi}{2},\ \frac{7\pi}{4},
        \]
        βρίσκουμε
        \begin{align*}
          a
          &=\frac14\left(
             -9-\frac{7\sqrt2}{2}\right)
            =-\frac94-\frac{7\sqrt2}{8}
            \simeq-3{,}49,\\
          b
          &=\frac14\left(
             -2-\frac{3\sqrt2}{2}\right)
            =-\frac12-\frac{3\sqrt2}{8}
            \simeq-1{,}03.
        \end{align*}

        Άρα το αρμονικό μοντέλο είναι
        \[
          \boxed{
          M(t)=-3{,}49\syn\left(\frac{\pi t}{12}\right)
               -1{,}03\hm\left(\frac{\pi t}{12}\right)
               +30{,}75}.
        \]

  \item Θέτουμε $\omega=\dfrac{\pi}{12}$. Από την ταυτότητα
        \[
          A\syn\bigl(\omega(t-C)\bigr)
          =A\syn(\omega t)\syn(\omega C)
           +A\hm(\omega t)\hm(\omega C)
        \]
        πρέπει να ισχύουν
        \[
          a=A\syn(\omega C),
          \qquad
          b=A\hm(\omega C).
        \]
        Επομένως,
        \[
          A=\sqrt{a^2+b^2}
          \simeq3{,}64.
        \]
        Από τις δύο προηγούμενες σχέσεις, με $C\in[0,24)$, προκύπτει
        \[
          C\simeq13{,}10.
        \]
        Συνεπώς,
        \[
          \boxed{
          M(t)\simeq
          3{,}64\syn\left(\frac{\pi}{12}(t-13{,}10)\right)
          +30{,}75}.
        \]

  \item Παραγωγίζουμε:
        \[
          M'(t)
          =-\frac{3{,}64\pi}{12}
           \hm\left(\frac{\pi}{12}(t-13{,}10)\right).
        \]
        Στο $[0,24]$ η παράγωγος μηδενίζεται για
        \[
          t_1\simeq1{,}10
          \quad\text{και}\quad
          t_2\simeq13{,}10.
        \]
        Η $M$ είναι γνησίως φθίνουσα στο $[0,1{,}10]$, γνησίως
        αύξουσα στο $[1{,}10,13{,}10]$ και γνησίως φθίνουσα στο
        $[13{,}10,24]$.

        Η ελάχιστη και η μέγιστη τιμή του μοντέλου είναι
        \[
          M_{\min}=30{,}75-3{,}64
          \simeq27{,}11\degree C
        \]
        και
        \[
          M_{\max}=30{,}75+3{,}64
          \simeq34{,}39\degree C.
        \]
        Επειδή $t=0$ αντιστοιχεί στις $00{:}20$ UTC, το μοντέλο
        προβλέπει ελάχιστο περίπου στις $01{:}26$ UTC και μέγιστο περίπου
        στις $13{:}26$ UTC. Στην Ελλάδα, όπου στις 15 Ιουλίου ισχύει
        θερινή ώρα UTC$+3$, οι ώρες αυτές είναι περίπου $04{:}26$ και
        $16{:}26$ αντίστοιχα.

  \item Οι προβλέψεις του μοντέλου και τα απόλυτα σφάλματα είναι
        προσεγγιστικά
        \[
        \begin{array}{c|cccccccc}
          t&0&3&6&9&12&15&18&21\\ \hline
          M(t)&27{,}26&27{,}56&29{,}72&32{,}49&
                34{,}24&33{,}95&31{,}78&29{,}01\\
          |\theta-M(t)|
              &1{,}26&0{,}44&0{,}28&0{,}49&
               0{,}76&0{,}95&0{,}22&0{,}99
        \end{array}
        \]
        σε βαθμούς Κελσίου. Άρα,
        \[
          \boxed{\max|\theta_i-M(t_i)|\simeq1{,}26\degree C}.
        \]

        Τα σφάλματα δεν είναι μηδενικά, επειδή το μοντέλο είναι
        προσαρμογή ενός μόνο αρμονικού σε οκτώ πραγματικές μετρήσεις και
        όχι παρεμβολή που εξαναγκάζεται να διέρχεται από όλα τα σημεία.
        Η πραγματική θερμοκρασία επηρεάζεται επίσης από νέφωση, άνεμο,
        υγρασία και μεταφορά αερίων μαζών, παράγοντες που δεν
        περιλαμβάνονται στη συνάρτηση.
\end{alist}
