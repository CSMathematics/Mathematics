\subsection*{Λύση Προβλήματος 32 --- Καμπύλη \en{Lorenz} και δείκτης \en{Gini}}

\begin{alist}
  \item Για
        \[
          f(x)=\frac{e^{2{,}15x}-1}{e^{2{,}15}-1}
        \]
        έχουμε
        \[
          f''(x)=
          \frac{(2{,}15)^2e^{2{,}15x}}{e^{2{,}15}-1}>0.
        \]
        Άρα η $f$ είναι κυρτή στο $(0,1)$.

        Επειδή $f(0)=0$, $f(1)=1$ και η $f$ είναι κυρτή, το γράφημά της
        βρίσκεται κάτω από τη χορδή που ενώνει τα σημεία $(0,0)$ και
        $(1,1)$. Η χορδή αυτή είναι η ευθεία $y=x$. Επομένως,
        \[
          f(x)\leq x,\qquad 0\leq x\leq1.
        \]

  \item Ο δείκτης \en{Gini} του μοντέλου είναι
        \[
          G=2\int_0^1\bigl(x-f(x)\bigr)\,dx.
        \]
        Υπολογίζουμε πρώτα
        \begin{align*}
          \int_0^1 f(x)\,dx
          &=
          \frac{1}{e^{2{,}15}-1}
          \int_0^1\left(e^{2{,}15x}-1\right)\,dx\\
          &=
          \frac{1}{e^{2{,}15}-1}
          \left[
            \frac{e^{2{,}15x}}{2{,}15}-x
          \right]_0^1\\
          &=
          \frac{1}{2{,}15}
          -\frac{1}{e^{2{,}15}-1}.
        \end{align*}
        Άρα
        \begin{align*}
          G
          &=2\left[
            \frac12-
            \left(
              \frac{1}{2{,}15}
              -\frac{1}{e^{2{,}15}-1}
            \right)
          \right]\\
          &=
          1-\frac{2}{2{,}15}
          +\frac{2}{e^{2{,}15}-1}\\
          &\approx0{,}333.
        \end{align*}
        Η τιμή της Παγκόσμιας Τράπεζας για την Ελλάδα το 2023 είναι
        $33{,}4$ μονάδες, δηλαδή $0{,}334$ στην κλίμακα $0$ έως $1$.
        Επομένως, το εκθετικό μοντέλο δίνει δείκτη πολύ κοντά στην
        πραγματική τιμή:
        \[
          0{,}333\approx0{,}334.
        \]
\end{alist}
