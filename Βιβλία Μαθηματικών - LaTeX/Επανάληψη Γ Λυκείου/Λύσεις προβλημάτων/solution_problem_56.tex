% Λύση του 56ου προβλήματος της ενότητας «Άλυτα Προβλήματα».
\subsection*{Λύση Προβλήματος 56 --- Νόμος Poiseuille}

\begin{alist}
  \item Από τον τύπο
        \[
          \Phi(t)=c\,\rho(t)^4,
        \]
        με \(c>0\), παραγωγίζουμε ως προς \(t\). Με τον κανόνα
        σύνθετης συνάρτησης παίρνουμε
        \[
          \Phi'(t)
          =
          c\cdot 4\rho(t)^3\rho'(t).
        \]
        Άρα
        \[
          {
          \Phi'(t)=4c\rho(t)^3\rho'(t)
          }.
        \]

  \item Επειδή
        \[
          \Phi(t)=c\rho(t)^4,
        \]
        διαιρούμε την προηγούμενη σχέση με τη \(\Phi(t)\):
        \[
          \frac{\Phi'(t)}{\Phi(t)}
          =
          \frac{4c\rho(t)^3\rho'(t)}{c\rho(t)^4}.
        \]
        Απλοποιώντας, καταλήγουμε στο
        \[
          {
          \frac{\Phi'(t)}{\Phi(t)}
          =
          4\frac{\rho'(t)}{\rho(t)}
          }.
        \]
        Δηλαδή ο σχετικός ρυθμός μεταβολής της παροχής είναι τετραπλάσιος
        από τον σχετικό ρυθμό μεταβολής της ακτίνας.

  \item Δίνεται ότι
        \[
          \frac{\rho'(t)}{\rho(t)}=-0{,}01
          \quad \text{ανά λεπτό}.
        \]
        Άρα, από το προηγούμενο ερώτημα,
        \[
          \frac{\Phi'(t)}{\Phi(t)}
          =
          4(-0{,}01)
          =
          -0{,}04
          \quad \text{ανά λεπτό}.
        \]
        Επομένως η παροχή μειώνεται με σχετικό ρυθμό
        \[
          {4\%\ \text{ανά λεπτό}}.
        \]

  \item Αν η νέα ακτίνα είναι
        \[
          \rho_1=0{,}95\rho_0,
        \]
        τότε, επειδή \(\Phi=c\rho^4\), έχουμε
        \[
          \frac{\Phi_1}{\Phi_0}
          =
          \frac{c\rho_1^4}{c\rho_0^4}
          =
          \left(\frac{\rho_1}{\rho_0}\right)^4
          =
          (0{,}95)^4.
        \]
        Υπολογίζουμε:
        \[
          (0{,}95)^4=0{,}81450625.
        \]
        Άρα απομένει περίπου
        \[
          {81{,}45\%}
        \]
        της αρχικής παροχής. Η μείωση είναι περίπου
        \[
          100\%-81{,}45\%=18{,}55\%.
        \]
\end{alist}
