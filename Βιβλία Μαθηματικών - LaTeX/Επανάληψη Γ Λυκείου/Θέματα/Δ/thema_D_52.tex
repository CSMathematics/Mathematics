\begin{thema}{Δ}
Δίνεται η πολυωνυμική συνάρτηση $\displaystyle P(x) = x^3 - 3x^2 - 9x + \mu, \; x \in \mathbb{R}$, όπου $\mu \in \mathbb{R}$ σταθερά.
\begin{erwthma}
    \item Να αποδείξετε ότι για κάθε $\mu \in \mathbb{R}$ η συνάρτηση $P$ παρουσιάζει ακριβώς ένα τοπικό μέγιστο, ένα τοπικό ελάχιστο και ένα σημείο καμπής.
    \item Να βρείτε την τιμή του $\mu$ ώστε το σημείο καμπής της γραφικής παράστασης της $P$ να βρίσκεται πάνω στον άξονα $x'x$.
\end{erwthma}
Στα παρακάτω ερωτήματα, θεωρήστε ότι $\mu = 11$:
\begin{erwthma}[resume]
    \item \begin{alist}
        \item Να αποδείξετε ότι για κάθε $x \in \mathbb{R}$ ισχύει $P(2-x) = -P(x)$ και να ερμηνεύσετε γεωμετρικά το αποτέλεσμα ως προς τη γραφική παράσταση της $P$.
        \item Να βρείτε το πλήθος των πραγματικών ριζών της εξίσωσης $P(x) = 0$.
    \end{alist}
    \item Να υπολογίσετε το ολοκλήρωμα:
    \[ \dint_0^2 \left(P(x)\right)^{2025} dx \]
\end{erwthma}
\end{thema}
