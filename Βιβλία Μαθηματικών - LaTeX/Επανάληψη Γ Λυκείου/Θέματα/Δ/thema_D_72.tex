\begin{thema}{Δ}
Στο σχήμα δίνεται η $C_{f'}$ μιας $f:[-2, 5]\to\mathbb{R}$ με $f(-2)=0$. Η $f'$ είναι μια λεία καμπύλη: $f'(-2)=3$, $f'(0)=0$, $f'(2)=-1$, $f'(4)=0$, $f'(5)=1$. Η $f'$ αλλάζει πρόσημο στα $x=0$ και $x=4$. Επιπλέον δίνεται ότι
\[
\int_{-2}^{4} f'(x)\d x>0.
\]

\begin{erwthma}
  \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία και τα τοπικά ακρότατα.
  \item Να αποδείξετε ότι $f(0) > 0$ και $f(4) < f(0)$.
  \item Να λύσετε την ανίσωση $f(x) \geq 0$ στο διάστημα $[-2, 5]$.
  \item Να αποδείξετε ότι η εξίσωση $f(x) = f(4)$ έχει ακριβώς δύο ρίζες στο διάστημα $[-2, 5]$.
\end{erwthma}
\end{thema}
