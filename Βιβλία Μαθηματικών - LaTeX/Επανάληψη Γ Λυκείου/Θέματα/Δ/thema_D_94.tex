\begin{thema}{Δ}
Δίνεται $f:(0,+\infty)\to\mathbb{R}$ με $f(x)=x-\ln x$.
\begin{erwthma}
    \item Μελετήστε μονοτονία, ακρότατα, κυρτότητα. \monades{6}
    \item Αποδείξτε $\ln x \leq x - 1$ $\forall x > 0$ (αξιοποιήστε το ελάχιστο). \monades{6}
    \item Θέτοντας $x = \frac{a}{b}$ ($a,b>0$), δείξτε $b\ln a - b\ln b \leq a - b$. \monades{6}
    \item Αποδείξτε $\dint_1^e (x-\ln x)dx > e - 1$. \monades{7}
\end{erwthma}
\end{thema}
