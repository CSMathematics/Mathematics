\begin{thema}{Δ}
Δίνεται η συνάρτηση $f(x) = \begin{cases} \frac{x \ln x}{x-1}, & 0 < x \ne 1 \\ 1, & x = 1 \end{cases}$
\begin{erwthma}
    \item Να αποδείξετε ότι η συνάρτηση $f$ είναι συνεχής στο σημείο $x_0 = 1$.
    \item Να αποδείξετε ότι η συνάρτηση $f$ είναι παραγωγίσιμη στο $x_0 = 1$ και να βρείτε την $f'(1)$.
    \item Να υπολογίσετε το $\displaystyle\lim_{x \to 0^+} f(x)$ και να αποδείξετε ότι η $f$ επεκτείνεται συνεχώς στο $x=0$ (δεν υπάρχει κατακόρυφη ασύμπτωτη).
    \item Να δείξετε ότι $f'(x) > 0$ για $x > 0$, $x \ne 1$, επομένως η $f$ είναι γνησίως αύξουσα.
\end{erwthma}
\end{thema}
