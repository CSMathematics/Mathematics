\begin{thema}{Δ}
Δίνεται η παραγωγίσιμη συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ για την οποία ισχύουν:
\begin{itemize}
    \item $e^x \le f(x) \le (x^2+1)e^x$ για κάθε $x \in \mathbb{R}$.
    \item $f'(x) - f(x) = 2x e^x$ για κάθε $x \in \mathbb{R}$.
\end{itemize}

\begin{erwthma}
    \item Να αποδείξετε ότι $f(0) = 1$, $f'(0) = 1$ και, στη συνέχεια, να αποδείξετε ότι $f(x) = (x^2 + 1)e^x$.
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία και να λύσετε στο $\mathbb{R}$ την ανίσωση: 
    \[ f\left( f(x^2 - 2) - 1 \right) > 1 \]
    \item Να αποδείξετε ότι για κάθε $x > 0$ ισχύει: 
    \[ f(x+2) + f(x) > 2f(x+1) \]
    \item Να υπολογίσετε το ολοκλήρωμα $\displaystyle I = \dint_0^1 f(x) dx$.
\end{erwthma}
\end{thema}
