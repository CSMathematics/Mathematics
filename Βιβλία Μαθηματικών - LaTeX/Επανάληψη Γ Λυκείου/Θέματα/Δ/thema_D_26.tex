\begin{thema}{Δ}
Δίνεται η συνάρτηση $f(x) = (x-1)(e^x - e)$ με $x \in \mathbb{R}$.
\begin{erwthma}
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία, τα ακρότατα και την κυρτότητα.
    \item Να αποδείξετε ότι η εξίσωση $f(x) = 2e$ έχει ακριβώς δύο ρίζες $x_1, x_2$ για τις οποίες ισχύει $x_1 < -1 < 1 < x_2$.
    \item Να αποδείξετε ότι υπάρχει τουλάχιστον ένα $\xi \in (x_1, x_2)$ τέτοιο ώστε να ισχύει: 
    \[ f'(\xi) + f(\xi) = 2e \]
    \item Να βρείτε την εξίσωση της τέμνουσας ευθείας $(\varepsilon)$ που διέρχεται από τα σημεία $A(0, f(0))$ και $B(1, f(1))$. Στη συνέχεια, αξιοποιώντας την κυρτότητα της $f$ στο διάστημα $[0, 1]$, να αποδείξετε ότι:
    \[ \dint_0^1 f(x) dx < \frac{e-1}{2} \]
\end{erwthma}
\end{thema}
