\begin{thema}{Δ}
Στο σχήμα δίνεται η $C_{f'}$ μιας δύο φορές παραγωγίσιμης $f:[0, 8]\to\mathbb{R}$. Η $f'$ είναι γνησίως φθίνουσα στο $[0, 4]$ (από $f'(0)=4$ στο $f'(4)=0$) και γνησίως αύξουσα στο $[4, 8]$ (από $f'(4)=0$ στο $f'(8)=2$). Δίνεται $f(0) = 0$.

\end{center}

\begin{erwthma}
  \item Να λύσετε την ανίσωση $f(x) \geq 0$ στο διάστημα $[0, 8]$.
  \item Να βρείτε τα διαστήματα κυρτότητας της $f$ και το σημείο καμπής της $C_f$.
  \item Να αποδείξετε ότι η εξίσωση $f'(x) = 2$ έχει ακριβώς δύο ρίζες στο διάστημα $[0, 8]$.
  \item Να αποδείξετε ότι $f(x) \leq f(4)$ για κάθε $x \in [0, 4]$ και $f(x) \geq f(4)$ για κάθε $x \in [4, 8]$.
\end{erwthma}
\end{thema}
