\begin{thema}{Δ}
Δίνονται οι συναρτήσεις $\displaystyle f(x) = e^x - 1, \; x \in \mathbb{R}$ και $\displaystyle g(x) = \ln(x+1), \; x > -1$.
\begin{erwthma}
    \item Να αποδείξετε ότι για κάθε $x > -1$ ισχύει $g(x) \le x \le f(x)$, και ότι οι γραφικές παραστάσεις των $f$ και $g$ έχουν ακριβώς ένα κοινό σημείο, στο οποίο δέχονται κοινή εφαπτομένη.
    \item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τις $C_f, C_g$ και την ευθεία $x = 1$.
    \item Θεωρούμε τη συνάρτηση:
    \[ h(x) = \begin{cases} 
        \displaystyle \frac{f(x)}{g(x)}, & x \in (-1, 0) \cup (0, +\infty) \\[2ex]
        1, & x = 0 
    \end{cases} \]
    Να αποδείξετε ότι η $h$ είναι συνεχής στο $x_0 = 0$ και στη συνέχεια ότι είναι παραγωγίσιμη στο σημείο αυτό, με $h'(0) = 1$.
    \item Αξιοποιώντας τον ορισμό της παραγώγου από το ερώτημα Δ3, να υπολογίσετε το όριο:
    \[ \lim_{x \to 0} \left( \frac{1}{g(x)} - \frac{1}{f(x)} \right) \]
\end{erwthma}
\end{thema}
