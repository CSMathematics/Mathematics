\begin{thema}{Δ}
Στο σχήμα δίνεται η γραφική παράσταση $C_f$ μιας \textbf{γνησίως αύξουσας} και συνεχούς $f:[0, 5]\to[0, 4]$ με $f(0) = 0$ και $f(5) = 4$. Το σκιασμένο εμβαδόν κάτω από τη $C_f$ (μεταξύ $C_f$, $x'x$, $x=0$ και $x=5$) είναι ίσο με $E_f = 8$.

\begin{erwthma}
  \item Η $f$ αντιστρέφεται; Ποιο το πεδίο ορισμού της $f^{-1}$;
  \item Χρησιμοποιώντας τη σχέση $\dint_0^5 f(x)\,dx + \dint_0^4 f^{-1}(y)\,dy = 5 \cdot 4$, να υπολογίσετε το $\dint_0^4 f^{-1}(y)\,dy$.
  \item Εφαρμόστε Θ.Μ.Τ. στη $f$ στο $[0, 5]$: υπάρχει $\xi$ ώστε $f'(\xi) = \frac{4}{5}$. Αν $f'(\xi_0)$ είναι η αντίστοιχη κλίση, να δείξετε ότι $(f^{-1})'(f(\xi_0)) = \frac{5}{4}$.
  \item Αν η $f$ είναι και κυρτή (κοίλη άνω) στο $[0, 5]$, αποδείξτε ότι $\dint_0^5 f(x)\,dx < 10$ και $\dint_0^4 f^{-1}(y)\,dy > 10$.
\end{erwthma}
\end{thema}
