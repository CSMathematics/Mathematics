\begin{thema}{Δ}
Δίνεται η $f(x)=e^x-x-1$, $x\in\mathbb{R}$.
\begin{erwthma}
    \item Η $f$ έχει ολικό ελάχιστο. Βρείτε το και δείξτε $e^x \geq x+1$ $\forall x$. \monades{6}
    \item Αν $g(x) = \frac{e^x-1}{x}$ για $x\neq 0$ και $g(0)=1$, δείξτε ότι η $g$ είναι συνεχής στο $0$. \monades{6}
    \item Εφαρμόστε ΘΜΤ στη $f$ στο $[0,1]$: δείξτε ότι $e^\xi = e-1$ για κάποιο $\xi\in(0,1)$. \monades{6}
    \item Αποδείξτε ότι $\dint_0^1 e^x dx > 1 + \dint_0^1 x\,dx = \frac{3}{2}$. \monades{7}
\end{erwthma}
\end{thema}
