\begin{thema}{Δ}
Στο σχήμα δίνεται η $C_f$ μιας συνεχούς στο $[0,5]$ και παραγωγίσιμης στο $(0,5)$ συνάρτησης $f:[0, 5]\to\mathbb{R}$ μαζί με τη χορδή $AB$ ($A(0,1)$, $B(5,6)$). Η $f$ είναι αυστηρά κοίλη στο $[0,5]$ και η $f'$ είναι γνησίως φθίνουσα στο $(0,5)$.

\end{center}

\begin{erwthma}
  \item Να αποδείξετε ότι $f(x) \geq x + 1$ για κάθε $x \in [0, 5]$.
  \item Να λύσετε την εξίσωση $f(x) = x + 1$ στο διάστημα $[0, 5]$.
  \item Να αποδείξετε ότι η εξίσωση $f'(x) = 1$ έχει μοναδική ρίζα στο διάστημα $(0, 5)$.
  \item Να αποδείξετε ότι $\dint_0^5 f(x)\,dx > \frac{35}{2}$.
\end{erwthma}
\end{thema}
