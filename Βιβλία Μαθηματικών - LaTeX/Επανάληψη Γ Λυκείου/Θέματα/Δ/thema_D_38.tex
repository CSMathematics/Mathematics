\begin{thema}{Δ}
Δίνεται η συνάρτηση $\displaystyle f(x) = \begin{cases} x^2 \hm\left(\frac{1}{x}\right), & x \ne 0 \\ 0, & x = 0 \end{cases}$.
\begin{erwthma}
    \item Να αποδείξετε ότι η $f$ είναι συνεχής και παραγωγίσιμη στο $x_0 = 0$ με $f'(0) = 0$.
    \item Να βρείτε την παράγωγο συνάρτηση $f'(x)$ για κάθε $x \in \mathbb{R}$.
    \item Να αποδείξετε ότι το όριο $\displaystyle \lim_{x\to 0} \cos\left(\frac{1}{x}\right)$ δεν υπάρχει, και να συμπεράνετε ότι η $f'$ δεν είναι συνεχής στο $x_0 = 0$.
    \item Να υπολογίσετε την τιμή του ολοκληρώματος $\displaystyle \dint_{1/\pi}^{2/\pi} f'(x) dx$.
\end{erwthma}
\end{thema}
