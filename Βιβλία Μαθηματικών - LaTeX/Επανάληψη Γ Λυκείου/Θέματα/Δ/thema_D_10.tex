\begin{thema}{Δ}
Δίνεται η συνάρτηση $f(x) = \frac{\ln x}{x}$, με $x > 0$.
\begin{erwthma}
    \item Να μελετήσετε τη μονοτονία της $f$, να βρείτε το ολικό μέγιστο και τις ασύμπτωτες της $C_f$.
    \item Με χρήση του παραπάνω ακροτάτου, να αποδείξετε ότι $x^e \le e^x$ για κάθε $x > 0$ και να συγκρίνετε τους αριθμούς $\pi^e$ και $e^\pi$.
    \item Αν $a > \beta \ge e$, να αποδείξετε ότι $\beta^a > a^\beta$.
    \item Να υπολογίσετε το όριο: $\displaystyle \lim_{x \to 0^+} \left( f(x) \cdot \hm x \right)$.
\end{erwthma}
\end{thema}
