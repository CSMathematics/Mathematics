\begin{thema}{Δ}
Δίνονται οι συναρτήσεις $f,g:\mathbb{R}\to\mathbb{R}$ για τις οποίες ισχύουν
\begin{itemize}
\item $g(0)=0$ και $g(1)=\dfrac{\pi}{4}$
\item $f'(x)=g(x)-\dfrac{x}{x^2+1}$ και $g'(x)=\dfrac{1}{x^2+1}$, για κάθε $x\in\mathbb{R}$
\end{itemize}
\begin{erwthma}
    \item Να αποδείξετε ότι η $f$ είναι κυρτή.
\item Να αποδείξετε ότι \[f(x)-f(\hm{x})<\left(x-\hm{x}\right)f'(x)\]
για κάθε $x>0$.
\item Να λύσετε την ανίσωση \[f\left(x^2+1\right)-f(2)<f\left(\frac{1}{x^2+1}\right)-f\left(\frac{1}{2}\right)\]
\item Αν επιπλέον ισχύει $\dint_{0}^{1}{xf(x)\d x}>\frac{\pi}{4}$, τότε να αποδείξετε ότι υπάρχει τουλάχιστον ένα $x_0\in(0,1)$ τέτοιο ώστε
\[x_0^3f(x_0)+x_0f(x_0)=1\]
\end{erwthma}
\end{thema}
