\begin{thema}{Δ}
Θεωρούμε την παραγωγίσιμη συνάρτηση $f: (0, +\infty) \to \mathbb{R}$ για την οποία ικανοποιείται η σχέση:
\[ f(x) + e^{f(x)} = e^{x-1} - \ln x + \frac{e^{e^{x-1}}}{x} \quad \text{για κάθε } x > 0 \]
\begin{erwthma}
    \item Να αποδείξετε ότι ο τύπος της συνάρτησης είναι $f(x) = e^{x-1} - \ln x$. \monades{6}
    \item Να αποδείξετε ότι για οποιαδήποτε $\alpha, \beta \in \mathbb{R}$ με $1 < \alpha < \beta$, ισχύει η ανισότητα $\frac{f(\beta) - f(\alpha)}{\beta - \alpha} > e^{\alpha-1} - \frac{1}{\alpha}$. \monades{6}
    \item Να αποδείξετε ότι υπάρχει ακριβώς ένα $x_0 \in (1, 2)$ τέτοιο ώστε η εφαπτομένη της γραφικής παράστασης $C_f$ στο σημείο $A(x_0, f(x_0))$ να διέρχεται από την αρχή των αξόνων $O(0,0)$. \monades{6}
    \item Να αποδείξετε την ανισότητα $\dint_1^e f(x) dx < e^{e-1} - 1$. \monades{7}
\end{erwthma}
\end{thema}
