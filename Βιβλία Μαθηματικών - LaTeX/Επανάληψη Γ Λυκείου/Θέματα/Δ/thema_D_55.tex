\begin{thema}{Δ}
Δίνεται η συνάρτηση $g(x) = e^x + e^{-x}, \; x \ge 0$.
\begin{erwthma}
    \item Να μελετήσετε την $g$ ως προς τη μονοτονία. Στη συνέχεια, να αποδείξετε ότι για κάθε $x \ge 1$, η εξίσωση $g(y) = 2x$ έχει ακριβώς μία λύση $y \in [0, +\infty)$.
    \item Θεωρούμε τη συνάρτηση $f: [1, +\infty) \to \mathbb{R}$ με $f(x) \ge 0$, η οποία για κάθε $x \ge 1$ ικανοποιεί τη σχέση:
    \[ e^{f(x)} + e^{-f(x)} = 2x \]
    Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα.
    \item Να αποδείξετε ότι ο τύπος της $f$ είναι $\displaystyle f(x) = \ln\left(x + \sqrt{x^2-1}\right)$ για κάθε $x \ge 1$ και να ορίσετε την αντίστροφη συνάρτηση $f^{-1}$.
    \item Να υπολογίσετε το ολοκλήρωμα:
    \[ I = \dint_1^{\frac{e^2+1}{2e}} f(x) dx \]
    αξιοποιώντας την αντίστροφη συνάρτηση $f^{-1}$.
\end{erwthma}
\end{thema}
