\begin{thema}{Δ}
Στο σχήμα δίνεται η γραφική παράσταση $C_{f'}$ της παραγώγου μιας δύο φορές παραγωγίσιμης συνάρτησης $f:[-2, 6]\to\mathbb{R}$. Η $f'$ μηδενίζεται στα $x = 0, 2, 5$ και αλλάζει πρόσημο σε κάθε ρίζα. Δίνονται επιπλέον οι τιμές: $f(-2) = 1$, $f(0) = 3$, $f(2) = -1$, $f(5) = 4$ και $f(6) = 2$.

\begin{center}

    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-3,-2,-1,1,2,3,4}

  \filldraw[secondcolor] (5,0) circle (2pt) node[above right,font=\footnotesize] {$5$};
  \node[secondcolor] at (6,2) {$C_{f'}$};
\end{tikzpicture}
\end{center}

\begin{erwthma}
  \item Να κατασκευάσετε τον πίνακα μεταβολών της $f$ και να βρείτε το ολικό μέγιστο καθώς και το ολικό ελάχιστο στο $[-2, 6]$.
  \item Να αποδείξετε ότι η εξίσωση $f(x) = 1$ έχει ακριβώς δύο ρίζες $\rho_1, \rho_2$ στο διάστημα $(0, 6)$.
  \item Εφαρμόζοντας το Θεώρημα \eng{Rolle} στα διαστήματα $[-2, \rho_1]$ και $[\rho_1, \rho_2]$, να δείξετε ότι η εξίσωση $f'(x) = 0$ έχει τουλάχιστον δύο ρίζες. Πώς επιβεβαιώνεται αυτό από το δοσμένο σχήμα;
  \item Με χρήση του Θ.Μ.Τ. στο διάστημα $[2, 5]$, να αποδείξετε ότι υπάρχει τουλάχιστον ένα $\xi \in (0, 5)$ τέτοιο ώστε $f'(\xi) = 1$.
\end{erwthma}
\end{thema}
