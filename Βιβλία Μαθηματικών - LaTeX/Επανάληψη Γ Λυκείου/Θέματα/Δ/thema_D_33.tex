\begin{thema}{Δ}
Δίνεται η συνάρτηση $f(x) = e^x + x - 1$, με $x \in \mathbb{R}$.
\begin{erwthma}
    \item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα, να βρείτε τη μοναδική ρίζα της εξίσωσης $f(x) = 0$ και να λύσετε την ανίσωση $f(f(\ln x) + 1) > e$.
    \item Να υπολογίσετε τα όρια $\displaystyle\lim_{x\to 0} \frac{f(x) - 2x}{x^2}$ και $\displaystyle\lim_{x\to +\infty} \frac{x f(x) - e^x}{f(x) + e^{x-1}}$.
    \item Αφού αιτιολογήσετε ότι η $f$ αντιστρέφεται στο $\mathbb{R}$, να λύσετε την εξίσωση $f(x) = f^{-1}(x)$ και, θέτοντας $x = f(u)$, να υπολογίσετε το ολοκλήρωμα $\displaystyle \dint_0^e f^{-1}(x) dx$.
    \item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τις γραφικές παραστάσεις των συναρτήσεων $f, f^{-1}$ και την ευθεία $x+y=e+1$.
\end{erwthma}
\end{thema}
