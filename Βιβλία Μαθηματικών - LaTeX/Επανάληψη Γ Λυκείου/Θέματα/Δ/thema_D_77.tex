\begin{thema}{Δ}
Στο σχήμα δίνεται η $C_f$ μιας συνεχούς στο $[0,5]$ και παραγωγίσιμης στο $(0,5)$ γνησίως αύξουσας $f:[0, 5]\to\mathbb{R}$ με $f(0)=-2$, $f(2) = 0$, $f(3) = 1$, $f(5) = 4$. Η $f$ είναι κυρτή στο $[0,5]$, η $f'$ είναι γνησίως αύξουσα στο $(0,5)$ και υπάρχει η αριστερή παράγωγος $f'_{-}(5)$.

\begin{erwthma}
  \item Να λύσετε την ανίσωση $f(x) > 0$ στο διάστημα $[0, 5]$.
  \item Να αποδείξετε ότι $f(x) \leq \frac{6}{5}x - 2$ για κάθε $x \in [0, 5]$.
  \item Να αποδείξετε ότι υπάρχει $\xi_1 \in (0, 5)$ τέτοιο ώστε $f'(\xi_1) = \frac{6}{5}$.
  \item Να αποδείξετε ότι $f'(2) < \frac{4}{3} < f'_{-}(5)$.
\end{erwthma}
\end{thema}
