\begin{thema}{Δ}
Δίνεται η παραγωγίσιμη $f:\mathbb{R}\to\mathbb{R}$ με $f(0)=0$, $f(2)=2$ και $f'(x) > 0$ $\forall x\in\mathbb{R}$.
\begin{erwthma}
    \item Δείξτε ότι η $f$ αντιστρέφεται. \monades{6}
    \item Εφαρμόστε ΘΜΤ στο $[0,2]$ και δείξτε ότι $f'(\xi) = 1$ για κάποιο $\xi$. \monades{6}
    \item Αν $g(x) = f(x) - x$, δείξτε ότι $g(0)=g(2)=0$ και εφαρμόστε Rolle. Τι σημαίνει $f'(\xi)=1$; \monades{6}
    \item Αποδείξτε ότι $\dint_0^2 f(x)\,dx + \dint_0^2 f^{-1}(y)\,dy = 4$. \monades{7}
\end{erwthma}
\end{thema}
