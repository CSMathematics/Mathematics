\begin{thema}{Δ}
Θεωρούμε την παραγωγίσιμη συνάρτηση $f: (0, +\infty) \to (0, +\infty)$ η οποία επαληθεύει τη συναρτησιακή εκθετική εξίσωση:
\[ f(x) e^{f(x)} = e^{x-1} e^{e^{x-1}} \quad \text{για κάθε } x > 0 \]
\begin{erwthma}
    \item Να αποδείξετε ότι ο τύπος της συνάρτησης προσδιορίζεται μονοσήμαντα και είναι ο $f(x) = e^{x-1}$. \monades{6}
    \item Να δείξετε ότι η $f(x)$ είναι γνησίως αύξουσα και αυστηρά κυρτή. Έπειτα, χρησιμοποιώντας την εφαπτομένη της στο σημείο με τετμημένη $1$, να δείξετε ότι $\dint_0^1 f(x) dx > \frac{1}{2}$. \monades{6}
    \item Αξιοποιώντας την κυρτότητα της $f$, να αποδείξετε ότι $\dint_0^2 f(x) dx > 2$ και στη συνέχεια να υπολογίσετε την ακριβή τιμή του ολοκληρώματος $\dint_0^2 f(x) dx$. \monades{6}
    \item Να λύσετε στους πραγματικούς αριθμούς την ανίσωση $f(x^2 + 1) > f(x + 1)$. \monades{7}
\end{erwthma}
\end{thema}
