\begin{thema}{Δ}
Δίνεται η συνάρτηση $f(x) = x\ln x - x + 1$, με $x > 0$.
\begin{erwthma}
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία και τα ακρότατα, και να βρείτε το σύνολο τιμών της.
    \item Να αποδείξετε ότι η γραφική παράσταση της $f$ δέχεται ακριβώς μία εφαπτομένη η οποία διέρχεται από το σημείο $A(0, -1)$, και να βρείτε την εξίσωσή της.
    \item Να αποδείξετε ότι $x\ln x \ge x - 1$ για κάθε $x > 0$, και στη συνέχεια να λύσετε στο $(0, +\infty)$ την εξίσωση $f(x^2) + f(x^3) = 0$.
    \item Έστω $F$ μία παράγουσα (αρχική) της $f$ στο $(0, +\infty)$. Να υπολογίσετε το όριο:
    \[ \lim_{x\to 1} \frac{F(x) - F(1)}{(x-1)^3} \]
\end{erwthma}
\end{thema}
