\begin{thema}{Δ}
Δίνεται $f:\mathbb{R}\to\mathbb{R}$ παραγωγίσιμη με $f(0)=1$ και $f'(x) \geq f(x)$ $\forall x\geq 0$.
\begin{erwthma}
    \item Θεωρήστε $g(x) = f(x)e^{-x}$. Δείξτε $g'(x)\geq 0$ $\forall x\geq 0$. \monades{6}
    \item Δείξτε ότι $f(x)\geq e^x$ $\forall x\geq 0$. \monades{6}
    \item Αποδείξτε ότι $\dint_0^1 f(x)dx \geq e-1$. \monades{6}
    \item Αν επιπλέον $f(1)=e$, δείξτε ότι $f(x)=e^x$ $\forall x\in[0,1]$. \monades{7}
\end{erwthma}
\end{thema}
