\begin{thema}{Δ}
Δίνεται η $f:\mathbb{R}\to\mathbb{R}$ παραγωγίσιμη με $f(0)=0$ και $f'(x) + f(x) > 0$ $\forall x$.
\begin{erwthma}
    \item Θεωρήστε $h(x)=e^x f(x)$. Δείξτε $h'(x) > 0$, δηλαδή η $h$ γνησίως αύξουσα. \monades{6}
    \item Δείξτε $f(x)>0$ $\forall x>0$ και $f(x)<0$ $\forall x<0$. \monades{6}
    \item Αποδείξτε ότι η εξίσωση $f(x)=0$ έχει μοναδική ρίζα στο $\mathbb{R}$. \monades{6}
    \item Αν $f(1)=\frac{1}{e}$, δείξτε $\dint_0^1 f(x)\,dx > 0$ και εκτιμήστε ότι $\dint_0^1 f(x)dx < \frac{1}{e}$. \monades{7}
\end{erwthma}
\end{thema}
