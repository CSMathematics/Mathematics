\begin{thema}{Δ}
Στα σχήματα δίνονται \textbf{χωριστά} η $C_f$ (μπλε) και η $C_{f'}$ (κόκκινη) μιας συνάρτησης $f:[-1, 5]\to\mathbb{R}$ με συνεχή παράγωγο. Από τη γραφική: $f(-1) = 2$, $f(1) = 4$ (μέγιστο), $f(3) = 0$ (ρίζα), $f(5) = -2$.

\begin{center}
\begin{tikzpicture}[scale=0.6]
  \draw[help lines, gray!30] (-2,-3) grid (6,5);
  \draw[-latex] (-2.3,0) -- (6.3,0) node[right] {$x$};
  \draw[-latex] (0,-3.3) -- (0,5.3) node[above] {$y$};
  \foreach \x in {-1,1,2,3,4,5}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-2,-1,1,2,3,4}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  \draw[very thick, maincolor, smooth, tension=0.5]
    plot coordinates {(-1,2) (0,3.5) (1,4) (2,2.5) (3,0) (4,-1.5) (5,-2)};
  \node[maincolor] at (-1,3.5) {$C_f$};
  \draw[very thick, secondcolor, smooth, tension=0.5]
    plot coordinates {(-1,2) (0,1) (1,0) (2,-1.5) (3,-1.5) (4,-1) (5,-0.3)};
  \node[secondcolor] at (4.5,-2.5) {$C_{f'}$};
  \filldraw[maincolor] (1,4) circle (2pt) node[above,font=\footnotesize] {$(1,4)$};
  \filldraw[maincolor] (3,0) circle (2pt);
  \filldraw[secondcolor] (1,0) circle (2pt);
\end{tikzpicture}
\end{center}

\begin{erwthma}
  \item Να λύσετε την ανίσωση $f(x) > 0$ στο διάστημα $[-1, 5]$.
  \item Να αποδείξετε ότι $f(x) \leq 4$ για κάθε $x \in [-1, 5]$.
  \item Να αποδείξετε ότι η εξίσωση $f(x) + f'(x) = 0$ έχει τουλάχιστον μία ρίζα στο διάστημα $(1, 3)$.
  \item Να αποδείξετε ότι $\dint_{-1}^{5} f(x)\,dx < 24$.
\end{erwthma}
\end{thema}
