\begin{thema}{Δ}
Στα σχήματα δίνονται \textbf{χωριστά} η $C_f$ (μπλε) και η $C_{f'}$ (κόκκινη) μιας $f:[-1, 5]\to\mathbb{R}$. Από τη γραφική: $f(-1) = 2$, $f(1) = 4$ (μέγιστο), $f(3) = 0$ (ρίζα), $f(5) = -2$.

\begin{center}

  \draw[help lines, gray!30] (-2,-3) grid (6,5);
  \draw[-latex] (-2.3,0) -- (6.3,0) node[right] {$x$};
  \draw[-latex] (0,-3.3) -- (0,5.3) node[above] {$y$};
  \foreach \x in {-1,1,2,3,4,5}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-2,-1,1,2,3,4}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  \draw[very thick, blue!70!black, smooth, tension=0.5]
    plot coordinates {(-1,2) (0,3.5) (1,4) (2,2.5) (3,0) (4,-1.5) (5,-2)};

  \filldraw[blue!70!black] (1,4) circle (2pt) node[above,font=\footnotesize] {$(1,4)$};
  \filldraw[blue!70!black] (3,0) circle (2pt);
  \filldraw[red!70!black] (1,0) circle (2pt);
\end{tikzpicture}
\end{center}

\begin{erwthma}
  \item Αποδείξτε ότι $f'(x) < 0$ $\forall x \in (1, 5)$ (γραφικά). Λύστε $f(x) > 0$ στο $[-1, 5]$.
  \item Γιατί ισχύει η ανίσωση $f(x) < 4$ $\forall x \neq 1$ στο $[-1, 5]$; (Χρησιμοποιήστε ότι $f(1)$ ολικό μέγιστο.)
  \item Αποδείξτε ότι η εξίσωση $f(x) + f'(x) = 0$ έχει τουλάχιστον μία λύση στο $(1, 5)$. (Θεωρήστε $g(x) = e^x f(x)$ και εφαρμόστε Rolle.)
  \item Δείξτε ότι $\dint_{-1}^{5} f(x)\,dx < 4 \cdot 6 = 24$ και εκτιμήστε γεωμετρικά (πρόσημο ολοκληρώματος).
\end{erwthma}
\end{thema}
