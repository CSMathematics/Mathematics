\begin{thema}{Δ}
Θεωρούμε τη συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ για την οποία ικανοποιείται η σχέση:
\[ f^3(x) + e^{f(x)} = x + e \quad \text{για κάθε } x \in \mathbb{R} \]
\begin{erwthma}
    \item Να αποδείξετε ότι η $f$ αντιστρέφεται στο $\mathbb{R}$ και να βρείτε τον τύπο της αντίστροφης συνάρτησης $f^{-1}$. \monades{6}
    \item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα και να βρείτε το πρόσημό της σε όλο το πεδίο ορισμού της. \monades{6}
    \item Να λύσετε την ανίσωση $f(e^{x-1} + x^3 - 2) > 1$. \monades{6}
    \item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x = e - 1$ και $x = 1+e$. \monades{7}
\end{erwthma}
\end{thema}
