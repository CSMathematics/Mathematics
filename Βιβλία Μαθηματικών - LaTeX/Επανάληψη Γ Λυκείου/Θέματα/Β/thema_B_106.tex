\begin{thema}{Β}
Στο σχήμα δίνεται η γραφική παράσταση μιας συνάρτησης $f$ ορισμένης στο $[-4, 4]$, η οποία είναι \textbf{περιττή}. Επίσης $f(1) = 2$ και $f(2) = 3$.

\begin{center}
\begin{tikzpicture}[scale=0.7]
  \draw[help lines, gray!30] (-5,-4) grid (5,4);
  \draw[-latex] (-5.3,0) -- (5.3,0) node[right] {$x$};
  \draw[-latex] (0,-4.3) -- (0,4.3) node[above] {$y$};
  \foreach \x in {-4,-3,-2,-1,1,2,3,4}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-3,-2,-1,1,2,3}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % odd function: smooth curve through origin
  \draw[very thick, blue!70!black, smooth, tension=0.5]
    plot coordinates {(0,0) (1,2) (2,3) (3,3.2) (4,3.5)};
  \filldraw[blue!70!black] (1,2) circle (2pt) node[above left,font=\footnotesize] {$(1,2)$};
  \filldraw[blue!70!black] (2,3) circle (2pt) node[above left,font=\footnotesize] {$(2,3)$};
  \filldraw[blue!70!black] (-1,-2) circle (2pt);
  \filldraw[blue!70!black] (-2,-3) circle (2pt);
  \filldraw[blue!70!black] (0,0) circle (2pt);
\end{tikzpicture}
\end{center}

\begin{erwthma}
  \item Συμπληρώστε την γραφική παράσταση και υπολογίστε τις τιμές $f(-2)$ και $f(0)$.
  \item Αν η $f$ είναι γνησίως αύξουσα στο $[0, 4]$, είναι γνησίως αύξουσα και στο $[-4, 0]$?
  \item Να υπολογίσετε το $\dint_{-2}^{2} f(x) dx$ αιτιολογώντας την απάντησή σας.
  \item 
\end{erwthma}
\end{thema}
