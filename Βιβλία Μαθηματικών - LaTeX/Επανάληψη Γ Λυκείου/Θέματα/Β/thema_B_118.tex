\begin{thema}{Β}
Δίνεται η γραφική παράσταση $C_f$ μιας γνησίως αύξουσας $f:[0, 4]\to\mathbb{R}$ με $f(0) = 0$. Σχεδιάστε νοερά και τη $C_{f^{-1}}$. Δίνεται επιπλέον ότι η γαλάζια σκιασμένη περιοχή (μεταξύ $C_f$ και άξονα $y$ από $y=0$ ως $y=f(4)=3$) έχει εμβαδόν $E_1 = 5$.

\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[help lines, gray!30] (-1,-1) grid (5,4);
  \draw[-latex] (-1.3,0) -- (5.3,0) node[right] {$x$};
  \draw[-latex] (0,-1.3) -- (0,4.3) node[above] {$y$};
  \foreach \x in {1,2,3,4}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {1,2,3}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % f
  \draw[very thick, blue!70!black, smooth, tension=0.5]

  % shaded area between f and y=3 (i.e. area to the left of f)
  \fill[yellow!20, opacity=0.5, smooth, tension=0.5]
    (0,0) -- plot coordinates {(0,0) (1,0.5) (2,1.2) (3,2.2) (4,3)} -- (4,3) -- (0,3) -- (0,0);

  \filldraw[blue!70!black] (0,0) circle (2pt);
  \filldraw[blue!70!black] (4,3) circle (2pt) node[above right,font=\footnotesize] {$(4,3)$};

\begin{erwthma}
  \item Ποιο είναι το εμβαδόν $E_2 = \dint_0^4 f(x) dx$, αν γνωρίζετε ότι $E_1 + E_2 = 4 \cdot 3 = 12$;
  \item Χρησιμοποιώντας τη σχέση εμβαδών, να υπολογίσετε το $\dint_0^3 f^{-1}(y) dy$.
  \item Γιατί ισχύει $\dint_0^4 f(x) dx + \dint_0^3 f^{-1}(y) dy = 4 \cdot 3$;
  \item Αν κάποιος ισχυριζόταν ότι $\dint_0^4 f(x)dx = 4$, τι θα σήμαινε αυτό για τη μέση τιμή της $f$;
\end{erwthma}
\end{thema}
