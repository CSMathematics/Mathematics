\begin{thema}{Β}
Δίνεται η γραφική παράσταση της συνάρτησης $f(x) = \left| x-2 \right| - 1$.
\begin{center}
\begin{tikzpicture}
    \begin{axis}[axis lines = middle,
        axis line style = {->, >=latex},
        grid = major,
        grid style = {very thin, gray!30},
        xlabel = {$x$},
        ylabel = {$y$},
        xtick = {-1,1,2,3,4},
        ytick = {-1,1,2},
        every axis x label/.style = {at={(current axis.right of origin)}, anchor=west},
        every axis y label/.style = {at={(current axis.above origin)}, anchor=south},
        width = 8cm, height = 5cm, samples = 100,
        xmin=-0.5, xmax=4.5, ymin=-1.5, ymax=2.5,
        xtick distance=1, ytick distance=1
    ]
        \addplot[very thick, \xrwma, domain=0:4] {abs(x-2)-1};
        \node[below left] at (axis cs:0,0) {$0$};
    \end{axis}
\end{tikzpicture}
\end{center}
\begin{erwthma}
  \item Βρείτε τα ακρότατα της συνάρτησης και τις ρίζες της εξίσωσης $f(x)=0$.
  \item Γράψτε τη συνάρτηση χωρίς την απόλυτη τιμή και βρείτε την παράγωγό της όπου αυτή ορίζεται.
  \item Υπάρχει σημείο στο οποίο η $f$ δεν είναι παραγωγίσιμη?
  \item Υπολογίστε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$ και τον άξονα $x'x$.
\end{erwthma}
\end{thema}
