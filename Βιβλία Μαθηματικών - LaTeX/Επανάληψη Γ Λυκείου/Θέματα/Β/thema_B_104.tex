\begin{thema}{Β}
Στο παρακάτω σχήμα δίνεται η γραφική παράσταση $C_f$ μιας συνεχούς συνάρτησης $f$ ορισμένης στο $[-2, 3]$. Η $C_f$ αποτελείται από ένα ημικύκλιο (κέντρο $(0,0)$, ακτίνα $2$ στο $[-2, 0]$, πάνω από τον $x'x$) και ένα ευθύγραμμο τμήμα από $(0, 2)$ στο $(3, -1)$.

\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[help lines, gray!30] (-3,-2) grid (4,3);
  \draw[-latex] (-3.3,0) -- (4.3,0) node[right] {$x$};
  \draw[-latex] (0,-2.3) -- (0,3.3) node[above] {$y$};
  \foreach \x in {-2,-1,1,2,3}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-1,1,2}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % semicircle
  \draw[very thick, blue!70!black] (-2,0) arc (180:0:1 and 2);
  % line segment
  \draw[very thick, blue!70!black] (0,2) -- (3,-1);
  \filldraw[blue!70!black] (-2,0) circle (2pt);
  \filldraw[blue!70!black] (0,2) circle (2pt) node[above right,font=\footnotesize] {$(0,2)$};
  \filldraw[blue!70!black] (3,-1) circle (2pt) node[below right,font=\footnotesize] {$(3,-1)$};
\end{tikzpicture}
\end{center}

\begin{erwthma}
  \item Να βρείτε τις ρίζες της $f$ και να μελετήσετε τη μονοτονία.
  \item Να υπολογίσετε γεωμετρικά το $\dint_{-2}^{0} f(x) dx$ (εμβαδόν ημικυκλίου).
  \item Να βρείτε τα σημεία στα οποία η $f$ δεν είναι παραγωγίσιμη.
  \item Να υπολογίσετε το $\dint_{0}^{3} f(x) dx$ γεωμετρικά (εμβαδόν τριγώνων).
\end{erwthma}
\end{thema}
