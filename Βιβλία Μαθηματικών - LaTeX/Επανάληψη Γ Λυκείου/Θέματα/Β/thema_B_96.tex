\begin{thema}{Β}
Στο ίδιο σύστημα συντεταγμένων δίνονται οι γραφικές παραστάσεις δύο συναρτήσεων $f$και $g$.
\begin{center}
\begin{tikzpicture}
    \begin{axis}[axis lines = middle,
        axis line style = {->, >=latex},
        grid = major,
        grid style = {very thin, gray!30},
        xlabel = {$x$},
        ylabel = {$y$},
        xtick = {1,2,3},
        ytick = {1,2,3},
        every axis x label/.style = {at={(current axis.right of origin)}, anchor=west},
        every axis y label/.style = {at={(current axis.above origin)}, anchor=south},
        width = 6cm, height = 6cm, samples = 100,
        xmin=-0.5, xmax=2.5, ymin=-0.5, ymax=2.5,
        xtick distance=1, ytick distance=1
    ]
        \addplot[very thick, \xrwma, domain=0:2.2] {sqrt(x)} node[pos=0.8,below] {$C_f$};
        \addplot[very thick, black, domain=-0.5:1.6] {x^2} node[pos=0.8,right]{$C_g$};
        \node[below left] at (axis cs:0,0) {$0$};
    \end{axis}
\end{tikzpicture}
\end{center}
\begin{erwthma}
  \item Να βρείτε τις τετμημένες των σημείων τομής των $C_f$ και $C_g$ και να λύσετε γραφικά την ανίσωση $f(x) > g(x)$.
  \item Να γράψετε, το ορισμένο ολοκλήρωμα που εκφράζει το εμβαδόν του χωρίου που περικλείεται από τις γραφικές παραστάσεις των $f$ και $g$.
  \item Γνωρίζοντας ότι οι τύποι των συναρτήσεων είναι $f(x) = \sqrt{x}$ και $g(x) = x^2$ για $x \ge 0$, να υπολογίσετε το εμβαδόν του παραπάνω χωρίου.
  \item Έστω η συνάρτηση διαφοράς $h(x) = f(x) - g(x)$. Να αποδείξετε ότι η μέγιστη κατακόρυφη απόσταση μεταξύ των δύο γραφικών παραστάσεων στο διάστημα $(0, 1)$ είναι μικρότερη από $\frac{1}{2}$.
\end{erwthma}
\end{thema}
