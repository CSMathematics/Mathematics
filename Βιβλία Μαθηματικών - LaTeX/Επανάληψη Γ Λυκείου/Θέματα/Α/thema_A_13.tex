\begin{thema}{A}
\begin{erwthma}
    \item {Δίνονται δύο συναρτήσεις $ f,g $ ορισμένες σε ένα διάστημα $ [a,\beta] $ τέτοιες ώστε \begin{itemize} \item $ f(x)\geq g(x) $ για κάθε $ x\in[a,\beta] $ και \item $ f,g $ μη αρνητικές στο $ [a,\beta] $. \end{itemize} Να αποδείξετε ότι το εμβαδόν του χωρίου $ \varOmega $ μεταξύ των γραφικών παραστάσεων $ C_f,C_g $ και των ευθειών $ x=a,x=\beta $ δίνεται από τον τύπο \[ E(\varOmega)=\int_{a}^{\beta}{(f(x)-g(x))\d x} \]}
    \item Πότε λέμε ότι μια συνάρτηση $f$ στρέφει τα κοίλα προς τα άνω (ή είναι κυρτή) στο $\Delta$;
    \item Πότε μια συνάρτηση $f:A\to\mathbb{R}$ ονομάζεται συνάρτηση $1-1$;
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν, γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό}, αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη.
    \begin{alist}
        \item Στην κίνηση κινητού σε άξονα, αν ταχύτητα και επιτάχυνση είναι ομόσημες, η κίνηση επιταχύνεται.
        \item Ισχύει ότι $\dint_{a}^{\beta} (f(x) + g(x)) dx = \dint_{a}^{\beta} f(x) dx + \dint_{a}^{\beta} g(x) dx$, όταν οι $f, g$ είναι συνεχείς στο $[a, \beta]$.
        \item Αν $f'(x_0) = 0$, τότε υποχρεωτικά το $x_0$ είναι τοπικό ακρότατο.
        \item Υπάρχει παράγουσα για κάθε συνεχής συνάρτηση στο $\mathbb{R}$.
        \item Δύο συναρτήσεις είναι ίσες μόνο αν έχουν ίδιους τύπους.
    \end{alist}
    \item Η παράγωγος της ολοκληρωτικής συνάρτησης $F(x) = \dint_a^x f(t) dt$ (με $f$ συνεχή) είναι:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](4)
        \task $F'(x) = 0$
        \task $F'(x) = f(t)$
        \task $F'(x) = f(x)$
        \task $F'(x) = f(x) - f(a)$
    \end{tasks}
\end{erwthma}
\end{thema}
