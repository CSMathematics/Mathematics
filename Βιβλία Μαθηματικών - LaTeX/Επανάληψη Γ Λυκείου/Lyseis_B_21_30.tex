\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x-1}{x+1}$.

\begin{erwthma}
\item Το πεδίο ορισμού της $f$ απαιτεί $x+1 \ne 0 \Leftrightarrow x \ne -1$. Άρα $D_f = \mathbb{R} \setminus \{-1\}$.
Έστω $x_1, x_2 \in D_f$ με $f(x_1) = f(x_2)$. Τότε:
\[ \frac{x_1-1}{x_1+1} = \frac{x_2-1}{x_2+1} \Leftrightarrow (x_1-1)(x_2+1) = (x_2-1)(x_1+1) \]
\[ \Leftrightarrow x_1 x_2 + x_1 - x_2 - 1 = x_1 x_2 + x_2 - x_1 - 1 \Leftrightarrow 2x_1 = 2x_2 \Leftrightarrow x_1 = x_2 \]
Άρα η συνάρτηση $f$ είναι «1-1», συνεπώς αντιστρέφεται.

\item Θέτουμε $y = f(x)$ και λύνουμε ως προς $x$:
\[ y = \frac{x-1}{x+1} \Leftrightarrow y(x+1) = x-1 \Leftrightarrow yx + y = x - 1 \Leftrightarrow x(y-1) = -1 - y \]
Για τον περιορισμό πρέπει $y-1 \ne 0 \Leftrightarrow y \ne 1$. Το πεδίο ορισμού της αντίστροφης είναι $A_{f^{-1}} = \mathbb{R} \setminus \{1\}$.
Για $y \ne 1$, έχουμε $x = \frac{-y-1}{y-1} = \frac{y+1}{1-y}$. Επειδή πρέπει επιπλέον $x \ne -1$, ελέγχουμε αν υπάρχει τιμή του $y$ ώστε $\frac{y+1}{1-y} = -1 \Leftrightarrow y+1 = -1+y \Leftrightarrow 1 = -1$ (αδύνατον).
Άρα ο τύπος της αντίστροφης είναι $f^{-1}(x) = \frac{x+1}{1-x}$, για $x \ne 1$.

\item Το πεδίο ορισμού της συνάρτησης $h = f \circ f$ είναι:
\[ A_h = \{x \in D_f \mid f(x) \in D_f\} = \{x \ne -1 \mid f(x) \ne -1\} \]
Λύνουμε τον περιορισμό $f(x) \ne -1$:
\[ \frac{x-1}{x+1} \ne -1 \Leftrightarrow x-1 \ne -x-1 \Leftrightarrow 2x \ne 0 \Leftrightarrow x \ne 0 \]
Συνεπώς το πεδίο ορισμού είναι το $A_h = \mathbb{R} \setminus \{-1, 0\}$.
Για τον τύπο της ισχύει:
\[ h(x) = f(f(x)) = \frac{f(x)-1}{f(x)+1} = \frac{\frac{x-1}{x+1} - 1}{\frac{x-1}{x+1} + 1} = \frac{\frac{x-1-x-1}{x+1}}{\frac{x-1+x+1}{x+1}} = \frac{-2}{2x} = -\frac{1}{x} \]
Άρα $h(x) = -\frac{1}{x}$ για $x \in \mathbb{R} \setminus \{-1, 0\}$.

\item Ορίζουμε τη συνάρτηση $K(x) = x \cdot f\left(\frac{1}{x}\right)$ για $x \to +\infty$.
Ο τύπος της για $x$ αρκετά μεγάλο (π.χ. $x > 0$) είναι:
\[ K(x) = x \cdot \frac{\frac{1}{x}-1}{\frac{1}{x}+1} = x \cdot \frac{\frac{1-x}{x}}{\frac{1+x}{x}} = x \cdot \frac{1-x}{1+x} = \frac{x-x^2}{1+x} \]
Το ζητούμενο όριο υπολογίζεται ως έξης (με όριο ρητής):
\[ \lim_{x\to+\infty} K(x) = \lim_{x\to+\infty} \frac{-x^2+x}{x+1} = \lim_{x\to+\infty} \frac{-x^2}{x} = \lim_{x\to+\infty} (-x) = -\infty \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{2x^2-1}{x^2+1}$.

\begin{erwthma}
\item Ο παρονομαστής $x^2+1 > 0$ για κάθε $x \in \mathbb{R}$, άρα το πεδίο ορισμού της $f$ είναι $D_f = \mathbb{R}$.
Για να βρούμε τις οριζόντιες ασύμπτωτες, υπολογίζουμε τα όρια της $f(x)$ στο $\pm\infty$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{2x^2-1}{x^2+1} = \lim_{x\to+\infty} \frac{2x^2}{x^2} = 2 \]
Ομοίως, $\lim\limits_{x\to-\infty} f(x) = 2$.
Άρα η ευθεία $y = 2$ είναι οριζόντια ασύμπτωτη της $C_f$ και στο $+\infty$ και στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο:
\[ f'(x) = \left( \frac{2x^2-1}{x^2+1} \right)' = \frac{(2x^2-1)'(x^2+1) - (2x^2-1)(x^2+1)'}{(x^2+1)^2} \]
\[ f'(x) = \frac{4x(x^2+1) - (2x^2-1)2x}{(x^2+1)^2} = \frac{4x^3+4x-4x^3+2x}{(x^2+1)^2} = \frac{6x}{(x^2+1)^2} \]
Η ρίζα της παραγώγου είναι $f'(x) = 0 \Leftrightarrow 6x = 0 \Leftrightarrow x = 0$.
Επειδή $(x^2+1)^2 > 0$ για κάθε $x$, το πρόσημο της $f'$ ταυτίζεται με το πρόσημο του $6x$. Άρα:

Οπότε, η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$ και γνησίως αύξουσα στο $[0, +\infty)$.
Παρουσιάζει ολικό ελάχιστο στη θέση $x_0 = 0$, με ελάχιστη τιμή $f(0) = \frac{-1}{1} = -1$.

\item \textbf{Σημεία τομής με τους άξονες:}
Με τον άξονα $y'y$: Θέτουμε $x=0$ και προκύπτει $f(0) = -1$. Άρα τέμνει τον άξονα $y'y$ στο σημείο $A(0, -1)$.
Με τον άξονα $x'x$: Θέτουμε $f(x) = 0$.
\[ \frac{2x^2-1}{x^2+1} = 0 \Leftrightarrow 2x^2-1 = 0 \Leftrightarrow 2x^2 = 1 \Leftrightarrow x^2 = \frac{1}{2} \Leftrightarrow x = \pm\frac{1}{\sqrt{2}} = \pm\frac{\sqrt{2}}{2} \]
Άρα τέμνει τον $x'x$ στα σημεία $B_1\left(\frac{\sqrt{2}}{2}, 0\right)$ και $B_2\left(-\frac{\sqrt{2}}{2}, 0\right)$.
\textbf{Εξίσωση εφαπτομένης στο $(0, -1)$:}
Είναι $f'(0) = \frac{6(0)}{(0^2+1)^2} = 0$. Η εξίσωση της εφαπτομένης στο $x_0 = 0$ είναι:
\[ y - f(0) = f'(0)(x-0) \Leftrightarrow y - (-1) = 0 \Leftrightarrow y = -1 \]

\item Το χωρίο περικλείεται από την $C_f$, την ευθεία $y=2$ και τις ευθείες $x=0, x=1$.
Για κάθε $x \in \mathbb{R}$, παρατηρούμε ότι: 
$2 - f(x) = 2 - \frac{2x^2-1}{x^2+1} = \frac{2x^2+2-2x^2+1}{x^2+1} = \frac{3}{x^2+1} > 0$.
Οπότε $y = 2$ είναι πάνω από την $C_f$. Το εμβαδόν δίνεται από το ολοκλήρωμα:
\[ E = \int_0^1 (2 - f(x)) dx = \int_0^1 \frac{3}{x^2+1} dx \]
Θέτουμε κατά την γνωστή μέθοδο $x = \varepsilon\phi u$ (οπότε $dx = \frac{1}{\sigma\upsilon\nu^2 u} du$).
Αφού $\frac{1}{\sigma\upsilon\nu^2 u} = 1+\varepsilon\phi^2 u \Rightarrow dx = (1+\varepsilon\phi^2 u) du = (1+x^2)du$.
Τα όρια αλλάζουν ώς εξής: Για $x=0 \Rightarrow \varepsilon\phi u = 0 \Rightarrow u=0$. Για $x=1 \Rightarrow \varepsilon\phi u = 1 \Rightarrow u = \frac{\pi}{4}$.
Άρα το εμβαδόν υπολογίζεται:
\[ E = \int_0^{\pi/4} \frac{3}{\varepsilon\phi^2 u + 1}(1+\varepsilon\phi^2 u)du = \int_0^{\pi/4} 3 du = 3[u]_0^{\pi/4} = \frac{3\pi}{4} \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = (x-1)e^{-x} + 2$.

\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι $D_f = \mathbb{R}$. Είναι δύο φορές παραγωγίσιμη.
\textbf{Μονοτονία:}
\[ f'(x) = (x-1)'e^{-x} + (x-1)(e^{-x})' = e^{-x} - (x-1)e^{-x} = e^{-x}(1 - x + 1) = e^{-x}(2-x) \]
Η ρίζα της $f'$ είναι $x = 2$. Αφού $e^{-x} > 0$:
- $f'(x) > 0 \Leftrightarrow 2-x > 0 \Leftrightarrow x < 2$.
- $f'(x) < 0 \Leftrightarrow 2-x < 0 \Leftrightarrow x > 2$.

\textbf{Κυρτότητα:}
\[ f''(x) = (e^{-x})'(2-x) + e^{-x}(2-x)' = -e^{-x}(2-x) - e^{-x} = e^{-x}(-2+x-1) = e^{-x}(x-3) \]
Η ρίζα της $f''$ είναι $x = 3$. Αφού $e^{-x} > 0$:
- $f''(x) > 0 \Leftrightarrow x-3 > 0 \Leftrightarrow x > 3$.
- $f''(x) < 0 \Leftrightarrow x < 3$.

Άρα η $f$ είναι κοίλη στο $(-\infty, 3]$ και κυρτή στο $[3, +\infty)$.

\item \textbf{Ακρότατα:} Από τη μελέτη μονοτονίας, η $f$ έχει ολικό μέγιστο για $x_0=2$, με τιμή:
\[ f(2) = (2-1)e^{-2} + 2 = \frac{1}{e^2} + 2 \]
\textbf{Σημεία Καμπής:} Από τη μελέτη κυρτότητας, η $C_f$ αλλάζει κυρτότητα στο $x_1=3$ όπου υπάρχει και εφαπτομένη, άρα παρουσιάζει σημείο καμπής στο σημείο $\Sigma(3, f(3))$ με:
\[ f(3) = (3-1)e^{-3} + 2 = \frac{2}{e^3} + 2 \]

\item Αναζητούμε το όριο στο $+\infty$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \left( \frac{x-1}{e^x} + 2 \right) \]
Για το επιμέρους όριο $\lim_{x\to+\infty} \frac{x-1}{e^x}$, επειδή έχουμε απροσδιοριστία $\frac{+\infty}{+\infty}$, εφαρμόζουμε De L' Hospital:
\[ \lim_{x\to+\infty} \frac{x-1}{e^x} \xlongequal[\text{DLH}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{(x-1)'}{(e^x)'} = \lim_{x\to+\infty} \frac{1}{e^x} = 0 \]
Κατά συνέπεια:
\[ \lim_{x\to+\infty} f(x) = 0 + 2 = 2 \]
Αυτό αποδεικνύει ότι η ευθεία $y = 2$ είναι οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$.

\item Στο διάστημα $[1, 2]$ ισχύει $x-1 \ge 0$, και $e^{-x} > 0$, επομένως $f(x) \ge 2 > 0$.
Το εμβαδόν είναι:
\[ E = \int_1^2 f(x) dx = \int_1^2 ((x-1)e^{-x} + 2) dx = \int_1^2 (x-1)e^{-x} dx + \int_1^2 2 dx \]
Υπολογίζουμε το πρώτο ολοκλήρωμα με παραγοντική:
\[ \int_1^2 (x-1)(-e^{-x})' dx = \left[ -(x-1)e^{-x} \right]_1^2 - \int_1^2 (x-1)'(-e^{-x}) dx \]
\[ = \left( -(2-1)e^{-2} \right) - \left( -(1-1)e^{-1} \right) + \int_1^2 e^{-x} dx = -e^{-2} - 0 + \left[ -e^{-x} \right]_1^2 \]
\[ = -e^{-2} - (e^{-2} - e^{-1}) = -2e^{-2} + e^{-1} \]
Το δεύτερο ολοκλήρωμα είναι:
\[ \int_1^2 2 dx = [2x]_1^2 = 4 - 2 = 2 \]
Προσθέτοντας, έχουμε το τελικό εμβαδόν:
\[ E = 2 + e^{-1} - 2e^{-2} = 2 + \frac{1}{e} - \frac{2}{e^2} \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x \ln x - x + 1$.

\begin{erwthma}
\item Πρέπει $x > 0$. Άρα $D_f = (0, +\infty)$.
Για το όριο $\lim\limits_{x \to 0^+} f(x)$, γνωρίζουμε το βασικό όριο $\lim\limits_{x \to 0^+} (x \ln x) = 0$. Επομένως:
\[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} (x \ln x - x + 1) = 0 - 0 + 1 = 1 \]

\item Η $f$ είναι παραγωγίσιμη στο $(0, +\infty)$ με:
\[ f'(x) = (x\ln x)' - (x)' + (1)' = \left( 1 \cdot \ln x + x \cdot \frac{1}{x} \right) - 1 = \ln x + 1 - 1 = \ln x \]
Έχουμε $f'(x) = 0 \Leftrightarrow \ln x = 0 \Leftrightarrow x = 1$.
Επίσης $f'(x) > 0 \Leftrightarrow \ln x > 0 \Leftrightarrow x > 1$, και όμοια $f'(x) < 0 \Leftrightarrow 0 < x < 1$.
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Η $f$ παρουσιάζει ολικό ελάχιστο στο $x_0 = 1$ με τιμή $f(1) = 1\ln 1 - 1 + 1 = 0$.

\item Από το προηγούμενο ερώτημα δείξαμε ότι το ολικό ελάχιστο της $f$ είναι το $0$. Επομένως για κάθε $x > 0$:
\[ f(x) \ge 0 \Leftrightarrow x \ln x - x + 1 \ge 0 \Leftrightarrow x \ln x \ge x - 1 \]
με την ισότητα να ισχύει μόνο για $x = 1$.

\item Στο διάστημα $[1, e]$, ξέρουμε ότι $f(x) \ge 0$. Άρα το εμβαδόν είναι:
\[ E = \int_1^e f(x) dx = \int_1^e (x \ln x - x + 1) dx \]
Υπολογίζουμε πρώτα με παραγοντική το ολοκλήρωμα $\int_1^e x \ln x dx$:
\[ \int_1^e x \ln x dx = \int_1^e \left(\frac{x^2}{2}\right)' \ln x dx = \left[ \frac{x^2}{2} \ln x \right]_1^e - \int_1^e \frac{x^2}{2} \cdot \frac{1}{x} dx \]
\[ = \frac{e^2}{2} \ln e - \frac{1^2}{2} \ln 1 - \frac{1}{2}\int_1^e x dx = \frac{e^2}{2} - \frac{1}{2} \left[ \frac{x^2}{2} \right]_1^e = \frac{e^2}{2} - \frac{e^2}{4} + \frac{1}{4} = \frac{e^2+1}{4} \]
Στη συνέχεια υπολογίζουμε το υπόλοιπο κομμάτι:
\[ \int_1^e (-x+1) dx = \left[ -\frac{x^2}{2} + x \right]_1^e = \left(-\frac{e^2}{2} + e\right) - \left(-\frac{1}{2} + 1\right) = -\frac{e^2}{2} + e - \frac{1}{2} \]
Συνολικά έχουμε:
\[ E = \frac{e^2+1}{4} - \frac{2e^2}{4} + \frac{4e}{4} - \frac{2}{4} = \frac{-e^2 + 4e - 1}{4} \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 - 2\ln x$.

\begin{erwthma}
\item Πρέπει $x > 0$. Άρα $D_f = (0, +\infty)$.
\textbf{Μονοτονία και ακρότατα:}
Η $f$ είναι παραγωγίσιμη με $f'(x) = 2x - \frac{2}{x} = \frac{2x^2-2}{x}$.
Επειδή $x>0$, έχουμε $f'(x) = 0 \Leftrightarrow 2x^2-2 = 0 \Leftrightarrow x^2 = 1 \Leftrightarrow x = 1$.
Επίσης $f'(x) > 0 \Leftrightarrow 2x^2-2 > 0 \Leftrightarrow x > 1$, και όμοια $f'(x) < 0 \Leftrightarrow 0 < x < 1$.
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Παρουσιάζει ολικό ελάχιστο στο $x_0 = 1$ με τιμή $f(1) = 1^2 - 2\ln 1 = 1$.
\textbf{Κυρτότητα:}
Η $f'$ είναι παραγωγίσιμη με $f''(x) = (2x - 2x^{-1})' = 2 + 2x^{-2} = 2 + \frac{2}{x^2}$.
Προφανώς $f''(x) > 0$ για κάθε $x > 0$.

\item Αφού το ολικό ελάχιστο της $f$ είναι 1, ισχύει $f(x) \ge 1$ για κάθε $x > 0$.
Οπότε:
\[ x^2 - 2\ln x \ge 1 \Leftrightarrow x^2 \ge 2\ln x + 1 \]

\item Αναζητούμε σημείο $M(x_0, f(x_0))$ της $C_f$ στο οποίο ο συντελεστής διεύθυνσης της εφαπτομένης να ισούται με τη διεύθυνση της ευθείας $\varepsilon: y=2x-3$, δηλαδή $\lambda_\varepsilon = 2$.
Πρέπει $f'(x_0) = 2 \Leftrightarrow \frac{2x_0^2-2}{x_0} = 2 \Leftrightarrow 2x_0^2 - 2 = 2x_0 \Leftrightarrow x_0^2 - x_0 - 1 = 0$.
Το τριώνυμο έχει διακρίνουσα $\Delta = 1 - 4(1)(-1) = 5$. Άρα οι ρίζες είναι $x = \frac{1 \pm \sqrt{5}}{2}$.
Επειδή $D_f = (0, +\infty)$, δεκτή γίνεται μόνο η ρίζα $x_0 = \frac{1+\sqrt{5}}{2}$ αφού η άλλη είναι αρνητική. Επομένως υπάρχει μοναδικό σημείο.
Η εξίσωση της εφαπτομένης είναι $y - f(x_0) = 2(x - x_0) \Leftrightarrow y = 2x - 2x_0 + f(x_0)$.
Είναι $f(x_0) = x_0^2 - 2\ln x_0$. Όμως $x_0^2 = x_0 + 1$, άρα $f(x_0) = x_0 + 1 - 2\ln x_0$.
Τελικά $y = 2x - 2x_0 + x_0 + 1 - 2\ln x_0 = 2x - x_0 + 1 - 2\ln x_0$, οπότε παραμένοντας στο ακριβές:
\[ y = 2x + \frac{1-\sqrt{5}}{2} - 2\ln\left( \frac{1+\sqrt{5}}{2} \right) \]

\item Θεωρούμε τη συνάρτηση $g(x) = f(x) - 2$ στο $[1, e]$.
- Η $g$ είναι συνεχής στο $[1, e]$.
- $g(1) = f(1) - 2 = 1 - 2 = -1 < 0$.

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \ln(e^x + 1) - x$.

\begin{erwthma}
\item Το πεδίο ορισμού είναι $D_f = \mathbb{R}$ αφού $e^x+1>0$.
Από ιδιότητες λογαρίθμων:
\[ f(x) = \ln(e^x + 1) - \ln(e^x) = \ln\left( \frac{e^x+1}{e^x} \right) = \ln(1 + e^{-x}) \]
Τα όρια υπολογίζονται πλέον εύκολα:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} \ln(1 + e^{-x}) \]
Επειδή $\lim_{x\to-\infty} e^{-x} = +\infty$, το όριο είναι $\ln(+\infty) = +\infty$.
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \ln(1 + e^{-x}) = \ln(1+0) = 0 \]

\item \textbf{Στο $+\infty$}: 
Αφού $\lim_{x\to+\infty} f(x) = 0$, η ευθεία $y = 0$ (άξονας $x'x$) είναι οριζόντια ασύμπτωτη.
\textbf{Στο $-\infty$}:
Αναζητούμε πλάγια ασύμπτωτη $y = \lambda x + \beta$.
- Υπολογίζουμε το $\lambda = \lim_{x\to-\infty} \frac{f(x)}{x} = \lim_{x\to-\infty} \left( \frac{\ln(e^x+1)}{x} - 1 \right)$.
Επειδή $\lim_{x\to-\infty} \ln(e^x+1) = \ln(0+1) = 0$ και ο παρονομαστής $x \to -\infty$, το όριο $\frac{\ln(e^x+1)}{x}$ είναι $0$. Οπότε $\lambda = -1$.
- Υπολογίζουμε τη $\beta = \lim_{x\to-\infty} (f(x) - \lambda x) = \lim_{x\to-\infty} (\ln(e^x+1) - x - (-x)) = \lim_{x\to-\infty} \ln(e^x+1) = \ln 1 = 0$.
Άρα η ευθεία $y = -x$ είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη με $f'(x) = \frac{e^x}{e^x+1} - 1 = \frac{e^x - (e^x+1)}{e^x+1} = \frac{-1}{e^x+1}$.

Άρα η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$, άρα είναι «1-1» και επομένως αντιστρέφεται.

\[ f(D_f) = \left( \lim_{x\to+\infty} f(x), \lim_{x\to-\infty} f(x) \right) = (0, +\infty) \]

\[ y = \ln(1+e^{-x}) \Leftrightarrow e^y = 1+e^{-x} \Leftrightarrow e^{-x} = e^y - 1 \]

\[ -x = \ln(e^y - 1) \Leftrightarrow x = -\ln(e^y-1) \]
Άρα ο τύπος της είναι $f^{-1}(x) = -\ln(e^x - 1)$, με $x \in (0, +\infty)$.

Επίσης $g'(x) = f'(x) - 1 < 0$ αφού $f'(x) < 0$. Άρα η $g$ είναι γνησίως φθίνουσα, που σημαίνει ότι η ρίζα είναι μοναδική.
\end{erwthma}
