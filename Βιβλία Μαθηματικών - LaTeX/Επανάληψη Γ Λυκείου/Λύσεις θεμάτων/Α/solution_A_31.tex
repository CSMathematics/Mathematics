\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Γεωμετρικά, όταν η εφαπτομένη της $C_f$ στο σημείο $M$ διαπερνά την καμπύλη, το σημείο $M$ χαρακτηρίζεται σημείο καμπής.
\item \textbf{Σωστό}. Το εμβαδόν του χωρίου που περικλείεται από τις $C_f,C_g$ στο $[a,\beta]$ δίνεται από
\[ E=\dint_a^\beta |f(x)-g(x)|\d x. \]
\item \textbf{Σωστό}. Καθώς $x\to0$, έχουμε $|x|\to0^+$, άρα
\[ \dfrac{1}{|x|}\to+\infty. \]
\item \textbf{Σωστό}. Για παράδειγμα η συνάρτηση
\[ F(x)=\dfrac{x|x|}{2} \]
είναι παραγωγίσιμη στο $\mathbb{R}$ και έχει $F'(x)=|x|$.
\item \textbf{Σωστό}. Αν άλλαζε πρόσημο, από το θεώρημα ενδιάμεσων τιμών θα μηδενιζόταν σε κάποιο σημείο του διαστήματος, άτοπο.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Αν $F_1,F_2$ είναι δύο παράγουσες της ίδιας συνάρτησης $f$ στο διάστημα $\Delta$, τότε
\[ (F_1-F_2)'(x)=F_1'(x)-F_2'(x)=f(x)-f(x)=0, \]
οπότε $F_1(x)-F_2(x)=c$ για κάθε $x\in\Delta$.
\end{erwthma}
\end{thema}
