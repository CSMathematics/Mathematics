\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Αν η $f$ είναι $1-1$ και παραγωγίσιμη με $f'(x)\neq0$, τότε η αντίστροφη είναι παραγωγίσιμη στις αντίστοιχες τιμές και ισχύει
\[ (f^{-1})'(y)=\dfrac{1}{f'(x)},\quad y=f(x). \]
\item \textbf{Σωστό}. Για $f(x)=x^3$ έχουμε $f'(x)=3x^2$, άρα $f'(0)=0$ και $f(0)=0$. Η εφαπτομένη στο $x=0$ είναι
\[ y-f(0)=f'(0)(x-0)\Rightarrow y=0, \]
δηλαδή ο άξονας $x'x$.
\item \textbf{Σωστό}. Η $f(x)=x^2$ δεν είναι $1-1$ στο $\mathbb{R}$, αφού $f(-1)=f(1)=1$ ενώ $-1\neq1$.
\item \textbf{Λάθος}. Για κυρτή συνάρτηση η γραφική παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της, αλλά όχι υποχρεωτικά μόνο με κοινό σημείο το σημείο επαφής. Για παράδειγμα, η γραμμική συνάρτηση είναι κυρτή και κάθε εφαπτομένη της ταυτίζεται με τη γραφική της παράσταση.
\item \textbf{Σωστό}. Οι δύο συνθήκες
\[ \lim\limits_{x\to+\infty}\dfrac{f(x)}{x}=\lambda
\quad\textrm{και}\quad
\lim\limits_{x\to+\infty}\left(f(x)-\lambda x\right)=\mu \]
δίνουν την ασύμπτωτη $y=\lambda x+\mu$ στο $+\infty$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{γ}. Για συνεχή συνάρτηση ισχύει, για οποιαδήποτε $a,\beta$,
\[ \dint_a^\beta f(x)\d x=-\dint_\beta^a f(x)\d x. \]
\end{erwthma}
\end{thema}
