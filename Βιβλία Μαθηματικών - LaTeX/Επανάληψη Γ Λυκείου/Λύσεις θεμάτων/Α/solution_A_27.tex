\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Αφού $f(x)\ge2$ στο $[1,3]$, ισχύει
\[ \dint_1^3 f(x)\d x\ge\dint_1^3 2\d x=4. \]
\item \textbf{Λάθος}. Η συνέχεια σε ανοικτό διάστημα δεν εξασφαλίζει ολικά ακρότατα. Για παράδειγμα, η $f(x)=x$ στο $(0,1)$ δεν έχει ούτε ολικό μέγιστο ούτε ολικό ελάχιστο.
\item \textbf{Σωστό}. Αν $F(x)=G(x)+c$, τότε $F'(x)=G'(x)$, άρα οι $F,G$ είναι παράγουσες της ίδιας συνάρτησης.
\item \textbf{Σωστό}. Αν το $x_0$ ήταν θέση ακρότατου και η $f$ ήταν παραγωγίσιμη στο $x_0$, τότε από το θεώρημα \eng{Fermat} θα είχαμε $f'(x_0)=0$, άτοπο.
\item \textbf{Σωστό}. Ισχύει
\[ (\hm{x})'=\syn{x}. \]
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Αν $f'(x)=0$ για κάθε $x$ σε ένα διάστημα $\Delta$, τότε η $f$ είναι σταθερή στο $\Delta$.
\end{erwthma}
\end{thema}
