\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Λάθος}. Μια συνάρτηση $1-1$ δεν είναι υποχρεωτικά γνησίως μονότονη. Για παράδειγμα, στο $A=\{1,2,3\}$ η συνάρτηση με
\[ f(1)=1,\quad f(2)=3,\quad f(3)=2 \]
είναι $1-1$, αλλά δεν είναι γνησίως μονότονη.
\item \textbf{Σωστό}. Αν $f'(x)=0$ σε κάθε εσωτερικό σημείο ενός διαστήματος, τότε από το αντίστοιχο κριτήριο η $f$ είναι σταθερή στο διάστημα.
\item \textbf{Σωστό}. Αφού
\[ \lim_{x\to+\infty}\bigl(f(x)-(x+2)\bigr)=0, \]
η ευθεία $y=x+2$ είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$.
\item \textbf{Λάθος}. Μπορεί το γινόμενο να έχει όριο $0$ χωρίς να έχουν τα επιμέρους όρια τιμή $0$. Για παράδειγμα, αν
\[
f(x)=
\begin{cases}
0, & x\in\mathbb{Q},\\
1, & x\notin\mathbb{Q},
\end{cases}
\quad
g(x)=
\begin{cases}
1, & x\in\mathbb{Q},\\
0, & x\notin\mathbb{Q},
\end{cases}
\]
τότε $f(x)g(x)=0$ για κάθε $x$, ενώ κανένα από τα δύο όρια δεν είναι $0$.
\item \textbf{Σωστό}. Η $f(x)=x^3$ είναι γνησίως αύξουσα στο $\mathbb{R}$, αν και $f'(0)=0$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Αφού η $f$ είναι συνεχής και γνησίως αύξουσα στο $[a,\beta]$, παίρνει όλες τις τιμές από $f(a)$ έως $f(\beta)$, άρα το σύνολο τιμών της είναι
\[ [f(a),f(\beta)]. \]
\end{erwthma}
\end{thema}
