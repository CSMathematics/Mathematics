\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Αφού η $f$ είναι συνεχής και γνησίως αύξουσα στο $[a,\beta]$, παίρνει όλες τις τιμές από το $f(a)$ έως το $f(\beta)$, άρα το σύνολο τιμών της είναι $[f(a),f(\beta)]$.
\item \textbf{Λάθος}. Αν μια συνάρτηση είναι συνεχής στο $\mathbb{R}$, τότε ορίζεται στο $0$, άρα η γραφική της παράσταση έχει το σημείο $(0,f(0))$ πάνω στον άξονα $y'y$.
\item \textbf{Λάθος}. Ο σωστός κανόνας παραγώγισης γινομένου είναι
\[ (f(x)g(x))'=f'(x)g(x)+f(x)g'(x). \]
\item \textbf{Σωστό}. Από τη γραμμικότητα του ολοκληρώματος ισχύει
\[ \dint_a^\beta c f(x)\d x=c\dint_a^\beta f(x)\d x. \]
\item \textbf{Σωστό}. Στο $\mathbb{R}^*$ η συνάρτηση $\dfrac{|x|}{x}$ είναι ίση με $-1$ για $x<0$ και με $1$ για $x>0$, άρα έχει παράγωγο $0$ σε κάθε σημείο του $\mathbb{R}^*$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{δ}. Ο κανόνας \eng{L'Hospital} εφαρμόζεται άμεσα σε μορφές $\frac{0}{0}$ ή $\frac{\pm\infty}{\pm\infty}$, όχι στη μορφή $0\cdot\infty$.
\end{erwthma}
\end{thema}
