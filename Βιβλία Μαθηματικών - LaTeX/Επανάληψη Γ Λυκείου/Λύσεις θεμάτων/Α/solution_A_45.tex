\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Ισχύει $(e^x)'=e^x$ και, γενικότερα, για $a>0$ με $a\neq1$ ισχύει
\[ (a^x)'=a^x\ln a. \]
\item \textbf{Σωστό}. Επειδή $x^2\to0^+$ όταν $x\to0$, έχουμε
\[ \lim_{x\to0}\dfrac{1}{x^2}=+\infty. \]
\item \textbf{Σωστό}. Αν $\lim\limits_{x\to x_0}f(x)=\ell\in\mathbb{R}$ και $\lim\limits_{x\to x_0}g(x)=+\infty$, τότε
\[ \lim_{x\to x_0}(f(x)+g(x))=+\infty. \]
\item \textbf{Λάθος}. Επειδή $-1\leq \hm x\leq1$, έχουμε
\[ -\dfrac{1}{x}\leq\dfrac{\hm x}{x}\leq\dfrac{1}{x} \]
για $x>0$, άρα από το κριτήριο παρεμβολής
\[ \lim_{x\to+\infty}\dfrac{\hm x}{x}=0. \]
\item \textbf{Σωστό}. Αν μια οριζόντια ευθεία έτεμνε την $C_f$ σε δύο διαφορετικά σημεία, τότε η $f$ θα έπαιρνε την ίδια τιμή σε δύο διαφορετικά στοιχεία, άτοπο αφού είναι $1-1$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Από το θεώρημα \eng{Bolzano}, αφού η $f$ είναι συνεχής στο $[a,\beta]$ και $f(a)f(\beta)<0$, η εξίσωση $f(x)=0$ έχει τουλάχιστον μία ρίζα στο $(a,\beta)$.
\end{erwthma}
\end{thema}
