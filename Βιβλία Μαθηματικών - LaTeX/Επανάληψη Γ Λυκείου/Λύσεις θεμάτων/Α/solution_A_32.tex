\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Λάθος}. Η θετικότητα κοντά στο $x_0$ δεν εξασφαλίζει θετικό όριο. Για παράδειγμα, για $f(x)=(x-x_0)^2$ έχουμε $f(x)>0$ για $x\neq x_0$, αλλά
\[ \lim_{x\to x_0}f(x)=0. \]
\item \textbf{Σωστό}. Η $e^x$ είναι ορισμένη και συνεχής σε όλο το $\mathbb{R}$, άρα δεν υπάρχει πραγματικός αριθμός $x_0$ όπου να εμφανίζεται κατακόρυφη ασύμπτωτη.
\item \textbf{Σωστό}. Αν $f(a)f(\beta)=0$, τότε $f(a)=0$ ή $f(\beta)=0$, οπότε υπάρχει $x_0\in[a,\beta]$ με $f(x_0)=0$.
\item \textbf{Λάθος}. Μια εφαπτομένη μπορεί να τέμνει ξανά την καμπύλη σε άλλο σημείο.
\item \textbf{Σωστό}. Κάθε εσωτερικό σημείο στο οποίο μηδενίζεται η παράγωγος είναι κρίσιμο σημείο της συνάρτησης.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Επειδή μια αρχική της $\dfrac{1}{x}$ στο $(0,+\infty)$ είναι η $\ln x$, έχουμε
\[ \dint_1^e \dfrac{1}{x}\d x=\ln e-\ln 1=1. \]
\end{erwthma}
\end{thema}
