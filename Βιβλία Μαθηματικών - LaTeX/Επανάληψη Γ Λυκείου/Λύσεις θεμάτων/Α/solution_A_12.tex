\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Δύο συναρτήσεις είναι ίσες όταν έχουν το ίδιο πεδίο ορισμού και ίσες τιμές για κάθε στοιχείο του πεδίου ορισμού τους.
\item \textbf{Σωστό}. Η σχέση $\lim\limits_{x\to x_0}f(x)=\ell$ σημαίνει ακριβώς ότι οι τιμές $f(x)$ πλησιάζουν το $\ell$ όσο το $x$ πλησιάζει το $x_0$.
\item \textbf{Σωστό}. Αν $\lim\limits_{x\to x_0}g(x)=\alpha$ και η $f$ είναι συνεχής στο $\alpha$, τότε
\[ \lim_{x\to x_0}f(g(x))=f(\alpha)=f\left(\lim_{x\to x_0}g(x)\right). \]
\item \textbf{Λάθος}. Η συνθήκη $f''(x_0)=0$ δεν αρκεί για σημείο καμπής. Για παράδειγμα, για $f(x)=x^4$ έχουμε $f''(0)=0$, αλλά δεν αλλάζει κυρτότητα στο $0$.
\item \textbf{Σωστό}. Αν $f,g$ είναι κυρτές στο $\varDelta$, τότε οι $f',g'$ είναι γνησίως αύξουσες, άρα και $(f+g)'=f'+g'$ είναι αύξουσα. Επομένως η $f+g$ είναι κυρτή στο $\varDelta$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $(x_0,f(x_0))$ είναι
\[ y-f(x_0)=f'(x_0)(x-x_0). \]
\end{erwthma}
\end{thema}
