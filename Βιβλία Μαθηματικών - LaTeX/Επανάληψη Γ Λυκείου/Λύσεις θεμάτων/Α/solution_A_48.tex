\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Λάθος}. Για συνέχεια στο $x_0$ δεν αρκεί να υπάρχει το όριο και να ορίζεται η $f(x_0)$· πρέπει να ισχύει
\[ \lim_{x\to x_0}f(x)=f(x_0). \]
\item \textbf{Σωστό}. Η μορφή $+\infty-\infty$ είναι απροσδιόριστη.
\item \textbf{Λάθος}. Γενικά η σύνθεση δεν είναι αντιμεταθετική, δηλαδή δεν ισχύει πάντοτε $f\circ g=g\circ f$.
\item \textbf{Σωστό}. Για παράδειγμα, αν $f(x)=x$ και $g(x)=2x$ κοντά στο $0$ με κατάλληλο περιορισμό ώστε $x>0$, τότε $f(x)<g(x)$, αλλά και τα δύο όρια στο $0$ είναι $0$.
\item \textbf{Σωστό}. Αν $f(x)\geq0$ στο $[a,\beta]$, τότε το ορισμένο ολοκλήρωμα είναι μη αρνητικό:
\[ \dint_a^\beta f(x)\d x\geq0. \]
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Στο θεώρημα \eng{Rolle} απαιτείται η $f$ να είναι παραγωγίσιμη στο ανοικτό διάστημα $(a,\beta)$, όχι σε όλο το κλειστό $[a,\beta]$.
\end{erwthma}
\end{thema}
