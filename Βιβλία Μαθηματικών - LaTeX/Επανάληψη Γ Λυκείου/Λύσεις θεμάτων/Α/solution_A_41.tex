\begin{thema}{Α}
\begin{erwthma}
\item \apodeixiref{thewrhma:paragwgos_athroismatos}

\item Το κριτήριο παρεμβολής διατυπώνεται ως εξής: Αν για τις συναρτήσεις $f,g,h$ ισχύει
\[ h(x)\leq f(x)\leq g(x) \]
κοντά στο $x_0$ και
\[ \lim_{x\to x_0}h(x)=\lim_{x\to x_0}g(x)=l, \]
τότε
\[ \lim_{x\to x_0}f(x)=l. \]

\item \orismosref{orismos:shmeio_kamphs}

\item Οι χαρακτηρισμοί των προτάσεων είναι:
\begin{alist}
\item \textbf{Σωστό}. Εφόσον $f=(f+g)-g$ και οι $f+g$, $g$ είναι συνεχείς στο $x_0$, τότε και η $f$ είναι συνεχής στο $x_0$.
\item \textbf{Λάθος}. Η $|x|$ δεν είναι παραγωγίσιμη στο $0$, αφού οι πλευρικές παράγωγοι είναι $-1$ και $1$.
\item \textbf{Σωστό}. Αν η θετική $f$ μετατοπιστεί σε $f+c$, τότε το νέο εμβαδόν αυξάνεται κατά
\[ \dint_a^\beta c\d x=c(\beta-a). \]
\item \textbf{Σωστό}. Για $f(x)=\dfrac{1}{x}$ ισχύει
\[ \lim_{x\to-\infty}\dfrac{1}{x}=0, \]
άρα ο άξονας $x'x$ είναι οριζόντια ασύμπτωτη στο $-\infty$.
\item \textbf{Σωστό}. Είναι ακριβώς το θεώρημα \eng{Bolzano}: αφού η $f$ είναι συνεχής στο $[a,\beta]$ και $f(a)f(\beta)<0$, υπάρχει ρίζα στο $(a,\beta)$.
\end{alist}

\item Σωστή απάντηση είναι η \textbf{β}. Το ολοκλήρωμα περιττής συνάρτησης σε συμμετρικό διάστημα είναι μηδέν, δηλαδή
\[ \dint_{-a}^{a} f(x)\d x=0. \]
\end{erwthma}
\end{thema}
