\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^{2x} - 4e^x + 3$.

\begin{erwthma}
\item Για να βρούμε τα σημεία τομής με τον άξονα $x'x$ θέτουμε $f(x) = 0$:
\[ e^{2x} - 4e^x + 3 = 0 \]
Θέτουμε $u = e^x > 0$ οπότε προκύπτει $u^2 - 4u + 3 = 0$. Οι ρίζες του τριωνύμου είναι $u=1$ και $u=3$.
\begin{itemize}
\item Αν $e^x = 1 \Leftrightarrow x = 0$.
\item Αν $e^x = 3 \Leftrightarrow x = \ln 3$.
\end{itemize}
Άρα τα σημεία τομής είναι τα $A(0, 0)$ και $B(\ln 3, 0)$.
\item Αφού $\lim\limits_{x\to-\infty} e^{2x} = 0$ και $\lim\limits_{x\to-\infty} e^x = 0$, έχουμε:
\[ \lim_{x\to-\infty} f(x) = 0 - 0 + 3 = 3 \]
Άρα η ευθεία $y = 3$ είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.
Επίσης:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} e^{2x}\left(1 - \frac{4}{e^x} + \frac{3}{e^{2x}}\right) = (+\infty)(1 - 0 + 0) = +\infty \]
Άρα δεν υπάρχει οριζόντια ασύμπτωτη στο $+\infty$.

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ f'(x)=\left(e^{2x}-4e^x+3\right)' = 2e^{2x} - 4e^x = 2e^x(e^x - 2) \]
Οι ρίζες και τα πρόσημα της παραγώγου είναι 
\begin{itemize}
\item $e^x - 2 = 0 \Leftrightarrow x = \ln 2$, αφού $2e^x > 0$:
\item $f'(x) > 0 \Leftrightarrow e^x - 2 > 0 \Leftrightarrow e^x > 2 \Leftrightarrow x > \ln 2$.
\item $f'(x) < 0 \Leftrightarrow x < \ln 2$.
\end{itemize}

Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty, \ln 2]$ και γνησίως αύξουσα στο $[\ln 2, +\infty)$.
Παρουσιάζει ολικό ελάχιστο στο $x_0 = \ln 2$ με τιμή:
\[ f(\ln 2) = (e^{\ln 2})^2 - 4e^{\ln 2} + 3 = 2^2 - 4(2) + 3 = 4 - 8 + 3 = -1 \]

\item Στο ερώτημα (Β1) βρήκαμε τις ρίζες $x=0$ και $x=\ln 3$. Επειδή $\ln 2 \in (0, \ln 3)$ και $f(\ln 2)=-1 < 0$, η συνεχής $f$ διατηρεί πρόσημο και είναι $f(x) \le 0$ στο $[0, \ln 3]$.
Επομένως το εμβαδόν μεταξύ της $C_f$, του $x'x$ και των $x=0, x=\ln 3$ είναι:
\begin{align*}
E &= \int_0^{\ln 3} -f(x)\,\d x = \int_0^{\ln 3} (-e^{2x} + 4e^x - 3)\,\d x =\\
&= \left[ -\frac{e^{2x}}{2} + 4e^x - 3x \right]_0^{\ln 3}= \left( \frac{15}{2} - 3\ln 3 \right) - \frac{7}{2} = \frac{8}{2} - 3\ln 3 = 4 - 3\ln 3
\end{align*}

\end{erwthma}
\end{thema}
