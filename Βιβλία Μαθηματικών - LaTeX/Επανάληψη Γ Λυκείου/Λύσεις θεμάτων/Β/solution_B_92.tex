\begin{thema}{Β}
\begin{erwthma}
\item Από το σχήμα της $f'$ βλέπουμε ότι $f'(x)=0$ για $x=-1$ και $x=3$. Επίσης $f'(x)>0$ στα $(-\infty,-1)\cup(3,+\infty)$ και $f'(x)<0$ στο $(-1,3)$. Άρα η $f$ είναι γνησίως αύξουσα στα $(-\infty,-1]$, $[3,+\infty)$ και γνησίως φθίνουσα στο $[-1,3]$.

\item Στο $x=-1$ η $f'$ αλλάζει πρόσημο από $+$ σε $-$, άρα η $f$ παρουσιάζει τοπικό μέγιστο. Στο $x=3$ η $f'$ αλλάζει πρόσημο από $-$ σε $+$, άρα η $f$ παρουσιάζει τοπικό ελάχιστο.

\item Για την κυρτότητα της $f$ μελετάμε τη μονοτονία της $f'$. Από το σχήμα η $f'$ είναι γνησίως φθίνουσα στο $(-\infty,1]$ και γνησίως αύξουσα στο $[1,+\infty)$. Άρα η $f$ είναι κοίλη στο $(-\infty,1]$ και κυρτή στο $[1,+\infty)$. Επειδή στο $x=1$ αλλάζει η κυρτότητα, η $C_f$ έχει σημείο καμπής το $K(1,f(1))$.

\item Δίνεται $f(0)=2$ και από το σχήμα της $f'$ διαβάζουμε $f'(0)=-3$. Η εφαπτομένη στο $x_0=0$ είναι
\[
y-f(0)=f'(0)(x-0)\Leftrightarrow y-2=-3x\Leftrightarrow y=-3x+2.
\]
\end{erwthma}
\end{thema}
