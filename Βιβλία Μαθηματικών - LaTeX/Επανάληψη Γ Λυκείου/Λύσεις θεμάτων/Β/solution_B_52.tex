\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{e^x}{x}$ με $x \neq 0$.
\begin{erwthma}
\item Εξετάζουμε τα σημεία τομής με τον άξονα $x'x$. Θέτουμε $f(x) = 0$:
\[ \frac{e^x}{x} = 0 \Leftrightarrow e^x = 0 \]
Η εξίσωση $e^x = 0$ είναι \textbf{αδύνατη} στο $\mathbb{R}$, καθώς η εκθετική συνάρτηση λαμβάνει μόνο θετικές τιμές ($e^x > 0$ για κάθε $x \in \mathbb{R}$). 
Άρα η $C_f$ δεν έχει κανένα κοινό σημείο με τον άξονα $x'x$.

\item Υπολογίζουμε την παράγωγο της $f$:
\[ f'(x) = \frac{(e^x)'x - e^x(x)'}{x^2} = \frac{xe^x - e^x}{x^2} = \frac{e^x(x-1)}{x^2} \]
Η ρίζα της παραγώγου προκύπτει από τον αριθμητή: $x - 1 = 0 \Leftrightarrow x = 1$.
Το πρόσημο εξαρτάται μόνο από το $x-1$ αφού $e^x > 0$ και $x^2 > 0$ (για $x \neq 0$).
- $f'(x) > 0 \Leftrightarrow x > 1$.
- $f'(x) < 0 \Leftrightarrow x < 1$ (και προφανώς $x \ne 0$).

Η συνάρτηση είναι γνησίως φθίνουσα στα διαστήματα $(-\infty, 0)$ και $(0, 1]$, και γνησίως αύξουσα στο $[1, +\infty)$.
Παρουσιάζει \textbf{τοπικό ελάχιστο} στο $x = 1$ με τιμή $f(1) = \frac{e^1}{1} = e$.

\item \textbf{Όρια στο $\pm\infty$:}
- Στο $-\infty$: $\lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} \frac{e^x}{x} = 0$, διότι $\lim e^x = 0$ και $\lim \frac{1}{x} = 0$ (άρα $0 \cdot 0 = 0$).
- Στο $+\infty$: $\lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{e^x}{x} \xlongequal[\text{DLH}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{1} = +\infty$.
\textbf{Πλευρικά όρια στο 0:}
- Στο $0^+$: $\lim_{x\to0^+} \frac{e^x}{x} = \frac{e^0}{0^+} = \frac{1}{0^+} = +\infty$.
- Στο $0^-$: $\lim_{x\to0^-} \frac{e^x}{x} = \frac{e^0}{0^-} = \frac{1}{0^-} = -\infty$.

\item Στο διάστημα $\Delta_1 = (0, 1)$, η $f$ είναι συνεχής και γνησίως φθίνουσα.
Υπολογίζουμε το σύνολο τιμών $f(\Delta_1) = \left( \lim_{x\to1^-} f(x), \lim_{x\to0^+} f(x) \right) = (e, +\infty)$.
Παρατηρούμε ότι το $3$ ανήκει στο διάστημα $(e, +\infty)$ διότι $3 > e \approx 2,71$.
Εφόσον $3 \in f(\Delta_1)$ και η $f$ είναι γνησίως φθίνουσα στο $\Delta_1$, υπάρχει \textbf{ακριβώς μία} λύση της εξίσωσης $f(x_1) = 3$ στο διάστημα $(0, 1)$.
\end{erwthma}
\end{thema}
