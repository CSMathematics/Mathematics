\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{e^x}{x}$ για $x > 0$.
\begin{erwthma}
\item Θέτουμε
\[
g(x)=3f(x)-(e^2+e)=\frac{3e^x}{x}-(e^2+e), \quad x>0.
\]
Η $g$ είναι συνεχής στο $[1,2]$ και έχουμε
\begin{align*}
g(1)&=3e-(e^2+e)=2e-e^2=e(2-e)<0,\\
g(2)&=\frac{3e^2}{2}-(e^2+e)=\frac{e^2}{2}-e=e\left(\frac{e}{2}-1\right)>0.
\end{align*}
Άρα $g(1)g(2)<0$. Από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα
$x_0\in(1,2)$ τέτοιο ώστε $g(x_0)=0$, δηλαδή
\[
3f(x_0)=e^2+e.
\]

\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=\left(\frac{e^x}{x}\right)'=\frac{xe^x-e^x}{x^2}=\frac{e^x(x-1)}{x^2}.
\]
Για κάθε $x\in(1,2)$ ισχύει $e^x>0$, $x^2>0$ και $x-1>0$, άρα $f'(x)>0$.
Επομένως η $f$ είναι γνησίως αύξουσα στο $(1,2)$.

Αφού η $f$ είναι γνησίως αύξουσα στο $(1,2)$, η εξίσωση
\[
3f(x)=e^2+e
\]
δεν μπορεί να έχει πάνω από μία λύση στο $(1,2)$. Σε συνδυασμό με το προηγούμενο
ερώτημα, έχει ακριβώς μία λύση στο $(1,2)$.

\item Έχουμε
\[
\lim_{x\to+\infty}f(x)=\lim_{x\to+\infty}\frac{e^x}{x}
\dlh{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}e^x=+\infty.
\]

\item Έστω ότι η ζητούμενη εφαπτομένη εφάπτεται στην $C_f$ στο σημείο
$A(a,f(a))$, με $a>0$. Η εξίσωση της εφαπτομένης είναι
\[
y-f(a)=f'(a)(x-a).
\]
Αφού διέρχεται από την αρχή των αξόνων $O(0,0)$, έχουμε
\[
0-f(a)=f'(a)(0-a)\Leftrightarrow f(a)=af'(a).
\]
Άρα
\[
\frac{e^a}{a}=a\cdot \frac{e^a(a-1)}{a^2}
\Leftrightarrow \frac{e^a}{a}=\frac{e^a(a-1)}{a}
\Leftrightarrow 1=a-1
\Leftrightarrow a=2.
\]
Επομένως η εφαπτομένη έχει συντελεστή διεύθυνσης
\[
f'(2)=\frac{e^2(2-1)}{2^2}=\frac{e^2}{4}.
\]
Επειδή διέρχεται από το $O(0,0)$, η εξίσωσή της είναι
\[
y=\frac{e^2}{4}x.
\]
\end{erwthma}
\end{thema}
