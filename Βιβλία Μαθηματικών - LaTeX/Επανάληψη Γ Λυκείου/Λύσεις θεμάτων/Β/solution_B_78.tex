%=========== ΛΥΣΗ - 78ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ ισχύει
\[
1-x^2\leq f(x)\leq 1+x^2.
\]

Επειδή
\[
\lim_{x\to 0}(1-x^2)=1
\]
και
\[
\lim_{x\to 0}(1+x^2)=1,
\]
από το Κριτήριο Παρεμβολής προκύπτει
\[
\lim_{x\to 0}f(x)=1.
\]

Επίσης, θέτοντας $x=0$ στην αρχική ανισότητα, παίρνουμε
\[
1\leq f(0)\leq 1.
\]
Άρα
\[
f(0)=1.
\]

\item Από την αρχική ανισότητα και από το $f(0)=1$ έχουμε
\[
1-x^2\leq f(x)\leq 1+x^2.
\]
Άρα
\[
-x^2\leq f(x)-1\leq x^2.
\]
Επομένως,
\[
|f(x)-1|\leq x^2.
\]

Για $x\neq 0$ έχουμε
\[
\left|\frac{f(x)-f(0)}{x}\right|
=\left|\frac{f(x)-1}{x}\right|
\leq \frac{x^2}{|x|}
=|x|.
\]

Επειδή
\[
\lim_{x\to 0}|x|=0,
\]
από το Κριτήριο Παρεμβολής προκύπτει
\[
\lim_{x\to 0}\frac{f(x)-f(0)}{x}=0.
\]

Άρα η $f$ είναι παραγωγίσιμη στο $x_0=0$ και
\[
f'(0)=0.
\]

\item Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $x_0=0$ έχει εξίσωση
\[
y-f(0)=f'(0)(x-0).
\]

Επειδή
\[
f(0)=1
\]
και
\[
f'(0)=0,
\]
παίρνουμε
\[
y-1=0.
\]
Άρα η εφαπτομένη είναι
\[
y=1.
\]

\item Θα εξετάσουμε το όριο
\[
\lim_{x\to 0} f(x)\hm\left(\frac{1}{x}\right).
\]

Γνωρίζουμε ότι
\[
\lim_{x\to 0}f(x)=1.
\]
Η συνάρτηση
\[
\hm\left(\frac{1}{x}\right)
\]
δεν έχει όριο καθώς $x\to 0$.

Πιο συγκεκριμένα, για
\[
x_n=\frac{1}{\frac{\pi}{2}+2\pi n}
\]
έχουμε $x_n\to 0$ και
\[
\hm\left(\frac{1}{x_n}\right)=1.
\]
Άρα
\[
f(x_n)\hm\left(\frac{1}{x_n}\right)\to 1.
\]

Για
\[
y_n=\frac{1}{\frac{3\pi}{2}+2\pi n}
\]
έχουμε $y_n\to 0$ και
\[
\hm\left(\frac{1}{y_n}\right)=-1.
\]
Άρα
\[
f(y_n)\hm\left(\frac{1}{y_n}\right)\to -1.
\]

Εφόσον προκύπτουν δύο διαφορετικά οριακά αποτελέσματα, το όριο
\[
\lim_{x\to 0} f(x)\hm\left(\frac{1}{x}\right)
\]
δεν υπάρχει.
\end{erwthma}
\end{thema}
