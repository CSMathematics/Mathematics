\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^3}{x^2-1}$.

\begin{erwthma}
\item Το πεδίο ορισμού είναι $D_f = \mathbb{R} - \{-1, 1\}$.
Κατακόρυφες ασύμπτωτες αναζητούμε στα σημεία μη ορισμού $x=1$ και $x=-1$.
\begin{itemize}
\item Για $x \to 1^+$: $x^2-1 \to 0$
\[\lim\limits_{x\to1^+}{f(x)}=\lim\limits_{x\to1^+}{\frac{x^3}{x^2-1}}=\lim\limits_{x\to1^+}{\frac{x^3}{(x-1)(x+1)}}=\lim\limits_{x\to 1^+}{\left(\frac{x^3}{x+1}\frac{1}{x-1}\right)}=\frac{1}{2}(+\infty)=+\infty\]
Οπότε η ευθεία $x = 1$ είναι κατακόρυφη ασύμπτωτη.
\item Ομοίως για $x \to -1^+$: 
\[\lim\limits_{x\to-1^+}{f(x)}=\lim\limits_{x\to-1^+}{\frac{x^3}{x^2-1}}=\lim\limits_{x\to-1^+}{\frac{x^3}{(x-1)(x+1)}}=\lim\limits_{x\to -1^+}{\left(\frac{x^3}{x-1}\frac{1}{x+1}\right)}=\frac{1}{2}(+\infty)=+\infty\]
Άρα και η ευθεία $x = -1$ είναι κατακόρυφη ασύμπτωτη.
\end{itemize}


\item Εξετάζουμε τη διαφορά $f(x) - x$:
\[ f(x) - x = \frac{x^3}{x^2-1} - \frac{x(x^2-1)}{x^2-1} = \frac{x^3 - x^3 + x}{x^2-1} = \frac{x}{x^2-1} \]
Υπολογίζουμε το όριο στο $\pm\infty$:
\[ \lim_{x\to \pm\infty} (f(x) - x) = \lim_{x\to \pm\infty} \frac{x}{x^2-1} = \lim_{x\to \pm\infty} \frac{x}{x^2} = \lim_{x\to \pm\infty} \frac{1}{x} = 0 \]
Κατά συνέπεια, σύμφωνα με τον ορισμό, η ευθεία $y = x$ είναι πλάγια ασύμπτωτη της $C_f$ τόσο στο $+\infty$ όσο και στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη με:
\begin{align*}
f'(x) &= \frac{(x^3)'(x^2-1) - x^3(x^2-1)'}{(x^2-1)^2} = \frac{3x^2(x^2-1) - x^3(2x)}{(x^2-1)^2} =\\&= \frac{3x^4 - 3x^2 - 2x^4}{(x^2-1)^2} = \frac{x^4 - 3x^2}{(x^2-1)^2} = \frac{x^2(x^2-3)}{(x^2-1)^2}
\end{align*}

Οι ρίζες και τα πρόσημα της $f'$ είναι:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow\frac{x^2(x^2-3)}{(x^2-1)^2}=0\Leftrightarrow x^2=0\ \text{ή}\ x^2-3=0\Leftrightarrow x=0\ \text{ή}\  x=\pm\sqrt{3}$.
\item Το πρόσημο εξαρτάται από τον παράγοντα $(x^2-3)$, καθώς $x^2 \ge 0$ και $(x^2-1)^2 > 0$. Συνεπώς:
\item $f'(x) > 0 \Leftrightarrow x \in (-\infty, -\sqrt{3}) \cup (\sqrt{3}, +\infty)$.
\item $f'(x) < 0 \Leftrightarrow x \in (-\sqrt{3}, -1) \cup (-1, 0) \cup (0, 1) \cup (1, \sqrt{3})$.
\end{itemize}

Η $f$ παρουσιάζει:
\begin{itemize}
\item τοπικό μέγιστο στο $x = -\sqrt{3}$ ίσο με $f(-\sqrt{3}) = \frac{-3\sqrt{3}}{2}$, 
\item τοπικό ελάχιστο στο $x = \sqrt{3}$ ίσο με $f(\sqrt{3}) = \frac{3\sqrt{3}}{2}$.
\end{itemize}
\item Για να βρούμε το σύνολο τιμών της συνάρτησης $f$ καθώς και το πλήθος των ριζών της εξίσωσης $f(x) = \lambda$, θα βασιστούμε στον πίνακα μεταβολών και τα συμπεράσματα των προηγούμενων ερωτημάτων. 

Αρχικά, υπενθυμίζουμε ότι το πεδίο ορισμού είναι $D_f = (-\infty, -1) \cup (-1, 1) \cup (1, +\infty)$.
Από τη μελέτη της παραγώγου $f'(x) = \frac{x^2(x^2-3)}{(x^2-1)^2}$ γνωρίζουμε τα εξής κρίσιμα σημεία και όρια: $f(-\sqrt{3})= -\frac{3\sqrt{3}}{2}$, $f(\sqrt{3}) = \frac{3\sqrt{3}}{2}$, $\lim_{x \to -1^+} f(x) = +\infty$ και $\lim_{x \to 1^+} f(x) = +\infty$.

Για να βρούμε το σύνολο τιμών $f(D_f)$, χωρίζουμε το πεδίο ορισμού σε διαστήματα στα οποία η συνάρτηση είναι γνησίως μονότονη και συνεχής. Υπολογίζουμε την εικόνα του κάθε διαστήματος:

\begin{itemize}
    \item Στο διάστημα $\Delta_1 = (-\infty, -\sqrt{3}]$, η $f$ είναι συνεχής και γνησίως αύξουσα. Άρα η εικόνα του είναι:
    \[ f(\Delta_1) = \left( \lim_{x \to -\infty} f(x), f(-\sqrt{3}) \right] = \left( -\infty, -\frac{3\sqrt{3}}{2} \right] \]
    αφού \[\lim\limits_{x\to-\infty}{f(x)}=\lim\limits_{x\to-\infty}{\frac{x^3}{x^2-1}}=\lim\limits_{x\to-\infty}{\frac{x^3}{x^2}}=\lim\limits_{x\to-\infty}{x}=-\infty\]
    \item Στο διάστημα $\Delta_2 = [-\sqrt{3}, -1)$, η $f$ είναι συνεχής και γνησίως φθίνουσα. Άρα:
    \[ f(\Delta_2) = \left( \lim_{x \to -1^-} f(x), f(-\sqrt{3}) \right] = \left( -\infty, -\frac{3\sqrt{3}}{2} \right] \]
    
    \item Στο διάστημα $\Delta_3 = (-1, 1)$, η $f$ είναι συνεχής. Επειδή $f'(x) \leq 0$ με μηδενισμό μόνο στο $x=0$, η $f$ είναι γνησίως φθίνουσα σε όλο το $\Delta_3$. Άρα:
    \[ f(\Delta_3) = \left( \lim_{x \to 1^-} f(x), \lim_{x \to -1^+} f(x) \right) = (-\infty, +\infty) = \mathbb{R} \]
    
    \item Στο διάστημα $\Delta_4 = (1, \sqrt{3}]$, η $f$ είναι συνεχής και γνησίως φθίνουσα. Άρα:
    \[ f(\Delta_4) = \left[ f(\sqrt{3}), \lim_{x \to 1^+} f(x) \right) = \left[ \frac{3\sqrt{3}}{2}, +\infty \right) \]
    
    \item Στο διάστημα $\Delta_5 = [\sqrt{3}, +\infty)$, η $f$ είναι συνεχής και γνησίως αύξουσα. Άρα:
    \[ f(\Delta_5) = \left[ f(\sqrt{3}), \lim_{x \to +\infty} f(x) \right) = \left[ \frac{3\sqrt{3}}{2}, +\infty \right) \]
αφού \[\lim\limits_{x\to+\infty}{f(x)}=\lim\limits_{x\to+\infty}{\frac{x^3}{x^2-1}}=\lim\limits_{x\to+\infty}{\frac{x^3}{x^2}}=\lim\limits_{x\to+\infty}{x}=+\infty\]
\end{itemize}

Το συνολικό σύνολο τιμών της $f$ είναι η ένωση των παραπάνω εικόνων:
\[ f(D_f) = f(\Delta_1) \cup f(\Delta_2) \cup f(\Delta_3) \cup f(\Delta_4) \cup f(\Delta_5) = \mathbb{R} \]

Η εξίσωση $f(x) = \lambda$ ισοδυναμεί με την εύρεση των σημείων τομής της γραφικής παράστασης $C_f$ με την οριζόντια ευθεία $y = \lambda$. Διακρίνουμε τις εξής περιπτώσεις ανάλογα με την τιμή του $\lambda \in \mathbb{R}$:

\begin{itemize}
    \item \textbf{Αν $\lambda < -\frac{3\sqrt{3}}{2}$}: 
    Το $\lambda$ ανήκει στα $f(\Delta_1), f(\Delta_2)$ και $f(\Delta_3)$. Λόγω γνησίας μονοτονίας σε κάθε διάστημα, η εξίσωση έχει ακριβώς \textbf{3 ρίζες} (μία σε κάθε ένα από τα $\Delta_1, \Delta_2, \Delta_3$).
    
    \item \textbf{Αν $\lambda = -\frac{3\sqrt{3}}{2}$}: 
    Το $\lambda$ ανήκει στο $f(\Delta_1)$ και $f(\Delta_2)$ στο κοινό τους άκρο $x = -\sqrt{3}$ (το οποίο είναι διπλή ρίζα/ακρότατο), και στο $f(\Delta_3)$. Άρα η εξίσωση έχει ακριβώς \textbf{2 ρίζες}.
    
    \item \textbf{Αν $-\frac{3\sqrt{3}}{2} < \lambda < \frac{3\sqrt{3}}{2}$}: 
    Το $\lambda$ ανήκει μόνο στο $f(\Delta_3)$ (καθώς είναι πάνω από το τοπικό μέγιστο και κάτω από το τοπικό ελάχιστο). Άρα η εξίσωση έχει ακριβώς \textbf{1 ρίζα}.
    
    \item \textbf{Αν $\lambda = \frac{3\sqrt{3}}{2}$}: 
    Αντίστοιχα με το τοπικό μέγιστο, το $\lambda$ πιάνει το τοπικό ελάχιστο στο $x = \sqrt{3}$ και ένα σημείο στο $\Delta_3$. Άρα η εξίσωση έχει ακριβώς \textbf{2 ρίζες}.
    
    \item \textbf{Αν $\lambda > \frac{3\sqrt{3}}{2}$}: 
    Το $\lambda$ ανήκει στα $f(\Delta_3), f(\Delta_4)$ και $f(\Delta_5)$. Λόγω γνησίας μονοτονίας, η εξίσωση έχει ακριβώς \textbf{3 ρίζες}.
\end{itemize}
\end{erwthma}
\end{thema}
