\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^3 + x - 1$.
\begin{erwthma}
\item Η $f$ είναι συνεχής στο $[0,1]$ και παραγωγίσιμη στο $(0,1)$, ως πολυωνυμική.
Από το θεώρημα μέσης τιμής υπάρχει $\xi\in(0,1)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(1)-f(0)}{1-0}.
\]
Επειδή $f(1)=1$ και $f(0)=-1$, έχουμε
\[
f'(\xi)=2.
\]
Όμως $f'(x)=3x^2+1$, άρα
\[
3\xi^2+1=2 \Leftrightarrow \xi^2=\frac{1}{3}
\Leftrightarrow \xi=\frac{1}{\sqrt{3}},
\]
αφού $\xi\in(0,1)$. Επομένως η ζητούμενη τετμημένη είναι
$\xi=\dfrac{1}{\sqrt{3}}$.

\item Η $f$ είναι συνεχής στο $[0,1]$ και
\[
f(0)=-1<0,\qquad f(1)=1>0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα
$x_0\in(0,1)$ τέτοιο ώστε $f(x_0)=0$.

\item Για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=3x^2+1>0.
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, οπότε η εξίσωση
$f(x)=0$ έχει το πολύ μία ρίζα στο $\mathbb{R}$. Επειδή από το προηγούμενο
ερώτημα έχει μία ρίζα στο $(0,1)$, αυτή είναι μοναδική στο $\mathbb{R}$.

\item Στο διάστημα $[1,2]$ έχουμε $f(1)=1>0$ και, επειδή η $f$ είναι γνησίως
αύξουσα, ισχύει $f(x)\ge f(1)>0$. Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\int_1^2 f(x)\,\d x
=\int_1^2 (x^3+x-1)\,\d x\\
&=\left[\frac{x^4}{4}+\frac{x^2}{2}-x\right]_1^2
=4-\left(\frac14+\frac12-1\right)
=\frac{17}{4}.
\end{align*}
\end{erwthma}
\end{thema}
