\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^3 + x - 1$ με $x \in \mathbb{R}$.
\begin{erwthma}
\item Η συνάρτηση $f$ είναι πολυωνυμική, άρα παραγωγίσιμη στο $\mathbb{R}$ με:
\[ f'(x) = 3x^2 + 1 \]
Επειδή $3x^2 \ge 0 \Rightarrow 3x^2+1 \ge 1 > 0$, έχουμε $f'(x) > 0$ για κάθε $x \in \mathbb{R}$.
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το $\mathbb{R}$.

\item Θεωρούμε την $f$ στο κλειστό διάστημα $[0, 1]$.
Η $f$ είναι συνεχής στο $[0, 1]$.
Υπολογίζουμε τις τιμές στα άκρα:
- $f(0) = 0^3 + 0 - 1 = -1 < 0$.
- $f(1) = 1^3 + 1 - 1 = 1 > 0$.
Αφού $f(0)f(1) < 0$, σύμφωνα με το Θεώρημα Bolzano έχει τουλάχιστον μία ρίζα $x_0 \in (0, 1)$ ώστε $f(x_0)=0$.
Δεδομένου ότι η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, η ρίζα αυτή είναι \textbf{ακριβώς μία}.

\item Εξετάζουμε το πρόσημο της $f(x)$:
- Για $x < x_0$, επειδή η $f$ είναι γνησίως αύξουσα, ισχύει $f(x) < f(x_0) \Rightarrow f(x) < 0$.
- Για $x > x_0$, ομοίως ισχύει $f(x) > f(x_0) \Rightarrow f(x) > 0$.

\item Υπολογίζουμε το ζητούμενο όριο: $L = \lim_{x\to+\infty} \left[ f(x) \hm \left( \frac{1}{f(x)} \right) \right]$.
Κατ' αρχάς, $\lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} (x^3 + x - 1) = \lim_{x\to+\infty} x^3 = +\infty$.
Θέτουμε $u = \frac{1}{f(x)}$. Καθώς το $x \to +\infty$, το $f(x) \to +\infty$, άρα $u \to 0^+$.
Τότε το $f(x) = \frac{1}{u}$ και το όριο μετατρέπεται στο εξής:
\[ L = \lim_{u\to0^+} \left[ \frac{1}{u} \hm u \right] = \lim_{u\to0^+} \frac{\hm u}{u} = 1 \]
\end{erwthma}
\end{thema}
