\begin{thema}{Β}
Στο παρακάτω αναβαθμισμένο σχήμα τονίζονται οι ασύμπτωτες, τα σημαντικά σημεία, και η φθίνουσα μονοτονία κάθε κλάδου.

\begin{erwthma}

\item Χρησιμοποιούμε τη γραφική παράσταση και τη γνωστή μας θέση των ασύμπτωτων:
\begin{itemize}
    \item \textbf{$\lim\limits_{x \to 1^-} f(x)$:} Καθώς $x$ πλησιάζει το $1$ από αριστερά, ο αριστερός κλάδος βυθίζεται. Άρα: $\lim\limits_{x \to 1^-} f(x) = -\infty$.
    \item \textbf{$\lim\limits_{x \to 1^+} f(x)$:} Καθώς $x$ πλησιάζει το $1$ από δεξιά, ο δεξιός κλάδος εκτοξεύεται. Άρα: $\lim\limits_{x \to 1^+} f(x) = +\infty$.
    \item \textbf{$\lim\limits_{x \to +\infty} f(x)$:} Ο δεξιός κλάδος πλησιάζει ασυμπτωτικά τη γραμμή $y = 2$ από πάνω. Άρα: $\lim\limits_{x \to +\infty} f(x) = 2$.
\end{itemize}

\item Οι ασύμπτωτες προκύπτουν κατευθείαν από τα παραπάνω:
\begin{itemize}
    \item \textbf{Κατακόρυφη ασύμπτωτη $x = 1$:} Επειδή $\lim\limits_{x \to 1^{\pm}} f(x) = \pm\infty$.
    \item \textbf{Οριζόντια ασύμπτωτη $y = 2$:} Επειδή $\lim\limits_{x \to +\infty} f(x) = 2$.
    Παρομοίως, εξετάζοντας τον αριστερό κλάδο: $\lim\limits_{x \to -\infty} f(x) = 2$ (η καμπύλη πλησιάζει τη γραμμή $y=2$ από κάτω). Η ίδια ευθεία είναι ασύμπτωτη και στο $-\infty$.
\end{itemize}

\item Μελετώντας τη μονοτονία σε κάθε κλάδο:
\begin{itemize}
    \item \textbf{Αριστερός κλάδος $(-\infty, 1)$:} Η καμπύλη ξεκινά από τα "χαμηλά" (κοντά στο $y=2$, λίγο κάτω) και, καθώς $x$ αυξάνεται προς το 1, η τεταγμένη μειώνεται δραματικά (προς $-\infty$). Συνεπώς, η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty, 1)$.
    \item \textbf{Δεξιός κλάδος $(1, +\infty)$:} Η καμπύλη ξεκινά από πολύ ψηλά ($+\infty$) και, καθώς $x$ αυξάνεται, πέφτει ομαλά πλησιάζοντας ασυμπτωτικά το $y=2$ (από πάνω). Συνεπώς, η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(1, +\infty)$.
\end{itemize}

\item Εξετάζουμε τη συνέχεια:
\begin{itemize}
    \item \textbf{Στο $x_0 = 0$:} Το σημείο $0$ ανήκει στο πεδίο ορισμού ($0 \ne 1$). Εφόσον η $f$ δίνεται ως ρητή (ή αλγεβρική), ορίζεται με ενιαίο τύπο στο $(-\infty, 1)$. Τόσο η τιμή $f(0) = 1$ υπάρχει, όσο και το $\lim\limits_{x \to 0} f(x) = f(0) = 1$ (αφού δεν υπάρχει κατάτμηση πεδίου ορισμού γύρω στο 0). Συνεπώς, η $f$ \textbf{είναι συνεχής} στο $x_0 = 0$.
    \item \textbf{Στο $x_0 = 1$:} Η τιμή $x = 1$ \textbf{δεν ανήκει} στο πεδίο ορισμού ($\varDelta_f = \mathbb{R} \setminus \{1\}$). Συνεπώς, δεν τίθεται καν ζήτημα συνέχειας στο $x_0 = 1$: η $f$ \textbf{δεν ορίζεται} εκεί. Η ασυνέχεια στο 1 δεν είναι "αποκαταστάσιμη" (αφού τα πλευρικά όρια είναι $\pm\infty$).
\end{itemize}

\end{erwthma}
\end{thema}
