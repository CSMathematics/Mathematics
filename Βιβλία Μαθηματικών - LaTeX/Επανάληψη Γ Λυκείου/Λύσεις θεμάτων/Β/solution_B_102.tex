\begin{thema}{Β}
Από το σχήμα φαίνεται ότι η $C_f$ έχει κατακόρυφη ασύμπτωτη την ευθεία $x=1$ και οριζόντια ασύμπτωτη την ευθεία $y=2$. Στα επόμενα ερωτήματα χρησιμοποιούμε και τον τύπο
\[
f(x)=\frac{2x-1}{x-1}=2+\frac{1}{x-1},\quad x\neq 1.
\]

\begin{erwthma}
\item Από το σχήμα η κατακόρυφη ασύμπτωτη είναι η ευθεία $x=1$. Τα αντίστοιχα πλευρικά όρια είναι
\[
\lim_{x\to 1^-}f(x)=-\infty
\quad\text{και}\quad
\lim_{x\to 1^+}f(x)=+\infty.
\]

Επίσης, η οριζόντια ασύμπτωτη είναι η ευθεία $y=2$. Τα αντίστοιχα όρια είναι
\[
\lim_{x\to -\infty}f(x)=2
\quad\text{και}\quad
\lim_{x\to +\infty}f(x)=2.
\]

\item Το σημείο τομής των ασύμπτωτων είναι το $K(1,2)$. Θα δείξουμε ότι καμία εφαπτομένη της $C_f$ δεν διέρχεται από το $K$.

Για κάθε $x\neq 1$ είναι
\[
f'(x)=\left(2+\frac{1}{x-1}\right)'=-\frac{1}{(x-1)^2}.
\]
Έστω $x_0\neq 1$. Η εφαπτομένη της $C_f$ στο σημείο $M(x_0,f(x_0))$ έχει εξίσωση
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]
Αν η εφαπτομένη περνούσε από το $K(1,2)$, τότε θα ίσχυε
\[
2-f(x_0)=f'(x_0)(1-x_0).
\]
Όμως
\[
f(x_0)-2=\frac{1}{x_0-1}\Rightarrow 2-f(x_0)=-\frac{1}{x_0-1},
\]
ενώ
\[
f'(x_0)(1-x_0)
=-\frac{1}{(x_0-1)^2}(1-x_0)
=\frac{1}{x_0-1}.
\]
Άρα θα έπρεπε
\[
-\frac{1}{x_0-1}=\frac{1}{x_0-1},
\]
που είναι αδύνατο, αφού $x_0\neq 1$. Επομένως καμία εφαπτομένη της $C_f$ δεν διέρχεται από το σημείο τομής των ασύμπτωτων.

\item Για $x\neq 1$ έχουμε
\[
f(x)-2=\frac{1}{x-1}.
\]
Επομένως
\[
\frac{1}{f(x)-2}=\frac{1}{\frac{1}{x-1}}=x-1.
\]
Άρα
\[
\lim_{x\to 1}\frac{1}{f(x)-2}
=\lim_{x\to 1}(x-1)=0.
\]
Συνεπώς το όριο υπάρχει και είναι ίσο με $0$.

\item Για να ορίζεται η $f\circ f$, πρέπει
\[
x\in D_f
\quad\text{και}\quad
f(x)\in D_f.
\]
Επειδή $D_f=\mathbb{R}-\{1\}$, πρέπει $x\neq 1$ και $f(x)\neq 1$.

Λύνουμε:
\[
f(x)=1
\Leftrightarrow
\frac{2x-1}{x-1}=1
\Leftrightarrow
2x-1=x-1
\Leftrightarrow
x=0.
\]
Άρα πρέπει να εξαιρέσουμε και το $x=0$. Επομένως
\[
D_{f\circ f}=\mathbb{R}-\{0,1\}.
\]
\end{erwthma}
\end{thema}
