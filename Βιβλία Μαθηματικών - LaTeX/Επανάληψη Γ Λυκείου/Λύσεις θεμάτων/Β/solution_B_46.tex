\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{\sqrt{x+1}-1}{x}$ για $x \ge -1, x \ne 0$.
\begin{erwthma}
\item Για τον υπολογισμό του ορίου στο 0, πολλαπλασιάζουμε με τη συζυγή παράσταση:
\begin{align*}
\lim_{x\to0} f(x)
&= \lim_{x\to0} \frac{\sqrt{x+1}-1}{x}
= \lim_{x\to0} \frac{(\sqrt{x+1}-1)(\sqrt{x+1}+1)}{x(\sqrt{x+1}+1)}\\
&= \lim_{x\to0} \frac{x}{x(\sqrt{x+1}+1)}
= \lim_{x\to0} \frac{1}{\sqrt{x+1}+1}
= \frac{1}{2}.
\end{align*}

\item Αφού η $g$ είναι συνεχής στο $0$, πρέπει:
\[ g(0)=\lim_{x\to0} f(x). \]
Όμως $g(0)=a-\frac{1}{2}$ και από το προηγούμενο ερώτημα $\lim\limits_{x\to0} f(x)=\frac{1}{2}$. Άρα:
\[ a-\frac{1}{2}=\frac{1}{2}\Leftrightarrow a=1. \]
Επομένως η συνεχής επέκταση της $f$ είναι:
\[ g(x)=\begin{cases}
\dfrac{\sqrt{x+1}-1}{x}, & x\in D_f\\[4pt]
\dfrac{1}{2}, & x=0
\end{cases} \]

\item Εξετάζουμε τη συνεχή συνάρτηση $g$ στο κλειστό διάστημα $[0, 3]$.
Υπολογίζουμε:
\[ g(0)=\frac{1}{2}, \qquad g(3)=\frac{\sqrt{4}-1}{3}=\frac{1}{3}. \]
Επειδή
\[ \frac{1}{3}<\frac{2}{5}<\frac{1}{2}, \]
από το Θεώρημα Ενδιάμεσων Τιμών υπάρχει τουλάχιστον ένα $\xi\in(0,3)$ τέτοιο ώστε
\[ g(\xi)=\frac{2}{5}. \]

\item Για $x\ne0$ έχουμε:
\[ f(x)=\frac{\sqrt{x+1}-1}{x}\cdot\frac{\sqrt{x+1}+1}{\sqrt{x+1}+1}
=\frac{1}{\sqrt{x+1}+1}. \]
Ο ίδιος τύπος ισχύει και για $x=0$, αφού $g(0)=\frac{1}{2}$. Άρα:
\[ g(x)=\frac{1}{\sqrt{x+1}+1}, \quad x\in[-1,+\infty). \]
Η συνάρτηση $x\mapsto \sqrt{x+1}+1$ είναι γνησίως αύξουσα στο $[-1,+\infty)$ και παίρνει θετικές τιμές. Επομένως η $g$ είναι γνησίως φθίνουσα στο $[-1,+\infty)$.
Άρα η $g$ είναι ένα προς ένα και συνεπώς αντιστρέφεται στο πεδίο ορισμού της.
\end{erwthma}
\end{thema}
