\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=axe^x+\beta,\quad x\in\mathbb{R}.
\]

\begin{erwthma}
\item Επειδή
\[
\lim_{x\to-\infty}xe^x
=\lim_{x\to-\infty}\frac{x}{e^{-x}}
\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to-\infty}\frac{1}{-e^{-x}}=0,
\]
έχουμε
\[
\lim_{x\to-\infty}f(x)
=\lim_{x\to-\infty}\left(axe^x+\beta\right)
=\beta.
\]
Αφού το όριο είναι ίσο με $2$, προκύπτει
\[
\beta=2.
\]
Επίσης, επειδή η $C_f$ διέρχεται από το σημείο $A(1,e+2)$, έχουμε
\[
f(1)=e+2\Leftrightarrow ae+2=e+2\Leftrightarrow a=1.
\]
Άρα
\[
f(x)=xe^x+2.
\]

\item Παραγωγίζουμε δύο φορές:
\[
f'(x)=e^x+xe^x=e^x(x+1)
\quad\text{και}\quad
f''(x)=e^x(x+1)+e^x=e^x(x+2).
\]
Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f''$ είναι το πρόσημο του $x+2$:
\begin{itemize}
\item $f''(x)<0\Leftrightarrow x<-2$,
\item $f''(x)=0\Leftrightarrow x=-2$,
\item $f''(x)>0\Leftrightarrow x>-2$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:

Άρα η $f$ είναι \textbf{κοίλη} στο $(-\infty,-2]$ και \textbf{κυρτή} στο $[-2,+\infty)$. Επειδή η κυρτότητα αλλάζει στο $x=-2$, η $C_f$ έχει σημείο καμπής το
\[
M(-2,f(-2))\equiv M\left(-2,2-\frac{2}{e^2}\right).
\]

\item Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)-3=xe^x-1,\quad x\in[0,1].
\]
Η $h$ είναι συνεχής στο $[0,1]$ και
\[
h(0)=-1<0,\quad h(1)=e-1>0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα $x_0\in(0,1)$ τέτοιο ώστε
\[
h(x_0)=0\Leftrightarrow f(x_0)=3.
\]
Επομένως η εξίσωση $f(x)=3$ έχει τουλάχιστον μία ρίζα στο $(0,1)$.

\item Για $x\in[0,1]$ ισχύει $xe^x\geq0$, άρα $f(x)=xe^x+2>0$. Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\int_0^1 f(x)\d x
=\int_0^1\left(xe^x+2\right)\d x\\
&=\left[(x-1)e^x+2x\right]_0^1
=2-(-1)=3.
\end{align*}
\end{erwthma}
\end{thema}
