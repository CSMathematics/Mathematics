\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=\frac{x+a}{\beta x-2}.
\]

\begin{erwthma}
\item Η $C_f$ έχει κατακόρυφη ασύμπτωτη την ευθεία $x=2$, άρα ο παρονομαστής μηδενίζεται για $x=2$:
\[
2\beta-2=0\Leftrightarrow \beta=1.
\]
Με $\beta=1$ έχουμε
\[
f(x)=\frac{x+a}{x-2},\quad x\neq2.
\]
Αφού το σημείο $M(-2,0)$ ανήκει στη $C_f$, ισχύει
\[
f(-2)=0\Leftrightarrow \frac{-2+a}{-4}=0\Leftrightarrow a=2.
\]
Επομένως
\[
f(x)=\frac{x+2}{x-2}.
\]

\item Έστω $x_1,x_2\in\mathbb{R}\setminus\{2\}$ με $f(x_1)=f(x_2)$. Τότε
\begin{align*}
\frac{x_1+2}{x_1-2}=\frac{x_2+2}{x_2-2}
&\Leftrightarrow (x_1+2)(x_2-2)=(x_2+2)(x_1-2)\\
&\Leftrightarrow -2x_1+2x_2=2x_1-2x_2\\
&\Leftrightarrow x_1=x_2.
\end{align*}
Άρα η $f$ είναι $1-1$, επομένως αντιστρέφεται.

Για να βρούμε την αντίστροφη, θέτουμε $y=f(x)$:
\[
y=\frac{x+2}{x-2}
\Leftrightarrow y(x-2)=x+2
\Leftrightarrow x(y-1)=2y+2
\Leftrightarrow x=\frac{2y+2}{y-1}.
\]
Άρα
\[
f^{-1}(x)=\frac{2x+2}{x-1},\quad x\in\mathbb{R}\setminus\{1\}.
\]

\item Για $x\neq2$ έχουμε
\[
f(x)=2x
\Leftrightarrow \frac{x+2}{x-2}=2x
\Leftrightarrow x+2=2x(x-2)
\Leftrightarrow 2x^2-5x-2=0.
\]
Η διακρίνουσα είναι
\[
\Delta=(-5)^2-4\cdot2\cdot(-2)=41>0.
\]
Άρα η εξίσωση έχει δύο πραγματικές και άνισες λύσεις. Καμία από αυτές δεν είναι το $2$, αφού το $2$ δεν μηδενίζει το τριώνυμο $2x^2-5x-2$. Επομένως η εξίσωση $f(x)=2x$ έχει ακριβώς δύο λύσεις.

\item Στο διάστημα $[0,1]$ είναι $x+2>0$ και $x-2<0$, άρα
\[
f(x)=\frac{x+2}{x-2}<0.
\]
Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\int_0^1 |f(x)|\d x
=-\int_0^1\frac{x+2}{x-2}\d x\\
&=-\int_0^1\left(1+\frac{4}{x-2}\right)\d x
=-\left[x+4\ln|x-2|\right]_0^1\\
&=-\left(1-4\ln2\right)=4\ln2-1.
\end{align*}
\end{erwthma}
\end{thema}
