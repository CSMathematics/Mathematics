\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=\frac{ax^2+1}{x}=ax+\frac{1}{x},\quad x\neq0.
\]

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο πεδίο ορισμού της και
\[
f'(x)=a-\frac{1}{x^2}.
\]
Επειδή η $f$ παρουσιάζει τοπικό ακρότατο στο εσωτερικό σημείο $x_0=1$, από το θεώρημα \eng{Fermat} έχουμε
\[
f'(1)=0\Leftrightarrow a-1=0\Leftrightarrow a=1.
\]

\item Για $a=1$ είναι
\[
f(x)=x+\frac{1}{x}
\quad\text{και}\quad
f'(x)=1-\frac{1}{x^2}=\frac{x^2-1}{x^2}.
\]
Επειδή $x^2>0$ για κάθε $x\neq0$, το πρόσημο της $f'$ καθορίζεται από το $x^2-1$:
\begin{itemize}
\item $f'(x)>0\Leftrightarrow x<-1$ ή $x>1$,
\item $f'(x)=0\Leftrightarrow x=-1$ ή $x=1$,
\item $f'(x)<0\Leftrightarrow -1<x<0$ ή $0<x<1$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Άρα η $f$ παρουσιάζει:
\begin{itemize}
\item τοπικό μέγιστο στο $x=-1$, με τιμή $f(-1)=-2$,
\item τοπικό ελάχιστο στο $x=1$, με τιμή $f(1)=2$.
\end{itemize}
Η $f$ δεν έχει ολικά ακρότατα στο πεδίο ορισμού της, αφού
\[
\lim_{x\to0^+}f(x)=+\infty
\quad\text{και}\quad
\lim_{x\to0^-}f(x)=-\infty.
\]

\item Η $f$ είναι συνεχής στο $[1,2]$ και παραγωγίσιμη στο $(1,2)$. Από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(1,2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(2)-f(1)}{2-1}
=\frac{\frac{5}{2}-2}{1}
=\frac{1}{2}.
\]

\item Για την κατακόρυφη ασύμπτωτη εξετάζουμε το σημείο $x=0$, που δεν ανήκει στο πεδίο ορισμού. Έχουμε
\[
\lim_{x\to0^+}f(x)=+\infty
\quad\text{και}\quad
\lim_{x\to0^-}f(x)=-\infty.
\]
Άρα η ευθεία
\[
x=0
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Επίσης
\[
f(x)-x=\frac{1}{x}\xrightarrow[x\to\pm\infty]{}0.
\]
Επομένως η ευθεία
\[
y=x
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$ και στο $-\infty$.
\end{erwthma}
\end{thema}
