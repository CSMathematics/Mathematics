\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=
\begin{cdcases}
\dfrac{\ln x}{x-1}, & x>0,\ x\neq1\\
a, & x=1
\end{cdcases}
\]
και είναι συνεχής στο $x_0=1$.

\begin{erwthma}
\item Αφού η $f$ είναι συνεχής στο $1$, πρέπει
\[
a=f(1)=\lim_{x\to1}\frac{\ln x}{x-1}
\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{1}{x}=1.
\]
Άρα
\[
a=1.
\]

\item Εξετάζουμε το άκρο $0$ του πεδίου ορισμού. Έχουμε
\[
\lim_{x\to0^+}\frac{\ln x}{x-1}=+\infty,
\]
διότι $\ln x\to-\infty$ και $x-1\to-1$. Επομένως η ευθεία
\[
x=0
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Στο $x=1$ δεν υπάρχει κατακόρυφη ασύμπτωτη, αφού
\[
\lim_{x\to1}\frac{\ln x}{x-1}=1.
\]

\item Με $a=1$ έχουμε $f(1)=1$. Εξετάζουμε το όριο του λόγου μεταβολής:
\begin{align*}
\lim_{x\to1}\frac{f(x)-f(1)}{x-1}
&=\lim_{x\to1}\frac{\frac{\ln x}{x-1}-1}{x-1}\\
&=\lim_{x\to1}\frac{\ln x-x+1}{(x-1)^2}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{\frac{1}{x}-1}{2(x-1)}
=\lim_{x\to1}\frac{-(x-1)}{2x(x-1)}
=-\frac{1}{2}.
\end{align*}
Άρα η $f$ είναι παραγωγίσιμη στο $x_0=1$ και
\[
f'(1)=-\frac{1}{2}.
\]

\item Για $x\neq1$ είναι
\[
f'(x)=\left(\frac{\ln x}{x-1}\right)'
=\frac{\frac{1}{x}(x-1)-\ln x}{(x-1)^2}
=\frac{x-1-x\ln x}{x(x-1)^2}.
\]
Θεωρούμε τη συνάρτηση
\[
g(x)=x-1-x\ln x,\quad x>0.
\]
Τότε
\[
g'(x)=1-(\ln x+1)=-\ln x.
\]
Για $0<x<1$ έχουμε $\ln x<0$, άρα $g'(x)>0$. Επομένως η $g$ είναι γνησίως αύξουσα στο $(0,1]$ και για κάθε $x\in(0,1)$ ισχύει
\[
g(x)<g(1)=0.
\]
Στο $(0,1)$ ο παρονομαστής $x(x-1)^2$ είναι θετικός, ενώ ο αριθμητής $g(x)$ είναι αρνητικός. Άρα
\[
f'(x)<0,\quad 0<x<1.
\]
Συνεπώς η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0,1)$.
\end{erwthma}
\end{thema}
