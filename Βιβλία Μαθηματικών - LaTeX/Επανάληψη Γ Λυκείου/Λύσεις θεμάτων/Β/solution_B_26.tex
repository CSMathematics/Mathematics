\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x-1}{x+1}$.

\begin{erwthma}
\item Το πεδίο ορισμού της $f$ απαιτεί $x+1 \ne 0 \Leftrightarrow x \ne -1$. Άρα $D_f = \mathbb{R} - \{-1\}$.
Έστω $x_1, x_2 \in D_f$ με $f(x_1) = f(x_2)$. Τότε:
\[ \frac{x_1-1}{x_1+1} = \frac{x_2-1}{x_2+1} \Leftrightarrow (x_1-1)(x_2+1) = (x_2-1)(x_1+1) \]
\[ \Leftrightarrow x_1 x_2 + x_1 - x_2 - 1 = x_1 x_2 + x_2 - x_1 - 1 \Leftrightarrow 2x_1 = 2x_2 \Leftrightarrow x_1 = x_2 \]
Άρα η συνάρτηση $f$ είναι $1-1$, συνεπώς αντιστρέφεται.

\item Θέτουμε $y = f(x)$ και λύνουμε ως προς $x$:
\[ y = \frac{x-1}{x+1} \Leftrightarrow y(x+1) = x-1 \Leftrightarrow yx + y = x - 1 \Leftrightarrow x(y-1) = -1 - y \]
Για τον περιορισμό πρέπει $y-1 \ne 0 \Leftrightarrow y \ne 1$. Το πεδίο ορισμού της αντίστροφης είναι $D_{f^{-1}} = \mathbb{R} - \{1\}$.
Για $y \ne 1$, έχουμε $x = \frac{-y-1}{y-1} = \frac{y+1}{1-y}$. Επειδή πρέπει επιπλέον $x \ne -1$, ελέγχουμε αν υπάρχει τιμή του $y$ ώστε $\frac{y+1}{1-y} = -1 \Leftrightarrow y+1 = -1+y \Leftrightarrow 1 = -1$ (αδύνατον).
Άρα ο τύπος της αντίστροφης είναι $f^{-1}(x) = \frac{x+1}{1-x}$, για $x \ne 1$.

\item Το πεδίο ορισμού της συνάρτησης $h = f \circ f$ είναι:
\[ D_h = \{x \in D_f \mid f(x) \in D_f\} = \{x \ne -1 \mid f(x) \ne -1\} \]
Λύνουμε τον περιορισμό $f(x) \ne -1$:
\[ \frac{x-1}{x+1} \ne -1 \Leftrightarrow x-1 \ne -x-1 \Leftrightarrow 2x \ne 0 \Leftrightarrow x \ne 0 \]
Συνεπώς το πεδίο ορισμού είναι το $D_h = \mathbb{R} - \{-1, 0\}$.
Για τον τύπο της ισχύει:
\[ h(x) = f(f(x)) = \frac{f(x)-1}{f(x)+1} = \frac{\frac{x-1}{x+1} - 1}{\frac{x-1}{x+1} + 1} = \frac{\frac{x-1-x-1}{x+1}}{\frac{x-1+x+1}{x+1}} = \frac{-2}{2x} = -\frac{1}{x} \]
Άρα $h(x) = -\frac{1}{x}$ για $x \in \mathbb{R} - \{-1, 0\}$.

\item Ορίζουμε τη συνάρτηση $K(x) = x \cdot f\left(\frac{1}{x}\right)$ για $x \to +\infty$.
Ο τύπος της για $x$ αρκετά μεγάλο (π.χ. $x > 0$) είναι:
\[ K(x) = x \cdot \frac{\frac{1}{x}-1}{\frac{1}{x}+1} = x \cdot \frac{\frac{1-x}{x}}{\frac{1+x}{x}} = x \cdot \frac{1-x}{1+x} = \frac{x-x^2}{1+x} \]
Το ζητούμενο όριο υπολογίζεται ως έξης (με όριο ρητής):
\[ \lim_{x\to+\infty} K(x) = \lim_{x\to+\infty} \frac{-x^2+x}{x+1} = \lim_{x\to+\infty} \frac{-x^2}{x} = \lim_{x\to+\infty} (-x) = -\infty \]
\end{erwthma}
\end{thema}
