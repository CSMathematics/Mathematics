\begin{thema}{Β}
Στο παρακάτω αναβαθμισμένο σχήμα σημειώνονται τα σημεία τομής $f=g$ και τα κρίσιμα σημεία.
\begin{center}

  \draw[very thick, blue!70!black] (0,1) -- (2,3) -- (4,1) -- (5,2);
  \node[blue!70!black] at (1,3) {$C_f$};
  % g
  \draw[very thick, red!70!black] (0,0) -- (1,2) -- (2,4) -- (3,3) -- (5,1);

\begin{erwthma}

\item Υπολογίζουμε τις τιμές διαδοχικά:
\begin{itemize}
    \item $(f \circ g)(1) = f(g(1)) = f(2) = 3$.
    \item $(g \circ f)(2) = g(f(2)) = g(3) = 3$.
    \item $(f \circ f)(0) = f(f(0)) = f(1)$. Η $f$ στο $[0, 2]$ έχει κλίση $= 1$ και $f(0)=1$, άρα $f(x) = x+1$, οπότε $f(1) = 2$. Τελικά: $(f \circ f)(0) = 2$.
\end{itemize}

\item Πράξεις μεταξύ $f$ και $g$:
\begin{itemize}
    \item $(f+g)(2) = f(2) + g(2) = 3 + 4 = 7$.
    \item $(f \cdot g)(0) = f(0) \cdot g(0) = 1 \cdot 0 = 0$.
    \item $\frac{f}{g}(3) = \frac{f(3)}{g(3)}$. Η $f$ στο $[2, 4]$: κλίση $-1$, $f(2)=3$, οπότε $f(x) = -x+5$, άρα $f(3) = 2$. Και $g(3) = 3$. Οπότε $\frac{f}{g}(3) = \frac{2}{3}$.
\end{itemize}

\item Η $(g \circ f)(4) = g(f(4)) = g(1)$. Ελέγχουμε: $f(4) = 1$ (δίνεται), και $g(1) = 2$ (δίνεται). Οπότε ναι, ορίζεται, και $(g \circ f)(4) = \mathbf{2}$.

\item Η εξίσωση $f(x) = g(x)$ λύνεται γραφικά βρίσκοντας τα σημεία τομής των $C_f$ και $C_g$.
Εξετάζουμε τμηματικά:
\begin{itemize}
    \item Στο $[0, 1]$: $f(x) = x+1$ και $g(x) = 2x$. Θέτουμε $x+1 = 2x \Rightarrow x = 1$. Ελέγχουμε: $f(1) = 2$, $g(1) = 2$. \checkmark. Ρίζα: $x = 1$.
    \item Στο $[1, 2]$: $f(x) = x+1$ και $g(x) = 2x$. Θέτουμε $x+1 = 2x \Rightarrow x = 1$, ήδη βρέθηκε.
    \item Στο $[2, 3]$: $f(x) = -x+5$ και $g(x) = -x+6$ (κλίση $g$: $\frac{3-4}{3-2} = -1$, $g(2)=4 \Rightarrow g(x) = -x+6$). Θέτουμε: $-x+5 = -x+6 \Rightarrow 5 = 6$, αδύνατο. Δεν υπάρχει λύση στο $(2, 3)$.
    \item Στο $[3, 5]$: $f$ και $g$ αλλάζουν κλάδο. Ελέγχουμε αναλυτικά.
    
    $f$ στο $[4, 5]$: κλίση $1$, $f(4) = 1 \Rightarrow f(x) = x-3$. 
    $g$ στο $[3, 5]$: κλίση $\frac{1-3}{5-3} = -1$, $g(3)=3 \Rightarrow g(x) = -x+6$.
    Στο $[4, 5]$: $x-3 = -x+6 \Rightarrow 2x = 9 \Rightarrow x = 4{,}5$. $f(4{,}5) = 1{,}5$ και $g(4{,}5) = 1{,}5$. \checkmark
\end{itemize}
Οι λύσεις είναι: $x = 1$ και $x = 4{,}5$.

\end{erwthma}
\end{thema}
