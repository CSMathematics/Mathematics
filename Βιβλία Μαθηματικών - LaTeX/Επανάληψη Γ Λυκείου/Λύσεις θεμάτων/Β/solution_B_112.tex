\begin{thema}{Β}
Από το σχήμα διαβάζουμε τις τιμές των $f,g$ και, όπου χρειάζεται, βρίσκουμε τους τύπους των ευθύγραμμων τμημάτων.

\begin{erwthma}
\item Από το σχήμα έχουμε
\[
g(1)=2,\quad f(2)=3.
\]
Άρα
\[
(f\circ g)(1)=f(g(1))=f(2)=3.
\]
Επίσης
\[
f(2)=3,\quad g(3)=3,
\]
οπότε
\[
(g\circ f)(2)=g(f(2))=g(3)=3.
\]
Τέλος, επειδή $f(0)=1$, έχουμε
\[
(f\circ f)(0)=f(f(0))=f(1).
\]
Στο διάστημα $[0,2]$ η $f$ είναι ευθεία με κλίση
\[
m=\frac{3-1}{2-0}=1,
\]
άρα $f(x)=x+1$. Επομένως $f(1)=2$ και
\[
(f\circ f)(0)=2.
\]

\item Από το σχήμα έχουμε
\[
f(2)=3,\quad g(2)=4.
\]
Άρα
\[
(f+g)(2)=f(2)+g(2)=3+4=7.
\]
Επίσης
\[
f(0)=1,\quad g(0)=0,
\]
οπότε
\[
(f\cdot g)(0)=f(0)g(0)=1\cdot 0=0.
\]
Για το πηλίκο,
\[
\frac{f}{g}(3)=\frac{f(3)}{g(3)}.
\]
Στο διάστημα $[2,4]$ η $f$ είναι ευθεία με κλίση
\[
\frac{1-3}{4-2}=-1,
\]
άρα $f(x)=-x+5$. Επομένως $f(3)=2$. Από το σχήμα $g(3)=3$, οπότε
\[
\frac{f}{g}(3)=\frac{2}{3}.
\]

\item Η σύνθεση $(g\circ f)(4)=g(f(4))$ ορίζεται, εφόσον από το σχήμα $f(4)=1$ και το $1$ ανήκει στο πεδίο ορισμού της $g$.

Άρα
\[
g(f(4))=g(1)=2.
\]

\item Λύνουμε γραφικά την εξίσωση $f(x)=g(x)$ βρίσκοντας τις τομές των δύο γραφικών παραστάσεων. Για ακρίβεια, ελέγχουμε τα ευθύγραμμα τμήματα.

Στο $[0,2]$ έχουμε
\[
f(x)=x+1.
\]
Για τη $g$, στο $[0,2]$ από το σχήμα έχουμε $g(x)=2x$. Άρα
\[
x+1=2x \Rightarrow x=1.
\]

Στο $[2,3]$ έχουμε
\[
f(x)=-x+5,\quad g(x)=-x+6.
\]
Η εξίσωση
\[
-x+5=-x+6
\]
είναι αδύνατη.

Στο $[3,4]$ έχουμε
\[
f(x)=-x+5,\quad g(x)=-x+6.
\]
Πάλι δεν υπάρχει λύση.

Στο $[4,5]$ έχουμε
\[
f(x)=x-3,\quad g(x)=-x+6.
\]
Άρα
\[
x-3=-x+6 \Rightarrow 2x=9 \Rightarrow x=\frac{9}{2}.
\]
Επομένως οι λύσεις της εξίσωσης είναι
\[
x=1
\quad\text{και}\quad
x=\frac{9}{2}.
\]
\end{erwthma}
\end{thema}
