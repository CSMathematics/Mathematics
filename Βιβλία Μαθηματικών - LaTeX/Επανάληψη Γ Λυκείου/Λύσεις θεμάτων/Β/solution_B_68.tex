%=========== ΛΥΣΗ - 68ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Η συνάρτηση είναι
\[
f(x)=\frac{x^2+ax}{x-2},\quad x\neq 2.
\]

Παραγωγίζουμε:
\[
f'(x)=\frac{(2x+a)(x-2)-(x^2+ax)}{(x-2)^2}.
\]
Άρα
\[
f'(x)=\frac{2x^2-4x+ax-2a-x^2-ax}{(x-2)^2}
=\frac{x^2-4x-2a}{(x-2)^2}.
\]

Η εφαπτομένη στο σημείο με τετμημένη $x_0=1$ είναι παράλληλη στον άξονα $x'x$, άρα
\[
f'(1)=0.
\]
Επομένως,
\[
\frac{1-4-2a}{(1-2)^2}=0
\]
οπότε
\[
-3-2a=0.
\]
Άρα
\[
a=-\frac{3}{2}.
\]

Συνεπώς,
\[
f(x)=\frac{x^2-\frac{3}{2}x}{x-2}.
\]

\item Για $a=-\dfrac{3}{2}$ έχουμε
\[
f'(x)=\frac{x^2-4x+3}{(x-2)^2}
=\frac{(x-1)(x-3)}{(x-2)^2}.
\]

Επειδή
\[
(x-2)^2>0,\quad x\neq 2,
\]
το πρόσημο της $f'$ καθορίζεται από το γινόμενο
\[
(x-1)(x-3).
\]

Έχουμε:
\[
f'(x)>0 \Leftrightarrow x<1 \text{ ή } x>3,
\]
και
\[
f'(x)<0 \Leftrightarrow 1<x<2 \text{ ή } 2<x<3.
\]

Άρα η $f$ είναι γνησίως αύξουσα στα διαστήματα
\[
(-\infty,1]\quad \text{και}\quad [3,+\infty),
\]
και γνησίως φθίνουσα στα διαστήματα
\[
[1,2)\quad \text{και}\quad (2,3].
\]

Επομένως, η $f$ παρουσιάζει τοπικό μέγιστο στο $x=1$ και τοπικό ελάχιστο στο $x=3$.

\item Έστω $x_0\neq 2$ η τετμημένη ενός σημείου της $C_f$. Η εφαπτομένη της $C_f$ στο σημείο
\[
\bigl(x_0,f(x_0)\bigr)
\]
έχει εξίσωση
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]

Θέλουμε η εφαπτομένη αυτή να διέρχεται από το σημείο
\[
M(2,0).
\]
Άρα πρέπει
\[
0-f(x_0)=f'(x_0)(2-x_0).
\]

Για ευκολία θέτουμε $x_0=t$, με $t\neq 2$. Τότε
\[
f(t)=\frac{t^2-\frac{3}{2}t}{t-2}
\]
και
\[
f'(t)=\frac{(t-1)(t-3)}{(t-2)^2}.
\]

Η συνθήκη γίνεται
\[
-\frac{t^2-\frac{3}{2}t}{t-2}
=
\frac{(t-1)(t-3)}{(t-2)^2}(2-t).
\]
Επειδή
\[
2-t=-(t-2),
\]
παίρνουμε
\[
-\frac{t^2-\frac{3}{2}t}{t-2}
=-\frac{(t-1)(t-3)}{t-2}.
\]

Πολλαπλασιάζοντας με $-(t-2)$, όπου $t\neq 2$, προκύπτει
\[
t^2-\frac{3}{2}t=(t-1)(t-3).
\]
Άρα
\[
t^2-\frac{3}{2}t=t^2-4t+3.
\]
Επομένως,
\[
\frac{5}{2}t=3
\]
οπότε
\[
t=\frac{6}{5}.
\]

Η τιμή αυτή είναι διαφορετική από το $2$, άρα είναι αποδεκτή. Επειδή η εξίσωση που προέκυψε έχει μοναδική λύση, υπάρχει ακριβώς ένα σημείο της $C_f$ στο οποίο η εφαπτομένη διέρχεται από το $M(2,0)$.

\item Κάνουμε διαίρεση πολυωνύμων:
\[
\frac{x^2-\frac{3}{2}x}{x-2}
=x+\frac{1}{2}+\frac{1}{x-2}.
\]

Άρα
\[
f(x)-\left(x+\frac{1}{2}\right)=\frac{1}{x-2}.
\]

Επειδή
\[
\lim_{x\to+\infty}\frac{1}{x-2}=0
\]
και
\[
\lim_{x\to-\infty}\frac{1}{x-2}=0,
\]
η ευθεία
\[
y=x+\frac{1}{2}
\]
είναι πλάγια ασύμπτωτη της $C_f$ τόσο στο $+\infty$ όσο και στο $-\infty$.
\end{erwthma}
\end{thema}
