\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{2x^2-1}{x^2+1}$.

\begin{erwthma}
\item Ο παρονομαστής $x^2+1 > 0$ για κάθε $x \in \mathbb{R}$, άρα το πεδίο ορισμού της $f$ είναι $D_f = \mathbb{R}$.
Για να βρούμε τις οριζόντιες ασύμπτωτες, υπολογίζουμε τα όρια της $f(x)$ στο $\pm\infty$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \dfrac{2x^2-1}{x^2+1} = \lim_{x\to+\infty} \dfrac{2x^2}{x^2} = 2 \]
Ομοίως, $\lim\limits_{x\to-\infty} f(x) = 2$.
Άρα η ευθεία $y = 2$ είναι οριζόντια ασύμπτωτη της $C_f$ και στο $+\infty$ και στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο:
\[ f'(x) = \left( \dfrac{2x^2-1}{x^2+1} \right)' = \dfrac{(2x^2-1)'(x^2+1) - (2x^2-1)(x^2+1)'}{(x^2+1)^2} \]
\[ f'(x) = \dfrac{4x(x^2+1) - (2x^2-1)2x}{(x^2+1)^2} = \dfrac{4x^3+4x-4x^3+2x}{(x^2+1)^2} = \dfrac{6x}{(x^2+1)^2} \]
Η ρίζα της παραγώγου είναι $f'(x) = 0 \Leftrightarrow 6x = 0 \Leftrightarrow x = 0$.
Επειδή $(x^2+1)^2 > 0$ για κάθε $x$, το πρόσημο της $f'$ ταυτίζεται με το πρόσημο του $6x$. Άρα:
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Οπότε, η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$ και γνησίως αύξουσα στο $[0, +\infty)$.
Παρουσιάζει ολικό ελάχιστο στη θέση $x_0 = 0$, με ελάχιστη τιμή $f(0) = \dfrac{-1}{1} = -1$.

\item \textbf{Σημεία τομής με τους άξονες:}
Με τον άξονα $y'y$: Θέτουμε $x=0$ και προκύπτει $f(0) = -1$. Άρα τέμνει τον άξονα $y'y$ στο σημείο $A(0, -1)$.
Με τον άξονα $x'x$: Θέτουμε $f(x) = 0$.
\[ \dfrac{2x^2-1}{x^2+1} = 0 \Leftrightarrow 2x^2-1 = 0 \Leftrightarrow 2x^2 = 1 \Leftrightarrow x^2 = \dfrac{1}{2} \Leftrightarrow x = \pm\dfrac{1}{\sqrt{2}} = \pm\dfrac{\sqrt{2}}{2} \]
Άρα τέμνει τον $x'x$ στα σημεία $B_1\left(\frac{\sqrt{2}}{2}, 0\right)$ και $B_2\left(-\frac{\sqrt{2}}{2}, 0\right)$.
\textbf{Εξίσωση εφαπτομένης στο $(0, -1)$:}
Είναι $f'(0) = \dfrac{6\cdot 0}{(0^2+1)^2} = 0$. Η εξίσωση της εφαπτομένης στο $x_0 = 0$ είναι:
\[ y - f(0) = f'(0)(x-0) \Leftrightarrow y - (-1) = 0 \Leftrightarrow y = -1 \]

\item Το χωρίο περικλείεται από την $C_f$, την ευθεία $y=2$ και τις ευθείες $x=0, x=1$.
Για κάθε $x \in \mathbb{R}$, παρατηρούμε ότι: 
\[2 - f(x) = 2 - \dfrac{2x^2-1}{x^2+1} = \dfrac{2x^2+2-2x^2+1}{x^2+1} = \dfrac{3}{x^2+1} > 0\]
Οπότε $y = 2$ είναι πάνω από την $C_f$. Το εμβαδόν δίνεται από το ολοκλήρωμα:
\[ E = \int_0^1 (2 - f(x)) \d x = \int_0^1 \dfrac{3}{x^2+1} \d x \]
Θέτουμε $x = \ef u$ (οπότε $\d x = \dfrac{1}{\syn^2 u} \d u$).
Αφού $\dfrac{1}{\syn^2 u} = 1+\ef^2 u \Rightarrow \d x = (1+\ef^2 u) \d u$.
Τα όρια αλλάζουν ώς εξής: 
\begin{itemize}
\item Για $x=0 \Rightarrow \ef u = 0 \Rightarrow u=0$.
\item Για $x=1 \Rightarrow \ef u = 1 \Rightarrow u = \dfrac{\pi}{4}$.
\end{itemize}
Άρα το εμβαδόν υπολογίζεται:
\[ E = \int_0^{\pi/4} \dfrac{3}{\ef^2 u + 1}(1+\ef^2 u)\d u = \int_0^{\pi/4} 3 \d u = 3[u]_0^{\pi/4} = \dfrac{3\pi}{4} \]
\end{erwthma}
\end{thema}
