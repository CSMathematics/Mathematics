\begin{thema}{Β}
\begin{erwthma}
\item Από τη σχέση $2f(x)+f(-x)=3x^2+x$, θέτοντας όπου $x$ το $-x$, παίρνουμε
\[
f(x)+2f(-x)=3x^2-x.
\]
Άρα
\[
\begin{cases}
2f(x)+f(-x)=3x^2+x\\
f(x)+2f(-x)=3x^2-x
\end{cases}
\]
Πολλαπλασιάζουμε την πρώτη εξίσωση με $2$ και αφαιρούμε τη δεύτερη:
\[
4f(x)+2f(-x)-\bigl(f(x)+2f(-x)\bigr)=6x^2+2x-(3x^2-x),
\]
οπότε $3f(x)=3x^2+3x$ και τελικά $f(x)=x^2+x$.

\item Είναι $f'(x)=2x+1$, οπότε
\[
f'(x)=0\Leftrightarrow x=-\frac12.
\]
Επίσης $f'(x)<0$ για $x<-\frac12$ και $f'(x)>0$ για $x>-\frac12$. Άρα η $f$ είναι γνησίως φθίνουσα στο $\left(-\infty,-\frac12\right]$ και γνησίως αύξουσα στο $\left[-\frac12,+\infty\right)$. Επιπλέον $f''(x)=2>0$, άρα η $f$ είναι κυρτή στο $\mathbb{R}$.

\item Έχουμε $f(-1)=1-1=0$, άρα $A(-1,0)\in C_f$. Επίσης $f'(-1)=-1$, οπότε η εφαπτομένη στο $A$ είναι
\[
y-0=-1(x+1)\Leftrightarrow \varepsilon:\ y=-x-1.
\]
Για $x=0$ δίνει $y=-1$, άρα διέρχεται από το $B(0,-1)$.

\item Το χωρίο οριοθετείται από $C_f:\ y=x^2+x$, την $\varepsilon:\ y=-x-1$ και τον άξονα $y'y$, δηλαδή για $x\in[-1,0]$. Στο διάστημα αυτό
\[
f(x)-(-x-1)=x^2+2x+1=(x+1)^2\ge0.
\]
Άρα
\[
E=\int_{-1}^{0}(x+1)^2\,\mathrm{d}x
=\left[\frac{(x+1)^3}{3}\right]_{-1}^{0}
=\frac13.
\]
\end{erwthma}
\end{thema}
