%=========== ΛΥΣΗ - 76ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Το πλάτος του παραθύρου είναι $2x$, άρα η ακτίνα του ημικυκλίου είναι $x$.

Το μήκος του πλαισίου αποτελείται από:
\begin{itemize}
\item την κάτω πλευρά του ορθογωνίου, μήκους $2x$,
\item τις δύο κατακόρυφες πλευρές, συνολικού μήκους $2y$,
\item το τόξο του ημικυκλίου, μήκους $\pi x$.
\end{itemize}

Άρα
\[
2x+2y+\pi x=10.
\]
Επομένως,
\[
2y=10-(2+\pi)x
\]
και
\[
y=\frac{10-(2+\pi)x}{2}.
\]

Επειδή $x>0$ και $y>0$, πρέπει
\[
10-(2+\pi)x>0
\]
οπότε
\[
0<x<\frac{10}{2+\pi}.
\]

Το εμβαδόν του παραθύρου είναι το άθροισμα του εμβαδού του ορθογωνίου και του εμβαδού του ημικυκλίου:
\[
E=2xy+\frac{1}{2}\pi x^2.
\]
Άρα
\[
E(x)=2x\cdot \frac{10-(2+\pi)x}{2}+\frac{\pi}{2}x^2.
\]
Επομένως,
\[
E(x)=10x-(2+\pi)x^2+\frac{\pi}{2}x^2
\]
δηλαδή
\[
E(x)=10x-2x^2-\frac{\pi}{2}x^2,
\]
με
\[
x\in\left(0,\frac{10}{2+\pi}\right).
\]

\item Έχουμε
\[
E(x)=10x-\left(2+\frac{\pi}{2}\right)x^2.
\]
Άρα
\[
E'(x)=10-(4+\pi)x.
\]

Η παράγωγος μηδενίζεται όταν
\[
10-(4+\pi)x=0
\]
δηλαδή όταν
\[
x=\frac{10}{4+\pi}.
\]

Για
\[
0<x<\frac{10}{4+\pi}
\]
ισχύει $E'(x)>0$, ενώ για
\[
\frac{10}{4+\pi}<x<\frac{10}{2+\pi}
\]
ισχύει $E'(x)<0$.

Άρα η $E$ είναι γνησίως αύξουσα στο
\[
\left(0,\frac{10}{4+\pi}\right]
\]
και γνησίως φθίνουσα στο
\[
\left[\frac{10}{4+\pi},\frac{10}{2+\pi}\right).
\]

Επομένως, το εμβαδόν μεγιστοποιείται για
\[
x=\frac{10}{4+\pi}.
\]

Στη θέση αυτή,
\[
y=\frac{10-(2+\pi)x}{2}
=\frac{10-(2+\pi)\frac{10}{4+\pi}}{2}.
\]
Άρα
\[
y=\frac{1}{2}\cdot \frac{10(4+\pi)-10(2+\pi)}{4+\pi}
=\frac{1}{2}\cdot \frac{20}{4+\pi}
=\frac{10}{4+\pi}.
\]

Επομένως, στη θέση μέγιστου εμβαδού ισχύει
\[
y=x.
\]

\item Η μέγιστη τιμή της $E$ είναι
\[
E\left(\frac{10}{4+\pi}\right)
=10\cdot \frac{10}{4+\pi}
-\left(2+\frac{\pi}{2}\right)\left(\frac{10}{4+\pi}\right)^2.
\]
Επειδή
\[
2+\frac{\pi}{2}=\frac{4+\pi}{2},
\]
παίρνουμε
\[
E\left(\frac{10}{4+\pi}\right)
=\frac{100}{4+\pi}
-\frac{4+\pi}{2}\cdot \frac{100}{(4+\pi)^2}
=\frac{100}{4+\pi}-\frac{50}{4+\pi}.
\]
Άρα
\[
E_{\max}=\frac{50}{4+\pi}.
\]

Επιπλέον,
\[
\lim_{x\to 0^+}E(x)=0.
\]

Επειδή η $E$ είναι συνεχής στο πεδίο ορισμού της, αυξάνεται μέχρι τη μέγιστη τιμή και έπειτα φθίνει, ενώ οι τιμές κοντά στο $0$ πλησιάζουν το $0$, το σύνολο τιμών της είναι
\[
E\left(\left(0,\frac{10}{2+\pi}\right)\right)
=\left(0,\frac{50}{4+\pi}\right].
\]

\item Θέλουμε να εξετάσουμε πόσες λύσεις έχει η εξίσωση
\[
E(x)=5
\]
στο διάστημα
\[
\left(0,\frac{10}{2+\pi}\right).
\]

Αρχικά,
\[
\frac{50}{4+\pi}>5
\]
αφού
\[
50>5(4+\pi)\Leftrightarrow 10>4+\pi,
\]
που ισχύει.

Άρα, επειδή η $E$ είναι γνησίως αύξουσα στο
\[
\left(0,\frac{10}{4+\pi}\right],
\]
και
\[
\lim_{x\to0^+}E(x)=0,\qquad E_{\max}>5,
\]
υπάρχει ακριβώς μία λύση στο πρώτο διάστημα.

Στο φθίνον τμήμα, δηλαδή στο
\[
\left[\frac{10}{4+\pi},\frac{10}{2+\pi}\right),
\]
η $E$ μειώνεται μέχρι το όριο
\[
\lim_{x\to\left(\frac{10}{2+\pi}\right)^-}E(x)
=\frac{\pi}{2}\left(\frac{10}{2+\pi}\right)^2
=\frac{50\pi}{(2+\pi)^2}.
\]

Θα δείξουμε ότι το όριο αυτό είναι μεγαλύτερο από $5$:
\[
\frac{50\pi}{(2+\pi)^2}>5
\Leftrightarrow
10\pi>(2+\pi)^2.
\]
Η τελευταία ανισότητα γράφεται
\[
10\pi>\pi^2+4\pi+4
\Leftrightarrow
\pi(6-\pi)>4,
\]
η οποία ισχύει, αφού $3<\pi<4$, άρα $\pi(6-\pi)>3\cdot 2=6>4$.

Επομένως, στο φθίνον τμήμα όλες οι τιμές της $E$ είναι μεγαλύτερες από $5$, οπότε δεν υπάρχει δεύτερη λύση.

Συνεπώς, μπορεί να κατασκευαστεί ακριβώς ένα παράθυρο με συνολικό εμβαδόν
\[
5\ \tau.\mu.
\]
\end{erwthma}
\end{thema}
