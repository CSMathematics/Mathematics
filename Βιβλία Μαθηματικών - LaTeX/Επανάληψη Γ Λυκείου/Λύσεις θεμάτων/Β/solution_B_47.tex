\begin{thema}{Β}
Δίνεται η πολυωνυμική συνάρτηση $f(x) = x^4 - 2x^2 + 1$.
\begin{erwthma}
\item Η συνάρτηση $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος \eng{Rolle} στο διάστημα $[-1,1]$:
\begin{itemize}
\item Είναι συνεχής στο $[-1,1]$, ως πολυωνυμική.
\item Είναι παραγωγίσιμη στο $(-1,1)$.
\item Ισχύει $f(-1)=1-2+1=0$ και $f(1)=1-2+1=0$, άρα $f(-1)=f(1)$.
\end{itemize}
Επομένως υπάρχει τουλάχιστον ένα $\xi\in(-1,1)$ τέτοιο ώστε $f'(\xi)=0$. Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $\xi$ είναι οριζόντια.

\item Η $f$ είναι παραγωγίσιμη με:
\[ f'(x)=4x^3-4x=4x(x-1)(x+1). \]
Οι ρίζες της $f'$ είναι $x=-1$, $x=0$ και $x=1$.
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$-1$,$0$,$1$,$+\infty$}%
\tkzTabLine{ , -, z, +, z, -, z, +, }
\tkzTabLine{ , \searrow, t, \nearrow, t, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}
Οπότε η $f$ είναι γνησίως φθίνουσα στα $(-\infty,-1]$ και $[0,1]$, ενώ είναι γνησίως αύξουσα στα $[-1,0]$ και $[1,+\infty)$.
Παρουσιάζει τοπικά ελάχιστα στα $x=-1$ και $x=1$, με τιμές $f(-1)=f(1)=0$, και τοπικό μέγιστο στο $x=0$, με τιμή $f(0)=1$.

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x)=12x^2-4=4(3x^2-1). \]
Έχουμε:
\[ f''(x)=0 \Leftrightarrow 3x^2-1=0 \Leftrightarrow x=\pm\frac{\sqrt{3}}{3}. \]
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:
\begin{center}

\tkzTabLine{ , \rotatebox[origin= c]{180}{$\curvearrowleft$}, t, \curvearrowright, t, \rotatebox[origin= c]{180}{$\curvearrowleft$}, }
\end{tikzpicture}
\end{center}
Η $C_f$ παρουσιάζει σημεία καμπής για $x=\pm\frac{\sqrt{3}}{3}$.
Επειδή
\[ f\left(\pm\frac{\sqrt{3}}{3}\right)=\frac{1}{9}-\frac{2}{3}+1=\frac{4}{9}, \]
τα σημεία καμπής είναι:
\[ A\left(-\frac{\sqrt{3}}{3},\frac{4}{9}\right), \quad B\left(\frac{\sqrt{3}}{3},\frac{4}{9}\right). \]

\item Τα σημεία τομής της $C_f$ με την ευθεία $y=9$ προκύπτουν από την εξίσωση:
\[ x^4-2x^2+1=9 \Leftrightarrow x^4-2x^2-8=0. \]
Θέτοντας $u=x^2$, παίρνουμε:
\[ u^2-2u-8=0 \Leftrightarrow u=4 \ \text{ή}\ u=-2. \]
Επειδή $u=x^2\ge0$, έχουμε $x^2=4$, άρα $x=\pm2$.
Στο διάστημα $[-2,2]$ ισχύει $9-f(x)\ge0$, οπότε το εμβαδόν είναι:
\begin{align*}
E&=\int_{-2}^{2}\left(9-f(x)\right)\,\d x
=\int_{-2}^{2}\left(8+2x^2-x^4\right)\,\d x\\
&=2\int_0^2\left(8+2x^2-x^4\right)\,\d x
=2\left[8x+\frac{2x^3}{3}-\frac{x^5}{5}\right]_0^2\\
&=2\left(16+\frac{16}{3}-\frac{32}{5}\right)=\frac{448}{15}.
\end{align*}
\end{erwthma}
\end{thema}
