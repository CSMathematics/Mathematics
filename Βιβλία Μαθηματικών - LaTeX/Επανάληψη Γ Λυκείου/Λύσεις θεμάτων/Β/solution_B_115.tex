\begin{thema}{Β}
Από το σχήμα η γραφική παράσταση της $f$ είναι το ευθύγραμμο τμήμα που περνά από τα σημεία
\[
(0,3),\ (1,2),\ (2,1),\ (3,0),\ (4,-1).
\]
Άρα, στο διάστημα $[0,4]$, η $f$ έχει τύπο
\[
f(x)=3-x.
\]

\begin{erwthma}
\item Η $f$ είναι γνησίως φθίνουσα στο $[0,4]$, άρα είναι $1-1$ και επομένως αντιστρέφεται.

Από το σχήμα έχουμε
\[
f(0)=3
\quad\text{και}\quad
f(4)=-1.
\]
Επειδή η $f$ είναι γνησίως φθίνουσα, το σύνολο τιμών της είναι
\[
f([0,4])=[-1,3].
\]
Άρα το πεδίο ορισμού της $f^{-1}$ είναι
\[
D_{f^{-1}}=[-1,3].
\]

\item Από το σχήμα διαβάζουμε
\[
f(4)=-1,\quad f(3)=0,\quad f(1)=2.
\]
Άρα
\[
f^{-1}(-1)=4,\quad f^{-1}(0)=3,\quad f^{-1}(2)=1.
\]

\item Η αντίστροφη μιας γνησίως φθίνουσας συνάρτησης είναι επίσης γνησίως φθίνουσα.

Πράγματι, έστω $y_1<y_2$ στο $[-1,3]$. Αν
\[
x_1=f^{-1}(y_1),\quad x_2=f^{-1}(y_2),
\]
τότε
\[
f(x_1)=y_1,\quad f(x_2)=y_2.
\]
Επειδή η $f$ είναι γνησίως φθίνουσα και $y_1<y_2$, πρέπει να είναι $x_1>x_2$. Άρα
\[
f^{-1}(y_1)>f^{-1}(y_2),
\]
οπότε η $f^{-1}$ είναι γνησίως φθίνουσα.

\item Από τον τύπο
\[
f(x)=3-x,\quad 0\leq x\leq 4,
\]
βρίσκουμε την αντίστροφη. Θέτουμε
\[
y=3-x.
\]
Τότε
\[
x=3-y.
\]
Άρα
\[
f^{-1}(x)=3-x,\quad -1\leq x\leq 3.
\]
Στο κοινό πεδίο ορισμού των $f$ και $f^{-1}$, δηλαδή στο $[0,3]$, ισχύει
\[
f(x)=3-x=f^{-1}(x).
\]
Επομένως
\[
f^{-1}(x)=f(x),\quad x\in[0,3].
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f^{-1}(x)=3-x,\quad -1\leq x\leq 3.
\]
Άρα
\begin{align*}
\dint_{-1}^{3} f^{-1}(x)\d x
&=\dint_{-1}^{3}(3-x)\d x\\
&=\left[3x-\frac{x^2}{2}\right]_{-1}^{3}\\
&=\left(9-\frac{9}{2}\right)-\left(-3-\frac{1}{2}\right)\\
&=\frac{9}{2}+\frac{7}{2}=8.
\end{align*}
\end{erwthma}
\end{thema}
