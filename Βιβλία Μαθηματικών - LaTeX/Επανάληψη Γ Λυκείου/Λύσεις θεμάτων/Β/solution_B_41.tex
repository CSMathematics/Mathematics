\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = \ln(x^2+1)$ και $g(x) = x$, ορισμένες στο $\mathbb{R}$.
\begin{erwthma}
\item Η συνάρτηση $h$ ορίζεται με τύπο
\[ h(x)=f(x)-g(x)=\ln(x^2+1)-x \]
και έχει πεδίο ορισμού το $D_h=\mathbb{R}$.
Επειδή $h(x)+x=\ln(x^2+1)$, έχουμε:
\[ \lim_{x \to 0} \frac{h(x)+x}{x^2} = \lim_{x \to 0} \frac{\ln(x^2+1)}{x^2} \dlh{\frac{0}{0}} \lim_{x \to 0} \frac{\frac{2x}{x^2+1}}{2x} = \lim_{x \to 0} \frac{1}{x^2+1} = 1. \]

\item Η $h$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ h'(x) = \frac{2x}{x^2+1} - 1 = \frac{2x-(x^2+1)}{x^2+1} = -\frac{(x-1)^2}{x^2+1}. \]
Επειδή $(x-1)^2\ge 0$ και $x^2+1>0$, ισχύει $h'(x)\le 0$ για κάθε $x\in\mathbb{R}$, με ισότητα μόνο για $x=1$.
Άρα η $h$ είναι γνησίως φθίνουσα στο $\mathbb{R}$.

\item Η $h'$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ h''(x)=\left(\frac{2x}{x^2+1}-1\right)'=\frac{2(x^2+1)-4x^2}{(x^2+1)^2}=\frac{2(1-x^2)}{(x^2+1)^2}. \]
Επειδή $(x^2+1)^2>0$, το πρόσημο της $h''$ εξαρτάται από το $1-x^2$.
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $h''$ και η κυρτότητα της $h$:

Άρα η $h$ είναι κοίλη στα $(-\infty,-1]$ και $[1,+\infty)$ και κυρτή στο $[-1,1]$.
Η $C_h$ παρουσιάζει σημεία καμπής στα:
\[ A(-1,h(-1))=(-1,\ln2+1) \quad \text{και} \quad B(1,h(1))=(1,\ln2-1). \]

\item Αναζητούμε πλάγια ασύμπτωτη της μορφής $y=\lambda x+\beta$ στο $+\infty$.
Έχουμε:
\[ \lambda = \lim_{x\to+\infty}\frac{h(x)}{x} = \lim_{x\to+\infty}\left(\frac{\ln(x^2+1)}{x}-1\right). \]
Με \eng{De L'Hospital}:
\[ \lim_{x\to+\infty}\frac{\ln(x^2+1)}{x} \dlh{\frac{+\infty}{+\infty}} \lim_{x\to+\infty}\frac{2x}{x^2+1}=0, \]
οπότε $\lambda=-1$.
Στη συνέχεια:
\[ \beta=\lim_{x\to+\infty}(h(x)-\lambda x)=\lim_{x\to+\infty}(\ln(x^2+1)-x+x)=+\infty. \]
Επειδή το $\beta$ δεν είναι πραγματικός αριθμός, δεν υπάρχει πλάγια ασύμπτωτη της $C_h$ στο $+\infty$.
\end{erwthma}
\end{thema}
