\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=x^3-3x^2+1,\quad x\in\mathbb{R}.
\]

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=3x^2-6x=3x(x-2).
\]
Βρίσκουμε ρίζες και πρόσημο της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow x=0$ ή $x=2$,
\item $f'(x)>0\Leftrightarrow x<0$ ή $x>2$,
\item $f'(x)<0\Leftrightarrow 0<x<2$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%

\tkzTabLine{ , +, z, -, z, +, }
\tkzTabLine{ , \nearrow, t, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στα $(-\infty,0]$, $[2,+\infty)$ και \textbf{γνησίως φθίνουσα} στο $[0,2]$. Παρουσιάζει τοπικό μέγιστο στο $x=0$, με
\[
f(0)=1,
\]
και τοπικό ελάχιστο στο $x=2$, με
\[
f(2)=-3.
\]

\item Επειδή
\[
\lim_{x\to-\infty}f(x)=-\infty
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty,
\]
και η $f$ είναι συνεχής στο $\mathbb{R}$, το σύνολο τιμών της είναι
\[
f(\mathbb{R})=\mathbb{R}.
\]
Επομένως η $f$ δεν έχει ολικά ακρότατα, αφού δεν είναι ούτε άνω ούτε κάτω φραγμένη.

\item Η εξίσωση γράφεται
\[
x^3-3x^2+\lambda=0
\Leftrightarrow f(x)=1-\lambda.
\]
Θέτουμε $\mu=1-\lambda$. Από τη μονοτονία του πρώτου ερωτήματος έχουμε:
\[
f((-\infty,0])=(-\infty,1],\quad
f([0,2])=[-3,1],\quad
f([2,+\infty))=[-3,+\infty).
\]
Άρα:
\begin{itemize}
\item αν $\mu>1$, δηλαδή $\lambda<0$, υπάρχει ακριβώς μία ρίζα,
\item αν $\mu=1$, δηλαδή $\lambda=0$, υπάρχουν ακριβώς δύο ρίζες,
\item αν $-3<\mu<1$, δηλαδή $0<\lambda<4$, υπάρχουν ακριβώς τρεις ρίζες,
\item αν $\mu=-3$, δηλαδή $\lambda=4$, υπάρχουν ακριβώς δύο ρίζες,
\item αν $\mu<-3$, δηλαδή $\lambda>4$, υπάρχει ακριβώς μία ρίζα.
\end{itemize}

\item Η εφαπτομένη της $C_f$ είναι οριζόντια στα σημεία όπου $f'(x)=0$. Από το πρώτο ερώτημα,
\[
f'(x)=0\Leftrightarrow x=0\ \text{ή}\ x=2.
\]
Τα αντίστοιχα σημεία είναι
\[
A(0,f(0))\equiv A(0,1)
\quad\text{και}\quad
B(2,f(2))\equiv B(2,-3).
\]
\end{erwthma}
\end{thema}
