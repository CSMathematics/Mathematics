\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = e^x + x - 1$ με πεδίο ορισμού το $\mathbb{R}$ και $g(x) = \ln x$ με πεδίο ορισμού το $(0, +\infty)$.

\begin{erwthma}

% -------------------------------------------------------
% α) Πεδίο ορισμού και τύπος h = f∘g
% -------------------------------------------------------
\item Η συνάρτηση $h = f \circ g$ έχει:
\begin{itemize}
    \item {τύπο:} $h(x) = f(g(x)) = f(\ln x) = e^{\ln x} + \ln x - 1 = x + \ln x - 1$
    \item {πεδίο ορισμού:} $D_h = \{x \in D_g \mid g(x) \in D_f\}$.
    \begin{itemize}[label=\tiny\faPlay]
        \item $x \in D_g \Leftrightarrow x > 0$ και
        \item $g(x) \in D_f \Leftrightarrow \ln x \in \mathbb{R}$, το οποίο ισχύει για κάθε $x > 0$.
    \end{itemize}
\end{itemize}
Άρα $h(x) = x + \ln x - 1$, $x \in (0, +\infty)$.

% -------------------------------------------------------
% β) Υπολογισμός ορίου
% -------------------------------------------------------
\item Για το όριο έχουμε
\[\lim\limits_{x \to 1^+} \frac{2x+1}{h(x)}=\frac{3}{0}\]
Ελέγχουμε το πρόσημο του $h(x)$ καθώς $x \to 1^+$:
Είναι $h'(x) = 1 + \dfrac{1}{x} > 0$ για κάθε $x \in (0,+\infty)$, άρα η $h$ είναι γνησίως αύξουσα.
Αφού $h(1) = 1 + 0 - 1 = 0$, για $x > 1$ (κοντά στο 1) ισχύει $h(x) > h(1) = 0$, δηλαδή $h(x) \to 0^+$.

Επομένως:
\[
\lim_{x \to 1^+} \frac{2x+1}{h(x)} = \frac{3}{0^+} = +\infty.
\]

% -------------------------------------------------------
% γ) Ύπαρξη x₀ ∈ (1,e) με x₀ + ln x₀ = 2
% -------------------------------------------------------
\item Ορίζουμε $\varphi(x) = x + \ln x - 2$, η οποία είναι συνεχής στο κλειστό διάστημα $[1, e]$.

Υπολογίζουμε τις τιμές στα άκρα:
\[
\varphi(1) = 1 + \ln 1 - 2 = 1 + 0 - 2 = -1 < 0,
\]
\[
\varphi(e) = e + \ln e - 2 = e + 1 - 2 = e - 1 > 0.
\]
άρα $\varphi(1) \cdot \varphi(e) < 0$ οπότε από το \textbf{Θεώρημα Bolzano} υπάρχει τουλάχιστον ένας $x_0 \in (1, e)$ τέτοιος ώστε $\varphi(x_0) = 0$, δηλαδή:
\[
x_0 + \ln x_0 = 2.
\]

% -------------------------------------------------------
% δ) Απόδειξη ανισότητας ολοκληρώματος
% -------------------------------------------------------
\item Θα δείξουμε ότι $\displaystyle\int_1^2 h(x)\,\d x < 1 + \ln 2$.

Υπολογίζουμε πρώτα το ολοκλήρωμα. Για το $\int_1^2 \ln x\,\d x$ εφαρμόζουμε ολοκλήρωση κατά παράγοντες:
\[
\int_1^2 \ln x\,\d x = \bigl[x\ln x\bigr]_1^2 - \int_1^2 1\,\d x = 2\ln 2 - [x]_1^2 = 2\ln2 - 1.
\]
Άρα:
\[
\int_1^2 h(x)\,\d x = \int_1^2 (x + \ln x - 1)\,\d x = \left[\frac{x^2}{2}\right]_1^2 + (2\ln2 - 1) - [x]_1^2
= \frac{3}{2} + 2\ln2 - 1 - 1 = 2\ln2 - \frac{1}{2}.
\]
Θέλουμε να δείξουμε ότι $2\ln 2 - \dfrac{1}{2} < 1 + \ln 2$, δηλαδή $\ln 2 < \frac{3}{2}$. Αυτό ισχύει αφού $\ln 2 < \ln e = 1 < \dfrac{3}{2}$.
Επομένως:
\[
\int_1^2 h(x)\,\d x = 2\ln2 - \frac{1}{2} < 1 + \ln2.
\]

\end{erwthma}
\end{thema}
