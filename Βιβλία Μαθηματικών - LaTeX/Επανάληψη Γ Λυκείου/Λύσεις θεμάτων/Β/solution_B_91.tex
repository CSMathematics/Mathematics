\begin{thema}{Β}
\begin{erwthma}
\item Από το σχήμα υπάρχει κατακόρυφη ασύμπτωτη $x=1$, άρα
\[
D_f=\mathbb{R}-\{1\}=(-\infty,1)\cup(1,+\infty).
\]
Επίσης
\[
\lim_{x\to1^-}f(x)=-\infty,\quad
\lim_{x\to1^+}f(x)=+\infty,\quad
\lim_{x\to-\infty}f(x)=2,\quad
\lim_{x\to+\infty}f(x)=2.
\]

\item Οι ασύμπτωτες της $C_f$ είναι η κατακόρυφη $x=1$ και η οριζόντια $y=2$.

\item Έστω $f(x)=\dfrac{ax+\beta}{x-\gamma}$. Από την κατακόρυφη ασύμπτωτη $x=1$ παίρνουμε $\gamma=1$, ενώ από την οριζόντια ασύμπτωτη $y=2$ παίρνουμε $a=2$. Άρα
\[
f(x)=\frac{2x+\beta}{x-1}.
\]
Από το σχήμα $f(0)=1$, οπότε $\dfrac{\beta}{-1}=1\Leftrightarrow \beta=-1$. Επομένως
\[
f(x)=\frac{2x-1}{x-1}.
\]

\item Γράφουμε
\[
f(x)=\frac{2x-1}{x-1}=2+\frac1{x-1}.
\]
Άρα για $x<1$ είναι $f(x)<2$, ενώ για $x>1$ είναι $f(x)>2$. Τα σύνολα τιμών των δύο κλάδων είναι ξένα. Επιπλέον
\[
f'(x)=-\frac1{(x-1)^2}<0,\quad x\ne1,
\]
άρα η $f$ είναι γνησίως φθίνουσα σε κάθε κλάδο. Συνεπώς είναι $1-1$ και αντιστρέφεται.

Θέτουμε $y=\dfrac{2x-1}{x-1}$. Τότε
\[
y(x-1)=2x-1\Leftrightarrow x(y-2)=y-1\Leftrightarrow x=\frac{y-1}{y-2}.
\]
Άρα
\[
f^{-1}(x)=\frac{x-1}{x-2},\quad x\ne2.
\]
Τέλος,
\[
\frac{f^{-1}(x)+x^2-1}{x-1}
=\frac{\frac{x-1}{x-2}+(x-1)(x+1)}{x-1}
=\frac1{x-2}+x+1,
\]
οπότε
\[
\lim_{x\to1}\frac{f^{-1}(x)+x^2-1}{x-1}=\frac1{1-2}+1+1=1.
\]
\end{erwthma}
\end{thema}
