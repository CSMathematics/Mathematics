Mε βάση το εμπλουτισμένο σχήμα:
\begin{erwthma}

\item Από το γράφημα είναι φανερό ότι η καμπύλη σπάει σε δύο κλάδους, με τη συνάρτηση να μην ορίζεται όταν το $x$ παίρνει την τιμή $1$, εξαιτίας της κατακόρυφης ασύμπτωτης. Άρα, το πεδίο ορισμού της είναι το προσδιοριστέο σύνολο $\varDelta_f = \mathbb{R} \setminus \{1\} = (-\infty, 1) \cup (1, +\infty)$.
Αναγνωρίζουμε τα όρια εξετάζοντας την πορεία της καμπύλης στα επιμέρους άκρα:
\begin{itemize}
    \item Καθώς $x \to 1^-$, ο αριστερός κλάδος βυθίζεται, άρα $\lim\limits_{x \to 1^-} f(x) = -\infty$.
    \item Καθώς $x \to 1^+$, ο δεξιός κλάδος εκτοξεύεται ψηλά, άρα $\lim\limits_{x \to 1^+} f(x) = +\infty$.
    \item Καθώς $x \to -\infty$, ο κλάδος προσεγγίζει τη διακεκομμένη ευθεία από κάτω, άρα $\lim\limits_{x \to -\infty} f(x) = 2$.
    \item Καθώς $x \to +\infty$, ο άλλος κλάδος προσεγγίζει τη διακεκομμένη ευθεία από πάνω, άρα $\lim\limits_{x \to +\infty} f(x) = 2$.
\end{itemize}

\item Με βάση τα παραπάνω όρια, προκύπτουν άμεσα οι ασύμπτωτες:
\begin{itemize}
    \item Η κατακόρυφη ευθεία \textbf{$x = 1$} αποτελεί \textbf{κατακόρυφη ασύμπτωτη} επειδή ισχύει $\lim\limits_{x \to 1} f(x) = \pm\infty$.
    \item Η οριζόντια ευθεία \textbf{$y = 2$} αποτελεί \textbf{οριζόντια ασύμπτωτη} στο $-\infty$ και στο $+\infty$, αφού το αντίστοιχο όριο είναι $2$.
\end{itemize}

\item Παρατηρούμε τη σχετική θέση της καμπύλης $C_f$ σε σχέση με την ευθεία $y = 2$.
\begin{itemize}
    \item \textbf{Εξίσωση $f(x) = 2$:} Η καμπύλη τείνει στο $y=2$ ασυμπτωτικά, ωστόσο πουθενά δεν τέμνει τη συγκεκριμένη διακεκομμένη ευθεία. Συνεπώς, η εξίσωση $f(x)=2$ είναι \textbf{αδύνατη}.
    \item \textbf{Ανίσωση $f(x) > 2$:} Σε ποιες θέσεις η $C_f$ βρίσκεται "πάνω" από την ευθεία $y=2$; Αποκλειστικά στον δεξιό της κλάδο, δηλαδή για κάθε $x > 1$. Συνεπώς, η λύση είναι το σύνολο \textbf{$x \in (1, +\infty)$}. (Στον αντιδιαμετρικό αριστερό κλάδο, η συνάρτηση είναι σταθερά μικρότερη του $2$).
\end{itemize}

\item Για τη μελέτη της μονοτονίας, διατρέχουμε την καμπύλη από τα αριστερά προς τα δεξιά:
\begin{itemize}
    \item Στο διάστημα \textbf{$(-\infty, 1)$}, καθώς το $x$ αυξάνεται, το ύψος $y$ μειώνεται συνεχόμενα (από το $2$ προς το $-\infty$). Συνεπώς, η $f$ είναι \textbf{γνησίως φθίνουσα}.
    \item Στο διάστημα \textbf{$(1, +\infty)$}, καθώς το $x$ αυξάνεται, το ύψος $y$ μειώνεται (από το $+\infty$ προς το $2$). Συνεπώς, η $f$ είναι επίσης \textbf{γνησίως φθίνουσα}.
\end{itemize}
Σημειώνεται πως, αν και η συνάρτηση είναι γνησίως φθίνουσα σε καθένα από τα δύο αυτά διαστήματα του πεδίου ορισμού ξεχωριστά, δεν επιτρέπεται η ένωσή τους (δεν είναι γνησίως φθίνουσα συνολικά στην ένωση, αφού π.χ. $f(0) \approx 1$ και $f(2) = 3$, παρατηρώντας αύξηση ανάμεσα σε σημεία διαφορετικών διαστημάτων).

\end{erwthma}
\end{thema}
