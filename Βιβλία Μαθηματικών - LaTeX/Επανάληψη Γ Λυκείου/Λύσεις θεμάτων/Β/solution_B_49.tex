\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x - \hm x}{x + \hm x}$ με $x\ne0$.
\begin{erwthma}
\item Για το όριο στο 0 διαιρούμε αριθμητή και παρονομαστή με $x \neq 0$:
\[ \lim_{x\to0} f(x) = \lim_{x\to0} \frac{x - \hm x}{x + \hm x} = \lim_{x\to0} \frac{1 - \frac{\hm x}{x}}{1 + \frac{\hm x}{x}}. \]
Γνωρίζοντας το βασικό όριο $\lim\limits_{x\to0} \frac{\hm x}{x} = 1$, έχουμε:
\[ \lim_{x\to0} f(x) = \frac{1 - 1}{1 + 1} = 0. \]

\item Για το όριο στο $+\infty$, βγάζουμε κοινό παράγοντα το $x$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{x\left(1 - \frac{\hm x}{x}\right)}{x\left(1 + \frac{\hm x}{x}\right)}
= \lim_{x\to+\infty} \frac{1 - \frac{\hm x}{x}}{1 + \frac{\hm x}{x}}. \]
Από την ιδιότητα $|\hm x| \le 1$ προκύπτει, για $x>0$, ότι:
\[ -\frac{1}{x} \le \frac{\hm x}{x} \le \frac{1}{x}. \]
Άρα, από το Κριτήριο Παρεμβολής, $\lim\limits_{x\to+\infty} \frac{\hm x}{x}=0$.
Επομένως:
\[ \lim_{x\to+\infty} f(x) = \frac{1 - 0}{1 + 0} = 1. \]

\item Η συνάρτηση $h(x) = x - \hm x$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ h'(x) = 1 - \syn x. \]
Επειδή $\syn x \le 1$, ισχύει $h'(x)\ge0$ για κάθε $x\in\mathbb{R}$.
Επιπλέον, η ισότητα $h'(x)=0$ ισχύει μόνο για $x=2\kappa\pi$, $\kappa\in\mathbb{Z}$, δηλαδή σε μεμονωμένα σημεία. Επομένως η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$.

\item Η εξίσωση $x=\hm x$ ισοδυναμεί με:
\[ x-\hm x=0 \Leftrightarrow h(x)=0. \]
Παρατηρούμε ότι $h(0)=0-\hm0=0$, άρα το $x=0$ είναι ρίζα της εξίσωσης.
Επειδή η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$, μπορεί να έχει το πολύ μία ρίζα. Άρα το $x=0$ είναι η μοναδική ρίζα της εξίσωσης στο $\mathbb{R}$.
\end{erwthma}
\end{thema}
