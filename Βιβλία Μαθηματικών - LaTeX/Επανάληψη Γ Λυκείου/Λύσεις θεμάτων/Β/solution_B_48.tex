\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^x(x-1) + 1$, με $x \in \mathbb{R}$.
\begin{erwthma}
\item Για το όριο στο $-\infty$, γράφουμε το γινόμενο σε μορφή πηλίκου:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} (e^x(x-1) + 1) = \lim_{x\to-\infty} \frac{x-1}{e^{-x}} + 1 \]
Επειδή $\lim_{x\to-\infty} (x-1) = -\infty$ και $\lim_{x\to-\infty} e^{-x} = +\infty$, το κλάσμα δημιουργεί απροσδιοριστία $\frac{-\infty}{+\infty}$. Εφαρμόζουμε τον κανόνα De L' Hospital:
\[ \lim_{x\to-\infty} \frac{x-1}{e^{-x}} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to-\infty} \frac{(x-1)'}{(e^{-x})'} = \lim_{x\to-\infty} \frac{1}{-e^{-x}} = \lim_{x\to-\infty} (-e^x) = 0 \]
Συνεπώς το ζητούμενο όριο ισούται με $\lim_{x\to-\infty} f(x) = 0 + 1 = 1$.

\item Υπολογίζουμε την πρώτη και δεύτερη παράγωγο της $f$:
\[ f'(x) = e^x(x-1) + e^x(1) + 0 = x e^x - e^x + e^x = x e^x \]
\[ f''(x) = 1 \cdot e^x + x e^x = e^x(x+1) \]
\textbf{Μονοτονία:}
Ρίζα της $f'$ είναι το $x=0$.
- $f'(x) > 0 \Leftrightarrow x > 0$, άρα η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$.
- $f'(x) < 0 \Leftrightarrow x < 0$, άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$.
Παρουσιάζει ολικό ελάχιστο στο $x=0$ με τιμή $f(0) = e^0(0-1)+1 = 0$.
\textbf{Κυρτότητα:}
Ρίζα της $f''$ είναι το $x=-1$.
- $f''(x) > 0 \Leftrightarrow x > -1$, άρα η $f$ είναι κυρτή στο $[-1, +\infty)$.
- $f''(x) < 0 \Leftrightarrow x < -1$, άρα η $f$ είναι κοίλη στο $(-\infty, -1]$.
Παρουσιάζει σημείο καμπής στο σημείο $x=-1$ με τιμή $f(-1) = e^{-1}(-2) + 1 = 1 - \frac{2}{e}$.

\item Θεωρούμε τη συνάρτηση $h(x) = f(x) - \frac{1}{2}$ στο κλειστό $[0, 1]$.
Η $h$ είναι συνεχής. Στα άκρα:
- $h(0) = f(0) - 0,5 = 0 - 0,5 = -0,5 < 0$.
- $h(1) = f(1) - 0,5 = 1 - 0,5 = 0,5 > 0$.
Επειδή $h(0)h(1) < 0$, σύμφωνα με το Θεώρημα Bolzano υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $h(x_0)=0 \Leftrightarrow f(x_0) = \frac{1}{2}$.
Διαπιστώσαμε ότι η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$, άρα και στο υποσύνολό του $[0, 1]$. Έτσι το $x_0$ είναι μοναδικό.

\item Το σημείο επαφής είναι το $M(0, f(0))$, δηλαδή το $(0, 0)$.
Ο συντελεστής διεύθυνσης της εφαπτομένης είναι $f'(0) = 0 \cdot e^0 = 0$. Επομένως, πρόκειται για οριζόντια εφαπτομένη.
Η εξίσωσή της δίνεται από τον τύπο $y - f(0) = f'(0)(x-0) \Leftrightarrow y - 0 = 0(x-0) \Leftrightarrow y = 0$. 
Οπότε η εφαπτομένη ταυτίζεται με τον άξονα $x'x$.
\end{erwthma}
\end{thema}
