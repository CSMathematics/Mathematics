\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x)=\dfrac{x}{e^x}$.

\begin{erwthma}

% -------------------------------------------------------
% α) Μονοτονία, ακρότατα, σύνολο τιμών
% -------------------------------------------------------
\item Η $f$ ορίζεται για κάθε $x\in\mathbb{R}$ ($e^x>0$ πάντα), άρα $D_f=\mathbb{R}$.
Είναι παραγωγίσιμη με:
\[
f'(x)=\left(\frac{x}{e^x}\right)'=\frac{(x)'\cdot e^x-x\cdot(e^x)'}{(e^x)^2}
=\frac{e^x-xe^x}{e^{2x}}=\frac{e^x(1-x)}{e^{2x}}=\frac{1-x}{e^x}.
\]
Βρίσκουμε ρίζες και πρόσημο της $f'$:
\begin{itemize}
    \item $f'(x)=0\Leftrightarrow 1-x=0\Leftrightarrow x=1$
    \item $f'(x)>0\Leftrightarrow \dfrac{1-x}{e^x}>0\stackrel{e^x>0}{\Leftrightarrow}1-x>0\Leftrightarrow x<1$
    \item $f'(x)<0\Leftrightarrow x>1$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}

\end{tikzpicture}
\end{center}
Συνεπώς η $f$ είναι γνησίως αύξουσα στο $(-\infty,1]$ και γνησίως φθίνουσα στο $[1,+\infty)$.
Στο $x_0=1$ παρουσιάζει \textbf{ολικό μέγιστο} με τιμή $f(1)=\dfrac{1}{e}$.

Για το σύνολο τιμών έχουμε:
\begin{itemize}
\item $f\nearrow\varDelta_1=(-\infty,1]$ άρα $f(\varDelta_1)=\left(\lim\limits_{x\to -\infty}{f(x)},f(1)\right]=\left(-\infty,\dfrac{1}{e}\right]$ αφού:
\[\lim\limits_{x\to-\infty}\dfrac{x}{e^x}=\lim\limits_{x\to -\infty}{\left(xe^{-x}\right)}=(-\infty)(+\infty)=-\infty\] 
\item $f\searrow\varDelta_2=[1,+\infty)$ άρα $f(\varDelta_2)=\left(\lim\limits_{x\to +\infty}{f(x)},f(1)\right]=\left(0,\dfrac{1}{e}\right]$ αφού:
\[ \lim\limits_{x\to+\infty}\dfrac{x}{e^x}\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}\lim\limits_{x\to +\infty}{\dfrac{1}{e^x}}=0\]
\end{itemize}

Άρα $f(\mathbb{R})=f(\varDelta_1)\cup f(\varDelta_2)=\left(-\infty,\,\dfrac{1}{e}\right]$.

% -------------------------------------------------------
% β) Κυρτότητα, σημεία καμπής
% -------------------------------------------------------
\item Για κάθε $x\in\mathbb{R}$:
\[
f''(x)=\frac{(1-x)'\cdot e^x-(1-x)\cdot e^x}{e^{2x}}
=\frac{-e^x-(1-x)e^x}{e^{2x}}
=\frac{e^x(-1-1+x)}{e^{2x}}
=\frac{x-2}{e^x}.
\]
Βρίσκουμε ρίζες και πρόσημο της $f''$:
\begin{itemize}
    \item $f''(x)=0\Leftrightarrow x-2=0\Leftrightarrow x=2$
    \item $f''(x)>0\Leftrightarrow \dfrac{x-2}{e^x}>0\stackrel{e^x>0}{\Leftrightarrow} x>2$
    \item $f''(x)<0\Leftrightarrow x<2$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:
\begin{center}

Επομένως η $f$ είναι κοίλη στο $(-\infty,2]$ και κυρτή στο $[2,+\infty)$.
Επίσης η $C_f$ έχει σημείο καμπής το $K(2,f(2))\equiv K\left(2,\,\dfrac{2}{e^2}\right)$.

% -------------------------------------------------------
% γ) Εφαπτομένη στο Α(0, f(0))
% -------------------------------------------------------
\item Υπολογίζουμε $f(0)=\dfrac{0}{e^0}=0$, άρα το σημείο είναι $A=(0,0)$.
Η κλίση εφαπτομένης είναι:
\[
f'(0)=\frac{1-0}{e^0}=1.
\]
επομένως έχει εξίσωση:
\[
y-f(0)=f'(0)(x-0)\Leftrightarrow y=x.
\]

% -------------------------------------------------------
% δ) Εμβαδόν
% -------------------------------------------------------
\item Για $x\in[0,\ln9]$ είναι $f(x)=\dfrac{x}{e^x}\geq0$ (αφού $x\geq0$, $e^x>0$), επομένως:
\[
E=\int_0^{\ln 9}\frac{x}{e^x}\,\d x=\int_0^{\ln 9}x\,e^{-x}\,\d x.
\]
Εφαρμόζουμε ολοκλήρωση κατά παράγοντες οπότε:
\begin{align*}
E&=\int_0^{\ln 9}x\left(-e^{-x}\right)'\,\d x=\Bigl[-xe^{-x}\Bigr]_0^{\ln 9}+\int_0^{\ln 9}e^{-x}\,\d x
=-\ln9\cdot e^{-\ln9}+\Bigl[-e^{-x}\Bigr]_0^{\ln 9}=\\&=-\frac{\ln 9}{9}+\left(-\frac{1}{9}+1\right)
=-\frac{\ln 9}{9}+\frac{8}{9}
=\frac{8-\ln 9}{9}\ \text{τ.μ.}
\end{align*}



\end{erwthma}
\end{thema}
