\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = e^x - 1$ και $g(x) = \ln x$. 
\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι το $D_f = \mathbb{R}$ και της $g$ είναι το $D_g = (0, +\infty)$.
\begin{itemize}
\item Για την $h = \frac{f}{g}$, απαιτείται $x \in D_f \cap D_g$ και $g(x) \ne 0 \Leftrightarrow \ln x \ne 0 \Leftrightarrow x \ne 1$.
Άρα το πεδίο ορισμού είναι το $D_h = (0, 1) \cup (1, +\infty)$. 
Τύπος: $h(x) = \frac{e^x - 1}{\ln x}$.
\item Για την $k = f \cdot g$, απαιτείται $x \in D_f \cap D_g \Rightarrow x \in (0, +\infty)$.
Άρα $D_k = (0, +\infty)$. Τύπος: $k(x) = (e^x - 1) \ln x$.
\end{itemize}

\item Υπολογίζουμε τα πλευρικά όρια στο $x_0 = 1$ για τη συνάρτηση $h(x)$:
Ο αριθμητής τείνει στο $e^1 - 1 = e - 1 > 0$.
Ο παρονομαστής $\ln x$ τείνει στο $0$. Συγκεκριμένα:
\begin{itemize}
\item $\lim_{x\to1^+} \ln x = 0^+$ (αφού $x>1 \Rightarrow \ln x > 0$), άρα $\lim_{x\to1^+} \frac{e^x-1}{\ln x} = \frac{e-1}{0^+} = +\infty$.
\item $\lim_{x\to1^-} \ln x = 0^-$ (αφού $x<1 \Rightarrow \ln x < 0$), άρα $\lim_{x\to1^-} \frac{e^x-1}{\ln x} = \frac{e-1}{0^-} = -\infty$.
\end{itemize}
Εφόσον τα πλευρικά όρια είναι διαφορετικά, το όριο $\lim_{x\to1} \frac{f(x)}{g(x)}$ \textbf{δεν υπάρχει}.

\item Εξετάζουμε την εξίσωση $k(x) = 1 \Leftrightarrow k(x) - 1 = 0$ στο $[1, e]$.
Θεωρούμε τη συνάρτηση $K(x) = k(x) - 1 = (e^x-1)\ln x - 1$, συνεχής στο $[1, e]$.
\begin{itemize}
\item $K(1) = (e-1)\ln 1 - 1 = 0 - 1 = -1 < 0$.
\item $K(e) = (e^e-1)\ln e - 1 = e^e - 1 - 1 = e^e - 2$. Επειδή $e > 2 \Rightarrow e^e > 2^2 = 4 \Rightarrow e^e-2 > 0$. 
Οπότε $K(e) > 0$.
\end{itemize}
Αφού $K(1)K(e) < 0$, σύμφωνα με το Θεώρημα \eng{Bolzano} έχει τουλάχιστον μία πραγματική ρίζα της εξίσωσης στο $(1, e)$.

\item Εξετάζουμε τις ασύμπτωτες της $h$.
Από το δεύτερο ερώτημα έχουμε:
\[ \lim_{x\to1^+} h(x)=+\infty \quad \text{και} \quad \lim_{x\to1^-} h(x)=-\infty. \]
Άρα η ευθεία $x=1$ είναι κατακόρυφη ασύμπτωτη της $C_h$.

Στο $0^+$ έχουμε:
\[ \lim_{x\to0^+} h(x)=\lim_{x\to0^+}\frac{e^x-1}{\ln x}=0, \]
οπότε δεν υπάρχει κατακόρυφη ασύμπτωτη στο $x=0$.

Στο $+\infty$:
\[ \lim_{x\to+\infty} h(x)=\lim_{x\to+\infty}\frac{e^x-1}{\ln x}\dlh{\frac{+\infty}{+\infty}}\lim_{x\to+\infty}\frac{e^x}{1/x}=\lim_{x\to+\infty}xe^x=+\infty. \]
Δεν υπάρχει οριζόντια ασύμπτωτη. Για πλάγια ασύμπτωτη ελέγχουμε:
\[ \lim_{x\to+\infty}\frac{h(x)}{x}=\lim_{x\to+\infty}\frac{e^x-1}{x\ln x}=+\infty, \]
άρα δεν υπάρχει ούτε πλάγια ασύμπτωτη στο $+\infty$.
\end{erwthma}
\end{thema}
