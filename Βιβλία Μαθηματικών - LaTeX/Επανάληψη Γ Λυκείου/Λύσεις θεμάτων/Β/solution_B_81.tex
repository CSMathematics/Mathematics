\begin{thema}{Β}
Δίνεται συνεχής συνάρτηση $f: (0, +\infty) \to \mathbb{R}$ τέτοια ώστε να ισχύει $f(e^x) = x e^{-x} + 2$ για κάθε $x \in \mathbb{R}$.

\begin{erwthma}

\item Θέτουμε $u = e^x$. Επειδή $x \in \mathbb{R}$, η μεταβλητή $u$ διατρέχει το διάστημα $(0, +\infty)$. 
Λύνοντας ως προς $x$, έχουμε $x = \ln u$.

Αντικαθιστώντας τη μεταβλητή στη δοσμένη σχέση προκύπτει:
\[ f(u) = (\ln u) e^{-\ln u} + 2 = (\ln u) \frac{1}{e^{\ln u}} + 2 = \frac{\ln u}{u} + 2 \]
Επομένως, για κάθε $x > 0$ (ή ομοίως με μεταβλητή $x$) ο τύπος της συνάρτησης είναι:
\[ f(x) = \frac{\ln x}{x} + 2 \]

\item Η $f(x)$ είναι παραγωγίσιμη στο $(0, +\infty)$ (ως πράξεις παραγωγίσιμων συναρτήσεων) με παράγωγο:
\[ f'(x) = \left( \frac{\ln x}{x} + 2 \right)' = \frac{(\ln x)' x - \ln x \cdot (x)'}{x^2} + 0 = \frac{\frac{1}{x} \cdot x - \ln x}{x^2} = \frac{1 - \ln x}{x^2} \]

Βρίσκουμε τις ρίζες και το πρόσημο της παραγώγου:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 1 - \ln x = 0 \Leftrightarrow \ln x = 1 \Leftrightarrow x = e$
    \item $f'(x) > 0 \Leftrightarrow \frac{1 - \ln x}{x^2} > 0 \stackrel{x^2 > 0}{\Leftrightarrow} 1 - \ln x > 0 \Leftrightarrow \ln x < \ln e \Leftrightarrow x < e$
    \item $f'(x) < 0 \Leftrightarrow x > e$
\end{itemize}

Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Συνεπώς η $f$ είναι γνησίως αύξουσα στο $(0, e]$ και γνησίως φθίνουσα στο $[e, +\infty)$. Στο $x_0 = e$ παρουσιάζει \textbf{ολικό μέγιστο} με τιμή $f(e) = \frac{\ln e}{e} + 2 = \frac{1}{e} + 2$.

Υπολογίζουμε το σύνολο τιμών $f(A)$:
\begin{itemize}
    \item Η $f$ είναι γνησίως αύξουσα και συνεχής στο $\varDelta_1 = (0, e]$, άρα $f(\varDelta_1) = \left( \lim\limits_{x\to0^+} f(x), f(e) \right]$.
    \[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} \left( \frac{1}{x} \ln x + 2 \right) = (+\infty)(-\infty) + 2 = -\infty \]
    Οπότε $f(\varDelta_1) = \left( -\infty,  \frac{1}{e} + 2 \right]$.
    \item Η $f$ είναι γνησίως φθίνουσα και συνεχής στο $\varDelta_2 = [e, +\infty)$, άρα $f(\varDelta_2) = \left( \lim\limits_{x\to+\infty} f(x), f(e) \right]$.
    \[ \lim_{x\to+\infty} \frac{\ln x}{x} \xlongequal[\text{DLH}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{(\ln x)'}{(x)'} = \lim_{x\to+\infty} \frac{\frac{1}{x}}{1} = 0 \]
    Επομένως, $\lim\limits_{x\to+\infty} f(x) = 0 + 2 = 2$.
    Οπότε $f(\varDelta_2) = \left( 2, \frac{1}{e} + 2 \right]$.
\end{itemize}
Το συνολικό σύνολο τιμών είναι η ένωση: 
\[ f(A) = f(\varDelta_1) \cup f(\varDelta_2) = \left( -\infty,  \frac{1}{e} + 2 \right] \]

\item Για $x > 0$ έχουμε την εξίσωση:
\[ \ln x = (\lambda - 2)x \Leftrightarrow \frac{\ln x}{x} = \lambda - 2 \Leftrightarrow \frac{\ln x}{x} + 2 = \lambda \Leftrightarrow f(x) = \lambda \]
Το πλήθος των ριζών της εξίσωσης αντιστοιχεί στο πλήθος των σημείων τομής της γραφικής παράστασης $C_f$ με την οριζόντια ευθεία $y = \lambda$. Εξετάζοντας τα επιμέρους σύνολα τιμών, προκύπτει:
\begin{itemize}
    \item Αν $\lambda > \frac{1}{e} + 2$, το $\lambda \notin f(A)$, άρα η εξίσωση είναι \textbf{αδύνατη} (0 ρίζες).
    \item Αν $\lambda = \frac{1}{e} + 2$, η ευθεία εφάπτεται στο ολικό μέγιστο της συνάρτησης, οπότε η εξίσωση έχει \textbf{ακριβώς 1 ρίζα} (τη $x = e$).
    \item Αν $2 < \lambda < \frac{1}{e} + 2$, το $\lambda$ ανήκει ταυτόχρονα στο $f(\varDelta_1)$ και στο $f(\varDelta_2)$. Καθώς η $f$ είναι γνησίως μονότονη σε κάθε υποδιάστημα, η εξίσωση έχει \textbf{ακριβώς 2 ρίζες} (μία στο $(0, e)$ και μία στο $(e, +\infty)$).
    \item Αν $\lambda \le 2$, το $\lambda \in f(\varDelta_1)$ αλλά $\lambda \notin f(\varDelta_2)$. Έτσι, η εξίσωση έχει \textbf{ακριβώς 1 ρίζα} στο $\varDelta_1$ (δηλαδή στο διάστημα $(0, e)$).
\end{itemize}

\item Ζητείται το όριο $\lim\limits_{x \to 0^+} \left[ x^2 f(x) \right]$.
Αντικαθιστώντας τον τύπο της $f$:
\[ x^2 f(x) = x^2 \left( \frac{\ln x}{x} + 2 \right) = x \ln x + 2x^2 \]
Υπολογίζουμε τα όρια των επιμέρους όρων:
\begin{itemize}
    \item $\lim\limits_{x\to0^+} 2x^2 = 0$.
    \item Για το δεύτερο, παρουσιάζεται απροσδιοριστία μορφής $0 \cdot (-\infty)$, την οποία μετατρέπουμε και εφαρμόζουμε DLH:
    \[ \lim_{x\to0^+} (x \ln x) = \lim_{x\to0^+} \frac{\ln x}{\frac{1}{x}} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to0^+} \frac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x\to0^+} (-x) = 0 \]
\end{itemize}
Εφόσον τα δύο όρια υπάρχουν και είναι πραγματικοί αριθμοί, προσθέτοντας βρίσκουμε τελικά ότι:
\[ \lim_{x \to 0^+} \left[ x^2 f(x) \right] = 0 + 0 = 0 \]

\end{erwthma}
\end{thema}
