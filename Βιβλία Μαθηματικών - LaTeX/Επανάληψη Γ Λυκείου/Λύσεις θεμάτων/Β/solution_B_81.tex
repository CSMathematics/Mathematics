\begin{thema}{Β}
Δίνεται συνεχής συνάρτηση $f:(0,+\infty)\to\mathbb{R}$ τέτοια ώστε
\[
f(e^x)=xe^{-x}+2,\quad x\in\mathbb{R}.
\]

\begin{erwthma}
\item Θέτουμε $u=e^x$. Τότε $u>0$ και $x=\ln u$, οπότε
\[
f(u)=(\ln u)e^{-\ln u}+2=\frac{\ln u}{u}+2.
\]
Επομένως
\[
f(x)=\frac{\ln x}{x}+2,\quad x>0.
\]

\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=\left(\frac{\ln x}{x}+2\right)'=\frac{1-\ln x}{x^2}.
\]
Επειδή $x^2>0$, το πρόσημο της $f'$ καθορίζεται από το $1-\ln x$:
\begin{itemize}
\item $f'(x)>0\Leftrightarrow 0<x<e$,
\item $f'(x)=0\Leftrightarrow x=e$,
\item $f'(x)<0\Leftrightarrow x>e$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,e]$ και \textbf{γνησίως φθίνουσα} στο $[e,+\infty)$. Παρουσιάζει ολικό μέγιστο στο $x=e$, με
\[
f(e)=2+\frac{1}{e}.
\]
Επιπλέον
\[
\lim_{x\to0^+}f(x)=-\infty
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=2.
\]
Στο $(0,e]$ η $f$ παίρνει όλες τις τιμές από $-\infty$ μέχρι $2+\frac{1}{e}$, ενώ στο $[e,+\infty)$ παίρνει τιμές στο $\left(2,2+\frac{1}{e}\right]$. Επομένως το σύνολο τιμών της είναι
\[
f((0,+\infty))=\left(-\infty,2+\frac{1}{e}\right].
\]

\item Για $x>0$ έχουμε
\[
\ln x=(\lambda-2)x
\Leftrightarrow \frac{\ln x}{x}+2=\lambda
\Leftrightarrow f(x)=\lambda.
\]
Άρα το πλήθος των ριζών της εξίσωσης είναι το πλήθος των λύσεων της $f(x)=\lambda$.

Από τη μονοτονία και τα σύνολα τιμών στα διαστήματα $(0,e]$ και $[e,+\infty)$ προκύπτει:
\begin{itemize}
\item αν $\lambda>2+\dfrac{1}{e}$, δεν υπάρχει ρίζα,
\item αν $\lambda=2+\dfrac{1}{e}$, υπάρχει ακριβώς μία ρίζα, η $x=e$,
\item αν $2<\lambda<2+\dfrac{1}{e}$, υπάρχουν ακριβώς δύο ρίζες, μία στο $(0,e)$ και μία στο $(e,+\infty)$,
\item αν $\lambda\leq2$, υπάρχει ακριβώς μία ρίζα, στο $(0,e]$.
\end{itemize}

\item Με τον τύπο της $f$ έχουμε
\[
x^2f(x)=x^2\left(\frac{\ln x}{x}+2\right)=x\ln x+2x^2.
\]
Επίσης
\[
\lim_{x\to0^+}x\ln x
=\lim_{x\to0^+}\frac{\ln x}{1/x}
\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to0^+}(-x)=0.
\]
Άρα
\[
\lim_{x\to0^+}x^2f(x)=0.
\]
\end{erwthma}
\end{thema}
