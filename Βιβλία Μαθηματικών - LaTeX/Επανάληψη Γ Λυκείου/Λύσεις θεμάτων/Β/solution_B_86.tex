\begin{thema}{Β}
Δίνεται συνάρτηση $f:\mathbb{R}\to\mathbb{R}$ τέτοια ώστε
\[
\left|xf(x)-\hm x\right|\leq |x|^3,\quad x\neq0,
\]
και η $f$ είναι συνεχής στο $0$.

\begin{erwthma}
\item Από τη δοσμένη ανισότητα έχουμε, για $x\neq0$,
\[
\left|f(x)-\frac{\hm x}{x}\right|\leq x^2,
\]
οπότε
\[
-x^2+\frac{\hm x}{x}\leq f(x)\leq x^2+\frac{\hm x}{x},\quad x\neq0.
\]

\item Από το προηγούμενο ερώτημα και επειδή
\[
\lim_{x\to0}\left(-x^2+\frac{\hm x}{x}\right)=1
\quad\text{και}\quad
\lim_{x\to0}\left(x^2+\frac{\hm x}{x}\right)=1,
\]
με το κριτήριο παρεμβολής προκύπτει
\[
\lim_{x\to0}f(x)=1.
\]
Επειδή η $f$ είναι συνεχής στο $0$, έχουμε
\[
f(0)=1.
\]

\item Για $x\neq0$ γράφουμε
\[
\frac{f(x)-f(0)}{x}
=\frac{f(x)-1}{x}.
\]
Από το φράγμα του πρώτου ερωτήματος προκύπτει
\[
\left|f(x)-\frac{\hm x}{x}\right|\leq x^2
\Rightarrow
\left|\frac{f(x)-\frac{\hm x}{x}}{x}\right|\leq |x|.
\]
Άρα
\[
\lim_{x\to0}\frac{f(x)-\frac{\hm x}{x}}{x}=0.
\]
Επίσης
\[
\lim_{x\to0}\frac{\frac{\hm x}{x}-1}{x}
=\lim_{x\to0}\frac{\hm x-x}{x^2}
\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{\syn x-1}{2x}
\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{-\hm x}{2}=0.
\]
Επομένως
\[
\lim_{x\to0}\frac{f(x)-f(0)}{x}=0,
\]
δηλαδή η $f$ είναι παραγωγίσιμη στο $0$ και
\[
f'(0)=0.
\]

\item Αν η $f$ είναι άρτια και παραγωγίσιμη στο $0$, τότε για κάθε $x\neq0$
\[
\frac{f(x)-f(0)}{x}
=-\frac{f(-x)-f(0)}{-x}.
\]
Περνώντας στο όριο για $x\to0$, παίρνουμε
\[
f'(0)=-f'(0)\Leftrightarrow f'(0)=0.
\]
Άρα η αρτιότητα επιβεβαιώνει ότι η παράγωγος στο $0$, εφόσον υπάρχει, είναι ίση με $0$.
\end{erwthma}
\end{thema}
