\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \ln(e^x + 1) - x$.

\begin{erwthma}
\item Το πεδίο ορισμού είναι $D_f = \mathbb{R}$ αφού $e^x+1>0$.
Από ιδιότητες λογαρίθμων:
\[ f(x) = \ln(e^x + 1) - \ln(e^x) = \ln\left( \frac{e^x+1}{e^x} \right) = \ln(1 + e^{-x}) \]
Τα όρια υπολογίζονται πλέον εύκολα:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} \ln(1 + e^{-x}) \]
Επειδή $\lim_{x\to-\infty} e^{-x} = +\infty$, το όριο είναι $\ln(+\infty) = +\infty$.
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \ln(1 + e^{-x}) = \ln(1+0) = 0 \]

\item \textbf{Στο $+\infty$}: 
Αφού $\lim_{x\to+\infty} f(x) = 0$, η ευθεία $y = 0$ (άξονας $x'x$) είναι οριζόντια ασύμπτωτη.
\textbf{Στο $-\infty$}:
Αναζητούμε πλάγια ασύμπτωτη $y = \lambda x + \beta$.
- Υπολογίζουμε το $\lambda = \lim_{x\to-\infty} \frac{f(x)}{x} = \lim_{x\to-\infty} \left( \frac{\ln(e^x+1)}{x} - 1 \right)$.
Επειδή $\lim_{x\to-\infty} \ln(e^x+1) = \ln(0+1) = 0$ και ο παρονομαστής $x \to -\infty$, το όριο $\frac{\ln(e^x+1)}{x}$ είναι $0$. Οπότε $\lambda = -1$.
- Υπολογίζουμε τη $\beta = \lim_{x\to-\infty} (f(x) - \lambda x) = \lim_{x\to-\infty} (\ln(e^x+1) - x - (-x)) = \lim_{x\to-\infty} \ln(e^x+1) = \ln 1 = 0$.
Άρα η ευθεία $y = -x$ είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη με $f'(x) = \frac{e^x}{e^x+1} - 1 = \frac{e^x - (e^x+1)}{e^x+1} = \frac{-1}{e^x+1}$.
Δεδομένου ότι το $e^x+1 > 0$, είναι $f'(x) < 0$ για κάθε $x \in \mathbb{R}$.
Άρα η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$, άρα είναι «1-1» και επομένως αντιστρέφεται.
Το πεδίο ορισμού της $f^{-1}$ είναι το σύνολο τιμών της $f$, το οποίο αφού η $f$ είναι γνησίως φθίνουσα και συνεχής, είναι:
\[ f(D_f) = \left( \lim_{x\to+\infty} f(x), \lim_{x\to-\infty} f(x) \right) = (0, +\infty) \]
Θέτουμε $y = f(x)$:
\[ y = \ln(1+e^{-x}) \Leftrightarrow e^y = 1+e^{-x} \Leftrightarrow e^{-x} = e^y - 1 \]
Για το $y > 0$, η ποσότητα $e^y - 1$ είναι θετική, οπότε μπορούμε να λογαριθμίσουμε:
\[ -x = \ln(e^y - 1) \Leftrightarrow x = -\ln(e^y-1) \]
Άρα ο τύπος της είναι $f^{-1}(x) = -\ln(e^x - 1)$, με $x \in (0, +\infty)$.

\item Εξετάζουμε την εξίσωση $f(x) = x \Leftrightarrow f(x) - x = 0$.
Θεωρούμε τη συνάρτηση $g(x) = f(x) - x$ με $D_g = \mathbb{R}$. H $g$ είναι συνεχής σε κάθε κλειστό διάστημα όπως το $[0, 1]$.
Βρίσκουμε τα $g(0)$ και $g(1)$:
- $g(0) = f(0) - 0 = \ln(1+1) = \ln 2 > 0$.
- $g(1) = \ln(1+e^{-1}) - 1 = \ln\left(\frac{e+1}{e}\right) - \ln e = \ln(e+1) - 2\ln e = \ln(e+1) - 2$.
Γνωρίζουμε ότι $e+1 < e^2$ (αφού $\approx 3.71 < 7.38$), οπότε $\ln(e+1) < \ln(e^2) = 2 \Rightarrow \ln(e+1) - 2 < 0$. Άρα $g(1) < 0$.
Ισχύει $g(0)g(1) < 0$. Από Θεώρημα Bolzano υπάρχει τουλάχιστον μία ρίζα $x_0 \in (0, 1)$.
Επίσης $g'(x) = f'(x) - 1 < 0$ αφού $f'(x) < 0$. Άρα η $g$ είναι γνησίως φθίνουσα, που σημαίνει ότι η ρίζα είναι μοναδική.
\end{erwthma}
\end{thema}
