%=========== ΛΥΣΗ - 98ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Η συνάρτηση γράφεται
\[
f(x)=|x-2|-1=
\begin{cases}
1-x, & x<2,\\
x-3, & x\geq 2.
\end{cases}
\]
Στον αριστερό κλάδο $1-x$ η κλίση είναι $-1$, άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty,2]$. Στον δεξιό κλάδο $x-3$ η κλίση είναι $1$, άρα η $f$ είναι γνησίως αύξουσα στο $[2,+\infty)$. Η ελάχιστη τιμή της είναι $f(2)=-1$, οπότε το σύνολο τιμών είναι $[-1,+\infty)$.

Για την εξίσωση $f(x)=\lambda$:
\begin{itemize}
\item αν $\lambda<-1$, δεν έχει καμία λύση,
\item αν $\lambda=-1$, έχει μία λύση, την $x=2$,
\item αν $\lambda>-1$, έχει δύο λύσεις, μία σε κάθε κλάδο.
\end{itemize}

\item Στο $x_0=2$ η γραφική παράσταση σχηματίζει γωνία. Το επιβεβαιώνουμε και από τις πλευρικές παραγώγους. Επειδή $f(2)=-1$, έχουμε
\[
f'_-(2)=\lim_{x\to2^-}\frac{f(x)-f(2)}{x-2}
=\lim_{x\to2^-}\frac{(1-x)+1}{x-2}
=\lim_{x\to2^-}\frac{2-x}{x-2}
=-1,
\]
ενώ
\[
f'_+(2)=\lim_{x\to2^+}\frac{f(x)-f(2)}{x-2}
=\lim_{x\to2^+}\frac{(x-3)+1}{x-2}
=\lim_{x\to2^+}\frac{x-2}{x-2}
=1.
\]
Επειδή $f'_-(2)\ne f'_+(2)$, δεν υπάρχει μοναδική εφαπτομένη. Επομένως η $C_f$ δεν δέχεται εφαπτομένη στο σημείο με τετμημένη $2$.

Στο $x_1=4$ βρισκόμαστε στον δεξιό κλάδο $f(x)=x-3$, που έχει κλίση $1$. Επειδή $f(4)=1$, η εφαπτομένη στο $(4,1)$ είναι
\[
y-1=1(x-4)\Leftrightarrow y=x-3.
\]

\item Επειδή $f(1)=0$, το όριο
\[
\lim_{x\to1}\frac{f(x)}{x-1}
=\lim_{x\to1}\frac{f(x)-f(1)}{x-1}
\]
εκφράζει την κλίση της εφαπτομένης στο σημείο της $C_f$ με τετμημένη $1$. Κοντά στο $1$ ισχύει $f(x)=1-x$, άρα
\[
\lim_{x\to1}\frac{f(x)}{x-1}
=\lim_{x\to1}\frac{1-x}{x-1}=-1.
\]

Αντίστοιχα, επειδή $f(3)=0$ και κοντά στο $3$ ισχύει $f(x)=x-3$, έχουμε
\[
\lim_{x\to3}\frac{f(x)}{x-3}
=\lim_{x\to3}\frac{x-3}{x-3}=1.
\]
Η τιμή αυτή εκφράζει την κλίση της εφαπτομένης στο σημείο της $C_f$ με τετμημένη $3$.

\item Οι τομές της $C_f$ με τον άξονα $x'x$ είναι στα $x=1$ και $x=3$, και στο $[1,3]$ η γραφική παράσταση βρίσκεται κάτω από τον άξονα. Άρα
\[
E=\int_1^3 |f(x)|\,\mathrm{d}x=-\int_1^3 f(x)\,\mathrm{d}x.
\]
Γεωμετρικά το χωρίο είναι τρίγωνο με βάση $3-1=2$ και ύψος $1$, οπότε
\[
E=\frac12\cdot2\cdot1=1.
\]
\end{erwthma}
\end{thema}
