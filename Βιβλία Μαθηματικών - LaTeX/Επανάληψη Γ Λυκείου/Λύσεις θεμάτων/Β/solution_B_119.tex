\begin{thema}{Β}
Στο παρακάτω αναβαθμισμένο σχήμα σημειώνονται οι εφαπτόμενες και τα κρίσιμα σημεία.
\begin{center}
\begin{tikzpicture}[scale=0.8]
  \draw[help lines, gray!30] (-3,-2) grid (4,4);
  \draw[-latex] (-3.3,0) -- (4.3,0) node[right] {$x$};
  \draw[-latex] (0,-2.3) -- (0,4.3) node[above] {$y$};
  \foreach \x in {-2,-1,1,2,3}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-1,1,2,3}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % curve
  \draw[very thick, blue!70!black, smooth, tension=0.5]
    plot coordinates {(-2,0) (-1,0.8) (0,2) (1,2.5) (2,0) (3,-1)};
  % tangent at A(0,2), slope 1
  \draw[dashed, red!60!black] (-1.5,0.5) -- (1.5,3.5);
  \node[red!60!black,font=\footnotesize] at (1.5,3.8) {$\varepsilon_A: y = x + 2$};
  % tangent at B(2,0), slope -2
  \draw[dashed, red!60!black] (1,2) -- (3,-2);
  \node[red!60!black,font=\footnotesize] at (3.2,-1.5) {$\varepsilon_B: y = -2x + 4$};
  
  \filldraw[blue!70!black] (0,2) circle (2pt) node[above left,font=\footnotesize] {$A(0,2)$};
  \filldraw[blue!70!black] (2,0) circle (2pt) node[below left,font=\footnotesize] {$B(2,0)$};
  
  % Σχόλια
  \node[anchor=south west, color=blue, font=\footnotesize] at (0.2, 2.1) {$f'(0)=1$};
  \node[anchor=north west, color=blue, font=\footnotesize] at (2.1, -0.3) {$f'(2)=-2$};
\end{tikzpicture}
\end{center}

\begin{erwthma}

\setcounter{enumi}{1}
\item Η εξίσωση εφαπτομένης σε σημείο $(x_0, f(x_0))$ είναι $y - f(x_0) = f'(x_0)(x - x_0)$.
\begin{itemize}
    \item \textbf{Εφαπτομένη $\varepsilon_A$ στο $A(0, 2)$} με κλίση $f'(0) = 1$:
    \[ y - 2 = 1 \cdot (x - 0) \Leftrightarrow \varepsilon_A: y = x + 2 \]
    \item \textbf{Εφαπτομένη $\varepsilon_B$ στο $B(2, 0)$} με κλίση $f'(2) = -2$:
    \[ y - 0 = -2(x - 2) \Leftrightarrow \varepsilon_B: y = -2x + 4 \]
\end{itemize}

\setcounter{enumi}{3}
\item Δίνεται $h(x) = f(x) \cdot x$. Εφαρμόζουμε τον \textbf{κανόνα γινομένου}:
\[ h'(x) = f'(x) \cdot x + f(x) \cdot 1 = x \cdot f'(x) + f(x) \]
\begin{itemize}
    \item $h'(0) = 0 \cdot f'(0) + f(0) = 0 + 2 = \mathbf{2}$.
    \item $h'(2) = 2 \cdot f'(2) + f(2) = 2(-2) + 0 = -4 + 0 = \mathbf{-4}$.
\end{itemize}

\end{erwthma}
\end{thema}
