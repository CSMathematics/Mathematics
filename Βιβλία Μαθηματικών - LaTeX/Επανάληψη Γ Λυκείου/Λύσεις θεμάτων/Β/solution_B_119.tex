\begin{thema}{Β}
Από το σχήμα έχουμε $f(-2)=0$, $f(2)=0$, $f(0)=2$, ενώ οι εφαπτόμενες στα $A(0,2)$ και $B(2,0)$ έχουν αντίστοιχα κλίσεις $1$ και $-2$.

\begin{erwthma}
\item Η $f$ είναι συνεχής στο $[-2,2]$ και παραγωγίσιμη στο $(-2,2)$.

Από το σχήμα έχουμε
\[
f(-2)=0
\quad\text{και}\quad
f(2)=0.
\]
Άρα
\[
f(-2)=f(2).
\]
Σύμφωνα με το Θεώρημα \eng{Rolle}, υπάρχει $\xi\in(-2,2)$ τέτοιο ώστε
\[
f'(\xi)=0.
\]
Γεωμετρικά, στο σημείο της $C_f$ με τετμημένη $\xi$ η εφαπτομένη είναι παράλληλη στον άξονα $x'x$, δηλαδή είναι οριζόντια.

\item Η εφαπτομένη στο $A(0,2)$ έχει κλίση $1$. Άρα
\[
y-2=1(x-0)\Rightarrow \varepsilon_A:\ y=x+2.
\]

Η εφαπτομένη στο $B(2,0)$ έχει κλίση $-2$. Άρα
\[
y-0=-2(x-2)\Rightarrow \varepsilon_B:\ y=-2x+4.
\]

\item Από το σχήμα έχουμε $f(0)=2$ και η κλίση της εφαπτομένης στο $A(0,2)$ είναι $f'(0)=1$. Επομένως
\[
\lim_{x\to 0}\frac{f(x)-2}{x}
=\lim_{x\to 0}\frac{f(x)-f(0)}{x-0}
=f'(0)=1.
\]

Επίσης $f(2)=0$ και η κλίση της εφαπτομένης στο $B(2,0)$ είναι $f'(2)=-2$. Άρα
\[
\lim_{x\to 2}\frac{f(x)}{x-2}
=\lim_{x\to 2}\frac{f(x)-f(2)}{x-2}
=f'(2)=-2.
\]

\item Δίνεται
\[
h(x)=x f(x).
\]
Με τον κανόνα γινομένου έχουμε
\[
h'(x)=f(x)+x f'(x).
\]
Άρα, επειδή $f(0)=2$ και $f'(0)=1$,
\[
h'(0)=f(0)+0\cdot f'(0)=2.
\]
Επίσης, επειδή $f(2)=0$ και $f'(2)=-2$,
\[
h'(2)=f(2)+2f'(2)=0+2(-2)=-4.
\]
\end{erwthma}
\end{thema}
