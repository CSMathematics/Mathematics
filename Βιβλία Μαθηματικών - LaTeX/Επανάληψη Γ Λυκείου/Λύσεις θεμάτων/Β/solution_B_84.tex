\begin{thema}{Β}
\begin{erwthma}
\item Θεωρούμε $g(x)=x+e^x,\ x\in\mathbb{R}$. Είναι $g'(x)=1+e^x>0$ για κάθε $x\in\mathbb{R}$, άρα η $g$ είναι γνησίως αύξουσα και επομένως $1-1$.

Η δοσμένη σχέση γράφεται $g(f(x))=x+1$. Για $x=0$ παίρνουμε $g(f(0))=1=g(0)$, οπότε, επειδή η $g$ είναι $1-1$, προκύπτει $f(0)=0$.

\item Παραγωγίζοντας τη σχέση $f(x)+e^{f(x)}=x+1$, έχουμε
\[
f'(x)+e^{f(x)}f'(x)=1\Leftrightarrow f'(x)\left(1+e^{f(x)}\right)=1
\Leftrightarrow f'(x)=\frac{1}{1+e^{f(x)}}.
\]
Επειδή $e^{f(x)}>0$, έχουμε $f'(x)>0$ για κάθε $x\in\mathbb{R}$.

\item Από το προηγούμενο ερώτημα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$ και από το πρώτο $f(0)=0$. Άρα
\[
f(x^2-x)>0\Leftrightarrow f(x^2-x)>f(0)\Leftrightarrow x^2-x>0
\Leftrightarrow x(x-1)>0.
\]
Επομένως $x\in(-\infty,0)\cup(1,+\infty)$.

\item Θέλουμε $f(x_0)=\dfrac12$. Από τη σχέση $g(f(x))=x+1$ παίρνουμε
\[
g\left(\frac12\right)=x_0+1\Leftrightarrow \frac12+\sqrt{e}=x_0+1
\Leftrightarrow x_0=\sqrt{e}-\frac12.
\]
Η τιμή αυτή είναι μοναδική, αφού η $f$ είναι γνησίως αύξουσα. Επιπλέον, επειδή $\dfrac94<e<\dfrac{25}{4}$, έχουμε $\dfrac32<\sqrt e<\dfrac52$, άρα
\[
1<\sqrt e-\frac12<2.
\]
Επομένως $x_0\in(1,2)$.
\end{erwthma}
\end{thema}
