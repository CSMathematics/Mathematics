\begin{thema}{Β}
Στο παρακάτω αναβαθμισμένο σχήμα σχεδιάστηκε η γραφική παράσταση $C_{f^{-1}}$ (ως αντικατοπτρισμός μέσω $y=x$) μαζί με την αρχική.
\begin{center}
\begin{tikzpicture}[scale=0.7]
  \draw[help lines, gray!30] (-3,-3) grid (5,5);
  \draw[-latex] (-3.3,0) -- (5.3,0) node[right] {$x$};
  \draw[-latex] (0,-3.3) -- (0,5.3) node[above] {$y$};
  \foreach \x in {-2,-1,1,2,3,4}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-2,-1,1,2,3,4}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % y=x
  \draw[dashed, gray!60] (-3,-3) -- (5,5);
  \node[gray!60,font=\footnotesize] at (4.5,4.5) {$y=x$};
  % f
  \draw[very thick, blue!70!black] (0,-2) -- (1,0) -- (2,1) -- (3,2) -- (4,3);
  \node[blue!70!black] at (0.5,1.5) {$C_f$};
  % f^{-1} (αντικατοπτρισμός: αντιστροφή x,y)
  \draw[very thick, red!70!black, dashed] (-2,0) -- (0,1) -- (1,2) -- (2,3) -- (3,4);
  \node[red!70!black] at (2.5,4.5) {$C_{f^{-1}}$};
  
  \filldraw[blue!70!black] (0,-2) circle (2pt);
  \filldraw[blue!70!black] (1,0) circle (2pt);
  \filldraw[blue!70!black] (2,1) circle (2pt);
  \filldraw[blue!70!black] (3,2) circle (2pt);
  \filldraw[blue!70!black] (4,3) circle (2pt);
  
  \filldraw[red!70!black] (-2,0) circle (2pt);
  \filldraw[red!70!black] (0,1) circle (2pt);
  \filldraw[red!70!black] (1,2) circle (2pt);
  \filldraw[red!70!black] (2,3) circle (2pt);
  \filldraw[red!70!black] (3,4) circle (2pt);
\end{tikzpicture}
\end{center}

\begin{erwthma}

\item Η $f$ είναι \textbf{γνησίως αύξουσα} στο $[0, 4]$ (δίνεται). Μια γνησίως μονότονη (αύξουσα ή φθίνουσα) συνάρτηση είναι \textbf{1-1}: 
\[ x_1 \ne x_2 \Rightarrow f(x_1) \ne f(x_2) \]
(αφού η αυξητική μονοτονία εγγυάται ότι δεν μπορούν δύο διαφορετικές τετμημένες να δώσουν ίδια τεταγμένη).
Επιπλέον, η $f$ ορίζεται σε κλειστό διάστημα, είναι συνεχής, και λαμβάνει κάθε ενδιάμεση τιμή μεταξύ $f(0) = -2$ και $f(4) = 3$, δηλαδή $f([0,4]) = [-2, 3]$. Εφόσον η $f$ είναι 1-1, είναι \textbf{αντιστρέψιμη}.

\item Η $f^{-1}$ αντιστρέφει ζεύγη: αν $f(a) = b$, τότε $f^{-1}(b) = a$.
\begin{itemize}
    \item $f(1) = 0 \Rightarrow f^{-1}(0) = 1$.
    \item $f(2) = 1 \Rightarrow f^{-1}(1) = 2$.
    \item $f(4) = 3 \Rightarrow f^{-1}(3) = 4$.
\end{itemize}

\item Η $C_{f^{-1}}$ κατασκευάζεται αντιστρέφοντας τα ζεύγη $(x, y)$ σε $(y, x)$, δηλαδή αντικατοπτρίζοντας ως προς τη διχοτόμο $y = x$. 
Τα σημεία της $C_{f^{-1}}$ είναι: $(-2, 0)$, $(0, 1)$, $(1, 2)$, $(2, 3)$, $(3, 4)$.
Η $C_{f^{-1}}$ έχει σχεδιαστεί (κόκκινη διακεκομμένη) στο αναβαθμισμένο σχήμα.

\item Ζητείται η κλίση της εφαπτομένης της $C_{f^{-1}}$ στο σημείο με τετμημένη $x = 1$.
Αυτό σημαίνει $(f^{-1})'(1)$. Γνωρίζουμε τον τύπο:
\[ (f^{-1})'(y_0) = \frac{1}{f'(f^{-1}(y_0))} \]
Εδώ $y_0 = 1$ και $f^{-1}(1) = 2$ (από ερ. β). Χρειαζόμαστε $f'(2)$: Η $C_f$ στο $[1, 2]$ έχει κλίση $\frac{1-0}{2-1} = 1$ και στο $[2, 3]$ επίσης $\frac{2-1}{3-2} = 1$. Οπότε $f'(2) = 1$.
\[ (f^{-1})'(1) = \frac{1}{f'(2)} = \frac{1}{1} = 1 \]
Η κλίση της εφαπτομένης είναι $\mathbf{1}$.

\end{erwthma}
\end{thema}
