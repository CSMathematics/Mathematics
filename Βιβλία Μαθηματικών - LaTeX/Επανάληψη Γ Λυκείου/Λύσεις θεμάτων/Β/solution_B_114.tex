\begin{thema}{Β}
Από το σχήμα η $f$ είναι γνησίως αύξουσα στο $[0,4]$ και η γραφική της παράσταση περνά από τα σημεία
\[
(0,-2),\ (1,0),\ (2,1),\ (3,2),\ (4,3).
\]

\begin{erwthma}
\item Επειδή η $f$ είναι γνησίως αύξουσα στο $[0,4]$, είναι $1-1$. Άρα είναι αντιστρέψιμη.

Από το σχήμα έχουμε
\[
f(0)=-2
\quad\text{και}\quad
f(4)=3.
\]
Επειδή η $f$ είναι γνησίως αύξουσα, το σύνολο τιμών της είναι
\[
f([0,4])=[-2,3].
\]
Επομένως
\[
D_{f^{-1}}=[-2,3]
\quad\text{και}\quad
f^{-1}([-2,3])=[0,4].
\]

\item Από το σχήμα διαβάζουμε:
\[
f(1)=0,\quad f(2)=1,\quad f(4)=3.
\]
Άρα
\[
f^{-1}(0)=1,\quad f^{-1}(1)=2,\quad f^{-1}(3)=4.
\]

\item Για μια γνησίως αύξουσα συνάρτηση $f$ ισχύει
\[
\dint_a^b f(x)\d x+\dint_{f(a)}^{f(b)} f^{-1}(x)\d x
=b f(b)-a f(a).
\]
Εδώ $a=0$, $b=4$, $f(0)=-2$ και $f(4)=3$. Άρα
\[
\dint_0^4 f(x)\d x+\dint_{-2}^{3} f^{-1}(x)\d x
=4\cdot 3-0\cdot(-2)=12.
\]
Δίνεται ότι
\[
\dint_0^4 f(x)\d x=2.
\]
Επομένως
\[
2+\dint_{-2}^{3} f^{-1}(x)\d x=12
\]
και άρα
\[
\dint_{-2}^{3} f^{-1}(x)\d x=10.
\]

\item Ζητείται η κλίση της εφαπτομένης της $C_{f^{-1}}$ στο σημείο με τετμημένη $x=1$, δηλαδή η τιμή $(f^{-1})'(1)$.

Από το προηγούμενο ερώτημα έχουμε
\[
f^{-1}(1)=2.
\]
Η κλίση της $C_f$ στο σημείο με τετμημένη $2$ είναι
\[
f'(2)=1,
\]
αφού αριστερά και δεξιά του $2$ η γραφική παράσταση είναι ευθύγραμμα τμήματα με κλίση $1$.

Άρα
\[
(f^{-1})'(1)=\frac{1}{f'(f^{-1}(1))}
=\frac{1}{f'(2)}
=1.
\]
Επομένως η ζητούμενη κλίση είναι $1$.
\end{erwthma}
\end{thema}
