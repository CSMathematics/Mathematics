\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x)=(x-a)e^x+\beta$.

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=\left[(x-a)e^x+\beta\right]'=e^x+(x-a)e^x=e^x(x-a+1).
\]
Επειδή η $f$ παρουσιάζει τοπικό ακρότατο στο εσωτερικό σημείο $x_0=1$, από το θεώρημα \eng{Fermat} έχουμε
\[
f'(1)=0\Leftrightarrow e(2-a)=0\Leftrightarrow a=2.
\]
Επιπλέον, η τιμή της $f$ στο σημείο αυτό είναι ίση με $2$. Άρα
\[
f(1)=2\Leftrightarrow (1-2)e+\beta=2\Leftrightarrow \beta=e+2.
\]
Επομένως
\[
a=2,\quad \beta=e+2.
\]

\item Για $a=2$ και $\beta=e+2$ είναι
\[
f(x)=(x-2)e^x+e+2
\quad\text{και}\quad
f'(x)=e^x(x-1).
\]
Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f'$ είναι το πρόσημο του $x-1$:
\begin{itemize}
\item $f'(x)<0\Leftrightarrow x<1$,
\item $f'(x)=0\Leftrightarrow x=1$,
\item $f'(x)>0\Leftrightarrow x>1$.
\end{itemize}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$. Συνεπώς παρουσιάζει \textbf{ολικό ελάχιστο} στο $x=1$, με τιμή
\[
f(1)=2.
\]

\item Από το προηγούμενο ερώτημα έχουμε $f(x)\geq f(1)$ για κάθε $x\in\mathbb{R}$, δηλαδή
\[
(x-2)e^x+e+2\geq 2\Leftrightarrow e^x(x-2)+e\geq 0.
\]
Επομένως
\[
e^x(x-2)+e\geq 0,\quad x\in\mathbb{R}.
\]

\item Για τις οριζόντιες ασύμπτωτες εξετάζουμε τα όρια της $f$ στα $\pm\infty$.

Στο $+\infty$ έχουμε
\[
\lim_{x\to+\infty}f(x)=\lim_{x\to+\infty}\left[(x-2)e^x+e+2\right]=+\infty,
\]
οπότε η $C_f$ δεν έχει οριζόντια ασύμπτωτη στο $+\infty$.

Στο $-\infty$ έχουμε $\lim\limits_{x\to-\infty}e^x=0$ και
\[
\lim_{x\to-\infty}xe^x
=\lim_{x\to-\infty}\frac{x}{e^{-x}}
\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to-\infty}\frac{1}{-e^{-x}}=0.
\]
Άρα
\[
\lim_{x\to-\infty}f(x)
=\lim_{x\to-\infty}\left(xe^x-2e^x+e+2\right)=e+2.
\]
Επομένως η ευθεία
\[
y=e+2
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.
\end{erwthma}
\end{thema}
