\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x)=\ln\!\left(1+\dfrac{1}{x}\right)$ και $g(x)=e^x$. Η συνάρτηση $g$ ορίζεται για κάθε $x\in\mathbb{R}$ ενώ η συνάρτηση $f$ όταν
\[1+\frac{1}{x}>0\Leftrightarrow \frac{x+1}{x}>0\Leftrightarrow x(x+1)>0\]
\begin{itemize}
\item $x=0$
\item $x+1=0\Leftrightarrow x=-1$
\end{itemize}

\end{center}
οπότε $D_f=(-\infty,-1)\cup (0,+\infty)$.
\begin{erwthma}
% -------------------------------------------------------
% α) Ορισμός h = f∘g
% -------------------------------------------------------
\item Η συνάρτηση $h=f\circ g$ έχει:
\begin{itemize}
    \item Τύπο: $h(x)=f(g(x))=f(e^x)=\ln\!\left(1+\dfrac{1}{e^x}\right)$
    \item Πεδίο ορισμού: $D_h=\{x\in D_g\mid g(x)\in D_f\}$.
    \begin{itemize}[label=\tiny\faPlay]
        \item $x\in D_g\Leftrightarrow x\in\mathbb{R}$ και
        \item $g(x)\in D_f\Leftrightarrow e^x\in(-\infty,-1)\cup(0,+\infty)$. Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, η συνθήκη ικανοποιείται πάντα.
    \end{itemize}
\end{itemize}
Άρα $D_h=\mathbb{R}$ και $h(x)=\ln\!\left(1+\dfrac{1}{e^x}\right)=\ln\!\left(1+e^{-x}\right)$, $x\in\mathbb{R}$.

\end{erwthma}

Αν $h(x)=\ln\!\left(1+\dfrac{1}{e^x}\right)$ με $x\in\mathbb{R}$:

\begin{erwthma}[resume]

% -------------------------------------------------------
% β) Αντιστρεψιμότητα και τύπος h^{-1}
% -------------------------------------------------------
\item Η $h$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως σύνθεση παραγωγίσιμων συναρτήσεων:
\[
h'(x)=\left[\ln\left(1+e^{-x}\right)\right]'=\frac{\left(1+e^{-x}\right)'}{1+e^{-x}}=\frac{-e^{-x}}{1+e^{-x}}<0
\]
για κάθε $x\in\mathbb{R}$, άρα έχουμε $h'(x)<0$. Η $h$ λοιπόν είναι γνησίως φθίνουσα στο $\mathbb{R}$, οπότε είναι $1-1$ και αντιστρέφεται. Αφού $h\searrow\mathbb{R}$ τότε
\[D_{h^{-1}}=h(D_h)=\left(\lim\limits_{x\to+\infty}{h(x)},\lim\limits_{x\to -\infty}{h(x)}\right)=(0,+\infty)\]
\begin{itemize}
\item $\lim\limits_{x\to+\infty}{h(x)}=\lim\limits_{x\to +\infty}{\ln\left(1+e^{-x}\right)}=\ln{1}=0$
\item $\lim\limits_{x\to -\infty}{h(x)}=\lim\limits_{x\to-\infty}{\ln\left(1+e^{-x}\right)}=+\infty$
\end{itemize}
Θέτουμε $y=h(x)\Leftrightarrow x=h^{-1}(y)$. Είναι:
\begin{align*}
y=h(x)&\Leftrightarrow y=\ln\left(1+e^{-x}\right)\Leftrightarrow e^y=1+e^{-x}\Leftrightarrow e^{-x}=e^y-1\Leftrightarrow\\&\Leftrightarrow -x=\ln(e^y-1)\Leftrightarrow x=-\ln(e^y-1)\Leftrightarrow h^{-1}(y)=-\ln(e^y-1)
\end{align*}
Άρα:
\[
h^{-1}(x)=-\ln(e^x-1),\quad x\in(0,+\infty).
\]

% -------------------------------------------------------
% γ) Ασύμπτωτες C_h και C_{h^{-1}}
% -------------------------------------------------------
\item \textbf{Ασύμπτωτες της $C_{h}$.}
Αφού όπως είδαμε πριν $\lim\limits_{x\to +\infty}{h(x)}=0$ τότε η ευθεία $y=0$ είναι οριζόντια ασύμπτωτη της $C_h$ στο $+\infty$. Εξετάζουμε επίσης για πλάγια ασύμπτωτη στο $-\infty$:
\begin{align*}
\lim\limits_{x\to -\infty}{\frac{h(x)}{x}}&=\lim\limits_{x\to -\infty}{\frac{\ln\left(1+e^{-x}\right)}{x}}\dlh{\frac{+\infty}{-\infty}}\lim\limits_{x\to -\infty}{\frac{-e^{-x}}{1+e^{-x}}}=-1=\lambda\\
\lim\limits_{x\to -\infty}{\left(h(x)+x\right)}&=\lim\limits_{x\to-\infty}{\left[\ln\left(1+e^{-x}\right)+x\right]}=\lim\limits_{x\to-\infty}{\ln{\left(e^x+1\right)}}=\ln1=0=\beta.
\end{align*}
άρα η ευθεία $y=-x$ είναι πλάγια ασύμπτωτη της $C_h$ στο $-\infty$.

\textbf{Ασύμπτωτες της $C_{h^{-1}}$.}
Το πεδίο ορισμού της $h^{-1}$ είναι $(0,+\infty)$.
\begin{itemize}
    \item Στο $x=0$:
\[\lim\limits_{x\to0^+}{h^{-1}(x)}=\lim\limits_{x\to0^+}{\left[-\ln\left(e^x-1\right)\right]}=+\infty\]
 άρα η ευθεία $x=0$ είναι κατακόρυφη ασύμπτωτη.
    \item Στο $+\infty$:
    \begin{align*}
	\lambda&=\lim_{x\to+\infty}\frac{h^{-1}(x)}{x}=\lim_{x\to+\infty}\frac{-\ln(e^x-1)}{x}\dlh{\frac{-\infty}{+\infty}}\lim_{x\to+\infty}{\frac{-e^x}{e^x-1}}=-1\\
    \beta&=\lim_{x\to+\infty}\bigl[h^{-1}(x)+x\bigr]=\lim_{x\to+\infty}\bigl[-\ln(e^x-1)+x\bigr]=\lim_{x\to+\infty}\ln\frac{e^x}{e^x-1}=\ln1=0.
	\end{align*}
    Η ευθεία $y=-x$ είναι πλάγια ασύμπτωτη της $C_{h^{-1}}$ στο $+\infty$.
\end{itemize}

% -------------------------------------------------------
% δ) Εφαπτομένη παράλληλη με ζ: y = -x/2 + 1
% -------------------------------------------------------
\item Η εφαπτομένη της $C_h$ πρέπει να έχει κλίση $-\dfrac{1}{2}$ (ίδια με την $\zeta: y=-\dfrac{x}{2}+1$).
Λύνουμε $h'(x)=-\dfrac{1}{2}$:
\[
\frac{-e^{-x}}{1+e^{-x}}=-\frac{1}{2}\Leftrightarrow 2e^{-x}=1+e^{-x}\Leftrightarrow e^{-x}=1\Leftrightarrow x=0.
\]
Σημείο επαφής: $h(0)=\ln\!\left(1+1\right)=\ln 2$.
Εξίσωση εφαπτομένης:
\[
y-\ln2=\left(-\frac{1}{2}\right)(x-0)\Leftrightarrow y=-\frac{x}{2}+\ln2.
\]

\end{erwthma}
\end{thema}
