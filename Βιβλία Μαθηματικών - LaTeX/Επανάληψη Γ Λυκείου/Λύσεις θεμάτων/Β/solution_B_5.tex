\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x)=x\ln{x}-x\ ,\ x>0$ και $g(x)=e^x\ ,\ x\in\mathbb{R}$.

\begin{erwthma}
\item Η συνάρτηση $h=f\circ g$ έχει:
\begin{itemize}
\item τύπο $h(x) = f(g(x)) = (e^x)\ln(e^x) - e^x = e^x \cdot x - e^x = e^x(x-1)$
\item πεδίο ορισμού $D_h = \{x\in D_g \mid g(x)\in D_f\}$
\begin{itemize}[label={\tiny \faPlay}]
\item $x\in D_g\Leftrightarrow x\in\mathbb{R}$ και
\item $g(x)\in D_f\Leftrightarrow e^x>0 \Leftrightarrow x\in\mathbb{R}$ αφού $e^x>0$ ισχύει για κάθε πραγματικό αριθμό $x$.
\end{itemize}
\end{itemize}
Άρα $h(x) = e^x(x-1)\ ,\ x\in\mathbb{R}$.

\item Η $h(x) = e^x(x-1)$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[h'(x) = (e^x(x-1))' = (e^x)'(x-1) + e^x(x-1)' = e^x(x-1) + e^x\cdot 1 = e^x(x-1+1) = xe^x\]
Βρίσκουμε ρίζες και πρόσημο της $h'$:
\begin{itemize}
    \item $h'(x) = 0 \Leftrightarrow xe^x = 0 \stackrel{e^x>0}{\Leftrightarrow} x = 0$
    \item $h'(x) > 0 \Leftrightarrow xe^x > 0 \stackrel{e^x>0}{\Leftrightarrow} x > 0$
    \item $h'(x) < 0 \Leftrightarrow x < 0$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $h'$ και η μονοτονία της $h$:
\begin{center}
\begin{tikzpicture}

\tkzTabVar{ +/, -/, +/, }
\end{tikzpicture}
\end{center}
Άρα η $h$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$ και γνησίως αύξουσα στο $[0, +\infty)$. Συνεπώς η $h$ παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή $h(0) = e^0(0-1) = -1$.

\item Η $h'$ είναι επίσης παραγωγίσιμη στο $\mathbb{R}$ με:
\[h''(x) = (xe^x)' = 1\cdot e^x + x\cdot e^x = (x+1)e^x\]
Για τις ρίζες και το πρόσημο της $h''$:
\begin{itemize}
    \item $h''(x) = 0 \Leftrightarrow (x+1)e^x = 0 \Leftrightarrow x = -1$
    \item $h''(x) > 0 \Leftrightarrow (x+1)e^x > 0 \Leftrightarrow x > -1$
    \item $h''(x) < 0 \Leftrightarrow x < -1$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $h''$ και η κυρτότητα της $h$:
\begin{center}

\tkzTabLine{ , \curvearrowright, t, \rotatebox[origin= c]{180}{$\curvearrowleft$}, }
\end{tikzpicture}
\end{center}
Άρα η $h$ είναι κοίλη (στρέφει τα κοίλα κάτω) στο $(-\infty, -1]$ και κυρτή (στρέφει τα κοίλα άνω) στο $[-1, +\infty)$.
Αφού η συνάρτηση αλλάζει κυρτότητα εκατέρωθεν του $x = -1$ και δέχεται εφαπτομένη, το σημείο της $C_h$: $M(-1, h(-1)) \equiv M(-1, -2e^{-1}) \equiv M\left(-1, -\dfrac{2}{e}\right)$ είναι σημείο καμπής.

\item Έστω $h(x)=0\Leftrightarrow e^x(x-1)=0\Leftrightarrow x-1=0\Leftrightarrow x=1$.
Άρα το εμβαδόν είναι:
\[ E = \int_{-2}^{1} |h(x)| \d x \]
Για κάθε $x\in[-2,1]$ είναι
\[x-1\leq 0\Leftrightarrow e^x(x-1)\leq 0\Leftrightarrow h(x)\leq 0\]
οπότε το εμβαδόν γίνεται
\begin{align*}
E&= \int_{-2}^{1} -h(x) \d x = -\int_{-2}^{1} e^x(x-1) \d x = -\int_{-2}^{1} e^x(x-1) \d x =\\&= -\left( [e^x(x-1)]_{-2}^{1} - \int_{-2}^{1} e^x \d x \right) = - ( 0 - (-3e^{-2}) - [e^x]_{-2}^{1} )=\\&= -(3e^{-2} - e + e^{-2}) = - (4e^{-2} - e) = e - 4e^{-2} = e - \dfrac{4}{e^2}
\end{align*}

\end{erwthma}
\end{thema}
