\begin{thema}{Β}
\begin{erwthma}
\item Η $g(x)=x^2$ είναι γνησίως αύξουσα στο $[0,+\infty)$, αφού για $0\le x_1<x_2$ ισχύει $x_1^2<x_2^2$. Άρα είναι $1-1$ και αντιστρέφεται. Αν $y=x^2$ με $x\ge0$, τότε $x=\sqrt y$, οπότε
\[
g^{-1}(x)=\sqrt x=f(x),\quad x\ge0.
\]
Για τις τομές λύνουμε:
\[
f(x)=g(x)\Leftrightarrow \sqrt x=x^2,\ x\ge0
\Leftrightarrow x=x^4\Leftrightarrow x(x^3-1)=0.
\]
Άρα $x=0$ ή $x=1$, δηλαδή τα σημεία τομής είναι $(0,0)$ και $(1,1)$. Γεωμετρικά, οι $C_f$ και $C_g$ είναι συμμετρικές ως προς την ευθεία $y=x$.

\item Είναι $g'(x)=2x$, άρα στο $x_0=1$ έχουμε $g(1)=1$ και $g'(1)=2$. Η εφαπτομένη είναι
\[
y-1=2(x-1)\Leftrightarrow y=2x-1.
\]
Επίσης, για $x\ge0$,
\[
g(x)\ge2x-1\Leftrightarrow x^2-2x+1\ge0\Leftrightarrow (x-1)^2\ge0.
\]

\item Από τη δοσμένη σχέση,
\[
x^2\hm\left(\frac1x\right)\le h(x)\le \sqrt x,\quad x\in(0,1].
\]
Επειδή $-1\le\hm\left(\frac1x\right)\le1$, έχουμε $-x^2\le x^2\hm\left(\frac1x\right)\le x^2$, άρα
\[
\lim_{x\to0^+}x^2\hm\left(\frac1x\right)=0.
\]
Επίσης $\lim\limits_{x\to0^+}\sqrt x=0$. Από το κριτήριο παρεμβολής προκύπτει $\lim\limits_{x\to0^+}h(x)=0$.

\item Οι καμπύλες τέμνονται στα $x=0$ και $x=1$. Στο $(0,1)$ ισχύει $\sqrt x>x^2$, άρα
\[
E=\int_0^1(\sqrt x-x^2)\,\mathrm{d}x
=\left[\frac23x^{3/2}-\frac{x^3}{3}\right]_0^1
=\frac13.
\]
\end{erwthma}
\end{thema}
