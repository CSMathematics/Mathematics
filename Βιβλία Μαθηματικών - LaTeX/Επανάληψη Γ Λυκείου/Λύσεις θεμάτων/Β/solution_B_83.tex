\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=e^x-\lambda x,\quad \lambda>0.
\]

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=e^x-\lambda.
\]
Επειδή $\lambda>0$, έχουμε
\begin{itemize}
\item $f'(x)=0\Leftrightarrow e^x=\lambda\Leftrightarrow x=\ln\lambda$,
\item $f'(x)<0\Leftrightarrow x<\ln\lambda$,
\item $f'(x)>0\Leftrightarrow x>\ln\lambda$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = red7,colorL = red9,colorV = red7]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$\ln\lambda$,$+\infty$}%
\tkzTabLine{ , -, z, +, }
\tkzTabLine{ , \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,\ln\lambda]$ και \textbf{γνησίως αύξουσα} στο $[\ln\lambda,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο
\[
x_0=\ln\lambda.
\]

\item Από το προηγούμενο ερώτημα, η ελάχιστη τιμή της $f$ είναι η
\[
f(\ln\lambda)=e^{\ln\lambda}-\lambda\ln\lambda
=\lambda-\lambda\ln\lambda.
\]

\item Για $\lambda=e$ έχουμε
\[
f(x)=e^x-ex.
\]
Από τα προηγούμενα, η $f$ παρουσιάζει ολικό ελάχιστο στο
\[
x_0=\ln e=1,
\]
με τιμή
\[
f(1)=e-e=0.
\]
Άρα $f(x)\geq0$ για κάθε $x\in\mathbb{R}$. Επειδή η $f$ είναι γνησίως φθίνουσα στο $(-\infty,1]$ και γνησίως αύξουσα στο $[1,+\infty)$, η τιμή $0$ λαμβάνεται μόνο στο $x=1$. Επομένως η εξίσωση $f(x)=0$ έχει μοναδική ρίζα
\[
x=1.
\]

\item Έχουμε
\[
\lim_{x\to+\infty}\frac{f(x)}{x^2}
=\lim_{x\to+\infty}\left(\frac{e^x}{x^2}-\frac{\lambda}{x}\right).
\]
Επειδή $\lim\limits_{x\to+\infty}\frac{\lambda}{x}=0$ και
\[
\lim_{x\to+\infty}\frac{e^x}{x^2}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{e^x}{2x}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{e^x}{2}=+\infty,
\]
προκύπτει
\[
\lim_{x\to+\infty}\frac{f(x)}{x^2}=+\infty.
\]
\end{erwthma}
\end{thema}
