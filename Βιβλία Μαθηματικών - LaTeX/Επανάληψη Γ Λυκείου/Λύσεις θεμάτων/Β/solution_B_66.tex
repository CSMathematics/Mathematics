\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2+ax+\beta}{x-1}$. Η $C_f$ έχει πλάγια ασύμπτωτη στο $+\infty$ την ευθεία $y=x+2$ και η εφαπτομένη της στο σημείο με τετμημένη $0$ είναι οριζόντια.

\begin{erwthma}

% -------------------------------------------------------
% α) Εύρεση a και β
% -------------------------------------------------------
\item Εκτελούμε διαίρεση πολυωνύμων ώστε να βρούμε την πλάγια ασύμπτωτη.
Διαιρούμε το $x^2+ax+\beta$ με το $x-1$:
\[
\frac{x^2+ax+\beta}{x-1} = x+(a+1)+\frac{\beta+a+1}{x-1}.
\]
Η πλάγια ασύμπτωτη στο $\pm\infty$ είναι η ευθεία $y=x+(a+1)$.
Αφού δίνεται ότι αυτή ισούται με $y=x+2$, έχουμε:
\[
a+1=2 \Rightarrow a=1.
\]

Άρα $f(x)=x+2+\dfrac{\beta+2}{x-1}$ και $f'(x)=1-\dfrac{\beta+2}{(x-1)^2}$.

Η εφαπτομένη στο $x_0=0$ είναι οριζόντια, άρα $f'(0)=0$:
\[
f'(0)=1-\frac{\beta+2}{(0-1)^2}=1-(\beta+2)=-\beta-1=0 \Rightarrow \beta=-1.
\]
Επομένως $a=1$ και $\beta=-1$. \quad $\square$

% -------------------------------------------------------
% β) Κατακόρυφη ασύμπτωτη
% -------------------------------------------------------
\item Με $a=1$, $\beta=-1$: $f(x)=\dfrac{x^2+x-1}{x-1}$.
Το πεδίο ορισμού είναι $D_f=\mathbb{R}\setminus\{1\}$.

Υπολογίζουμε τα πλευρικά όρια στο $x=1$:
\[
\lim_{x\to1^+}\frac{x^2+x-1}{x-1}=\frac{1+1-1}{0^+}=\frac{1}{0^+}=+\infty,\qquad \lim_{x\to1^-}\frac{x^2+x-1}{x-1}=-\infty.
\]
Άρα η ευθεία $x=1$ είναι \textbf{κατακόρυφη ασύμπτωτη} της $C_f$.

% -------------------------------------------------------
% γ) [Κενό ερώτημα στο πρωτότυπο]
% -------------------------------------------------------
\item \textit{(Το ερώτημα δεν συμπληρώθηκε στο πρωτότυπο αρχείο.)}

% -------------------------------------------------------
% δ) Τοπικά ακρότατα
% -------------------------------------------------------
\item Από το (α): $f'(x)=1-\dfrac{\beta+2}{(x-1)^2}=1-\dfrac{1}{(x-1)^2}=\dfrac{(x-1)^2-1}{(x-1)^2}=\dfrac{x^2-2x}{(x-1)^2}=\dfrac{x(x-2)}{(x-1)^2}$.

Βρίσκουμε ρίζες και πρόσημο της $f'$ ($(x-1)^2>0$ για $x\ne1$):
\begin{itemize}
    \item $f'(x)=0 \Leftrightarrow x(x-2)=0 \Leftrightarrow x=0$ ή $x=2$
    \item $f'(x)>0 \Leftrightarrow x(x-2)>0 \Leftrightarrow x<0$ ή $x>2$
    \item $f'(x)<0 \Leftrightarrow 0<x<1$ ή $1<x<2$
\end{itemize}

    \item Στο $x=0$: \textbf{τοπικό μέγιστο} με τιμή $f(0)=\dfrac{0+0-1}{0-1}=1$.
    \item Στο $x=2$: \textbf{τοπικό ελάχιστο} με τιμή $f(2)=\dfrac{4+2-1}{2-1}=5$.
\end{itemize}

\end{erwthma}
\end{thema}
