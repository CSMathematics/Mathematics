%=========== ΛΥΣΗ - 66ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=\frac{x^2+ax+\beta}{x-1},\quad x\neq1.
\]

\begin{erwthma}
\item Κάνουμε τη διαίρεση του $x^2+ax+\beta$ με το $x-1$:
\[
\frac{x^2+ax+\beta}{x-1}
=x+a+1+\frac{a+\beta+1}{x-1}.
\]
Άρα η πλάγια ασύμπτωτη της $C_f$ στο $+\infty$ είναι η ευθεία $y=x+a+1$. Επειδή δίνεται ότι είναι η $y=x+2$, έχουμε
\[
x+a+1=x+2\Leftrightarrow a=1.
\]
Οπότε
\[
f(x)=x+2+\frac{\beta+2}{x-1}
\quad\text{και}\quad
f'(x)=1-\frac{\beta+2}{(x-1)^2}.
\]
Η εφαπτομένη στο σημείο με τετμημένη $0$ είναι οριζόντια, άρα
\[
f'(0)=0\Leftrightarrow 1-(\beta+2)=0\Leftrightarrow \beta=-1.
\]
Επομένως
\[
a=1,\quad \beta=-1.
\]

\item Με $a=1$ και $\beta=-1$ έχουμε
\[
f(x)=\frac{x^2+x-1}{x-1}=x+2+\frac{1}{x-1},\quad x\neq1.
\]
Επειδή
\[
\lim_{x\to1^+}f(x)=+\infty
\quad\text{και}\quad
\lim_{x\to1^-}f(x)=-\infty,
\]
η ευθεία
\[
x=1
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

\item Για $x\neq1$ είναι
\[
f'(x)=1-\frac{1}{(x-1)^2}
=\frac{(x-1)^2-1}{(x-1)^2}
=\frac{x(x-2)}{(x-1)^2}.
\]
Επειδή $(x-1)^2>0$ για κάθε $x\neq1$, το πρόσημο της $f'$ καθορίζεται από το $x(x-2)$:
\begin{itemize}
\item $f'(x)>0\Leftrightarrow x<0$ ή $x>2$,
\item $f'(x)=0\Leftrightarrow x=0$ ή $x=2$,
\item $f'(x)<0\Leftrightarrow 0<x<1$ ή $1<x<2$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στα $(-\infty,0]$, $[2,+\infty)$ και \textbf{γνησίως φθίνουσα} στα $[0,1)$, $(1,2]$.

\item Από τον πίνακα μονοτονίας:
\begin{itemize}
\item στο $x=0$ η $f'$ αλλάζει πρόσημο από $+$ σε $-$, άρα η $f$ παρουσιάζει \textbf{τοπικό μέγιστο}
\[
f(0)=\frac{-1}{-1}=1,
\]
\item στο $x=2$ η $f'$ αλλάζει πρόσημο από $-$ σε $+$, άρα η $f$ παρουσιάζει \textbf{τοπικό ελάχιστο}
\[
f(2)=\frac{4+2-1}{2-1}=5.
\]
\end{itemize}
\end{erwthma}
\end{thema}
