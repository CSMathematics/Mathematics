\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 \ln x$ με $x > 0$.
\begin{erwthma}
\item Υπολογίζουμε το όριο στο $0^+$ γράφοντας το γινόμενο ως πηλίκο για να εφαρμόσουμε De L' Hospital:
\[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} (x^2 \ln x) = \lim_{x\to0^+} \frac{\ln x}{\frac{1}{x^2}} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to0^+} \frac{\frac{1}{x}}{-\frac{2}{x^3}} = \lim_{x\to0^+} \left( \frac{1}{x} \cdot \frac{x^3}{-2} \right) = \lim_{x\to0^+} \left( -\frac{x^2}{2} \right) = 0 \]

\item Για να αποτελεί η $g$ συνεχή επέκταση της $f$ στο $0$, πρέπει να ισχύει $g(0) = \lim_{x\to0^+} f(x)$.
Σύμφωνα με το προηγούμενο ερώτημα $\lim_{x\to0^+} f(x) = 0$, συνεπώς ορίζουμε $g(0) = 0$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για την $f$ στο διάστημα $[1, e]$:
- Η $f$ είναι συνεχής στο $[1, e]$.
- Η $f$ είναι παραγωγίσιμη στο $(1, e)$.
Ορίζονται τα σημεία $A(1, f(1))$ και $B(e, f(e))$. 
Υπολογίζουμε $f(1) = 1^2 \ln 1 = 0 \Rightarrow A(1, 0)$ και $f(e) = e^2 \ln e = e^2 \Rightarrow B(e, e^2)$.
Ο συντελεστής διεύθυνσης της τέμνουσας $AB$ είναι: $\lambda_{AB} = \frac{f(e) - f(1)}{e - 1} = \frac{e^2 - 0}{e-1} = \frac{e^2}{e-1}$.
Από το Θ.Μ.Τ. υπάρχει τουλάχιστον ένα $\xi \in (1, e)$ τέτοιο ώστε $f'(\xi) = \frac{e^2}{e-1} \Leftrightarrow f'(\xi) = \lambda_{AB}$.
Από τη γεωμετρική ερμηνεία, η εφαπτομένη της $C_f$ στο $x=\xi$ είναι παράλληλη προς την τέμνουσα $AB$.

\item Υπολογίζουμε την πρώτη και zweite παράγωγο:
\[ f'(x) = 2x \ln x + x^2 \frac{1}{x} = 2x \ln x + x \]
\[ f''(x) = 2\ln x + 2x \frac{1}{x} + 1 = 2\ln x + 2 + 1 = 2\ln x + 3 \]
Η δεύτερη παράγωγος μηδενίζεται: $f''(x) = 0 \Leftrightarrow 2\ln x = -3 \Leftrightarrow \ln x = -1,5 \Leftrightarrow x = e^{-3/2}$.
- Για $x > e^{-3/2}$, ισχύει $\ln x > -1,5 \Rightarrow 2\ln x + 3 > 0 \Rightarrow f''(x) > 0$. Η $f$ είναι \textbf{κυρτή} στο $[e^{-3/2}, +\infty)$.
- Για $0 < x < e^{-3/2}$, ισχύει $f''(x) < 0$. Η $f$ είναι \textbf{κοίλη} στο $(0, e^{-3/2}]$.

\end{erwthma}
\end{thema}
