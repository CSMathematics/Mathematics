\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 \ln x$ με $x > 0$.
\begin{erwthma}
\item Έχουμε
\[
\lim_{x\to0^+}x^2\ln x
=\lim_{x\to0^+}\frac{\ln x}{\frac{1}{x^2}}
\dlh{\frac{-\infty}{+\infty}}
\lim_{x\to0^+}\frac{\frac{1}{x}}{-\frac{2}{x^3}}
=\lim_{x\to0^+}\left(-\frac{x^2}{2}\right)=0.
\]

\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=2x\ln x+x=x(2\ln x+1).
\]
Η $C_f$ έχει οριζόντια εφαπτομένη στο σημείο με τετμημένη $x_0$ όταν
$f'(x_0)=0$. Άρα
\[
x_0(2\ln x_0+1)=0
\Leftrightarrow 2\ln x_0+1=0
\Leftrightarrow x_0=e^{-\frac12}.
\]
Επειδή $e^{-\frac12}\in(0,1)$, η $C_f$ έχει οριζόντια εφαπτομένη στο σημείο
\[
M\left(e^{-\frac12},f\left(e^{-\frac12}\right)\right)
\equiv M\left(e^{-\frac12},-\frac{1}{2e}\right).
\]

\item Η $f$ είναι συνεχής στο $[1,e]$ και παραγωγίσιμη στο $(1,e)$. Από το
θεώρημα μέσης τιμής υπάρχει $\xi\in(1,e)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(e)-f(1)}{e-1}.
\]
Επειδή $f(1)=0$ και $f(e)=e^2$, έχουμε
\[
f'(\xi)=\frac{e^2}{e-1}.
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $\xi$ είναι παράλληλη στο
τμήμα με άκρα $A(1,f(1))$ και $B(e,f(e))$.

\item Η δεύτερη παράγωγος της $f$ είναι
\[
f''(x)=(2x\ln x+x)'=2\ln x+3.
\]
Για τη ρίζα και το πρόσημο της $f''$ έχουμε
\begin{itemize}
\item $f''(x)=0 \Leftrightarrow 2\ln x+3=0 \Leftrightarrow x=e^{-\frac32}$,
\item $f''(x)<0 \Leftrightarrow 0<x<e^{-\frac32}$,
\item $f''(x)>0 \Leftrightarrow x>e^{-\frac32}$.
\end{itemize}
Επομένως η $f$ είναι κοίλη στο $\left(0,e^{-\frac32}\right]$ και κυρτή στο
$\left[e^{-\frac32},+\infty\right)$. Η $C_f$ παρουσιάζει σημείο καμπής στο
\[
K\left(e^{-\frac32},f\left(e^{-\frac32}\right)\right)
\equiv K\left(e^{-\frac32},-\frac{3}{2e^3}\right).
\]
\end{erwthma}
\end{thema}
