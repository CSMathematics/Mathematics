\begin{thema}{Β}
\begin{erwthma}
\item Από το σχήμα βλέπουμε ότι για $x\to+\infty$ η γραφική παράσταση πλησιάζει την ευθεία $y=2$, ενώ για $x\to-\infty$ πλησιάζει τον άξονα $x'x$. Άρα
\[
\lim_{x\to+\infty}g(x)=2,\quad \lim_{x\to-\infty}g(x)=0.
\]

\item Καθώς $x\to-\infty$, από το σχήμα έχουμε $g(x)\to0^+$, δηλαδή οι τιμές του $g(x)$ πλησιάζουν το $0$ από δεξιά. Θέτοντας $u=g(x)$, παίρνουμε $u\to0^+$, άρα
\[
\lim_{x\to-\infty}g(g(x))=\lim_{u\to0^+}g(u)=0.
\]

\item Από το σχήμα $g(x)\to0^+$ για $x\to0^-$ και $g(x)\to0^-$ για $x\to0^+$. Επομένως
\[
\lim_{x\to0^-}\frac1{g(x)}=+\infty,\quad
\lim_{x\to0^+}\frac1{g(x)}=-\infty.
\]
Τα πλευρικά όρια είναι διαφορετικά, άρα το $\lim\limits_{x\to0}\dfrac1{g(x)}$ δεν υπάρχει.
\end{erwthma}
\end{thema}
