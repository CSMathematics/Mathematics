\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \sqrt{x^2+1} - x$.

\begin{erwthma}

% -------------------------------------------------------
% α) Πεδίο ορισμού και όρια στα ±∞
% -------------------------------------------------------
\item Επειδή $x^2+1>0$ για κάθε $x\in\mathbb{R}$, η $f$ ορίζεται παντού, άρα $D_f=\mathbb{R}$.

Για το $\lim\limits_{x\to+\infty}f(x)$: πολλαπλασιάζουμε και διαιρούμε με την συζυγή:
\begin{align*}
\lim\limits_{x\to+\infty}f(x)&=\lim\limits_{x\to+\infty}\left( \sqrt{x^2+1}-x\right) =\lim\limits_{x\to+\infty}\frac{(\sqrt{x^2+1}-x)(\sqrt{x^2+1}+x)}{\sqrt{x^2+1}+x}=\\&=\lim\limits_{x\to+\infty}\frac{x^2+1-x^2}{\sqrt{x^2+1}+x}=\lim\limits_{x\to+\infty}\frac{1}{\sqrt{x^2+1}+x}=\\
&=\lim_{x\to+\infty} f(x)=\lim_{x\to+\infty}\frac{1}{\sqrt{x^2+1}+x}=0.
\end{align*}

Για το $\lim\limits_{x\to-\infty}f(x)$:
\begin{align*}
\lim\limits_{x\to-\infty}{f(x)}&=\lim\limits_{x\to-\infty}{\left(\sqrt{x^2+1}-x\right)}=+\infty-(-\infty)=+\infty
\end{align*}

% -------------------------------------------------------
% β) Γνησίως φθίνουσα και αυστηρά θετική
% -------------------------------------------------------
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[
f'(x)=\left(\sqrt{x^2+1}-x\right)'=\frac{2x}{2\sqrt{x^2+1}}-1=\frac{x}{\sqrt{x^2+1}}-1=\frac{x-\sqrt{x^2+1}}{\sqrt{x^2+1}}.
\]
Επειδή $\sqrt{x^2+1}>0$ πάντα, αρκεί να ελέγξουμε το πρόσημο του αριθμητή $x-\sqrt{x^2+1}$.
Για κάθε $x\in\mathbb{R}$:
\[
\sqrt{x^2+1}>\sqrt{x^2}=|x|\geq x,
\]
άρα $x-\sqrt{x^2+1}<0$, οπότε $f'(x)<0$ για κάθε $x\in\mathbb{R}$ οπότε η $f$ είναι \textbf{γνησίως φθίνουσα} στο $\mathbb{R}$.

Από τον τύπο που βρήκαμε στο (α):
\[
f(x)=\frac{1}{\sqrt{x^2+1}+x}.
\]
Για κάθε $x\in\mathbb{R}$: $\sqrt{x^2+1}>|x|\geq-x$, οπότε $\sqrt{x^2+1}+x>0$, άρα $f(x)=\dfrac{1}{\sqrt{x^2+1}+x}>0$.

% -------------------------------------------------------
% γ) Ασύμπτωτες
% -------------------------------------------------------
\item \textbf{Οριζόντια ασύμπτωτη στο $+\infty$.}
Από το (α): $\lim\limits_{x\to+\infty}f(x)=0$, οπότε η ευθεία $y=0$ είναι {οριζόντια ασύμπτωτη} στο $+\infty$.

\textbf{Πλάγια ασύμπτωτη στο $-\infty$.}
\begin{align*}
\lambda&=\lim_{x\to-\infty}\frac{f(x)}{x}=\lim_{x\to-\infty}\frac{\sqrt{x^2+1}-x}{x}=\lim\limits_{x\to-\infty}
\frac{\sqrt{x^2+1}-x}{x}=\lim_{x\to-\infty}\left( \frac{\sqrt{x^2+1}}{x}-1\right) =\\&=\lim_{x\to-\infty}\left( -\sqrt{1+\frac{1}{x^2}}-1\right)=-1-1=-2.
\end{align*}


Άρα $\lambda=-2$. Υπολογίζουμε $\beta$:
\begin{align*}
\beta&=\lim_{x\to-\infty}\bigl[f(x)-(-2x)\bigr]=\lim_{x\to-\infty}\bigl[\sqrt{x^2+1}-x+2x\bigr]=\lim_{x\to-\infty}\bigl[\sqrt{x^2+1}+x\bigr]\xlongequal{+\infty-\infty}\\
&=\lim\limits_{x\to-\infty}{\frac{1}{\sqrt{x^2+1}+x}}=\frac{1}{+\infty}=0.
\end{align*}




Άρα $\beta=0$ και η ευθεία $y=-2x$ είναι {πλάγια ασύμπτωτη} στο $-\infty$.

% -------------------------------------------------------
% δ) Κυρτότητα
% -------------------------------------------------------
\item Παραγωγίζουμε $f'(x)=\dfrac{x}{\sqrt{x^2+1}}-1$:
\[
f''(x)=\frac{\sqrt{x^2+1}-x\cdot\dfrac{x}{\sqrt{x^2+1}}}{x^2+1}=\frac{\dfrac{x^2+1-x^2}{\sqrt{x^2+1}}}{x^2+1}=\frac{1}{(x^2+1)\sqrt{x^2+1}}>0
\]
για κάθε $x\in\mathbb{R}$ άρα η $f$ είναι \textbf{κυρτή} στο $\mathbb{R}$. Δεν υπάρχουν σημεία καμπής.

\end{erwthma}
\end{thema}
