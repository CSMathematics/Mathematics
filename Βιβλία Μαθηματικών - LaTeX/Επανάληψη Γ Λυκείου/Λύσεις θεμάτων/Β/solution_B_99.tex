\begin{thema}{Β}
\begin{erwthma}
\item Από το σχήμα βλέπουμε ότι η $f$ είναι γνησίως αύξουσα στο $(-\infty,0]$, γνησίως φθίνουσα στο $[0,2]$ και γνησίως αύξουσα στο $[2,+\infty)$. Άρα
\[
f'(x)>0 \text{ στα } (-\infty,0)\cup(2,+\infty),\qquad
f'(x)<0 \text{ στο } (0,2).
\]
Στο $x=0$ η $f$ παρουσιάζει τοπικό μέγιστο και στο $x=2$ τοπικό ελάχιστο. Επειδή η $f$ είναι πολυωνυμική, άρα παραγωγίσιμη, από το θεώρημα \eng{Fermat} έχουμε
\[
f'(0)=0,\qquad f'(2)=0.
\]
Επομένως η εξίσωση $f'(x)=0$ έχει τουλάχιστον δύο ρίζες.

\item Η $f$ είναι πολυωνυμική, άρα η $f'$ είναι επίσης πολυωνυμική. Συνεπώς η $f'$ είναι συνεχής στο $[0,2]$ και παραγωγίσιμη στο $(0,2)$. Επιπλέον, από το προηγούμενο ερώτημα, $f'(0)=f'(2)=0$. Άρα η $f'$ ικανοποιεί τις υποθέσεις του θεωρήματος \eng{Rolle} στο $[0,2]$ και υπάρχει $\xi\in(0,2)$ τέτοιο ώστε
\[
(f')'(\xi)=0\Leftrightarrow f''(\xi)=0.
\]

\item Η ευθεία που διέρχεται από τα $A(1,1)$ και $B(-2,7)$ έχει κλίση
\[
\lambda_{AB}=\frac{7-1}{-2-1}=-2.
\]
Από το σχήμα διαβάζουμε $f(0)=1$ και $f(2)=-3$. Επειδή η $f$ είναι πολυωνυμική, εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[0,2]$. Υπάρχει λοιπόν $\xi\in(0,2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(2)-f(0)}{2-0}=\frac{-3-1}{2}=-2.
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $\xi$ είναι παράλληλη στην ευθεία $AB$.
\end{erwthma}
\end{thema}
