\begin{thema}{Β}
(Κριτήριο Παρεμβολής και Ασύμπτωτες)
\begin{erwthma}
\item Δίνεται η ανισοτική σχέση για κάθε $x > 0$:
\[ \frac{x^2 - x}{x+1} \le f(x) \le x - 2 + \frac{2}{x} \]
Διαιρούμε την ανισότητα με $x > 0$:
\[ \frac{x^2 - x}{x(x+1)} \le \frac{f(x)}{x} \le \frac{x - 2 + \frac{2}{x}}{x} \]
\[ \frac{x(x - 1)}{x(x+1)} \le \frac{f(x)}{x} \le 1 - \frac{2}{x} + \frac{2}{x^2} \]
\[ \frac{x - 1}{x+1} \le \frac{f(x)}{x} \le 1 - \frac{2}{x} + \frac{2}{x^2} \]
Υπολογίζουμε τα όρια των φραγμάτων καθώς το $x \to +\infty$:
$\lim_{x\to+\infty} \frac{x - 1}{x+1} = \lim_{x\to+\infty} \frac{x}{x} = 1$.
$\lim_{x\to+\infty} \left( 1 - \frac{2}{x} + \frac{2}{x^2} \right) = 1 - 0 + 0 = 1$.
Από το **Κριτήριο Παρεμβολής**, προκύπτει ότι:
\[ \lim_{x\to+\infty} \frac{f(x)}{x} = 1 \]

\item Για το δεύτερο όριο, αφαιρούμε από όλους τους όρους της αρχικής ανισότητας το $x$:
\[ \frac{x^2 - x}{x+1} - x \le f(x) - x \le x - 2 + \frac{2}{x} - x \]
Απλοποιούμε το αριστερό μέλος:
\[ \frac{x^2 - x - x(x+1)}{x+1} = \frac{x^2 - x - x^2 - x}{x+1} = \frac{-2x}{x+1} \]
Απλοποιούμε το δεξί μέλος:
\[ -2 + \frac{2}{x} \]
Οπότε παίρνουμε:
\[ \frac{-2x}{x+1} \le f(x) - x \le -2 + \frac{2}{x} \]
Υπολογίζουμε ξανά τα όρια στο $+\infty$:
$\lim_{x\to+\infty} \frac{-2x}{x+1} = \lim_{x\to+\infty} \frac{-2x}{x} = -2$.
$\lim_{x\to+\infty} \left( -2 + \frac{2}{x} \right) = -2 + 0 = -2$.
Εκ νέου από το **Κριτήριο Παρεμβολής**, συμπεραίνουμε ότι:
\[ \lim_{x\to+\infty} (f(x) - x) = -2 \]

\item Από τα προηγούμενα δύο ερωτήματα, διαπιστώσαμε ότι $\lim_{x\to+\infty} \frac{f(x)}{x} = \lambda = 1$ και $\lim_{x\to+\infty} (f(x) - \lambda x) = \beta = -2$.
Συνεπώς, η ευθεία με εξίσωση $y = \lambda x + \beta \Rightarrow y = x - 2$ αποτελεί **πλάγια ασύμπτωτη** της γραφικής παράστασης της $f$ στο $+\infty$.

\item Δίνεται επιπλέον το όριο: $\lim_{x\to1} \frac{f(x)}{x-1} = 2$.
Θέτουμε $g(x) = \frac{f(x)}{x-1}$, οπότε $f(x) = (x-1)g(x)$ με $\lim_{x\to1} g(x) = 2$.
Παίρνοντας όρια έχουμε: $\lim_{x\to1} f(x) = \lim_{x\to1} [(x-1)g(x)] = 0 \cdot 2 = 0$.
Καθώς η $f$ δηλώνεται **συνεχής** (στο $(0, +\infty)$ άρα και στο $x_0=1$), ισχύει: $f(1) = \lim_{x\to1} f(x) = 0$.
Με αυτήν την τιμή, το δοσμένο όριο μετασχηματίζεται άμεσα σε ορισμό της παραγώγου:
\[ \lim_{x\to1} \frac{f(x) - f(1)}{x-1} = 2 \Rightarrow f'(1) = 2 \]
Η εφαπτομένη της $C_f$ στο σημείο $x_0=1$ δίνεται από την εξίσωση:
\[ y - f(1) = f'(1)(x - 1) \Leftrightarrow y - 0 = 2(x - 1) \Leftrightarrow y = 2x - 2 \]
\end{erwthma}
\end{thema}
