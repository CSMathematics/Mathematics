%=========== ΛΥΣΗ - 79ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Για κάθε $x>0$ ισχύει
\[
\frac{x^2-x}{x+1}\leq f(x)\leq x-2+\frac{2}{x}.
\]

Διαιρούμε με $x>0$:
\[
\frac{x^2-x}{x(x+1)}\leq \frac{f(x)}{x}\leq
\frac{x-2+\frac{2}{x}}{x}.
\]
Άρα
\[
\frac{x-1}{x+1}\leq \frac{f(x)}{x}\leq
1-\frac{2}{x}+\frac{2}{x^2}.
\]

Επειδή
\[
\lim_{x\to+\infty}\frac{x-1}{x+1}=1
\]
και
\[
\lim_{x\to+\infty}\left(1-\frac{2}{x}+\frac{2}{x^2}\right)=1,
\]
από το Κριτήριο Παρεμβολής προκύπτει
\[
\lim_{x\to+\infty}\frac{f(x)}{x}=1.
\]

\item Από την αρχική ανισότητα αφαιρούμε $x$:
\[
\frac{x^2-x}{x+1}-x\leq f(x)-x\leq x-2+\frac{2}{x}-x.
\]

Για το αριστερό άκρο έχουμε
\[
\frac{x^2-x}{x+1}-x
=\frac{x^2-x-x(x+1)}{x+1}
=\frac{-2x}{x+1}.
\]
Για το δεξί άκρο έχουμε
\[
x-2+\frac{2}{x}-x=-2+\frac{2}{x}.
\]

Άρα
\[
\frac{-2x}{x+1}\leq f(x)-x\leq -2+\frac{2}{x}.
\]

Επειδή
\[
\lim_{x\to+\infty}\frac{-2x}{x+1}=-2
\]
και
\[
\lim_{x\to+\infty}\left(-2+\frac{2}{x}\right)=-2,
\]
από το Κριτήριο Παρεμβολής προκύπτει
\[
\lim_{x\to+\infty}(f(x)-x)=-2.
\]

\item Από τα προηγούμενα ερωτήματα έχουμε
\[
\lim_{x\to+\infty}\frac{f(x)}{x}=1
\]
και
\[
\lim_{x\to+\infty}(f(x)-x)=-2.
\]

Επομένως, η $C_f$ έχει στο $+\infty$ πλάγια ασύμπτωτη την ευθεία
\[
y=x-2.
\]

\item Δίνεται ότι
\[
\lim_{x\to 1}\frac{f(x)}{x-1}=2.
\]

Θέτουμε
\[
g(x)=\frac{f(x)}{x-1}.
\]
Τότε
\[
f(x)=(x-1)g(x)
\]
και
\[
\lim_{x\to 1}g(x)=2.
\]
Άρα
\[
\lim_{x\to 1}f(x)=\lim_{x\to 1}(x-1)g(x)=0.
\]

Επειδή η $f$ είναι συνεχής στο $x=1$, έχουμε
\[
f(1)=0.
\]

Επομένως,
\[
\lim_{x\to 1}\frac{f(x)}{x-1}
=\lim_{x\to 1}\frac{f(x)-f(1)}{x-1}
=f'(1).
\]
Άρα
\[
f'(1)=2.
\]

Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $x_0=1$ είναι
\[
y-f(1)=f'(1)(x-1).
\]
Άρα
\[
y=2(x-1)
\]
δηλαδή
\[
y=2x-2.
\]
\end{erwthma}
\end{thema}
