\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{ax + 1}{x - 2}$ με πεδίο ορισμού $D_f = \mathbb{R} - \{2\}$. Η $C_f$ διέρχεται από το σημείο $M(3, 4)$.

\begin{erwthma}
\item Αφού η $C_f$ διέρχεται από το $M(3, 4)$, ισχύει ότι $f(3) = 4$:
\[ f(3) =4\Leftrightarrow \dfrac{a\cdot3 + 1}{3 - 2} =4\Leftrightarrow \dfrac{3a + 1}{1} =4\Leftrightarrow 3a + 1=4 \Leftrightarrow 3a = 3 \Leftrightarrow a = 1\]
Επομένως, ο τύπος της συνάρτησης είναι $f(x) = \dfrac{x+1}{x-2}$.

\item Βρίσκουμε τις ασύμπτωτες:
\textbf{Κατακόρυφη:} Στο $x_0 = 2$, βρίσκουμε τα πλευρικά όρια:
\[ \lim_{x\to 2^+} f(x) = \lim_{x\to 2^+} \dfrac{x+1}{x-2} =\lim\limits_{x\to 2^+}\left[(x+1)\frac{1}{x-2}\right] = 3\cdot(+\infty)=+\infty\]
Επομένως, η ευθεία $x = 2$ είναι κατακόρυφη ασύμπτωτη της γραφικής παράστασης της $f$.
\textbf{Οριζόντια:} Ερευνώντας το όριο καθώς το $x \to +\infty$:
\[ \lim_{x\to +\infty} f(x) = \lim_{x\to +\infty} \dfrac{x+1}{x-2} = \lim_{x\to +\infty} \dfrac{x}{x} = 1 \]
Άρα η ευθεία $y = 1$ είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$ ομοίως και στο $-\infty$.

\item Το πεδίο ορισμού της $g$ είναι το $D_g = (0, +\infty)$.
Η συνάρτηση $h = f \circ g$ έχει:
\begin{itemize}
\item τύπο:
$h(x) = f(g(x)) = f(\ln x) = \dfrac{\ln x + 1}{\ln x - 2}$ και
\item πεδίο ορισμού
\[ D_h = \{x \in D_g \mid g(x) \in D_f\} \]
\begin{itemize}[label=\tiny\faPlay]
\item $x\in D_g\Leftrightarrow x\in (0,+\infty)$ και 
\item $g(x)\in D_f\Leftrightarrow\ln{x}\in\mathbb{R}-\{2\}\Leftrightarrow\ln x \ne 2 \Leftrightarrow \ln x \ne \ln(e^2) \Leftrightarrow x \ne e^2$.
\end{itemize}
 $h(x)=\dfrac{\ln x + 1}{\ln x - 2}\ ,\ D_h = (0, e^2) \cup (e^2, +\infty)$.
\end{itemize}


\item Θεωρούμε τη συνάρτηση $\phi(x)=h(x)-2026,\ x\in(e^2,e^3]$. Στο ανοιχτό άκρο $e^2$ του διαστήματος έχουμε:
\[\lim\limits_{x\to {e^{2}}^+}\phi(x)=\lim\limits_{x\to{e^2}^+}\left(\frac{\ln{x}+1}{\ln{x}-2}-2026\right)=+\infty\]
άρα ισχύει $\phi(x)>0$ κοντά στο $e^2$ δηλαδή υπάρχει $x_1$ κοντά στο $e^2$ τέτοιο ώστε $\phi(x_1)>0$. Η συνάρτηση $\phi$ είναι συνεχής στο $[x_1,e^3]$ ως πράξεις συνεχών συναρτήσεων. Επίσης
\begin{itemize}
\item $\phi(x_1)>0$
\item $\phi(e^3)=\frac{\ln{e^3}+1}{\ln{e^3}-2}-2026=\frac{3+1}{3-2}-2026=-2022<0$
\end{itemize}
Άρα σύμφωνα με το Θεώρημα \eng{Bolzano} η εξίσωση $\phi(x)=0\Leftrightarrow h(x) = 2026$ έχει τουλάχιστον μία ρίζα στο διάστημα $(x_1,e^3)\subseteq(e^2, e^3)$.
\end{erwthma}
\end{thema}
