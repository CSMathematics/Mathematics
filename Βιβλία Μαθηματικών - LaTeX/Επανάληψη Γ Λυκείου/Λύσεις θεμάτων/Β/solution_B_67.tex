\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=ax^2-x\ln x+\beta,\quad x>0.
\]

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=2ax-(x\ln x)'=2ax-\ln x-1.
\]
Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $x_0=1$ είναι οριζόντια, άρα
\[
f'(1)=0\Leftrightarrow 2a-\ln1-1=0\Leftrightarrow a=\frac{1}{2}.
\]
Επίσης, επειδή $f(1)=\dfrac{3}{2}$, έχουμε
\[
\frac{1}{2}-\ln1+\beta=\frac{3}{2}\Leftrightarrow \beta=1.
\]
Επομένως
\[
a=\frac{1}{2},\quad \beta=1
\]
και
\[
f(x)=\frac{x^2}{2}-x\ln x+1.
\]

\item Για $x>0$ έχουμε
\[
f'(x)=x-\ln x-1
\quad\text{και}\quad
f''(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Επειδή $x>0$, το πρόσημο της $f''$ είναι το πρόσημο του $x-1$:
\begin{itemize}
\item $f''(x)<0\Leftrightarrow 0<x<1$,
\item $f''(x)=0\Leftrightarrow x=1$,
\item $f''(x)>0\Leftrightarrow x>1$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:

Άρα η $f$ είναι \textbf{κοίλη} στο $(0,1]$ και \textbf{κυρτή} στο $[1,+\infty)$. Επειδή η κυρτότητα αλλάζει στο $x=1$, η $C_f$ έχει σημείο καμπής το
\[
A\left(1,\frac{3}{2}\right).
\]

\item Θεωρούμε τη συνάρτηση
\[
h(x)=x-\ln x-1,\quad x>0.
\]
Τότε $h(1)=0$ και
\[
h'(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Άρα η $h$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$, οπότε παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή $h(1)=0$. Επομένως
\[
h(x)\geq0,\quad x>0,
\]
με ισότητα μόνο για $x=1$. Δηλαδή
\[
f'(x)=x-\ln x-1\geq0,\quad x>0,
\]
και $f'(x)>0$ για κάθε $x\neq1$.

Συνεπώς η $f$ είναι γνησίως αύξουσα στα $(0,1]$ και $[1,+\infty)$. Επειδή τα δύο διαστήματα ενώνονται στο κοινό σημείο $1$, η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.

\item Έχουμε
\[
\lim_{x\to+\infty}\frac{f(x)}{x^2}
=\lim_{x\to+\infty}\left(\frac{1}{2}-\frac{\ln x}{x}+\frac{1}{x^2}\right)
=\frac{1}{2},
\]
αφού $\lim\limits_{x\to+\infty}\dfrac{\ln x}{x}=0$ και $\lim\limits_{x\to+\infty}\dfrac{1}{x^2}=0$.
\end{erwthma}
\end{thema}
