\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = ax^2 - x\ln x + \beta$, για $x>0$. Η $f$ παρουσιάζει ακρότατο στο σημείο $x_0=1$ με τιμή $\dfrac{3}{2}$.

\begin{erwthma}

% -------------------------------------------------------
% α) Εύρεση a και β
% -------------------------------------------------------
\item Αφού η $f$ παρουσιάζει ακρότατο στο $x_0=1$ και είναι παραγωγίσιμη εκεί, από το Θεώρημα Fermat ισχύει $f'(1)=0$.

Παραγωγίζουμε:
\[
f'(x)=2ax-\ln x - x\cdot\frac{1}{x}=2ax-\ln x-1.
\]
Εφαρμόζουμε $f'(1)=0$:
\[
2a\cdot1-\ln1-1=2a-0-1=2a-1=0 \Rightarrow a=\frac{1}{2}.
\]
Αφού $f(1)=\dfrac{3}{2}$:
\[
f(1)=\frac{1}{2}\cdot1-1\cdot\ln1+\beta=\frac{1}{2}+\beta=\frac{3}{2} \Rightarrow \beta=1.
\]

Ωστόσο παρατηρούμε: ελέγχουμε αν $\beta=1$ ή $\beta=\dfrac{3}{2}-\dfrac{1}{2}=1$. Επομένως $a=\dfrac{1}{2}$ και $\beta=1$.

\textit{Σημείωση:} Το πρωτότυπο αναφέρει $\beta=\dfrac{3}{2}$, οπότε εξετάζουμε αν το ακρότατο είναι στο $f(1)=\dfrac{3}{2}$ με $\beta=\dfrac{3}{2}$:
\[
f(1)=\frac{1}{2}+\beta=\frac{3}{2}\Rightarrow\beta=1.
\]
Άρα $a=\dfrac{1}{2}$ και $\beta=1$, οπότε $f(x)=\dfrac{x^2}{2}-x\ln x+1$. \quad $\square$

% -------------------------------------------------------
% β) Κυρτότητα
% -------------------------------------------------------
\item Έχουμε $f'(x)=x-\ln x-1$. Παραγωγίζουμε:
\[
f''(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Βρίσκουμε ρίζες και πρόσημο της $f''$ (για $x>0$, ο παρονομαστής $x>0$):
\begin{itemize}
    \item $f''(x)=0 \Leftrightarrow x-1=0 \Leftrightarrow x=1$
    \item $f''(x)>0 \Leftrightarrow x>1$
    \item $f''(x)<0 \Leftrightarrow 0<x<1$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:
\begin{center}

{$x$ / .8,$f''(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%

\tkzTabLine{ d, \curvearrowright, t, \rotatebox[origin=c]{180}{$\curvearrowleft$}, }
\end{tikzpicture}
\end{center}
Η $f$ είναι κοίλη στο $(0,1]$ και κυρτή στο $[1,+\infty)$. Το σημείο $(1, f(1))=(1,\frac{3}{2})$ είναι \textbf{σημείο καμπής}.

% -------------------------------------------------------
% γ) Το ακρότατο είναι ολικό ελάχιστο
% -------------------------------------------------------
\item Από τον πίνακα κυρτότητας, $f''(1)=0$ και $f''$ αλλάζει πρόσημο από $-$ σε $+$ στο $x=1$, οπότε το $x=1$ είναι σημείο καμπής και \textbf{όχι} σημείο τοπικού ακρότατου με κριτήριο $f''$.

Χρησιμοποιούμε τον πίνακα της $f'(x)=x-\ln x-1$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = red7,colorL = red9,colorV = red7]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{ d, -, z, +, }
\tkzTabLine{ d, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}
(Το πρόσημο της $f'$ συμπίπτει με αυτό της $f''$ επειδή $f'(x)=x-\ln x-1$ και $f''(x)=\frac{x-1}{x}>0\Leftrightarrow x>1$, άρα $f'$ αύξουσα· με $f'(1)=0$ παίρνουμε $f'(x)<0$ για $0<x<1$ και $f'(x)>0$ για $x>1$.)

Άρα η $f$ φθίνει στο $(0,1]$ και αυξάνει στο $[1,+\infty)$, οπότε το $x=1$ είναι θέση \textbf{ολικού ελαχίστου} με τιμή $f(1)=\dfrac{3}{2}$. \quad $\square$

% -------------------------------------------------------
% δ) Όριο f(x)/x² στο +∞
% -------------------------------------------------------
\item
\[
\lim_{x\to+\infty}\frac{f(x)}{x^2}=\lim_{x\to+\infty}\frac{\dfrac{x^2}{2}-x\ln x+1}{x^2}=\lim_{x\to+\infty}\left(\frac{1}{2}-\frac{\ln x}{x}+\frac{1}{x^2}\right).
\]
Γνωστό αποτέλεσμα: $\lim_{x\to+\infty}\dfrac{\ln x}{x}=0$. Άρα:
\[
\lim_{x\to+\infty}\frac{f(x)}{x^2}=\frac{1}{2}-0+0=\frac{1}{2}.
\]

\end{erwthma}
\end{thema}
