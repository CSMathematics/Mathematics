\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x)=\dfrac{2x}{x^2+1}$.

\begin{erwthma}

% -------------------------------------------------------
% α) Μονοτονία, ακρότατα
% -------------------------------------------------------
\item Η $f$ ορίζεται για κάθε $x\in\mathbb{R}$ ($x^2+1>0$ πάντα), άρα $D_f=\mathbb{R}$. Για κάθε $x\in\mathbb{R}$:
\[
f'(x)=\frac{(2x)'\cdot(x^2+1)-2x\cdot(x^2+1)'}{(x^2+1)^2}
=\frac{2(x^2+1)-2x\cdot2x}{(x^2+1)^2}
=\frac{2x^2+2-4x^2}{(x^2+1)^2}
=\frac{2(1-x^2)}{(x^2+1)^2}.
\]
Επειδή $(x^2+1)^2>0$ πάντα, το πρόσημο εξαρτάται αποκλειστικά από τον $1-x^2=(1-x)(1+x)$:
\begin{itemize}
    \item $f'(x)=0\Leftrightarrow 1-x^2=0\Leftrightarrow x=\pm1$
    \item $f'(x)>0\Leftrightarrow 1-x^2>0\Leftrightarrow -1<x<1$
    \item $f'(x)<0\Leftrightarrow 1-x^2<0\Leftrightarrow x<-1$ ή $x>1$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Συνεπώς:
\begin{itemize}
    \item Η $f$ είναι γνησίως φθίνουσα στο $(-\infty,-1]$, γνησίως αύξουσα στο $[-1,1]$ και γνησίως φθίνουσα στο $[1,+\infty)$.
    \item Στο $x=-1$ παρουσιάζει \textbf{τοπικό ελάχιστο} με τιμή $f(-1)=\dfrac{-2}{2}=-1$.
    \item Στο $x=1$ παρουσιάζει \textbf{τοπικό μέγιστο} με τιμή $f(1)=\dfrac{2}{2}=1$.
\end{itemize}

% -------------------------------------------------------
% β) Κυρτότητα, σημεία καμπής
% -------------------------------------------------------
\item Η $f'(x)=\dfrac{2-2x^2}{(x^2+1)^2}$ είναι παραγωγίσιμη στο $\mathbb{R}$.
Για κάθε $x\in\mathbb{R}$:
\[
f''(x)=\frac{(-4x)(x^2+1)^2-(2-2x^2)\cdot2(x^2+1)\cdot2x}{(x^2+1)^4}
\]
Βγάζουμε κοινό παράγοντα $2(x^2+1)$ στον αριθμητή:
\begin{align*}
f''(x)&=\frac{2(x^2+1)\left[-2x(x^2+1)-2x(2-2x^2)\right]}{(x^2+1)^4}
=\frac{-4x\left[(x^2+1)+(2-2x^2)\right]}{(x^2+1)^3}=\\&=\frac{-4x(3-x^2)}{(x^2+1)^3}
=\frac{4x(x^2-3)}{(x^2+1)^3}.
\end{align*}


Επειδή $(x^2+1)^3>0$, βρίσκουμε ρίζες και πρόσημο από τον αριθμητή $4x(x^2-3)=4x(x-\sqrt{3})(x+\sqrt{3})$:
\begin{itemize}
    \item $f''(x)=0\Leftrightarrow x=0$ ή $x=\pm\sqrt{3}$
    \item $f''(x)>0\Leftrightarrow x\in(-\sqrt{3},0)\cup(\sqrt{3},+\infty)$
    \item $f''(x)<0\Leftrightarrow x\in(-\infty,-\sqrt{3})\cup(0,\sqrt{3})$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:

Η $f$ αλλάζει κυρτότητα στα $x=-\sqrt{3},\,0,\,\sqrt{3}$, οπότε υπάρχουν τρία \textbf{σημεία καμπής}:
\[
f(0)=0,\quad f(\pm\sqrt{3})=\frac{\pm2\sqrt{3}}{3+1}=\pm\frac{\sqrt{3}}{2}.
\]
Τα σημεία καμπής είναι $\left(-\sqrt{3},-\dfrac{\sqrt{3}}{2}\right)$, $\;(0,0)$ και $\left(\sqrt{3},\dfrac{\sqrt{3}}{2}\right)$.

% -------------------------------------------------------
% γ) Ασύμπτωτες και γραφική παράσταση
% -------------------------------------------------------
\item \textbf{Κατακόρυφες ασύμπτωτες.} Δεν υπάρχουν, αφού η $f$ ορίζεται παντού.

\textbf{Οριζόντιες ασύμπτωτες.}
\[
\lim_{x\to\pm\infty}\frac{2x}{x^2+1}=\lim_{x\to\pm\infty}\frac{2x}{x^2}=\lim\limits_{x\to\pm\infty}{\dfrac{2}{x}}=0
\]
Η ευθεία $y=0$ είναι \textbf{οριζόντια ασύμπτωτη} τόσο στο $+\infty$ όσο και στο $-\infty$. Η $C_f$ δεν έχει πλάγιες ασύμπτωτες.

\textbf{Συμμετρία.} $f(-x)=\dfrac{-2x}{x^2+1}=-f(x)$, οπότε η $f$ είναι \textbf{περιττή} και η $C_f$ έχει κέντρο συμμετρίας την αρχή $O(0,0)$.

\textbf{Σύνολο τιμών.} Από τον πίνακα μονοτονίας, η $f$ παίρνει τοπικό ελάχιστο $-1$ και τοπικό μέγιστο $1$, ενώ τείνει στο $0$ στα $\pm\infty$. Άρα $f(\mathbb{R})=[-1,1]$.

% -------------------------------------------------------
% δ) Εμβαδόν
% -------------------------------------------------------
\item Το χωρίο ορίζεται από τον άξονα $x'x$ και τις ευθείες $x=-1$, $x=2$.

Ελέγχουμε το πρόσημο της $f$ στο $[-1,2]$:
\begin{itemize}
    \item $f(x)<0\Leftrightarrow \dfrac{2x}{x^2+1}<0\Leftrightarrow 2x<0\Leftrightarrow x<0$
    \item $f(x)>0\Leftrightarrow \dfrac{2x}{x^2+1}>0\Leftrightarrow 2x>0\Leftrightarrow x>0$
\end{itemize}
Άρα:
\[
E=\int_{-1}^{2}|f(x)|\,\d x
=\int_{-1}^{0}\frac{-2x}{x^2+1}\,\d x+\int_0^{2}\frac{2x}{x^2+1}\,\d x.
\]
Παρατηρούμε ότι $(x^2+1)'=2x$, οπότε $\dfrac{2x}{x^2+1}=\dfrac{(x^2+1)'}{x^2+1}$ και:
\[
\int\frac{2x}{x^2+1}\,\d x=\ln(x^2+1)+C.
\]
Επομένως:
\[
E=\Bigl[-\ln(x^2+1)\Bigr]_{-1}^{0}+\Bigl[\ln(x^2+1)\Bigr]_0^{2}
\]
\[
=(-\ln1+\ln2)+(\ln5-\ln1)=\ln2+\ln5=\ln10\ \text{τ.μ.}
\]

\end{erwthma}
\end{thema}
