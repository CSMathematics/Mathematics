\begin{thema}{Β}
Στο παρακάτω αναβαθμισμένο σχήμα έχουν προστεθεί σχόλια στα κρίσιμα σημεία της τμηματικά ορισμένης συνάρτησης (ακρότατα, σημείο ένωσης κλάδων).

\begin{erwthma}

\item Από τη γραφική παράσταση:
\begin{itemize}
    \item \textbf{Σύνολο τιμών:} Η $f$ ξεκινά από $f(0) = 0$ (ελάχιστο ύψος) και ανεβαίνει μέχρι $f(3) = 3$ (μέγιστο ύψος), χωρίς κανένα κενό. Συνεπώς, $f([0,3]) = [0, 3]$.
    \item \textbf{Ολικά Ακρότατα:} 
    \begin{itemize}
        \item \textbf{Ολικό ελάχιστο:} $f(0) = 0$, στη θέση $x = 0$.
        \item \textbf{Ολικό μέγιστο:} $f(3) = 3$, στη θέση $x = 3$.
    \end{itemize}
    Τοπικά ακρότατα στο εσωτερικό δεν εντοπίζονται (η συνάρτηση είναι συνεχώς αύξουσα).
    \item \textbf{Παραγωγισιμότητα στο $x_0 = 1$:} Γεωμετρικά η γραφική παράσταση δεν σχηματίζει οξεία γωνία (σαν V) στο σημείο $(1, 1)$. Ωστόσο, φαίνεται μια αλλαγή "κυρτότητας" (από παραβολικό κλάδο σε ευθύγραμμο). Η μετάβαση φαίνεται ομαλή, δηλαδή χωρίς κορυφή ή ακμή. Αυτό μας κάνει να πιστεύουμε ότι η $f$ \textbf{είναι παραγωγίσιμη} στο $x_0 = 1$. Θα το αποδείξουμε αναλυτικά στη συνέχεια.
\end{itemize}

\item Ελέγχουμε αναλυτικά την παραγωγισιμότητα στο $x_0 = 1$, εξετάζοντας τα πλευρικά όρια του λόγου διαφοράς (ή ισοδύναμα τις πλευρικές παραγώγους).

\textbf{Αριστερή παράγωγος ($x < 1$):} Ο τύπος εδώ είναι $f(x) = x^2$, οπότε $f'(x) = 2x$. Τότε:
\[ f'_-(1) = \lim_{x \to 1^-} f'(x) = \lim_{x \to 1^-} 2x = 2 \]

\textbf{Δεξιά παράγωγος ($x > 1$):} Ο τύπος εδώ είναι $f(x) = x$, οπότε $f'(x) = 1$. Τότε:
\[ f'_+(1) = \lim_{x \to 1^+} f'(x) = \lim_{x \to 1^+} 1 = 1 \]

Επειδή $f'_-(1) = 2 \ne 1 = f'_+(1)$, τα δύο πλευρικά όρια είναι \textbf{διαφορετικά}. Συνεπώς, η $f$ \textbf{δεν είναι παραγωγίσιμη} στο $x_0 = 1$!

\textit{(Σημείωση: Η γεωμετρική εικόνα μας παραπλάνησε! Η οξεία αλλαγή δεν είναι ακμή τύπου V, αλλά αλλαγή κλίσης: ο κλάδος $x^2$ φτάνει στο 1 με κλίση $2$, αλλά ο κλάδος $x$ ξεκινά από το 1 με κλίση $1$. Μοιάζουν ομαλές μεμονωμένα, αλλά η κλίση αλλάζει σαν σκαλοπάτι!)}

\item Εφόσον η $f$ είναι τμηματικά ορισμένη, σπάμε το ολοκλήρωμα στο σημείο $x=1$:
\[ \int_0^2 f(x) \,\mathrm{d}x = \int_0^1 f(x) \,\mathrm{d}x + \int_1^2 f(x) \,\mathrm{d}x = \int_0^1 x^2 \,\mathrm{d}x + \int_1^2 x \,\mathrm{d}x \]
Υπολογίζοντας καθένα ξεχωριστά:
\[ \int_0^1 x^2 \,\mathrm{d}x = \left[\frac{x^3}{3}\right]_0^1 = \frac{1}{3} - 0 = \frac{1}{3} \]
\[ \int_1^2 x \,\mathrm{d}x = \left[\frac{x^2}{2}\right]_1^2 = \frac{4}{2} - \frac{1}{2} = \frac{3}{2} \]
Συνολικά:
\[ \int_0^2 f(x) \,\mathrm{d}x = \frac{1}{3} + \frac{3}{2} = \frac{2}{6} + \frac{9}{6} = \frac{11}{6} \]

\end{erwthma}
\end{thema}
