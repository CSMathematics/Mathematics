%=========== ΛΥΣΗ - 97ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Από τη γραφική παράσταση βλέπουμε ότι η συνάρτηση παίρνει όλες τις τιμές από $0$ έως $3$, άρα
\[
f([0,3])=[0,3].
\]
Παρουσιάζει ολικό ελάχιστο στο $x=0$ με τιμή $f(0)=0$ και ολικό μέγιστο στο $x=3$ με τιμή $f(3)=3$.

Στο $x_0=1$ οι δύο κλάδοι ενώνονται στο σημείο $(1,1)$, άρα η συνάρτηση είναι συνεχής εκεί. Όμως η κλίση αλλάζει απότομα: ο παραβολικός κλάδος φτάνει με διαφορετική κλίση από αυτήν με την οποία συνεχίζει ο ευθύγραμμος κλάδος. Γεωμετρικά, λοιπόν, η $f$ δεν είναι παραγωγίσιμη στο $x_0=1$.

\item Δίνεται
\[
f(x)=
\begin{cases}
x^2, & 0\leq x\leq 1,\\
x, & 1<x\leq 3.
\end{cases}
\]
Έχουμε $f(1)=1$. Η αριστερή παράγωγος στο $1$ είναι
\[
f'_-(1)=\lim_{x\to1^-}\frac{f(x)-f(1)}{x-1}
=\lim_{x\to1^-}\frac{x^2-1}{x-1}
=\lim_{x\to1^-}\frac{(x-1)(x+1)}{x-1}
=2.
\]
Η δεξιά παράγωγος στο $1$ είναι
\[
f'_+(1)=\lim_{x\to1^+}\frac{f(x)-f(1)}{x-1}
=\lim_{x\to1^+}\frac{x-1}{x-1}
=1.
\]
Επειδή $f'_-(1)\ne f'_+(1)$, η $f$ δεν είναι παραγωγίσιμη στο $x=1$.

\item Χωρίζουμε το ολοκλήρωμα στο σημείο όπου αλλάζει ο τύπος της συνάρτησης:
\[
\int_0^2 f(x)\,\mathrm{d}x
=\int_0^1 x^2\,\mathrm{d}x+\int_1^2 x\,\mathrm{d}x
=\left[\frac{x^3}{3}\right]_0^1+\left[\frac{x^2}{2}\right]_1^2
=\frac13+\frac32=\frac{11}{6}.
\]
\end{erwthma}
\end{thema}
