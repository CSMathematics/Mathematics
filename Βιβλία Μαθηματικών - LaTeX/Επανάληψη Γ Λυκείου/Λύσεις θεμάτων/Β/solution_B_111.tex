\begin{thema}{Β}
Από το σχήμα διαβάζουμε τις τιμές των $f,g$ και, όπου χρειάζεται, τις εξισώσεις των ευθύγραμμων τμημάτων τους.

\begin{erwthma}
\item Από το σχήμα έχουμε
\[
g(0)=2,\quad g(2)=0,\quad f(0)=3,\quad f(2)=1.
\]
Επομένως
\[
(f\circ g)(0)=f(g(0))=f(2)=1,
\]
\[
(f\circ g)(2)=f(g(2))=f(0)=3.
\]
Επίσης
\[
(g\circ f)(0)=g(f(0))=g(3).
\]
Στο διάστημα $[2,4]$ η $g$ είναι το ευθύγραμμο τμήμα από το $(2,0)$ στο $(4,2)$, άρα έχει κλίση $1$ και τύπο $g(x)=x-2$. Έτσι
\[
g(3)=3-2=1.
\]
Άρα
\[
(g\circ f)(0)=1.
\]

\item Η σύνθεση $(f\circ g)(x)=f(g(x))$ ορίζεται όταν
\[
x\in D_g
\quad\text{και}\quad
g(x)\in D_f.
\]
Από την εκφώνηση έχουμε $D_f=D_g=[-2,4]$. Από το σχήμα η $g$ παίρνει τιμές από $0$ έως $2$, δηλαδή
\[
g([-2,4])=[0,2].
\]
Επειδή
\[
[0,2]\subseteq[-2,4],
\]
η $(f\circ g)(x)$ ορίζεται για κάθε $x\in[-2,4]$. Άρα
\[
D_{f\circ g}=[-2,4].
\]

\item Θα εξετάσουμε τις πλευρικές παραγώγους της $f\circ g$ στο $0$.

Για $x<0$ κοντά στο $0$, από το σχήμα έχουμε
\[
g(x)=x+2.
\]
Τότε $g(x)<2$, οπότε χρησιμοποιούμε τον κλάδο της $f$ στο $[0,2]$, όπου
\[
f(u)=-u+3.
\]
Άρα
\[
(f\circ g)(x)=f(x+2)=-(x+2)+3=1-x,\quad x<0.
\]
Επομένως
\[
(f\circ g)'_-(0)=-1.
\]

Για $x>0$ κοντά στο $0$, από το σχήμα έχουμε
\[
g(x)=-x+2.
\]
Τότε πάλι $g(x)<2$, οπότε
\[
(f\circ g)(x)=f(-x+2)=-(-x+2)+3=1+x,\quad x>0.
\]
Επομένως
\[
(f\circ g)'_+(0)=1.
\]

Επειδή
\[
(f\circ g)'_-(0)\neq (f\circ g)'_+(0),
\]
η παράγωγος $(f\circ g)'(0)$ δεν υπάρχει.

\item Για τη συνάρτηση $h(x)=f(x)+g(x)$ έχουμε
\[
h(0)=f(0)+g(0)=3+2=5
\]
και
\[
h(2)=f(2)+g(2)=1+0=1.
\]

Στο διάστημα $[0,2]$ από το σχήμα είναι
\[
f(x)=-x+3
\quad\text{και}\quad
g(x)=-x+2.
\]
Άρα
\[
h(x)=f(x)+g(x)=(-x+3)+(-x+2)=-2x+5,\quad 0\leq x\leq 2.
\]
Επομένως η $C_h$ στο $[0,2]$ είναι το ευθύγραμμο τμήμα με άκρα τα σημεία
\[
(0,5)\quad\text{και}\quad (2,1).
\]
\end{erwthma}
\end{thema}
