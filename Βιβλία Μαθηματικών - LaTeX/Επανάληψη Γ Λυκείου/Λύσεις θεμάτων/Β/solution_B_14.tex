\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = x^3 + x - 2$ και $g(x) = e^{x-1}$, ορισμένες στο $\mathbb{R}$.

\begin{erwthma}

% -------------------------------------------------------
% α) Τύπος h = f∘g
% -------------------------------------------------------
\item Η συνάρτηση $h = f \circ g$ έχει πεδίο ορισμού $D_h = \mathbb{R}$ (αφού $D_g = \mathbb{R}$ και $D_f = \mathbb{R}$) και τύπο:
\[
h(x) = f(g(x)) = f(e^{x-1}) = (e^{x-1})^3 + e^{x-1} - 2 = e^{3(x-1)} + e^{x-1} - 2,\quad x\in\mathbb{R}.
\]

% -------------------------------------------------------
% β) Απόδειξη 1-1 και λύση h(x²) = h(2x-1)
% -------------------------------------------------------
\item Η $h$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[
h'(x) = 3e^{3(x-1)} + e^{x-1}>0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$, οπότε είναι συνάρτηση $1-1$.

\textbf{Λύση της $h(x^2) = h(2x-1)$.}
Επειδή η $h$ είναι $1-1$, η εξίσωση ισοδυναμεί με:
\[
h(x^2) = h(2x-1)\Leftrightarrow x^2 = 2x - 1 \Leftrightarrow x^2 - 2x + 1 = 0 \Leftrightarrow (x-1)^2 = 0 \Leftrightarrow x = 1.
\]

% -------------------------------------------------------
% γ) Εφαπτομένη παράλληλη με ε: y = 4x + 2026
% -------------------------------------------------------
\item Η εφαπτομένη της $C_h$ πρέπει να έχει κλίση $4$ (ίδια με την $\varepsilon$).
Λύνουμε $h'(x) = 4$:
\[
3e^{3(x-1)} + e^{x-1} = 4.
\]
Θέτουμε $t = e^{x-1} > 0$:
\[
3t^3 + t = 4 \Leftrightarrow 3t^3 + t - 4 = 0\Leftrightarrow(t-1)(3t^2 + 3t + 4)=0\]

Η διακρίνουσα του $3t^2 + 3t + 4$ είναι $\Delta = 9 - 48 < 0$, άρα δεν έχει πραγματικές ρίζες.
Επομένως η μοναδική πραγματική λύση είναι $t = 1$, δηλαδή $e^{x-1} = 1 \Leftrightarrow x = 1$.

Σημείο επαφής: $h(1) = e^0 + e^0 - 2 = 1 + 1 - 2 = 0$, άρα το σημείο είναι $(1, 0)$.

Εξίσωση εφαπτομένης:
\[
y - 0 = 4(x - 1) \Leftrightarrow y = 4x - 4.
\]

% -------------------------------------------------------
% δ) Εμβαδόν χωρίου f, x'x, x=0, x=1
% -------------------------------------------------------
\item Ελέγχουμε το πρόσημο της $f(x) = x^3 + x - 2$ στο $[0,1]$.
Η $f$ είναι παραγωγίσιμη με $f'(x) = 3x^2 + 1 > 0$, άρα γνησίως αύξουσα. Έχουμε:
\[
f(0) = -2 < 0, \qquad f(1) = 1 + 1 - 2 = 0.
\]
Άρα $f(x) \leq 0$ για κάθε $x \in [0, 1]$, οπότε:
\[
E = \int_0^1 |f(x)|\,\d x = \int_0^1 \bigl(-(x^3 + x - 2)\bigr)\,\d x = \int_0^1 (-x^3 - x + 2)\,\d x.
\]
\[
E = \left[-\frac{x^4}{4} - \frac{x^2}{2} + 2x\right]_0^1 = -\frac{1}{4} - \frac{1}{2} + 2 = \frac{-1-2+8}{4} = \frac{5}{4}\ \text{τ.μ.}
\]

\end{erwthma}
\end{thema}
