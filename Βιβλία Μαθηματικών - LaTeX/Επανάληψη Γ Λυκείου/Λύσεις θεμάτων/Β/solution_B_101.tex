\begin{thema}{Β}
Η γραφική παράσταση αποτελείται από τρία ευθύγραμμα τμήματα. Θα βρούμε την εξίσωση της ευθείας σε κάθε τμήμα και θα χρησιμοποιήσουμε τα ίδια δεδομένα για τα επόμενα ερωτήματα.

\begin{erwthma}
\item Στο τμήμα $AB$ η κλίση είναι
\[
m_{AB}=\frac{3-(-1)}{-1-(-3)}=\frac{4}{2}=2.
\]
Άρα η ευθεία που περνά από το $A(-3,-1)$ έχει εξίσωση
\[
y+1=2(x+3)\Rightarrow y=2x+5.
\]
Στο τμήμα $B\Gamma$ η κλίση είναι
\[
m_{B\Gamma}=\frac{-1-3}{1-(-1)}=\frac{-4}{2}=-2,
\]
οπότε
\[
y-3=-2(x+1)\Rightarrow y=-2x+1.
\]
Στο τμήμα $\Gamma\Delta$ η κλίση είναι
\[
m_{\Gamma\Delta}=\frac{2-(-1)}{4-1}=\frac{3}{3}=1,
\]
οπότε
\[
y+1=x-1\Rightarrow y=x-2.
\]
Επομένως
\[
f(x)=
\begin{cases}
2x+5, & -3\leq x\leq -1,\\
-2x+1, & -1<x\leq 1,\\
x-2, & 1<x\leq 4.
\end{cases}
\]

\item Μελετάμε την εξίσωση $f(x)=\lambda$ γεωμετρικά, δηλαδή το πλήθος των σημείων τομής της $C_f$ με την οριζόντια ευθεία $y=\lambda$.

Το σύνολο τιμών του τμήματος $AB$ είναι $[-1,3]$, του $B\Gamma$ είναι $[-1,3]$ και του $\Gamma\Delta$ είναι $[-1,2]$. Προσέχουμε όμως ότι στις κορυφές τα γειτονικά τμήματα δίνουν το ίδιο σημείο.

Έτσι:
\begin{itemize}
\item για $\lambda<-1$ ή $\lambda>3$, η εξίσωση δεν έχει ρίζες,
\item για $\lambda=-1$, έχει δύο ρίζες, τις $x=-3$ και $x=1$,
\item για $-1<\lambda\leq 2$, έχει τρεις διαφορετικές ρίζες,
\item για $2<\lambda<3$, έχει δύο διαφορετικές ρίζες,
\item για $\lambda=3$, έχει μία ρίζα, την $x=-1$.
\end{itemize}

Έστω τώρα $\lambda\in(-1,2]$. Τότε η εξίσωση έχει μία ρίζα σε κάθε ευθύγραμμο τμήμα. Αν τις συμβολίσουμε από αριστερά προς τα δεξιά με $\rho_1,\rho_2,\rho_3$, τότε:
\[
2\rho_1+5=\lambda\Rightarrow \rho_1=\frac{\lambda-5}{2},
\]
\[
-2\rho_2+1=\lambda\Rightarrow \rho_2=\frac{1-\lambda}{2},
\]
και
\[
\rho_3-2=\lambda\Rightarrow \rho_3=\lambda+2.
\]
Άρα
\[
\rho_1+\rho_2+\rho_3
=\frac{\lambda-5}{2}+\frac{1-\lambda}{2}+\lambda+2
=\frac{-4}{2}+\lambda+2
=\lambda.
\]

\item Στο εσωτερικό του πεδίου ορισμού τα μόνα πιθανά σημεία μη παραγωγισιμότητας είναι οι κορυφές $B$ και $\Gamma$, δηλαδή τα $x=-1$ και $x=1$.

Στο $x=-1$ έχουμε $f(-1)=3$. Η αριστερή και η δεξιά παράγωγος είναι
\[
f'_-(-1)=\lim_{x\to -1^-}\frac{f(x)-f(-1)}{x-(-1)}
=\lim_{x\to -1^-}\frac{2x+5-3}{x+1}
=\lim_{x\to -1^-}\frac{2(x+1)}{x+1}=2
\]
και
\[
f'_+(-1)=\lim_{x\to -1^+}\frac{f(x)-f(-1)}{x-(-1)}
=\lim_{x\to -1^+}\frac{-2x+1-3}{x+1}
=\lim_{x\to -1^+}\frac{-2(x+1)}{x+1}=-2.
\]
Επειδή $f'_-(-1)\neq f'_+(-1)$, η $f$ δεν είναι παραγωγίσιμη στο $x=-1$.

Στο $x=1$ έχουμε $f(1)=-1$. Ομοίως,
\[
f'_-(1)=\lim_{x\to 1^-}\frac{f(x)-f(1)}{x-1}
=\lim_{x\to 1^-}\frac{-2x+1+1}{x-1}
=\lim_{x\to 1^-}\frac{-2(x-1)}{x-1}=-2
\]
και
\[
f'_+(1)=\lim_{x\to 1^+}\frac{f(x)-f(1)}{x-1}
=\lim_{x\to 1^+}\frac{x-2+1}{x-1}
=\lim_{x\to 1^+}\frac{x-1}{x-1}=1.
\]
Επειδή $f'_-(1)\neq f'_+(1)$, η $f$ δεν είναι παραγωγίσιμη στο $x=1$.

Στα άκρα $-3$ και $4$ δεν εξετάζουμε δίπλευρη παραγωγισιμότητα.

\item Για να υπολογίσουμε γεωμετρικά το $\dint_{-3}^{4}|f(x)|\d x$, βρίσκουμε πρώτα τις ρίζες της $f$:
\[
2x+5=0\Rightarrow x=-\frac{5}{2},\qquad
-2x+1=0\Rightarrow x=\frac{1}{2},\qquad
x-2=0\Rightarrow x=2.
\]
Άρα το ζητούμενο ολοκλήρωμα είναι το άθροισμα των εμβαδών που σχηματίζονται με τον άξονα $x'x$.

Στο $\left[-3,-\frac{5}{2}\right]$ έχουμε τρίγωνο με βάση $\frac{1}{2}$ και ύψος $1$, άρα
\[
E_1=\frac{1}{2}\cdot\frac{1}{2}\cdot 1=\frac{1}{4}.
\]
Στο $\left[-\frac{5}{2},\frac{1}{2}\right]$ έχουμε τρίγωνο με βάση $3$ και ύψος $3$, άρα
\[
E_2=\frac{1}{2}\cdot 3\cdot 3=\frac{9}{2}.
\]
Στο $\left[\frac{1}{2},2\right]$ έχουμε τρίγωνο με βάση $\frac{3}{2}$ και ύψος $1$, άρα
\[
E_3=\frac{1}{2}\cdot\frac{3}{2}\cdot 1=\frac{3}{4}.
\]
Στο $[2,4]$ έχουμε τρίγωνο με βάση $2$ και ύψος $2$, άρα
\[
E_4=\frac{1}{2}\cdot 2\cdot 2=2.
\]
Επομένως
\[
\dint_{-3}^{4}|f(x)|\d x
=E_1+E_2+E_3+E_4
=\frac{1}{4}+\frac{9}{2}+\frac{3}{4}+2
=\frac{15}{2}.
\]
\end{erwthma}
\end{thema}
