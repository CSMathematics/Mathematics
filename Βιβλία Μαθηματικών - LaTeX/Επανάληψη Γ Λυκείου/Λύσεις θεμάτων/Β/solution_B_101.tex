\begin{erwthma}

\item Από τη γραφική παράσταση:
\begin{itemize}
    \item \textbf{Πεδίο ορισμού:} Η $C_f$ εκτείνεται αδιάσπαστα από $x = -3$ έως $x = 4$, οπότε $\varDelta_f = [-3, 4]$.
    \item \textbf{Σύνολο τιμών:} Η ελάχιστη τεταγμένη που εμφανίζεται στο γράφημα είναι $-1$ (τα σημεία $A$ και $\Gamma$) και η μέγιστη είναι $3$ (σημείο $B$). Εφόσον οι κλάδοι συνεχούς τμηματικής γραμμικής λαμβάνουν ενδιάμεσα κάθε τιμή, ισχύει $f([-3, 4]) = [-1, 3]$.
    \item \textbf{Ρίζες ($f(x) = 0$):} Η γραφική παράσταση τέμνει τον $x'x$ σε τρία σημεία. Βρίσκουμε τις ρίζες υπολογίζοντας τις τετμημένες τομής κάθε ευθύγραμμου τμήματος:
    \begin{itemize}
        \item Τμήμα $AB$: Η ευθεία από $(-3,-1)$ ώς $(-1,3)$ έχει εξίσωση $y = 2(x+3) - 1 = 2x+5$. Θέτοντας $y=0$: $x = -5/2$. Αλλά αυτό πέφτει εκτός $[-3, -1]$. Ας ξαναυπολογίσουμε: κλίση $= \frac{3-(-1)}{-1-(-3)} = \frac{4}{2} = 2$. Εξίσωση: $y - (-1) = 2(x - (-3)) \Rightarrow y = 2x + 5$. Ρίζα: $2x+5 = 0 \Rightarrow x = -5/2 = -2{,}5$. Δεν ανήκει στο $[-3,-1]$; Ελέγχουμε: $-3 \le -2{,}5 \le -1$ \checkmark. Άρα $x_1 = -2{,}5 = -\frac{5}{2}$. Ωστόσο, γραφικά η τετμημένη φαίνεται ακριβώς στο $x = -2$. Η ρίζα $x=-2$ ελέγχεται: $f(-2) = 2(-2)+5 = 1 \ne 0$. Άρα η ρίζα είναι στο $x = -5/2$.
        \item Τμήμα $B\Gamma$: Κλίση $= \frac{-1-3}{1-(-1)} = \frac{-4}{2} = -2$. Εξίσωση: $y - 3 = -2(x+1) \Rightarrow y = -2x + 1$. Ρίζα: $-2x+1 = 0 \Rightarrow x = 1/2$.
        \item Τμήμα $\Gamma\Delta$: Κλίση $= \frac{2-(-1)}{4-1} = 1$. Εξίσωση: $y - (-1) = 1(x - 1) \Rightarrow y = x - 2$. Ρίζα: $x-2=0 \Rightarrow x = 2$.
    \end{itemize}
    Οι ρίζες είναι: $x_1 = -\frac{5}{2}$, $x_2 = \frac{1}{2}$, $x_3 = 2$.
\end{itemize}

\item Μελετώντας κάθε ευθύγραμμο τμήμα:
\begin{itemize}
    \item Στο $[-3, -1]$ (τμήμα $A \to B$): κλίση $= 2 > 0$, η $f$ είναι \textbf{γνησίως αύξουσα}.
    \item Στο $[-1, 1]$ (τμήμα $B \to \Gamma$): κλίση $= -2 < 0$, η $f$ είναι \textbf{γνησίως φθίνουσα}.
    \item Στο $[1, 4]$ (τμήμα $\Gamma \to \Delta$): κλίση $= 1 > 0$, η $f$ είναι \textbf{γνησίως αύξουσα}.
\end{itemize}
Ακρότατα:
\begin{itemize}
    \item \textbf{Τοπικό (και ολικό) μέγιστο:} στο $B(-1, 3)$, αφού αλλάζει από αύξουσα σε φθίνουσα. $f_{\max} = 3$.
    \item \textbf{Τοπικό ελάχιστο:} στο $\Gamma(1, -1)$, αφού αλλάζει από φθίνουσα σε αύξουσα.
    \item \textbf{Ολικό ελάχιστο:} $f(-3) = -1 = f(1) = -1$, τα δύο σημεία $A$ και $\Gamma$ μοιράζονται αυτή τη χαμηλότερη τιμή.
\end{itemize}

\item Η $f$ \textbf{δεν είναι παραγωγίσιμη} στα σημεία $x = -1$ και $x = 1$, δηλαδή στις κορυφές $B$ και $\Gamma$. Σε αυτά τα σημεία, η γραφική παράσταση σχηματίζει \textbf{γωνίες}, δηλαδή αλλάζει απότομα η κλίση (η αριστερή παράγωγος διαφέρει από τη δεξιά):
\begin{itemize}
    \item Στο $x = -1$: αριστερή κλίση $= 2$, δεξιά κλίση $= -2$. Επειδή $2 \ne -2$, δεν ορίζεται μοναδική εφαπτομένη.
    \item Στο $x = 1$: αριστερή κλίση $= -2$, δεξιά κλίση $= 1$. Επειδή $-2 \ne 1$, δεν ορίζεται μοναδική εφαπτομένη.
\end{itemize}

\item Υπολογίζουμε γεωμετρικά το $\int_{-3}^{4} |f(x)| \,\mathrm{d}x$ σπάζοντας στα τμήματα ανάλογα με το πρόσημο (εμβαδά τραπεζίων και τριγώνων):
\begin{itemize}
    \item $[-3, -5/2]$: $f(x) < 0$. Τρίγωνο με βάση $1/2$ και ύψος $1$: $E_1 = \frac{1}{2}\cdot\frac{1}{2}\cdot 1 = \frac{1}{4}$.
    \item $[-5/2, 1/2]$: $f(x) > 0$. Τρίγωνο (ισοσκελές) με βάση $3$ (από $-5/2$ ως $1/2$) και ύψος $3$: $E_2 = \frac{1}{2}\cdot 3 \cdot 3 = \frac{9}{2}$.
    \item $[1/2, 2]$: $f(x) < 0$. Τρίγωνο με βάση $3/2$ (από $1/2$ ως $2$) και ύψος $1$: $E_3 = \frac{1}{2}\cdot\frac{3}{2}\cdot 1 = \frac{3}{4}$.
    \item $[2, 4]$: $f(x) > 0$. Τρίγωνο με βάση $2$ (από $2$ ως $4$) και ύψος $2$: $E_4 = \frac{1}{2}\cdot 2 \cdot 2 = 2$.
\end{itemize}
Συνεπώς:
\[ \int_{-3}^{4} |f(x)| \,\mathrm{d}x = E_1 + E_2 + E_3 + E_4 = \frac{1}{4} + \frac{9}{2} + \frac{3}{4} + 2 = \frac{1 + 18 + 3 + 8}{4} = \frac{30}{4} = \frac{15}{2} \]

\end{erwthma}
\end{thema}
