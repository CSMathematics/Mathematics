\begin{thema}{Β}
Δίνονται σε πίνακα τιμές δύο συναρτήσεων $f,g$ ορισμένων στο $\{0,1,2,3,4\}$:

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
\hline
$f(x)$ & $3$ & $1$ & $4$ & $0$ & $2$ \\
\hline
$g(x)$ & $2$ & $0$ & $1$ & $4$ & $3$ \\
\hline
\end{tabular}
\end{center}

\begin{erwthma}

\item Υπολογίζουμε τις συνθέσεις βήμα-βήμα:
\begin{itemize}
    \item $(f \circ g)(0) = f(g(0)) = f(2) = 4$.
    \item $(g \circ f)(1) = g(f(1)) = g(1) = 0$.
    \item $(f \circ f)(2) = f(f(2)) = f(4) = 2$.
    \item $(g \circ g)(3) = g(g(3)) = g(4) = 3$.
\end{itemize}

\item Μια συνάρτηση είναι 1-1 (αμφιμονοσήμαντη) αν δεν επαναλαμβάνει τιμές.
\begin{itemize}
    \item \textbf{$f$}: Τιμές $\{3, 1, 4, 0, 2\}$. Κάθε τιμή εμφανίζεται ακριβώς μία φορά. \textbf{Ναι, η $f$ είναι 1-1.}
    \item \textbf{$g$}: Τιμές $\{2, 0, 1, 4, 3\}$. Κάθε τιμή εμφανίζεται ακριβώς μία φορά. \textbf{Ναι, η $g$ είναι 1-1.}
\end{itemize}

\item Εφόσον η $f$ είναι 1-1, \textbf{αντιστρέφεται}. Ο αντίστροφος πίνακας τιμών κατασκευάζεται αντιστρέφοντας τα ζεύγη $(x, f(x)) \to (f(x), x)$:

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$y$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
\hline
$f^{-1}(y)$ & $3$ & $1$ & $4$ & $0$ & $2$ \\
\hline
\end{tabular}
\end{center}

\textit{Εξήγηση: $f(3) = 0$ $\Rightarrow$ $f^{-1}(0) = 3$, $f(1) = 1$ $\Rightarrow$ $f^{-1}(1) = 1$, $f(4) = 2$ $\Rightarrow$ $f^{-1}(2) = 4$, $f(0) = 3$ $\Rightarrow$ $f^{-1}(3) = 0$, $f(2) = 4$ $\Rightarrow$ $f^{-1}(4) = 2$.}

Οπότε: $f^{-1}(0) = 3$ και $f^{-1}(3) = 0$.

\item Ζητάμε τις λύσεις της $f(g(x)) = 2$. Αφού $f^{-1}(2) = 4$, αυτό σημαίνει $g(x) = 4$. Από τον πίνακα: $g(3) = 4$, άρα $x = 3$.
Η εξίσωση $f(g(x)) = 2$ έχει \textbf{μοναδική λύση: $x = 3$}.

\end{erwthma}
\end{thema}
