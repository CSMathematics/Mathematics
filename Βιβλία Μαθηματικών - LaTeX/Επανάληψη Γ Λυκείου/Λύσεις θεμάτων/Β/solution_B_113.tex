\begin{thema}{Β}
Χρησιμοποιούμε τον πίνακα τιμών των συναρτήσεων $f,g$ στο σύνολο $\{0,1,2,3,4\}$.

\begin{erwthma}
\item Υπολογίζουμε κάθε σύνθεση διαδοχικά:
\[
(f\circ g)(0)=f(g(0))=f(2)=4.
\]
Επίσης
\[
(g\circ f)(1)=g(f(1))=g(1)=0.
\]
Ακόμη,
\[
(f\circ f)(2)=f(f(2))=f(4)=2.
\]
Τέλος,
\[
(g\circ g)(3)=g(g(3))=g(4)=3.
\]

\item Οι τιμές της $f$ είναι
\[
3,\ 1,\ 4,\ 0,\ 2.
\]
Παρατηρούμε ότι καμία τιμή δεν επαναλαμβάνεται. Άρα η $f$ είναι $1-1$.

\item Επειδή η $f$ είναι $1-1$, αντιστρέφεται. Για να φτιάξουμε τον πίνακα της $f^{-1}$, αντιστρέφουμε τα ζεύγη του πίνακα της $f$:
\[
f(0)=3,\quad f(1)=1,\quad f(2)=4,\quad f(3)=0,\quad f(4)=2.
\]
Άρα
\[
f^{-1}(3)=0,\quad f^{-1}(1)=1,\quad f^{-1}(4)=2,\quad f^{-1}(0)=3,\quad f^{-1}(2)=4.
\]
Ο αντίστροφος πίνακας τιμών είναι
\begin{center}
\begin{myhtblr}{}
$x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
$f^{-1}(x)$ & $3$ & $1$ & $4$ & $0$ & $2$
\end{myhtblr}
\end{center}
Επομένως
\[
f^{-1}(0)=3
\quad\text{και}\quad
f^{-1}(3)=0.
\]

\item Θέλουμε
\[
f(g(x))=2.
\]
Από τον πίνακα της $f$ βλέπουμε ότι
\[
f(4)=2.
\]
Επειδή η $f$ είναι $1-1$, η εξίσωση $f(u)=2$ έχει μοναδική λύση $u=4$. Άρα πρέπει
\[
g(x)=4.
\]
Από τον πίνακα της $g$ έχουμε
\[
g(3)=4.
\]
Επομένως η εξίσωση έχει μοναδική λύση
\[
x=3.
\]
\end{erwthma}
\end{thema}
