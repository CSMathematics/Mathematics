\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=x^3+ax+\beta,\quad x\in\mathbb{R}.
\]

\begin{erwthma}
\item Η ευθεία $\varepsilon:y=3x-4$ εφάπτεται στη $C_f$ στο σημείο με τετμημένη $x_0=1$. Άρα το σημείο επαφής ανήκει στην ευθεία, οπότε
\[
f(1)=3\cdot1-4=-1.
\]
Επιπλέον, η κλίση της εφαπτομένης είναι ίση με την παράγωγο της $f$ στο $1$, άρα
\[
f'(1)=3.
\]
Επειδή $f'(x)=3x^2+a$, έχουμε
\[
f'(1)=3\Leftrightarrow 3+a=3\Leftrightarrow a=0.
\]
Τότε, από τη σχέση $f(1)=-1$, προκύπτει
\[
1+\beta=-1\Leftrightarrow \beta=-2.
\]
Άρα
\[
f(x)=x^3-2.
\]

\item Η ευθεία $y=4x+1$ έχει συντελεστή διεύθυνσης $4$, επομένως αρκεί να δείξουμε ότι υπάρχει $\xi\in(0,2)$ με $f'(\xi)=4$.

Η $f$ είναι συνεχής στο $[0,2]$ και παραγωγίσιμη στο $(0,2)$, ως πολυωνυμική. Από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(0,2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(2)-f(0)}{2-0}
=\frac{6-(-2)}{2}=4.
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $\xi$ είναι παράλληλη στην ευθεία $y=4x+1$.

\item Με $f(x)=x^3-2$ και $f(2)=6$, έχουμε
\[
\lim_{x\to2}\frac{f(x)-6}{x-2}
=\lim_{x\to2}\frac{x^3-8}{x-2}
=\lim_{x\to2}(x^2+2x+4)=12.
\]

\item Αν $x_1<x_2$, τότε $x_1^3<x_2^3$, άρα
\[
f(x_1)=x_1^3-2<x_2^3-2=f(x_2).
\]
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$, άρα είναι $1-1$ και συνεπώς αντιστρέφεται.

Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\[
f(f(x))>f(6)\Leftrightarrow f(x)>6.
\]
Όμως $f(2)=6$, άρα
\[
f(x)>6\Leftrightarrow f(x)>f(2)\Leftrightarrow x>2.
\]
Επομένως η λύση της ανίσωσης είναι
\[
x>2.
\]
\end{erwthma}
\end{thema}
