\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \ln x$ με πεδίο ορισμού $(0, +\infty)$.
\begin{erwthma}
\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για τη συνάρτηση $f(t) = \ln t$ στο κλειστό διάστημα $[x, x+1]$ με $x > 0$:
- Η $f$ είναι συνεχής στο $[x, x+1]$.
- Η $f$ είναι παραγωγίσιμη στο $(x, x+1)$ με $f'(t) = \frac{1}{t}$.
Κατά συνέπεια, existe τουλάχιστον ένα $\xi \in (x, x+1)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(x+1) - f(x)}{x+1 - x} \Leftrightarrow \frac{1}{\xi} = \frac{\ln(x+1) - \ln x}{1} \Leftrightarrow \frac{1}{\xi} = \ln(x+1) - \ln x \]

\item Από την προηγούμενη σχέση έχουμε: 
\[ \ln(x+1) - \ln x = \ln\left( \frac{x+1}{x} \right) = \ln\left( 1 + \frac{1}{x} \right) \]
Άρα $\ln\left( 1 + \frac{1}{x} \right) = \frac{1}{\xi}$.
Επίσης, η επιλογή του $\xi$ μάς διασφαλίζει ότι $x < \xi < x+1$.
Επειδή όλες οι ποσότητες είναι θετικές, η αντιστροφή των όρων αναστρέφει τη φορά των ανισοτήτων:
\[ \frac{1}{x+1} < \frac{1}{\xi} < \frac{1}{x} \]
Αντικαθιστώντας το $\frac{1}{\xi}$ με $\ln\left( 1 + \frac{1}{x} \right)$:
\[ \frac{1}{x+1} < \ln\left( 1 + \frac{1}{x} \right) < \frac{1}{x} \]
Το ζητούμενο αποδείχθηκε.

\item Πολλαπλασιάζουμε την προκύπτουσα ανισότητα με $x > 0$:
\[ x \cdot \frac{1}{x+1} < x \cdot \ln\left( 1 + \frac{1}{x} \right) < x \cdot \frac{1}{x} \]
\[ \frac{x}{x+1} < x \ln\left( 1 + \frac{1}{x} \right) < 1 \]
Υπολογίζουμε τα όρια των ακραίων μελών καθώς το $x \to +\infty$:
- $\lim_{x\to+\infty} \frac{x}{x+1} = \lim_{x\to+\infty} \frac{x}{x} = 1$.
- $\lim_{x\to+\infty} 1 = 1$.
Άρα, από το Κριτήριο Παρεμβολής, προκύπτει ότι:
\[ \lim_{x \to +\infty} \left[ x \ln\left( 1 + \frac{1}{x} \right) \right] = 1 \]

\item Ως δεύτερος τρόπος για το όριο, θέτουμε $u = \frac{1}{x}$. Όταν $x \to +\infty$, ισχύει $u \to 0^+$.
Το όριο μετασχηματίζεται:
\[ \lim_{x\to+\infty} \left[ x \ln\left( 1 + \frac{1}{x} \right) \right] = \lim_{u\to0^+} \left[ \frac{1}{u} \ln(1+u) \right] = \lim_{u\to0^+} \frac{\ln(1+u)}{u} \]
Το όριο αυτό οδηγεί σε απροσδιοριστία $\frac{0}{0}$. Μπορούμε να εφαρμόσουμε τον κανόνα De L' Hospital:
\[ \lim_{u\to0^+} \frac{(\ln(1+u))'}{(u)'} = \lim_{u\to0^+} \frac{\frac{1}{1+u}}{1} = \frac{1}{1+0} = 1 \]
Ο δεύτερος τρόπος επιβεβαιώνει απολύτως το αποτέλεσμα.
\end{erwthma}
\end{thema}
