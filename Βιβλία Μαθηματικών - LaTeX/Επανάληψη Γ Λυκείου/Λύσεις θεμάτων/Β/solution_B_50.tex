\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x) = \begin{cases}
x^3 + ax + \beta, & x < 1 \\
\ln x + 2x, & x \ge 1
\end{cases}.
\]
\begin{erwthma}
\item Για να είναι η $f$ παραγωγίσιμη στο $x_0=1$, πρέπει πρώτα να είναι συνεχής στο $1$.
Έχουμε:
\[
\lim_{x\to1^-}f(x)=1+a+\beta
\quad \text{και} \quad
f(1)=\ln1+2=2.
\]
Άρα:
\[ 1+a+\beta=2 \Leftrightarrow a+\beta=1. \]

Επίσης, πρέπει να ισχύει $f'_-(1)=f'_+(1)$. Για $x<1$ είναι $f'(x)=3x^2+a$, ενώ για $x>1$ είναι $f'(x)=\frac{1}{x}+2$. Οπότε:
\[
f'_-(1)=3+a
\quad \text{και} \quad
f'_+(1)=3.
\]
Άρα $3+a=3\Leftrightarrow a=0$. Από τη σχέση $a+\beta=1$ παίρνουμε $\beta=1$.

\item Για $a=0$ και $\beta=1$ έχουμε:
\[
f(x)=\begin{cases}
x^3+1, & x<1\\
\ln x+2x, & x\ge1
\end{cases}.
\]
Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και:
\[
f'(x)=\begin{cases}
3x^2, & x<1\\
\dfrac{1}{x}+2, & x>1
\end{cases},
\quad \text{ενώ} \quad f'(1)=3.
\]
Τα κρίσιμα σημεία προκύπτουν από τη σχέση $f'(x)=0$. Για $x<1$ έχουμε $3x^2=0\Leftrightarrow x=0$, ενώ για $x>1$ είναι $\frac{1}{x}+2>0$.
Άρα το μοναδικό κρίσιμο σημείο είναι το $x=0$.

\item Από το προηγούμενο ερώτημα, για $x<1$ ισχύει $f'(x)=3x^2\ge0$, με μηδενισμό μόνο στο $x=0$, ενώ για $x\ge1$ η συνάρτηση είναι γνησίως αύξουσα, αφού $\frac{1}{x}+2>0$ για $x>1$ και $f'(1)=3$.
Επιπλέον, για $x<1$ έχουμε $f(x)=x^3+1<2=f(1)$, οπότε η $f$ είναι γνησίως αύξουσα σε όλο το $\mathbb{R}$.

Επειδή η $f$ είναι γνησίως αύξουσα και $f(0)=1$, έχουμε:
\[
f(x^2-x)>1 \Leftrightarrow f(x^2-x)>f(0)
\Leftrightarrow x^2-x>0.
\]
Άρα:
\[
x^2-x>0 \Leftrightarrow x(x-1)>0
\Leftrightarrow x\in(-\infty,0)\cup(1,+\infty).
\]

\item Στο διάστημα $[1,e]$ ισχύει $f(x)=\ln x+2x>0$. Άρα το ζητούμενο εμβαδόν είναι:
\begin{align*}
E&=\int_1^e f(x)\,\d x
=\int_1^e(\ln x+2x)\,\d x\\
&=\left[x\ln x-x+x^2\right]_1^e
=e^2.
\end{align*}
\end{erwthma}
\end{thema}
