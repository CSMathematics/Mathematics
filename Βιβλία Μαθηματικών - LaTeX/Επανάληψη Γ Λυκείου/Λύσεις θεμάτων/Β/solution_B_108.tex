\begin{thema}{Β}
Στο σχήμα δίνεται η γραφική παράσταση της παραγώγου $f'$ στο $[0,6]$. Παρατηρούμε ότι η $f'$ είναι μη αρνητική σε όλο το διάστημα και μηδενίζεται μόνο στο $x=3$.

\begin{erwthma}
\item Από το σχήμα βλέπουμε ότι
\[
f'(x)>0,\quad x\in[0,3)\cup(3,6],
\]
ενώ
\[
f'(3)=0.
\]
Η $f'$ δεν αλλάζει πρόσημο στο $x=3$, αφού είναι θετική και αριστερά και δεξιά του $3$.

Επομένως η $f$ δεν παρουσιάζει τοπικό ακρότατο στο $x=3$.

\item Η $f$ είναι γνησίως αύξουσα στο $[0,6]$. Πράγματι, για κάθε $x_1,x_2\in[0,6]$ με $x_1<x_2$, από το σχήμα έχουμε $f'(x)\geq 0$ στο $[x_1,x_2]$ και $f'(x)>0$ εκτός, το πολύ, από το σημείο $x=3$.

Άρα το εμβαδόν κάτω από τη γραφική παράσταση της $f'$ στο $[x_1,x_2]$ είναι θετικό, οπότε
\[
f(x_2)-f(x_1)=\dint_{x_1}^{x_2}f'(x)\d x>0.
\]
Επομένως
\[
x_1<x_2\Rightarrow f(x_1)<f(x_2),
\]
δηλαδή η $f$ είναι \textbf{γνησίως αύξουσα} στο $[0,6]$.

\item Δίνεται ότι $f(3)=2$. Τότε
\[
\lim_{x\to 3}\frac{f(x)-2}{x-3}
=\lim_{x\to 3}\frac{f(x)-f(3)}{x-3}
=f'(3).
\]
Από το σχήμα έχουμε $f'(3)=0$, άρα
\[
\lim_{x\to 3}\frac{f(x)-2}{x-3}=0.
\]

Για το δεύτερο όριο, από το σχήμα η $C_{f'}$ είναι παραβολή με κορυφή το $(3,0)$, οπότε
\[
f'(x)=a(x-3)^2,\quad a>0.
\]
Συγκεκριμένα, από το δοσμένο σχήμα προκύπτει $a=0{,}12$, αλλά η τιμή του $a$ δεν επηρεάζει το όριο.

Για $x\neq 3$, γράφουμε με τον κανόνα \eng{De L'Hospital}:
\[
\lim_{x\to 3}\frac{f(x)-2}{(x-3)^2}
\xlongequal{\text{\eng{DLH}}}
\lim_{x\to 3}\frac{f'(x)}{2(x-3)}
=\lim_{x\to 3}\frac{a(x-3)^2}{2(x-3)}
=\lim_{x\to 3}\frac{a}{2}(x-3)=0.
\]
Άρα
\[
\lim_{x\to 3}\frac{f(x)-2}{(x-3)^2}=0.
\]

\item Από το Β2 η $f$ είναι γνησίως αύξουσα στο $[0,6]$. Εφόσον $f(0)=1$, για κάθε $x\in[0,6]$ έχουμε
\[
x\geq 0 \Rightarrow f(x)\geq f(0)=1.
\]
Επομένως
\[
f(x)\geq 1,\quad x\in[0,6].
\]
\end{erwthma}
\end{thema}
