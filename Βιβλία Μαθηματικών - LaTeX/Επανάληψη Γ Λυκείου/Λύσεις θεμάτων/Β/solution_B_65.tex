%=========== ΛΥΣΗ - 65ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Επειδή η $C_f$ έχει οριζόντια ασύμπτωτη την ευθεία $y=-1$ στο $-\infty$, έχουμε
\[
\lim_{x\to-\infty} f(x)=-1.
\]
Όμως
\[
\lim_{x\to-\infty}\frac{ae^x+\beta}{e^x+1}
=\frac{a\cdot 0+\beta}{0+1}=\beta.
\]
Άρα
\[
\beta=-1.
\]

Επίσης, επειδή η $C_f$ έχει οριζόντια ασύμπτωτη την ευθεία $y=2$ στο $+\infty$, έχουμε
\[
\lim_{x\to+\infty} f(x)=2.
\]
Για το όριο αυτό γράφουμε
\[
\frac{ae^x+\beta}{e^x+1}
=\frac{a+\frac{\beta}{e^x}}{1+\frac{1}{e^x}}.
\]
Άρα
\[
\lim_{x\to+\infty}\frac{ae^x+\beta}{e^x+1}=a.
\]
Επομένως,
\[
a=2.
\]

Έτσι
\[
f(x)=\frac{2e^x-1}{e^x+1}.
\]

\item Για κάθε $x\in\mathbb{R}$ έχουμε
\[
f'(x)=\frac{2e^x(e^x+1)-(2e^x-1)e^x}{(e^x+1)^2}
=\frac{3e^x}{(e^x+1)^2}.
\]

Επειδή
\[
f'(x)>0,\quad x\in\mathbb{R},
\]
η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$. Άρα είναι $1-1$ και επομένως αντιστρέφεται.

Το σύνολο τιμών της είναι
\[
(-1,2),
\]
αφού
\[
\lim_{x\to-\infty}f(x)=-1
\quad \text{και}\quad
\lim_{x\to+\infty}f(x)=2.
\]

Για να βρούμε την αντίστροφη, θέτουμε
\[
y=\frac{2e^x-1}{e^x+1}.
\]
Τότε
\[
y(e^x+1)=2e^x-1
\]
οπότε
\[
ye^x+y=2e^x-1.
\]
Άρα
\[
y+1=e^x(2-y).
\]
Επειδή $y\in(-1,2)$, έχουμε $y+1>0$ και $2-y>0$, οπότε
\[
e^x=\frac{y+1}{2-y}.
\]
Λογαριθμίζοντας, παίρνουμε
\[
x=\ln\left(\frac{y+1}{2-y}\right).
\]

Επομένως,
\[
f^{-1}(x)=\ln\left(\frac{x+1}{2-x}\right),\quad x\in(-1,2).
\]

\item Έχουμε
\[
f(0)=\frac{2e^0-1}{e^0+1}=\frac{1}{2}.
\]
Άρα
\[
\lim_{x\to 0}\frac{f(x)-\frac{1}{2}}{x}
=\lim_{x\to 0}\frac{f(x)-f(0)}{x-0}
=f'(0).
\]

Από τον τύπο της παραγώγου,
\[
f'(0)=\frac{3e^0}{(e^0+1)^2}=\frac{3}{4}.
\]

Επομένως,
\[
\lim_{x\to 0}\frac{f(x)-\frac{1}{2}}{x}=\frac{3}{4}.
\]

\item Υπολογίζουμε
\[
I=\int_0^{\ln 2} f(x)\,\d x
=\int_0^{\ln 2}\frac{2e^x-1}{e^x+1}\,\d x.
\]

Γράφουμε
\[
\frac{2e^x-1}{e^x+1}
=\frac{2(e^x+1)-3}{e^x+1}
=2-\frac{3}{e^x+1}.
\]
Άρα
\[
I=\int_0^{\ln 2}2\,\d x-3\int_0^{\ln 2}\frac{1}{e^x+1}\,\d x.
\]

Για το δεύτερο ολοκλήρωμα, πολλαπλασιάζουμε αριθμητή και παρονομαστή με $e^{-x}$:
\[
\int_0^{\ln 2}\frac{1}{e^x+1}\,\d x
=\int_0^{\ln 2}\frac{e^{-x}}{1+e^{-x}}\,\d x.
\]
Επειδή
\[
\left(\ln(1+e^{-x})\right)'=-\frac{e^{-x}}{1+e^{-x}},
\]
έχουμε
\[
\int_0^{\ln 2}\frac{e^{-x}}{1+e^{-x}}\,\d x
=-\left[\ln(1+e^{-x})\right]_0^{\ln 2}.
\]

Άρα
\begin{align*}
I
&=2\ln 2+3\left[\ln(1+e^{-x})\right]_0^{\ln 2}\\
&=2\ln 2+3\left(\ln\left(1+\frac{1}{2}\right)-\ln 2\right)\\
&=2\ln 2+3\left(\ln\frac{3}{2}-\ln 2\right)\\
&=2\ln 2+3\ln\frac{3}{4}\\
&=3\ln 3-4\ln 2.
\end{align*}
\end{erwthma}
\end{thema}
