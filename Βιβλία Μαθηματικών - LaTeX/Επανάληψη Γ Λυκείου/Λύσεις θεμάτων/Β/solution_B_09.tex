\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = \dfrac{x}{x+1}$ με $D_f = \mathbb{R} - \{-1\}$ και $g(x) = e^x$ με $D_g = \mathbb{R}$.

\begin{erwthma}
\item Η συνάρτηση $h = f \circ g$ έχει:
\begin{itemize}
\item τύπο: $h(x) = f(g(x)) = f(e^x) = \dfrac{e^x}{e^x + 1}$.
\item πεδίο ορισμού:\[ D_h = \{x \in D_g \mid g(x) \in D_f\}  \]
\begin{itemize}[label=\tiny\faPlay]
\item $x\in D_g\Leftrightarrow x\in\mathbb{R}$ και 
\item $g(x)\in D_f\Leftrightarrow e^x\in\mathbb{R}-\{-1\}\Leftrightarrow e^x\neq -1\Leftrightarrow x\in\mathbb{R}$
\end{itemize}
\end{itemize}
οπότε $h(x)=\dfrac{e^x}{e^x+1}\ ,\ x\in\mathbb{R}$.
\item Η $h(x) = \dfrac{e^x}{e^x+1}$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ h'(x) = \left( \dfrac{e^x}{e^x+1} \right)' = \dfrac{e^x(e^x+1) - e^x(e^x+1)'}{(e^x+1)^2} = \dfrac{e^{2x} + e^x - e^{2x}}{(e^x+1)^2} = \dfrac{e^x}{(e^x+1)^2}>0 \]
Άρα η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$, άρα είναι συνάρτηση $1-1$ και συνεπώς αντιστρέφεται. Το πεδίο ορισμού της $h^{-1}$ είναι το σύνολο τιμών $h(\mathbb{R}) = \left( \lim\limits_{x\to-\infty} h(x), \lim\limits_{x\to+\infty} h(x) \right)$.
\begin{itemize}
\item $\lim\limits_{x\to-\infty} h(x) = \lim\limits_{x\to-\infty} \dfrac{e^x}{e^x+1} = \dfrac{0}{0+1} = 0$.
\item $\lim\limits_{x\to+\infty} h(x) = \lim\limits_{x\to+\infty} \dfrac{e^x}{e^x+1}\dlh{\frac{+\infty}{+\infty}} \lim\limits_{x\to+\infty} \dfrac{e^x}{e^{x}} = 1$.
\end{itemize}
Άρα $D_{h^{-1}} = (0, 1)$.
Λύνουμε την εξίσωση $y = h(x)$ ως προς $x$:
\begin{gather*}
y = \dfrac{e^x}{e^x+1} \Leftrightarrow y(e^x+1) = e^x \Leftrightarrow\\ ye^x + y = e^x \Leftrightarrow ye^x - e^x = -y \Leftrightarrow e^x(y-1) = -y \\
e^x = \dfrac{y}{1-y}\Leftrightarrow x = \ln\left(\dfrac{y}{1-y}\right)
\end{gather*}
Άρα ο τύπος της αντίστροφης είναι $h^{-1}(x) = \ln\left(\dfrac{x}{1-x}\right)$, με $x \in (0, 1)$.

\item Εξετάζουμε το όριο \[\displaystyle\lim\limits_{x \to 1^-} \dfrac{h^{-1}(x)}{\ln(1-x)} = \lim\limits_{x \to 1^-} \dfrac{\ln\left(\dfrac{x}{1-x}\right)}{\ln(1-x)} = \lim\limits_{x \to 1^-} \dfrac{\ln x - \ln(1-x)}{\ln(1-x)} = \lim\limits_{x \to 1^-} \left( \dfrac{\ln x}{\ln(1-x)} - 1 \right)=\frac{0}{-\infty}-1=-1\]
\item Θέτουμε τη συνάρτηση $\varphi(x) = h(x) - x = \dfrac{e^x}{e^x+1} - x$, με $x \in [0, 1]$.
\begin{itemize}
\item Η $\varphi(x)$ είναι συνεχής στο $[0, 1]$ ως διαφορά συνεχών συναρτήσεων.
Ισχύει:
\item $\varphi(0) = h(0) - 0 = \dfrac{e^0}{e^0+1} = \dfrac{1}{2} > 0$.
\item $\varphi(1) = h(1) - 1 = \dfrac{e}{e+1} - 1 = \dfrac{e - e - 1}{e+1} = \dfrac{-1}{e+1} < 0$. Άρα $\varphi(0) \cdot \varphi(1) < 0$,
\end{itemize}
 σύμφωνα με το Θεώρημα Bolzano, η εξίσωση $\varphi(x) = 0 \Leftrightarrow h(x) = x$ έχει τουλάχιστον μία ρίζα στο διάστημα $(0, 1)$.
\end{erwthma}
\end{thema}
