\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2+1}{x}$ με $x \ne 0$.

\begin{erwthma}
\item Η $f(x)$ είναι παραγωγίσιμη στο $\mathbb{R}^*$ με:
\[ f'(x) = \left( \frac{x^2+1}{x} \right)' = \left( x + \frac{1}{x} \right)' = 1 - \frac{1}{x^2} = \frac{x^2-1}{x^2} \]
Για τις ρίζες και το πρόσημο της $f'$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow \frac{x^2-1}{x^2} = 0 \Leftrightarrow x^2 - 1 = 0 \Leftrightarrow x = 1 \text{ ή } x = -1$.
    \item $f'(x) > 0 \Leftrightarrow \frac{x^2-1}{x^2} > 0 \stackrel{x^2>0}{\Leftrightarrow} x^2 - 1 > 0 \Leftrightarrow |x| > 1 \Leftrightarrow x < -1 \text{ ή } x > 1$.
    \item $f'(x) < 0 \Leftrightarrow -1 < x < 0 \text{ ή } 0 < x < 1$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Συνεπώς η $f$ είναι γνησίως αύξουσα στο $(-\infty, -1]$, γνησίως φθίνουσα στο $[-1, 0)$, γνησίως φθίνουσα στο $(0, 1]$ και γνησίως αύξουσα στο $[1, +\infty)$.
\begin{itemize}
\item Παρουσιάζει τοπικό μέγιστο στο $x = -1$, με τιμή $f(-1) = \frac{(-1)^2+1}{-1} = -2$.
\item Παρουσιάζει τοπικό ελάχιστο στο $x = 1$, με τιμή $f(1) = \frac{1^2+1}{1} = 2$.
\end{itemize}

\item Η $f'(x) = 1 - x^{-2}$ παραγωγίζεται στο $\mathbb{R}^*$ με:
\[ f''(x) = (1 - x^{-2})' = 2x^{-3} = \frac{2}{x^3} \]
Το πρόσημο της δεύτερης παραγώγου εξαρτάται μόνο από το $x^3$:
\begin{itemize}
    \item $f''(x) > 0 \Leftrightarrow x^3 > 0 \Leftrightarrow x > 0$. Άρα η $f$ είναι κυρτή στο $(0, +\infty)$.
    \item $f''(x) < 0 \Leftrightarrow x^3 < 0 \Leftrightarrow x < 0$. Άρα η $f$ είναι κοίλη στο $(-\infty, 0)$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:

\end{center}
Εφόσον η συνάρτηση αλλάζει κυρτότητα εκατέρωθεν του $x=0$, αλλά το $x=0$ δεν ανήκει στο πεδίο ορισμού της, η γραφική της παράσταση δεν έχει σημεία καμπής.

\item Βρίσκουμε τις ασύμπτωτες:
\textbf{Κατακόρυφη:} Εξετάζουμε το όριο στο $0$:
\[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} \left( (x^2+1)\frac{1}{x} \right) = 1 \cdot (+\infty) = +\infty \]
Άρα η ευθεία $x=0$ (άξονας $y'y$) είναι κατακόρυφη ασύμπτωτη της $C_f$.
\textbf{Οριζόντιες/Πλάγιες:} Στο $+\infty$ έχουμε:
\[\lim\limits_{x\to +\infty}{\frac{f(x)}{x}}=\lim\limits_{x\to +\infty}{\frac{x^2+1}{x^2}}=\lim\limits_{x\to+\infty}{\frac{x^2}{x^2}}=1=\lambda\]
Επίσης
\[\lim\limits_{x\to +\infty}{(f(x)-\lambda x)}=\lim\limits_{x\to +\infty}{\left(x+\frac{1}{x}-x\right)}=\lim\limits_{x\to +\infty}{\frac{1}{x}}=0=\beta\]
Επομένως, η ευθεία $y = x$ είναι πλάγια ασύμπτωτη της $C_f$ τόσο στο $+\infty$ όσο και στο $-\infty$.

\item Το χωρίο περικλείεται από την $C_f$, την ασύμπτωτη $y=x$ στο $+\infty$, και τις κατακόρυφες $x=1$, $x=e$. 
Από τον τύπο $f(x) - x = \frac{1}{x} > 0$ για κάθε $x \in [1, e]$, το ζητούμενο εμβαδόν είναι:
\[ E = \int_1^e |f(x) - x|\,\d x = \int_1^e \frac{1}{x}\,\d x = [\ln x]_1^e = \ln e - \ln 1 = 1 - 0 = 1 \]
\end{erwthma}
\end{thema}
