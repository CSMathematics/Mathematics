\begin{thema}{Β}
\begin{erwthma}
\item Η εφαπτομένη διέρχεται από τα σημεία $M(0,1)$ και $A(-1,0)$, άρα έχει κλίση
\[
\lambda=\frac{1-0}{0-(-1)}=1.
\]
Επομένως
\[
y-1=1(x-0)\Leftrightarrow \epsilon:\ y=x+1.
\]

\item Επειδή $f(x)=e^x$, έχουμε $f(0)=e^0=1$ και $f'(x)=e^x$, άρα $f'(0)=1$.

\item Αφού $f(0)=1$,
\[
\lim_{x\to0}\frac{f(x)-1}{x}
=\lim_{x\to0}\frac{f(x)-f(0)}{x-0}=f'(0)=1.
\]
Με \eng{De L'Hospital}, η μορφή είναι $\dfrac00$ και
\[
\lim_{x\to0}\frac{f(x)-1}{x}
\xlongequal{\text{\eng{DLH}}}
\lim_{x\to0}\frac{f'(x)}{1}
=\lim_{x\to0}e^x=1.
\]

\item Από το σχήμα η $C_f$ βρίσκεται πάνω από την εφαπτομένη της, άρα η $f$ είναι κυρτή. Για μια δύο φορές παραγωγίσιμη συνάρτηση αυτό αντιστοιχεί σε μη αρνητική δεύτερη παράγωγο. Επειδή εδώ $f(x)=e^x$, έχουμε ειδικότερα
\[
f''(x)=e^x>0,\quad x\in\mathbb{R}.
\]
\end{erwthma}
\end{thema}
