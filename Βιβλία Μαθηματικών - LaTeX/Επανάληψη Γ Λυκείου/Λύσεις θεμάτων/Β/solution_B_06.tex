\begin{thema}{Β}
Δίνονται οι συναρτήσεις $g:(1,+\infty)\to\mathbb{R}$ με τύπο $g(x)=\ln\left(x-\dfrac{1}{x}\right)$ και $h:\mathbb{R}\to\mathbb{R}$ με $h(x)=e^x$.

\begin{erwthma}
\item Η συνάρτηση $f=h\circ g$ έχει:
\begin{itemize}
\item τύπο: $f(x) = h(g(x)) = e^{\ln(x-\frac{1}{x})} = x - \dfrac{1}{x}$
\item πεδίο ορισμού: $D_f = \{x\in D_g \mid g(x)\in D_h\}$
\begin{itemize}[label=\tiny\faPlay]
\item $x\in D_g\Leftrightarrow x\in(1,+\infty)$ και 
\item $g(x)\in D_h\Leftrightarrow \ln{\left(1-\dfrac{1}{x}\right)}\in\mathbb{R}\Leftrightarrow x-\dfrac{1}{x}>0\xRightarrow{x>1}x^2-1>0\Leftrightarrow x>1$
\end{itemize}
\end{itemize}
Επομένως, $f(x) = x - \dfrac{1}{x}$ για $x \in (1, +\infty)$.

\item Το πεδίο ορισμού της $k(x) = x - \dfrac{1}{x}$ είναι $D_k = \mathbb{R} \setminus \{0\}$.
Εφόσον $D_f = (1, +\infty) \ne D_k$, οι συναρτήσεις $f$ και $k$ \textbf{δεν είναι ίσες}.
Για να είναι ίσες, θα πρέπει να ορίζονται στο ίδιο υποσύνολο του $\mathbb{R}$. Το ευρύτερο κοινό υποσύνολο είναι η τομή των πεδίων ορισμού τους, δηλαδή $D_f \cap D_k = (1, +\infty) \cap (\mathbb{R} \setminus \{0\}) = (1, +\infty)$.
Και για κάθε $x \in (1, +\infty)$ ισχύει ότι $f(x) = k(x) = x - \dfrac{1}{x}$.

\item Η $f(x) = x - \dfrac{1}{x}$ είναι παραγωγίσιμη στο $(1, +\infty)$ με:
\[f'(x) = 1 - \left(-\dfrac{1}{x^2}\right) = 1 + \dfrac{1}{x^2}>0\]
για κάθε $x \in (1, +\infty)$, η $f$ είναι γνησίως αύξουσα στο πεδίο ορισμού της, άρα είναι συνάρτηση '1-1' και συνεπώς αντιστρέφεται. Το πεδίο ορισμού της $f^{-1}$ είναι το σύνολο τιμών $f(A) = f((1, +\infty))$. Εφόσον $f$ γνησίως αύξουσα:
$f(A) = \left( \lim\limits_{x\to1^+} f(x), \lim\limits_{x\to+\infty} f(x) \right)$.

\begin{itemize}
\item $\lim\limits_{x\to1^+} f(x) = \lim\limits_{x\to1^+} \left(x - \dfrac{1}{x}\right) = 1 - 1 = 0$.
\item $\lim\limits_{x\to+\infty} f(x) = \lim\limits_{x\to+\infty} \left(x - \dfrac{1}{x}\right) = +\infty - 0 = +\infty$.
\end{itemize}

Άρα $D_{f^{-1}} = (0, +\infty)$. Για την εύρεση της $f^{-1}(y)$ θέτουμε $y=f(x)\Leftrightarrow x=f^{-1}(y)$:
\[ y = x - \dfrac{1}{x} \Leftrightarrow xy = x^2 - 1 \Leftrightarrow x^2 - yx - 1 = 0 \]
Πρόκειται για εξίσωση 2ου βαθμού ως προς $x$. Η διακρίνουσα είναι $\Delta = y^2 + 4 > 0$. Άρα έχει λύσεις:
$x = \dfrac{y \pm \sqrt{y^2+4}}{2}$.
Όμως $x \in (1, +\infty) \Rightarrow x > 1$.
Αν $x = \dfrac{y - \sqrt{y^2+4}}{2}$, τότε επειδή $\sqrt{y^2+4} > \sqrt{y^2} = |y| \ge y$, προκύπτει ότι $y - \sqrt{y^2+4} < 0$, άρα $x < 0$, που απορρίπτεται.
Οπότε $x = \dfrac{y + \sqrt{y^2+4}}{2}$, η οποία είναι δεκτή (για $y>0$ η ποσότητα είναι όντως μεγαλύτερη του 1).
Άρα ο τύπος της αντίστροφης είναι $f^{-1}(x) = \dfrac{x + \sqrt{x^2+4}}{2}$ με $x > 0$.

\item Υπολογίζουμε το όριο:
\begin{align*}
\lim\limits_{x\to+\infty} \dfrac{f^{-1}(x)}{x} &= \lim\limits_{x\to+\infty} \dfrac{\dfrac{x + \sqrt{x^2+4}}{2}}{x} = \lim\limits_{x\to+\infty} \dfrac{x + |x|\sqrt{1+4/x^2}}{2x} \\
&\stackrel{x>0}{=} \lim\limits_{x\to+\infty} \dfrac{x + x\sqrt{1+4/x^2}}{2x} = \lim\limits_{x\to+\infty} \dfrac{x(1 + \sqrt{1+4/x^2})}{2x} \\
&= \lim\limits_{x\to+\infty} \dfrac{1 + \sqrt{1+4/x^2}}{2} = \dfrac{1 + \sqrt{1+0}}{2} = \dfrac{2}{2} = 1
\end{align*}
\end{erwthma}
\end{thema}
