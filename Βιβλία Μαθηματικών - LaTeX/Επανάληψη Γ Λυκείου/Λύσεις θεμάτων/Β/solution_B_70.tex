\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=e^{x-a}+\beta x,\quad x\in\mathbb{R}.
\]

\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=e^{x-a}+\beta.
\]
Αφού η $f$ παρουσιάζει ακρότατο στο σημείο $M(a,0)$, έχουμε $f(a)=0$ και, από το θεώρημα \eng{Fermat}, $f'(a)=0$. Άρα
\[
f(a)=0\Leftrightarrow 1+a\beta=0
\quad\text{και}\quad
f'(a)=0\Leftrightarrow 1+\beta=0.
\]
Από τη δεύτερη σχέση παίρνουμε $\beta=-1$ και από την πρώτη
\[
1-a=0\Leftrightarrow a=1.
\]
Επομένως
\[
a=1,\quad \beta=-1
\]
και
\[
f(x)=e^{x-1}-x.
\]

\item Για $a=1$ και $\beta=-1$ έχουμε
\[
f'(x)=e^{x-1}-1.
\]
Βρίσκουμε ρίζες και πρόσημο της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow e^{x-1}=1\Leftrightarrow x=1$,
\item $f'(x)<0\Leftrightarrow x<1$,
\item $f'(x)>0\Leftrightarrow x>1$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}

\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$. Συνεπώς παρουσιάζει \textbf{ολικό ελάχιστο} στο $x=1$, με τιμή
\[
f(1)=0.
\]

\item Από το προηγούμενο ερώτημα, για κάθε $x\in\mathbb{R}$ ισχύει $f(x)\geq f(1)=0$. Επομένως
\[
e^{x-1}-x\geq0\Leftrightarrow e^{x-1}\geq x,\quad x\in\mathbb{R}.
\]

\item Στο διάστημα $[0,1]$ έχουμε $f(x)\geq0$, αφού το ολικό ελάχιστο της $f$ είναι $f(1)=0$. Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\int_0^1 f(x)\d x
=\int_0^1\left(e^{x-1}-x\right)\d x\\
&=\left[e^{x-1}-\frac{x^2}{2}\right]_0^1
=\left(1-\frac{1}{2}\right)-\frac{1}{e}
=\frac{1}{2}-\frac{1}{e}.
\end{align*}
\end{erwthma}
\end{thema}
