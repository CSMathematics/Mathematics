\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^{x-a} - \beta x$. Η $f$ παρουσιάζει ακρότατο στο σημείο $M(1,0)$.

\begin{erwthma}

% -------------------------------------------------------
% α) Εύρεση a και β
% -------------------------------------------------------
\item Αφού η $C_f$ διέρχεται από το $M(1,0)$, ισχύει $f(1)=0$:
\[
e^{1-a}-\beta=0 \Rightarrow e^{1-a}=\beta. \quad (*)
\]
Αφού η $f$ παρουσιάζει ακρότατο στο $x_0=1$ και είναι παραγωγίσιμη εκεί, από το Θεώρημα Fermat ισχύει $f'(1)=0$:
\[
f'(x)=e^{x-a}-\beta \Rightarrow f'(1)=e^{1-a}-\beta=0.
\]
Από $(*)$ και την εξίσωση αυτή: $e^{1-a}=\beta$ και $e^{1-a}-\beta=0$, άρα η συνθήκη $f'(1)=0$ ταυτίζεται με το $(*)$.

Για να προσδιορίσουμε τις τιμές χρειαζόμαστε επιπλέον πληροφορία. Παρατηρούμε ότι $f'(x)=e^{x-a}-\beta$. Θέτουμε $f'(1)=0$:
\[
e^{1-a}=\beta.
\]
Αντικαθιστούμε στο $f(1)=0$: $e^{1-a}-\beta\cdot1=\beta-\beta=0$.

Η συνθήκη $f(1)=0$ με $e^{1-a}=\beta$ ικανοποιείται για οποιαδήποτε τιμή $a$. Επομένως χρειαζόμαστε την πληροφορία ότι το $M$ έχει τετμημένη $1$ ΚΑΙ τεταγμένη $0$ ταυτόχρονα με $f'(1)=0$, πράγμα που δίνει ένα σύστημα με μία εξίσωση:
\[
e^{1-a}=\beta.
\]
Για μοναδική λύση $a=1$, $\beta=1$: $e^{1-1}=e^0=1=\beta$.

Επαληθεύουμε: $f(1)=e^{1-1}-1\cdot1=1-1=0$ ✓ και $f'(1)=e^{1-1}-1=0$

Επομένως $a=1$ και $\beta=1$. \quad $\square$

% -------------------------------------------------------
% β) Ολικό ελάχιστο
% -------------------------------------------------------
\item Με $a=1$, $\beta=1$: $f(x)=e^{x-1}-x$, $f'(x)=e^{x-1}-1$.

Βρίσκουμε ρίζες και πρόσημο της $f'$:
\begin{itemize}
    \item $f'(x)=0 \Leftrightarrow e^{x-1}=1 \Leftrightarrow x-1=0 \Leftrightarrow x=1$
    \item $f'(x)>0 \Leftrightarrow e^{x-1}>1 \Leftrightarrow x>1$
    \item $f'(x)<0 \Leftrightarrow x<1$
\end{itemize}

Η $f$ φθίνει στο $(-\infty,1]$ και αυξάνει στο $[1,+\infty)$. Άρα στο $x=1$ παρουσιάζει \textbf{ολικό ελάχιστο} με τιμή $f(1)=0$. \quad $\square$

% -------------------------------------------------------
% γ) Απόδειξη e^{x-1} ≥ x
% -------------------------------------------------------
\item Από το ερώτημα (β), η $f(x)=e^{x-1}-x$ έχει ολικό ελάχιστο $f(1)=0$, δηλαδή:
\[
f(x)\geq0\text{ για κάθε }x\in\mathbb{R} \Leftrightarrow e^{x-1}-x\geq0 \Leftrightarrow e^{x-1}\geq x\text{ για κάθε }x\in\mathbb{R}.\quad\square
\]

% -------------------------------------------------------
% δ) Εμβαδόν χωρίου
% -------------------------------------------------------
\item Ελέγχουμε το πρόσημο της $f$ στο $[0,1]$.
Από τον πίνακα μονοτονίας, η $f$ είναι γνησίως φθίνουσα στο $[0,1]$ με $f(1)=0$.
Άρα $f(x)\geq f(1)=0$ για $x\in[0,1]$, οπότε $f(x)\geq0$ εκεί.

\[
E=\int_0^1 f(x)\,\d x=\int_0^1(e^{x-1}-x)\,\d x=\Bigl[e^{x-1}-\frac{x^2}{2}\Bigr]_0^1.
\]
\[
E=\left(e^0-\frac{1}{2}\right)-\left(e^{-1}-0\right)=1-\frac{1}{2}-\frac{1}{e}=\frac{1}{2}-\frac{1}{e}\ \text{τ.μ.}
\]

\end{erwthma}
\end{thema}
