\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x(x-2)e^x$ με πεδίο ορισμού το $\mathbb{R}$.
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση $F(x) = f(x)-3 = x(x-2)e^x-3$, η οποία είναι συνεχής στο $[2,3]$.
Υπολογίζουμε τις τιμές της στα άκρα:
\begin{itemize}
\item $F(2)=2(2-2)e^2-3=-3<0$.
\item $F(3)=3(3-2)e^3-3=3e^3-3=3(e^3-1)>0$.
\end{itemize}
Επειδή $F(2)F(3)<0$, από το Θεώρημα \eng{Bolzano} υπάρχει τουλάχιστον ένα $x_0\in(2,3)$ τέτοιο ώστε $F(x_0)=0$, δηλαδή $f(x_0)=3$.

\item Γράφουμε την $f(x)$ ως πηλίκο για να υπολογίσουμε το όριο στο $-\infty$:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} (x^2-2x)e^x = \lim_{x\to-\infty} \frac{x^2-2x}{e^{-x}} \]
Επειδή έχουμε απροσδιοριστία μορφής $\frac{+\infty}{+\infty}$, εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x\to-\infty} \frac{x^2-2x}{e^{-x}} \dlh{\frac{+\infty}{+\infty}} \lim_{x\to-\infty} \frac{2x-2}{-e^{-x}} \dlh{\frac{-\infty}{-\infty}} \lim_{x\to-\infty} \frac{2}{e^{-x}} = 0 \]

\item Η $f$ είναι παραγωγίσιμη με:
\[ f'(x) = (x^2-2x)'e^x + (x^2-2x)(e^x)' = (2x-2)e^x + (x^2-2x)e^x = e^x(x^2-2) \]
Οι ρίζες της παραγώγου είναι $x_1=-\sqrt{2}$ και $x_2=\sqrt{2}$.
Επειδή $e^x > 0$, το πρόσημο της $f'$ εξαρτάται από το τριώνυμο $x^2-2$.
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Η συνάρτηση είναι γνησίως αύξουσα στα διαστήματα $(-\infty, -\sqrt{2}]$ και $[\sqrt{2}, +\infty)$, ενώ είναι γνησίως φθίνουσα στο $[-\sqrt{2}, \sqrt{2}]$.
Παρουσιάζει τοπικό μέγιστο στο $x_1 = -\sqrt{2}$ με τιμή
\[ f(-\sqrt{2}) = (2+2\sqrt{2})e^{-\sqrt{2}}, \]
και τοπικό ελάχιστο στο $x_2 = \sqrt{2}$ με τιμή
\[ f(\sqrt{2}) = (2-2\sqrt{2})e^{\sqrt{2}}. \]

\item Από τον πίνακα μεταβολών και τα όρια $\lim\limits_{x\to-\infty} f(x)=0$ και $\lim\limits_{x\to+\infty} f(x)=+\infty$, η μικρότερη τιμή της $f$ είναι η τιμή του τοπικού ελαχίστου:
\[ f(\sqrt{2}) = (2-2\sqrt{2})e^{\sqrt{2}}. \]
Επομένως το σύνολο τιμών της $f$ είναι:
\[ f(\mathbb{R}) = \left[(2-2\sqrt{2})e^{\sqrt{2}}, +\infty\right). \]
\end{erwthma}
\end{thema}
