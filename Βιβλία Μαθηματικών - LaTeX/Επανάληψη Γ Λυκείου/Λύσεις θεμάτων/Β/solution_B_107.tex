%=========== ΛΥΣΗ - 107ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
Στο σχήμα δίνεται η γραφική παράσταση της παραγώγου $f'$. Άρα η μονοτονία της $f$ προκύπτει από το πρόσημο της $f'$, ενώ η κυρτότητα της $f$ προκύπτει από τη μονοτονία της $f'$.

\begin{erwthma}
\item Από το σχήμα βλέπουμε ότι
\[
f'(x)>0,\quad x\in[-3,-1)
\]
και
\[
f'(x)<0,\quad x\in(-1,2).
\]
Επίσης
\[
f'(x)>0,\quad x\in(2,4].
\]
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $[-3,-1]$, \textbf{γνησίως φθίνουσα} στο $[-1,2]$ και \textbf{γνησίως αύξουσα} στο $[2,4]$.

Στο $x=-1$ η $f'$ αλλάζει πρόσημο από $+$ σε $-$, άρα η $f$ παρουσιάζει τοπικό μέγιστο. Στο $x=2$ η $f'$ αλλάζει πρόσημο από $-$ σε $+$, άρα η $f$ παρουσιάζει τοπικό ελάχιστο.

\item Δίνεται ότι $f(2)=3$. Τότε
\[
\lim_{x\to 2}\frac{f(x)-3}{x^2-4}
=\lim_{x\to 2}\frac{f(x)-f(2)}{(x-2)(x+2)}.
\]
Για $x\neq 2$ γράφουμε
\[
\frac{f(x)-f(2)}{(x-2)(x+2)}
=\frac{f(x)-f(2)}{x-2}\cdot\frac{1}{x+2}.
\]
Άρα
\[
\lim_{x\to 2}\frac{f(x)-3}{x^2-4}
=f'(2)\cdot\frac{1}{4}.
\]
Από το σχήμα της $f'$ έχουμε $f'(2)=0$. Επομένως
\[
\lim_{x\to 2}\frac{f(x)-3}{x^2-4}=0.
\]

\item Από το σχήμα βλέπουμε ότι η $f'$ είναι γνησίως φθίνουσα στο $\left[-3,\frac{1}{2}\right]$ και γνησίως αύξουσα στο $\left[\frac{1}{2},4\right]$.

Επομένως η $f$ είναι \textbf{κοίλη} στο $\left[-3,\frac{1}{2}\right]$ και \textbf{κυρτή} στο $\left[\frac{1}{2},4\right]$.

Στο $x=\frac{1}{2}$ η $f'$ αλλάζει μονοτονία, άρα η $f$ αλλάζει κυρτότητα. Επομένως η τετμημένη του σημείου καμπής της $C_f$ είναι
\[
x=\frac{1}{2}.
\]

\item Από το πρώτο ερώτημα η $f$ είναι γνησίως αύξουσα στο $[-3,-1]$ και γνησίως φθίνουσα στο $[-1,2]$. Άρα στο διάστημα $[-3,2]$ η $f$ παρουσιάζει μέγιστο στο $x=-1$.

Αφού δίνεται ότι $f(-1)=5$, έχουμε
\[
f(x)\leq 5,\quad x\in[-3,2].
\]
Επομένως δεν υπάρχει $x\in[-3,2]$ τέτοιο ώστε $f(x)=6$. Άρα η εξίσωση $f(x)=6$ είναι αδύνατη στο διάστημα $[-3,2]$.
\end{erwthma}
\end{thema}
