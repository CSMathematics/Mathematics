\begin{thema}{Β}
Από το σχήμα και την εκφώνηση γνωρίζουμε ότι η $C_f$ έχει οριζόντια ασύμπτωτη την ευθεία $y=-1$ στο $-\infty$ και πλάγια ασύμπτωτη την ευθεία $y=\dfrac{x}{2}$ στο $+\infty$.

\begin{erwthma}
\item Επειδή η $C_f$ έχει οριζόντια ασύμπτωτη την ευθεία $y=-1$ στο $-\infty$, έχουμε
\[
\lim_{x\to -\infty}f(x)=-1.
\]

Επειδή η $C_f$ έχει πλάγια ασύμπτωτη την ευθεία $y=\dfrac{x}{2}$ στο $+\infty$, ισχύει
\[
\lim_{x\to +\infty}\left(f(x)-\frac{x}{2}\right)=0.
\]
Άρα, για $x\to+\infty$, μπορούμε να γράψουμε
\[
f(x)=\frac{x}{2}+\varepsilon(x),\quad \text{όπου}\quad \varepsilon(x)\to 0.
\]
Επομένως
\[
\lim_{x\to+\infty}\frac{f(x)}{x}
=\lim_{x\to+\infty}\left(\frac{1}{2}+\frac{\varepsilon(x)}{x}\right)
=\frac{1}{2}.
\]

\item Από την πλάγια ασύμπτωτη, όπως είδαμε, έχουμε
\[
f(x)=\frac{x}{2}+\varepsilon(x),\quad \varepsilon(x)\to 0.
\]
Τότε
\begin{align*}
\lim_{x\to+\infty}\frac{x f(x)-x^2+3}{2x^2+x+1}
&=\lim_{x\to+\infty}
\frac{x\left(\frac{x}{2}+\varepsilon(x)\right)-x^2+3}{2x^2+x+1}\\
&=\lim_{x\to+\infty}
\frac{-\frac{x^2}{2}+x\varepsilon(x)+3}{2x^2+x+1}\\
&=\lim_{x\to+\infty}
\frac{-\frac{1}{2}+\frac{\varepsilon(x)}{x}+\frac{3}{x^2}}{2+\frac{1}{x}+\frac{1}{x^2}}
=-\frac{1}{4}.
\end{align*}

\item Από το σχήμα βλέπουμε ότι $f(0)=0$. Επειδή η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, ισχύει
\[
f(t)<0 \Leftrightarrow t<0.
\]
Θέτουμε
\[
t=|x-1|-2.
\]
Τότε
\[
f(|x-1|-2)<0
\Leftrightarrow
|x-1|-2<0
\Leftrightarrow
|x-1|<2.
\]
Άρα
\[
-2<x-1<2 \Leftrightarrow -1<x<3.
\]
Επομένως η λύση της ανίσωσης είναι
\[
x\in(-1,3).
\]

\item Αφού η $C_f$ βρίσκεται πάνω από την οριζόντια ασύμπτωτη $y=-1$, έχουμε
\[
f(x)>-1,\quad x\in\mathbb{R}.
\]
Επίσης, από το προηγούμενο ερώτημα, η $f$ είναι γνησίως αύξουσα και $f(0)=0$. Άρα για κάθε $x\in[-2,0)$ ισχύει $f(x)<f(0)=0$. Συνεπώς, στο διάστημα $[-2,0]$ έχουμε
\[
-1<f(x)\leq 0.
\]

Το χωρίο που περικλείεται από την $C_f$, τον άξονα $x'x$ και τις ευθείες $x=-2$, $x=0$ βρίσκεται κάτω από τον άξονα $x'x$. Επομένως
\[
E=\dint_{-2}^{0}|f(x)|\d x=\dint_{-2}^{0}(-f(x))\d x.
\]
Από τη σχέση $-1<f(x)\leq 0$ παίρνουμε
\[
0\leq -f(x)<1,\quad x\in[-2,0].
\]
Άρα
\[
E=\dint_{-2}^{0}(-f(x))\d x<\dint_{-2}^{0}1\d x=2.
\]
Συνεπώς $E<2$.
\end{erwthma}
\end{thema}
