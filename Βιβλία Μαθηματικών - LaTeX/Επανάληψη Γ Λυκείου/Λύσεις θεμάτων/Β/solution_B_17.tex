\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{e^x}{x^2}$ με $x \ne 0$.

\begin{erwthma}

% -------------------------------------------------------
% α) Όρια και ασύμπτωτες
% -------------------------------------------------------
\item \textbf{Υπολογισμός ορίων.}
Έχουμε:
\[
\lim\limits_{x\to 0}{f(x)}=\lim_{x \to 0} \frac{e^x}{x^2} = \frac{1}{0^+} = +\infty.
\]
και
\[
\lim\limits_{x\to -\infty}{f(x)}=\lim_{x \to -\infty} \frac{e^x}{x^2} = \frac{0}{+\infty}=0
\]

\textit{Κατακόρυφη ασύμπτωτη:} Το μόνο σημείο εκτός πεδίου ορισμού είναι το $x = 0$.
Αφού $\lim\limits_{x \to 0} f(x) = +\infty$, η ευθεία $x = 0$ είναι κατακόρυφη ασύμπτωτη.

\textit{Οριζόντιες ασύμπτωτες:}
\[
\lim_{x \to -\infty} f(x) = 0 \quad \text{(βρέθηκε παραπάνω)}, \qquad \lim_{x \to +\infty} \frac{e^x}{x^2} = +\infty.
\]
Άρα η ευθεία $y = 0$ είναι \textbf{οριζόντια ασύμπτωτη} μόνο στο $-\infty$. Στο $+\infty$ δεν υπάρχει οριζόντια ασύμπτωτη (ούτε πλάγια, αφού $\lim_{x\to+\infty}\frac{f(x)}{x}=+\infty$).

% -------------------------------------------------------
% β) Μονοτονία, τοπικά ακρότατα
% -------------------------------------------------------
\item Η $f$ είναι παραγωγίσιμη στο $(-\infty,0)\cup(0,+\infty)$ με:
\[
f'(x) = \frac{e^x \cdot x^2 - e^x \cdot 2x}{x^4} = \frac{e^x(x^2 - 2x)}{x^4} = \frac{e^x \cdot x(x-2)}{x^4} = \frac{e^x(x-2)}{x^3}.
\]
Βρίσκουμε ρίζες και πρόσημο της $f'$. Επειδή $e^x > 0$ πάντα:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow x - 2 = 0 \Leftrightarrow x = 2$ \quad (το $x=0$ δεν ανήκει στο πεδίο ορισμού)
    \item Πρόσημο $f'$: εξαρτάται από $\dfrac{x-2}{x^3}$.
    \begin{itemize}
        \item Για $x \in (-\infty, 0)$: $x-2 < 0$, $x^3 < 0$, άρα $f'(x) > 0$.
        \item Για $x \in (0, 2)$: $x-2 < 0$, $x^3 > 0$, άρα $f'(x) < 0$.
        \item Για $x \in (2, +\infty)$: $x-2 > 0$, $x^3 > 0$, άρα $f'(x) > 0$.
    \end{itemize}
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Συνεπώς:
\begin{itemize}
    \item Η $f$ είναι γνησίως αύξουσα στο $(-\infty, 0)$, γνησίως φθίνουσα στο $(0, 2]$ και γνησίως αύξουσα στο $[2, +\infty)$.
    \item Στο $x = 2$ παρουσιάζει \textbf{τοπικό ελάχιστο} με τιμή $f(2) = \dfrac{e^2}{4}$.
\end{itemize}

% -------------------------------------------------------
% γ) Εφαπτομένη στο Α(-1, f(-1))
% -------------------------------------------------------
\item Υπολογίζουμε: $f(-1) = \dfrac{e^{-1}}{(-1)^2} = \dfrac{1}{e}$.

Κλίση εφαπτομένης:
\[
f'(-1) = \frac{e^{-1}(-1-2)}{(-1)^3} = \frac{\frac{-3}{e}}{-1} = \frac{3}{e}.
\]
Εξίσωση εφαπτομένης:
\[
y - \frac{1}{e} = \frac{3}{e}(x+1) \Leftrightarrow y = \frac{3}{e}x + \frac{3}{e} + \frac{1}{e} \Leftrightarrow y = \frac{3}{e}x + \frac{4}{e}.
\]

% -------------------------------------------------------
% δ) Ύπαρξη και μοναδικότητα ρίζας
% -------------------------------------------------------
\item Θα αποδείξουμε ότι υπάρχει \textbf{μοναδικό} $x_0 \in (1,2)$ με $f(x_0) = 2$.

Από τον πίνακα μονοτονίας, η $f$ είναι γνησίως φθίνουσα στο $(0, 2]$, οπότε για κάθε $x \in (1, 2)$:
\[
f(2) < f(x) < f(1).
\]
Υπολογίζουμε:
\[
f(2) = \frac{e^2}{4} \approx \frac{7{,}39}{4} \approx 1{,}85, \qquad f(1) = \frac{e}{1} = e \approx 2{,}72.
\]
Εφόσον $f(2) \approx 1{,}85 < 2 < e \approx 2{,}72 = f(1)$, από το Θεώρημα Ενδιάμεσων Τιμών υπάρχει τουλάχιστον ένα $x_0 \in (1,2)$ με $f(x_0) = 2$.

Όμως η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(1,2) \subset (0,2)$, άρα είναι «1-1» σε αυτό, οπότε το $x_0$ είναι \textbf{μοναδικό}.

\end{erwthma}
\end{thema}
