\begin{thema}{Β}
Δίνεται η συνάρτηση $f:(0,+\infty)\to\mathbb{R}$ με τύπο $f(x)=\sqrt{x}-\ln x$.

\begin{erwthma}
\item Η συνάρτηση $f$ είναι ορισμένη στο $(0,+\infty)$ και παραγωγίσιμη σε αυτό ως άθροισμα παραγωγίσιμων συναρτήσεων με παράγωγο:
\[f'(x) = (\sqrt{x} - \ln x)' = \frac{1}{2\sqrt{x}} - \frac{1}{x} = \frac{\sqrt{x}}{2x} - \frac{2}{2x} = \frac{\sqrt{x}-2}{2x}\]
Για τις ρίζες και το πρόσημο της $f'$ έχουμε:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow \sqrt{x}-2 = 0 \Leftrightarrow \sqrt{x}=2 \Leftrightarrow x=4$
    \item $f'(x) > 0 \Leftrightarrow \dfrac{\sqrt{x}-2}{2x} > 0 \stackrel{x>0}{\Leftrightarrow} \sqrt{x}-2 > 0 \Leftrightarrow \sqrt{x}>2 \Leftrightarrow x>4$
    \item $f'(x) < 0 \Leftrightarrow \dots \Leftrightarrow 0<x<4$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Συνεπώς η $f$ είναι γνησίως φθίνουσα στο $(0, 4]$ και γνησίως αύξουσα στο $[4, +\infty)$. Η συνάρτηση $f$ παρουσιάζει ολικό ελάχιστο στη θέση $x_0 = 4$ με τιμή $f(4) = \sqrt{4} - \ln 4 = 2 - 2\ln 2$.
\item Για κάθε $x \in (0, +\infty)$ η $f'$ είναι παραγωγίσιμη με:
\[f''(x) = \left( \dfrac{1}{2} x^{-1/2} - x^{-1} \right)' = -\dfrac{1}{4} x^{-3/2} + x^{-2} = -\dfrac{1}{4x\sqrt{x}} + \dfrac{1}{x^2} = \dfrac{-x + 4\sqrt{x}}{4x^2\sqrt{x}} = \dfrac{4\sqrt{x}-x}{4x^2\sqrt{x}}\]
Βρίσκουμε τις ρίζες και το πρόσημο της $f''$:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow 4\sqrt{x}-x = 0 \Leftrightarrow 4\sqrt{x} = x \stackrel{x>0}{\Leftrightarrow} 16x = x^2 \Leftrightarrow x^2-16x=0 \Leftrightarrow x(x-16)=0 \stackrel{x>0}{\Rightarrow} x=16$
    \item $f''(x) > 0 \Leftrightarrow 4\sqrt{x}-x > 0 \Leftrightarrow x < 4\sqrt{x} \Leftrightarrow x^2 < 16x \Leftrightarrow x^2-16x < 0 \Leftrightarrow x(x-16) < 0 \stackrel{x>0}{\Rightarrow} 0<x<16$
    \item $f''(x) < 0 \dots \Leftrightarrow x>16$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:

Επομένως η $f$ είναι κυρτή στο διάστημα $(0, 16]$ και κοίλη στο διάστημα $[16, +\infty)$. Τα σημείο $M(16, f(16))$ δηλαδή το $M(16, 4-4\ln 2)$ είναι σημείο καμπής.
\item Έχουμε $f(1) = \sqrt{1} - \ln 1 = 1$ και $f'(1) = \dfrac{\sqrt{1}-2}{2\cdot 1} = -\dfrac{1}{2}$. Επομένως η εφαπτομένη της $C_f$ στο $x_0=1$ είναι η ευθεία με εξίσωση:
\[y - f(1) = f'(1)(x-1) \Leftrightarrow y - 1 = -\dfrac{1}{2}(x-1) \Leftrightarrow y = -\dfrac{1}{2}x + \dfrac{1}{2} + 1 \Leftrightarrow y = -\dfrac{1}{2}x + \dfrac{3}{2}\]
\item Αφού $\varepsilon: y = -\dfrac{1}{2}x + \dfrac{3}{2}$ είναι η εφαπτομένη της $C_f$ στο $x_0=1$ και η $f$ είναι κυρτή στο $(0, 16]$, συμπεραίνουμε ότι η γραφική παράσταση της $f$ βρίσκεται εξ ολοκλήρου πάνω (ή τέμνει) από την εφαπτομένη της στο διάστημα αυτό. Άρα για κάθε $x \in (0, 16]$, ισχύει:
\[f(x) \ge y \Leftrightarrow \sqrt{x} - \ln x \ge -\dfrac{1}{2}x + \dfrac{3}{2} \Leftrightarrow 2\sqrt{x} - 2\ln x \ge -x + 3 \Leftrightarrow 2\sqrt{x} + x - 3 \ge 2\ln x\]
\item Έχουμε $\displaystyle\lim_{x\to0^+} (xf(x)) = \lim_{x\to0^+} (x\sqrt{x} - x\ln x)$.
Αλλά $\displaystyle\lim_{x\to0^+} x\sqrt{x} = 0 \cdot 0 = 0$ και για το $\displaystyle\lim_{x\to0^+} (x\ln x)$ έχουμε απροσδιοριστία $0 \cdot (-\infty)$, οπότε:
\[\lim_{x\to0^+} (x\ln x) = \lim_{x\to0^+} \dfrac{\ln x}{\frac{1}{x}} \xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}} \lim_{x\to0^+} \dfrac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x\to0^+} (-x) = 0\]
Επομένως, $\displaystyle\lim_{x\to0^+} (xf(x)) = 0 - 0 = 0$.
\item Η $f$ είναι κυρτή στο $(0, 16]$ και η $\varepsilon$ είναι η εφαπτομένη της στο 1, άρα ισχύει $f(x) \ge -\dfrac{1}{2}x + \dfrac{3}{2}$ για κάθε $x \in (0, 16]$, δηλαδή η $C_f$ βρίσκεται πάντοτε πάνω από την εφαπτομένη, συνεπώς $f(x) - \left(-\dfrac{x}{2} + \dfrac{3}{2}\right) \ge 0$. Το εμβαδόν του χωρίου μεταξύ της $C_f$, της $\varepsilon$ από το σημείο επαφής μέχρι το $x=3$ είναι:
\[ E = \int_1^3 \left( f(x) - \left(-\dfrac{1}{2}x + \dfrac{3}{2}\right) \right) \d x = \int_1^3 \left( \sqrt{x} - \ln x + \dfrac{x}{2} - \dfrac{3}{2} \right) \d x \]
Έχουμε:
\begin{align*}
\int_1^3 \sqrt{x} \d x &= \int_1^3 x^{1/2} \d x = \left[ \dfrac{2}{3} x^{3/2} \right]_1^3 = \dfrac{2}{3} (3\sqrt{3} - 1) = 2\sqrt{3} - \dfrac{2}{3} \\
\int_1^3 \ln x \d x &= \int_1^3 (x)' \ln x \d x = [x \ln x]_1^3 - \int_1^3 x \dfrac{1}{x} \d x = 3\ln 3 - [x]_1^3 = 3\ln 3 - 2 \\
\int_1^3 \left( \dfrac{x}{2} - \dfrac{3}{2} \right) \d x &= \left[ \dfrac{x^2}{4} - \dfrac{3x}{2} \right]_1^3 = \left( \dfrac{9}{4} - \dfrac{9}{2} \right) - \left( \dfrac{1}{4} - \dfrac{3}{2} \right) = -\dfrac{9}{4} + \dfrac{5}{4} = -1
\end{align*}
Άρα:
\[ E = 2\sqrt{3} - \dfrac{2}{3} - (3\ln 3 - 2) - 1 = 2\sqrt{3} - \dfrac{2}{3} - 3\ln 3 + 2 - 1 = 2\sqrt{3} - 3\ln 3 + \dfrac{1}{3} \]
\end{erwthma}
\end{thema}
