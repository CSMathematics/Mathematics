\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = (x-2)\ln(x-2) + x - 2$.

\begin{erwthma}

% -------------------------------------------------------
% α) Πεδίο ορισμού και όριο στο 2+
% -------------------------------------------------------
\item \textbf{Πεδίο ορισμού.} Η $f$ ορίζεται όταν $x - 2 > 0$, δηλαδή $x > 2$.
Άρα $D_f = (2, +\infty)$.

\textbf{Υπολογισμός ορίου.}
\[
\lim_{x \to 2^+} f(x) = \lim_{x \to 2^+} \bigl[(x-2)\ln(x-2) + x - 2\bigr].
\]
Για τον όρο $(x-2)\ln(x-2)$ έχουμε απροσδιοριστία $0 \cdot (-\infty)$:
\[
\lim_{x \to 2^+} (x-2) \ln (x-2) = \lim_{x \to 2^+} \frac{\ln (x-2)}{\frac{1}{x-2}} \dlh{\frac{-\infty}{+\infty}} \lim_{x \to 2^+} \frac{\frac{1}{x-2}}{-\frac{1}{(x-2)^2}} = \lim_{x \to 2^+} (-x+2) = 0.
\]
Επομένως:
\[
\lim_{x \to 2^+} f(x) = 0 + 0 = 0.
\]

% -------------------------------------------------------
% β) Μονοτονία, τοπικά ακρότατα
% -------------------------------------------------------
\item Η $f$ είναι παραγωγίσιμη στο $(2, +\infty)$ ως άθροισμα παραγωγίσιμων συναρτήσεων:
\[
f'(x) = (x-2)' \ln(x-2) + (x-2) \cdot \frac{1}{x-2} + 1 = \ln(x-2) + 1 + 1 = \ln(x-2) + 2.
\]
Βρίσκουμε ρίζες και πρόσημο της $f'$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow \ln(x-2) = -2 \Leftrightarrow x - 2 = e^{-2} \Leftrightarrow x = 2 + e^{-2}$
    \item $f'(x) > 0 \Leftrightarrow \ln(x-2) > -2 \Leftrightarrow x - 2 > e^{-2} \Leftrightarrow x > 2 + e^{-2}$
    \item $f'(x) < 0 \Leftrightarrow 2 < x < 2 + e^{-2}$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:

Η $f$ είναι γνησίως φθίνουσα στο $\left(2,\,2+e^{-2}\right]$ και γνησίως αύξουσα στο $\left[2+e^{-2},+\infty\right)$.

Στο $x_0 = 2+e^{-2}$ παρουσιάζει \textbf{τοπικό (και ολικό) ελάχιστο}:
\[
f\!\left(2+e^{-2}\right) = e^{-2}\ln(e^{-2}) + e^{-2} = e^{-2}(-2) + e^{-2} = -2e^{-2} + e^{-2} = -e^{-2} = -\frac{1}{e^2}.
\]

% -------------------------------------------------------
% γ) Κυρτότητα
% -------------------------------------------------------
\item Παραγωγίζουμε την $f'(x) = \ln(x-2) + 2$:
\[
f''(x) =\left(\ln(x-2) + 2\right)'= \frac{1}{x-2}>0
\]
για κάθε $x \in (2, +\infty)$ άρα η $f$ είναι \textbf{κυρτή} σε ολόκληρο το πεδίο ορισμού της $(2, +\infty)$. Δεν υπάρχουν σημεία καμπής.

% -------------------------------------------------------
% δ) Εμβαδόν χωρίου
% -------------------------------------------------------
\item Εξετάζουμε το πρόσημο της $f$ στο $[3, e+2]$.

Αφού η $f$ είναι γνησίως αύξουσα στο $[3, e+2] \subset [2+e^{-2}, +\infty)$ και $f(3) = 1\cdot\ln1+1 = 1 > 0$, ισχύει $f(x) > 0$ για κάθε $x \in [3, e+2]$.

Άρα:
\[
E = \int_3^{e+2} f(x)\,\d x = \int_3^{e+2} \bigl[(x-2)\ln(x-2) + x - 2\bigr]\,\d x.
\]
Θέτουμε $t = x - 2$, οπότε $\d t = \d x$. Για $x=3$: $t=1$, για $x=e+2$: $t=e$.
\[
E = \int_1^{e} (t\ln t + t)\,\d t = \int_1^e t\ln t\,\d t + \int_1^e t\,\d t.
\]
Για το $\displaystyle\int_1^e t\ln t\,\d t$ εφαρμόζουμε ολοκλήρωση κατά παράγοντες:
\[
\int_1^e t\ln t\,\d t = \left[\frac{t^2}{2}\ln t\right]_1^e - \int_1^e \frac{t^2}{2}\cdot\frac{1}{t}\,\d t = \frac{e^2}{2} - 0 - \frac{1}{2}\int_1^e t\,\d t = \frac{e^2}{2} - \frac{1}{2}\left[\frac{t^2}{2}\right]_1^e = \frac{e^2}{2} - \frac{e^2-1}{4}.
\]
\[
\int_1^e t\,\d t = \left[\frac{t^2}{2}\right]_1^e = \frac{e^2-1}{2}.
\]
Άρα:
\[
E = \frac{e^2}{2} - \frac{e^2-1}{4} + \frac{e^2-1}{2}
= \frac{e^2}{2} + \frac{e^2-1}{4}
= \frac{2e^2 + e^2 - 1}{4}
= \frac{3e^2-1}{4}\ \text{τ.μ.}
\]

\end{erwthma}
\end{thema}
