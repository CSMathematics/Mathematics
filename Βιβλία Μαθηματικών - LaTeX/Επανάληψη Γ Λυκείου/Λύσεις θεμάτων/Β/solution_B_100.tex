\begin{thema}{Β}
Στο σχήμα δίνεται η γραφική παράσταση της παραγώγου $f'$. Επομένως, η μονοτονία της $f$ προκύπτει από το πρόσημο της $f'$, ενώ η κυρτότητα της $f$ προκύπτει από τη μονοτονία της $f'$.

\begin{center}
\begin{tikzpicture}
    \begin{axis}[
        axis lines=middle,
        axis line style={->, >=latex},
        grid=major,
        grid style={very thin, gray!30},
        xlabel={$x$},
        ylabel={$y=f'(x)$},
        xtick={1,2,3,4},
        ytick={1,2,3,4,5},
        every axis x label/.style={at={(current axis.right of origin)}, anchor=west},
        every axis y label/.style={at={(current axis.above origin)}, anchor=south},
        width=8cm,

        xmin=-0.5,
        xmax=4.5,
        ymin=-0.5,
        ymax=5.5,
        xtick distance=1,
        ytick distance=1
    ]
        \addplot[very thick, \xrwma, domain=0.2:4] {1/x};
        \node[below left] at (axis cs:0,0) {$0$};
    \end{axis}
\end{tikzpicture}
\end{center}

\begin{erwthma}
\item Από το σχήμα βλέπουμε ότι $f'(x)>0$ για κάθε $x\in(0,+\infty)$. Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.

Επίσης η γραφική παράσταση της $f'$ είναι γνησίως φθίνουσα στο $(0,+\infty)$. Επομένως, η $f$ είναι \textbf{κοίλη} στο $(0,+\infty)$.

\item Δίνεται ότι $f'(x)=\dfrac{1}{x}$ για κάθε $x>0$ και $f(1)=0$. Επειδή μια παράγουσα της $\dfrac{1}{x}$ στο $(0,+\infty)$ είναι η $\ln x$, έχουμε
\[
f(x)=\ln x+c,\quad x>0.
\]
Από τη συνθήκη $f(1)=0$ παίρνουμε
\[
0=f(1)=\ln 1+c=0+c \Rightarrow c=0.
\]
Άρα
\[
f(x)=\ln x,\quad x>0.
\]

\item Αφού $f(x)=\ln x$, αρκεί να αποδείξουμε ότι $\ln x\leq x-1$ για κάθε $x>0$. Θεωρούμε τη συνάρτηση
\[
\phi(x)=x-1-\ln x,\quad x>0.
\]
Τότε
\[
\phi'(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Επειδή $x>0$, το πρόσημο της $\phi'(x)$ καθορίζεται από τον αριθμητή $x-1$. Άρα $\phi'(x)<0$ για $0<x<1$ και $\phi'(x)>0$ για $x>1$.

Επομένως η $\phi$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$, οπότε παρουσιάζει ολικό ελάχιστο στο $x=1$. Είναι
\[
\phi(1)=1-1-\ln 1=0.
\]
Άρα $\phi(x)\geq 0$ για κάθε $x>0$, δηλαδή
\[
x-1-\ln x\geq 0 \Rightarrow \ln x\leq x-1.
\]
Συνεπώς
\[
f(x)\leq x-1,\quad x>0.
\]

\item Από το Β2 έχουμε $f(x)=\ln x$, άρα
\[
\lim_{x\to 0^+}x f(x)=\lim_{x\to 0^+}x\ln x.
\]
Το όριο είναι της μορφής $0\cdot(-\infty)$, οπότε το γράφουμε ως πηλίκο:
\[
\lim_{x\to 0^+}x\ln x=\lim_{x\to 0^+}\frac{\ln x}{1/x}.
\]
Έχουμε μορφή $\dfrac{-\infty}{+\infty}$ και, εφαρμόζοντας τον κανόνα \eng{De L'Hospital}, παίρνουμε
\[
\lim_{x\to 0^+}\frac{\ln x}{1/x}
\xlongequal{\text{\eng{DLH}}}
\lim_{x\to 0^+}\frac{1/x}{-1/x^2}
=\lim_{x\to 0^+}(-x)=0.
\]
Επομένως
\[
\lim_{x\to 0^+}x f(x)=0.
\]
\end{erwthma}
\end{thema}
