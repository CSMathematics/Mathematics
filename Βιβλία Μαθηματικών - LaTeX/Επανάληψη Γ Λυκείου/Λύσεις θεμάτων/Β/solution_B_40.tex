\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^3 - 3x + 1$ με $x \in \mathbb{R}$.
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη με:
\[ f'(x) = 3x^2 - 3 = 3(x^2 - 1) \]
Οι ρίζες της $f'$ είναι $x_1=-1$ και $x_2=1$.
Επίσης:
\begin{itemize}
\item $f'(x) > 0 \Leftrightarrow x \in (-\infty, -1) \cup (1, +\infty)$,
\item $f'(x) < 0 \Leftrightarrow x \in (-1, 1)$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%

\tkzTabLine{ , +, z, -, z, +, }
\tkzTabLine{ , \nearrow, t, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}
Οπότε η $f$ είναι γνησίως αύξουσα στα $(-\infty, -1]$ και $[1, +\infty)$, ενώ είναι γνησίως φθίνουσα στο $[-1, 1]$.

\item Η $f$ είναι συνεχής στο $[0,1]$ ως πολυωνυμική και:
\[ f(0)=1>0,\qquad f(1)=1-3+1=-1<0. \]
Άρα, από το Θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα $x_0\in(0,1)$ τέτοιο ώστε $f(x_0)=0$.
Επειδή το $(0,1)$ είναι υποσύνολο του $[-1,1]$ και η $f$ είναι γνησίως φθίνουσα στο $[-1,1]$, η ρίζα αυτή είναι μοναδική στο $(0,1)$.

\item Υπολογίζουμε την κλίση του τμήματος που ενώνει τα σημεία $A(0, f(0)) = (0, 1)$ και $B(2, f(2))$, όπου $f(2)=8-6+1=3$:
\[ \lambda_{AB} = \frac{f(2) - f(0)}{2 - 0} = \frac{3 - 1}{2} = 1. \]
Εφαρμόζουμε το Θεώρημα Μέσης Τιμής για την $f$ στο $[0,2]$. Τότε υπάρχει τουλάχιστον ένα $\xi \in (0,2)$ τέτοιο ώστε:
\[ f'(\xi)=\frac{f(2)-f(0)}{2-0}=1. \]
Άρα:
\[ 3\xi^2 - 3 = 1 \Leftrightarrow 3\xi^2 = 4 \Leftrightarrow \xi^2 = \frac{4}{3} \Leftrightarrow \xi = \pm \frac{2\sqrt{3}}{3}. \]
Επειδή ζητάμε $\xi \in (0,2)$, δεκτή είναι μόνο η τιμή
\[ \xi = \frac{2\sqrt{3}}{3}. \]

\item Επειδή $x_0\in(0,1)$, έχουμε $\syn x_0>0$. Επίσης $f(x_0)=0$ και η $f$ είναι γνησίως φθίνουσα στο $[-1,1]$, άρα:
\begin{itemize}
\item για $x\to x_0^-$ είναι $f(x)>0$, οπότε $\dfrac{\syn x}{f(x)}\to +\infty$,
\item για $x\to x_0^+$ είναι $f(x)<0$, οπότε $\dfrac{\syn x}{f(x)}\to -\infty$.
\end{itemize}
Τα πλευρικά όρια είναι διαφορετικά, επομένως το όριο
\[ \lim\limits_{x\to x_0}{\dfrac{\syn{x}}{f(x)}} \]
δεν υπάρχει.
\end{erwthma}
\end{thema}
