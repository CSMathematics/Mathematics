\begin{thema}{Β}
Έστω η συνάρτηση $f(x) = -x^2+4$ με $x \ge 0$. Μια τυχαία εφαπτομένη της γραφικής της παράστασης στο σημείο $M(x_0, f(x_0))$ με $x_0 \in (0, 2)$ τέμνει τους θετικούς ημιάξονες $Ox$ και $Oy$ στα σημεία $A$ και $B$ αντίστοιχα.

\begin{erwthma}

\item Η $f(x)$ είναι παραγωγίσιμη με $f'(x) = -2x$. 
Η εφαπτομένη $(\varepsilon)$ στο σημείο $M(x_0, f(x_0))$ έχει συντελεστή διεύθυνσης $\lambda = f'(x_0) = -2x_0$ και η εξίσωσή της είναι:
\[ y - f(x_0) = f'(x_0)(x - x_0) \Leftrightarrow y - (-x_0^2 + 4) = -2x_0 (x - x_0) \]
\[ y = -2x_0 x + 2x_0^2 - x_0^2 + 4 \Leftrightarrow (\varepsilon): \quad y = -2x_0 x + x_0^2 + 4 \]

Βρίσκουμε τα σημεία τομής της εφαπτομένης με τους άξονες:
\begin{itemize}
    \item \textbf{Τομή με τον άξονα $Oy$ ($x=0$):}  Για $x=0$, έχουμε $y = x_0^2 + 4$. 
    Άρα τέμνει τον θετικό ημιάξονα $Oy$ στο σημείο $B\left(0, x_0^2 + 4\right)$.
    \item \textbf{Τομή με τον άξονα $Ox$ ($y=0$):} Για $y=0$, λύνοντας την εξίσωση εντοπίζουμε:
    \[ 0 = -2x_0 x + x_0^2 + 4 \Leftrightarrow 2x_0 x = x_0^2 + 4 \Leftrightarrow x = \frac{x_0^2 + 4}{2x_0} \]
    Άρα τέμνει τον θετικό ημιάξονα $Ox$ στο σημείο $A\left(\frac{x_0^2 + 4}{2x_0}, 0\right)$.
\end{itemize}

\item Το τρίγωνο $OAB$ είναι ορθογώνιο στην αρχή των αξόνων $O(0,0)$. Τα μήκη των κάθετων πλευρών του ταυτίζονται με τις θετικές συντεταγμένες των σημείων $A$ και $B$:
\[ (OA) = x_A = \frac{x_0^2 + 4}{2x_0} \quad \text{και} \quad (OB) = y_B = x_0^2 + 4 \]
Το εμβαδόν του ορθογωνίου τριγώνου, συναρτήσει του $x_0$, είναι:
\[ E(x_0) = \frac{1}{2} (OA) (OB) = \frac{1}{2} \cdot \frac{x_0^2 + 4}{2x_0} \cdot (x_0^2 + 4) = \frac{(x_0^2+4)^2}{4x_0} \]
Το ζητούμενο αποδείχθηκε.

\item Θέλουμε να βρούμε το $x_0 \in (0, 2)$ που ελαχιστοποιεί τη συνάρτηση εμβαδού $E(t) = \frac{(t^2+4)^2}{4t}$ για $t \in (0, 2)$.
Παραγωγίζουμε χρησιμοποιώντας τον κανόνα του πηλίκου:
\[ E'(t) = \frac{ \left((t^2+4)^2\right)' (4t) - (t^2+4)^2 (4t)' }{(4t)^2} = \frac{ 2(t^2+4)(2t)(4t) - 4(t^2+4)^2 }{16t^2} \]
\[ E'(t) = \frac{ 16t^2(t^2+4) - 4(t^2+4)^2 }{16t^2} = \frac{4(t^2+4) \left[ 4t^2 - (t^2+4) \right]}{16t^2} = \frac{4(t^2+4)(3t^2 - 4)}{16t^2} = \frac{(t^2+4)(3t^2 - 4)}{4t^2} \]

Για να βρούμε τα ακρότατα, θέτουμε $E'(t) = 0$:
\[ E'(t) = 0 \Leftrightarrow 3t^2 - 4 = 0 \Leftrightarrow t^2 = \frac{4}{3} \stackrel{t>0}{\Leftrightarrow} t = \frac{2}{\sqrt{3}} = \frac{2\sqrt{3}}{3} \]
Εξετάζουμε εάν η ρίζα ανήκει στο $(0, 2)$: προφανώς $2/\sqrt{3} < 2$ αφού $\sqrt{3} > 1$. Οπότε το κρίσιμο σημείο είναι το $x_0 = \frac{2\sqrt{3}}{3}$.

Ελέγχουμε το πρόσημο της παραγώγου:
\begin{itemize}
    \item Αν $0 < t < \frac{2\sqrt{3}}{3}$, τότε $3t^2 < 4 \Leftrightarrow 3t^2 - 4 < 0$, άρα $E'(t) < 0$. Συνεπώς η $E(t)$ είναι γνησίως φθίνουσα στο $\left(0, \frac{2\sqrt{3}}{3}\right]$.
    \item Αν $\frac{2\sqrt{3}}{3} < t < 2$, τότε $3t^2 - 4 > 0$, άρα $E'(t) > 0$. Συνεπώς η $E(t)$ είναι γνησίως αύξουσα στο $\left[\frac{2\sqrt{3}}{3}, 2\right)$.
\end{itemize}
Συμπεραίνουμε πως η συνάρτηση παρουσιάζει ελάχιστο (το οποίο είναι και ολικό στο $(0, 2)$) για τη θέση $x_0 = \frac{2\sqrt{3}}{3}$. 

\item Για να βρούμε το αντίστοιχο ελάχιστο εμβαδόν, αντικαθιστούμε την τιμή $x_0 = \frac{2}{\sqrt{3}}$ στον τύπο της $E$:
\[ E_{\min} = E\left(\frac{2}{\sqrt{3}}\right) = \frac{\left( \left(\frac{2}{\sqrt{3}}\right)^2 + 4 \right)^2}{4 \cdot \frac{2}{\sqrt{3}}} = \frac{\left( \frac{4}{3} + 4 \right)^2}{\frac{8}{\sqrt{3}}} = \frac{\left( \frac{16}{3} \right)^2}{\frac{8}{\sqrt{3}}} \]
Εκτελούμε τις πράξεις:
\[ E_{\min} = \frac{\frac{256}{9}}{\frac{8}{\sqrt{3}}} = \frac{256\sqrt{3}}{72} = \frac{32\sqrt{3}}{9} \quad \text{τ.μ.} \]

\end{erwthma}
\end{thema}
