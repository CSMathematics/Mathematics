\begin{thema}{Β}
Στο παρακάτω σχήμα φαίνεται ενδεικτικά η κατασκευή για $x_0=1$.

    \end{axis}

\begin{erwthma}
\item Είναι $f(x)=-x^2+4$, άρα $f'(x)=-2x$. Στο $M(x_0,f(x_0))=\left(x_0,-x_0^2+4\right)$ η εφαπτομένη έχει εξίσωση
\[
y-(-x_0^2+4)=-2x_0(x-x_0)\Leftrightarrow \varepsilon:\ y=-2x_0x+x_0^2+4.
\]
Για $x=0$ βρίσκουμε $B(0,x_0^2+4)$, ενώ για $y=0$ βρίσκουμε
\[
-2x_0x+x_0^2+4=0\Leftrightarrow x=\frac{x_0^2+4}{2x_0},
\]
οπότε $A\left(\dfrac{x_0^2+4}{2x_0},0\right)$.

\item Το τρίγωνο $OAB$ είναι ορθογώνιο στο $O$ και
\[
OA=\frac{x_0^2+4}{2x_0},\quad OB=x_0^2+4.
\]
Άρα
\[
E(x_0)=\frac12\cdot OA\cdot OB
=\frac12\cdot\frac{x_0^2+4}{2x_0}\cdot(x_0^2+4)
=\frac{(x_0^2+4)^2}{4x_0}.
\]

\item Θέτουμε $x$ στη θέση του $x_0$ και μελετάμε
\[
E(x)=\frac{(x^2+4)^2}{4x}=\frac{x^3}{4}+2x+\frac4x,\quad x\in(0,2).
\]
Έχουμε
\[
E'(x)=\frac{3x^2}{4}+2-\frac{4}{x^2}
=\frac{3x^4+8x^2-16}{4x^2}
=\frac{(x^2+4)(3x^2-4)}{4x^2}.
\]
Η παραγοντοποίηση γίνεται επειδή $(x^2+4)(3x^2-4)=3x^4+8x^2-16$. Στο $(0,2)$ έχουμε $x^2+4>0$ και $4x^2>0$, άρα το πρόσημο του $E'$ είναι το πρόσημο του $3x^2-4$. Επομένως
\[
E'(x)=0\Leftrightarrow x=\frac{2}{\sqrt3}=\frac{2\sqrt3}{3}.
\]
Η $E$ είναι φθίνουσα στο $\left(0,\frac{2}{\sqrt3}\right]$ και αύξουσα στο $\left[\frac{2}{\sqrt3},2\right)$, οπότε το ελάχιστο προκύπτει για $x_0=\dfrac{2\sqrt3}{3}$.

\item
\[
E_{\min}=E\left(\frac{2}{\sqrt3}\right)
=\frac{\left(\frac43+4\right)^2}{4\cdot\frac{2}{\sqrt3}}
=\frac{\left(\frac{16}{3}\right)^2}{\frac{8}{\sqrt3}}
=\frac{32\sqrt3}{9}.
\]
\end{erwthma}
\end{thema}
