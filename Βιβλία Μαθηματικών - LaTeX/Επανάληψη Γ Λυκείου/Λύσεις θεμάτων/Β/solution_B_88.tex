\begin{thema}{Β}
\begin{erwthma}
\item Για να ορίζεται η $g(x)=\ln(f(x))$, πρέπει $f(x)>0$. Επειδή
\[
x^2-4x+3>0\Leftrightarrow (x-1)(x-3)>0,
\]
παίρνουμε
\[
D_g=(-\infty,1)\cup(3,+\infty).
\]

\item Στα άκρα του πεδίου ορισμού που αντιστοιχούν στις ρίζες της $f$ έχουμε
\[
\lim_{x\to1^-}f(x)=0^+\quad \text{και}\quad \lim_{x\to3^+}f(x)=0^+.
\]
Επομένως
\[
\lim_{x\to1^-}g(x)=-\infty\quad \text{και}\quad \lim_{x\to3^+}g(x)=-\infty.
\]
Γεωμετρικά, οι ευθείες $x=1$ και $x=3$ είναι κατακόρυφες ασύμπτωτες της $C_g$.

\item Είναι
\[
g'(x)=\frac{f'(x)}{f(x)}=\frac{2x-4}{x^2-4x+3},\quad x\in D_g.
\]
Στο $D_g$ είναι $x^2-4x+3>0$, άρα το πρόσημο της $g'$ καθορίζεται από το $2x-4$. Έτσι $g'(x)<0$ στο $(-\infty,1)$ και $g'(x)>0$ στο $(3,+\infty)$. Συνεπώς η $g$ είναι γνησίως φθίνουσα στο $(-\infty,1)$ και γνησίως αύξουσα στο $(3,+\infty)$.

\item Για $x\in D_g$ έχουμε
\begin{align*}
g(x)=0
&\Leftrightarrow \ln(x^2-4x+3)=0\\
&\Leftrightarrow x^2-4x+3=1\\
&\Leftrightarrow x^2-4x+2=0\\
&\Leftrightarrow x=2\pm\sqrt2.
\end{align*}
Επειδή $2-\sqrt2<1$ και $2+\sqrt2>3$, οι δύο λύσεις ανήκουν στο $D_g$. Άρα η εξίσωση $g(x)=0$ έχει ακριβώς δύο λύσεις.
\end{erwthma}
\end{thema}
