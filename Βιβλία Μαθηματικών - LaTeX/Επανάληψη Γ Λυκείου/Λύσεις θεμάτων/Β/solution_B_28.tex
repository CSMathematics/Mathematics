\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = (x-1)e^{-x} + 2$.

\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι $D_f = \mathbb{R}$. Είναι δύο φορές παραγωγίσιμη.
\textbf{Μονοτονία:}
\[ f'(x) = (x-1)'e^{-x} + (x-1)(e^{-x})' = e^{-x} - (x-1)e^{-x} = e^{-x}(1 - x + 1) = e^{-x}(2-x) \]
\begin{itemize}
\item $f'(x)=0\xLeftrightarrow{e^{-x}>0} 2-x=0\Leftrightarrow x=2$
\item $f'(x) > 0 \Leftrightarrow 2-x > 0 \Leftrightarrow x < 2$.
\item $f'(x) < 0 \Leftrightarrow 2-x < 0 \Leftrightarrow x > 2$.
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}

\end{center}
Οπότε, η $f$ είναι γνησίως αύξουσα στο $(-\infty, 2]$ και γνησίως φθίνουσα στο $[2, +\infty)$.

\textbf{Κυρτότητα:}
\[ f''(x) = (e^{-x})'(2-x) + e^{-x}(2-x)' = -e^{-x}(2-x) - e^{-x} = e^{-x}(-2+x-1) = e^{-x}(x-3) \]
\begin{itemize}
\item $f''(x)=0\xLeftrightarrow{e^{-x}>0}x-3=0\Leftrightarrow x=3$.
\item $f''(x) > 0 \Leftrightarrow x-3 > 0 \Leftrightarrow x > 3$.
\item $f''(x) < 0 \Leftrightarrow x < 3$.
\end{itemize}

Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f''$ και η κυρτότητα της $f$:
\begin{center}

\end{center}
Άρα η $f$ είναι κοίλη στο $(-\infty, 3]$ και κυρτή στο $[3, +\infty)$.

\item \textbf{Ακρότατα:} Από τη μελέτη μονοτονίας, η $f$ έχει ολικό μέγιστο για $x_0=2$, με τιμή:
\[ f(2) = (2-1)e^{-2} + 2 = \frac{1}{e^2} + 2 \]
Δεν παρουσιάζει τοπικό ελάχιστο.
\textbf{Σημεία Καμπής:} Από τη μελέτη κυρτότητας, η $C_f$ αλλάζει κυρτότητα στο $x_1=3$ όπου υπάρχει και εφαπτομένη, άρα παρουσιάζει σημείο καμπής στο σημείο $\Sigma(3, f(3))$ με:
\[ f(3) = (3-1)e^{-3} + 2 = \frac{2}{e^3} + 2 \]

\item Αναζητούμε το όριο στο $+\infty$:
\[ \lim\limits_{x\to+\infty} f(x) = \lim\limits_{x\to+\infty} \left( \frac{x-1}{e^x} + 2 \right) \]
Για το επιμέρους όριο $\lim\limits_{x\to+\infty} \frac{x-1}{e^x}$, επειδή έχουμε απροσδιοριστία $\frac{+\infty}{+\infty}$, εφαρμόζουμε:
\[ \lim\limits_{x\to+\infty} \frac{x-1}{e^x}\dlh{\frac{+\infty}{+\infty}} \lim\limits_{x\to+\infty} \frac{(x-1)'}{(e^x)'} = \lim\limits_{x\to+\infty} \frac{1}{e^x} = 0 \]
Κατά συνέπεια:
\[ \lim\limits_{x\to+\infty} f(x) = 0 + 2 = 2 \]
Αυτό αποδεικνύει ότι η ευθεία $y = 2$ είναι οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$.

\item Στο διάστημα $[1, 2]$ ισχύει $x-1 \ge 0$, και $e^{-x} > 0$, επομένως $f(x) \ge 2 > 0$.
Το εμβαδόν είναι:
\[ E = \int_1^2 f(x)\,\d x = \int_1^2 ((x-1)e^{-x} + 2)\,\d x = \int_1^2 (x-1)e^{-x}\,\d x + \int_1^2 2\,\d x \]
Υπολογίζουμε το πρώτο ολοκλήρωμα με παραγοντική:
\[ \int_1^2 (x-1)(-e^{-x})'\,\d x = \left[ -(x-1)e^{-x} \right]_1^2 - \int_1^2 (x-1)'(-e^{-x})\,\d x \]
\[ = \left( -(2-1)e^{-2} \right) - \left( -(1-1)e^{-1} \right) + \int_1^2 e^{-x}\,\d x = -e^{-2} - 0 + \left[ -e^{-x} \right]_1^2 \]
\[ = -e^{-2} - (e^{-2} - e^{-1}) = -2e^{-2} + e^{-1} \]
Το δεύτερο ολοκλήρωμα είναι:
\[ \int_1^2 2\,\d x = [2x]_1^2 = 4 - 2 = 2 \]
Προσθέτοντας, έχουμε το τελικό εμβαδόν:
\[ E = 2 + e^{-1} - 2e^{-2} = 2 + \frac{1}{e} - \frac{2}{e^2} \]
\end{erwthma}
\end{thema}
