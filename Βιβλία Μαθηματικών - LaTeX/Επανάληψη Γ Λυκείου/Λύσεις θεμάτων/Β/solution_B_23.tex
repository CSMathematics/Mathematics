\begin{thema}{Β}
Δίνονται οι συναρτήσεις $g(x) = e^x - x$ και $f(x) = \ln(g(x))$.

\begin{erwthma}

% -------------------------------------------------------
% α) Μονοτονία g, ολικό ελάχιστο, πεδίο ορισμού f
% -------------------------------------------------------
\item Η $g$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[
g'(x) =\left(e^x-x\right)'= e^x - 1.
\]
Βρίσκουμε ρίζες και πρόσημο της $g'$:
\begin{itemize}
    \item $g'(x)=0 \Leftrightarrow e^x=1 \Leftrightarrow x=0$
    \item $g'(x)>0 \Leftrightarrow e^x>1 \Leftrightarrow x>0$
    \item $g'(x)<0 \Leftrightarrow x<0$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $g'$ και η μονοτονία της $g$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = red7,colorL = red9,colorV = red7]%
{$x$ / .8,$g'(x)$ /0.8,$g(x)$/0.8}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{ , -, z, +, }
\tkzTabVar{ +/,-/\tiny{TE},+/, }
\end{tikzpicture}
\end{center}
Η $g$ παρουσιάζει \textbf{ολικό ελάχιστο} στο $x=0$ με τιμή $g(0)=e^0-0=1$.

\textbf{Πεδίο ορισμού της $f$.}
Η $f(x)=\ln(g(x))$ ορίζεται όταν $g(x)>0$.
Αφού το ολικό ελάχιστο της $g$ είναι $g(0)=1>0$, ισχύει $g(x)\geq1>0$ για κάθε $x\in\mathbb{R}$. Άρα $f$ ορίζεται για κάθε $x\in\mathbb{R}$, δηλαδή $D_f=\mathbb{R}$.

% -------------------------------------------------------
% β) Μονοτονία f και ακρότατα
% -------------------------------------------------------
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως σύνθεση παραγωγίσιμων συναρτήσεων:
\[
f'(x)=\left[\ln{\left(g(x)\right)}\right]'=\frac{g'(x)}{g(x)}=\frac{e^x-1}{e^x-x}.
\]
Επειδή $g(x)=e^x-x\geq1>0$ πάντα, το πρόσημο της $f'$ εξαρτάται αποκλειστικά από τον αριθμητή $e^x-1$:
\begin{itemize}
    \item $f'(x)=0 \Leftrightarrow e^x-1=0 \Leftrightarrow x=0$
    \item $f'(x)>0 \Leftrightarrow e^x>1 \Leftrightarrow x>0$
    \item $f'(x)<0 \Leftrightarrow x<0$
\end{itemize}
Στον παρακάτω πίνακα φαίνεται το πρόσημο της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = red7,colorL = red9,colorV = red7]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{ , -, z, +, }
\tkzTabVar{ +/,-/\tiny{TE},+/,}
\end{tikzpicture}
\end{center}
Η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$.
Στο $x=0$ παρουσιάζει \textbf{ολικό ελάχιστο} με τιμή $f(0)=\ln(g(0))=\ln1=0$.

% -------------------------------------------------------
% γ) Απόδειξη f(x) ≥ 0
% -------------------------------------------------------
\item Από το ερώτημα (β), το ολικό ελάχιστο της $f$ στο $\mathbb{R}$ είναι $f(0)=0$.
Άρα για κάθε $x\in\mathbb{R}$:
\[
f(x)\geq f(0)=0 \Leftrightarrow \ln(e^x-x)\geq0.\quad\square
\]

% -------------------------------------------------------
% δ) Εμβαδόν χωρίου g, x'x, x=0, x=1
% -------------------------------------------------------
\item Ελέγχουμε το πρόσημο της $g$ στο $[0,1]$.
Αφού $g(x)\geq1>0$ για κάθε $x\in\mathbb{R}$, ισχύει $g(x)>0$ σε ολόκληρο το $[0,1]$, οπότε η $C_g$ βρίσκεται πάνω από τον άξονα $x'x$. Άρα:
\[
E=\int_0^1 g(x)\,\d x = \int_0^1(e^x-x)\,\d x
=\Bigl[e^x-\frac{x^2}{2}\Bigr]_0^1
=\left(e-\frac{1}{2}\right)-(1-0)
=e-\frac{3}{2}\ \text{τ.μ.}
\]

\end{erwthma}
\end{thema}
