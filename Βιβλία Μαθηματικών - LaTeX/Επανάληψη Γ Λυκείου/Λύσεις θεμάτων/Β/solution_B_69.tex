\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \ln(x-a) + \beta x$. Η $f$ τέμνει τον άξονα $x'x$ στο σημείο με τετμημένη $2$ και η γραφική παράσταση έχει κατακόρυφη ασύμπτωτη την ευθεία $x=1$.

\begin{erwthma}

% -------------------------------------------------------
% α) Εύρεση a και β
% -------------------------------------------------------
\item Η κατακόρυφη ασύμπτωτη είναι $x=1$, δηλαδή το πεδίο ορισμού αρχίζει από το $x=1$.
Η $f$ ορίζεται για $x-a>0$, άρα $x>a$. Η αριστερή οριακή τιμή $a$ αντιστοιχεί στην κατακόρυφη ασύμπτωτη $x=a=1$, οπότε:
\[
a=1.
\]
Αφού η $C_f$ τέμνει τον $x'x$ στο $x=2$, ισχύει $f(2)=0$:
\[
\ln(2-1)+\beta\cdot2=\ln1+2\beta=0+2\beta=0 \Rightarrow \beta=0.
\]
Επομένως $a=1$ και $\beta=0$. \quad $\square$

% -------------------------------------------------------
% β) Πεδίο ορισμού και μονοτονία
% -------------------------------------------------------
\item Με $a=1$, $\beta=0$: $f(x)=\ln(x-1)$, $D_f=(1,+\infty)$.

Παραγωγίζουμε:
\[
f'(x)=\frac{1}{x-1}.
\]
Για $x>1$: $x-1>0$, άρα $f'(x)>0$ για κάθε $x\in(1,+\infty)$.

Η $f$ είναι \textbf{γνησίως αύξουσα} στο $(1,+\infty)$.

% -------------------------------------------------------
% γ) Αντίστροφη συνάρτηση
% -------------------------------------------------------
\item Αφού η $f$ είναι γνησίως αύξουσα, αντιστρέφεται.

\textbf{Σύνολο τιμών (= πεδίο ορισμού $f^{-1}$):}
\[
\lim_{x\to1^+}\ln(x-1)=-\infty,\qquad \lim_{x\to+\infty}\ln(x-1)=+\infty.
\]
Άρα $f((1,+\infty))=\mathbb{R}$, οπότε $D_{f^{-1}}=\mathbb{R}$.

\textbf{Τύπος $f^{-1}$:} Λύνουμε $y=\ln(x-1)$ ως προς $x$:
\[
e^y=x-1 \Leftrightarrow x=e^y+1.
\]
Άρα:
\[
f^{-1}(x)=e^x+1,\quad x\in\mathbb{R}.
\]

% -------------------------------------------------------
% δ) Εξίσωση f(x) = 1/(x-1) – ακριβώς μία ρίζα
% -------------------------------------------------------
\item Η εξίσωση $\ln(x-1)=\dfrac{1}{x-1}$ ισοδυναμεί με $\varphi(x)=0$, όπου:
\[
\varphi(x)=\ln(x-1)-\frac{1}{x-1},\quad x\in(1,+\infty).
\]
Η $\varphi$ είναι συνεχής στο $(1,+\infty)$.

\textbf{Ύπαρξη:} Ορίζουμε $t=x-1\in(0,+\infty)$ και εξετάζουμε $\psi(t)=\ln t-\dfrac{1}{t}$.
\[
\psi(1)=\ln1-1=0-1=-1<0,\qquad \psi(e)=1-\frac{1}{e}>0\quad\text{(αφού }1>\tfrac{1}{e}\text{)}.
\]
Αφού $\psi(1)\cdot\psi(e)<0$ και $\psi$ συνεχής στο $[1,e]$, από το \textbf{Θεώρημα Bolzano} υπάρχει $t_0\in(1,e)$ με $\psi(t_0)=0$, άρα $x_0=t_0+1\in(2,e+1)\subset(1,+\infty)$ είναι ρίζα.

\textbf{Μοναδικότητα:} Παραγωγίζουμε $\psi(t)=\ln t-\dfrac{1}{t}$:
\[
\psi'(t)=\frac{1}{t}+\frac{1}{t^2}=\frac{t+1}{t^2}>0\quad\text{για κάθε }t>0.
\]
Η $\psi$ είναι γνησίως αύξουσα, άρα «1-1», οπότε η ρίζα είναι μοναδική.

Επομένως η εξίσωση $f(x)=\dfrac{1}{x-1}$ έχει \textbf{ακριβώς μία ρίζα} στο $(1,+\infty)$. \quad $\square$

\end{erwthma}
\end{thema}
