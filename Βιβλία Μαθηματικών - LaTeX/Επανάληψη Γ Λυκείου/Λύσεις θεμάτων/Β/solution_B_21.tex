\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 e^{-x}$.

\begin{erwthma}
\item Η συνάρτηση $f$ έχει πεδίο ορισμού το $D_f = \mathbb{R}$.
Είναι παραγωγίσιμη για κάθε $x \in \mathbb{R}$ ως γινόμενο παραγωγίσιμων συναρτήσεων με:
\[ f'(x) = (x^2)' e^{-x} + x^2 (e^{-x})' = 2x e^{-x} - x^2 e^{-x} = x(2-x)e^{-x} \]
\begin{itemize}
    \item $f'(x)=0\Leftrightarrow x(2-x)e^{-x}=0\Leftrightarrow x=0$ ή $2-x=0\Leftrightarrow x=2$
    \item $f'(x)>0\Leftrightarrow x(2-x)e^{x}>0\Leftrightarrow x\in(0,2)$.
	\item $f'(x)<0\Leftrightarrow x(2-x)e^{x}<0\Leftrightarrow x\in(-\infty,0)\cup(2,+\infty)$.

\end{itemize}
Η $f$ είναι γνησίως αύξουσα στο $[0,2]$ και γνησίως φθίνουσα στα $(-\infty,0]$ και $2,+\infty$. Επίσης παρουσιάζει:
\begin{itemize}
    \item Τοπικό ελάχιστο στο $x=0$ με τιμή $f(0) = 0$.
    \item Τοπικό μέγιστο στο $x=2$ με τιμή $f(2) = 2^2 e^{-2} = \dfrac{4}{e^2}$.
\end{itemize}

\item Η $f'$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως πράξεις παραγωγίσιμων συναρτήσεων με:
\[ f''(x) = (2-2x)e^{-x} + (2x-x^2)e^{-x}(-1) = (2-2x-2x+x^2)e^{-x} = (x^2-4x+2)e^{-x} \]
Οι ρίζες της $f''$ είναι οι ρίζες του τριωνύμου $x^2-4x+2$:
\[ \Delta = (-4)^2 - 4 \cdot 1 \cdot 2 = 16 - 8 = 8 \]
\[ x_{1,2} = \frac{4 \pm \sqrt{8}}{2} = \frac{4 \pm 2\sqrt{2}}{2} = 2 \pm \sqrt{2} \]
Έστω $x_1 = 2-\sqrt{2}$ και $x_2 = 2+\sqrt{2}$. Το πρόσημο της $f''$ και η κυρτότητα της $f$:

\begin{itemize}
    \item H $f$ είναι κυρτή στα διαστήματα $(-\infty, x_1]$ και $[x_2, +\infty)$.
    \item και κοίλη στο διάστημα $[x_1, x_2]$.
\end{itemize}
Τα σημεία καμπής της $C_f$ είναι τα $A(x_1, f(x_1))$ και $B(x_2, f(x_2))$.

\item Κατακόρυφες ασύμπτωτες: Επειδή η $f$ είναι συνεχής στο $\mathbb{R}$, η $C_f$ δεν έχει κατακόρυφες ασύμπτωτες.
Οριζόντιες - Πλάγιες ασύμπτωτες:
\begin{itemize}
    \item Στο $+\infty$:
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{x^2}{e^x} \stackrel{\frac{+\infty}{+\infty}}{=} \lim_{x \to +\infty} \frac{2x}{e^x} \stackrel{\frac{+\infty}{+\infty}}{=} \lim_{x \to +\infty} \frac{2}{e^x} = 0 \]
    Άρα η ευθεία $y=0$ ($x'x$ άξονας) είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.
    \item Στο $-\infty$:
    \[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} (x^2 e^{-x}) = (+\infty) \cdot (+\infty) = +\infty \]
    Εξετάζουμε αν υπάρχει πλάγια ασύμπτωτη:
    \[ \lim_{x \to -\infty} \frac{f(x)}{x} = \lim_{x \to -\infty} (x e^{-x}) = (-\infty) \cdot (+\infty) = -\infty \]
    Οπότε η $C_f$ δεν έχει ασύμπτωτη στο $-\infty$.
\end{itemize}

\item Επειδή $f(x) = x^2 e^{-x} \geq 0$ για κάθε $x \in [0, 1]$, το ζητούμενο εμβαδόν είναι:
\[ E = \int_0^1 f(x) dx = \int_0^1 x^2 e^{-x} dx \]
Με παραγοντική ολοκλήρωση έχουμε:
\begin{align*}
E &= \int_0^1 x^2 (-e^{-x})' dx = [ -x^2 e^{-x} ]_0^1 + \int_0^1 2x e^{-x} dx \\
&= -e^{-1} + 2 \int_0^1 x (-e^{-x})' dx = -e^{-1} + 2 \left( [ -x e^{-x} ]_0^1 + \int_0^1 e^{-x} dx \right) \\
&= -e^{-1} + 2 \left( -e^{-1} + [ -e^{-x} ]_0^1 \right) = -e^{-1} + 2(-e^{-1} - e^{-1} + 1) \\
&= -5e^{-1} + 2 = 2 - \frac{5}{e} \text{ τ.μ.}
\end{align*}
\end{erwthma}
\end{thema}
