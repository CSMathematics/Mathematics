\begin{thema}{Β}
\begin{erwthma}
\item Από το σχήμα η $f$ έχει δύο κλάδους, έναν για $x<2$ και έναν για $x>2$, ενώ στο $x=2$ υπάρχει κατακόρυφη ασύμπτωτη. Άρα
\[
D_f=(-\infty,2)\cup(2,+\infty).
\]
Επίσης διαβάζουμε:
\[
\lim_{x\to2^-}f(x)=+\infty,\quad
\lim_{x\to2^+}f(x)=+\infty,\quad
\lim_{x\to-\infty}f(x)=0,\quad
\lim_{x\to+\infty}f(x)=1.
\]

\item Από τα δύο πλευρικά όρια στο $2$ έχουμε $\lim\limits_{x\to2}f(x)=+\infty$, άρα $\lim\limits_{x\to2}\dfrac1{f(x)}=0$. Επειδή η εκθετική συνάρτηση είναι συνεχής,
\[
\lim_{x\to2}e^{\frac1{f(x)}}=e^0=1.
\]

\item Από το σχήμα $f(1)=1$. Επειδή η $f$ είναι συνεχής στο πεδίο ορισμού της και $1\in D_f$, έχουμε $\lim\limits_{x\to1}f(x)=1$. Κοντά στο $x=1$ οι τιμές της $f$ είναι θετικές, άρα ο λογάριθμος ορίζεται και
\[
\lim_{x\to1}\ln(f(x))=\ln\left(\lim_{x\to1}f(x)\right)=\ln1=0.
\]

\item Από το σχήμα η $f$ είναι γνησίως αύξουσα στο $(-\infty,2)$ και γνησίως φθίνουσα στο $(2,+\infty)$. Η αλλαγή γίνεται στο $x=2$, όπου η συνάρτηση δεν ορίζεται, άρα η $f$ δεν παρουσιάζει τοπικό μέγιστο ή τοπικό ελάχιστο.
\end{erwthma}
\end{thema}
