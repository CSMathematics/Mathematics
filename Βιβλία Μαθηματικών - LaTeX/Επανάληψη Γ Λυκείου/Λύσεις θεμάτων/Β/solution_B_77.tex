%=========== ΛΥΣΗ - 77ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Το σημείο $M$ κινείται στη γραφική παράσταση της
\[
g(x)=\sqrt{x},\quad x\geq 0.
\]
Άρα
\[
M(x,\sqrt{x}),\quad x\geq 0.
\]

Το σταθερό σημείο είναι
\[
A(2,0).
\]
Επομένως, η απόσταση $AM$ είναι
\[
d(x)=\sqrt{(x-2)^2+(\sqrt{x}-0)^2}.
\]
Άρα
\[
d(x)=\sqrt{x^2-4x+4+x}
\]
και τελικά
\[
d(x)=\sqrt{x^2-3x+4},\quad x\geq 0.
\]

\item Για $x\geq 0$ έχουμε
\[
d(x)=\sqrt{x^2-3x+4}.
\]
Το τριώνυμο
\[
x^2-3x+4
\]
έχει διακρίνουσα
\[
\Delta=9-16=-7<0
\]
και είναι πάντα θετικό. Άρα η $d$ είναι παραγωγίσιμη στο $[0,+\infty)$ και
\[
d'(x)=\frac{2x-3}{2\sqrt{x^2-3x+4}}.
\]

Ο παρονομαστής είναι θετικός, επομένως το πρόσημο της $d'(x)$ καθορίζεται από το $2x-3$.

Έχουμε
\[
d'(x)=0 \Leftrightarrow 2x-3=0 \Leftrightarrow x=\frac{3}{2}.
\]

Άρα η $d$ είναι γνησίως φθίνουσα στο
\[
\left[0,\frac{3}{2}\right]
\]
και γνησίως αύξουσα στο
\[
\left[\frac{3}{2},+\infty\right).
\]

Επομένως, η απόσταση γίνεται ελάχιστη για
\[
x=\frac{3}{2}.
\]
Το αντίστοιχο σημείο της καμπύλης είναι
\[
M_0\left(\frac{3}{2},\sqrt{\frac{3}{2}}\right)
\]
δηλαδή
\[
M_0\left(\frac{3}{2},\frac{\sqrt{6}}{2}\right).
\]

\item Αναζητούμε πλάγια ασύμπτωτη της $C_d$ στο $+\infty$ της μορφής
\[
y=\lambda x+\beta.
\]

Έχουμε
\[
\lambda=\lim_{x\to+\infty}\frac{d(x)}{x}
=\lim_{x\to+\infty}\frac{\sqrt{x^2-3x+4}}{x}.
\]
Επειδή $x\to+\infty$, είναι $x>0$, οπότε
\[
\lambda=\lim_{x\to+\infty}\sqrt{1-\frac{3}{x}+\frac{4}{x^2}}=1.
\]

Επίσης,
\[
\beta=\lim_{x\to+\infty}\left(d(x)-x\right)
=\lim_{x\to+\infty}\left(\sqrt{x^2-3x+4}-x\right).
\]
Πολλαπλασιάζουμε με τη συζυγή παράσταση:
\[
\beta=
\lim_{x\to+\infty}
\frac{x^2-3x+4-x^2}{\sqrt{x^2-3x+4}+x}.
\]
Άρα
\[
\beta=
\lim_{x\to+\infty}
\frac{-3x+4}{\sqrt{x^2-3x+4}+x}.
\]
Διαιρώντας αριθμητή και παρονομαστή με $x>0$, παίρνουμε
\[
\beta=
\lim_{x\to+\infty}
\frac{-3+\frac{4}{x}}{\sqrt{1-\frac{3}{x}+\frac{4}{x^2}}+1}
=-\frac{3}{2}.
\]

Επομένως, η πλάγια ασύμπτωτη της $C_d$ στο $+\infty$ είναι
\[
y=x-\frac{3}{2}.
\]

\item Η κατακόρυφη ευθεία που διέρχεται από το $M_0$ έχει εξίσωση
\[
x=\frac{3}{2}.
\]

Το ζητούμενο εμβαδόν περικλείεται από τη γραφική παράσταση της
\[
g(x)=\sqrt{x},
\]
τον άξονα $x'x$ και την ευθεία
\[
x=\frac{3}{2}.
\]
Επειδή $\sqrt{x}\geq 0$ για $x\geq 0$, το εμβαδόν είναι
\[
E=\int_0^{3/2}\sqrt{x}\,\d x.
\]

Άρα
\[
E=\int_0^{3/2}x^{1/2}\,\d x
=\left[\frac{2}{3}x^{3/2}\right]_0^{3/2}.
\]
Επομένως,
\[
E=\frac{2}{3}\left(\frac{3}{2}\right)^{3/2}
=\frac{2}{3}\cdot \frac{3}{2}\sqrt{\frac{3}{2}}
=\sqrt{\frac{3}{2}}.
\]
Τελικά,
\[
E=\frac{\sqrt{6}}{2}\ \tau.\mu.
\]
\end{erwthma}
\end{thema}
