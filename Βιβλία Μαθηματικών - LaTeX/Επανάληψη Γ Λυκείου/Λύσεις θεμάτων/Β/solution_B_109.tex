\begin{thema}{Β}
Δίνεται ο πίνακας προσήμων (μεταβολής) της $f'(x)$ στο $[-3, 5]$:

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $-3$ & $\cdots$ & $0$ & $\cdots$ & $5$ \\
\hline
$f'(x)$ & & $+$ & $0$ & $-$ & \\
\hline
$f(x)$ & $-2$ & $\nearrow$ & $4$ & $\searrow$ & $1$ \\
\hline
\end{tabular}
\end{center}

\begin{erwthma}

\item Από τον πίνακα διαβάζουμε κατευθείαν:
\begin{itemize}
    \item Στο $[-3, 0]$: $f'(x) > 0$, άρα η $f$ \textbf{γνησίως αύξουσα}. Οι τιμές ξεκινούν από $f(-3) = -2$ και φτάνουν στο $f(0) = 4$.
    \item Στο $[0, 5]$: $f'(x) < 0$, άρα η $f$ \textbf{γνησίως φθίνουσα}. Οι τιμές ξεκινούν από $f(0) = 4$ και πέφτουν στο $f(5) = 1$.
\end{itemize}
Ακρότατα: Στο $x = 0$, η $f'$ αλλάζει πρόσημο από $(+)$ σε $(-)$, οπότε η $f$ παρουσιάζει \textbf{τοπικό μέγιστο} $f(0) = 4$.

\item Συγκρίνουμε τις τιμές στα ακραία σημεία και στα σημεία ακροτάτων:
\begin{itemize}
    \item Τιμές: $f(-3) = -2$, $f(0) = 4$, $f(5) = 1$.
    \item \textbf{Ολικό μέγιστο:} $f(0) = 4$ (η μεγαλύτερη τιμή).
    \item \textbf{Ολικό ελάχιστο:} $f(-3) = -2$ (η μικρότερη τιμή).
\end{itemize}

\item Ζητάμε τον αριθμό λύσεων της εξίσωσης $f(x) = 2$ στο $[-3, 5]$.
Εφαρμόζουμε το \textbf{Θεώρημα Bolzano} (ΘΕΤ) στη συνεχή $g(x) = f(x) - 2$:
\begin{itemize}
    \item \textbf{Στο $[-3, 0]$:} $f(-3) = -2 < 2$ και $f(0) = 4 > 2$. Αφού $f$ συνεχής, από Bolzano υπάρχει τουλάχιστον ένα $x_1 \in (-3, 0)$ με $f(x_1) = 2$. Η $f$ επιπλέον γνησίως αύξουσα στο $[-3, 0]$, οπότε η ρίζα είναι \textbf{μοναδική}.
    \item \textbf{Στο $[0, 5]$:} $f(0) = 4 > 2$ και $f(5) = 1 < 2$. Ομοίως, υπάρχει $x_2 \in (0, 5)$ με $f(x_2) = 2$. Η γνησίως φθίνουσα μονοτονία εγγυάται πάλι \textbf{μοναδικότητα}.
\end{itemize}
Συνολικά, η εξίσωση $f(x) = 2$ έχει \textbf{ακριβώς 2 λύσεις} στο $[-3, 5]$.

\item Από το Θεμελιώδες Θεώρημα Λογισμού:
\[ \int_{-3}^{5} f'(x)\,\mathrm{d}x = f(5) - f(-3) = 1 - (-2) = 3 \]
Η διαφορά $f(5) - f(-3) = 3 > 0$, δηλαδή \textbf{θετική}. Αυτό σημαίνει ότι η "καθαρή μεταβολή" της $f$ στο $[-3, 5]$ είναι ανοδική: η τελική τιμή υπερβαίνει την αρχική.

\end{erwthma}
\end{thema}
