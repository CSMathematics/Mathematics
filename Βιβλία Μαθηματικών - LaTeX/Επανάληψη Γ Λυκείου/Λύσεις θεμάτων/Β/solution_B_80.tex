%=========== ΛΥΣΗ - 80ο ΘΕΜΑ Β ===========

\begin{thema}{Β}
\begin{erwthma}
\item Δίνεται ότι για κάθε $x\in\mathbb{R}$ ισχύει
\[
f(x-2)=(x-2)e^{3-x}.
\]

Θέτουμε
\[
u=x-2.
\]
Τότε
\[
x=u+2.
\]
Άρα
\[
f(u)=u e^{3-(u+2)}
=u e^{1-u}.
\]

Επειδή το $u$ διατρέχει όλο το $\mathbb{R}$, προκύπτει ότι
\[
f(x)=x e^{1-x},\quad x\in\mathbb{R}.
\]

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=\left(xe^{1-x}\right)'
=e^{1-x}+x\left(e^{1-x}\right)'.
\]
Επειδή
\[
\left(e^{1-x}\right)'=-e^{1-x},
\]
έχουμε
\[
f'(x)=e^{1-x}-xe^{1-x}
=e^{1-x}(1-x).
\]

Επειδή
\[
e^{1-x}>0,\quad x\in\mathbb{R},
\]
το πρόσημο της $f'$ καθορίζεται από το $1-x$.

Άρα
\[
f'(x)>0 \Leftrightarrow x<1
\]
και
\[
f'(x)<0 \Leftrightarrow x>1.
\]

Επομένως, η $f$ είναι γνησίως αύξουσα στο
\[
(-\infty,1]
\]
και γνησίως φθίνουσα στο
\[
[1,+\infty).
\]

Συνεπώς, η $f$ παρουσιάζει ολικό μέγιστο στο $x=1$, με τιμή
\[
f(1)=1\cdot e^{1-1}=1.
\]

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\[
f''(x)=\left(e^{1-x}(1-x)\right)'.
\]
Άρα
\[
f''(x)=\left(e^{1-x}\right)'(1-x)+e^{1-x}(1-x)'.
\]
Επομένως,
\[
f''(x)=-e^{1-x}(1-x)-e^{1-x}
=e^{1-x}(x-2).
\]

Επειδή
\[
e^{1-x}>0,
\]
το πρόσημο της $f''$ καθορίζεται από το $x-2$.

Έχουμε
\[
f''(x)<0 \Leftrightarrow x<2
\]
και
\[
f''(x)>0 \Leftrightarrow x>2.
\]

Άρα η $f$ είναι κοίλη στο
\[
(-\infty,2]
\]
και κυρτή στο
\[
[2,+\infty).
\]

Στο $x=2$ η $f''$ μηδενίζεται και αλλάζει πρόσημο, οπότε η $C_f$ παρουσιάζει μοναδικό σημείο καμπής στο
\[
\bigl(2,f(2)\bigr).
\]

Επειδή
\[
f(2)=2e^{1-2}=\frac{2}{e},
\]
το σημείο καμπής είναι
\[
\left(2,\frac{2}{e}\right).
\]

\item Από το δεύτερο ερώτημα γνωρίζουμε ότι η $f$ παρουσιάζει ολικό μέγιστο στο $x=1$ με μέγιστη τιμή
\[
f(1)=1.
\]
Άρα, για κάθε $x\in\mathbb{R}$ ισχύει
\[
f(x)\leq 1.
\]

Επειδή
\[
2>1,
\]
δεν υπάρχει $x\in\mathbb{R}$ τέτοιο ώστε
\[
f(x)=2.
\]

Επομένως, η εξίσωση
\[
f(x)=2
\]
είναι αδύνατη στο $\mathbb{R}$.
\end{erwthma}
\end{thema}
