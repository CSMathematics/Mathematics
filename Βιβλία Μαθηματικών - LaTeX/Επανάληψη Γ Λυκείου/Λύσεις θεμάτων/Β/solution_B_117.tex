\begin{thema}{Β}
Στο παρακάτω σχήμα συμπληρώθηκε ολόκληρη η $C_f$ (αριστερός κλάδος) εκμεταλλευόμενοι την αξονική συμμετρία ως προς $y$.
\begin{center}
\begin{tikzpicture}[scale=0.7]
  \draw[help lines, gray!30] (-4,-1) grid (4,5);
  \draw[-latex] (-4.3,0) -- (4.3,0) node[right] {$x$};
  \draw[-latex] (0,-1.3) -- (0,5.3) node[above] {$y$};
  \foreach \x in {-3,-2,-1,1,2,3}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {1,2,3,4}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};
  % Δεξιός κλάδος
  \draw[very thick, blue!70!black] (0,4) -- (1,2) -- (2,0) -- (3,1);
  % Αριστερός κλάδος (συμμετρία)
  \draw[very thick, blue!70!black, dashed] (0,4) -- (-1,2) -- (-2,0) -- (-3,1);
  
  \filldraw[blue!70!black] (0,4) circle (2pt) node[above right, font=\footnotesize] {$(0,4)$};
  \filldraw[blue!70!black] (2,0) circle (2pt) node[below right, font=\footnotesize] {$(2,0)$};
  \filldraw[blue!70!black] (-2,0) circle (2pt) node[below left, font=\footnotesize] {$(-2,0)$};
  \filldraw[blue!70!black] (3,1) circle (2pt) node[above right, font=\footnotesize] {$(3,1)$};
  \filldraw[blue!70!black] (-3,1) circle (2pt) node[above left, font=\footnotesize] {$(-3,1)$};
  
  % Σχόλιο
  \node[anchor=south, color=orange, font=\footnotesize\bfseries] at (0, 4.3) {Άξονας Συμμετρίας: $y$};
  \draw[<->, >=stealth, orange, thick] (-1.5, 1.5) -- (1.5, 1.5);
\end{tikzpicture}
\end{center}

\begin{erwthma}

\item Η $f$ άρτια σημαίνει $f(-x) = f(x)$: η $C_f$ είναι συμμετρική ως προς τον \textbf{άξονα $y$}.
Αντιγράφουμε τα σημεία αντικατοπτρίζοντας:
\begin{itemize}
    \item $(1, 2) \to (-1, 2)$
    \item $(2, 0) \to (-2, 0)$
    \item $(3, 1) \to (-3, 1)$
\end{itemize}
Ο αριστερός κλάδος σχεδιάστηκε (διακεκομμένα) στο σχήμα.

\item Οι ρίζες ($f(x) = 0$) είναι στα σημεία όπου η $C_f$ τέμνει τον $x'x$:
\begin{itemize}
    \item Στο δεξιό τμήμα: $f(2) = 0$, άρα $x = 2$.
    \item Λόγω αρτιότητας: $f(-2) = f(2) = 0$, άρα $x = -2$.
\end{itemize}
Ρίζες: $x = -2$ και $x = 2$.

\item Εκμεταλλευόμαστε ότι η $f$ άρτια:
\[ \int_{-3}^{3} f(x)\,\mathrm{d}x = 2\int_0^3 f(x)\,\mathrm{d}x \]
Υπολογίζουμε γεωμετρικά $\int_0^3 f(x)\,\mathrm{d}x$ σπάζοντας σε τμήματα:
\begin{itemize}
    \item $[0, 1]$: $f(x) = -2x + 4$. $\int_0^1 (-2x+4)\,\mathrm{d}x = [-x^2 + 4x]_0^1 = -1+4 = 3$. Ή γεωμετρικά: τραπέζιο βάσεις $4, 2$, ύψος $1$: $\frac{(4+2)}{2}\cdot 1 = 3$.
    \item $[1, 2]$: $f(x) = -2x + 4$. $\int_1^2 (-2x+4)\,\mathrm{d}x = [-x^2+4x]_1^2 = (-4+8) - (-1+4) = 4-3 = 1$. Ή τραπέζιο $2, 0$, ύψος $1$: $\frac{(2+0)}{2}\cdot 1 = 1$.
    \item $[2, 3]$: $f(x) = x - 2$ (κλίση $1$, $f(2)=0$). $\int_2^3 (x-2)\,\mathrm{d}x = \frac{(x-2)^2}{2}\Big|_2^3 = \frac{1}{2}$. Ή τρίγωνο $1\times 1$: $\frac{1}{2}$.
\end{itemize}
\[ \int_0^3 f(x)\,\mathrm{d}x = 3 + 1 + \frac{1}{2} = \frac{9}{2} \]
\[ \int_{-3}^{3} f(x)\,\mathrm{d}x = 2 \cdot \frac{9}{2} = \mathbf{9} \]

\item 
\begin{itemize}
    \item Στο $[-3, 3]$: Η $f$ \textbf{δεν αντιστρέφεται}, γιατί δεν είναι 1-1. Πράγματι, $f(-2) = f(2) = 0$ (δύο διαφορετικά $x$ δίνουν ίδια τιμή). Αυτό ήταν αναμενόμενο: μια άρτια συνάρτηση ορισμένη σε συμμετρικό διάστημα δεν είναι ποτέ 1-1 (εκτός αν είναι σταθερή).
    \item Στο $[0, 2]$: Η $f$ είναι \textbf{γνησίως φθίνουσα} (πέφτει μονότονα από $4$ στο $0$), οπότε 1-1, και επομένως \textbf{αντιστρέφεται} στο $[0, 2]$.
\end{itemize}

\end{erwthma}
\end{thema}
