\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x>0$ οι αριθμοί $x^2-1$ και $\ln x$ είναι ομόσημοι ή μηδενίζονται ταυτόχρονα. Επομένως
\[
(x^2-1)\ln x\geq0.
\]
Επειδή $x^2+1>0$, προκύπτει
\[
{(x^2-1)\ln x+x^2+1>0}
\]
για κάθε $x>0$.

Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\begin{align*}
f'(x)
&=\left(x+\frac{1}{x}\right)'\ln x
+\left(x+\frac{1}{x}\right)(\ln x)'\\
&=\left(1-\frac{1}{x^2}\right)\ln x
+1+\frac{1}{x^2}\\
&=\frac{(x^2-1)\ln x+x^2+1}{x^2}.
\end{align*}
Από την προηγούμενη ανισότητα και επειδή $x^2>0$, έχουμε
\[
f'(x)>0
\]
για κάθε $x>0$. Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.

Επίσης
\[
\lim_{x\to0^+}f(x)
=\lim_{x\to0^+}\left(x\ln x+\frac{\ln x}{x}\right)
=-\infty
\]
και
\[
\lim_{x\to+\infty}f(x)=+\infty.
\]
Επομένως το σύνολο τιμών της $f$ είναι
\[
{f((0,+\infty))=\mathbb{R}}.
\]

\item Για κάθε $x>0$ είναι
\[
g(x)=\frac{f(x)}{x}
=\left(1+\frac{1}{x^2}\right)\ln x.
\]
Η $g$ είναι παραγωγίσιμη και
\begin{align*}
g'(x)
&=-\frac{2\ln x}{x^3}
+\left(1+\frac{1}{x^2}\right)\frac{1}{x}\\
&=\frac{x^2+1-2\ln x}{x^3}.
\end{align*}
Θεωρούμε τη βοηθητική συνάρτηση
\[
h(x)=x^2+1-2\ln x,\quad x>0.
\]
Έχουμε
\[
h'(x)=2x-\frac{2}{x}
=\frac{2(x^2-1)}{x}.
\]
Άρα η $h$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
h(1)=2.
\]
Συνεπώς
\[
h(x)>0
\]
για κάθε $x>0$. Επειδή $x^3>0$, προκύπτει
\[
g'(x)>0
\]
για κάθε $x>0$. Άρα η $g$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.

Επιπλέον,
\[
\lim_{x\to0^+}g(x)=-\infty
\quad\text{και}\quad
\lim_{x\to+\infty}g(x)=+\infty.
\]
Επομένως, για κάθε $\lambda>0$, η εξίσωση
\[
f(x)=\lambda x
\]
είναι ισοδύναμη με
\[
g(x)=\lambda
\]
και έχει ακριβώς μία ρίζα στο $(0,+\infty)$.

\item Επειδή
\[
g(1)=0
\]
και η $g$ είναι γνησίως αύξουσα, έχουμε $g(x)>0$ για κάθε $x\in(1,e]$. Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_1^e g(x)\d x\\
&=\int_1^e\left(1+\frac{1}{x^2}\right)\ln x\,\d x\\
&=\left[x\ln x-x-\frac{\ln x+1}{x}\right]_1^e\\
&=-\frac{2}{e}-(-2)\\
&={2-\frac{2}{e}}.
\end{align*}

\item Από το προηγούμενο ερώτημα έχουμε
\[
\frac{E}{e-1}
=\frac{2-\frac{2}{e}}{e-1}
=\frac{2}{e}.
\]
Το σημείο $M(x_0,f(x_0))$ ανήκει στην ευθεία που διέρχεται από το $O(0,0)$ και έχει κλίση $\frac{2}{e}$ αν και μόνο αν
\begin{align*}
f(x_0)&=\frac{2}{e}x_0\\
\Leftrightarrow g(x_0)&=\frac{2}{e}.
\end{align*}

Η $g$ είναι συνεχής και γνησίως αύξουσα στο $[1,e]$. Επίσης
\[
g(1)=0<\frac{2}{e}
\]
και
\[
g(e)=1+\frac{1}{e^2}>\frac{2}{e},
\]
αφού
\[
1+\frac{1}{e^2}-\frac{2}{e}
=\frac{(e-1)^2}{e^2}>0.
\]
Άρα, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει $x_0\in(1,e)$ τέτοιο ώστε
\[
g(x_0)=\frac{2}{e}.
\]
Το $x_0$ είναι μοναδικό, επειδή η $g$ είναι γνησίως αύξουσα.
\end{erwthma}
\end{thema}
