%=========== ΛΥΣΗ - 73ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Από το σχήμα της $C_{f''}$ έχουμε
\[
f''(x)>0,\quad x\in(0,2),
\]
\[
f''(2)=0,
\]
\[
f''(x)<0,\quad x\in(2,4),
\]
\[
f''(4)=0
\]
και
\[
f''(x)>0,\quad x\in(4,6).
\]

\begin{erwthma}
\item Από το πρόσημο της $f''$ προκύπτει ότι:
\begin{itemize}
\item η $f$ είναι κυρτή στο $[0,2]$,
\item η $f$ είναι κοίλη στο $[2,4]$,
\item η $f$ είναι κυρτή στο $[4,6]$.
\end{itemize}

Επειδή η $f''$ αλλάζει πρόσημο στα $x=2$ και $x=4$, η γραφική
παράσταση $C_f$ έχει σημεία καμπής τα
\[
{A(2,f(2))}
\quad\text{και}\quad
{B(4,f(4))}.
\]

\item Στο διάστημα $(0,2)$ ισχύει
\[
f''(x)>0.
\]
Άρα η $f'$ είναι γνησίως αύξουσα στο $[0,2]$.

Επειδή
\[
f'(0)=0,
\]
παίρνουμε
\[
f'(x)>0,\quad x\in(0,2].
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $[0,2]$.

\item Από το προηγούμενο ερώτημα η $f$ είναι γνησίως αύξουσα στο
$[0,2]$. Επομένως, για κάθε
\[
x\in(0,2],
\]
έχουμε
\[
f(x)>f(0).
\]
Επειδή
\[
f(0)=1,
\]
προκύπτει
\[
{f(x)>1,\quad x\in(0,2]}.
\]

\item Από το β΄ ερώτημα γνωρίζουμε ότι η $f'$ είναι γνησίως αύξουσα στο
$[0,2]$ και
\[
f'(0)=0.
\]
Άρα
\[
f'(2)>0.
\]

Επίσης, στο διάστημα $(4,6)$ ισχύει
\[
f''(x)>0,
\]
άρα η $f'$ είναι γνησίως αύξουσα στο $[4,6]$. Επειδή
\[
f'(6)=0,
\]
θα είναι
\[
f'(4)<0.
\]

Η $f'$ είναι συνεχής στο $[2,4]$ και
\[
f'(2)>0,\qquad f'(4)<0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\xi\in(2,4)\subset(2,6)
\]
τέτοιο ώστε
\[
f'(\xi)=0.
\]

Επομένως η εξίσωση
\[
f'(x)=0
\]
έχει τουλάχιστον μία ρίζα στο διάστημα $(2,6)$.
\end{erwthma}
\end{thema}
