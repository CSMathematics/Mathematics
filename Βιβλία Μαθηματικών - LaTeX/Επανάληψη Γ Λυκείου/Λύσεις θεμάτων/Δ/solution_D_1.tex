\begin{thema}{Δ}
Η συνάρτηση $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ ως πράξεις παραγωγίσιμων συναρτήσεων και για κάθε $x>0$ είναι
\[
f'(x)=(x\ln x-x+1)'=\ln x+x\cdot\frac{1}{x}-1=\ln x.
\]
\begin{erwthma}
\item Βρίσκουμε τη ρίζα και το πρόσημο της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow \ln x=0\Leftrightarrow x=1$.
\item $f'(x)>0\Leftrightarrow \ln x>0\Leftrightarrow x>1$.
\item $f'(x)<0\Leftrightarrow \ln x<0\Leftrightarrow 0<x<1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{d, -, z, +, }
\tkzTabLine{d, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}

Επομένως η $f$ είναι \textbf{γνησίως φθίνουσα} στο διάστημα $(0,1]$ και \textbf{γνησίως αύξουσα} στο διάστημα $[1,+\infty)$. Άρα παρουσιάζει ολικό ελάχιστο στο $x_0=1$, με τιμή
\[
f(1)=1\ln 1-1+1=0.
\]
Η $f$ δεν παρουσιάζει μέγιστο.

\item Από το προηγούμενο ερώτημα, για κάθε $x>0$ ισχύει
\[
f(x)\geq f(1)\Leftrightarrow x\ln x-x+1\geq 0
\Leftrightarrow x\ln x\geq x-1.
\]
Η ισότητα ισχύει μόνο όταν η $f$ λαμβάνει την ελάχιστη τιμή της, δηλαδή μόνο για $x=1$.

\item Έστω $M(x_0,f(x_0))$, με $x_0>0$, το σημείο επαφής. Η εφαπτομένη της $C_f$ στο $M$ έχει εξίσωση
\begin{align*}
y-f(x_0)&=f'(x_0)(x-x_0)\\
\Leftrightarrow y-\left(x_0\ln x_0-x_0+1\right)&=\ln x_0(x-x_0)\\
\Leftrightarrow y&=x\ln x_0-x_0+1.
\end{align*}
Η εφαπτομένη διέρχεται από την αρχή των αξόνων $O(0,0)$, οπότε
\[
0=0\cdot\ln x_0-x_0+1\Leftrightarrow x_0=1.
\]
Επειδή $f(1)=0$ και $f'(1)=\ln 1=0$, η ζητούμενη εφαπτομένη είναι
\[
{y=0}.
\]

\item Έστω $\alpha,\beta>1$ με $\alpha\neq\beta$. Η $F$ είναι συνεχής στο κλειστό διάστημα με άκρα τα $\alpha,\beta$ και παραγωγίσιμη στο εσωτερικό του. Επομένως, από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi$ μεταξύ των $\alpha,\beta$ τέτοιο ώστε
\[
\frac{F(\beta)-F(\alpha)}{\beta-\alpha}=F'(\xi)=f(\xi).
\]
Επειδή $\xi>1$ και η $f$ είναι γνησίως αύξουσα στο $[1,+\infty)$, έχουμε
\[
f(\xi)>f(1)=0.
\]
Άρα
\[
{\frac{F(\beta)-F(\alpha)}{\beta-\alpha}>0}.
\]
\end{erwthma}
\end{thema}
