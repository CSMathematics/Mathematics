%=========== ΛΥΣΗ - 44ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Έχουμε
\[
\lim_{x\to1}\frac{\ln x}{x-1}
\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{1}{x}
=1=f(1).
\]
Επομένως η $f$ είναι συνεχής στο $x_0=1$.

Για κάθε $x>0$ με $x\neq1$ είναι
\begin{align*}
f'(x)
&=\left(\frac{\ln x}{x-1}\right)'\\
&=\frac{\frac{x-1}{x}-\ln x}{(x-1)^2}.
\end{align*}
Θεωρούμε τη συνάρτηση
\[
h(x)=\frac{x-1}{x}-\ln x
=1-\frac{1}{x}-\ln x,\quad x>0.
\]
Έχουμε
\[
h'(x)=\frac{1}{x^2}-\frac{1}{x}
=\frac{1-x}{x^2}.
\]
Άρα η $h$ είναι γνησίως αύξουσα στο $(0,1]$ και γνησίως φθίνουσα στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=1$, με τιμή
\[
h(1)=0.
\]
Συνεπώς
\[
h(x)<0
\]
για κάθε $x>0$ με $x\neq1$. Επειδή $(x-1)^2>0$, έχουμε
\[
f'(x)<0
\]
για κάθε $x>0$ με $x\neq1$.

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στα διαστήματα $(0,1]$ και $[1,+\infty)$. Επειδή είναι συνεχής στο $1$, είναι γνησίως φθίνουσα σε ολόκληρο το $(0,+\infty)$.

\item Από τον ορισμό της παραγώγου έχουμε
\begin{align*}
f'(1)
&=\lim_{x\to1}\frac{f(x)-f(1)}{x-1}\\
&=\lim_{x\to1}
\frac{\frac{\ln x}{x-1}-1}{x-1}\\
&=\lim_{x\to1}
\frac{\ln x-x+1}{(x-1)^2}.
\end{align*}
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
f'(1)
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{\frac{1}{x}-1}{2(x-1)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{-\frac{1}{x^2}}{2}\\
&={-\frac{1}{2}}.
\end{align*}
Άρα η $f$ είναι παραγωγίσιμη στο $x_0=1$.

Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο
\[
A(1,f(1))\equiv A(1,1)
\]
είναι
\begin{align*}
y-f(1)&=f'(1)(x-1)\\
y-1&=-\frac{1}{2}(x-1),
\end{align*}
δηλαδή
\[
{y=-\frac{1}{2}x+\frac{3}{2}}.
\]

\item Έχουμε
\[
\lim_{x\to0^+}\frac{\ln x}{x-1}=+\infty
\]
και
\[
\lim_{x\to+\infty}\frac{\ln x}{x-1}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{1}{x}
=0.
\]
Επειδή η $f$ είναι συνεχής και γνησίως φθίνουσα στο $(0,+\infty)$, το σύνολο τιμών της είναι
\[
{f\big((0,+\infty)\big)=(0,+\infty)}.
\]

Για $x>1$ η εξίσωση
\[
2\ln x=x-1
\]
είναι ισοδύναμη με
\[
f(x)=\frac{1}{2}.
\]
Έχουμε
\[
f(1)=1>\frac{1}{2}
\]
και
\[
f(e^2)=\frac{2}{e^2-1}<\frac{1}{2},
\]
αφού $e^2>5$. Επομένως, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει μία τουλάχιστον ρίζα στο $(1,e^2)$. Η ρίζα αυτή είναι μοναδική, διότι η $f$ είναι γνησίως φθίνουσα.

\item Επειδή η $f$ είναι γνησίως φθίνουσα και $f(1)=1$, διακρίνουμε τις εξής περιπτώσεις:
\begin{itemize}
\item Αν $x>1$, τότε
\[
f(x)<f(1)
\Leftrightarrow
\frac{\ln x}{x-1}<1
\Leftrightarrow
\ln x<x-1.
\]
\item Αν $0<x<1$, τότε
\[
f(x)>f(1)
\Leftrightarrow
\frac{\ln x}{x-1}>1.
\]
Επειδή $x-1<0$, πολλαπλασιάζοντας με $x-1$ αλλάζει η φορά της ανισότητας και παίρνουμε
\[
\ln x<x-1.
\]
\end{itemize}
Για $x=1$ ισχύει η ισότητα. Επομένως
\[
{\ln x\leq x-1}
\]
για κάθε $x>0$, με ισότητα μόνο για $x=1$.

Τέλος,
\begin{align*}
\lim_{x\to0^+}\left(f(x)\hm x\right)
&=\lim_{x\to0^+}
\left(
\frac{\ln x}{x-1}\hm x
\right)\\
&=\lim_{x\to0^+}
\left[
\frac{x\ln x}{x-1}\cdot\frac{\hm x}{x}
\right].
\end{align*}
Επειδή
\[
\lim_{x\to0^+}x\ln x=0,
\quad
\lim_{x\to0^+}(x-1)=-1
\]
και
\[
\lim_{x\to0^+}\frac{\hm x}{x}=1,
\]
προκύπτει
\[
{\lim_{x\to0^+}\left(f(x)\hm x\right)=0}.
\]
\end{erwthma}
\end{thema}
