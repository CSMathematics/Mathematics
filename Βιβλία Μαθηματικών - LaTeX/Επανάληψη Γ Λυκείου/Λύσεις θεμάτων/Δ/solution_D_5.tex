\begin{thema}{Δ}
Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως πράξεις παραγωγίσιμων συναρτήσεων και για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f'(x)
&=(xe^x-e^x+1)'\\
&=e^x+xe^x-e^x\\
&=xe^x.
\end{align*}
\begin{erwthma}
\item Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f'$ είναι το πρόσημο του $x$. Επομένως:
\begin{itemize}
\item $f'(x)<0$ για $x<0$,
\item $f'(0)=0$,
\item $f'(x)>0$ για $x>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{, -, z, +, }
\tkzTabVar{+/{1},-/{0},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$. Παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
f(0)=0\cdot e^0-e^0+1=0.
\]
Επομένως
\[
f(x)\geq0
\]
για κάθε $x\in\mathbb{R}$, με την ισότητα να ισχύει μόνο για $x=0$. Συνεπώς η εξίσωση $f(x)=0$ έχει μοναδική λύση
\[
{x=0}.
\]

\item Έχουμε
\begin{align*}
\lim_{x\to-\infty}f(x)
&=\lim_{x\to-\infty}\left[(x-1)e^x+1\right]\\
&=\lim_{x\to-\infty}\frac{x-1}{e^{-x}}+1\\
&\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to-\infty}\frac{1}{-e^{-x}}+1\\
&=1.
\end{align*}
Επομένως η ευθεία
\[
{y=1}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.

Επιπλέον,
\[
\lim_{x\to+\infty}f(x)
=\lim_{x\to+\infty}\left[(x-1)e^x+1\right]
=+\infty.
\]
Για $x<0$ είναι $x-1<0$ και $e^x>0$, άρα
\[
f(x)=(x-1)e^x+1<1.
\]
Με βάση τη μονοτονία και τα παραπάνω όρια:
\begin{itemize}
\item στο $(-\infty,0]$ η $f$ είναι γνησίως φθίνουσα και έχει σύνολο τιμών $[0,1)$,
\item στο $[0,+\infty)$ η $f$ είναι γνησίως αύξουσα και έχει σύνολο τιμών $[0,+\infty)$.
\end{itemize}

Επομένως, για την εξίσωση $f(x)=c$ ισχύουν τα εξής:
\begin{itemize}
\item Αν $c<0$, η εξίσωση δεν έχει πραγματικές λύσεις.
\item Αν $c=0$, έχει ακριβώς μία λύση, την $x=0$.
\item Αν $0<c<1$, έχει ακριβώς δύο λύσεις, μία στο $(-\infty,0)$ και μία στο $(0,1)$.
\item Αν $c=1$, έχει ακριβώς μία λύση, την $x=1$, αφού
\[
f(x)=1\Leftrightarrow (x-1)e^x=0\Leftrightarrow x=1.
\]
\item Αν $c>1$, έχει ακριβώς μία λύση στο $(1,+\infty)$.
\end{itemize}

\item Η εξίσωση
\[
xe^x-e^x+1=2026
\]
είναι ισοδύναμη με την εξίσωση
\[
f(x)=2026.
\]
Επειδή $2026>1$, από το προηγούμενο ερώτημα η εξίσωση έχει ακριβώς μία πραγματική ρίζα, η οποία ανήκει στο διάστημα $(1,+\infty)$.

\item Επειδή η $F$ είναι παράγουσα της $f$, έχουμε
\[
F'(x)=f(x)
\]
για κάθε $x\in\mathbb{R}$. Επιπλέον $F(0)=0$ και $f(0)=0$. Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0}\frac{F(x)}{x^3}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{F'(x)}{3x^2}\\
&=\lim_{x\to0}\frac{f(x)}{3x^2}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{f'(x)}{6x}\\
&=\lim_{x\to0}\frac{xe^x}{6x}\\
&=\lim_{x\to0}\frac{e^x}{6}\\
&={\frac{1}{6}}.
\end{align*}
\end{erwthma}
\end{thema}
