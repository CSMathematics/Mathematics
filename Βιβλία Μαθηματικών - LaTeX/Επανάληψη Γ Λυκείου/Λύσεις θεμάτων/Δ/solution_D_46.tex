\begin{thema}{Δ}
\begin{erwthma}
\item Παραγωγίζοντας τη δοσμένη σχέση, παίρνουμε
\[
e^{f(x)}f'(x)+f'(x)=1.
\]
Επομένως
\[
f'(x)=\frac{1}{e^{f(x)}+1}>0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Παραγωγίζοντας ξανά, έχουμε
\begin{align*}
f''(x)
&=\left(e^{f(x)}+1\right)^{-1}'\\
&=-\left(e^{f(x)}+1\right)^{-2}
e^{f(x)}f'(x)\\
&=-\frac{e^{f(x)}}{\left(e^{f(x)}+1\right)^3}<0.
\end{align*}
Άρα η $f$ είναι \textbf{κοίλη} στο $\mathbb{R}$.

\item Θεωρούμε τη συνάρτηση
\[
h(t)=e^t+t,\quad t\in\mathbb{R}.
\]
Επειδή
\[
h'(t)=e^t+1>0,
\]
η $h$ είναι γνησίως αύξουσα. Επιπλέον,
\[
\lim_{t\to-\infty}h(t)=-\infty
\quad\text{και}\quad
\lim_{t\to+\infty}h(t)=+\infty.
\]
Από τη σχέση
\[
h(f(x))=x+1
\]
προκύπτει ότι, καθώς το $x+1$ διατρέχει το $\mathbb{R}$, η $f(x)$ διατρέχει επίσης το $\mathbb{R}$. Άρα
\[
{f(\mathbb{R})=\mathbb{R}}.
\]

Για $x=0$ η δοσμένη σχέση δίνει
\[
e^{f(0)}+f(0)=1.
\]
Επειδή $h(0)=1$ και η $h$ είναι $1-1$, έχουμε
\[
{f(0)=0}.
\]
Αφού η $f$ είναι γνησίως αύξουσα,
\[
f(u)<0\Leftrightarrow u<0.
\]
Επομένως
\begin{align*}
f\left(f(x^2-4)\right)<0
&\Leftrightarrow f(x^2-4)<0\\
&\Leftrightarrow x^2-4<0\\
&\Leftrightarrow {-2<x<2}.
\end{align*}

\item Έχουμε
\[
f(0)=0
\]
και
\[
f'(0)=\frac{1}{e^{f(0)}+1}=\frac{1}{2}.
\]
Άρα η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $O(0,0)$ είναι
\[
{y=\frac{x}{2}}.
\]

Για το ζητούμενο όριο έχουμε μορφή $\frac{0}{0}$. Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0}\frac{2f(x)-x}{x^2}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{2f'(x)-1}{2x}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}f''(x)\\
&=f''(0).
\end{align*}
Από τον τύπο της δεύτερης παραγώγου έχουμε
\[
f''(0)
=-\frac{e^{f(0)}}{\left(e^{f(0)}+1\right)^3}
=-\frac{1}{8}.
\]
Επομένως
\[
{\lim_{x\to0}\frac{2f(x)-x}{x^2}=-\frac{1}{8}}.
\]

\item Επειδή η $f$ είναι κοίλη στο $\mathbb{R}$, η γραφική της παράσταση βρίσκεται κάτω από την εφαπτομένη της στο $x_0=0$. Άρα
\[
f(x)\leq\frac{x}{2}
\]
για κάθε $x\in[0,e]$, με ισότητα μόνο για $x=0$.

Το ζητούμενο εμβαδόν είναι
\[
E=\int_0^e\left(\frac{x}{2}-f(x)\right)\d x.
\]
Για να υπολογίσουμε το ολοκλήρωμα της $f$, θέτουμε
\[
t=f(x).
\]
Από τη δοσμένη σχέση έχουμε
\[
x=e^t+t-1
\]
και
\[
\d x=(e^t+1)\d t.
\]
Για $x=0$ είναι $t=f(0)=0$, ενώ για $x=e$ είναι $t=1$, αφού
\[
e^1+1=e+1.
\]
Επομένως
\begin{align*}
\int_0^e f(x)\d x
&=\int_0^1t(e^t+1)\d t\\
&=\left[te^t-e^t+\frac{t^2}{2}\right]_0^1\\
&=\frac{3}{2}.
\end{align*}
Άρα
\begin{align*}
E
&=\int_0^e\frac{x}{2}\d x-\int_0^ef(x)\d x\\
&=\frac{e^2}{4}-\frac{3}{2}\\
&={\frac{e^2-6}{4}}.
\end{align*}
\end{erwthma}
\end{thema}
