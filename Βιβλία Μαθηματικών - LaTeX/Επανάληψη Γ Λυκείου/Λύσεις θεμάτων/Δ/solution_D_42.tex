\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και για κάθε $x>0$ είναι
\[
f'(x)=e^{x-1}+\frac{1}{x}+1>0.
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.

Επιπλέον,
\[
\lim_{x\to0^+}f(x)=-\infty
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty.
\]
Επομένως το σύνολο τιμών της $f$ είναι
\[
{f\big((0,+\infty)\big)=\mathbb{R}}.
\]

Επειδή
\[
f(1)=e^0+\ln1+1-2=0,
\]
ο αριθμός $1$ είναι ρίζα της εξίσωσης $f(x)=0$. Η ρίζα αυτή είναι μοναδική, διότι η $f$ είναι γνησίως αύξουσα. Άρα
\[
{x=1}.
\]

\item Για κάθε $x>0$ είναι
\[
f''(x)=e^{x-1}-\frac{1}{x^2}
=\frac{x^2e^{x-1}-1}{x^2}.
\]
Θεωρούμε τη συνάρτηση
\[
h(x)=x^2e^{x-1},\quad x>0.
\]
Έχουμε
\[
h'(x)=x(x+2)e^{x-1}>0,
\]
οπότε η $h$ είναι γνησίως αύξουσα στο $(0,+\infty)$. Επειδή
\[
h(1)=1,
\]
προκύπτει:
\begin{itemize}
\item $f''(x)<0$ για $0<x<1$,
\item $f''(1)=0$,
\item $f''(x)>0$ για $x>1$.
\end{itemize}
Άρα η $f$ είναι \textbf{κοίλη} στο $(0,1]$ και \textbf{κυρτή} στο $[1,+\infty)$. Επομένως η $C_f$ έχει ακριβώς ένα σημείο καμπής
\[
{A(1,f(1))\equiv A(1,0)}.
\]

Επειδή
\[
f'(1)=1+1+1=3,
\]
η εξίσωση της εφαπτομένης της $C_f$ στο σημείο καμπής είναι
\begin{align*}
y-f(1)&=f'(1)(x-1)\\
y&=3(x-1),
\end{align*}
δηλαδή
\[
{y=3x-3}.
\]

\item Η $f$ είναι κυρτή στο $[1,+\infty)$, οπότε η γραφική της παράσταση βρίσκεται πάνω από την εφαπτομένη της στο σημείο $A(1,0)$. Άρα, για κάθε $x\in[1,e]$,
\[
f(x)\geq3x-3,
\]
με ισότητα μόνο για $x=1$.

Το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_1^e\left[f(x)-(3x-3)\right]\d x\\
&=\int_1^e
\left[e^{x-1}+\ln x-2x+1\right]\d x\\
&=\left[e^{x-1}+x\ln x-x^2\right]_1^e\\
&={e^{e-1}+e-e^2}.
\end{align*}

\item Για να ορίζεται η εξίσωση
\[
e^{\ln x-1}+\ln(\ln x)+\ln x=2,
\]
πρέπει
\[
\ln x>0
\Leftrightarrow
x>1.
\]
Για $x>1$ είναι
\begin{align*}
e^{\ln x-1}+\ln(\ln x)+\ln x=2
&\Leftrightarrow
f(\ln x)=0.
\end{align*}
Από το πρώτο ερώτημα, η μοναδική ρίζα της $f$ είναι ο αριθμός $1$. Επομένως
\begin{align*}
f(\ln x)=0
&\Leftrightarrow \ln x=1\\
&\Leftrightarrow {x=e}.
\end{align*}

Για τη δεύτερη εξίσωση θεωρούμε τη συνάρτηση
\[
g(x)=xf(x),\quad x\in[1,e].
\]
Η $g$ είναι συνεχής στο $[1,e]$ και παραγωγίσιμη στο $(1,e)$. Για κάθε $x\in(1,e)$ είναι
\[
f(x)>f(1)=0
\quad\text{και}\quad
f'(x)>0.
\]
Άρα
\[
g'(x)=f(x)+xf'(x)>0,
\]
οπότε η $g$ είναι γνησίως αύξουσα στο $[1,e]$.

Στα άκρα του διαστήματος έχουμε
\[
g(1)=f(1)=0<e^2-1
\]
και
\begin{align*}
g(e)
&=ef(e)\\
&=e\left(e^{e-1}+e-1\right)\\
&>e(1+e-1)\\
&=e^2>e^2-1.
\end{align*}
Επομένως, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει $x_0\in(1,e)$ τέτοιο ώστε
\[
g(x_0)=e^2-1.
\]
Η ρίζα αυτή είναι μοναδική, διότι η $g$ είναι γνησίως αύξουσα στο $[1,e]$. Συνεπώς η εξίσωση
\[
xf(x)=e^2-1
\]
έχει ακριβώς μία ρίζα στο διάστημα $(1,e)$.
\end{erwthma}
\end{thema}
