%=========== ΛΥΣΗ - 77ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι γνησίως αύξουσα στο $[0,5]$ και
\[
f(2)=0.
\]
Άρα
\[
x>2
\Leftrightarrow
f(x)>f(2)
\Leftrightarrow
f(x)>0.
\]
Επομένως η λύση της ανίσωσης $f(x)>0$ στο $[0,5]$ είναι
\[
{x\in(2,5]}.
\]

\item Η χορδή που ενώνει τα σημεία
\[
A(0,f(0))=(0,-2)
\quad\text{και}\quad
B(5,f(5))=(5,4)
\]
έχει συντελεστή διεύθυνσης
\[
\lambda_{AB}=\frac{4-(-2)}{5-0}=\frac65.
\]
Άρα η εξίσωση της χορδής $AB$ είναι
\[
y=\frac65x-2.
\]

Επειδή η $f$ είναι κυρτή στο $[0,5]$, η γραφική της παράσταση βρίσκεται
κάτω από τη χορδή που ενώνει τα άκρα της. Επομένως
\[
{f(x)\leq \frac65x-2,\quad x\in[0,5]}.
\]

\item Η $f$ είναι συνεχής στο $[0,5]$ και παραγωγίσιμη στο $(0,5)$.
Άρα, από το θεώρημα μέσης τιμής, υπάρχει
\[
\xi_1\in(0,5)
\]
τέτοιο ώστε
\[
f'(\xi_1)=\frac{f(5)-f(0)}{5-0}
=\frac{4-(-2)}{5}
=\frac65.
\]
Επομένως υπάρχει $\xi_1\in(0,5)$ τέτοιο ώστε
\[
{f'(\xi_1)=\frac65}.
\]

\item Εφαρμόζουμε το θεώρημα μέσης τιμής στο διάστημα $[2,5]$.
Υπάρχει
\[
\xi_2\in(2,5)
\]
τέτοιο ώστε
\[
f'(\xi_2)=\frac{f(5)-f(2)}{5-2}
=\frac{4-0}{3}
=\frac43.
\]

Επειδή η $f'$ είναι γνησίως αύξουσα στο $(0,5)$ και
\[
2<\xi_2,
\]
παίρνουμε
\[
f'(2)<f'(\xi_2)=\frac43.
\]

Επίσης, επειδή
\[
\xi_2<5
\]
και η $f'$ είναι γνησίως αύξουσα, ενώ υπάρχει η αριστερή παράγωγος στο
$5$, έχουμε
\[
f'(\xi_2)<f'_{-}(5).
\]
Άρα
\[
\frac43<f'_{-}(5).
\]

Συνεπώς
\[
{f'(2)<\frac43<f'_{-}(5)}.
\]
\end{erwthma}
\end{thema}
