\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=1+\syn x\geq0.
\]
Η ισότητα $f'(x)=0$ ισχύει μόνο για
\[
x=(2k+1)\pi,\quad k\in\mathbb{Z}.
\]
Επειδή η $f'$ είναι μη αρνητική και δεν μηδενίζεται σε κανένα διάστημα, η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Έχουμε
\[
f(\pi)=\pi+\hm\pi=\pi.
\]
Επομένως, λόγω της γνησίως αύξουσας μονοτονίας της $f$,
\begin{align*}
f(x^2-x+\pi)>\pi
&\Leftrightarrow
f(x^2-x+\pi)>f(\pi)\\
&\Leftrightarrow
x^2-x+\pi>\pi\\
&\Leftrightarrow
x(x-1)>0\\
&\Leftrightarrow
{x\in(-\infty,0)\cup(1,+\infty)}.
\end{align*}

\item Για κάθε $x\in\mathbb{R}$ είναι
\[
f''(x)=-\hm x.
\]
Στο διάστημα $[0,2\pi]$ έχουμε:
\begin{itemize}
\item $f''(x)<0$ για $x\in(0,\pi)$,
\item $f''(\pi)=0$,
\item $f''(x)>0$ για $x\in(\pi,2\pi)$.
\end{itemize}
Άρα η $f$ είναι \textbf{κοίλη} στο $[0,\pi]$ και \textbf{κυρτή} στο $[\pi,2\pi]$. Επομένως η $C_f$ έχει στο διάστημα αυτό σημείο καμπής
\[
{A(\pi,f(\pi))\equiv A(\pi,\pi)}.
\]

Επειδή
\[
f'(\pi)=1+\syn\pi=0,
\]
η εξίσωση της εφαπτομένης της $C_f$ στο $A$ είναι
\[
y-\pi=0(x-\pi),
\]
δηλαδή
\[
{y=\pi}.
\]

\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
f(2\pi-x)+f(x)
&=2\pi-x+\hm(2\pi-x)+x+\hm x\\
&=2\pi-x-\hm x+x+\hm x\\
&={2\pi}.
\end{align*}

Ισοδύναμα,
\[
f(2\pi-x)=2\pi-f(x).
\]
Άρα, αν το σημείο $M(x,f(x))$ ανήκει στη $C_f$, τότε και το σημείο
\[
M'(2\pi-x,2\pi-f(x))
\]
ανήκει στη $C_f$. Τα σημεία $M$ και $M'$ είναι συμμετρικά ως προς το σημείο
\[
{A(\pi,\pi)}.
\]
Επομένως το $A$ είναι κέντρο συμμετρίας της γραφικής παράστασης $C_f$.

\item Επειδή η $f$ είναι γνησίως αύξουσα και $f(\pi)=\pi$, έχουμε
\[
f(x)<\pi
\]
για $x\in[0,\pi)$ και
\[
f(x)>\pi
\]
για $x\in(\pi,2\pi]$.

Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_0^\pi\left[\pi-f(x)\right]\d x
+\int_\pi^{2\pi}\left[f(x)-\pi\right]\d x.
\end{align*}
Λόγω της κεντρικής συμμετρίας, τα δύο επιμέρους εμβαδά είναι ίσα. Επομένως
\begin{align*}
E
&=2\int_0^\pi\left(\pi-x-\hm x\right)\d x\\
&=2\left[\pi x-\frac{x^2}{2}+\syn x\right]_0^\pi\\
&=2\left(\frac{\pi^2}{2}-2\right)\\
&={\pi^2-4}.
\end{align*}
\end{erwthma}
\end{thema}
