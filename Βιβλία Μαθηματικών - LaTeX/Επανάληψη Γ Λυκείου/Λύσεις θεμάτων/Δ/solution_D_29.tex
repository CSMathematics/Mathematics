\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $[0,\pi]$ και
\begin{align*}
f'(x)
&=(x\hm x+\syn x)'\\
&=\hm x+x\syn x-\hm x\\
&=x\syn x.
\end{align*}
Επειδή $x\geq0$ στο $[0,\pi]$, το πρόσημο της $f'$ είναι το πρόσημο του $\syn x$. Επομένως:
\begin{itemize}
\item $f'(x)>0$ για $0<x<\frac{\pi}{2}$,
\item $f'\left(\frac{\pi}{2}\right)=0$,
\item $f'(x)<0$ για $\frac{\pi}{2}<x<\pi$.
\end{itemize}
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\left[0,\frac{\pi}{2}\right]$ και \textbf{γνησίως φθίνουσα} στο $\left[\frac{\pi}{2},\pi\right]$.

Οι τιμές της στα κρίσιμα σημεία και στα άκρα είναι
\[
f(0)=1,
\]
\[
f\left(\frac{\pi}{2}\right)=\frac{\pi}{2}
\]
και
\[
f(\pi)=-1.
\]
Επομένως η $f$ παρουσιάζει ολικό μέγιστο στο $x=\frac{\pi}{2}$, με τιμή
\[
{\frac{\pi}{2}},
\]
και ολικό ελάχιστο στο $x=\pi$, με τιμή
\[
{-1}.
\]

\item Η $f$ είναι συνεχής στο $\left[\frac{\pi}{2},\pi\right]$ και
\[
f\left(\frac{\pi}{2}\right)=\frac{\pi}{2}>0,
\quad
f(\pi)=-1<0.
\]
Άρα, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει $x_0\in\left(\frac{\pi}{2},\pi\right)$ τέτοιο ώστε
\[
f(x_0)=0.
\]
Η ρίζα αυτή είναι μοναδική, διότι η $f$ είναι γνησίως φθίνουσα στο $\left[\frac{\pi}{2},\pi\right]$. Επιπλέον, στο $\left[0,\frac{\pi}{2}\right]$ είναι
\[
f(x)\geq f(0)=1>0,
\]
οπότε δεν υπάρχουν άλλες ρίζες.

\item Για κάθε $x\in[0,\pi]$ έχουμε
\begin{align*}
f(\pi-x)
&=(\pi-x)\hm(\pi-x)+\syn(\pi-x)\\
&=(\pi-x)\hm x-\syn x.
\end{align*}
Επομένως
\begin{align*}
f(x)+f(\pi-x)
&=x\hm x+\syn x+(\pi-x)\hm x-\syn x\\
&={\pi\hm x}.
\end{align*}

Θέτουμε
\[
I=\int_0^\pi f(x)\d x.
\]
Με την αλλαγή μεταβλητής $u=\pi-x$ έχουμε
\[
I=\int_0^\pi f(\pi-x)\d x.
\]
Άρα
\begin{align*}
2I
&=\int_0^\pi\left[f(x)+f(\pi-x)\right]\d x\\
&=\pi\int_0^\pi\hm x\,\d x\\
&=\pi\left[-\syn x\right]_0^\pi\\
&=2\pi.
\end{align*}
Επομένως
\[
{\int_0^\pi f(x)\d x=\pi}.
\]

\item Η $H$ είναι παραγωγίσιμη στο $[0,\pi]$ και
\begin{align*}
H'(x)
&=(2\hm x-x\syn x)'\\
&=2\syn x-\syn x+x\hm x\\
&=x\hm x+\syn x\\
&=f(x).
\end{align*}
Άρα η $H$ είναι παράγουσα της $f$ στο $[0,\pi]$.

Εφαρμόζοντας το Θεώρημα Μέσης Τιμής για την $H$ στο $\left[0,\frac{\pi}{2}\right]$, υπάρχει $\xi_1\in\left(0,\frac{\pi}{2}\right)$ τέτοιο ώστε
\begin{align*}
f(\xi_1)
&=H'(\xi_1)\\
&=\frac{H\left(\frac{\pi}{2}\right)-H(0)}{\frac{\pi}{2}-0}\\
&=\frac{2}{\frac{\pi}{2}}\\
&=\frac{4}{\pi}.
\end{align*}

Ομοίως, εφαρμόζοντας το Θεώρημα Μέσης Τιμής στο $\left[\frac{\pi}{2},\pi\right]$, υπάρχει $\xi_2\in\left(\frac{\pi}{2},\pi\right)$ τέτοιο ώστε
\begin{align*}
f(\xi_2)
&=H'(\xi_2)\\
&=\frac{H(\pi)-H\left(\frac{\pi}{2}\right)}{\pi-\frac{\pi}{2}}\\
&=\frac{\pi-2}{\frac{\pi}{2}}\\
&=2-\frac{4}{\pi}.
\end{align*}
Επομένως
\[
{f(\xi_1)+f(\xi_2)=2}.
\]
\end{erwthma}
\end{thema}
