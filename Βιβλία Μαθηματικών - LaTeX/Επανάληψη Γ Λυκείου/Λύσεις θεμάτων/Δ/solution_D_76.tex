%=========== ΛΥΣΗ - 76ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Από το σχήμα της $C_{f'}$ και τα δεδομένα έχουμε
\[
f'(0)=4,\quad f'(4)=0,\quad f'(8)=2.
\]
Επίσης, η $f'$ είναι γνησίως φθίνουσα στο $[0,4]$ και γνησίως αύξουσα
στο $[4,8]$.

\begin{erwthma}
\item Στο διάστημα $[0,4]$ η $f'$ είναι γνησίως φθίνουσα από το $4$ έως
το $0$. Άρα
\[
f'(x)\geq 0,\quad x\in[0,4].
\]
Στο διάστημα $[4,8]$ η $f'$ είναι γνησίως αύξουσα από το $0$ έως το
$2$. Άρα
\[
f'(x)\geq 0,\quad x\in[4,8].
\]
Επομένως
\[
f'(x)\geq 0,\quad x\in[0,8].
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $[0,8]$, αφού η παράγωγος είναι
θετική σε κάθε υποδιάστημα, εκτός από το μεμονωμένο σημείο $x=4$.

Επειδή
\[
f(0)=0,
\]
για κάθε $x\in[0,8]$ έχουμε
\[
f(x)\geq f(0)=0.
\]
Άρα η λύση της ανίσωσης $f(x)\geq 0$ στο $[0,8]$ είναι
\[
{x\in[0,8]}.
\]

\item Επειδή η $f'$ είναι γνησίως φθίνουσα στο $[0,4]$, η $f$ είναι
κοίλη στο $[0,4]$.

Επειδή η $f'$ είναι γνησίως αύξουσα στο $[4,8]$, η $f$ είναι κυρτή στο
$[4,8]$.

Στο $x=4$ η κυρτότητα αλλάζει από κοίλη σε κυρτή. Άρα η $C_f$ έχει
σημείο καμπής το
\[
{A(4,f(4))}.
\]

\item Στο $[0,4]$ η $f'$ είναι συνεχής και γνησίως φθίνουσα, με
\[
f'(0)=4
\quad\text{και}\quad
f'(4)=0.
\]
Επειδή
\[
0<2<4,
\]
υπάρχει μοναδικό
\[
\xi_1\in(0,4)
\]
τέτοιο ώστε
\[
f'(\xi_1)=2.
\]

Στο $[4,8]$ η $f'$ είναι γνησίως αύξουσα, με
\[
f'(4)=0
\quad\text{και}\quad
f'(8)=2.
\]
Άρα η μοναδική λύση της εξίσωσης $f'(x)=2$ στο $[4,8]$ είναι
\[
x=8.
\]

Επομένως η εξίσωση
\[
f'(x)=2
\]
έχει ακριβώς δύο ρίζες στο $[0,8]$:
\[
{\xi_1\in(0,4)\ \text{και}\ 8}.
\]

\item Από το α΄ ερώτημα γνωρίζουμε ότι η $f$ είναι γνησίως αύξουσα στο
$[0,8]$.

Άρα, αν
\[
x\in[0,4],
\]
τότε
\[
x\leq 4
\]
και επομένως
\[
f(x)\leq f(4).
\]

Επίσης, αν
\[
x\in[4,8],
\]
τότε
\[
x\geq 4
\]
και επομένως
\[
f(x)\geq f(4).
\]

Συνεπώς
\[
{f(x)\leq f(4),\quad x\in[0,4]}
\]
και
\[
{f(x)\geq f(4),\quad x\in[4,8]}.
\]
\end{erwthma}
\end{thema}
