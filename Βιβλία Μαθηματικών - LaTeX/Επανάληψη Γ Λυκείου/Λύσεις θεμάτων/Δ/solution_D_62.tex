%=========== ΛΥΣΗ - 62ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Παρατηρούμε αρχικά ότι
\[
p'(x)=\left(-x^2+4x-1\right)'=-2x+4=q(x).
\]
Επομένως
\[
{f(x)=p(x)=-x^2+4x-1}
\]
και
\[
{f'(x)=q(x)=-2x+4}.
\]

Το ίδιο συμπέρασμα προκύπτει και γραφικά. Πράγματι,
\[
f'(x)=0\Leftrightarrow -2x+4=0\Leftrightarrow x=2,
\]
ενώ η μπλε παραβολή παρουσιάζει μέγιστο στο σημείο $(2,3)$. Επιπλέον,
\[
\begin{array}{lll}
f'(x)>0, & x<2, & \text{και η }f\text{ είναι αύξουσα},\\
f'(x)<0, & x>2, & \text{και η }f\text{ είναι φθίνουσα}.
\end{array}
\]
Άρα η μπλε καμπύλη είναι η $C_f$ και η μωβ ευθεία είναι η $C_{f'}$.

\item Από το γράφημα ή από τους τύπους των συναρτήσεων διαβάζουμε
\[
f(0)=-1
\qquad\text{και}\qquad
f'(0)=4.
\]
Η εφαπτομένη της $C_f$ στο σημείο
\[
A(0,f(0))=A(0,-1)
\]
έχει εξίσωση
\begin{align*}
y-f(0)&=f'(0)(x-0)\\
y+1&=4x.
\end{align*}
Επομένως
\[
{y=4x-1}.
\]

\item Για να υπολογίσουμε το εμβαδόν, πρέπει πρώτα να βρούμε ποια
γραφική παράσταση βρίσκεται ψηλότερα στο $[1,3]$. Έχουμε
\begin{align*}
f(x)-f'(x)
&=-x^2+4x-1-(-2x+4)\\
&=-x^2+6x-5\\
&=-(x-1)(x-5).
\end{align*}
Για κάθε $x\in(1,3]$ ισχύει
\[
x-1>0
\qquad\text{και}\qquad
x-5<0,
\]
οπότε
\[
f(x)-f'(x)>0.
\]
Άρα η $C_f$ βρίσκεται πάνω από τη $C_{f'}$ στο $[1,3]$, ενώ οι δύο
καμπύλες συναντώνται στο αριστερό άκρο $x=1$.

Στο παρακάτω σχήμα φαίνεται η διάταξη των δύο καμπυλών στο ζητούμενο
διάστημα:
\begin{center}
\begin{tikzpicture}[scale=0.72]
  \draw[help lines,gray!25] (-1,-3) grid (4,5);
  \draw[-latex] (-1.3,0) -- (4.3,0) node[right] {$x$};
  \draw[-latex] (0,-3.3) -- (0,5.3) node[above] {$y$};

  \fill[yellow!25,opacity=0.65]
    (1,2)
    -- plot[domain=1:3,samples=50] (\x,{-(\x)*(\x)+4*(\x)-1})
    -- (3,-2)
    -- plot[domain=3:1,samples=2] (\x,{-2*(\x)+4})
    -- cycle;

  \draw[very thick,maincolor,domain=-0.4:4,samples=70]
    plot (\x,{-(\x)*(\x)+4*(\x)-1});
  \draw[very thick,secondcolor,domain=-0.4:3.5,samples=2]
    plot (\x,{-2*(\x)+4});

  \draw[dashed] (1,0) -- (1,2);
  \draw[dashed] (3,-2) -- (3,2);
  \filldraw[black] (1,2) circle (1.8pt);

  \node[maincolor,fill=white,inner sep=1pt] at (3.45,3.1) {$C_f$};
  \node[secondcolor,fill=white,inner sep=1pt] at (3.25,-2.45) {$C_{f'}$};
  \node[font=\footnotesize,fill=yellow!15,inner sep=1pt] at (2,0.9)
    {$f(x)>f'(x)$};

  \foreach \x in {1,2,3}
    \draw (\x,0.1) -- (\x,-0.1)
      node[below,font=\footnotesize,fill=white,inner sep=1pt] {$\x$};
\end{tikzpicture}
\end{center}

Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_1^3\left[f(x)-f'(x)\right]\d x\\
&=\int_1^3\left(-x^2+6x-5\right)\d x\\
&=\left.-\frac{x^3}{3}+3x^2-5x\right|_1^3\\
&=3-\left(-\frac{7}{3}\right)\\
&={\frac{16}{3}}.
\end{align*}

\item Για κάθε $x\in[-1,4]$ είναι
\[
g(x)=f(x)-f'(x)=-x^2+6x-5.
\]
Η $g$ είναι συνεχής ως πολυωνυμική. Επιπλέον,
\[
g(0)=-5<0
\qquad\text{και}\qquad
g(2)=3>0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα
$\xi\in(0,2)$ τέτοιο ώστε $g(\xi)=0$.

Για τη μοναδικότητα υπολογίζουμε
\[
g'(x)=-2x+6.
\]
Έχουμε:
\begin{itemize}
\item $g'(x)>0$ για $x\in(-1,3)$, άρα η $g$ είναι γνησίως αύξουσα
στο $[-1,3]$. Επομένως η ρίζα που εξασφαλίστηκε με το θεώρημα
\eng{Bolzano} είναι η μοναδική ρίζα της $g$ στο διάστημα αυτό.

\item $g'(x)<0$ για $x\in(3,4)$, άρα η $g$ είναι γνησίως φθίνουσα
στο $[3,4]$. Επειδή
\[
g(4)=3>0,
\]
για κάθε $x\in[3,4]$ ισχύει $g(x)\geq g(4)>0$. Συνεπώς η $g$ δεν
έχει ρίζα στο $[3,4]$.
\end{itemize}

Άρα η εξίσωση $g(x)=0$ έχει ακριβώς μία ρίζα στο $(-1,4)$.
Μάλιστα,
\begin{align*}
g(x)=0
&\Leftrightarrow -x^2+6x-5=0\\
&\Leftrightarrow (x-1)(x-5)=0\\
&\Leftrightarrow x=1\ \text{ή}\ x=5.
\end{align*}
Από τις δύο λύσεις μόνο η $x=1$ ανήκει στο $(-1,4)$. Επομένως η
μοναδική ζητούμενη ρίζα είναι
\[
{x=1}.
\]
\end{erwthma}
\end{thema}
