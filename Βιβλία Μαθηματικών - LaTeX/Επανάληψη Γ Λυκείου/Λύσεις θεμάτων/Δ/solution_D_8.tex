\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\in[0,\pi]$ είναι $e^{-x}>0$. Επομένως
\begin{align*}
f(x)=0
&\Leftrightarrow e^{-x}\hm x=0\\
&\Leftrightarrow \hm x=0\\
&\Leftrightarrow x=0\ \text{ή}\ x=\pi.
\end{align*}
Άρα οι ρίζες της $f$ στο διάστημα $[0,\pi]$ είναι
\[
{x=0\quad\text{και}\quad x=\pi}.
\]

\item Η $f$ είναι συνεχής στο $[0,\pi]$ και παραγωγίσιμη στο $(0,\pi)$. Για κάθε $x\in(0,\pi)$ είναι
\begin{align*}
f'(x)
&=(e^{-x}\hm x)'\\
&=-e^{-x}\hm x+e^{-x}\syn x\\
&=e^{-x}(\syn x-\hm x).
\end{align*}
Επειδή $e^{-x}>0$, το πρόσημο της $f'$ είναι το πρόσημο της παράστασης $\syn x-\hm x$. Έχουμε
\begin{align*}
f'(x)=0
&\Leftrightarrow \syn x-\hm x=0\\
&\Leftrightarrow \syn x=\hm x\\
&\Leftrightarrow \ef x=1\\
&\Leftrightarrow x=\frac{\pi}{4},
\end{align*}
αφού $x\in(0,\pi)$.

Επίσης:
\begin{itemize}
\item $f'(x)>0$ για $0<x<\frac{\pi}{4}$,
\item $f'(x)<0$ για $\frac{\pi}{4}<x<\pi$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1.1}%
{$0$,$\frac{\pi}{4}$,$\pi$}%
\tkzTabLine{, +, z, -, }
\tkzTabVar{-/{0},+/{\dfrac{\sqrt{2}}{2}e^{-\pi/4}},-/{0}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\left[0,\frac{\pi}{4}\right]$ και \textbf{γνησίως φθίνουσα} στο $\left[\frac{\pi}{4},\pi\right]$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=\frac{\pi}{4}$, με τιμή
\[
f\left(\frac{\pi}{4}\right)
=e^{-\pi/4}\hm\frac{\pi}{4}
={e^{-\pi/4}\frac{\sqrt{2}}{2}}.
\]
Επίσης παρουσιάζει ολικό ελάχιστο στα άκρα $x=0$ και $x=\pi$, με τιμή
\[
f(0)=f(\pi)=0.
\]

\item Το σημείο $x_0=\frac{\pi}{4}$ ανήκει στο εσωτερικό $(0,\pi)$ του διαστήματος και, όπως αποδείξαμε, η $f$ παρουσιάζει σε αυτό ολικό μέγιστο. Επομένως, για κάθε $x\in[0,\pi]$, ισχύει
\[
f(x)\leq f\left(\frac{\pi}{4}\right),
\]
δηλαδή
\[
{f(x)\leq e^{-\pi/4}\frac{\sqrt{2}}{2}}.
\]
Η ισότητα ισχύει μόνο για $x=\frac{\pi}{4}$.

\item Θέτουμε
\[
I=\int_0^\pi e^{-x}\hm x\,\d x.
\]
Ολοκληρώνοντας κατά παράγοντες, με
\[
u=\hm x,\quad \d v=e^{-x}\d x,
\]
παίρνουμε
\[
\d u=\syn x\,\d x,\quad v=-e^{-x}.
\]
Άρα
\begin{align*}
I
&=\left[-e^{-x}\hm x\right]_0^\pi
+\int_0^\pi e^{-x}\syn x\,\d x\\
&=\int_0^\pi e^{-x}\syn x\,\d x.
\end{align*}
Ολοκληρώνουμε ξανά κατά παράγοντες, θέτοντας
\[
u=\syn x,\quad \d v=e^{-x}\d x.
\]
Τότε
\[
\d u=-\hm x\,\d x,\quad v=-e^{-x},
\]
οπότε
\begin{align*}
I
&=\left[-e^{-x}\syn x\right]_0^\pi
-\int_0^\pi e^{-x}\hm x\,\d x\\
&=\left[-e^{-x}\syn x\right]_0^\pi-I.
\end{align*}
Επομένως
\begin{align*}
2I
&=\left[-e^{-x}\syn x\right]_0^\pi\\
&=-e^{-\pi}\syn\pi+\syn0\\
&=e^{-\pi}+1.
\end{align*}
Άρα
\[
{\int_0^\pi f(x)\d x=\frac{1+e^{-\pi}}{2}}.
\]
\end{erwthma}
\end{thema}
