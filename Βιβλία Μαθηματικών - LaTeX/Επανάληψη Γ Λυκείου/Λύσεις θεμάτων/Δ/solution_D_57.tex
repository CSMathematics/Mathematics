\begin{thema}{Δ}
\begin{erwthma}
\item Για να είναι η $P$ συνεχής στο $t=10$, πρέπει
\[
\lim_{t\to10^-}P(t)
=\lim_{t\to10^+}P(t)
=P(10).
\]
Έχουμε
\[
P(10)=1000e^5
\]
και
\[
\lim_{t\to10^+}P(t)=Ae^{-5}.
\]
Άρα
\begin{align*}
Ae^{-5}&=1000e^5\\
\Leftrightarrow
{A=1000e^{10}}.
\end{align*}

\item Για $0<t<10$ είναι
\[
P'(t)=500e^{t/2},
\]
οπότε
\[
P'_-(10)=500e^5.
\]

Για $t>10$, με $A=1000e^{10}$, είναι
\[
P'(t)=-\frac{A}{2}e^{-t/2},
\]
οπότε
\[
P'_+(10)
=-\frac{1000e^{10}}{2}e^{-5}
=-500e^5.
\]
Επειδή
\[
P'_-(10)\neq P'_+(10),
\]
η $P$ δεν είναι παραγωγίσιμη στο $t=10$.

Γεωμετρικά, η γραφική παράσταση της $P$ έχει στο σημείο
\[
\left(10,1000e^5\right)
\]
γωνιακό σημείο. Ο πληθυσμός αυξάνεται μέχρι τη χρονική στιγμή $t=10$ και αμέσως μετά αρχίζει να μειώνεται.

\item Για $t>10$ είναι
\[
P(t)=1000e^{10}e^{-t/2}.
\]
Ζητούμε $t_0>10$ τέτοιο ώστε
\[
P(t_0)=1000.
\]
Άρα
\begin{align*}
1000e^{10}e^{-t_0/2}&=1000\\
\Leftrightarrow
e^{10-t_0/2}&=1\\
\Leftrightarrow
10-\frac{t_0}{2}&=0\\
\Leftrightarrow
{t_0=20}.
\end{align*}
Επομένως
\[
t_0=10+k
\]
με
\[
{k=10\ \text{ώρες}}.
\]

\item Έχουμε
\begin{align*}
\int_0^{10}P(t)\d t
&=\int_0^{10}1000e^{t/2}\d t\\
&=\left[2000e^{t/2}\right]_0^{10}\\
&={2000(e^5-1)}.
\end{align*}

Για κάθε $s\in[0,10]$ έχουμε
\begin{align*}
P(10-s)
&=1000e^{(10-s)/2}\\
&=1000e^{5-s/2}
\end{align*}
και
\begin{align*}
P(10+s)
&=1000e^{10}e^{-(10+s)/2}\\
&=1000e^{5-s/2}.
\end{align*}
Άρα
\[
{P(10-s)=P(10+s)}
\]
για κάθε $s\in[0,10]$.

Επομένως η γραφική παράσταση της $P$ στο διάστημα $[0,20]$ είναι συμμετρική ως προς την ευθεία $t=10$. Συνεπώς τα εμβαδά κάτω από τη γραφική παράσταση στα διαστήματα $[0,10]$ και $[10,20]$ είναι ίσα:
\[
{
\int_0^{10}P(t)\d t
=\int_{10}^{t_0}P(t)\d t}.
\]
Άρα και το δεύτερο ολοκλήρωμα είναι ίσο με
\[
2000(e^5-1).
\]
\end{erwthma}
\end{thema}
