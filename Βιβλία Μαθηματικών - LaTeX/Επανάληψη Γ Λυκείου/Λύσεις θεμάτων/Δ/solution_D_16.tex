%=========== ΛΥΣΗ - 16ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για τη συνέχεια της $f$ στο $x_0=1$ υπολογίζουμε
\begin{align*}
\lim_{x\to1}f(x)
&=\lim_{x\to1}\frac{x\ln x}{x-1}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{\ln x+1}{1}\\
&=1.
\end{align*}
Επειδή
\[
\lim_{x\to1}f(x)=1=f(1),
\]
η $f$ είναι συνεχής στο $x_0=1$.

\item Από τον ορισμό της παραγώγου έχουμε
\begin{align*}
f'(1)
&=\lim_{x\to1}\frac{f(x)-f(1)}{x-1}\\
&=\lim_{x\to1}
\frac{\frac{x\ln x}{x-1}-1}{x-1}\\
&=\lim_{x\to1}
\frac{x\ln x-x+1}{(x-1)^2}.
\end{align*}
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
f'(1)
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{\ln x}{2(x-1)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{\frac{1}{x}}{2}\\
&=\frac{1}{2}.
\end{align*}
Άρα η $f$ είναι παραγωγίσιμη στο $x_0=1$ και
\[
{f'(1)=\frac{1}{2}}.
\]

\item Έχουμε
\begin{align*}
\lim_{x\to0^+}f(x)
&=\lim_{x\to0^+}\frac{x\ln x}{x-1}.
\end{align*}
Επειδή
\begin{align*}
\lim_{x\to0^+}x\ln x
&=\lim_{x\to0^+}\frac{\ln x}{\frac{1}{x}}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to0^+}\frac{\frac{1}{x}}{-\frac{1}{x^2}}\\
&=\lim_{x\to0^+}(-x)\\
&=0,
\end{align*}
προκύπτει
\[
\lim_{x\to0^+}f(x)=\frac{0}{-1}=0.
\]
Επομένως, αν ορίσουμε
\[
f(0)=0,
\]
η $f$ επεκτείνεται συνεχώς στο $x=0$. Άρα η ευθεία $x=0$ δεν είναι κατακόρυφη ασύμπτωτη της $C_f$.

\item Για $x>0$ με $x\neq1$ είναι
\begin{align*}
f'(x)
&=\left(\frac{x\ln x}{x-1}\right)'\\
&=\frac{(\ln x+1)(x-1)-x\ln x}{(x-1)^2}\\
&=\frac{x-1-\ln x}{(x-1)^2}.
\end{align*}
Θεωρούμε τη βοηθητική συνάρτηση
\[
h(x)=x-1-\ln x,\quad x>0.
\]
Η $h$ είναι παραγωγίσιμη και
\[
h'(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Άρα η $h$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
h(1)=0.
\]
Συνεπώς
\[
h(x)>0
\]
για κάθε $x>0$ με $x\neq1$. Επειδή $(x-1)^2>0$, έχουμε
\[
f'(x)>0
\]
για κάθε $x>0$, $x\neq1$.

Από το δεύτερο ερώτημα είναι επίσης
\[
f'(1)=\frac{1}{2}>0.
\]
Άρα
\[
f'(x)>0
\]
για κάθε $x>0$, και επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,+\infty)$.
\end{erwthma}
\end{thema}
