\begin{thema}{Δ}
\begin{erwthma}
\item Η $g$ είναι παραγωγίσιμη στο $(1,+\infty)$ και για κάθε $y>1$ είναι
\[
g'(y)=\frac{1}{y-1}+1=\frac{y}{y-1}>0.
\]
Άρα η $g$ είναι \textbf{γνησίως αύξουσα} στο $(1,+\infty)$.

Από τη δοσμένη σχέση έχουμε
\[
g(f(0))=0.
\]
Επίσης
\[
g(2)=\ln1+2-2=0.
\]
Επειδή η $g$ είναι γνησίως αύξουσα, η εξίσωση $g(y)=0$ έχει το πολύ μία ρίζα. Επομένως
\[
{f(0)=2}.
\]

\item Για κάθε $x\in(-2,2)$ έχουμε
\[
f(x)f'(x)=-x.
\]
Άρα
\begin{align*}
\left[f^2(x)+x^2\right]'
&=2f(x)f'(x)+2x\\
&=-2x+2x\\
&=0.
\end{align*}
Επομένως υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
f^2(x)+x^2=c
\]
για κάθε $x\in(-2,2)$. Για $x=0$ έχουμε $f(0)=2$, οπότε
\[
c=4.
\]
Άρα
\[
f^2(x)=4-x^2.
\]

Για κάθε $x\in(-2,2)$ είναι $4-x^2>0$, άρα η $f$ δεν μηδενίζεται στο διάστημα αυτό. Επειδή η $f$ είναι συνεχής και $f(0)=2>0$, διατηρεί θετικό πρόσημο στο $(-2,2)$. Συνεπώς
\[
f(x)=\sqrt{4-x^2}
\]
για κάθε $x\in(-2,2)$.

Τέλος, λόγω της συνέχειας της $f$ στο $[-2,2]$,
\[
f(\pm2)=\lim_{x\to\pm2}f(x)=0.
\]
Επομένως
\[
{f(x)=\sqrt{4-x^2}},\quad x\in[-2,2].
\]

\item Το ορθογώνιο έχει βάση
\[
K\Lambda=2x
\]
και ύψος
\[
\Lambda M=f(x)=\sqrt{4-x^2}.
\]
Άρα το εμβαδόν του είναι
\[
E(x)=2x\sqrt{4-x^2},\quad x\in(0,2).
\]
Επειδή $E(x)>0$, η $E$ μεγιστοποιείται στο ίδιο σημείο με την $E^2$. Έχουμε
\[
E^2(x)=4x^2(4-x^2).
\]
Θεωρούμε τη συνάρτηση
\[
H(x)=x^2(4-x^2),\quad x\in(0,2).
\]
Για κάθε $x\in(0,2)$ είναι
\[
H'(x)=4x(2-x^2).
\]
Επομένως:
\begin{itemize}
\item $H'(x)>0$ για $0<x<\sqrt2$,
\item $H'(\sqrt2)=0$,
\item $H'(x)<0$ για $\sqrt2<x<2$.
\end{itemize}
Άρα το εμβαδόν του ορθογωνίου γίνεται μέγιστο για
\[
{x=\sqrt2}.
\]
Η μέγιστη τιμή του είναι
\[
E(\sqrt2)=2\sqrt2\sqrt2={4}.
\]

\item Η $f$ είναι άρτια, οπότε η συνάρτηση
\[
h(x)=x^3f(x)
\]
είναι περιττή. Επομένως
\[
{\int_{-2}^{2}x^3f(x)\d x=0}.
\]

Για το ζητούμενο όριο έχουμε
\begin{align*}
\lim_{x\to0}\frac{f(x)-2}{x\hm x}
&=\lim_{x\to0}
\frac{\sqrt{4-x^2}-2}{x\hm x}\\
&=\lim_{x\to0}
\frac{-x^2}
{x\hm x\left(\sqrt{4-x^2}+2\right)}\\
&=-\lim_{x\to0}
\left[
\frac{x}{\hm x}\cdot
\frac{1}{\sqrt{4-x^2}+2}
\right]\\
&=-1\cdot\frac14\\
&={-\frac14}.
\end{align*}
\end{erwthma}
\end{thema}
