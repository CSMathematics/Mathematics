%=========== ΛΥΣΗ - 50ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
    \item Η συνάρτηση $f(x) = x \ln x$ με πεδίο ορισμού το $(0, +\infty)$ είναι παραγωγίσιμη ως γινόμενο παραγωγίσιμων συναρτήσεων με παράγωγο:
    \[f'(x) = (x)' \ln x + x (\ln x)' = \ln x + x \cdot \frac{1}{x} = \ln x + 1\]
    Βρίσκουμε τις ρίζες και το πρόσημο της $f'$:
    \begin{itemize}
        \item $f'(x) = 0 \Leftrightarrow \ln x + 1 = 0 \Leftrightarrow \ln x = -1 \Leftrightarrow x = e^{-1} = \frac{1}{e}$
        \item $f'(x) > 0 \Leftrightarrow \ln x + 1 > 0 \Leftrightarrow \ln x > -1 \Leftrightarrow x > \frac{1}{e}$
        \item $f'(x) < 0 \Leftrightarrow \ln x + 1 < 0 \Leftrightarrow \ln x < -1 \Leftrightarrow 0 < x < \frac{1}{e}$
    \end{itemize}
    Κατασκευάζουμε τον πίνακα μονοτονίας της $f$:
    \begin{center}
    \begin{tikzpicture}
    \tkzTabInit[color,lgt=2,espcl=2,colorC = red7,colorL = red9,colorV = red7]%
    {$x$ / .8, $f'(x)$ / .8, $f(x)$ / 1.2}%
    {$0$, $\frac{1}{e}$, $+\infty$}%
    \tkzTabLine{d, -, z, +, }
    \tkzTabLine{d, \searrow, t, \nearrow, }
    \end{tikzpicture}
    \end{center}
    Όπως φαίνεται από τον πίνακα, η $f$ είναι \textbf{γνησίως φθίνουσα} στο διάστημα $\left(0, \frac{1}{e}\right]$ και \textbf{γνησίως αύξουσα} στο $\left[\frac{1}{e}, +\infty\right)$.
    Επομένως, παρουσιάζει ολικό ελάχιστο στο $x = \frac{1}{e}$ με τιμή:
    \[f\left(\frac{1}{e}\right) = \frac{1}{e} \ln\left(\frac{1}{e}\right) = \frac{1}{e} \cdot (-1) = -\frac{1}{e}\]
    Άρα, για κάθε $x>0$ ισχύει $f(x) \ge f\left(\frac{1}{e}\right) \Rightarrow f(x) \ge -\frac{1}{e}$.

    \item Δίνεται ότι η συνάρτηση $g$ είναι παραγωγίσιμη με $g'(x) = f(x) = x \ln x$ για κάθε $x>0$.
    Παρατηρούμε ότι:
    \[g'(1) = 1 \cdot \ln 1 = 0\]
    Επίσης:
    \begin{itemize}
        \item Για $0 < x < 1$ είναι $x > 0$ και $\ln x < 0$, άρα $x \ln x < 0 \Leftrightarrow g'(x) < 0$.
        \item Για $x > 1$ είναι $x > 0$ και $\ln x > 0$, άρα $x \ln x > 0 \Leftrightarrow g'(x) > 0$.
    \end{itemize}
    Συνεπώς η $g$ είναι \textbf{γνησίως φθίνουσα} στο $(0, 1]$ και \textbf{γνησίως αύξουσα} στο $[1, +\infty)$.
    Εφόσον η $g'$ μηδενίζεται στο $x=1$ και εκατέρωθεν αλλάζει πρόσημο από αρνητικό σε θετικό, η συνάρτηση $g$ παρουσιάζει τοπικό ελάχιστο στο $x=1$.

    \item Για κάθε $x>0$ έχουμε:
    \begin{align*}
        g'(x) &= x \ln x \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2}\right)' \ln x \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2} \ln x\right)' - \frac{x^2}{2}(\ln x)' \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2} \ln x\right)' - \frac{x^2}{2} \cdot \frac{1}{x} \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2} \ln x\right)' - \frac{x}{2} \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2} \ln x\right)' - \left(\frac{x^2}{4}\right)' \\
        \Leftrightarrow g'(x) &= \left(\frac{x^2}{2} \ln x - \frac{x^2}{4}\right)'
    \end{align*}
    Σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c \in \mathbb{R}$ τέτοια ώστε:
    \[g(x) = \frac{x^2}{2} \ln x - \frac{x^2}{4} + c \ ,\ \text{για κάθε } x>0\]
    Η γραφική παράσταση της $g$ διέρχεται από το σημείο $\left(1, -\frac{1}{4}\right)$, οπότε:
    \[g(1) = -\frac{1}{4} \Leftrightarrow \frac{1^2}{2} \ln 1 - \frac{1^2}{4} + c = -\frac{1}{4} \Leftrightarrow -\frac{1}{4} + c = -\frac{1}{4} \Leftrightarrow c = 0\]
    Επομένως, ο ακριβής τύπος της $g$ είναι:
    \[g(x) = \frac{x^2}{2} \ln x - \frac{x^2}{4} \ ,\ x>0\]

    \item Έχουμε να υπολογίσουμε το ολοκλήρωμα $\dint_1^e g(x) \,\d x$. Με βάση τον τύπο της $g$ έχουμε:
    \begin{align*}
        \dint_1^e g(x) \,\d x &= \dint_1^e \left(\frac{x^2}{2} \ln x - \frac{x^2}{4}\right) \,\d x \\
        &= \dint_1^e \frac{x^2}{2} \ln x \,\d x - \dint_1^e \frac{x^2}{4} \,\d x \\
        &= \dint_1^e \left(\frac{x^3}{6}\right)' \ln x \,\d x - \left[\frac{x^3}{12}\right]_1^e \\
        &= \left[\frac{x^3}{6} \ln x\right]_1^e - \dint_1^e \frac{x^3}{6} (\ln x)' \,\d x - \left(\frac{e^3}{12} - \frac{1}{12}\right) \\
        &= \left(\frac{e^3}{6} \ln e - \frac{1^3}{6} \ln 1\right) - \dint_1^e \frac{x^3}{6} \cdot \frac{1}{x} \,\d x - \frac{e^3-1}{12} \\
        &= \frac{e^3}{6} - \dint_1^e \frac{x^2}{6} \,\d x - \frac{e^3-1}{12} \\
        &= \frac{e^3}{6} - \left[\frac{x^3}{18}\right]_1^e - \frac{e^3-1}{12} \\
        &= \frac{e^3}{6} - \left(\frac{e^3}{18} - \frac{1}{18}\right) - \frac{e^3-1}{12} \\
        &= \frac{3e^3}{18} - \frac{e^3}{18} + \frac{1}{18} - \frac{e^3-1}{12} \\
        &= \frac{2e^3+1}{18} - \frac{e^3-1}{12} \\
        &= \frac{2(2e^3+1)}{36} - \frac{3(e^3-1)}{36} \\
        &= \frac{4e^3+2-3e^3+3}{36} \\
        &= \frac{e^3+5}{36}
    \end{align*}
    Επειδή $e > 0$, είναι προφανές ότι $e^3+5 > 0$, άρα προκύπτει $\frac{e^3+5}{36} > 0$.
    Επομένως, δείξαμε ότι $\dint_1^e g(x) \,\d x > 0$.
\end{erwthma}
\end{thema}
