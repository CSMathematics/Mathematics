%=========== ΛΥΣΗ - 78ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Από τη γραφική παράσταση της $f$ παρατηρούμε ότι η $C_f$ τέμνει
τον άξονα $x'x$ στο σημείο με τετμημένη
\[
x=3.
\]
Επίσης η $C_f$ βρίσκεται πάνω από τον άξονα $x'x$ στο διάστημα
$[-1,3)$ και κάτω από αυτόν στο διάστημα $(3,5]$.

Άρα
\[
f(x)>0
\Leftrightarrow
x\in[-1,3).
\]
Επομένως η λύση της ανίσωσης $f(x)>0$ στο $[-1,5]$ είναι
\[
{x\in[-1,3)}.
\]

\item Από τη γραφική παράσταση της $f$ βλέπουμε ότι η μέγιστη τιμή της
είναι
\[
f(1)=4.
\]
Άρα, για κάθε
\[
x\in[-1,5],
\]
ισχύει
\[
f(x)\leq 4.
\]
Δηλαδή
\[
{f(x)\leq 4,\quad x\in[-1,5]}.
\]

\item Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)+f'(x),\quad x\in[1,3].
\]
Επειδή η $f$ έχει συνεχή παράγωγο, οι $f$ και $f'$ είναι συνεχείς, άρα
και η $h$ είναι συνεχής στο $[1,3]$.

Από το σχήμα έχουμε
\[
f(1)=4
\quad\text{και}\quad
f'(1)=0.
\]
Άρα
\[
h(1)=f(1)+f'(1)=4+0=4>0.
\]

Επίσης
\[
f(3)=0
\quad\text{και}\quad
f'(3)<0.
\]
Άρα
\[
h(3)=f(3)+f'(3)=f'(3)<0.
\]

Επομένως
\[
h(1)>0
\quad\text{και}\quad
h(3)<0.
\]
Από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\xi\in(1,3)
\]
τέτοιο ώστε
\[
h(\xi)=0.
\]
Δηλαδή
\[
f(\xi)+f'(\xi)=0.
\]
Άρα η εξίσωση
\[
f(x)+f'(x)=0
\]
έχει τουλάχιστον μία ρίζα στο διάστημα $(1,3)$.

\item Από το β΄ ερώτημα γνωρίζουμε ότι
\[
f(x)\leq 4,\quad x\in[-1,5].
\]
Μάλιστα η ανισότητα είναι γνήσια σε υποδιάστημα, αφού
\[
f(5)=-2<4.
\]
Λόγω συνέχειας της $f$, υπάρχει διάστημα κοντά στο $5$ στο οποίο
\[
f(x)<4.
\]
Άρα
\[
\int_{-1}^{5} f(x)\d x
<
\int_{-1}^{5} 4\d x.
\]
Επειδή
\[
\int_{-1}^{5}4\d x=4\cdot\bigl(5-(-1)\bigr)=24,
\]
παίρνουμε
\[
{\int_{-1}^{5} f(x)\d x<24}.
\]
\end{erwthma}
\end{thema}
