\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και για κάθε $x>0$ είναι
\begin{align*}
f'(x)
&=\left(\frac{\ln x}{x}\right)'\\
&=\frac{\frac{1}{x}\cdot x-\ln x}{x^2}\\
&=\frac{1-\ln x}{x^2}.
\end{align*}
Επειδή $x^2>0$, το πρόσημο της $f'$ είναι το πρόσημο της παράστασης $1-\ln x$. Έχουμε:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow \ln x=1\Leftrightarrow x=e$,
\item $f'(x)>0\Leftrightarrow \ln x<1\Leftrightarrow 0<x<e$,
\item $f'(x)<0\Leftrightarrow \ln x>1\Leftrightarrow x>e$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$0$,$e$,$+\infty$}%
\tkzTabLine{d, +, z, -, }
\tkzTabVar{D-/{-\infty},+/{\frac{1}{e}},-/{0}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,e]$ και \textbf{γνησίως φθίνουσα} στο $[e,+\infty)$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=e$, με τιμή
\[
{f(e)=\frac{\ln e}{e}=\frac{1}{e}}.
\]

Για τις ασύμπτωτες της $C_f$ έχουμε
\[
\lim_{x\to0^+}\frac{\ln x}{x}=-\infty,
\]
οπότε η ευθεία
\[
{x=0}
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Επίσης
\[
\lim_{x\to+\infty}\frac{\ln x}{x}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{\frac{1}{x}}{1}
=0.
\]
Άρα η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

\item Από το προηγούμενο ερώτημα, για κάθε $x>0$ ισχύει
\[
f(x)\leq f(e).
\]
Επομένως
\begin{align*}
\frac{\ln x}{x}
&\leq\frac{1}{e}\\
\Leftrightarrow e\ln x&\leq x\\
\Leftrightarrow \ln x^e&\leq\ln e^x\\
\Leftrightarrow {x^e\leq e^x}.
\end{align*}
Η ισότητα ισχύει μόνο για $x=e$.

Επειδή $\pi>e$ και η $f$ είναι γνησίως φθίνουσα στο $[e,+\infty)$, έχουμε
\[
f(\pi)<f(e).
\]
Άρα
\begin{align*}
\frac{\ln\pi}{\pi}
&<\frac{1}{e}\\
\Leftrightarrow e\ln\pi&<\pi\\
\Leftrightarrow \ln\pi^e&<\ln e^\pi\\
\Leftrightarrow {\pi^e<e^\pi}.
\end{align*}

\item Έστω $a>\beta\geq e$. Επειδή η $f$ είναι γνησίως φθίνουσα στο $[e,+\infty)$, έχουμε
\[
a>\beta\Rightarrow f(a)<f(\beta).
\]
Επομένως
\begin{align*}
\frac{\ln a}{a}
&<\frac{\ln\beta}{\beta}\\
\Leftrightarrow \beta\ln a&<a\ln\beta\\
\Leftrightarrow \ln a^\beta&<\ln\beta^a\\
\Leftrightarrow {\beta^a>a^\beta}.
\end{align*}

\item Για κάθε $x>0$ γράφουμε
\[
f(x)\hm x
=\frac{\ln x}{x}\hm x
=\ln x\cdot\frac{\hm x}{x}.
\]
Επειδή
\[
\lim_{x\to0^+}\ln x=-\infty
\quad\text{και}\quad
\lim_{x\to0^+}\frac{\hm x}{x}=1,
\]
προκύπτει
\[
{\lim_{x\to0^+}\left(f(x)\hm x\right)=-\infty}.
\]
\end{erwthma}
\end{thema}
