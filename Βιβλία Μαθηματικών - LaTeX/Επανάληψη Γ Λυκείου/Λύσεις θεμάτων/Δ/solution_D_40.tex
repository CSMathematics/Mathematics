\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
\left(e^{-x}f(x)\right)'
&=-e^{-x}f(x)+e^{-x}f'(x)\\
&=e^{-x}\left[f'(x)-f(x)\right]\\
&=1.
\end{align*}
Επομένως
\[
e^{-x}f(x)=x+c,
\]
όπου $c\in\mathbb{R}$. Επειδή $f(0)=0$, έχουμε
\[
0=c.
\]
Άρα
\[
{f(x)=xe^x},\quad x\in\mathbb{R}.
\]

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=(x+1)e^x.
\]
Επειδή $e^x>0$, το πρόσημο της $f'$ είναι το πρόσημο της παράστασης $x+1$. Έχουμε:
\begin{itemize}
\item $f'(x)<0$ για $x<-1$,
\item $f'(-1)=0$,
\item $f'(x)>0$ για $x>-1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$-\infty$,$-1$,$+\infty$}%
\tkzTabLine{, -, z, +, }
\tkzTabVar{+/{0},-/{-\frac{1}{e}},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,-1]$ και \textbf{γνησίως αύξουσα} στο $[-1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=-1$, με τιμή
\[
{f(-1)=-\frac{1}{e}}.
\]
Δεν παρουσιάζει μέγιστο. Συνεπώς το σύνολο τιμών της είναι
\[
{f(\mathbb{R})=
\left[-\frac{1}{e},+\infty\right)}.
\]

\item Έστω ότι η εφαπτομένη της $C_f$ στο σημείο
\[
M\left(x_0,f(x_0)\right)
\]
διέρχεται από την αρχή των αξόνων. Η εξίσωσή της είναι
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]
Για $x=0$ και $y=0$ παίρνουμε
\[
f(x_0)=x_0f'(x_0).
\]
Άρα
\begin{align*}
x_0e^{x_0}
&=x_0(x_0+1)e^{x_0}\\
\Leftrightarrow
x_0&=x_0(x_0+1)\\
\Leftrightarrow
x_0^2&=0\\
\Leftrightarrow
x_0&=0.
\end{align*}
Επομένως υπάρχει ακριβώς μία τέτοια εφαπτομένη. Επειδή
\[
f(0)=0
\quad\text{και}\quad
f'(0)=1,
\]
η εξίσωσή της είναι
\[
{y=x}.
\]

\item Η δοσμένη εξίσωση γράφεται
\[
(x^2+1)e^{x^2}=(x+1)e^x.
\]
Επειδή το αριστερό μέλος είναι θετικό, πρέπει
\[
x+1>0
\Leftrightarrow
x>-1.
\]
Για κάθε $t\in\mathbb{R}$ έχουμε
\[
f'(t)=(t+1)e^t.
\]
Άρα η εξίσωση είναι ισοδύναμη με
\[
f'(x^2)=f'(x).
\]
Επίσης
\[
f''(t)=(t+2)e^t>0
\]
για κάθε $t>-1$. Επομένως η $f'$ είναι γνησίως αύξουσα στο $(-1,+\infty)$. Επειδή $x>-1$ και $x^2\geq0$, προκύπτει
\begin{align*}
f'(x^2)=f'(x)
&\Leftrightarrow x^2=x\\
&\Leftrightarrow x(x-1)=0\\
&\Leftrightarrow {x=0\ \text{ή}\ x=1}.
\end{align*}

Για κάθε $x>0$ έχουμε
\[
x>\ln(x+1).
\]
Πράγματι, η συνάρτηση
\[
h(x)=x-\ln(x+1),\quad x\geq0,
\]
έχει
\[
h'(x)=1-\frac{1}{x+1}=\frac{x}{x+1}>0
\]
για κάθε $x>0$ και $h(0)=0$.

Επειδή $x>0$ και $\ln(x+1)>0$, ενώ η $f$ είναι γνησίως αύξουσα στο $[-1,+\infty)$, έχουμε
\[
x>\ln(x+1)
\Rightarrow
f(x)>f(\ln(x+1)).
\]
Άρα
\begin{align*}
xe^x
&>\ln(x+1)e^{\ln(x+1)}\\
&=(x+1)\ln(x+1).
\end{align*}
Επομένως
\[
{xe^x>(x+1)\ln(x+1)}
\]
για κάθε $x>0$.
\end{erwthma}
\end{thema}
