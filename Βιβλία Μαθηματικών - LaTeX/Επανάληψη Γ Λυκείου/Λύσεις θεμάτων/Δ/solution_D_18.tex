\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f'(x)
&=\left[(x-2)e^x+x+2\right]'\\
&=e^x+(x-2)e^x+1\\
&=(x-1)e^x+1.
\end{align*}
Για τη μελέτη του προσήμου της $f'$ υπολογίζουμε
\[
f''(x)=xe^x.
\]
Επειδή $e^x>0$, έχουμε:
\begin{itemize}
\item $f''(x)<0$ για $x<0$,
\item $f''(0)=0$,
\item $f''(x)>0$ για $x>0$.
\end{itemize}
Άρα η $f'$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
f'(0)=-1+1=0.
\]
Συνεπώς
\[
f'(x)\geq0
\]
για κάθε $x\in\mathbb{R}$, με την ισότητα να ισχύει μόνο για $x=0$. Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Επειδή
\[
f(0)=-2+2=0,
\]
η εξίσωση $f(x)=0$ έχει μοναδική λύση
\[
{x=0}.
\]
Τέλος, από τη γνήσια αύξηση της $f$ προκύπτει:
\[
{
\begin{cases}
f(x)<0, & x<0,\\
f(x)=0, & x=0,\\
f(x)>0, & x>0.
\end{cases}}
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f''(x)=xe^x.
\]
Η $f''$ αλλάζει πρόσημο μόνο στο $x=0$. Επομένως η $f$ είναι κοίλη στο $(-\infty,0]$, κυρτή στο $[0,+\infty)$ και η $C_f$ έχει ακριβώς ένα σημείο καμπής, το
\[
{O(0,0)}.
\]

Η κλίση της εφαπτομένης της $C_f$ στο σημείο με τετμημένη $x$ είναι $f'(x)$. Όπως δείξαμε, η $f'$ παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
f'(0)=0.
\]
Άρα η εφαπτομένη στο σημείο καμπής έχει την ελάχιστη δυνατή κλίση. Η εξίσωσή της είναι
\[
y-f(0)=f'(0)(x-0)
\Leftrightarrow
{y=0}.
\]

\item Για να δείξουμε ότι η ευθεία
\[
(\varepsilon):y=x+2
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$, υπολογίζουμε
\begin{align*}
\lim_{x\to-\infty}\left[f(x)-(x+2)\right]
&=\lim_{x\to-\infty}(x-2)e^x\\
&=\lim_{x\to-\infty}\frac{x-2}{e^{-x}}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}}
\lim_{x\to-\infty}\frac{1}{-e^{-x}}\\
&=0.
\end{align*}
Επομένως η ευθεία
\[
{y=x+2}
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$.

\item Για κάθε $x\in[-1,0]$ είναι
\[
f(x)-(x+2)=(x-2)e^x<0,
\]
αφού $x-2<0$ και $e^x>0$. Άρα η ασύμπτωτη $(\varepsilon)$ βρίσκεται πάνω από τη $C_f$ στο διάστημα $[-1,0]$.

Το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_{-1}^{0}\left[(x+2)-f(x)\right]\d x\\
&=\int_{-1}^{0}(2-x)e^x\d x\\
&=\left[(3-x)e^x\right]_{-1}^{0}\\
&=3-\frac{4}{e}.
\end{align*}
Επομένως
\[
{E=3-\frac{4}{e}}.
\]
\end{erwthma}
\end{thema}
