%=========== ΛΥΣΗ - 72ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Από το σχήμα της $C_{f'}$ έχουμε:
\[
f'(x)>0,\quad x\in[-2,0),
\]
\[
f'(0)=0,
\]
\[
f'(x)<0,\quad x\in(0,4),
\]
\[
f'(4)=0
\]
και
\[
f'(x)>0,\quad x\in(4,5].
\]

\begin{erwthma}
\item Από το πρόσημο της $f'$ προκύπτει ότι η $f$ είναι:
\[
\begin{array}{c}
\text{γνησίως αύξουσα στο }[-2,0],\\
\text{γνησίως φθίνουσα στο }[0,4],\\
\text{γνησίως αύξουσα στο }[4,5].
\end{array}
\]

Επομένως:
\begin{itemize}
\item στο $x=0$ η $f'$ αλλάζει πρόσημο από $+$ σε $-$, άρα η $f$
παρουσιάζει τοπικό μέγιστο,
\item στο $x=4$ η $f'$ αλλάζει πρόσημο από $-$ σε $+$, άρα η $f$
παρουσιάζει τοπικό ελάχιστο.
\end{itemize}

\item Έχουμε
\[
f(0)-f(-2)=\int_{-2}^{0}f'(x)\d x.
\]
Στο διάστημα $[-2,0)$ ισχύει
\[
f'(x)>0,
\]
άρα
\[
\int_{-2}^{0}f'(x)\d x>0.
\]
Επειδή
\[
f(-2)=0,
\]
παίρνουμε
\[
f(0)>0.
\]

Επίσης,
\[
f(4)-f(0)=\int_0^4 f'(x)\d x.
\]
Στο διάστημα $(0,4)$ ισχύει
\[
f'(x)<0,
\]
άρα
\[
\int_0^4 f'(x)\d x<0.
\]
Επομένως
\[
f(4)-f(0)<0
\]
και άρα
\[
{f(4)<f(0)}.
\]

\item Από το επιπλέον δεδομένο της εκφώνησης έχουμε
\[
\int_{-2}^{4}f'(x)\d x>0.
\]
Όμως
\[
f(4)-f(-2)=\int_{-2}^{4}f'(x)\d x.
\]
Άρα
\[
f(4)-f(-2)>0.
\]
Επειδή
\[
f(-2)=0,
\]
προκύπτει
\[
f(4)>0.
\]

Τώρα, επειδή η $f$ είναι γνησίως αύξουσα στο $[-2,0]$, έχουμε
\[
f(x)\geq f(-2)=0,\quad x\in[-2,0].
\]

Επειδή η $f$ είναι γνησίως φθίνουσα στο $[0,4]$, η ελάχιστη τιμή της
στο διάστημα αυτό είναι η $f(4)$. Άρα
\[
f(x)\geq f(4)>0,\quad x\in[0,4].
\]

Τέλος, επειδή η $f$ είναι γνησίως αύξουσα στο $[4,5]$, η ελάχιστη τιμή
της στο διάστημα αυτό είναι επίσης η $f(4)$. Άρα
\[
f(x)\geq f(4)>0,\quad x\in[4,5].
\]

Συνεπώς
\[
f(x)\geq 0,\quad x\in[-2,5].
\]
Άρα η λύση της ανίσωσης $f(x)\geq 0$ στο $[-2,5]$ είναι
\[
{x\in[-2,5]}.
\]

\item Από το προηγούμενο ερώτημα γνωρίζουμε ότι
\[
f(4)>0.
\]
Επειδή
\[
f(-2)=0<f(4)
\]
και, από το β΄ ερώτημα,
\[
f(4)<f(0),
\]
έχουμε
\[
f(-2)<f(4)<f(0).
\]

Η $f$ είναι συνεχής και γνησίως αύξουσα στο $[-2,0]$. Άρα υπάρχει
μοναδικό
\[
\alpha\in(-2,0)
\]
τέτοιο ώστε
\[
f(\alpha)=f(4).
\]

Επιπλέον, η ίδια η τιμή $x=4$ είναι προφανώς λύση της εξίσωσης
\[
f(x)=f(4).
\]

Θα δείξουμε ότι δεν υπάρχουν άλλες λύσεις.

Στο διάστημα $[0,4]$ η $f$ είναι γνησίως φθίνουσα. Άρα, για κάθε
$x\in[0,4)$, ισχύει
\[
f(x)>f(4).
\]
Επομένως στο $[0,4]$ η μοναδική λύση είναι η
\[
x=4.
\]

Στο διάστημα $[4,5]$ η $f$ είναι γνησίως αύξουσα. Άρα, για κάθε
$x\in(4,5]$, ισχύει
\[
f(x)>f(4).
\]
Επομένως στο $[4,5]$ η μοναδική λύση είναι πάλι η
\[
x=4.
\]

Άρα η εξίσωση
\[
f(x)=f(4)
\]
έχει ακριβώς δύο ρίζες στο $[-2,5]$, τις
\[
{\alpha\in(-2,0)\ \text{και}\ 4}.
\]
\end{erwthma}
\end{thema}
