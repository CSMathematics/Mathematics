%=========== ΛΥΣΗ - 39ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x>0$ έχουμε
\begin{align*}
\left(\frac{f(x)}{x}\right)'
&=\frac{xf'(x)-f(x)}{x^2}\\
&=\frac{x^2e^{-x}}{x^2}\\
&=e^{-x}.
\end{align*}
Επομένως
\[
\frac{f(x)}{x}=-e^{-x}+c,
\]
όπου $c\in\mathbb{R}$. Επειδή $f(1)=1$, έχουμε
\[
1=-\frac{1}{e}+c
\Leftrightarrow
c=1+\frac{1}{e}.
\]
Άρα
\[
{f(x)=x\left(1+\frac{1}{e}-e^{-x}\right)},\quad x>0.
\]

\item Για κάθε $x>0$ γράφουμε
\[
f(x)=\left(1+\frac{1}{e}\right)x-xe^{-x}.
\]
Έχουμε
\[
\lim_{x\to+\infty}
\left[
f(x)-\left(1+\frac{1}{e}\right)x
\right]
=\lim_{x\to+\infty}(-xe^{-x})
=0.
\]
Επομένως η ευθεία
\[
{(\varepsilon):\ y=\left(1+\frac{1}{e}\right)x}
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$.

\item Για κάθε $x>0$ είναι
\begin{align*}
f'(x)
&=1+\frac{1}{e}+\left(x-1\right)e^{-x}.
\end{align*}
και
\begin{align*}
f''(x)
&=\left[(x-1)e^{-x}\right]'\\
&=e^{-x}-(x-1)e^{-x}\\
&=(2-x)e^{-x}.
\end{align*}
Επειδή $e^{-x}>0$, έχουμε:
\begin{itemize}
\item $f''(x)>0$ για $0<x<2$,
\item $f''(2)=0$,
\item $f''(x)<0$ για $x>2$.
\end{itemize}
Άρα η $f$ είναι \textbf{κυρτή} στο $(0,2]$ και \textbf{κοίλη} στο $[2,+\infty)$. Επομένως η $C_f$ έχει σημείο καμπής
\[
K\left(2,f(2)\right),
\]
όπου
\[
f(2)=2\left(1+\frac{1}{e}-\frac{1}{e^2}\right).
\]
Άρα
\[
{
K\left(
2,\,
2+\frac{2}{e}-\frac{2}{e^2}
\right)}.
\]

Ο συντελεστής διεύθυνσης της ασύμπτωτης $(\varepsilon)$ είναι
\[
\lambda=1+\frac{1}{e}.
\]
Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $x_0>0$ είναι παράλληλη στην $(\varepsilon)$, αν και μόνο αν
\begin{align*}
f'(x_0)&=1+\frac{1}{e}\\
\Leftrightarrow
(x_0-1)e^{-x_0}&=0\\
\Leftrightarrow
x_0&=1.
\end{align*}
Άρα υπάρχει ακριβώς μία τέτοια εφαπτομένη. Επειδή
\[
f(1)=1
\quad\text{και}\quad
f'(1)=1+\frac{1}{e},
\]
η εξίσωσή της είναι
\begin{align*}
y-1
&=\left(1+\frac{1}{e}\right)(x-1),
\end{align*}
δηλαδή
\[
{y=\left(1+\frac{1}{e}\right)x-\frac{1}{e}}.
\]

\item Για κάθε $x>0$ έχουμε
\[
\left(1+\frac{1}{e}\right)x-f(x)=xe^{-x}>0.
\]
Άρα η ασύμπτωτη $(\varepsilon)$ βρίσκεται πάνω από τη $C_f$ και το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_1^2
\left[
\left(1+\frac{1}{e}\right)x-f(x)
\right]\d x\\
&=\int_1^2xe^{-x}\d x\\
&=\left[-(x+1)e^{-x}\right]_1^2\\
&={\frac{2}{e}-\frac{3}{e^2}}.
\end{align*}

Επίσης
\begin{align*}
\int_1^2f(x)\d x
&=\int_1^2
\left(1+\frac{1}{e}\right)x\d x-E\\
&=\frac{3}{2}\left(1+\frac{1}{e}\right)
-\left(\frac{2}{e}-\frac{3}{e^2}\right)\\
&=\frac{3}{2}-\frac{1}{2e}+\frac{3}{e^2}.
\end{align*}
Επομένως
\begin{align*}
\int_1^2f(x)\d x-\frac{3e+1}{2e}
&=\frac{3}{e^2}-\frac{1}{e}\\
&=\frac{3-e}{e^2}>0,
\end{align*}
αφού $e<3$. Συνεπώς
\[
{
\int_1^2f(x)\d x>\frac{3e+1}{2e}}.
\]
\end{erwthma}
\end{thema}
