\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f'(x)
&=2xe^{-x}-(x^2+1)e^{-x}\\
&=-(x-1)^2e^{-x}\leq0.
\end{align*}
Η ισότητα $f'(x)=0$ ισχύει μόνο για $x=1$. Επειδή η $f'$ είναι μη θετική και δεν μηδενίζεται σε κανένα διάστημα, η $f$ είναι \textbf{γνησίως φθίνουσα} στο $\mathbb{R}$.

Επιπλέον,
\[
\lim_{x\to-\infty}f(x)=+\infty
\]
και
\[
\lim_{x\to+\infty}f(x)=0.
\]
Επειδή $f(x)>0$ για κάθε $x\in\mathbb{R}$, το σύνολο τιμών της είναι
\[
{f(\mathbb{R})=(0,+\infty)}.
\]

\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
f''(x)
&=\left[-(x-1)^2e^{-x}\right]'\\
&=-2(x-1)e^{-x}+(x-1)^2e^{-x}\\
&=(x-1)(x-3)e^{-x}.
\end{align*}
Επειδή $e^{-x}>0$, έχουμε:
\begin{itemize}
\item $f''(x)>0$ για $x<1$ ή $x>3$,
\item $f''(x)=0$ για $x=1$ ή $x=3$,
\item $f''(x)<0$ για $1<x<3$.
\end{itemize}
Άρα η $f$ είναι \textbf{κυρτή} στα διαστήματα $(-\infty,1]$, $[3,+\infty)$ και \textbf{κοίλη} στο $[1,3]$.

Επομένως η $C_f$ έχει ακριβώς δύο σημεία καμπής:
\[
{K_1\left(1,\frac{2}{e}\right)}
\quad\text{και}\quad
{K_2\left(3,\frac{10}{e^3}\right)}.
\]

Το σημείο καμπής με τη μικρότερη τετμημένη είναι το
\[
K_1\left(1,\frac{2}{e}\right).
\]
Επειδή
\[
f'(1)=0,
\]
η εξίσωση της εφαπτομένης της $C_f$ στο $K_1$ είναι
\[
{y=\frac{2}{e}}.
\]

\item Επειδή $f(x)>0$ για κάθε $x\in\mathbb{R}$, το ζητούμενο εμβαδόν είναι
\[
E=\int_0^1(x^2+1)e^{-x}\d x.
\]
Μια παράγουσα της $f$ είναι
\[
F(x)=-(x^2+2x+3)e^{-x}.
\]
Επομένως
\begin{align*}
E
&=\left[-(x^2+2x+3)e^{-x}\right]_0^1\\
&=-\frac{6}{e}+3\\
&={3-\frac{6}{e}}.
\end{align*}

\item Έχουμε
\begin{align*}
L
&=\lim_{x\to+\infty}\frac{\ln\left((x^2+1)e^{-x}\right)}{x}\\
&=\lim_{x\to+\infty}
\left[
\frac{\ln(x^2+1)}{x}-1
\right].
\end{align*}
Επειδή
\[
\lim_{x\to+\infty}\frac{\ln(x^2+1)}{x}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{2x}{x^2+1}
=0,
\]
προκύπτει
\[
{L=-1}.
\]

Η εξίσωση
\[
f(x)=e^{Lx}+\frac{x}{4}
\]
γράφεται
\begin{align*}
(x^2+1)e^{-x}
&=e^{-x}+\frac{x}{4}\\
\Leftrightarrow
x^2e^{-x}&=\frac{x}{4}.
\end{align*}
Για $x\in(0,1)$ έχουμε $x>0$, οπότε η εξίσωση είναι ισοδύναμη με
\[
xe^{-x}=\frac14.
\]
Θεωρούμε τη συνάρτηση
\[
h(x)=xe^{-x},\quad x\in[0,1].
\]
Για κάθε $x\in(0,1)$ είναι
\[
h'(x)=(1-x)e^{-x}>0,
\]
οπότε η $h$ είναι γνησίως αύξουσα στο $[0,1]$. Επιπλέον,
\[
h(0)=0<\frac14
\]
και
\[
h(1)=\frac1e>\frac14,
\]
αφού $e<4$. Επομένως, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει μία τουλάχιστον ρίζα στο $(0,1)$. Η ρίζα αυτή είναι μοναδική, διότι η $h$ είναι γνησίως αύξουσα.

Συνεπώς η εξίσωση έχει ακριβώς μία ρίζα στο διάστημα $(0,1)$.
\end{erwthma}
\end{thema}
