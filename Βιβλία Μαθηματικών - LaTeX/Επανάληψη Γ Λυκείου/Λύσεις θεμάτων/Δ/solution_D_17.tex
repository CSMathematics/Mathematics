\begin{thema}{Δ}
\begin{erwthma}
\item Η συνάρτηση $g(x)=f(x)e^{-x}$ είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}$ και
\begin{align*}
g'(x)
&=f'(x)e^{-x}-f(x)e^{-x}\\
&=e^{-x}\left(f'(x)-f(x)\right).
\end{align*}
Επομένως
\begin{align*}
g''(x)
&=-e^{-x}\left(f'(x)-f(x)\right)
+e^{-x}\left(f''(x)-f'(x)\right)\\
&=e^{-x}\left(f''(x)-2f'(x)+f(x)\right)\\
&=0
\end{align*}
για κάθε $x\in\mathbb{R}$.

\item Επειδή $g''(x)=0$ για κάθε $x\in\mathbb{R}$, η $g'$ είναι σταθερή. Άρα υπάρχουν σταθερές $a,\beta\in\mathbb{R}$ τέτοιες ώστε
\[
g(x)=ax+\beta.
\]
Από τη σχέση $f(0)=0$ έχουμε
\[
g(0)=f(0)e^0=0,
\]
οπότε $\beta=0$.

Επίσης
\[
g'(x)=e^{-x}\left(f'(x)-f(x)\right),
\]
άρα
\[
g'(0)=f'(0)-f(0)=1.
\]
Επειδή $g'(x)=a$, προκύπτει $a=1$. Επομένως
\[
g(x)=x.
\]
Άρα
\[
f(x)e^{-x}=x
\Leftrightarrow
{f(x)=xe^x},
\quad x\in\mathbb{R}.
\]

\item Για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=e^x+xe^x=e^x(x+1).
\]
Επειδή $e^x>0$, το πρόσημο της $f'$ είναι το πρόσημο του $x+1$. Επομένως:
\begin{itemize}
\item $f'(x)<0$ για $x<-1$,
\item $f'(-1)=0$,
\item $f'(x)>0$ για $x>-1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

\tkzTabLine{, -, z, +, }
\tkzTabVar{+/{0},-/{-\frac{1}{e}},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,-1]$ και \textbf{γνησίως αύξουσα} στο $[-1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=-1$, με τιμή
\[
{f(-1)=-\frac{1}{e}}.
\]

Για την κυρτότητα έχουμε
\[
f''(x)=\left[e^x(x+1)\right]'=e^x(x+2).
\]
Επειδή $e^x>0$, ισχύουν:
\begin{itemize}
\item $f''(x)<0$ για $x<-2$,
\item $f''(-2)=0$,
\item $f''(x)>0$ για $x>-2$.
\end{itemize}
Άρα η $f$ είναι \textbf{κοίλη} στο $(-\infty,-2]$ και \textbf{κυρτή} στο $[-2,+\infty)$. Επειδή η $f''$ αλλάζει πρόσημο στο $x=-2$, η $C_f$ έχει μοναδικό σημείο καμπής το
\[
{K\left(-2,-\frac{2}{e^2}\right)}.
\]

\item Η ανίσωση γράφεται
\[
f(x^2)<f(3x).
\]
Διακρίνουμε περιπτώσεις:
\begin{itemize}
\item Αν $x<0$, τότε $x^2>0$, οπότε
\[
f(x^2)=x^2e^{x^2}>0,
\]
ενώ
\[
f(3x)=3xe^{3x}<0.
\]
Άρα η ανίσωση είναι αδύνατη.

\item Αν $x=0$, τότε $f(x^2)=f(3x)=0$, οπότε η αυστηρή ανίσωση δεν ισχύει.

\item Αν $x>0$, τότε $x^2,3x\in(0,+\infty)\subset[-1,+\infty)$, όπου η $f$ είναι γνησίως αύξουσα. Επομένως
\begin{align*}
f(x^2)<f(3x)
&\Leftrightarrow x^2<3x\\
&\Leftrightarrow x(x-3)<0\\
&\Leftrightarrow 0<x<3.
\end{align*}
\end{itemize}
Άρα το σύνολο λύσεων είναι
\[
{x\in(0,3)}.
\]
\end{erwthma}
\end{thema}
