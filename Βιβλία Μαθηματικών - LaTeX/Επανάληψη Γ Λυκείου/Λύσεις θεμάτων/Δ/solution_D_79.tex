%=========== ΛΥΣΗ - 79ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Από την εκφώνηση έχουμε
\[
f''(x)=2-x,\quad x\in[0,4].
\]
Άρα
\[
f''(x)>0\Leftrightarrow x<2
\]
και
\[
f''(x)<0\Leftrightarrow x>2.
\]

\begin{erwthma}
\item Από το πρόσημο της $f''$ προκύπτει ότι:
\begin{itemize}
\item η $f$ είναι κυρτή στο $[0,2]$,
\item η $f$ είναι κοίλη στο $[2,4]$.
\end{itemize}

Επειδή η $f''$ αλλάζει πρόσημο στο $x=2$, η $C_f$ έχει σημείο καμπής
στο σημείο με τετμημένη $2$.

Για να βρούμε την τεταγμένη του, υπολογίζουμε πρώτα την $f'$.
Έχουμε
\[
f'(x)=\int (2-x)\d x=2x-\frac{x^2}{2}+c.
\]
Επειδή
\[
f'(0)=-1,
\]
παίρνουμε
\[
c=-1.
\]
Άρα
\[
f'(x)=2x-\frac{x^2}{2}-1.
\]

Οπότε
\[
f(x)=\int\left(2x-\frac{x^2}{2}-1\right)\d x
=x^2-\frac{x^3}{6}-x+c_1.
\]
Επειδή
\[
f(0)=3,
\]
παίρνουμε
\[
c_1=3.
\]
Άρα
\[
f(x)=x^2-\frac{x^3}{6}-x+3.
\]
Επομένως
\[
f(2)=4-\frac{8}{6}-2+3
=5-\frac43
=\frac{11}{3}.
\]

Άρα το σημείο καμπής της $C_f$ είναι
\[
{A\left(2,\frac{11}{3}\right)}.
\]

\item Από το θεμελιώδες θεώρημα του ολοκληρωτικού λογισμού έχουμε
\[
f'(2)-f'(0)=\int_0^2 f''(x)\d x.
\]
Στο διάστημα $[0,2]$ η $C_{f''}$ σχηματίζει με τους άξονες τρίγωνο
εμβαδού $2$, και επειδή $f''(x)\geq 0$ στο $[0,2]$, ισχύει
\[
\int_0^2 f''(x)\d x=2.
\]
Άρα
\[
f'(2)-f'(0)=2.
\]
Επειδή
\[
f'(0)=-1,
\]
προκύπτει
\[
f'(2)+1=2
\]
και επομένως
\[
{f'(2)=1}.
\]

\item Από το α΄ ερώτημα έχουμε
\[
f'(x)=2x-\frac{x^2}{2}-1.
\]
Άρα
\[
f'(x)=0
\Leftrightarrow
2x-\frac{x^2}{2}-1=0.
\]
Πολλαπλασιάζοντας με $2$, παίρνουμε
\[
4x-x^2-2=0
\Leftrightarrow
x^2-4x+2=0.
\]
Οι ρίζες είναι
\[
x=\frac{4\pm\sqrt{16-8}}{2}
=\frac{4\pm2\sqrt2}{2}
=2\pm\sqrt2.
\]
Από αυτές, στο διάστημα $(0,2)$ ανήκει μόνο η
\[
2-\sqrt2.
\]
Άρα η εξίσωση
\[
f'(x)=0
\]
έχει ακριβώς μία ρίζα στο $(0,2)$, την
\[
{x=2-\sqrt2}.
\]

\item Από το προηγούμενο ερώτημα οι ρίζες της εξίσωσης $f'(x)=0$ στο
$[0,4]$ είναι
\[
2-\sqrt2
\quad\text{και}\quad
2+\sqrt2.
\]
Επειδή
\[
f'(x)=2x-\frac{x^2}{2}-1
=-\frac12\left(x^2-4x+2\right),
\]
το τριώνυμο αυτό είναι θετικό ανάμεσα στις ρίζες του.

Άρα
\[
f'(x)>0
\Leftrightarrow
x\in(2-\sqrt2,\,2+\sqrt2).
\]
Επομένως η λύση της ανίσωσης $f'(x)>0$ στο $[0,4]$ είναι
\[
{x\in(2-\sqrt2,\,2+\sqrt2)}.
\]
\end{erwthma}
\end{thema}
