\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\begin{align*}
f'(x)
&=(e^x-e)+(x-1)e^x\\
&=xe^x-e.
\end{align*}
Θεωρούμε τη συνάρτηση
\[
h(x)=xe^x.
\]
Για $x<0$ είναι $h(x)<0<e$, ενώ στο $(0,+\infty)$ η $h$ είναι γνησίως αύξουσα, αφού
\[
h'(x)=(x+1)e^x>0.
\]
Επειδή $h(1)=e$, η εξίσωση $f'(x)=0$ έχει μοναδική λύση την $x=1$. Επομένως:
\begin{itemize}
\item $f'(x)<0$ για $x<1$,
\item $f'(1)=0$,
\item $f'(x)>0$ για $x>1$.
\end{itemize}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$. Παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
{f(1)=0}.
\]

Για την κυρτότητα έχουμε
\[
f''(x)=(x+1)e^x.
\]
Επομένως:
\begin{itemize}
\item $f''(x)<0$ για $x<-1$,
\item $f''(-1)=0$,
\item $f''(x)>0$ για $x>-1$.
\end{itemize}
Άρα η $f$ είναι \textbf{κοίλη} στο $(-\infty,-1]$ και \textbf{κυρτή} στο $[-1,+\infty)$. Η $C_f$ έχει μοναδικό σημείο καμπής το
\[
{K\left(-1,2e-\frac{2}{e}\right)}.
\]

\item Έχουμε
\[
\lim_{x\to-\infty}f(x)=+\infty,
\quad
f(1)=0
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty.
\]
Στο διάστημα $(-\infty,1]$ η $f$ είναι συνεχής και γνησίως φθίνουσα. Επίσης
\[
f(-1)=2e-\frac{2}{e}<2e.
\]
Άρα η εξίσωση $f(x)=2e$ έχει ακριβώς μία ρίζα
\[
x_1<-1.
\]

Στο διάστημα $[1,+\infty)$ η $f$ είναι συνεχής και γνησίως αύξουσα, με $f(1)=0<2e$ και $\lim_{x\to+\infty}f(x)=+\infty$. Επομένως έχει ακριβώς μία ακόμη ρίζα
\[
x_2>1.
\]
Συνεπώς η εξίσωση έχει ακριβώς δύο ρίζες, για τις οποίες
\[
{x_1<-1<1<x_2}.
\]

\item Θεωρούμε τις συναρτήσεις
\[
F(x)=e^xf(x)
\quad\text{και}\quad
G(x)=e^x.
\]
Οι $F,G$ είναι συνεχείς στο $[x_1,x_2]$ και παραγωγίσιμες στο $(x_1,x_2)$, ενώ $G'(x)=e^x\neq0$. Επομένως, από το Θεώρημα Μέσης Τιμής του \eng{Cauchy}, υπάρχει $\xi\in(x_1,x_2)$ τέτοιο ώστε
\begin{align*}
\frac{F(x_2)-F(x_1)}{G(x_2)-G(x_1)}
&=\frac{F'(\xi)}{G'(\xi)}.
\end{align*}
Επειδή $f(x_1)=f(x_2)=2e$, έχουμε
\[
\frac{F(x_2)-F(x_1)}{G(x_2)-G(x_1)}
=\frac{2e(e^{x_2}-e^{x_1})}{e^{x_2}-e^{x_1}}
=2e.
\]
Επίσης
\[
\frac{F'(\xi)}{G'(\xi)}
=\frac{e^\xi\left(f'(\xi)+f(\xi)\right)}{e^\xi}
=f'(\xi)+f(\xi).
\]
Άρα
\[
{f'(\xi)+f(\xi)=2e}.
\]

\item Έχουμε
\[
A(0,f(0))\equiv A(0,e-1)
\]
και
\[
B(1,f(1))\equiv B(1,0).
\]
Ο συντελεστής διεύθυνσης της τέμνουσας $AB$ είναι
\[
\lambda=\frac{0-(e-1)}{1-0}=1-e.
\]
Άρα η εξίσωση της τέμνουσας είναι
\[
(\varepsilon):{y=(e-1)(1-x)}.
\]

Στο διάστημα $[0,1]$ έχουμε
\[
f''(x)=(x+1)e^x>0,
\]
οπότε η $f$ είναι αυστηρά κυρτή. Επομένως η $C_f$ βρίσκεται κάτω από την τέμνουσα των άκρων της, δηλαδή για κάθε $x\in(0,1)$ ισχύει
\[
f(x)<(e-1)(1-x).
\]
Ολοκληρώνοντας κατά μέλη στο $[0,1]$, παίρνουμε
\begin{align*}
\int_0^1f(x)\d x
&<\int_0^1(e-1)(1-x)\d x\\
&=(e-1)\left[x-\frac{x^2}{2}\right]_0^1\\
&={\frac{e-1}{2}}.
\end{align*}
\end{erwthma}
\end{thema}
