%=========== ΛΥΣΗ - 37ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Από τη δοσμένη σχέση έχουμε
\begin{align*}
(f(x)-x)(f(x)+x)&=x^2+1\\
f^2(x)-x^2&=x^2+1\\
f^2(x)&=2x^2+1.
\end{align*}
Επειδή
\[
2x^2+1>0
\]
για κάθε $x\in\mathbb{R}$, η $f$ δεν μηδενίζεται στο $\mathbb{R}$.

Η $f$ είναι συνεχής και το $\mathbb{R}$ είναι διάστημα. Επομένως η $f$ διατηρεί σταθερό πρόσημο στο $\mathbb{R}$, αφού διαφορετικά, από το Θεώρημα Ενδιάμεσων Τιμών, θα υπήρχε σημείο στο οποίο θα μηδενιζόταν.

Επειδή η $C_f$ διέρχεται από το σημείο $A(0,1)$, έχουμε
\[
f(0)=1>0.
\]
Άρα $f(x)>0$ για κάθε $x\in\mathbb{R}$ και συνεπώς
\[
{f(x)=\sqrt{2x^2+1}},\quad x\in\mathbb{R}.
\]

\item Η $f$ είναι συνεχής στα διαστήματα $[-1,0]$, $[0,1]$ και παραγωγίσιμη στα αντίστοιχα εσωτερικά τους.

Εφαρμόζοντας το Θεώρημα Μέσης Τιμής στο $[-1,0]$, υπάρχει $\xi_1\in(-1,0)$ τέτοιο ώστε
\begin{align*}
f'(\xi_1)
&=\frac{f(0)-f(-1)}{0-(-1)}\\
&=1-\sqrt{3}.
\end{align*}

Ομοίως, εφαρμόζοντας το Θεώρημα Μέσης Τιμής στο $[0,1]$, υπάρχει $\xi_2\in(0,1)$ τέτοιο ώστε
\begin{align*}
f'(\xi_2)
&=\frac{f(1)-f(0)}{1-0}\\
&=\sqrt{3}-1.
\end{align*}
Επομένως
\[
{f'(\xi_1)+f'(\xi_2)=0}.
\]

\item Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f'(x)
&=\left(\sqrt{2x^2+1}\right)'\\
&=\frac{2x}{\sqrt{2x^2+1}}.
\end{align*}
Επίσης
\begin{align*}
f''(x)
&=\left[2x(2x^2+1)^{-\frac{1}{2}}\right]'\\
&=2(2x^2+1)^{-\frac{1}{2}}
-4x^2(2x^2+1)^{-\frac{3}{2}}\\
&=\frac{2(2x^2+1)-4x^2}
{(2x^2+1)^{\frac{3}{2}}}\\
&=\frac{2}{(2x^2+1)^{\frac{3}{2}}}>0.
\end{align*}
Άρα η $f$ είναι \textbf{αυστηρά κυρτή} στο $\mathbb{R}$ και η $C_f$ δεν έχει σημεία καμπής.

\item Επειδή οι αριθμοί $a<\beta<\gamma$ είναι διαδοχικοί όροι αριθμητικής προόδου, ισχύει
\[
2\beta=a+\gamma
\Leftrightarrow
\beta=\frac{a+\gamma}{2}.
\]
Η $f$ είναι αυστηρά κυρτή στο $\mathbb{R}$ και $a\neq\gamma$. Επομένως, από τον ορισμό της αυστηρής κυρτότητας, έχουμε
\[
f\left(\frac{a+\gamma}{2}\right)
<\frac{f(a)+f(\gamma)}{2}.
\]
Επειδή $\beta=\frac{a+\gamma}{2}$, προκύπτει
\[
f(\beta)<\frac{f(a)+f(\gamma)}{2},
\]
άρα
\[
{f(a)+f(\gamma)>2f(\beta)}.
\]
\end{erwthma}
\end{thema}
