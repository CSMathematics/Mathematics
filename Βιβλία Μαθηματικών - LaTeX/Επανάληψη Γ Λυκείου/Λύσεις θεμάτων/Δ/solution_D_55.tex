\begin{thema}{Δ}
\begin{erwthma}
\item Η $g$ είναι παραγωγίσιμη στο $[0,+\infty)$ και
\[
g'(x)=e^x-e^{-x}.
\]
Για κάθε $x>0$ είναι $e^x>e^{-x}$, άρα
\[
g'(x)>0.
\]
Επομένως η $g$ είναι \textbf{γνησίως αύξουσα} στο $[0,+\infty)$.

Επιπλέον,
\[
g(0)=2
\]
και
\[
\lim_{x\to+\infty}g(x)=+\infty.
\]
Άρα
\[
g\big([0,+\infty)\big)=[2,+\infty).
\]

Έστω $x\geq1$. Τότε $2x\in[2,+\infty)$, οπότε, λόγω της συνέχειας της $g$, υπάρχει τουλάχιστον ένα $y\in[0,+\infty)$ τέτοιο ώστε
\[
g(y)=2x.
\]
Η λύση αυτή είναι μοναδική, διότι η $g$ είναι γνησίως αύξουσα.

\item Η δοσμένη σχέση γράφεται
\[
g(f(x))=2x.
\]
Έστω $1\leq x_1<x_2$. Τότε
\[
2x_1<2x_2,
\]
άρα
\[
g(f(x_1))<g(f(x_2)).
\]
Επειδή $f(x_1),f(x_2)\geq0$ και η $g$ είναι γνησίως αύξουσα στο $[0,+\infty)$, προκύπτει
\[
f(x_1)<f(x_2).
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $[1,+\infty)$.

\item Θέτουμε
\[
z=e^{f(x)}>0.
\]
Τότε
\begin{align*}
e^{f(x)}+e^{-f(x)}=2x
&\Leftrightarrow
z+\frac{1}{z}=2x\\
&\Leftrightarrow
z^2-2xz+1=0.
\end{align*}
Επομένως
\[
z=x\pm\sqrt{x^2-1}.
\]
Επειδή $f(x)\geq0$, είναι
\[
z=e^{f(x)}\geq1.
\]
Για $x>1$ έχουμε
\[
x-\sqrt{x^2-1}<1,
\]
οπότε δεκτή είναι μόνο η τιμή
\[
z=x+\sqrt{x^2-1}.
\]
Ο τύπος αυτός ισχύει και για $x=1$. Άρα
\[
{
f(x)=\ln\left(x+\sqrt{x^2-1}\right)},\quad x\geq1.
\]

Η $f$ είναι γνησίως αύξουσα στο $[1,+\infty)$, με
\[
f(1)=0
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty.
\]
Επομένως αντιστρέφεται και
\[
D_{f^{-1}}=[0,+\infty).
\]

Από τη σχέση
\[
e^{f(x)}+e^{-f(x)}=2x,
\]
θέτοντας $y=f(x)$ παίρνουμε
\[
x=\frac{e^y+e^{-y}}{2}.
\]
Άρα
\[
{
f^{-1}(y)=\frac{e^y+e^{-y}}{2}},\quad y\geq0.
\]

\item Θέτουμε
\[
b=\frac{e^2+1}{2e}
=\frac{e+e^{-1}}{2}.
\]
Από τον τύπο της αντίστροφης έχουμε
\[
b=f^{-1}(1),
\]
οπότε
\[
f(b)=1.
\]

Χρησιμοποιώντας τον τύπο ολοκληρώματος αντίστροφης συνάρτησης, παίρνουμε
\begin{align*}
I
&=\int_1^b f(x)\d x\\
&=b\cdot f(b)-1\cdot f(1)
-\int_{f(1)}^{f(b)}f^{-1}(y)\d y\\
&=b-\int_0^1\frac{e^y+e^{-y}}{2}\d y\\
&=b-\left[\frac{e^y-e^{-y}}{2}\right]_0^1\\
&=\frac{e+e^{-1}}{2}
-\frac{e-e^{-1}}{2}\\
&={\frac{1}{e}}.
\end{align*}
\end{erwthma}
\end{thema}
