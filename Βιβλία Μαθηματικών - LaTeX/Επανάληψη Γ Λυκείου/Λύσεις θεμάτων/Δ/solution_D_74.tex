%=========== ΛΥΣΗ - 74ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Η χορδή $AB$ έχει άκρα
\[
A(0,1)
\quad\text{και}\quad
B(5,6).
\]
Ο συντελεστής διεύθυνσής της είναι
\[
\lambda_{AB}=\frac{6-1}{5-0}=1.
\]
Άρα η εξίσωση της χορδής είναι
\[
y=x+1.
\]

\begin{erwthma}
\item Επειδή η $f$ είναι αυστηρά κοίλη στο $[0,5]$, η γραφική της
παράσταση βρίσκεται πάνω από τη χορδή που ενώνει τα άκρα της.

Επομένως
\[
f(x)\geq x+1,\quad x\in[0,5].
\]
Μάλιστα, για κάθε εσωτερικό σημείο του διαστήματος, ισχύει η αυστηρή
ανισότητα
\[
f(x)>x+1,\quad x\in(0,5).
\]
Στα άκρα έχουμε ισότητα, αφού
\[
f(0)=1
\quad\text{και}\quad
f(5)=6.
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f(x)>x+1,\quad x\in(0,5).
\]
Άρα η εξίσωση
\[
f(x)=x+1
\]
δεν έχει λύσεις στο εσωτερικό του διαστήματος.

Στα άκρα:
\[
f(0)=1=0+1
\]
και
\[
f(5)=6=5+1.
\]
Επομένως οι λύσεις της εξίσωσης στο $[0,5]$ είναι
\[
{x=0\ \text{ή}\ x=5}.
\]

\item Η $f$ είναι συνεχής στο $[0,5]$ και παραγωγίσιμη στο $(0,5)$.
Άρα, από το θεώρημα μέσης τιμής, υπάρχει
\[
\xi\in(0,5)
\]
τέτοιο ώστε
\[
f'(\xi)=\frac{f(5)-f(0)}{5-0}
=\frac{6-1}{5}
=1.
\]
Άρα η εξίσωση
\[
f'(x)=1
\]
έχει τουλάχιστον μία ρίζα στο $(0,5)$.

Θα δείξουμε ότι η ρίζα αυτή είναι μοναδική. Έστω ότι υπάρχουν
\[
\xi_1,\xi_2\in(0,5),\quad \xi_1<\xi_2,
\]
τέτοια ώστε
\[
f'(\xi_1)=1
\quad\text{και}\quad
f'(\xi_2)=1.
\]
Τότε
\[
f'(\xi_1)=f'(\xi_2).
\]
Όμως η $f'$ είναι γνησίως φθίνουσα στο $(0,5)$, άρα από
\[
\xi_1<\xi_2
\]
θα έπρεπε να ισχύει
\[
f'(\xi_1)>f'(\xi_2),
\]
άτοπο.

Επομένως η εξίσωση
\[
f'(x)=1
\]
έχει μοναδική ρίζα στο $(0,5)$.

\item Από το α΄ ερώτημα γνωρίζουμε ότι
\[
f(x)>x+1,\quad x\in(0,5).
\]
Άρα
\[
\int_0^5 f(x)\d x
>
\int_0^5 (x+1)\d x.
\]
Υπολογίζουμε
\[
\int_0^5 (x+1)\d x
=\frac{5^2}{2}+5
=\frac{25}{2}+5
=\frac{35}{2}.
\]
Επομένως
\[
{\int_0^5 f(x)\d x>\frac{35}{2}}.
\]
\end{erwthma}
\end{thema}
