%=========== ΛΥΣΗ - 47ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για $x=y=1$ η δοσμένη σχέση δίνει
\[
g(1)=g(1)+g(1),
\]
οπότε
\[
{g(1)=0}.
\]

Έστω $x>0$. Για $h$ αρκετά κοντά στο $0$ είναι $x+h>0$ και έχουμε
\begin{align*}
\frac{g(x+h)-g(x)}{h}
&=\frac{g\left(x\left(1+\frac{h}{x}\right)\right)-g(x)}{h}\\
&=\frac{g(x)+g\left(1+\frac{h}{x}\right)-g(x)}{h}\\
&=\frac{1}{x}
\frac{g\left(1+\frac{h}{x}\right)-g(1)}{\frac{h}{x}}.
\end{align*}
Καθώς $h\to0$, έχουμε $\frac{h}{x}\to0$. Επειδή η $g$ είναι παραγωγίσιμη στο $1$ με $g'(1)=1$, προκύπτει
\begin{align*}
g'(x)
&=\lim_{h\to0}\frac{g(x+h)-g(x)}{h}\\
&=\frac{1}{x}g'(1)\\
&={\frac{1}{x}}.
\end{align*}
Άρα η $g$ είναι παραγωγίσιμη σε κάθε $x>0$.

\item Για κάθε $x>0$ είναι
\[
\left[g(x)-\ln x\right]'
=g'(x)-\frac{1}{x}
=0.
\]
Άρα υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
g(x)-\ln x=c.
\]
Για $x=1$ έχουμε
\[
c=g(1)-\ln1=0.
\]
Επομένως
\[
{g(x)=\ln x},\quad x>0.
\]

Συνεπώς
\[
f(x)=\frac{\ln x}{x},\quad x>0.
\]
Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=\frac{1-\ln x}{x^2}.
\]
Επειδή $x^2>0$, έχουμε:
\begin{itemize}
\item $f'(x)>0$ για $0<x<e$,
\item $f'(e)=0$,
\item $f'(x)<0$ για $x>e$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$0$,$e$,$+\infty$}%
\tkzTabLine{d, +, z, -, }
\tkzTabVar{D-/{-\infty},+/{\frac{1}{e}},-/{0}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,e]$ και \textbf{γνησίως φθίνουσα} στο $[e,+\infty)$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=e$, με τιμή
\[
{f(e)=\frac{1}{e}}.
\]
Η $f$ δεν παρουσιάζει ελάχιστο, αφού
\[
\lim_{x\to0^+}f(x)=-\infty.
\]

\item Έστω $a>e$. Η εξίσωση
\[
x^a=a^x
\]
είναι ισοδύναμη με
\begin{align*}
a\ln x&=x\ln a\\
\Leftrightarrow
\frac{\ln x}{x}&=\frac{\ln a}{a}\\
\Leftrightarrow
f(x)&=f(a).
\end{align*}
Ο αριθμός $x=a$ είναι μία λύση.

Επειδή $a>e$ και η $f$ είναι γνησίως φθίνουσα στο $[e,+\infty)$, η λύση αυτή είναι η μοναδική στο $[e,+\infty)$. Επιπλέον,
\[
f(1)=0<f(a)<f(e)=\frac{1}{e}.
\]
Η $f$ είναι συνεχής και γνησίως αύξουσα στο $[1,e]$, οπότε υπάρχει ακριβώς μία ακόμη λύση στο $(1,e)$. Στο $(0,1)$ είναι $f(x)<0<f(a)$, άρα δεν υπάρχουν άλλες λύσεις.

Συνεπώς, για κάθε $a>e$, η εξίσωση $x^a=a^x$ έχει ακριβώς δύο θετικές ρίζες.

Η εξίσωση
\[
x^e=e^x
\]
είναι ισοδύναμη με
\[
f(x)=f(e)=\frac{1}{e}.
\]
Επειδή το $\frac{1}{e}$ είναι η μοναδική μέγιστη τιμή της $f$, η εξίσωση έχει μοναδική λύση
\[
{x=e}.
\]

\item Έχουμε
\begin{align*}
\int_1^ef(x)\d x
&=\int_1^e\frac{\ln x}{x}\d x\\
&=\left[\frac{1}{2}\ln^2x\right]_1^e\\
&=\frac{1}{2}.
\end{align*}
Άρα ζητούμε $\xi\in(1,e)$ τέτοιο ώστε
\[
f(\xi)=\frac{1}{4}.
\]
Έχουμε
\[
f(1)=0<\frac{1}{4}
\]
και
\[
f(e)=\frac{1}{e}>\frac{1}{4},
\]
αφού $e<4$. Επομένως, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει $\xi\in(1,e)$ τέτοιο ώστε
\[
f(\xi)=\frac{1}{4}.
\]
Η τιμή αυτή είναι μοναδική, διότι η $f$ είναι γνησίως αύξουσα στο $(1,e)$.

Τέλος,
\[
f(\sqrt{e})
=\frac{\ln\sqrt{e}}{\sqrt{e}}
=\frac{1}{2\sqrt{e}}.
\]
Επειδή $e<4$, έχουμε $\sqrt{e}<2$, άρα
\[
f(\sqrt{e})=\frac{1}{2\sqrt{e}}>\frac{1}{4}=f(\xi).
\]
Η $f$ είναι γνησίως αύξουσα στο $(1,e)$, επομένως
\[
{\xi<\sqrt{e}}.
\]
\end{erwthma}
\end{thema}
