\begin{thema}{Δ}
\begin{erwthma}
\item Επειδή $f(x)>0$ για κάθε $x\in\mathbb{R}$, ορίζεται η συνάρτηση $\ln f(x)$. Από τη δοσμένη σχέση έχουμε
\begin{align*}
f'(x)=f(x)e^x
&\Leftrightarrow \frac{f'(x)}{f(x)}=e^x\\
&\Leftrightarrow (\ln f(x))'=(e^x)'.
\end{align*}
Σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
\ln f(x)=e^x+c.
\]
Για $x=0$ και $f(0)=e$ παίρνουμε
\[
1=1+c\Leftrightarrow c=0.
\]
Άρα
\[
\ln f(x)=e^x
\]
και επομένως
\[
{f(x)=e^{e^x}},
\quad x\in\mathbb{R}.
\]

\item Από τη δοσμένη σχέση είναι
\[
f'(x)=f(x)e^x>0
\]
για κάθε $x\in\mathbb{R}$. Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Για την κυρτότητα έχουμε
\begin{align*}
f''(x)
&=\left(f(x)e^x\right)'\\
&=f'(x)e^x+f(x)e^x\\
&=f(x)e^{2x}+f(x)e^x\\
&=f(x)e^x(e^x+1)>0.
\end{align*}
Άρα η $f$ είναι \textbf{αυστηρά κυρτή} στο $\mathbb{R}$.

Στο σημείο $A(0,f(0))\equiv A(0,e)$ έχουμε
\[
f'(0)=f(0)e^0=e.
\]
Επομένως η εφαπτομένη της $C_f$ στο $A$ έχει εξίσωση
\begin{align*}
y-f(0)&=f'(0)(x-0)\\
\Leftrightarrow y-e&=ex\\
\Leftrightarrow {y=e(x+1)}.
\end{align*}

\item Από τον τύπο της $f$ έχουμε
\begin{align*}
\lim_{x\to-\infty}f(x)
&=\lim_{x\to-\infty}e^{e^x}\\
&=e^0\\
&={1}.
\end{align*}
Άρα η ευθεία $y=1$ είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.

Επειδή η $f$ είναι κυρτή στο $\mathbb{R}$, η γραφική της παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της. Ειδικότερα, από την εφαπτομένη στο $x_0=0$, για κάθε $x\in\mathbb{R}$ ισχύει
\[
{f(x)\geq e(x+1)}.
\]
Η ισότητα ισχύει μόνο για $x=0$, αφού η $f$ είναι αυστηρά κυρτή.

\item Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)+e^x.
\]
Η $h$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
h'(x)=f'(x)+e^x=f(x)e^x+e^x=e^x(f(x)+1)>0.
\]
Άρα η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$. Επίσης
\[
h(0)=f(0)+1=e+1.
\]
Επομένως η εξίσωση
\[
f(x)+e^x=e+1
\]
είναι ισοδύναμη με την $h(x)=h(0)$ και, λόγω της γνήσιας μονοτονίας της $h$, έχει μοναδική λύση
\[
{x=0}.
\]
\end{erwthma}
\end{thema}
