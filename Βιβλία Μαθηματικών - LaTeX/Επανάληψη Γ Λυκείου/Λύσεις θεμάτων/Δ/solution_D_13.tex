%=========== ΛΥΣΗ - 13ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\neq0$ έχουμε
\begin{align*}
\left(xf(x)+e^x\right)'
&=xf'(x)+f(x)+e^x\\
&=0.
\end{align*}
Επομένως η συνάρτηση
\[
g(x)=xf(x)+e^x
\]
είναι σταθερή σε καθένα από τα διαστήματα $(-\infty,0)$ και $(0,+\infty)$.

Στο $(0,+\infty)$ υπάρχει σταθερά $c_1\in\mathbb{R}$ τέτοια ώστε
\[
xf(x)+e^x=c_1.
\]
Για $x=1$ και $f(1)=1-e$ παίρνουμε
\[
1-e+e=c_1\Leftrightarrow c_1=1.
\]
Άρα
\[
f(x)=\frac{1-e^x}{x},\quad x>0.
\]

Στο $(-\infty,0)$ υπάρχει σταθερά $c_2\in\mathbb{R}$ τέτοια ώστε
\[
xf(x)+e^x=c_2.
\]
Επειδή η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, είναι συνεχής στο $0$. Επομένως
\[
\lim_{x\to0^-}xf(x)=0
\quad\text{και}\quad
\lim_{x\to0^-}e^x=1.
\]
Άρα, λαμβάνοντας όρια στη σχέση $xf(x)+e^x=c_2$, προκύπτει
\[
c_2=1.
\]
Συνεπώς
\[
{f(x)=\frac{1-e^x}{x}},\quad x\neq0.
\]

\item Από τη συνέχεια της $f$ στο $x=0$ έχουμε
\begin{align*}
f(0)
&=\lim_{x\to0}f(x)\\
&=\lim_{x\to0}\frac{1-e^x}{x}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}(-e^x)\\
&=-1.
\end{align*}
Άρα
\[
{f(0)=-1}.
\]

\item Για κάθε $x\neq0$ είναι
\begin{align*}
f'(x)
&=\left(\frac{1-e^x}{x}\right)'\\
&=\frac{-xe^x-(1-e^x)}{x^2}\\
&=\frac{e^x(1-x)-1}{x^2}.
\end{align*}
Θεωρούμε τη βοηθητική συνάρτηση
\[
h(x)=e^x(1-x)-1.
\]
Για κάθε $x\in\mathbb{R}$ είναι
\[
h'(x)=e^x(1-x)-e^x=-xe^x.
\]
Επομένως η $h$ είναι γνησίως αύξουσα στο $(-\infty,0]$ και γνησίως φθίνουσα στο $[0,+\infty)$. Άρα παρουσιάζει ολικό μέγιστο στο $x=0$, με τιμή
\[
h(0)=0.
\]
Συνεπώς
\[
h(x)<0
\]
για κάθε $x\neq0$. Επειδή $x^2>0$, προκύπτει
\[
f'(x)<0
\]
για κάθε $x\neq0$.

Άρα η $f$ είναι γνησίως φθίνουσα στα διαστήματα $(-\infty,0]$ και $[0,+\infty)$. Επιπλέον, αν $x_1<0<x_2$, τότε
\[
f(x_1)>f(0)>f(x_2).
\]
Επομένως η $f$ είναι \textbf{γνησίως φθίνουσα} σε ολόκληρο το $\mathbb{R}$.

\item Έχουμε
\begin{align*}
\lim_{x\to-\infty}f(x)
&=\lim_{x\to-\infty}\frac{1-e^x}{x}\\
&=\frac{1-0}{-\infty}\\
&=0.
\end{align*}
Επομένως η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.
\end{erwthma}
\end{thema}
