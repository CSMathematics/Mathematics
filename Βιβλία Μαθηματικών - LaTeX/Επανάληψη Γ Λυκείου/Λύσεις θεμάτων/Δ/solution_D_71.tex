%=========== ΛΥΣΗ - 71ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Επειδή η $f$ παρουσιάζει τοπικό μέγιστο στο εσωτερικό σημείο $x=2$ και
τοπικό ελάχιστο στο εσωτερικό σημείο $x=4$, από το θεώρημα \eng{Fermat}
έχουμε
\[
f'(2)=0
\quad\text{και}\quad
f'(4)=0.
\]

Επίσης,
\[
f''(x)<0,\quad x\in(0,3),
\]
άρα η $f'$ είναι γνησίως φθίνουσα στο $(0,3)$. Επομένως, επειδή
$f'(2)=0$, έχουμε
\[
f'(x)>0,\quad x\in(0,2)
\]
και
\[
f'(x)<0,\quad x\in(2,3).
\]

Ακόμη,
\[
f''(x)>0,\quad x\in(3,6),
\]
άρα η $f'$ είναι γνησίως αύξουσα στο $(3,6)$. Επομένως, επειδή
$f'(4)=0$, έχουμε
\[
f'(x)<0,\quad x\in(3,4)
\]
και
\[
f'(x)>0,\quad x\in(4,6).
\]

Συμπεραίνουμε ότι η $f$ είναι:
\[
\begin{array}{c}
\text{γνησίως αύξουσα στο }[0,2],\\
\text{γνησίως φθίνουσα στο }[2,4],\\
\text{γνησίως αύξουσα στο }[4,6].
\end{array}
\]

\begin{erwthma}
\item Στο διάστημα $[0,2]$ η $f$ είναι γνησίως αύξουσα και
\[
f(0)=0,\qquad f(2)=4.
\]
Άρα η μόνη ρίζα της $f(x)=0$ στο $[0,2]$ είναι η
\[
x=0.
\]

Στο διάστημα $[2,4]$ η $f$ είναι συνεχής και
\[
f(2)=4>0,
\qquad
f(4)=-2<0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\alpha\in(2,4)
\]
τέτοιο ώστε
\[
f(\alpha)=0.
\]
Επειδή η $f$ είναι γνησίως φθίνουσα στο $[2,4]$, η ρίζα αυτή είναι
μοναδική στο διάστημα $[2,4]$.

Στο διάστημα $[4,6]$ η $f$ είναι συνεχής και
\[
f(4)=-2<0,
\qquad
f(6)=1>0.
\]
Άρα, πάλι από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\beta\in(4,6)
\]
τέτοιο ώστε
\[
f(\beta)=0.
\]
Επειδή η $f$ είναι γνησίως αύξουσα στο $[4,6]$, η ρίζα αυτή είναι
μοναδική στο διάστημα $[4,6]$.

Επομένως η εξίσωση $f(x)=0$ έχει ακριβώς τρεις ρίζες στο $[0,6]$:
\[
{0,\ \alpha,\ \beta},
\quad
\alpha\in(2,4),\ \beta\in(4,6).
\]

\item Από το προηγούμενο ερώτημα γνωρίζουμε ότι
\[
f(0)=0,\qquad f(\alpha)=0,\qquad f(\beta)=0,
\]
όπου
\[
\alpha\in(2,4)
\quad\text{και}\quad
\beta\in(4,6).
\]

Επειδή η $f$ είναι γνησίως αύξουσα στο $[0,2]$, έχουμε
\[
f(x)>f(0)=0,\quad x\in(0,2].
\]
Επειδή η $f$ είναι γνησίως φθίνουσα στο $[2,4]$, από
\[
f(2)>0
\quad\text{και}\quad
f(\alpha)=0
\]
παίρνουμε
\[
f(x)>0,\quad x\in[2,\alpha).
\]
Άρα
\[
f(x)>0,\quad x\in(0,\alpha).
\]

Στο διάστημα $(\alpha,4]$ η $f$ είναι γνησίως φθίνουσα και
\[
f(\alpha)=0,
\]
οπότε
\[
f(x)<0,\quad x\in(\alpha,4].
\]
Στο διάστημα $[4,\beta)$ η $f$ είναι γνησίως αύξουσα και
\[
f(4)=-2<0,\qquad f(\beta)=0,
\]
οπότε
\[
f(x)<0,\quad x\in[4,\beta).
\]
Άρα
\[
f(x)<0,\quad x\in(\alpha,\beta).
\]

Τέλος, επειδή η $f$ είναι γνησίως αύξουσα στο $[4,6]$ και
\[
f(\beta)=0,
\]
έχουμε
\[
f(x)>0,\quad x\in(\beta,6].
\]

Επειδή ζητάμε τη λύση στο $(0,6)$, προκύπτει
\[
{x\in(0,\alpha)\cup(\beta,6)}.
\]

\item Από το θεώρημα μέσης τιμής για την $f$ στο $[0,2]$ υπάρχει
\[
x_1\in(0,2)
\]
τέτοιο ώστε
\[
f'(x_1)=\frac{f(2)-f(0)}{2-0}
=\frac{4-0}{2}
=2.
\]
Επίσης
\[
f'(2)=0.
\]
Η $f'$ είναι συνεχής στο $[x_1,2]$, οπότε, επειδή
\[
0<\frac16<2,
\]
από το θεώρημα ενδιάμεσων τιμών υπάρχει
\[
\xi_1\in(x_1,2)\subset(0,2)
\]
τέτοιο ώστε
\[
f'(\xi_1)=\frac16.
\]

Από το θεώρημα μέσης τιμής για την $f$ στο $[4,6]$ υπάρχει
\[
x_2\in(4,6)
\]
τέτοιο ώστε
\[
f'(x_2)=\frac{f(6)-f(4)}{6-4}
=\frac{1-(-2)}{2}
=\frac32.
\]
Επίσης
\[
f'(4)=0.
\]
Η $f'$ είναι συνεχής στο $[4,x_2]$, οπότε, επειδή
\[
0<\frac16<\frac32,
\]
υπάρχει
\[
\xi_2\in(4,x_2)\subset(4,6)
\]
τέτοιο ώστε
\[
f'(\xi_2)=\frac16.
\]

Τα σημεία $\xi_1$ και $\xi_2$ ανήκουν σε ξένα διαστήματα, άρα είναι
διαφορετικά. Επομένως η εξίσωση
\[
f'(x)=\frac16
\]
έχει τουλάχιστον δύο ρίζες στο $(0,6)$.

\item Στο διάστημα $[0,2]$ η $f$ είναι κοίλη, αφού
\[
f''(x)<0,\quad x\in(0,2).
\]
Άρα η γραφική της παράσταση βρίσκεται πάνω από τη χορδή που ενώνει τα
σημεία
\[
A(0,f(0))=(0,0)
\quad\text{και}\quad
B(2,f(2))=(2,4).
\]

Η εξίσωση της χορδής $AB$ είναι
\[
y=2x.
\]
Επομένως
\[
f(x)\geq 2x,\quad x\in[0,2].
\]

Μάλιστα η ισότητα δεν μπορεί να ισχύει σε όλο το $[0,2]$. Αν ίσχυε
\[
f(x)=2x,\quad x\in[0,2],
\]
τότε θα είχαμε
\[
f'(2)=2,
\]
που είναι άτοπο, αφού από το θεώρημα \eng{Fermat} γνωρίζουμε ότι
\[
f'(2)=0.
\]
Άρα υπάρχει υποδιάστημα του $[0,2]$ στο οποίο
\[
f(x)>2x.
\]

Συνεπώς
\[
\int_0^2 f(x)\d x
>
\int_0^2 2x\d x
=4.
\]
Δηλαδή
\[
{\int_0^2 f(x)\d x>4}.
\]
\end{erwthma}
\end{thema}
