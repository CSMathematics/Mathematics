%=========== ΛΥΣΗ - 31ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση
\[
h(t)=t^3+t,\quad t\in\mathbb{R}.
\]
Για κάθε $t\in\mathbb{R}$ είναι
\[
h'(t)=3t^2+1>0,
\]
οπότε η $h$ είναι γνησίως αύξουσα και επομένως $1-1$.

Αρχικά αποδεικνύουμε ότι η $f$ είναι $1-1$. Έστω $x_1,x_2\in\mathbb{R}$ με
\[
f(x_1)=f(x_2).
\]
Τότε
\[
f^3(x_1)+f(x_1)=f^3(x_2)+f(x_2),
\]
άρα, από τη δοσμένη σχέση,
\[
x_1=x_2.
\]
Επομένως η $f$ είναι $1-1$.

Στη συνέχεια αποδεικνύουμε ότι η $f$ είναι επί του $\mathbb{R}$. Έστω $y\in\mathbb{R}$ και θέτουμε
\[
x=y^3+y.
\]
Τότε
\[
h(f(x))=f^3(x)+f(x)=x=y^3+y=h(y).
\]
Επειδή η $h$ είναι $1-1$, έχουμε $f(x)=y$. Άρα η $f$ είναι επί και συνεπώς αντιστρέφεται στο $\mathbb{R}$.

Από τη σχέση
\[
f^3(x)+f(x)=x
\]
έχουμε
\[
h(f(x))=x,
\]
οπότε
\[
{f^{-1}(x)=x^3+x},\quad x\in\mathbb{R}.
\]

\item Έστω $x_1,x_2\in\mathbb{R}$ με $x_1<x_2$. Τότε
\[
h(f(x_1))=x_1<x_2=h(f(x_2)).
\]
Επειδή η $h$ είναι γνησίως αύξουσα, προκύπτει
\[
f(x_1)<f(x_2).
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Για $x=0$ η δοσμένη σχέση δίνει
\[
f^3(0)+f(0)=0
\Leftrightarrow
f(0)\left(f^2(0)+1\right)=0
\Leftrightarrow
f(0)=0.
\]
Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\[
{
\begin{aligned}
&f(x)<0 &&\text{για }x<0,\\
&f(0)=0,\\
&f(x)>0 &&\text{για }x>0.
\end{aligned}}
\]

Με βάση το πρόσημο και τη μονοτονία της $f$, έχουμε
\begin{align*}
f(f(x)-2)<0
&\Leftrightarrow f(x)-2<0\\
&\Leftrightarrow f(x)<2\\
&\Leftrightarrow x<f^{-1}(2)\\
&\Leftrightarrow x<2^3+2\\
&\Leftrightarrow {x<10}.
\end{align*}

\item Για κάθε $x,x_0\in\mathbb{R}$, αφαιρώντας κατά μέλη τις σχέσεις
\[
f^3(x)+f(x)=x
\]
και
\[
f^3(x_0)+f(x_0)=x_0,
\]
παίρνουμε
\begin{align*}
x-x_0
&=f^3(x)-f^3(x_0)+f(x)-f(x_0)\\
&=\left[f(x)-f(x_0)\right]
\left[f^2(x)+f(x)f(x_0)+f^2(x_0)+1\right].
\end{align*}
Όμως
\begin{align*}
f^2(x)+f(x)f(x_0)+f^2(x_0)
&=\left(f(x)+\frac{f(x_0)}{2}\right)^2
+\frac{3}{4}f^2(x_0)\\
&\geq0.
\end{align*}
Επομένως
\[
f^2(x)+f(x)f(x_0)+f^2(x_0)+1\geq1.
\]
Λαμβάνοντας απόλυτες τιμές, προκύπτει
\begin{align*}
|x-x_0|
&=|f(x)-f(x_0)|
\left[f^2(x)+f(x)f(x_0)+f^2(x_0)+1\right]\\
&\geq|f(x)-f(x_0)|.
\end{align*}
Άρα
\[
{|f(x)-f(x_0)|\leq|x-x_0|}
\]
για κάθε $x,x_0\in\mathbb{R}$.

Έστω τώρα $x_0\in\mathbb{R}$. Από την προηγούμενη ανισότητα έχουμε
\[
0\leq|f(x)-f(x_0)|\leq|x-x_0|.
\]
Επειδή
\[
\lim_{x\to x_0}|x-x_0|=0,
\]
από το κριτήριο παρεμβολής προκύπτει
\[
\lim_{x\to x_0}|f(x)-f(x_0)|=0,
\]
δηλαδή
\[
\lim_{x\to x_0}f(x)=f(x_0).
\]
Επομένως η $f$ είναι συνεχής στο $\mathbb{R}$.

\item Επειδή η $f$ είναι γνησίως αύξουσα και επί του $\mathbb{R}$, ισχύει
\[
\lim_{x\to+\infty}f(x)=+\infty.
\]
Από τη δοσμένη σχέση έχουμε
\[
x=f^3(x)+f(x)=f(x)\left(f^2(x)+1\right),
\]
οπότε, για $x>0$,
\[
\frac{f(x)}{x}=\frac{1}{f^2(x)+1}.
\]
Άρα
\[
{\lim_{x\to+\infty}\frac{f(x)}{x}=0}.
\]

Η $C_f$ δεν έχει οριζόντια ασύμπτωτη στο $+\infty$, διότι
\[
\lim_{x\to+\infty}f(x)=+\infty.
\]
Επίσης, αν υπήρχε πλάγια ασύμπτωτη $y=ax+\beta$, θα ήταν
\[
a=\lim_{x\to+\infty}\frac{f(x)}{x}=0,
\]
οπότε θα επρόκειτο για οριζόντια ασύμπτωτη, η οποία δεν υπάρχει. Συνεπώς η $C_f$ δεν έχει ασύμπτωτη στο $+\infty$.

\item Στο διάστημα $[0,2]$ είναι $f(x)\geq0$. Επίσης
\[
f(0)=0
\]
και
\[
f(2)=1,
\]
αφού
\[
f^3(2)+f(2)=2
\]
και ο αριθμός $1$ είναι η μοναδική λύση της εξίσωσης $u^3+u=2$.

Άρα το ζητούμενο εμβαδόν είναι
\[
E=\int_0^2 f(x)\d x.
\]
Θέτουμε
\[
x=f^{-1}(u)=u^3+u.
\]
Τότε
\[
\d x=(3u^2+1)\d u,
\]
ενώ για $x=0$ είναι $u=f(0)=0$ και για $x=2$ είναι $u=f(2)=1$. Επομένως
\begin{align*}
E
&=\int_0^1 u(3u^2+1)\d u\\
&=\int_0^1(3u^3+u)\d u\\
&=\left[\frac{3}{4}u^4+\frac{1}{2}u^2\right]_0^1\\
&=\frac{3}{4}+\frac{1}{2}\\
&={\frac{5}{4}}.
\end{align*}
\end{erwthma}
\end{thema}
