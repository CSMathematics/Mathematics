\begin{thema}{Δ}
Στο σχήμα της $C_{f'}$ σημειώνουμε τις πληροφορίες που θα χρησιμοποιήσουμε:

\noindent
\textbf{Προσοχή:} Το εμβαδόν ενός χωρίου είναι θετικό, αλλά το ορισμένο
ολοκλήρωμα είναι \emph{προσημασμένο}. Έτσι, το μεσαίο χωρίο δίνει
$\int_1^5 f'(x)\d x=-E_2=-5$, επειδή βρίσκεται κάτω από τον άξονα $x'x$.

\begin{erwthma}
\item Από το σχήμα παρατηρούμε ότι
\[
\begin{array}{lll}
f'(x)>0, & x\in(-1,1), & \text{άρα }f\nearrow[-1,1],\\
f'(x)<0, & x\in(1,5),  & \text{άρα }f\searrow[1,5],\\
f'(x)>0, & x\in(5,7),  & \text{άρα }f\nearrow[5,7].
\end{array}
\]
Επιπλέον, η $f'$ αλλάζει πρόσημο από θετικό σε αρνητικό στο $x=1$,
οπότε η $f$ παρουσιάζει τοπικό μέγιστο στο $x=1$. Στο $x=5$ η $f'$
αλλάζει πρόσημο από αρνητικό σε θετικό, οπότε η $f$ παρουσιάζει τοπικό
ελάχιστο.

Ο πίνακας μεταβολών είναι:
\begin{center}
\begin{tikzpicture}

colorC=maincolor!50!white,colorL=maincolor!20!white,

\tkzTabLine{z,+,z,-,z,+,z}
\tkzTabVar{-/{$f(-1)$},+/{$f(1)$},-/{$f(5)$},+/{$f(7)$}}
\end{tikzpicture}
\end{center}

Η πληροφορία ότι η $f'$ έχει ελάχιστη τιμή στο $x=3$ δεν σημαίνει ότι
η $f$ έχει εκεί ακρότατο. Πράγματι, $f'(3)<0$, άρα η $f$ εξακολουθεί
να είναι γνησίως φθίνουσα γύρω από το $x=3$.

\item Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού ισχύει
\[
f(b)-f(a)=\int_a^b f'(x)\d x.
\]
Επομένως, χρησιμοποιώντας τα προσημασμένα εμβαδά του σχήματος, έχουμε
\begin{align*}
f(1)-f(-1)
&=\int_{-1}^{1}f'(x)\d x
=E_1=2,
\end{align*}
οπότε
\[
{f(1)=2}.
\]

Στη συνέχεια,
\begin{align*}
f(5)-f(1)
&=\int_{1}^{5}f'(x)\d x
=-E_2=-5,
\end{align*}
άρα
\[
{f(5)=2-5=-3}.
\]

Τέλος,
\begin{align*}
f(7)-f(5)
&=\int_{5}^{7}f'(x)\d x
=E_3=4,
\end{align*}
οπότε
\[
{f(7)=-3+4=1}.
\]

Συνολικά, στα άκρα των διαστημάτων μονοτονίας έχουμε
\[
{f(-1)=0,\quad f(1)=2,\quad f(5)=-3,\quad f(7)=1}.
\]
Επειδή η $f$ είναι μονότονη σε καθένα από τα τρία επιμέρους διαστήματα,
η μεγαλύτερη από τις παραπάνω τιμές είναι το ολικό μέγιστο και η
μικρότερη το ολικό ελάχιστο. Άρα
\[
{\max_{[-1,7]}f=f(1)=2}
\qquad\text{και}\qquad
{\min_{[-1,7]}f=f(5)=-3}.
\]

\item Η $f$ είναι συνεχής στο $[1,5]$ και
\[
f(1)=2>0,\qquad f(5)=-3<0.
\]
Επομένως
\[
f(1)f(5)<0.
\]
Από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα
$\alpha\in(1,5)$ τέτοιο ώστε
\[
{f(\alpha)=0}.
\]

\item Θα εξετάσουμε χωριστά τα τρία διαστήματα μονοτονίας.
\begin{itemize}
\item Στο $[-1,1]$ η $f$ είναι γνησίως αύξουσα και $f(-1)=0$. Άρα
\[
f(x)>f(-1)=0,\qquad x\in(-1,1],
\]
οπότε η μοναδική ρίζα της στο διάστημα αυτό είναι η $x=-1$.

\item Στο $[1,5]$ γνωρίζουμε από το προηγούμενο ερώτημα ότι υπάρχει
ρίζα $\alpha\in(1,5)$. Επειδή η $f$ είναι γνησίως φθίνουσα στο
$[1,5]$, η ρίζα αυτή είναι μοναδική.

\item Στο $[5,7]$ η $f$ είναι συνεχής και
\[
f(5)=-3<0<1=f(7).
\]
Από το θεώρημα \eng{Bolzano} υπάρχει $\beta\in(5,7)$ με $f(\beta)=0$.
Επειδή η $f$ είναι γνησίως αύξουσα στο $[5,7]$, η ρίζα $\beta$ είναι
μοναδική.
\end{itemize}

Άρα οι ρίζες είναι
\[
x=-1,\qquad \alpha\in(1,5),\qquad \beta\in(5,7),
\]
και επομένως η εξίσωση $f(x)=0$ έχει στο $[-1,7]$
\[
{\text{ακριβώς τρεις ρίζες}}.
\]
\end{erwthma}
\end{thema}
