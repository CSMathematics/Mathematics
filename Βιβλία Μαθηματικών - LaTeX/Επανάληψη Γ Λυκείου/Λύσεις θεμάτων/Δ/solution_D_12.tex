%=========== ΛΥΣΗ - 12ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f'(x)
&=\left(-x+\sqrt{x^2+1}\right)'\\
&=-1+\frac{x}{\sqrt{x^2+1}}\\
&=\frac{x-\sqrt{x^2+1}}{\sqrt{x^2+1}}.
\end{align*}
Επειδή
\[
\sqrt{x^2+1}>|x|\geq x,
\]
έχουμε
\[
x-\sqrt{x^2+1}<0.
\]
Επομένως
\[
f'(x)<0
\]
για κάθε $x\in\mathbb{R}$ και άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $\mathbb{R}$.

\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
f(x)
&=\sqrt{x^2+1}-x\\
&=\frac{\left(\sqrt{x^2+1}-x\right)
\left(\sqrt{x^2+1}+x\right)}
{\sqrt{x^2+1}+x}\\
&=\frac{1}{\sqrt{x^2+1}+x}.
\end{align*}
Άρα
\begin{align*}
\lim_{x\to+\infty}f(x)
&=\lim_{x\to+\infty}
\frac{1}{\sqrt{x^2+1}+x}\\
&=0.
\end{align*}
Επομένως η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

\item Για $x<0$ είναι $|x|=-x$, οπότε
\begin{align*}
\lim_{x\to-\infty}f(x)
&=\lim_{x\to-\infty}
\left(-x+\sqrt{x^2+1}\right)\\
&=\lim_{x\to-\infty}
\left[-x-x\sqrt{1+\frac{1}{x^2}}\right]\\
&=+\infty.
\end{align*}

Για να εξετάσουμε αν η ευθεία $y=-2x$ είναι ασύμπτωτη, υπολογίζουμε
\begin{align*}
\lim_{x\to-\infty}\left[f(x)-(-2x)\right]
&=\lim_{x\to-\infty}
\left(x+\sqrt{x^2+1}\right)\\
&=\lim_{x\to-\infty}
\frac{1}{\sqrt{x^2+1}-x}\\
&=0.
\end{align*}
Επομένως η ευθεία
\[
{y=-2x}
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$.

\item Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f(x)f(-x)
&=\left(-x+\sqrt{x^2+1}\right)
\left(x+\sqrt{x^2+1}\right)\\
&=x^2+1-x^2\\
&=1.
\end{align*}

Θεωρούμε τη συνάρτηση
\[
g(x)=\frac{1}{1+f(x)}.
\]
Από τη σχέση $f(x)f(-x)=1$ και επειδή $f(x)>0$, έχουμε
\begin{align*}
g(-x)
&=\frac{1}{1+f(-x)}\\
&=\frac{1}{1+\frac{1}{f(x)}}\\
&=\frac{f(x)}{1+f(x)}\\
&=1-\frac{1}{1+f(x)}\\
&=1-g(x).
\end{align*}
Επομένως
\[
g(x)+g(-x)=1.
\]
Με την αλλαγή μεταβλητής $x=-t$ έχουμε
\[
\int_{-1}^{0}g(x)\d x=\int_0^1g(-t)\d t.
\]
Άρα
\begin{align*}
I
&=\int_{-1}^{1}g(x)\d x\\
&=\int_0^1g(-x)\d x+\int_0^1g(x)\d x\\
&=\int_0^1\left[g(-x)+g(x)\right]\d x\\
&=\int_0^1 1\d x\\
&={1}.
\end{align*}
\end{erwthma}
\end{thema}
