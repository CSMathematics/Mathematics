%=========== ΛΥΣΗ - 64ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,6)$, αφού είναι δύο φορές
παραγωγίσιμη στο $[0,6]$.

Στο $x=2$ η $f$ παρουσιάζει τοπικό μέγιστο και το $2$ είναι εσωτερικό
σημείο του πεδίου ορισμού. Άρα, από το θεώρημα \eng{Fermat},
\[
{f'(2)=0}.
\]

Ομοίως, στο $x=4$ η $f$ παρουσιάζει τοπικό ελάχιστο και το $4$ είναι
εσωτερικό σημείο του πεδίου ορισμού. Άρα, πάλι από το θεώρημα
\eng{Fermat},
\[
{f'(4)=0}.
\]

\item Το θεώρημα \eng{Rolle} για την $f$ στο $[0,6]$ δεν εφαρμόζεται,
διότι δεν ισχύει η βασική προϋπόθεση
\[
f(0)=f(6).
\]
Πράγματι,
\[
f(0)=2
\qquad\text{και}\qquad
f(6)=3.
\]

Θέλουμε τώρα να αποδείξουμε ότι η εξίσωση
\[
f(x)=2
\]
έχει τουλάχιστον δύο λύσεις στο $(0,6)$.

Θεωρούμε τη συνάρτηση
\[
g(x)=f(x)-2.
\]
Η $g$ είναι συνεχής στο $[0,6]$, ως διαφορά συνεχών συναρτήσεων.

Στο διάστημα $[2,4]$ έχουμε
\[
g(2)=f(2)-2=5-2=3>0
\]
και
\[
g(4)=f(4)-2=1-2=-1<0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\rho_1\in(2,4)
\]
τέτοιο ώστε
\[
g(\rho_1)=0\Leftrightarrow f(\rho_1)=2.
\]

Στο διάστημα $[4,6]$ έχουμε
\[
g(4)=-1<0
\]
και
\[
g(6)=f(6)-2=3-2=1>0.
\]
Άρα, ξανά από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\rho_2\in(4,6)
\]
τέτοιο ώστε
\[
g(\rho_2)=0\Leftrightarrow f(\rho_2)=2.
\]

Επομένως η εξίσωση $f(x)=2$ έχει τουλάχιστον δύο ρίζες
\[
{\rho_1\in(2,4)\quad\text{και}\quad \rho_2\in(4,6)}.
\]

\item Η $f$ είναι συνεχής στο $[0,2]$ και παραγωγίσιμη στο $(0,2)$.
Άρα εφαρμόζεται το Θεώρημα Μέσης Τιμής στο $[0,2]$. Επομένως υπάρχει
$\xi_1\in(0,2)$ τέτοιο ώστε
\[
f'(\xi_1)=\frac{f(2)-f(0)}{2-0}
=\frac{5-2}{2}
=\frac32.
\]
Άρα
\[
{\exists\,\xi_1\in(0,2):\ f'(\xi_1)=\frac32}.
\]

Ομοίως, η $f$ είναι συνεχής στο $[4,6]$ και παραγωγίσιμη στο $(4,6)$.
Με το Θεώρημα Μέσης Τιμής στο $[4,6]$ υπάρχει $\xi_2\in(4,6)$ τέτοιο ώστε
\[
f'(\xi_2)=\frac{f(6)-f(4)}{6-4}
=\frac{3-1}{2}
=1.
\]
Άρα
\[
{\exists\,\xi_2\in(4,6):\ f'(\xi_2)=1}.
\]

Μένει να δείξουμε ότι η εξίσωση $f'(x)=1$ έχει λύση και στο $(0,2)$.
Επειδή η $f$ είναι δύο φορές παραγωγίσιμη, η $f'$ είναι συνεχής.
Θεωρούμε τη συνάρτηση
\[
h(x)=f'(x)-1.
\]
Η $h$ είναι συνεχής στο $[\xi_1,2]$. Επιπλέον,
\[
h(\xi_1)=f'(\xi_1)-1=\frac32-1=\frac12>0,
\]
ενώ, από το Δ1,
\[
h(2)=f'(2)-1=0-1=-1<0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
x_1\in(\xi_1,2)\subset(0,2)
\]
τέτοιο ώστε
\[
h(x_1)=0\Leftrightarrow f'(x_1)=1.
\]

Συνεπώς
\[
{\exists\,x_1\in(0,2):\ f'(x_1)=1}.
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f'(x_1)=1,\quad x_1\in(0,2),
\]
και
\[
f'(\xi_2)=1,\quad \xi_2\in(4,6).
\]
Επειδή
\[
x_1<2<4<\xi_2,
\]
ισχύει $x_1<\xi_2$.

Η $f'$ είναι συνεχής στο $[x_1,\xi_2]$ και παραγωγίσιμη στο
$(x_1,\xi_2)$, αφού η $f$ είναι δύο φορές παραγωγίσιμη. Επίσης
\[
f'(x_1)=f'(\xi_2)=1.
\]
Άρα, εφαρμόζοντας το θεώρημα \eng{Rolle} στη συνάρτηση $f'$ στο
$[x_1,\xi_2]$, υπάρχει
\[
x_0\in(x_1,\xi_2)\subset(0,6)
\]
τέτοιο ώστε
\[
(f')'(x_0)=0.
\]
Δηλαδή
\[
{f''(x_0)=0}.
\]

Γεωμετρικά, στο σημείο της $C_f$ με τετμημένη $x_0$ ο ρυθμός μεταβολής
της κλίσης της εφαπτομένης μηδενίζεται. Το σημείο
\[
M(x_0,f(x_0))
\]
είναι \textbf{υποψήφιο σημείο καμπής} της $C_f$. Για να είναι πράγματι
σημείο καμπής, θα πρέπει επιπλέον να αλλάζει η κυρτότητα της $f$ γύρω
από το $x_0$.
\end{erwthma}
\end{thema}
