%=========== ΛΥΣΗ - 28ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η συνάρτηση
\[
g(x)=\left(f'(x)-f(x)\right)e^{-x}
\]
είναι παραγωγίσιμη στο $\mathbb{R}$ και
\begin{align*}
g'(x)
&=\left(f''(x)-f'(x)\right)e^{-x}
-\left(f'(x)-f(x)\right)e^{-x}\\
&=e^{-x}\left(f''(x)-2f'(x)+f(x)\right)\\
&=0.
\end{align*}
Άρα η $g$ είναι σταθερή στο $\mathbb{R}$. Επειδή
\[
g(0)=f'(0)-f(0)=1,
\]
έχουμε
\[
\left(f'(x)-f(x)\right)e^{-x}=1.
\]
Επομένως
\[
f'(x)-f(x)=e^x.
\]
Παρατηρούμε ότι
\[
\left(f(x)e^{-x}\right)'
=e^{-x}\left(f'(x)-f(x)\right)
=1.
\]
Άρα υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
f(x)e^{-x}=x+c.
\]
Για $x=0$ και $f(0)=0$ προκύπτει $c=0$. Συνεπώς
\[
{f(x)=xe^x},
\quad x\in\mathbb{R}.
\]

\item Για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=e^x(x+1).
\]
Επειδή $e^x>0$, η $f$ είναι γνησίως φθίνουσα στο $(-\infty,-1]$ και γνησίως αύξουσα στο $[-1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x=-1$, με τιμή
\[
{f(-1)=-\frac{1}{e}}.
\]

Επίσης
\[
f''(x)=e^x(x+2).
\]
Άρα η $f$ είναι κοίλη στο $(-\infty,-2]$ και κυρτή στο $[-2,+\infty)$. Η $C_f$ έχει μοναδικό σημείο καμπής το
\[
{K\left(-2,-\frac{2}{e^2}\right)}.
\]

\item Έστω $M(t,f(t))$ το σημείο επαφής. Η εφαπτομένη της $C_f$ στο $M$ έχει εξίσωση
\[
y-te^t=e^t(t+1)(x-t).
\]
Για να διέρχεται από το σημείο $A(2,0)$, πρέπει
\begin{align*}
-te^t&=e^t(t+1)(2-t)\\
\Leftrightarrow -t&=(t+1)(2-t)\\
\Leftrightarrow t^2-2t-2&=0.
\end{align*}
Η εξίσωση αυτή έχει ακριβώς δύο ρίζες
\[
{x_1=1-\sqrt{3}}
\quad\text{και}\quad
{x_2=1+\sqrt{3}}.
\]
Άρα από το $A$ άγονται ακριβώς δύο εφαπτόμενες προς τη $C_f$.

Οι κλίσεις τους είναι
\[
\lambda_1=f'(x_1)=e^{x_1}(x_1+1)
\]
και
\[
\lambda_2=f'(x_2)=e^{x_2}(x_2+1).
\]
Από τους τύπους του \eng{Vieta} έχουμε
\[
x_1+x_2=2
\quad\text{και}\quad
x_1x_2=-2.
\]
Επομένως
\begin{align*}
\lambda_1\lambda_2
&=e^{x_1+x_2}(x_1+1)(x_2+1)\\
&=e^2\left(x_1x_2+x_1+x_2+1\right)\\
&=e^2(-2+2+1)\\
&={e^2}.
\end{align*}

\item Θεωρούμε τη συνάρτηση
\[
h(x)=xe^x-x^2-1.
\]
Η ζητούμενη εξίσωση είναι ισοδύναμη με την $h(x)=0$.

Για $x\leq0$ έχουμε
\[
xe^x\leq0<x^2+1,
\]
άρα δεν υπάρχουν ρίζες.

Για $x>0$ είναι
\[
h'(x)=(x+1)e^x-2x.
\]
Αν $0<x<1$, τότε
\[
\frac{2x}{x+1}<1<e^x,
\]
ενώ, αν $x\geq1$, τότε
\[
\frac{2x}{x+1}<2<e^x.
\]
Άρα σε κάθε περίπτωση
\[
e^x>\frac{2x}{x+1},
\]
οπότε
\[
h'(x)>0.
\]
Επομένως η $h$ είναι γνησίως αύξουσα στο $(0,+\infty)$.

Τέλος,
\[
h(0)=-1<0
\quad\text{και}\quad
h(1)=e-2>0.
\]
Άρα, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει μία ρίζα στο $(0,1)$, η οποία είναι μοναδική λόγω της γνήσιας αύξησης της $h$.
\end{erwthma}
\end{thema}
