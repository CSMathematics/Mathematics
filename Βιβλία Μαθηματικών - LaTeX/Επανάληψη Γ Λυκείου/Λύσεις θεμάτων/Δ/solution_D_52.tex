\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
P'(x)
&=3x^2-6x-9\\
&=3(x+1)(x-3).
\end{align*}
Επομένως:
\begin{itemize}
\item $P'(x)>0$ για $x<-1$ ή $x>3$,
\item $P'(x)=0$ για $x=-1$ ή $x=3$,
\item $P'(x)<0$ για $-1<x<3$.
\end{itemize}
Άρα η $P$ είναι γνησίως αύξουσα στα διαστήματα $(-\infty,-1]$, $[3,+\infty)$ και γνησίως φθίνουσα στο $[-1,3]$. Επομένως παρουσιάζει ακριβώς ένα τοπικό μέγιστο στο $x=-1$ και ακριβώς ένα τοπικό ελάχιστο στο $x=3$.

Οι τιμές των ακροτάτων είναι
\[
P(-1)=\mu+5
\]
και
\[
P(3)=\mu-27.
\]

Επιπλέον,
\[
P''(x)=6x-6=6(x-1).
\]
Άρα:
\begin{itemize}
\item $P''(x)<0$ για $x<1$,
\item $P''(1)=0$,
\item $P''(x)>0$ για $x>1$.
\end{itemize}
Επομένως η $P$ είναι κοίλη στο $(-\infty,1]$ και κυρτή στο $[1,+\infty)$. Συνεπώς η $C_P$ έχει ακριβώς ένα σημείο καμπής
\[
{K(1,P(1))\equiv K(1,\mu-11)}.
\]

\item Το σημείο καμπής της $C_P$ βρίσκεται πάνω στον άξονα $x'x$, αν και μόνο αν
\[
P(1)=0.
\]
Άρα
\[
\mu-11=0
\Leftrightarrow
{\mu=11}.
\]
\end{erwthma}

Για $\mu=11$ είναι
\[
P(x)=x^3-3x^2-9x+11.
\]

\begin{erwthma}[resume]
\item
\begin{alist}
\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
P(2-x)
&=(2-x)^3-3(2-x)^2-9(2-x)+11\\
&=-x^3+3x^2+9x-11\\
&=-P(x).
\end{align*}
Άρα
\[
{P(2-x)=-P(x)}
\]
για κάθε $x\in\mathbb{R}$.

Η σχέση αυτή γράφεται
\[
P(1-h)=-P(1+h),
\]
οπότε η γραφική παράσταση της $P$ έχει κέντρο συμμετρίας το σημείο
\[
{K(1,0)},
\]
το οποίο είναι και το σημείο καμπής της.

\item Έχουμε
\begin{align*}
P(x)
&=x^3-3x^2-9x+11\\
&=(x-1)(x^2-2x-11).
\end{align*}
Επομένως
\begin{align*}
P(x)=0
&\Leftrightarrow
x-1=0
\quad\text{ή}\quad
x^2-2x-11=0\\
&\Leftrightarrow
x=1
\quad\text{ή}\quad
x=1\pm2\sqrt3.
\end{align*}
Άρα η εξίσωση $P(x)=0$ έχει
\[
{\text{ακριβώς τρεις διαφορετικές πραγματικές ρίζες}}.
\]
\end{alist}

\item Θέτουμε
\[
I=\int_0^2\left(P(x)\right)^{2025}\d x.
\]
Με την αλλαγή μεταβλητής $u=2-x$ έχουμε
\[
I=\int_0^2\left(P(2-u)\right)^{2025}\d u.
\]
Επειδή
\[
P(2-u)=-P(u)
\]
και ο αριθμός $2025$ είναι περιττός, προκύπτει
\begin{align*}
I
&=\int_0^2\left[-P(u)\right]^{2025}\d u\\
&=-\int_0^2\left(P(u)\right)^{2025}\d u\\
&=-I.
\end{align*}
Άρα
\[
2I=0
\]
και επομένως
\[
{\int_0^2\left(P(x)\right)^{2025}\d x=0}.
\]
\end{erwthma}
\end{thema}
