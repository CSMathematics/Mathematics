%=========== ΛΥΣΗ - 32ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x>0$ έχουμε
\[
\left(xf'(x)\right)'=xf''(x)+f'(x)=-2.
\]
Επομένως
\[
xf'(x)=-2x+c_1,
\]
όπου $c_1\in\mathbb{R}$. Για $x=1$ και με δεδομένο ότι $f'(1)=-1$, παίρνουμε
\[
-1=-2+c_1
\Leftrightarrow
c_1=1.
\]
Άρα
\[
xf'(x)=-2x+1
\Leftrightarrow
f'(x)=\frac{1}{x}-2.
\]
Ολοκληρώνοντας, προκύπτει
\[
f(x)=\ln x-2x+c_2,
\]
όπου $c_2\in\mathbb{R}$. Επειδή $f(1)=1$, έχουμε
\[
1=\ln1-2+c_2
\Leftrightarrow
c_2=3.
\]
Επομένως
\[
{f(x)=\ln x-2x+3},\quad x>0.
\]

\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
f'(x)=\frac{1}{x}-2=\frac{1-2x}{x}.
\]
Επειδή $x>0$, έχουμε:
\begin{itemize}
\item $f'(x)>0\Leftrightarrow 0<x<\dfrac{1}{2}$,
\item $f'(x)=0\Leftrightarrow x=\dfrac{1}{2}$,
\item $f'(x)<0\Leftrightarrow x>\dfrac{1}{2}$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$0$,$\frac{1}{2}$,$+\infty$}%
\tkzTabLine{d, +, z, -, }
\tkzTabVar{D-/{-\infty},+/{2-\ln2},-/{-\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\left(0,\frac{1}{2}\right]$ και \textbf{γνησίως φθίνουσα} στο $\left[\frac{1}{2},+\infty\right)$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=\frac{1}{2}$, με τιμή
\[
{f\left(\frac{1}{2}\right)=2-\ln2}.
\]

Επιπλέον,
\[
\lim_{x\to0^+}f(x)
=\lim_{x\to0^+}\left(\ln x-2x+3\right)
=-\infty
\]
και
\[
\lim_{x\to+\infty}f(x)
=\lim_{x\to+\infty}\left(\ln x-2x+3\right)
=-\infty.
\]
Επειδή
\[
f\left(\frac{1}{2}\right)=2-\ln2>0,
\]
ενώ
\[
f\left(e^{-3}\right)
=-3-2e^{-3}+3
=-2e^{-3}<0
\]
και
\[
f(2)=\ln2-1<0,
\]
από το Θεώρημα Ενδιάμεσων Τιμών υπάρχει μία τουλάχιστον ρίζα
\[
\rho_1\in\left(e^{-3},\frac{1}{2}\right)
\]
και μία τουλάχιστον ρίζα
\[
\rho_2\in\left(\frac{1}{2},2\right).
\]
Οι ρίζες αυτές είναι μοναδικές, διότι η $f$ είναι γνησίως αύξουσα στο $\left(0,\frac{1}{2}\right]$ και γνησίως φθίνουσα στο $\left[\frac{1}{2},+\infty\right)$. Άρα η εξίσωση $f(x)=0$ έχει ακριβώς δύο ρίζες, για τις οποίες
\[
{\rho_1<\frac{1}{2}<\rho_2}.
\]

\item Για κάθε $x>0$ είναι
\[
f''(x)=-\frac{1}{x^2}<0.
\]
Επομένως η $f$ είναι \textbf{κοίλη} στο $(0,+\infty)$.

Στο σημείο με τετμημένη $x_0=1$ έχουμε
\[
f(1)=1
\quad\text{και}\quad
f'(1)=-1.
\]
Άρα η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $A(1,1)$ είναι
\begin{align*}
y-f(1)&=f'(1)(x-1)\\
y-1&=-(x-1),
\end{align*}
δηλαδή
\[
{y=-x+2}.
\]

Επειδή η $f$ είναι κοίλη, η γραφική της παράσταση βρίσκεται κάτω από την εφαπτομένη της στο $x_0=1$. Επομένως, για κάθε $x>0$,
\[
f(x)\leq -x+2.
\]
Άρα
\begin{align*}
\ln x-2x+3&\leq-x+2\\
\Leftrightarrow\quad
{\ln x\leq x-1}.
\end{align*}
Η ισότητα ισχύει μόνο για $x=1$.

\item Από τη μονοτονία και τις ρίζες της $f$ γνωρίζουμε ότι
\[
f(x)>0
\]
για κάθε $x\in(\rho_1,\rho_2)$. Επομένως
\[
\int_{\rho_1}^{\rho_2}f(x)\d x>0.
\]

Από την ανισότητα του προηγούμενου ερωτήματος έχουμε, για κάθε $x>0$,
\[
\ln x\leq x-1.
\]
Άρα
\[
f(x)=\ln x-2x+3\leq2-x.
\]
Η ισότητα ισχύει μόνο για $x=1$, οπότε, ολοκληρώνοντας στο διάστημα $[\rho_1,\rho_2]$, παίρνουμε
\begin{align*}
\int_{\rho_1}^{\rho_2}f(x)\d x
&<\int_{\rho_1}^{\rho_2}(2-x)\d x\\
&=\left[2x-\frac{x^2}{2}\right]_{\rho_1}^{\rho_2}\\
&=2(\rho_2-\rho_1)-\frac{\rho_2^2-\rho_1^2}{2}\\
&=\frac{1}{2}(\rho_2-\rho_1)
\left(4-\rho_1-\rho_2\right).
\end{align*}
Συνεπώς
\[
{
0<\int_{\rho_1}^{\rho_2}f(x)\d x
<\frac{1}{2}(\rho_2-\rho_1)
\left(4-\rho_1-\rho_2\right)}.
\]
\end{erwthma}
\end{thema}
