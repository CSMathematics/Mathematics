\begin{thema}{Δ}
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση
\[
\varphi(x)=x-\ln(x+1),\quad x>-1.
\]
Για κάθε $x>-1$ είναι
\[
\varphi'(x)=1-\frac{1}{x+1}=\frac{x}{x+1}.
\]
Άρα η $\varphi$ είναι γνησίως φθίνουσα στο $(-1,0]$ και γνησίως αύξουσα στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
\varphi(0)=0.
\]
Συνεπώς
\[
{\ln(x+1)\leq x}
\]
για κάθε $x>-1$, με ισότητα μόνο για $x=0$.

Ομοίως, θεωρούμε τη συνάρτηση
\[
\psi(x)=e^x-x-1,\quad x\in\mathbb{R}.
\]
Έχουμε
\[
\psi'(x)=e^x-1.
\]
Άρα η $\psi$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
\psi(0)=0.
\]
Συνεπώς
\[
{x\leq e^x-1}
\]
για κάθε $x\in\mathbb{R}$, με ισότητα μόνο για $x=0$.

Άρα, για κάθε $x>-1$,
\[
{g(x)\leq x\leq f(x)}.
\]

Αν οι $C_f$ και $C_g$ έχουν κοινό σημείο με τετμημένη $x>-1$, τότε
\[
f(x)=g(x).
\]
Από τις προηγούμενες ανισότητες προκύπτει
\[
g(x)\leq x\leq f(x),
\]
οπότε αναγκαστικά
\[
f(x)=g(x)=x.
\]
Η ισότητα συμβαίνει μόνο για $x=0$. Άρα οι δύο γραφικές παραστάσεις έχουν μοναδικό κοινό σημείο
\[
{O(0,0)}.
\]

Επιπλέον,
\[
f'(x)=e^x
\quad\text{και}\quad
g'(x)=\frac{1}{x+1},
\]
οπότε
\[
f'(0)=g'(0)=1.
\]
Συνεπώς στο $O(0,0)$ οι δύο καμπύλες δέχονται κοινή εφαπτομένη
\[
{y=x}.
\]

\item Για κάθε $x\in(0,1]$ είναι
\[
f(x)>g(x).
\]
Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_0^1\left[f(x)-g(x)\right]\d x\\
&=\int_0^1\left[e^x-1-\ln(x+1)\right]\d x\\
&=\left[
e^x-x-(x+1)\ln(x+1)+(x+1)
\right]_0^1\\
&={e-1-2\ln2}.
\end{align*}

\item Για τη συνέχεια της $h$ στο $0$ έχουμε
\begin{align*}
\lim_{x\to0}h(x)
&=\lim_{x\to0}\frac{e^x-1}{\ln(x+1)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}e^x(x+1)\\
&=1\\
&=h(0).
\end{align*}
Άρα η $h$ είναι συνεχής στο $x_0=0$.

Από τον ορισμό της παραγώγου έχουμε
\begin{align*}
h'(0)
&=\lim_{x\to0}\frac{h(x)-h(0)}{x}\\
&=\lim_{x\to0}
\frac{e^x-1-\ln(x+1)}
{x\ln(x+1)}.
\end{align*}
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
h'(0)
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}
\frac{e^x-\frac{1}{x+1}}
{\ln(x+1)+\frac{x}{x+1}}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}
\frac{e^x+\frac{1}{(x+1)^2}}
{\frac{1}{x+1}+\frac{1}{(x+1)^2}}\\
&={1}.
\end{align*}
Επομένως η $h$ είναι παραγωγίσιμη στο $x_0=0$.

\item Για $x\neq0$ γράφουμε
\begin{align*}
\frac{1}{g(x)}-\frac{1}{f(x)}
&=\frac{f(x)-g(x)}{f(x)g(x)}\\
&=\frac{\frac{f(x)}{g(x)}-1}{f(x)}\\
&=\frac{h(x)-h(0)}{x}\cdot\frac{x}{f(x)}.
\end{align*}
Από το προηγούμενο ερώτημα έχουμε
\[
\lim_{x\to0}\frac{h(x)-h(0)}{x}=h'(0)=1,
\]
ενώ
\[
\lim_{x\to0}\frac{x}{f(x)}
=\lim_{x\to0}\frac{x}{e^x-1}
=1.
\]
Επομένως
\[
{
\lim_{x\to0}
\left(
\frac{1}{g(x)}-\frac{1}{f(x)}
\right)=1}.
\]
\end{erwthma}
\end{thema}
