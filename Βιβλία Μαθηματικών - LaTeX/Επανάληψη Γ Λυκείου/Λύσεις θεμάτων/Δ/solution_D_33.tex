%=========== ΛΥΣΗ - 33ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=e^x+1>0.
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Επειδή
\[
f(0)=e^0+0-1=0,
\]
ο αριθμός $0$ είναι ρίζα της εξίσωσης $f(x)=0$. Η ρίζα αυτή είναι μοναδική, λόγω της γνησίως αύξουσας μονοτονίας της $f$.

Για να ορίζεται η παράσταση $f(\ln x)$, πρέπει $x>0$. Επίσης
\[
f(1)=e.
\]
Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\begin{align*}
f\left(f(\ln x)+1\right)>e
&\Leftrightarrow
f\left(f(\ln x)+1\right)>f(1)\\
&\Leftrightarrow
f(\ln x)+1>1\\
&\Leftrightarrow
f(\ln x)>0\\
&\Leftrightarrow
f(\ln x)>f(0)\\
&\Leftrightarrow
\ln x>0\\
&\Leftrightarrow
{x>1}.
\end{align*}

\item Για το πρώτο όριο έχουμε
\[
f(x)-2x=e^x-x-1.
\]
Επομένως, εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0}\frac{f(x)-2x}{x^2}
&=\lim_{x\to0}\frac{e^x-x-1}{x^2}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{e^x-1}{2x}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{e^x}{2}\\
&={\frac{1}{2}}.
\end{align*}

Για το δεύτερο όριο είναι
\begin{align*}
\frac{xf(x)-e^x}{f(x)+e^{x-1}}
&=\frac{x(e^x+x-1)-e^x}{e^x+x-1+e^{x-1}}\\
&=\frac{(x-1)e^x+x^2-x}
{e^x\left(1+\frac{1}{e}\right)+x-1}\\
&=\frac{x-1+\frac{x^2-x}{e^x}}
{1+\frac{1}{e}+\frac{x-1}{e^x}}.
\end{align*}
Επειδή
\[
\lim_{x\to+\infty}\frac{x^2-x}{e^x}=0
\quad\text{και}\quad
\lim_{x\to+\infty}\frac{x-1}{e^x}=0,
\]
προκύπτει
\[
{
\lim_{x\to+\infty}
\frac{xf(x)-e^x}{f(x)+e^{x-1}}
=+\infty}.
\]

\item Έχουμε
\[
\lim_{x\to-\infty}f(x)
=\lim_{x\to-\infty}(e^x+x-1)
=-\infty
\]
και
\[
\lim_{x\to+\infty}f(x)=+\infty.
\]
Η $f$ είναι συνεχής και γνησίως αύξουσα στο $\mathbb{R}$, οπότε
\[
f(\mathbb{R})=\mathbb{R}
\]
και συνεπώς αντιστρέφεται στο $\mathbb{R}$.

Θέλουμε να λύσουμε την εξίσωση
\[
f(x)=f^{-1}(x).
\]
Αν $f(x)>x$, τότε, επειδή η $f$ είναι γνησίως αύξουσα,
\[
f(f(x))>f(x)>x,
\]
ενώ από τη δοσμένη εξίσωση, εφαρμόζοντας την $f$ στα δύο μέλη, θα είχαμε
\[
f(f(x))=x,
\]
άτοπο. Ομοίως, η περίπτωση $f(x)<x$ οδηγεί σε άτοπο. Άρα
\[
f(x)=x.
\]
Επομένως
\begin{align*}
f(x)=f^{-1}(x)
&\Leftrightarrow f(x)=x\\
&\Leftrightarrow e^x+x-1=x\\
&\Leftrightarrow e^x=1\\
&\Leftrightarrow {x=0}.
\end{align*}

Για τον υπολογισμό του ολοκληρώματος θέτουμε
\[
x=f(u)=e^u+u-1.
\]
Τότε
\[
\d x=(e^u+1)\d u.
\]
Για $x=0$ είναι $u=f^{-1}(0)=0$, ενώ για $x=e$ είναι
\[
u=f^{-1}(e)=1,
\]
αφού $f(1)=e$. Άρα
\begin{align*}
\int_0^e f^{-1}(x)\d x
&=\int_0^1u(e^u+1)\d u\\
&=\int_0^1ue^u\d u+\int_0^1u\d u\\
&=\left[ue^u-e^u\right]_0^1
+\left[\frac{u^2}{2}\right]_0^1\\
&=1+\frac{1}{2}\\
&={\frac{3}{2}}.
\end{align*}

\item Από το προηγούμενο ερώτημα, οι γραφικές παραστάσεις $C_f$ και $C_{f^{-1}}$ τέμνονται μόνο στο σημείο
\[
O(0,0).
\]

Τα κοινά σημεία της $C_f$ και της ευθείας
\[
y=e+1-x
\]
προκύπτουν από την εξίσωση
\[
f(x)=e+1-x.
\]
Ισοδύναμα,
\[
e^x+2x=e+2.
\]
Η συνάρτηση
\[
h(x)=e^x+2x
\]
είναι γνησίως αύξουσα, αφού
\[
h'(x)=e^x+2>0.
\]
Επειδή $h(1)=e+2$, η μοναδική λύση είναι $x=1$. Άρα το κοινό σημείο είναι
\[
A(1,e).
\]
Λόγω συμμετρίας των $C_f$ και $C_{f^{-1}}$ ως προς την ευθεία $y=x$, η $C_{f^{-1}}$ τέμνει την ευθεία $x+y=e+1$ στο σημείο
\[
B(e,1).
\]

Για $0<x\leq1$ είναι $f(x)>f^{-1}(x)$, ενώ για $1\leq x<e$ η ευθεία $y=e+1-x$ βρίσκεται πάνω από τη $C_{f^{-1}}$. Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_0^1\left[f(x)-f^{-1}(x)\right]\d x\\
&\quad+\int_1^e\left[e+1-x-f^{-1}(x)\right]\d x\\
&=\int_0^1f(x)\d x
+\int_1^e(e+1-x)\d x
-\int_0^e f^{-1}(x)\d x.
\end{align*}
Έχουμε
\begin{align*}
\int_0^1f(x)\d x
&=\int_0^1(e^x+x-1)\d x\\
&=\left[e^x+\frac{x^2}{2}-x\right]_0^1\\
&=e-\frac{3}{2},
\end{align*}
ενώ
\begin{align*}
\int_1^e(e+1-x)\d x
&=\left[(e+1)x-\frac{x^2}{2}\right]_1^e\\
&=\frac{e^2-1}{2}.
\end{align*}
Από το προηγούμενο ερώτημα γνωρίζουμε ότι
\[
\int_0^e f^{-1}(x)\d x=\frac{3}{2}.
\]
Συνεπώς
\begin{align*}
E
&=e-\frac{3}{2}+\frac{e^2-1}{2}-\frac{3}{2}\\
&={\frac{e^2+2e-7}{2}}.
\end{align*}
\end{erwthma}
\end{thema}
