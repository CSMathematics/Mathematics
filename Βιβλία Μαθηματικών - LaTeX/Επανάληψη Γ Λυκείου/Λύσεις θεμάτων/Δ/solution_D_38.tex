%=========== ΛΥΣΗ - 38ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\neq0$ έχουμε
\[
\left|f(x)\right|
=\left|x^2\hm\left(\frac{1}{x}\right)\right|
\leq x^2.
\]
Επομένως
\[
0\leq|f(x)|\leq x^2.
\]
Επειδή
\[
\lim_{x\to0}x^2=0,
\]
από το κριτήριο παρεμβολής προκύπτει
\[
\lim_{x\to0}|f(x)|=0,
\]
άρα
\[
\lim_{x\to0}f(x)=0=f(0).
\]
Συνεπώς η $f$ είναι συνεχής στο $x_0=0$.

Για την παραγωγισιμότητα της $f$ στο $0$ έχουμε
\begin{align*}
f'(0)
&=\lim_{x\to0}\frac{f(x)-f(0)}{x-0}\\
&=\lim_{x\to0}\frac{x^2\hm\left(\frac{1}{x}\right)}{x}\\
&=\lim_{x\to0}x\hm\left(\frac{1}{x}\right).
\end{align*}
Όμως
\[
\left|x\hm\left(\frac{1}{x}\right)\right|\leq|x|
\]
και $\lim\limits_{x\to0}|x|=0$. Επομένως, από το κριτήριο παρεμβολής,
\[
{f'(0)=0}.
\]
Άρα η $f$ είναι παραγωγίσιμη στο $x_0=0$.

\item Για κάθε $x\neq0$ είναι
\begin{align*}
f'(x)
&=\left[x^2\hm\left(\frac{1}{x}\right)\right]'\\
&=2x\hm\left(\frac{1}{x}\right)
+x^2\syn\left(\frac{1}{x}\right)
\left(-\frac{1}{x^2}\right)\\
&=2x\hm\left(\frac{1}{x}\right)
-\syn\left(\frac{1}{x}\right).
\end{align*}
Σε συνδυασμό με το προηγούμενο ερώτημα, προκύπτει
\[
{
f'(x)=
\begin{cases}
2x\hm\left(\dfrac{1}{x}\right)-\syn\left(\dfrac{1}{x}\right),
&x\neq0,\\[2mm]
0,&x=0.
\end{cases}}
\]

\item Θεωρούμε τις ακολουθίες
\[
x_n=\frac{1}{2n\pi}
\quad\text{και}\quad
y_n=\frac{1}{(2n+1)\pi},
\quad n\in\mathbb{N}^*.
\]
Τότε
\[
\lim_{n\to+\infty}x_n
=\lim_{n\to+\infty}y_n
=0.
\]
Όμως
\[
\syn\left(\frac{1}{x_n}\right)
=\syn(2n\pi)
=1,
\]
ενώ
\[
\syn\left(\frac{1}{y_n}\right)
=\syn((2n+1)\pi)
=-1.
\]
Επειδή προκύπτουν δύο διαφορετικά διαδοχικά όρια, το όριο
\[
{\lim_{x\to0}\syn\left(\frac{1}{x}\right)}
\]
δεν υπάρχει.

Επιπλέον,
\[
\left|2x\hm\left(\frac{1}{x}\right)\right|\leq2|x|,
\]
οπότε
\[
\lim_{x\to0}2x\hm\left(\frac{1}{x}\right)=0.
\]
Αν υπήρχε το $\lim\limits_{x\to0}f'(x)$, τότε από τη σχέση
\[
\syn\left(\frac{1}{x}\right)
=2x\hm\left(\frac{1}{x}\right)-f'(x)
\]
θα υπήρχε και το $\lim\limits_{x\to0}\syn\left(\frac{1}{x}\right)$, άτοπο. Άρα το $\lim\limits_{x\to0}f'(x)$ δεν υπάρχει και, επειδή $f'(0)=0$, η $f'$ δεν είναι συνεχής στο $x_0=0$.

\item Η $f'$ είναι συνεχής στο διάστημα $\left[\frac{1}{\pi},\frac{2}{\pi}\right]$. Επομένως, από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού, έχουμε
\begin{align*}
\int_{1/\pi}^{2/\pi}f'(x)\d x
&=f\left(\frac{2}{\pi}\right)
-f\left(\frac{1}{\pi}\right)\\
&=\frac{4}{\pi^2}
\hm\left(\frac{\pi}{2}\right)
-\frac{1}{\pi^2}\hm\pi\\
&={\frac{4}{\pi^2}}.
\end{align*}
\end{erwthma}
\end{thema}
