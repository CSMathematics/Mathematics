\begin{thema}{Δ}
\begin{erwthma}
\item Από το σχήμα έχουμε
\[
f''(x)=2>0,\quad x\in(-2,1),
\]
άρα η $f$ είναι \textbf{κυρτή} στο $[-2,1]$.

Επίσης,
\[
f''(x)=-1<0,\quad x\in(1,4),
\]
άρα η $f$ είναι \textbf{κοίλη} στο $[1,4]$.

Στο $x=1$ αλλάζει η κυρτότητα της $f$, από κυρτή σε κοίλη. Επειδή
$f(1)=3$, η $C_f$ έχει σημείο καμπής το
\[
{A(1,3)}.
\]

Στον πίνακα φαίνεται συνοπτικά η κυρτότητα:
\begin{center}

{$-2$,$1$,$4$}%
\tkzTabLine{,+,d,-,}
\tkzTabLine{,\rotatebox[origin=c]{180}{$\curvearrowleft$},t,\curvearrowright,}
\end{tikzpicture}
\end{center}

\item Για $x\in[-2,1]$ έχουμε
\[
f''(x)=2.
\]
Άρα
\[
f'(x)=2x+c_1.
\]
Από τη συνθήκη $f'(1)=0$ βρίσκουμε
\[
0=2+c_1\Rightarrow c_1=-2.
\]
Επομένως
\[
f'(x)=2x-2=2(x-1),\quad x\in[-2,1].
\]

Για $x\in[1,4]$ έχουμε
\[
f''(x)=-1.
\]
Άρα
\[
f'(x)=-x+c_2.
\]
Από τη συνθήκη $f'(1)=0$ βρίσκουμε
\[
0=-1+c_2\Rightarrow c_2=1.
\]
Επομένως
\[
f'(x)=-x+1=-(x-1),\quad x\in[1,4].
\]

Συνολικά,
\[
{
f'(x)=
\begin{cases}
2x-2, & -2\leq x\leq 1,\\
-x+1, & 1\leq x\leq 4.
\end{cases}}
\]

\item Μελετούμε το πρόσημο της $f'$.

Για $x\in[-2,1)$ είναι
\[
f'(x)=2(x-1)<0.
\]
Για $x\in(1,4]$ είναι
\[
f'(x)=-(x-1)<0.
\]
Επίσης $f'(1)=0$.

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $[-2,1]$ και
\textbf{γνησίως φθίνουσα} στο $[1,4]$. Επομένως είναι γνησίως
φθίνουσα σε όλο το $[-2,4]$.

Ο πίνακας μεταβολών είναι:

Στο $x=1$ έχουμε $f'(1)=0$, αλλά η $f'$ δεν αλλάζει πρόσημο. Άρα στο
$x=1$ η $f$ \textbf{δεν} παρουσιάζει ακρότατο.

Αφού η $f$ είναι γνησίως φθίνουσα στο $[-2,4]$, παρουσιάζει:
\begin{itemize}
\item ολικό μέγιστο στο αριστερό άκρο $x=-2$,
\item ολικό ελάχιστο στο δεξί άκρο $x=4$.
\end{itemize}
Οι αντίστοιχες τιμές θα υπολογιστούν στο επόμενο ερώτημα.

\item Για $x\in[-2,1]$ έχουμε
\[
f'(x)=2x-2.
\]
Άρα
\[
f(x)=x^2-2x+c_3.
\]
Από τη συνθήκη $f(1)=3$ παίρνουμε
\[
3=1-2+c_3\Rightarrow c_3=4.
\]
Οπότε
\[
f(x)=x^2-2x+4={(x-1)}^2+3,\quad x\in[-2,1].
\]

Για $x\in[1,4]$ έχουμε
\[
f'(x)=-x+1.
\]
Άρα
\[
f(x)=-\frac{x^2}{2}+x+c_4.
\]
Από τη συνθήκη $f(1)=3$ παίρνουμε
\[
3=-\frac12+1+c_4\Rightarrow c_4=\frac52.
\]
Οπότε
\[
f(x)=-\frac{x^2}{2}+x+\frac52
=3-\frac{{(x-1)}^2}{2},\quad x\in[1,4].
\]

Συνολικά,
\[
{
f(x)=
\begin{cases}
{(x-1)}^2+3, & -2\leq x\leq 1,\\[1mm]
3-\dfrac{{(x-1)}^2}{2}, & 1\leq x\leq 4.
\end{cases}}
\]

Άρα
\[
f(-2)={(-3)}^2+3=12
\qquad\text{και}\qquad
f(4)=3-\frac{9}{2}=-\frac32.
\]
Επομένως
\[
{\max_{[-2,4]}f=f(-2)=12}
\qquad\text{και}\qquad
{\min_{[-2,4]}f=f(4)=-\frac32}.
\]

Τέλος,
\begin{align*}
\int_{-2}^{4} f(x)\d x
&=\int_{-2}^{1}\left[{(x-1)}^2+3\right]\d x
+\int_{1}^{4}\left[3-\frac{{(x-1)}^2}{2}\right]\d x.
\end{align*}

Για το πρώτο ολοκλήρωμα θέτουμε $u=x-1$. Όταν $x=-2$, τότε $u=-3$,
ενώ όταν $x=1$, τότε $u=0$. Έτσι,
\begin{align*}
\int_{-2}^{1}\left[{(x-1)}^2+3\right]\d x
&=\int_{-3}^{0}(u^2+3)\d u\\
&=\left.\left(\frac{u^3}{3}+3u\right)\right|_{-3}^{0}\\
&=18.
\end{align*}

Για το δεύτερο ολοκλήρωμα θέτουμε πάλι $u=x-1$. Όταν $x=1$, τότε
$u=0$, ενώ όταν $x=4$, τότε $u=3$. Έτσι,
\begin{align*}
\int_{1}^{4}\left[3-\frac{{(x-1)}^2}{2}\right]\d x
&=\int_{0}^{3}\left(3-\frac{u^2}{2}\right)\d u\\
&=\left.\left(3u-\frac{u^3}{6}\right)\right|_{0}^{3}\\
&=\frac{9}{2}.
\end{align*}

Επομένως
\[
{\int_{-2}^{4} f(x)\d x=18+\frac92=\frac{45}{2}}.
\]
\end{erwthma}
\end{thema}
