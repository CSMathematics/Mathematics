\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f''(x)
&=g'(x)-\left(\frac{x}{x^2+1}\right)'\\
&=\frac{1}{x^2+1}
-\frac{1-x^2}{(x^2+1)^2}\\
&=\frac{x^2+1-1+x^2}{(x^2+1)^2}\\
&=\frac{2x^2}{(x^2+1)^2}\geq0.
\end{align*}
Άρα η $f$ είναι \textbf{κυρτή} στο $\mathbb{R}$.

Μάλιστα, η $f'$ είναι γνησίως αύξουσα στο $\mathbb{R}$, αφού $f''(x)>0$ για κάθε $x\neq0$. Επομένως η $f$ είναι αυστηρά κυρτή στο $\mathbb{R}$.

\item Επειδή η $f$ είναι αυστηρά κυρτή, η γραφική της παράσταση βρίσκεται αυστηρά πάνω από κάθε εφαπτομένη της, εκτός από το σημείο επαφής. Άρα, για $u\neq x$,
\[
f(u)>f(x)+f'(x)(u-x).
\]

Για κάθε $x>0$ ισχύει
\[
\hm x<x,
\]
οπότε θέτοντας $u=\hm x$ παίρνουμε
\[
f(\hm x)>f(x)+f'(x)(\hm x-x).
\]
Επομένως
\[
{
f(x)-f(\hm x)
<(x-\hm x)f'(x)}
\]
για κάθε $x>0$.

\item Από τις σχέσεις
\[
g'(x)=\frac{1}{x^2+1}
\quad\text{και}\quad
g(0)=0
\]
προκύπτει
\[
g(x)=\arctan x.
\]
Πράγματι, η συνάρτηση $\arctan x$ έχει την ίδια παράγωγο και την ίδια τιμή στο $0$. Επομένως
\[
f'(x)=\arctan x-\frac{x}{x^2+1}.
\]
Η $f'$ είναι περιττή και γνησίως αύξουσα, με $f'(0)=0$. Άρα
\[
f'(x)>0
\]
για κάθε $x>0$.

Θεωρούμε τη συνάρτηση
\[
H(t)=f(t)-f\left(\frac{1}{t}\right),\quad t\geq1.
\]
Για κάθε $t>1$ είναι
\begin{align*}
H'(t)
&=f'(t)+\frac{1}{t^2}f'\left(\frac{1}{t}\right)>0.
\end{align*}
Άρα η $H$ είναι γνησίως αύξουσα στο $[1,+\infty)$.

Η δοσμένη ανίσωση γράφεται
\begin{align*}
f(x^2+1)-f(2)
&<f\left(\frac{1}{x^2+1}\right)-f\left(\frac{1}{2}\right)\\
\Leftrightarrow
H(x^2+1)&<H(2).
\end{align*}
Επειδή $x^2+1\geq1$ και η $H$ είναι γνησίως αύξουσα, έχουμε
\begin{align*}
H(x^2+1)<H(2)
&\Leftrightarrow x^2+1<2\\
&\Leftrightarrow x^2<1\\
&\Leftrightarrow {-1<x<1}.
\end{align*}

\item Θεωρούμε τη συνεχή συνάρτηση
\[
\varphi(x)=x(x^2+1)f(x)-1,\quad x\in[0,1].
\]
Έχουμε
\[
\varphi(0)=-1<0.
\]

Θα αποδείξουμε ότι υπάρχει $c\in(0,1)$ τέτοιο ώστε $\varphi(c)>0$. Αν, αντίθετα, ίσχυε
\[
\varphi(x)\leq0
\]
για κάθε $x\in(0,1)$, τότε
\[
x(x^2+1)f(x)\leq1,
\]
άρα
\[
xf(x)\leq\frac{1}{x^2+1}.
\]
Ολοκληρώνοντας στο $[0,1]$, θα παίρναμε
\begin{align*}
\int_0^1xf(x)\d x
&\leq\int_0^1\frac{1}{x^2+1}\d x\\
&=\left[\arctan x\right]_0^1\\
&=\frac{\pi}{4},
\end{align*}
σε αντίθεση με τη δοσμένη υπόθεση.

Επομένως υπάρχει $c\in(0,1)$ με $\varphi(c)>0$. Αφού
\[
\varphi(0)<0<\varphi(c),
\]
από το Θεώρημα Ενδιάμεσων Τιμών υπάρχει τουλάχιστον ένα
\[
x_0\in(0,c)\subset(0,1)
\]
τέτοιο ώστε
\[
\varphi(x_0)=0.
\]
Άρα
\[
{x_0^3f(x_0)+x_0f(x_0)=1}.
\]
\end{erwthma}
\end{thema}
