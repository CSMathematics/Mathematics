\begin{thema}{Δ}
Για κάθε $x\in\mathbb{R}$ είναι
\[
\sqrt{x^2+1}>\sqrt{x^2}=|x|\geq -x,
\]
οπότε
\[
x+\sqrt{x^2+1}>0.
\]
Επομένως η συνάρτηση $f$ ορίζεται σε ολόκληρο το $\mathbb{R}$.
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ έχουμε
\begin{align*}
f(-x)
&=\ln\left(-x+\sqrt{x^2+1}\right).
\end{align*}
Επειδή
\[
\left(\sqrt{x^2+1}-x\right)\left(\sqrt{x^2+1}+x\right)=1,
\]
προκύπτει
\[
\sqrt{x^2+1}-x=\frac{1}{x+\sqrt{x^2+1}}.
\]
Άρα
\begin{align*}
f(-x)
&=\ln\left(\frac{1}{x+\sqrt{x^2+1}}\right)\\
&=-\ln\left(x+\sqrt{x^2+1}\right)\\
&=-f(x).
\end{align*}
Επομένως η $f$ είναι \textbf{περιττή}.

Επίσης
\[
f(0)=\ln\left(0+\sqrt{0^2+1}\right)=\ln1=0,
\]
άρα η $C_f$ διέρχεται από την αρχή των αξόνων $O(0,0)$.

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\begin{align*}
f'(x)
&=\frac{\left(x+\sqrt{x^2+1}\right)'}{x+\sqrt{x^2+1}}\\
&=\frac{1+\dfrac{x}{\sqrt{x^2+1}}}{x+\sqrt{x^2+1}}\\
&=\frac{\sqrt{x^2+1}+x}
{\sqrt{x^2+1}\left(x+\sqrt{x^2+1}\right)}\\
&=\frac{1}{\sqrt{x^2+1}}.
\end{align*}
Επειδή
\[
f'(x)=\frac{1}{\sqrt{x^2+1}}>0
\]
για κάθε $x\in\mathbb{R}$, η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f''(x)
&=\left[\left(x^2+1\right)^{-\frac{1}{2}}\right]'\\
&=-\frac{1}{2}\left(x^2+1\right)^{-\frac{3}{2}}\cdot2x\\
&=-\frac{x}{(x^2+1)^{\frac{3}{2}}}.
\end{align*}
Επειδή $(x^2+1)^{\frac{3}{2}}>0$, έχουμε:
\begin{itemize}
\item $f''(x)>0$ για $x<0$,
\item $f''(0)=0$,
\item $f''(x)<0$ για $x>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f''$ και την κυρτότητα της $f$:
\begin{center}

Άρα η $f$ είναι \textbf{κυρτή} στο $(-\infty,0]$ και \textbf{κοίλη} στο $[0,+\infty)$. Επειδή η $f''$ αλλάζει πρόσημο στο $x=0$, η $C_f$ έχει μοναδικό σημείο καμπής το
\[
{O(0,0)}.
\]

\item Έχουμε
\begin{align*}
f(x^2-4)+f(x-2)<0
&\Leftrightarrow f(x^2-4)<-f(x-2)\\
&\Leftrightarrow f(x^2-4)<f(2-x),
\end{align*}
αφού η $f$ είναι περιττή.

Επειδή η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, προκύπτει
\begin{align*}
f(x^2-4)<f(2-x)
&\Leftrightarrow x^2-4<2-x\\
&\Leftrightarrow x^2+x-6<0\\
&\Leftrightarrow (x+3)(x-2)<0\\
&\Leftrightarrow -3<x<2.
\end{align*}
Άρα το σύνολο λύσεων της ανίσωσης είναι
\[
{x\in(-3,2)}.
\]

\item Από το δεύτερο ερώτημα έχουμε
\[
f'(0)=\frac{1}{\sqrt{0^2+1}}=1.
\]
Επομένως η εφαπτομένη της $C_f$ στο σημείο $O(0,0)$ έχει εξίσωση
\[
y=f(0)+f'(0)(x-0)\Leftrightarrow y=x.
\]
Από το τρίτο ερώτημα γνωρίζουμε ότι η $f$ είναι κοίλη στο $[0,+\infty)$. Άρα η γραφική της παράσταση βρίσκεται κάτω από κάθε εφαπτομένη της στο διάστημα αυτό. Ειδικότερα, για κάθε $x\geq0$ ισχύει
\[
{f(x)\leq x}.
\]
Η ισότητα ισχύει μόνο για $x=0$, αφού για $x>0$ είναι
\[
f'(x)=\frac{1}{\sqrt{x^2+1}}<1.
\]
\end{erwthma}
\end{thema}
