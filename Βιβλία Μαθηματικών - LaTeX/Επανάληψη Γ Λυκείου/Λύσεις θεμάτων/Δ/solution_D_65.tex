%=========== ΛΥΣΗ - 65ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Από το σχήμα της $C_{f'}$ διαβάζουμε το πρόσημο της $f'$:
\[
\begin{array}{lll}
f'(x)>0, & x\in(-2,0), & \text{άρα } f\nearrow[-2,0],\\
f'(x)<0, & x\in(0,2),  & \text{άρα } f\searrow[0,2],\\
f'(x)>0, & x\in(2,5),  & \text{άρα } f\nearrow[2,5],\\
f'(x)<0, & x\in(5,6),  & \text{άρα } f\searrow[5,6].
\end{array}
\]

Ο πίνακας μεταβολών της $f$ είναι:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2.2,espcl=1.5,
colorC=maincolor!50!white,colorL=maincolor!20!white,
colorV=maincolor!50!white]%
{$x$/.8,$f'(x)$/.8,$f(x)$/1.2}%
{$-2$,$0$,$2$,$5$,$6$}%
\tkzTabLine{,+,z,-,z,+,z,-,}
\tkzTabVar{-/{$1$},+/{$3$},-/{$-1$},+/{$4$},-/{$2$}}
\end{tikzpicture}
\end{center}

Άρα η μεγαλύτερη από τις τιμές
\[
f(-2)=1,\quad f(0)=3,\quad f(2)=-1,\quad f(5)=4,\quad f(6)=2
\]
είναι το $4$, ενώ η μικρότερη είναι το $-1$. Επομένως
\[
{\max_{[-2,6]} f=f(5)=4}
\]
και
\[
{\min_{[-2,6]} f=f(2)=-1}.
\]

\item Θα χρησιμοποιήσουμε τη μονοτονία που βρήκαμε στο προηγούμενο
ερώτημα.

Στο διάστημα $[0,2]$ η $f$ είναι συνεχής και γνησίως φθίνουσα, με
\[
f(0)=3>1
\qquad\text{και}\qquad
f(2)=-1<1.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\rho_1\in(0,2)
\]
τέτοιο ώστε $f(\rho_1)=1$. Επειδή η $f$ είναι γνησίως φθίνουσα στο
$[0,2]$, η ρίζα αυτή είναι μοναδική στο $(0,2)$.

Στο διάστημα $[2,5]$ η $f$ είναι συνεχής και γνησίως αύξουσα, με
\[
f(2)=-1<1
\qquad\text{και}\qquad
f(5)=4>1.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\rho_2\in(2,5)
\]
τέτοιο ώστε $f(\rho_2)=1$. Επειδή η $f$ είναι γνησίως αύξουσα στο
$[2,5]$, η ρίζα αυτή είναι μοναδική στο $(2,5)$.

Τέλος, στο διάστημα $[5,6]$ η $f$ είναι γνησίως φθίνουσα και
\[
f(6)=2>1.
\]
Άρα για κάθε $x\in[5,6]$ ισχύει
\[
f(x)\geq f(6)=2>1,
\]
οπότε δεν υπάρχει ρίζα της εξίσωσης $f(x)=1$ στο $(5,6)$.

Συνεπώς η εξίσωση $f(x)=1$ έχει στο $(0,6)$ ακριβώς δύο ρίζες:
\[
{\rho_1\in(0,2)\quad\text{και}\quad \rho_2\in(2,5)}.
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f(-2)=1=f(\rho_1).
\]
Η $f$ είναι συνεχής στο $[-2,\rho_1]$ και παραγωγίσιμη στο
$(-2,\rho_1)$. Άρα, από το θεώρημα \eng{Rolle}, υπάρχει
\[
\xi_1\in(-2,\rho_1)
\]
τέτοιο ώστε
\[
{f'(\xi_1)=0}.
\]

Επίσης,
\[
f(\rho_1)=1=f(\rho_2).
\]
Η $f$ είναι συνεχής στο $[\rho_1,\rho_2]$ και παραγωγίσιμη στο
$(\rho_1,\rho_2)$. Επομένως, ξανά από το θεώρημα \eng{Rolle}, υπάρχει
\[
\xi_2\in(\rho_1,\rho_2)
\]
τέτοιο ώστε
\[
{f'(\xi_2)=0}.
\]

Άρα η εξίσωση $f'(x)=0$ έχει τουλάχιστον δύο ρίζες. Αυτό επιβεβαιώνεται
από το δοσμένο σχήμα της $C_{f'}$, όπου φαίνεται ότι η $f'$ μηδενίζεται
στα
\[
x=0,\qquad x=2,\qquad x=5.
\]
Μάλιστα, επειδή
\[
\rho_1\in(0,2)\quad\text{και}\quad \rho_2\in(2,5),
\]
οι ρίζες που εξασφαλίζει το θεώρημα \eng{Rolle} αντιστοιχούν στις
τετμημένες
\[
{\xi_1=0\quad\text{και}\quad \xi_2=2}.
\]

\item Η $f$ είναι συνεχής στο $[2,5]$ και παραγωγίσιμη στο $(2,5)$.
Άρα εφαρμόζεται το Θεώρημα Μέσης Τιμής στο $[2,5]$. Υπάρχει λοιπόν
$\eta\in(2,5)$ τέτοιο ώστε
\[
f'(\eta)=\frac{f(5)-f(2)}{5-2}
=\frac{4-(-1)}{3}
=\frac{5}{3}.
\]
Άρα
\[
f'(\eta)=\frac53>1.
\]

Επιπλέον, από το σχήμα γνωρίζουμε ότι
\[
f'(2)=0<1.
\]
Επειδή η $f$ είναι δύο φορές παραγωγίσιμη, η $f'$ είναι συνεχής.
Θεωρούμε τη συνάρτηση
\[
h(x)=f'(x)-1.
\]
Η $h$ είναι συνεχής στο $[2,\eta]$ και
\[
h(2)=f'(2)-1=-1<0,
\]
ενώ
\[
h(\eta)=f'(\eta)-1=\frac53-1=\frac23>0.
\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει
\[
\xi\in(2,\eta)\subset(0,5)
\]
τέτοιο ώστε
\[
h(\xi)=0.
\]
Δηλαδή
\[
{f'(\xi)=1}
\]
για κάποιο $\xi\in(0,5)$.
\end{erwthma}
\end{thema}
