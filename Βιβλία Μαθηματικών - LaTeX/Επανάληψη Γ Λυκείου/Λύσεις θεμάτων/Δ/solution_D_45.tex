%=========== ΛΥΣΗ - 45ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(-1,+\infty)$ και για κάθε $x>-1$ είναι
\begin{align*}
f'(x)
&=e^x-\frac{1}{x+1}\\
&=\frac{(x+1)e^x-1}{x+1}.
\end{align*}
Θεωρούμε τη συνάρτηση
\[
h(x)=(x+1)e^x,\quad x>-1.
\]
Έχουμε
\[
h'(x)=(x+2)e^x>0,
\]
οπότε η $h$ είναι γνησίως αύξουσα στο $(-1,+\infty)$. Επειδή $x+1>0$ και $h(0)=1$, προκύπτει:
\begin{itemize}
\item $f'(x)<0$ για $-1<x<0$,
\item $f'(0)=0$,
\item $f'(x)>0$ για $x>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$-1$,$0$,$+\infty$}%
\tkzTabLine{d, -, z, +, }
\tkzTabVar{D+/{+\infty},-/{0},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-1,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
{f(0)=0}.
\]
Συνεπώς, για κάθε $x>-1$,
\[
f(x)\geq0,
\]
δηλαδή
\[
{e^x-1\geq\ln(x+1)}.
\]
Η ισότητα ισχύει μόνο για $x=0$.

\item Από το προηγούμενο ερώτημα και τα όρια
\[
\lim_{x\to-1^+}f(x)=+\infty,
\qquad
\lim_{x\to+\infty}f(x)=+\infty,
\]
προκύπτει ότι
\[
{f\big((-1,+\infty)\big)=[0,+\infty)}.
\]

Η εξίσωση
\[
2026f(x)=2025
\]
είναι ισοδύναμη με
\[
f(x)=\frac{2025}{2026}>0.
\]
Στο διάστημα $(-1,0]$ η $f$ είναι συνεχής και γνησίως φθίνουσα από το $+\infty$ στο $0$, οπότε η εξίσωση έχει ακριβώς μία ρίζα σε αυτό. Ομοίως, στο $[0,+\infty)$ η $f$ είναι συνεχής και γνησίως αύξουσα από το $0$ στο $+\infty$, οπότε έχει ακριβώς μία ακόμη ρίζα.

Συνεπώς η εξίσωση έχει
\[
{\text{ακριβώς δύο πραγματικές ρίζες}}.
\]

\item Για κάθε $x>-1$ είναι
\[
f''(x)=e^x+\frac{1}{(x+1)^2}>0.
\]
Άρα η $f$ είναι \textbf{αυστηρά κυρτή} στο $(-1,+\infty)$.

Για $x\geq0$, ο αριθμός $2x$ είναι το μέσο των $x$ και $3x$. Από την κυρτότητα της $f$ έχουμε
\[
f(2x)\leq\frac{f(x)+f(3x)}{2},
\]
δηλαδή
\[
f(x)+f(3x)\geq2f(2x).
\]
Επειδή η $f$ είναι αυστηρά κυρτή, η ισότητα ισχύει αν και μόνο αν
\[
x=3x,
\]
δηλαδή αν και μόνο αν $x=0$. Επομένως η μοναδική λύση της εξίσωσης στο $[0,+\infty)$ είναι
\[
{x=0}.
\]

\item Η $f$ είναι συνεχής και γνησίως αύξουσα στο $[0,+\infty)$, με
\[
f([0,+\infty))=[0,+\infty).
\]
Άρα αντιστρέφεται στο διάστημα αυτό.

Παρατηρούμε ότι
\[
f(1)=e-1-\ln2.
\]
Θέτουμε
\[
x=f(t).
\]
Τότε
\[
\d x=f'(t)\d t,
\]
ενώ για $x=0$ είναι $t=g(0)=0$ και για $x=f(1)$ είναι $t=1$. Επομένως
\begin{align*}
\int_0^{e-1-\ln2}g(x)\d x
&=\int_0^1t f'(t)\d t\\
&=\left[t f(t)\right]_0^1-\int_0^1f(t)\d t.
\end{align*}
Μια παράγουσα της $f$ είναι
\[
F(t)=e^t-(t+1)\ln(t+1)+1.
\]
Άρα
\begin{align*}
\int_0^1f(t)\d t
&=\left[e^t-(t+1)\ln(t+1)+1\right]_0^1\\
&=e-2\ln2-1.
\end{align*}
Συνεπώς
\begin{align*}
\int_0^{e-1-\ln2}g(x)\d x
&=f(1)-\left(e-2\ln2-1\right)\\
&=e-1-\ln2-e+2\ln2+1\\
&={\ln2}.
\end{align*}

Για το ζητούμενο όριο έχουμε μορφή $\frac{0}{0}$. Εφαρμόζοντας τρεις φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0^+}\frac{x^2-f(x)}{xf(x)}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0^+}\frac{2x-f'(x)}{f(x)+xf'(x)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0^+}\frac{2-f''(x)}{2f'(x)+xf''(x)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0^+}\frac{-f'''(x)}{3f''(x)+xf'''(x)}.
\end{align*}
Επειδή
\[
f''(0)=2
\]
και
\[
f'''(x)=e^x-\frac{2}{(x+1)^3}
\Rightarrow
f'''(0)=-1,
\]
προκύπτει
\[
{
\lim_{x\to0^+}\frac{x^2-f(x)}{xf(x)}
=\frac{1}{6}}.
\]
\end{erwthma}
\end{thema}
