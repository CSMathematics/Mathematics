\begin{thema}{Δ}
\begin{erwthma}
\item Για $x=y=0$ η δοσμένη σχέση δίνει
\[
f(0)=f^2(0),
\]
οπότε
\[
f(0)=0
\quad\text{ή}\quad
f(0)=1.
\]
Αν ήταν $f(0)=0$, τότε για κάθε $x\in\mathbb{R}$ θα είχαμε
\[
f(x)=f(x+0)=f(x)f(0)=0,
\]
οπότε $f'(0)=0$, άτοπο. Άρα
\[
{f(0)=1}.
\]

Για κάθε $x\in\mathbb{R}$ έχουμε
\[
f(x)=f\left(\frac{x}{2}+\frac{x}{2}\right)
=f^2\left(\frac{x}{2}\right)\geq0.
\]
Επίσης
\[
1=f(0)=f(x-x)=f(x)f(-x),
\]
οπότε $f(x)\neq0$. Συνεπώς
\[
{f(x)>0}
\]
για κάθε $x\in\mathbb{R}$.

Επειδή $f'(0)=1>0$, υπάρχει $\delta>0$ τέτοιο ώστε
\[
0<h<\delta\Rightarrow f(h)>f(0)=1.
\]
Έστω τώρα $x<y$ και θέτουμε $d=y-x>0$. Επιλέγουμε $n\in\mathbb{N}^*$ τέτοιο ώστε
\[
0<\frac{d}{n}<\delta.
\]
Τότε
\[
f\left(\frac{d}{n}\right)>1
\]
και, από τη συναρτησιακή σχέση,
\[
f(d)
=f\left(n\frac{d}{n}\right)
=\left[f\left(\frac{d}{n}\right)\right]^n
>1.
\]
Άρα
\[
f(y)=f(x+d)=f(x)f(d)>f(x),
\]
αφού $f(x)>0$. Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Έστω $x\in\mathbb{R}$. Από τον ορισμό της παραγώγου έχουμε
\begin{align*}
f'(x)
&=\lim_{h\to0}\frac{f(x+h)-f(x)}{h}\\
&=\lim_{h\to0}
\frac{f(x)f(h)-f(x)}{h}\\
&=f(x)\lim_{h\to0}\frac{f(h)-f(0)}{h}\\
&=f(x)f'(0).
\end{align*}
Επειδή $f'(0)=1$, προκύπτει
\[
{f'(x)=f(x)}
\]
για κάθε $x\in\mathbb{R}$.

Επομένως
\[
\left(e^{-x}f(x)\right)'
=e^{-x}\left[f'(x)-f(x)\right]
=0.
\]
Άρα
\[
e^{-x}f(x)=c.
\]
Για $x=0$ έχουμε $c=f(0)=1$, οπότε
\[
{f(x)=e^x},\quad x\in\mathbb{R}.
\]

\item Για $x\to0$ ο αριθμητής και ο παρονομαστής τείνουν στο $0$. Εφαρμόζοντας τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0}
\frac{f(3x)-f(x)}{f(x)-f(-x)}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}
\frac{3f'(3x)-f'(x)}
{f'(x)+f'(-x)}\\
&=\frac{3f'(0)-f'(0)}
{f'(0)+f'(0)}\\
&={1}.
\end{align*}

\item Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\begin{align*}
f(x^2-x)>f(x)
&\Leftrightarrow
x^2-x>x\\
&\Leftrightarrow
x^2-2x>0\\
&\Leftrightarrow
x(x-2)>0.
\end{align*}
Επομένως
\[
{x\in(-\infty,0)\cup(2,+\infty)}.
\]

Τέλος, επειδή $f(x)=e^x$, έχουμε
\begin{align*}
\int_0^1f(x)\d x
&=\int_0^1e^x\d x\\
&=\left[e^x\right]_0^1\\
&={e-1}.
\end{align*}
\end{erwthma}
\end{thema}
