%=========== ΛΥΣΗ - 68ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η ευθεία που διέρχεται από τα σημεία
\[
O(0,0)\quad\text{και}\quad A(4,3)
\]
έχει συντελεστή διεύθυνσης
\[
\lambda_{OA}=\frac{3-0}{4-0}=\frac34.
\]

Η $f$ είναι συνεχής στο $[0,4]$ και παραγωγίσιμη στο $(0,4)$. Άρα,
από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi_1\in(0,4)$ τέτοιο ώστε
\[
f'(\xi_1)=\frac{f(4)-f(0)}{4-0}
=\frac{3-0}{4}
=\frac34.
\]
Επομένως η εφαπτομένη της $C_f$ στο σημείο
\[
M(\xi_1,f(\xi_1))
\]
έχει συντελεστή διεύθυνσης $\dfrac34$, άρα είναι παράλληλη στην $OA$.

Ομοίως, η $g$ είναι συνεχής στο $[0,4]$ και παραγωγίσιμη στο $(0,4)$.
Άρα υπάρχει $\xi_2\in(0,4)$ τέτοιο ώστε
\[
g'(\xi_2)=\frac{g(4)-g(0)}{4-0}
=\frac{3-0}{4}
=\frac34.
\]
Επομένως και η εφαπτομένη της $C_g$ στο σημείο
\[
N(\xi_2,g(\xi_2))
\]
είναι παράλληλη στην ευθεία $OA$.

\item Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)-g(x),\quad x\in[0,4].
\]
Η $h$ είναι συνεχής στο $[0,4]$ και παραγωγίσιμη στο $(0,4)$, ως
διαφορά συνεχών και παραγωγίσιμων συναρτήσεων. Επιπλέον,
\[
h(0)=f(0)-g(0)=0
\]
και
\[
h(4)=f(4)-g(4)=0.
\]
Άρα, από το θεώρημα \eng{Rolle}, υπάρχει $\xi\in(0,4)$ τέτοιο ώστε
\[
h'(\xi)=0.
\]
Δηλαδή
\[
f'(\xi)-g'(\xi)=0
\]
και επομένως
\[
{f'(\xi)=g'(\xi)}.
\]
Άρα οι εφαπτόμενες των $C_f$ και $C_g$ στα σημεία με την ίδια τετμημένη
$\xi$ έχουν ίσους συντελεστές διεύθυνσης, οπότε είναι παράλληλες.

\item Για κάθε $x\in(0,4)$ δίνεται ότι
\[
f(x)>g(x).
\]
Άρα
\[
f(x)-g(x)>0,\quad x\in(0,4).
\]
Επειδή η $f-g$ είναι συνεχής και θετική στο εσωτερικό του διαστήματος,
έχουμε
\[
\int_0^4 \left[f(x)-g(x)\right]\d x>0.
\]
Επομένως
\[
\int_0^4 f(x)\d x-\int_0^4 g(x)\d x>0,
\]
δηλαδή
\[
{\int_0^4 f(x)\d x>\int_0^4 g(x)\d x}.
\]
Γεωμετρικά, η διαφορά των δύο ολοκληρωμάτων είναι το εμβαδόν του
κίτρινου χωρίου ανάμεσα στις δύο γραφικές παραστάσεις.

\item Επειδή οι $f$ και $g$ είναι γνησίως αύξουσες στο $[0,4]$, είναι
$1-1$, άρα αντιστρέφονται. Επιπλέον, επειδή είναι συνεχείς και
\[
f(0)=g(0)=0,\qquad f(4)=g(4)=3,
\]
έχουμε
\[
f([0,4])=g([0,4])=[0,3].
\]
Άρα
\[
D_{f^{-1}}=D_{g^{-1}}=[0,3].
\]

Θα δείξουμε ότι
\[
f^{-1}(y)<g^{-1}(y),\quad y\in(0,3).
\]
Πράγματι, έστω $y\in(0,3)$ και
\[
a=f^{-1}(y).
\]
Τότε
\[
a\in(0,4)
\]
και
\[
f(a)=y.
\]
Από την υπόθεση $f(x)>g(x)$ για κάθε $x\in(0,4)$, παίρνουμε
\[
f(a)>g(a).
\]
Άρα
\[
y>g(a).
\]
Επειδή η $g$ είναι γνησίως αύξουσα, από την ανισότητα $g(a)<y$
προκύπτει
\[
a<g^{-1}(y).
\]
Δηλαδή
\[
f^{-1}(y)<g^{-1}(y),\quad y\in(0,3).
\]

Επομένως
\[
\int_0^3 f^{-1}(x)\d x<\int_0^3 g^{-1}(x)\d x.
\]
Άρα
\[
{\int_0^3 f^{-1}(x)\d x<\int_0^3 g^{-1}(x)\d x}.
\]
\end{erwthma}
\end{thema}
