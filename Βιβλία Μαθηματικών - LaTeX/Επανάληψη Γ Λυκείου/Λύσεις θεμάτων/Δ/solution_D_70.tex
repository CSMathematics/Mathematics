\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι συνεχής στο $[0,4]$ και παραγωγίσιμη στο $(0,4)$.
Άρα, από το θεώρημα μέσης τιμής, υπάρχει
\[
\xi_1\in(0,4)
\]
τέτοιο ώστε
\[
f'(\xi_1)=\frac{f(4)-f(0)}{4-0}
=\frac{3-0}{4}
=\frac34.
\]

Επίσης, η $g$ είναι συνεχής στο $[0,3]$ και παραγωγίσιμη στο $(0,3)$.
Άρα, πάλι από το θεώρημα μέσης τιμής, υπάρχει
\[
\xi_2\in(0,3)
\]
τέτοιο ώστε
\[
g'(\xi_2)=\frac{g(3)-g(0)}{3-0}
=\frac{4-0}{3}
=\frac43.
\]

Επομένως
\[
f'(\xi_1)\cdot g'(\xi_2)
=\frac34\cdot\frac43
=1.
\]
Άρα υπάρχουν $\xi_1 \in (0,4)$ και $\xi_2\in(0,3)$ τέτοια, ώστε
\[
{f'(\xi_1)\cdot g'(\xi_2)=1}.
\]

\item Επειδή $g=f^{-1}$, οι καμπύλες $C_f$ και $C_g$ είναι συμμετρικές
ως προς την ευθεία $y=x$.

Θεωρούμε το ορθογώνιο
\[
R=[0,4]\times[0,3].
\]
Το εμβαδόν του είναι
\[
E_R=4\cdot 3=12.
\]

Η καμπύλη $C_f$ χωρίζει το ορθογώνιο αυτό σε δύο χωρία:
\begin{itemize}
\item το χωρίο κάτω από τη $C_f$, με εμβαδόν
\[
E_1=\int_0^4 f(x)\d x,
\]
\item και το χωρίο αριστερά από τη $C_f$, το οποίο περιγράφεται με την
αντίστροφη $g=f^{-1}$ και έχει εμβαδόν
\[
E_2=\int_0^3 g(y)\d y.
\]
\end{itemize}
Τα δύο χωρία καλύπτουν όλο το ορθογώνιο $R$ και δεν έχουν κοινό
εσωτερικό. Άρα
\[
E_1+E_2=E_R.
\]
Δηλαδή
\[
\int_0^4 f(x)\d x+\int_0^3 g(y)\d y=12.
\]
Αν αλλάξουμε απλώς το γράμμα της μεταβλητής στο δεύτερο ολοκλήρωμα,
παίρνουμε
\[
{\int_0^4 f(x)\d x+\int_0^3 g(x)\d x=12}.
\]

\item Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)+g(x)-2x,\quad x\in[0,2].
\]
Η $h$ είναι συνεχής στο $[0,2]$ και δύο φορές παραγωγίσιμη στο $(0,2)$.

Από τα δεδομένα έχουμε
\[
h(0)=f(0)+g(0)-0=0,
\]
\[
h(1)=f(1)+g(1)-2=1+1-2=0
\]
και
\[
h(2)=f(2)+g(2)-4=2+2-4=0.
\]

Επειδή
\[
h(0)=h(1),
\]
από το θεώρημα \eng{Rolle} στο $[0,1]$ υπάρχει
\[
\alpha\in(0,1)
\]
τέτοιο ώστε
\[
h'(\alpha)=0.
\]

Επίσης, επειδή
\[
h(1)=h(2),
\]
από το θεώρημα \eng{Rolle} στο $[1,2]$ υπάρχει
\[
\beta\in(1,2)
\]
τέτοιο ώστε
\[
h'(\beta)=0.
\]

Τώρα εφαρμόζουμε το θεώρημα \eng{Rolle} στη συνάρτηση $h'$ στο
$[\alpha,\beta]$. Επειδή
\[
h'(\alpha)=h'(\beta)=0,
\]
υπάρχει
\[
\xi\in(\alpha,\beta)\subset(0,2)
\]
τέτοιο ώστε
\[
h''(\xi)=0.
\]

Όμως
\[
h''(x)=f''(x)+g''(x),
\]
αφού $(2x)''=0$. Άρα
\[
h''(\xi)=0
\Leftrightarrow
f''(\xi)+g''(\xi)=0.
\]
Επομένως υπάρχει $\xi\in(0,2)$ τέτοιο, ώστε
\[
{f''(\xi)+g''(\xi)=0}.
\]

\item Αν η $f$ είναι κοίλη στο $[0,4]$, τότε η γραφική της παράσταση
βρίσκεται πάνω από τη χορδή που ενώνει τα άκρα
\[
A(0,f(0))=(0,0)
\quad\text{και}\quad
B(4,f(4))=(4,3).
\]

Η εξίσωση της χορδής $AB$ είναι
\[
y=\frac34x.
\]
Άρα, για κάθε $x\in[0,4]$,
\[
f(x)\geq \frac34x.
\]

Η ανισότητα είναι γνήσια σε κάποιο υποδιάστημα. Πράγματι,
\[
f(1)=1>\frac34.
\]
Επομένως, λόγω συνέχειας της $f$, υπάρχει διάστημα γύρω από το $1$ στο
οποίο
\[
f(x)>\frac34x.
\]
Άρα
\[
\int_0^4 f(x)\d x
>
\int_0^4 \frac34x\d x
=\frac34\cdot\frac{4^2}{2}
=6.
\]
Δηλαδή
\[
{\int_0^4 f(x)\d x>6}.
\]

Από το β΄ ερώτημα γνωρίζουμε ότι
\[
\int_0^4 f(x)\d x+\int_0^3 g(x)\d x=12.
\]
Άρα
\[
\int_0^3 g(x)\d x
=12-\int_0^4 f(x)\d x.
\]
Επειδή
\[
\int_0^4 f(x)\d x>6,
\]
προκύπτει
\[
\int_0^3 g(x)\d x<12-6=6.
\]
Επομένως
\[
{\int_0^3 g(x)\d x<6}.
\]
\end{erwthma}
\end{thema}
