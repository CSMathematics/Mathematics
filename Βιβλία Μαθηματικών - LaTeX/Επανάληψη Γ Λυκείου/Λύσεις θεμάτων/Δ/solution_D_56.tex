\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και για κάθε $x>0$ είναι
\[
f'(x)=\ln x.
\]
Επομένως:
\begin{itemize}
\item $f'(x)<0$ για $0<x<1$,
\item $f'(1)=0$,
\item $f'(x)>0$ για $x>1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{d, -, z, +, }
\tkzTabVar{D+/{1},-/{0},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=1$, με τιμή
\[
{f(1)=0}.
\]
Συνεπώς
\[
f(x)\geq0
\]
για κάθε $x>0$, δηλαδή
\[
x\ln x-x+1\geq0.
\]
Άρα
\[
{x\ln x\geq x-1}
\]
για κάθε $x>0$. Η ισότητα ισχύει μόνο για $x=1$.

\item Για κάθε $x>0$ είναι
\[
f''(x)=\frac{1}{x}>0.
\]
Άρα η $f$ είναι \textbf{αυστηρά κυρτή} στο $(0,+\infty)$.

Στο σημείο του ελαχίστου έχουμε
\[
f(1)=0
\quad\text{και}\quad
f'(1)=0.
\]
Επομένως η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $A(1,0)$ είναι
\[
{y=0}.
\]

Από το προηγούμενο ερώτημα γνωρίζουμε ότι
\[
f(x)\geq0
\]
για κάθε $x>0$, με ισότητα μόνο για $x=1$. Άρα η ευθεία $y=0$ τέμνει την $C_f$ μόνο στο σημείο
\[
{A(1,0)}.
\]

\item Από την ανισότητα του πρώτου ερωτήματος έχουμε
\[
x\ln x\geq x-1.
\]
Επειδή $x>0$, διαιρώντας με $x$ παίρνουμε
\[
{\ln x\geq1-\frac{1}{x}}
\]
για κάθε $x>0$.

Η ισότητα ισχύει αν και μόνο αν
\[
x\ln x=x-1,
\]
δηλαδή αν και μόνο αν $f(x)=0$. Από το πρώτο ερώτημα, η $f$ μηδενίζεται μόνο στο $x=1$. Επομένως η εξίσωση
\[
\ln x=1-\frac{1}{x}
\]
έχει
\[
{\text{ακριβώς μία λύση, την }x=1}.
\]

\item Μια παράγουσα της $f$ είναι
\[
F(x)=\frac{x^2}{2}\ln x-\frac{3x^2}{4}+x.
\]
Πράγματι,
\[
F'(x)=x\ln x-x+1=f(x).
\]
Επομένως
\begin{align*}
\int_1^2f(x)\d x
&=\left[
\frac{x^2}{2}\ln x-\frac{3x^2}{4}+x
\right]_1^2\\
&=\left(2\ln2-1\right)-\frac14\\
&={2\ln2-\frac54}.
\end{align*}
\end{erwthma}
\end{thema}
