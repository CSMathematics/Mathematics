%=========== ΛΥΣΗ - 36ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και για κάθε $x\in\mathbb{R}$ είναι
\[
f'(x)=e^{x-1}>0.
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$ και επομένως $1-1$.

Επιπλέον,
\[
\lim_{x\to-\infty}f(x)=0
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty,
\]
οπότε
\[
f(\mathbb{R})=(0,+\infty).
\]
Συνεπώς η $f$ αντιστρέφεται και η αντίστροφή της έχει πεδίο ορισμού
\[
D_g=(0,+\infty).
\]

Για να βρούμε τον τύπο της, θέτουμε
\begin{align*}
y=e^{x-1}
&\Leftrightarrow \ln y=x-1\\
&\Leftrightarrow x=\ln y+1.
\end{align*}
Άρα
\[
{g(x)=f^{-1}(x)=\ln x+1},\quad x>0.
\]

\item Θεωρούμε τη συνάρτηση
\[
h(x)=x-1-\ln x,\quad x>0.
\]
Για κάθε $x>0$ είναι
\[
h'(x)=1-\frac{1}{x}=\frac{x-1}{x}.
\]
Άρα η $h$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=1$, με τιμή
\[
h(1)=0.
\]
Συνεπώς
\[
h(x)\geq0
\Leftrightarrow
\ln x+1\leq x
\Leftrightarrow
{g(x)\leq x}
\]
για κάθε $x>0$.

Επίσης, από τη γνωστή ανισότητα
\[
e^u\geq u+1,
\]
θέτοντας $u=x-1$, παίρνουμε
\[
e^{x-1}\geq x,
\]
δηλαδή
\[
{x\leq f(x)}
\]
για κάθε $x>0$.

Άρα
\[
{g(x)\leq x\leq f(x)}
\]
για κάθε $x>0$. Και στις δύο ανισότητες η ισότητα ισχύει μόνο για $x=1$.

Αν οι $C_f$ και $C_g$ έχουν κοινό σημείο με τετμημένη $x>0$, τότε
\[
f(x)=g(x).
\]
Όμως
\[
g(x)\leq x\leq f(x),
\]
οπότε αναγκαστικά
\[
f(x)=g(x)=x.
\]
Από την προηγούμενη μελέτη, η ισότητα ισχύει μόνο για $x=1$. Άρα το μοναδικό κοινό σημείο είναι
\[
{A(1,1)}.
\]

Έχουμε
\[
f'(1)=1
\]
και
\[
g'(x)=\frac{1}{x}
\Rightarrow
g'(1)=1.
\]
Επομένως οι $C_f$ και $C_g$ δέχονται στο $A(1,1)$ κοινή εφαπτομένη, με εξίσωση
\[
y-1=x-1,
\]
δηλαδή
\[
{y=x}.
\]

\item Από το προηγούμενο ερώτημα, για κάθε $x\in[1,e]$ ισχύει
\[
g(x)\leq f(x).
\]
Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_1^e\left[f(x)-g(x)\right]\d x\\
&=\int_1^e\left[e^{x-1}-\ln x-1\right]\d x\\
&=\left[e^{x-1}-x\ln x\right]_1^e\\
&=-\left(1-0\right)+e^{e-1}-e\\
&={e^{e-1}-e-1}.
\end{align*}

\item Για το ζητούμενο όριο έχουμε
\[
f(1)-g(1)=0.
\]
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to1}\frac{f(x)-g(x)}{(x-1)^2}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{e^{x-1}-\frac{1}{x}}{2(x-1)}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to1}\frac{e^{x-1}+\frac{1}{x^2}}{2}\\
&={1}.
\end{align*}

Από το δεύτερο ερώτημα γνωρίζουμε ότι
\[
t\leq e^{t-1}
\]
για κάθε $t>0$, με ισότητα μόνο για $t=1$. Θέτοντας $t=x^2$, έχουμε
\[
x^2\leq e^{x^2-1}.
\]
Για κάθε $x\in(1,2]$ είναι $x^2>1$, οπότε η ανισότητα είναι αυστηρή:
\[
e^{x^2-1}>x^2.
\]
Ολοκληρώνοντας στο $[1,2]$, προκύπτει
\begin{align*}
\int_1^2e^{x^2-1}\d x
&>\int_1^2x^2\d x\\
&=\left[\frac{x^3}{3}\right]_1^2\\
&={\frac{7}{3}}.
\end{align*}
\end{erwthma}
\end{thema}
