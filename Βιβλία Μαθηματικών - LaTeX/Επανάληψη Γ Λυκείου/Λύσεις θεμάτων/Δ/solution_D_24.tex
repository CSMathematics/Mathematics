\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=e^x+(x-1)e^x=xe^x.
\]
Επειδή $e^x>0$, το πρόσημο της $f'$ είναι το πρόσημο του $x$. Επομένως η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$. Παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
f(0)=0.
\]
Άρα
\[
f(x)\geq0
\]
για κάθε $x\in\mathbb{R}$.

Για $x<1$ έχουμε
\begin{align*}
(x-1)e^x+1\geq0
&\Leftrightarrow (1-x)e^x\leq1\\
&\Leftrightarrow {e^x\leq\frac{1}{1-x}},
\end{align*}
αφού $1-x>0$.

\item Έχουμε
\[
\lim_{x\to-\infty}f(x)=1,
\quad
f(0)=0
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty.
\]
Επιπλέον, για $x<0$ είναι
\[
f(x)=(x-1)e^x+1<1.
\]
Άρα το σύνολο τιμών της $f$ στο $(-\infty,0]$ είναι $[0,1)$, ενώ στο $[0,+\infty)$ είναι $[0,+\infty)$.

Επομένως η εξίσωση $f(x)=\lambda$:
\begin{itemize}
\item δεν έχει πραγματικές ρίζες, αν $\lambda<0$,
\item έχει μία ρίζα, την $x=0$, αν $\lambda=0$,
\item έχει δύο ρίζες, μία αρνητική και μία θετική, αν $0<\lambda<1$,
\item έχει μία ρίζα, την $x=1$, αν $\lambda=1$,
\item έχει μία θετική ρίζα, αν $\lambda>1$.
\end{itemize}

\item Έστω $M(t,f(t))$ το σημείο επαφής. Η εφαπτομένη της $C_f$ στο $M$ έχει εξίσωση
\[
y-f(t)=f'(t)(x-t).
\]
Για να διέρχεται από το $O(0,0)$, πρέπει
\begin{align*}
f(t)-tf'(t)&=0\\
\Leftrightarrow (t-1)e^t+1-t^2e^t&=0\\
\Leftrightarrow (t^2-t+1)e^t&=1.
\end{align*}
Θεωρούμε τη συνάρτηση
\[
h(t)=(t^2-t+1)e^t-1.
\]
Έχουμε
\[
h'(t)=t(t+1)e^t.
\]
Άρα η $h$ είναι γνησίως αύξουσα στο $(-\infty,-1]$, γνησίως φθίνουσα στο $[-1,0]$ και γνησίως αύξουσα στο $[0,+\infty)$.

Επιπλέον,
\[
\lim_{t\to-\infty}h(t)=-1,
\quad
h(-1)=\frac{3}{e}-1>0
\]
και
\[
h(0)=0.
\]
Επομένως υπάρχει ακριβώς μία ρίζα $a\in(-\infty,-1)$ και μία ακόμη ρίζα η $t=0$. Μάλιστα,
\[
h(-2)=\frac{7}{e^2}-1<0<h(-1),
\]
οπότε
\[
a\in(-2,-1).
\]

Συνεπώς υπάρχουν ακριβώς δύο ζητούμενες εφαπτόμενες. Για $t=0$ είναι
\[
{y=0},
\]
ενώ για $t=a$ είναι
\[
{y=ae^a x},
\]
όπου $a\in(-2,-1)$ είναι η μοναδική αρνητική ρίζα της εξίσωσης
\[
(a^2-a+1)e^a=1.
\]

\item Θεωρούμε τη συνάρτηση
\[
g(x)=x^2-f(x),\quad x\in[0,1].
\]
Έχουμε
\[
g(0)=g(1)=0
\]
και
\[
g'(x)=2x-xe^x=x(2-e^x).
\]
Άρα η $g$ είναι γνησίως αύξουσα στο $[0,\ln2]$ και γνησίως φθίνουσα στο $[\ln2,1]$. Εφόσον μηδενίζεται στα άκρα, προκύπτει
\[
g(x)\geq0
\]
για κάθε $x\in[0,1]$, με αυστηρή ανισότητα για $x\in(0,1)$. Επομένως
\[
{f(x)\leq x^2}
\]
για κάθε $x\in[0,1]$.

Για $x\in(0,1)$ έχουμε $x^2\in(0,1)$, οπότε
\[
f(x^2)<x^4.
\]
Ολοκληρώνοντας κατά μέλη στο $[0,1]$, παίρνουμε
\[
\int_0^1f(x^2)\d x
<\int_0^1x^4\d x
={\frac{1}{5}}.
\]
\end{erwthma}
\end{thema}
