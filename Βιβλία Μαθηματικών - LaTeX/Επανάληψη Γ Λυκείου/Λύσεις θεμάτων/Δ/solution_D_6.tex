%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x>0$ είναι
\[
xf'(x)+f(x)=(xf(x))'.
\]
Επομένως η δοσμένη σχέση γράφεται
\[
(xf(x))'=xe^x.
\]
Επίσης
\[
xe^x=(xe^x-e^x)'.
\]
Άρα, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
xf(x)=xe^x-e^x+c.
\]
Για $x=1$ και $f(1)=e$ έχουμε
\[
e=e-e+c\Leftrightarrow c=e.
\]
Συνεπώς
\[
{f(x)=\frac{xe^x-e^x+e}{x}},\quad x>0.
\]

\item Για κάθε $x>0$ είναι
\begin{align*}
f'(x)
&=\frac{xe^x\cdot x-\left(xe^x-e^x+e\right)}{x^2}\\
&=\frac{e^x(x^2-x+1)-e}{x^2}.
\end{align*}
Θεωρούμε τη βοηθητική συνάρτηση
\[
h(x)=e^x(x^2-x+1)-e,\quad x>0.
\]
Η $h$ είναι παραγωγίσιμη και
\begin{align*}
h'(x)
&=e^x(x^2-x+1)+e^x(2x-1)\\
&=e^x(x^2+x)\\
&=x(x+1)e^x>0
\end{align*}
για κάθε $x>0$. Επομένως η $h$ είναι γνησίως αύξουσα στο $(0,+\infty)$. Επειδή
\[
h(1)=e-e=0,
\]
έχουμε:
\begin{itemize}
\item $h(x)<0$ για $0<x<1$,
\item $h(1)=0$,
\item $h(x)>0$ για $x>1$.
\end{itemize}
Επειδή $x^2>0$, η $f'$ έχει το ίδιο πρόσημο με την $h$. Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}

{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%

\tkzTabLine{d, -, z, +, }
\tkzTabVar{D+/{+\infty},-/{e},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=1$, με τιμή
\[
{f(1)=e}.
\]

\item Για το όριο στο $0^+$ έχουμε
\begin{align*}
\lim_{x\to0^+}f(x)
&=\lim_{x\to0^+}\frac{xe^x-e^x+e}{x}\\
&=\frac{e-1}{0^+}\\
&=+\infty.
\end{align*}

Για το όριο στο $+\infty$ γράφουμε
\[
f(x)=e^x-\frac{e^x}{x}+\frac{e}{x}
=e^x\left(1-\frac{1}{x}\right)+\frac{e}{x}.
\]
Επομένως
\[
{\lim_{x\to+\infty}f(x)=+\infty}.
\]
Συνολικά,
\[
{\lim_{x\to0^+}f(x)=+\infty
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty}.
\]

\item Από τη δοσμένη σχέση, για $x=1$, έχουμε
\[
f'(1)+f(1)=e.
\]
Επειδή $f(1)=e$, προκύπτει
\[
f'(1)=0.
\]
Η εφαπτομένη της $C_f$ στο σημείο $A(1,f(1))\equiv A(1,e)$ έχει εξίσωση
\begin{align*}
y-f(1)&=f'(1)(x-1)\\
\Leftrightarrow y-e&=0\\
\Leftrightarrow {y=e}.
\end{align*}
Η εφαπτομένη είναι παράλληλη στον άξονα $x'x$ και τέμνει τον άξονα $y'y$ στο σημείο $(0,e)$. Από το δεύτερο ερώτημα, το $e$ είναι η ελάχιστη τιμή της $f$, όπως ζητήθηκε.
\end{erwthma}
\end{thema}
