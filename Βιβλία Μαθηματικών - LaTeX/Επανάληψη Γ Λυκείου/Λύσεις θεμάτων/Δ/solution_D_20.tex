\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(-1,+\infty)$ και για κάθε $x>-1$ είναι
\begin{align*}
f'(x)
&=\left(\frac{e^x}{x+1}\right)'\\
&=\frac{e^x(x+1)-e^x}{(x+1)^2}\\
&=\frac{xe^x}{(x+1)^2}.
\end{align*}
Επειδή $e^x>0$ και $(x+1)^2>0$, το πρόσημο της $f'$ είναι το πρόσημο του $x$. Επομένως:
\begin{itemize}
\item $f'(x)<0$ για $-1<x<0$,
\item $f'(0)=0$,
\item $f'(x)>0$ για $x>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-1$,$0$,$+\infty$}%
\tkzTabLine{d, -, z, +, }
\tkzTabVar{D+/{+\infty},-/{1},+/{+\infty}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-1,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
{f(0)=1}.
\]

\item Από το προηγούμενο ερώτημα γνωρίζουμε ότι
\[
f(x)\geq1
\]
για κάθε $x>-1$. Επίσης
\[
\lim_{x\to-1^+}\frac{e^x}{x+1}=+\infty
\quad\text{και}\quad
\lim_{x\to+\infty}\frac{e^x}{x+1}=+\infty.
\]
Με βάση τη συνέχεια και τη μονοτονία της $f$, το σύνολο τιμών της είναι
\[
{f((-1,+\infty))=[1,+\infty)}.
\]

\item Για κάθε $x\in\mathbb{R}$ είναι $x^2,x^4\geq0$, άρα
\[
f(x^2)\geq1
\quad\text{και}\quad
f(x^4)\geq1.
\]
Επομένως
\[
f(x^2)+f(x^4)\geq2.
\]
Η ισότητα ισχύει αν και μόνο αν
\[
f(x^2)=f(x^4)=1.
\]
Η $f$ λαμβάνει την ελάχιστη τιμή $1$ μόνο στο $0$. Άρα
\[
x^2=0
\quad\text{και}\quad
x^4=0,
\]
οπότε
\[
{x=0}.
\]

\item Για κάθε $x>-1$ έχουμε
\[
g(x)=\frac{1}{f(x)}=(x+1)e^{-x}.
\]
Στο διάστημα $[0,1]$ είναι
\[
g(x)>0.
\]
Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\int_0^1g(x)\d x\\
&=\int_0^1(x+1)e^{-x}\d x\\
&=\left[-(x+2)e^{-x}\right]_0^1\\
&=2-\frac{3}{e}.
\end{align*}
Άρα
\[
{E=2-\frac{3}{e}}.
\]
\end{erwthma}
\end{thema}
