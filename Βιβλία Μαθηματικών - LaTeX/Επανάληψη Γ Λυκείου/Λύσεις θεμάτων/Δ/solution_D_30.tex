\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(-1,+\infty)$ και για κάθε $x>-1$ είναι
\begin{align*}
f'(x)
&=\left[e^x-\ln(x+1)-1\right]'\\
&=e^x-\frac{1}{x+1}\\
&=\frac{(x+1)e^x-1}{x+1}.
\end{align*}
Θεωρούμε τη συνάρτηση
\[
h(x)=(x+1)e^x,\quad x>-1.
\]
Έχουμε
\[
h'(x)=(x+2)e^x>0
\]
για κάθε $x>-1$, οπότε η $h$ είναι γνησίως αύξουσα στο $(-1,+\infty)$. Επίσης, $x+1>0$ στο πεδίο ορισμού της $f$, οπότε το πρόσημο της $f'$ είναι το πρόσημο της παράστασης $h(x)-1$. Επειδή $h(0)=1$, προκύπτει:
\begin{itemize}
\item για $-1<x<0$ είναι $h(x)<h(0)=1$, άρα $f'(x)<0$,
\item $f'(0)=0$,
\item για $x>0$ είναι $h(x)>h(0)=1$, άρα $f'(x)>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-1,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
{f(0)=0}.
\]
Συνεπώς, για κάθε $x>-1$ ισχύει
\[
f(x)\geq f(0)=0,
\]
δηλαδή
\[
e^x-\ln(x+1)-1\geq0
\Leftrightarrow
{e^x\geq\ln(x+1)+1}.
\]
Η ισότητα ισχύει μόνο για $x=0$.

\item Από το προηγούμενο ερώτημα είναι
\[
g(x)\geq f(x)\geq0
\]
για κάθε $x>-1$, ενώ $g(0)=0$. Επομένως η $g$ παρουσιάζει ολικό ελάχιστο στο εσωτερικό σημείο $x_0=0$. Επειδή η $g$ είναι παραγωγίσιμη στο $0$, από το θεώρημα \eng{Fermat} έχουμε
\[
{g'(0)=0}.
\]
Η εξίσωση της εφαπτομένης της $C_g$ στο σημείο $A(0,g(0))\equiv A(0,0)$ είναι
\[
y-g(0)=g'(0)(x-0),
\]
άρα
\[
{y=0}.
\]

\item Έχουμε $f(0)=0$ και $f'(0)=0$. Επομένως, εφαρμόζοντας δύο φορές τον κανόνα \eng{De l'Hospital}, παίρνουμε
\begin{align*}
\lim_{x\to0}\frac{f(x)}{x^2}
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{f'(x)}{2x}\\
&\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to0}\frac{f''(x)}{2}.
\end{align*}
Για κάθε $x>-1$ είναι
\[
f''(x)=e^x+\frac{1}{(x+1)^2},
\]
οπότε
\[
{\lim_{x\to0}\frac{f(x)}{x^2}
=\frac{f''(0)}{2}
=\frac{1+1}{2}
=1}.
\]

Για κάθε $x\neq0$ είναι $x^2>0$ και, αφού $g(x)\geq f(x)$, έχουμε
\[
\frac{g(x)}{x^2}\geq\frac{f(x)}{x^2}.
\]
Περνώντας στο όριο καθώς $x\to0$, προκύπτει
\[
L\geq1,
\]
δηλαδή
\[
{L\geq1}.
\]

\item Από το πρώτο ερώτημα γνωρίζουμε ότι $f(x)\geq0$ για κάθε $x>-1$. Άρα το ζητούμενο εμβαδόν είναι
\[
E=\int_0^{e-1}f(x)\d x.
\]
Μια παράγουσα της $f$ είναι
\[
F(x)=e^x-(x+1)\ln(x+1)+1,
\]
διότι
\[
F'(x)=e^x-\ln(x+1)-1=f(x).
\]
Επομένως
\begin{align*}
E
&=\left[e^x-(x+1)\ln(x+1)+1\right]_0^{e-1}\\
&=e^{e-1}-e\ln e+1-\left(1-\ln1+1\right)\\
&=e^{e-1}-e+1-2\\
&={e^{e-1}-e-1}.
\end{align*}
\end{erwthma}
\end{thema}
