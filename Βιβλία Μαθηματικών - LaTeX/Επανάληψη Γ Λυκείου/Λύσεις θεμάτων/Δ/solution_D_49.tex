\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ η δοσμένη σχέση γράφεται
\[
(x^2+1)f'(x)+2xf(x)=2x.
\]
Όμως
\[
\left[(x^2+1)f(x)\right]'
=(x^2+1)f'(x)+2xf(x),
\]
οπότε
\[
\left[(x^2+1)f(x)\right]'=2x.
\]
Επομένως
\[
(x^2+1)f(x)=x^2+c,
\]
όπου $c\in\mathbb{R}$. Επειδή $f(0)=0$, έχουμε
\[
c=0.
\]
Άρα
\[
{f(x)=\frac{x^2}{x^2+1}},\quad x\in\mathbb{R}.
\]

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=\frac{2x}{(x^2+1)^2}.
\]
Επειδή $(x^2+1)^2>0$, το πρόσημο της $f'$ είναι το πρόσημο του $x$. Έχουμε:
\begin{itemize}
\item $f'(x)<0$ για $x<0$,
\item $f'(0)=0$,
\item $f'(x)>0$ για $x>0$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{, -, z, +, }
\tkzTabVar{+/{1},-/{0},+/{1}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x_0=0$, με τιμή
\[
{f(0)=0}.
\]
Δεν παρουσιάζει τοπικό μέγιστο.

Για κάθε $x\in\mathbb{R}$ είναι
\begin{align*}
f''(x)
&=\left[\frac{2x}{(x^2+1)^2}\right]'\\
&=\frac{2(1-3x^2)}{(x^2+1)^3}.
\end{align*}
Επειδή $(x^2+1)^3>0$, έχουμε:
\begin{itemize}
\item $f''(x)>0$ για $-\dfrac{1}{\sqrt3}<x<\dfrac{1}{\sqrt3}$,
\item $f''(x)=0$ για $x=\pm\dfrac{1}{\sqrt3}$,
\item $f''(x)<0$ για $x<-\dfrac{1}{\sqrt3}$ ή $x>\dfrac{1}{\sqrt3}$.
\end{itemize}
Άρα η $f$ είναι \textbf{κυρτή} στο
\[
\left[-\frac{1}{\sqrt3},\frac{1}{\sqrt3}\right]
\]
και \textbf{κοίλη} στα διαστήματα
\[
\left(-\infty,-\frac{1}{\sqrt3}\right]
\quad\text{και}\quad
\left[\frac{1}{\sqrt3},+\infty\right).
\]
Η $C_f$ έχει σημεία καμπής
\[
{
K_1\left(-\frac{1}{\sqrt3},\frac14\right)}
\quad\text{και}\quad
{
K_2\left(\frac{1}{\sqrt3},\frac14\right)}.
\]

\item Έστω ότι η εφαπτομένη της $C_f$ στο σημείο
\[
A\left(x_0,f(x_0)\right)
\]
διέρχεται από το σημείο
\[
M\left(0,\frac12\right).
\]
Η εξίσωσή της είναι
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]
Θέτοντας $x=0$ και $y=\frac12$, παίρνουμε
\[
\frac12-f(x_0)=-x_0f'(x_0),
\]
δηλαδή
\[
f(x_0)-x_0f'(x_0)=\frac12.
\]
Επομένως
\begin{align*}
\frac{x_0^2}{x_0^2+1}
-\frac{2x_0^2}{(x_0^2+1)^2}
&=\frac12\\
\Leftrightarrow
\frac{x_0^2(x_0^2-1)}
{(x_0^2+1)^2}
&=\frac12\\
\Leftrightarrow
2x_0^2(x_0^2-1)
&=(x_0^2+1)^2\\
\Leftrightarrow
x_0^4-4x_0^2-1&=0.
\end{align*}
Θέτουμε $u=x_0^2\geq0$. Τότε
\[
u^2-4u-1=0,
\]
οπότε
\[
u=2\pm\sqrt5.
\]
Επειδή $u\geq0$, δεκτή είναι μόνο η τιμή
\[
u=2+\sqrt5.
\]
Άρα
\[
{x_0=\pm\sqrt{2+\sqrt5}}.
\]
Επομένως υπάρχουν ακριβώς δύο εφαπτόμενες της $C_f$ που διέρχονται από το $M$.

Αν θέσουμε
\[
a=\sqrt{2+\sqrt5},
\]
οι εξισώσεις τους είναι
\[
{
y-\frac12
=\frac{2a}{(a^2+1)^2}x}
\]
και
\[
{
y-\frac12
=-\frac{2a}{(a^2+1)^2}x}.
\]

\item Για κάθε $x\in(0,1)$ έχουμε
\[
1<1+x^2<2.
\]
Επομένως
\[
\frac{x^2}{2}
<\frac{x^2}{x^2+1}
<x^2,
\]
δηλαδή
\[
\frac{x^2}{2}<f(x)<x^2.
\]
Ολοκληρώνοντας στο $[0,1]$, παίρνουμε
\begin{align*}
\int_0^1\frac{x^2}{2}\d x
&<\int_0^1f(x)\d x
<\int_0^1x^2\d x,
\end{align*}
άρα
\[
{
\frac16<\int_0^1f(x)\d x<\frac13}.
\]

Θέτουμε
\[
I=\int_0^1f(x)\d x.
\]
Από την προηγούμενη ανισότητα έχουμε
\[
0<I<\frac13<\frac12=f(1),
\]
ενώ $f(0)=0<I$. Η $f$ είναι συνεχής στο $[0,1]$, οπότε από το Θεώρημα Ενδιάμεσων Τιμών υπάρχει $\xi\in(0,1)$ τέτοιο ώστε
\[
f(\xi)=I.
\]
Η τιμή $\xi$ είναι μοναδική, διότι η $f$ είναι γνησίως αύξουσα στο $[0,1]$.

Συνεπώς υπάρχει μοναδικό $\xi\in(0,1)$ τέτοιο ώστε
\[
{f(\xi)=\int_0^1f(x)\d x}.
\]
\end{erwthma}
\end{thema}
