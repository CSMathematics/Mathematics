\begin{thema}{Δ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=3x^2+1>0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $f$ είναι γνησίως αύξουσα και επομένως συνάρτηση 1-1.

Επίσης
\[
\lim_{x\to-\infty}f(x)=-\infty
\quad\text{και}\quad
\lim_{x\to+\infty}f(x)=+\infty,
\]
οπότε
\[
f(\mathbb{R})=\mathbb{R}.
\]
Συνεπώς η $f$ αντιστρέφεται στο $\mathbb{R}$.

Επειδή η $f$ είναι γνησίως αύξουσα, τα κοινά σημεία των $C_f$ και $C_{f^{-1}}$ βρίσκονται στην ευθεία $y=x$. Επομένως λύνουμε την εξίσωση
\begin{align*}
f(x)=x
&\Leftrightarrow x^3+x=x\\
&\Leftrightarrow x^3=0\\
&\Leftrightarrow x=0.
\end{align*}
Άρα οι δύο γραφικές παραστάσεις έχουν μοναδικό κοινό σημείο το
\[
{O(0,0)}.
\]

\item Επειδή
\[
f(0)=0
\quad\text{και}\quad
f(2)=10,
\]
έχουμε
\[
f^{-1}(0)=0
\quad\text{και}\quad
f^{-1}(10)=2.
\]
Η $f^{-1}$ είναι γνησίως αύξουσα, οπότε $f^{-1}(x)\geq0$ για $x\in[0,10]$. Το ζητούμενο εμβαδόν είναι
\[
E=\int_0^{10}f^{-1}(x)\d x.
\]
Χρησιμοποιώντας τον τύπο ολοκληρώματος αντίστροφης συνάρτησης, παίρνουμε
\begin{align*}
E
&=10\cdot2-\int_0^2f(x)\d x\\
&=20-\int_0^2(x^3+x)\d x\\
&=20-\left[\frac{x^4}{4}+\frac{x^2}{2}\right]_0^2\\
&=20-(4+2)\\
&={14}.
\end{align*}

\item Το εμβαδόν του τριγώνου $OAM$ είναι
\[
E=\frac{1}{2}xy.
\]
Επειδή το σημείο $M$ κινείται πάνω στη $C_f$, έχουμε
\[
y=x^3+x.
\]
Άρα
\[
E=\frac{1}{2}x(x^3+x)
=\frac{1}{2}(x^4+x^2).
\]
Παραγωγίζοντας ως προς τον χρόνο $t$, παίρνουμε
\[
\frac{\d E}{\d t}
=(2x^3+x)\frac{\d x}{\d t}.
\]
Τη χρονική στιγμή $t_0$ είναι $y=10$, οπότε
\[
x^3+x=10.
\]
Επειδή η $f$ είναι 1-1 και $f(2)=10$, έχουμε $x=2$. Επίσης
\[
\frac{\d x}{\d t}=2.
\]
Επομένως
\begin{align*}
\left.\frac{\d E}{\d t}\right|_{t=t_0}
&=\left(2\cdot2^3+2\right)\cdot2\\
&=18\cdot2\\
&={36\ \text{τετρ. μονάδες/$\sec$}}.
\end{align*}

\item Για κάθε $x\in\mathbb{R}$ είναι
\[
f(-x)=(-x)^3+(-x)=-(x^3+x)=-f(x).
\]
Άρα η $f$ είναι περιττή. Επειδή είναι και αντιστρέψιμη, η αντίστροφή της είναι επίσης περιττή, δηλαδή
\[
f^{-1}(-x)=-f^{-1}(x).
\]

Η δοσμένη εξίσωση γράφεται
\begin{align*}
f^{-1}(x^3)+f^{-1}(x-10)=0
&\Leftrightarrow f^{-1}(x^3)=-f^{-1}(x-10)\\
&\Leftrightarrow f^{-1}(x^3)=f^{-1}(10-x).
\end{align*}
Επειδή η $f^{-1}$ είναι 1-1, προκύπτει
\begin{align*}
x^3&=10-x\\
\Leftrightarrow x^3+x&=10\\
\Leftrightarrow f(x)&=f(2)\\
\Leftrightarrow {x=2}.
\end{align*}
\end{erwthma}
\end{thema}
