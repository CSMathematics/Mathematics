%=========== ΛΥΣΗ - 35ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
\begin{erwthma}
\item Για κάθε $x>0$ η δοσμένη σχέση γράφεται
\[
x^2f'(x)+2xf(x)=\frac{1}{x}.
\]
Όμως
\[
\left(x^2f(x)\right)'=x^2f'(x)+2xf(x),
\]
οπότε
\[
\left(x^2f(x)\right)'=\frac{1}{x}.
\]
Επομένως
\[
x^2f(x)=\ln x+c,
\]
όπου $c\in\mathbb{R}$. Επειδή $f(1)=0$, έχουμε
\[
0=\ln1+c
\Leftrightarrow
c=0.
\]
Άρα
\[
{f(x)=\frac{\ln x}{x^2}},\quad x>0.
\]

\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\begin{align*}
f'(x)
&=\left(\frac{\ln x}{x^2}\right)'\\
&=\frac{1-2\ln x}{x^3}.
\end{align*}
Επειδή $x^3>0$, το πρόσημο της $f'$ είναι το πρόσημο της παράστασης $1-2\ln x$. Έχουμε:
\begin{itemize}
\item $f'(x)>0\Leftrightarrow 0<x<\sqrt{e}$,
\item $f'(x)=0\Leftrightarrow x=\sqrt{e}$,
\item $f'(x)<0\Leftrightarrow x>\sqrt{e}$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε το πρόσημο της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.8,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/1}%
{$0$,$\sqrt{e}$,$+\infty$}%
\tkzTabLine{d, +, z, -, }
\tkzTabVar{D-/{-\infty},+/{\frac{1}{2e}},-/{0}}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,\sqrt{e}]$ και \textbf{γνησίως φθίνουσα} στο $[\sqrt{e},+\infty)$. Επομένως παρουσιάζει ολικό μέγιστο στο $x_0=\sqrt{e}$, με τιμή
\[
{f(\sqrt{e})=\frac{1}{2e}}.
\]
Δεν παρουσιάζει ελάχιστο, αφού
\[
\lim_{x\to0^+}f(x)=-\infty.
\]
Επιπλέον,
\[
\lim_{x\to+\infty}f(x)=0.
\]
Συνεπώς το σύνολο τιμών της $f$ είναι
\[
{f\big((0,+\infty)\big)
=\left(-\infty,\frac{1}{2e}\right]}.
\]

\item Έχουμε
\[
\lim_{x\to0^+}\frac{\ln x}{x^2}=-\infty,
\]
οπότε η ευθεία
\[
{x=0}
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Επίσης
\[
\lim_{x\to+\infty}\frac{\ln x}{x^2}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{1}{2x^2}
=0.
\]
Άρα η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

Η εξίσωση
\[
10\ln x=x^2
\]
είναι ισοδύναμη με
\[
f(x)=\frac{1}{10}.
\]
Έχουμε
\[
f(1)=0<\frac{1}{10}
\]
και
\[
f(\sqrt{e})=\frac{1}{2e}>\frac{1}{10},
\]
αφού $e<5$. Επομένως, από το Θεώρημα Ενδιάμεσων Τιμών, υπάρχει μία τουλάχιστον ρίζα
\[
x_1\in(1,\sqrt{e}).
\]
Η ρίζα αυτή είναι μοναδική, διότι η $f$ είναι γνησίως αύξουσα στο $(0,\sqrt{e}]$.

Επίσης
\[
\lim_{x\to+\infty}f(x)=0<\frac{1}{10}.
\]
Άρα υπάρχει $M>\sqrt{e}$ τέτοιο ώστε
\[
f(M)<\frac{1}{10}.
\]
Επομένως υπάρχει μία τουλάχιστον ρίζα
\[
x_2\in(\sqrt{e},M),
\]
η οποία είναι μοναδική, διότι η $f$ είναι γνησίως φθίνουσα στο $[\sqrt{e},+\infty)$.

Συνεπώς η εξίσωση $10\ln x=x^2$ έχει ακριβώς δύο θετικές ρίζες.

\item Επειδή
\[
f(1)=0
\]
και $f(x)>0$ για κάθε $x>1$, το ζητούμενο εμβαδόν είναι
\[
E=\int_1^e\frac{\ln x}{x^2}\d x.
\]
Μια παράγουσα της $f$ είναι
\[
F(x)=-\frac{\ln x+1}{x},
\]
οπότε
\begin{align*}
E
&=\left[-\frac{\ln x+1}{x}\right]_1^e\\
&=-\frac{2}{e}+1\\
&={1-\frac{2}{e}}.
\end{align*}

Από τη γνωστή ανισότητα
\[
e^u\geq u+1,
\]
με ισότητα μόνο για $u=0$, θέτοντας $u=f(x)$ παίρνουμε
\[
e^{f(x)}\geq f(x)+1.
\]
Για κάθε $x\in(1,e]$ είναι $f(x)>0$, άρα
\[
e^{f(x)}>f(x)+1.
\]
Ολοκληρώνοντας στο $[1,e]$, προκύπτει
\begin{align*}
\int_1^e e^{f(x)}\d x
&>\int_1^e\left[f(x)+1\right]\d x\\
&=\int_1^e f(x)\d x+\int_1^e1\d x\\
&=E+e-1.
\end{align*}
Επομένως
\[
{\int_1^e e^{f(x)}\d x>E+e-1}.
\]
\end{erwthma}
\end{thema}
