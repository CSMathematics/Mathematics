%=========== ΛΥΣΗ - 80ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Θεωρούμε τη συνάρτηση
\[
h(x)=f(x)-x,\quad x\in[0,6].
\]
Η $h$ είναι συνεχής στο $[0,6]$ και δύο φορές παραγωγίσιμη στο $(0,6)$.

Η εξίσωση
\[
f(x)=x
\]
είναι ισοδύναμη με
\[
h(x)=0.
\]
Από την εκφώνηση γνωρίζουμε ότι η $C_f$ τέμνει την ευθεία $y=x$
ακριβώς στα
\[
x=0,\quad x=3,\quad x=5.
\]
Άρα
\[
h(0)=h(3)=h(5)=0.
\]

\begin{erwthma}
\item Από τα παραπάνω, η εξίσωση
\[
f(x)=x
\]
έχει λύσεις
\[
{x=0,\ x=3,\ x=5}.
\]

Για την ανίσωση $f(x)>x$, δηλαδή
\[
h(x)>0,
\]
μελετάμε το πρόσημο της $h$ στα διαστήματα που ορίζουν οι ρίζες της.

Έχουμε
\[
h(2)=f(2)-2=3-2=1>0.
\]
Άρα, επειδή η $h$ είναι συνεχής και δεν έχει άλλη ρίζα στο $(0,3)$,
ισχύει
\[
h(x)>0,\quad x\in(0,3).
\]

Επίσης
\[
h(4)=f(4)-4=3-4=-1<0.
\]
Άρα
\[
h(x)<0,\quad x\in(3,5).
\]

Τέλος
\[
h(6)=f(6)-6=\frac{13}{2}-6=\frac12>0.
\]
Άρα
\[
h(x)>0,\quad x\in(5,6].
\]

Επομένως
\[
f(x)>x
\Leftrightarrow
h(x)>0
\Leftrightarrow
x\in(0,3)\cup(5,6].
\]
Άρα η λύση της ανίσωσης είναι
\[
{x\in(0,3)\cup(5,6]}.
\]

\item Επειδή
\[
h(0)=h(3),
\]
από το θεώρημα \eng{Rolle} στο $[0,3]$ υπάρχει
\[
\xi_1\in(0,3)
\]
τέτοιο ώστε
\[
h'(\xi_1)=0.
\]

Επειδή
\[
h(3)=h(5),
\]
από το θεώρημα \eng{Rolle} στο $[3,5]$ υπάρχει
\[
\xi_2\in(3,5)
\]
τέτοιο ώστε
\[
h'(\xi_2)=0.
\]

Όμως
\[
h'(x)=f'(x)-1.
\]
Άρα
\[
h'(\xi_1)=0
\Leftrightarrow
f'(\xi_1)=1
\]
και
\[
h'(\xi_2)=0
\Leftrightarrow
f'(\xi_2)=1.
\]

Επειδή
\[
\xi_1\in(0,3)
\quad\text{και}\quad
\xi_2\in(3,5),
\]
οι ρίζες αυτές είναι διαφορετικές και ανήκουν στο $(0,5)$.

Επομένως η εξίσωση
\[
f'(x)=1
\]
έχει τουλάχιστον δύο ρίζες στο διάστημα $(0,5)$.

\item Από το προηγούμενο ερώτημα υπάρχουν
\[
\xi_1\in(0,3)
\quad\text{και}\quad
\xi_2\in(3,5)
\]
τέτοια ώστε
\[
h'(\xi_1)=h'(\xi_2)=0.
\]
Η $h'$ είναι συνεχής στο $[\xi_1,\xi_2]$ και παραγωγίσιμη στο
$(\xi_1,\xi_2)$.

Άρα, από το θεώρημα \eng{Rolle} στη συνάρτηση $h'$ στο
$[\xi_1,\xi_2]$, υπάρχει
\[
\eta\in(\xi_1,\xi_2)\subset(0,5)
\]
τέτοιο ώστε
\[
h''(\eta)=0.
\]
Όμως
\[
h''(x)=f''(x),
\]
αφού $(x)''=0$. Επομένως
\[
{f''(\eta)=0}
\]
για κάποιο $\eta\in(0,5)$.

\item Από το α΄ ερώτημα γνωρίζουμε ότι
\[
f(x)>x,\quad x\in(0,3).
\]
Άρα
\[
\int_0^3 f(x)\d x
>
\int_0^3 x\d x.
\]
Υπολογίζουμε
\[
\int_0^3 x\d x=\frac{3^2}{2}=\frac92.
\]
Επομένως
\[
{\int_0^3 f(x)\d x>\frac92}.
\]
\end{erwthma}
\end{thema}
