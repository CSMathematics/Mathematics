%=========== ΛΥΣΗ - 40ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το πεδίο ορισμού της $f(x) = x + \frac{1}{x}$ για $x > 0$ είναι το $(0, +\infty)$.

\textbf{Κατακόρυφη Ασύμπτωτη:}
Εξετάζουμε το όριο στο ανοιχτό άκρο $x \to 0^+$:
\[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \left( x + \frac{1}{x} \right) = 0 + (+\infty) = +\infty \]
Επομένως, η ευθεία $\boldsymbol{x = 0}$ (ο άξονας $y'y$) είναι \textbf{κατακόρυφη ασύμπτωτη} της $C_f$.

\textbf{Πλάγια Ασύμπτωτη στο $+\infty$:}
Αναζητούμε ευθεία $y = \lambda x + \beta$. Υπολογίζουμε:
\begin{align*}
\lambda &= \lim_{x \to +\infty} \frac{f(x)}{x} = \lim_{x \to +\infty} \frac{x + \frac{1}{x}}{x} = \lim_{x \to +\infty} \left( 1 + \frac{1}{x^2} \right) = 1 \\
\beta &= \lim_{x \to +\infty} (f(x) - \lambda x) = \lim_{x \to +\infty} \left( x + \frac{1}{x} - x \right) = \lim_{x \to +\infty} \frac{1}{x} = 0
\end{align*}
Επομένως, η \textbf{πλάγια ασύμπτωτη} είναι η ευθεία $\boldsymbol{y = x}$.

\item Για $a > 0$, η κατακόρυφη ευθεία $x = a$ τέμνει:
\begin{itemize}
    \item Τη γραφική παράσταση $C_f$ στο σημείο $A\left(a, a + \dfrac{1}{a}\right)$
    \item Την πλάγια ασύμπτωτη $y = x$ στο σημείο $B(a, a)$
\end{itemize}
Τα $A$ και $B$ έχουν ίδια τετμημένη, άρα το τμήμα $AB$ είναι κατακόρυφο και το μήκος του ισούται με:
\[ (AB) = |y_A - y_B| = \left| a + \frac{1}{a} - a \right| = \frac{1}{a} \]
(η απόλυτη τιμή καταργείται γιατί $a > 0$). Άρα: $\boldsymbol{(AB) = \dfrac{1}{a}}$.

\item Ζητείται το $\lim\limits_{a \to +\infty} [a \cdot (AB)]$.
Από το Γ2, $(AB) = \dfrac{1}{a}$, οπότε:
\[ a \cdot (AB) = a \cdot \frac{1}{a} = 1 \]
Η παράσταση είναι σταθερή, άρα:
\[ \lim_{a \to +\infty} [a \cdot (AB)] = \lim_{a \to +\infty} 1 = \boldsymbol{1} \]

\item Το σημείο $M(x(t), y(t))$ κινείται πάνω στην $C_f$, οπότε $y(t) = f(x(t)) = x(t) + \dfrac{1}{x(t)}$.
Παραγωγίζουμε ως προς $t$ (κανόνας αλυσίδας):
\[ \frac{\d y}{\d t} = \frac{\d}{\d t}\left[ x(t) + \frac{1}{x(t)} \right] = \frac{\d x}{\d t} - \frac{1}{x^2(t)} \cdot \frac{\d x}{\d t} = \frac{\d x}{\d t} \cdot \left( 1 - \frac{1}{x^2(t)} \right) \]
Δίνεται $\dfrac{\d x}{\d t} = 2 \ \mu/\si{sec}$ (σταθερός ρυθμός).
Για τη χρονική στιγμή $t_0$ που $x(t_0) = 3$:
\[ \frac{\d y}{\d t}\bigg|_{t_0} = 2 \cdot \left( 1 - \frac{1}{3^2} \right) = 2 \cdot \frac{8}{9} = \boldsymbol{\frac{16}{9} \ \mu/\si{sec}} \]
\end{erwthma}
\end{thema}
