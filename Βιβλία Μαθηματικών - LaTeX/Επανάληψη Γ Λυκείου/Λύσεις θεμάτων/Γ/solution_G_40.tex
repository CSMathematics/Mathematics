%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.40 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Δίνεται η $f(x) = x + \frac{1}{x}$ με πεδίο ορισμού $(0, +\infty)$.
\textbf{Κατακόρυφες Ασύμπτωτες:} 
Εξετάζουμε το όριο στο ανοιχτό άκρο του πεδίου ορισμού $0^+$:
\[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} \left( x + \frac{1}{x} \right) = 0 + (+\infty) = +\infty \]
Αφού το όριο είναι άπειρο, η ευθεία $\boldsymbol{x = 0}$ (δηλαδή ο άξονας $y'y$) είναι \textbf{κατακόρυφη ασύμπτωτη} της $C_f$.

\textbf{Πλάγιες / Οριζόντιες Ασύμπτωτες στο $+\infty$:}
Αναζητούμε ευθεία της μορφής $y = \lambda x + \beta$. Υπολογίζουμε την κλίση $\lambda$:
\begin{align*}
\lambda &= \lim_{x \to +\infty} \frac{f(x)}{x} = \lim_{x \to +\infty} \frac{x + \frac{1}{x}}{x} = \lim_{x \to +\infty} \left( 1 + \frac{1}{x^2} \right) \\
&= 1 + 0 = 1
\end{align*}
Υπολογίζουμε τη σταθερά $\beta$:
\begin{align*}
\beta &= \lim_{x \to +\infty} (f(x) - \lambda x) = \lim_{x \to +\infty} \left( x + \frac{1}{x} - 1 \cdot x \right) \\
&= \lim_{x \to +\infty} \frac{1}{x} = 0
\end{align*}
Επομένως, η ευθεία \textbf{$\boldsymbol{y = x}$} $\left( \text{η διχοτόμος του 1ου 4ου} \right)$ είναι η \textbf{πλάγια ασύμπτωτη} της $C_f$ στο $+\infty$.

\item Θεωρούμε κατακόρυφη ευθεία $x = a$ (με $a > 0$). 
Το σημείο βρισκόμενο πάνω στη δοσμένη συνάρτηση $C_f$ έχει συντεταγμένες:
\[ A \left( a, f(a) \right) = A \left( a, a + \frac{1}{a} \right) \]
Το σημείο πάνω στην πλάγια ασύμπτωτη ευθεία ($y=x$) έχει συντεταγμένες:
\[ B(a, a) \]
Επειδή και τα δύο σημεία βρίσκονται στην ίδια κατακόρυφη ευθεία (ίδια τετμημένη), η απόστασή τους (το μήκος του ευθύγραμμου τμήματος) ταυτίζεται με την απόλυτη διαφορά των τεταγμένων τους:
\[ (AB) = |y_A - y_B| = \left| a + \frac{1}{a} - a \right| = \left| \frac{1}{a} \right| \]
Λόγω της συνθήκης $a > 0$, η απόλυτη τιμή καταργείται αυτόματα:
\[ \boldsymbol{(AB) = \frac{1}{a}} \]

\item Ζητείται η αξιολόγηση του ορίου $\lim_{a \to +\infty} [a \cdot (AB)]$.
Γνωρίζοντας από το ερώτημα 2 ότι $(AB) = \frac{1}{a}$, το γινόμενο μετασχηματίζεται άμεσα:
\[ a \cdot (AB) = a \cdot \frac{1}{a} = 1 \]
Ως εκ τούτου, πρόκειται για όριο μιας απολύτως σταθερής έκφρασης:
\[ \lim_{a \to +\infty} [a \cdot (AB)] = \lim_{a \to +\infty} 1 = \boldsymbol{1} \]

\item Έχουμε την παραμετρική διαδρομή του κινητού $M(x(t), y(t))$ πάνω στην καμπύλη. 
Η τεταγμένη $y(t)$ συνδέεται με τη χρονική εξίσωση:
\[ y(t) = f(x(t)) = x(t) + \frac{1}{x(t)} \]
Παραγωγίζουμε ως προς το χρόνο $t$ για να βρούμε τους ρυθμούς μεταβολής, κάνοντας χρήση του κανόνα σύνθετης (αλυσίδα):
\[ y'(t) = x'(t) + \left( -\frac{1}{x^2(t)} \right) \cdot x'(t) = x'(t) \left( 1 - \frac{1}{x^2(t)} \right) \]
Μας δίνονται τα παρακάτω δεδομένα για την κρίσιμη χρονική στιγμή $t_0$:
- Η τετμημένη αυξάνεται με σταθερό ρυθμό, άρα $x'(t_0) = 2 \ \mu/\text{sec}$.
- Το σημείο διέρχεται από το $x(t_0) = 3$.
Αντικαθιστούμε τα δεδομένα στη διαφορική σχέση:
\begin{align*}
y'(t_0) &= 2 \cdot \left( 1 - \frac{1}{3^2} \right) = 2 \cdot \left( 1 - \frac{1}{9} \right) \\
&= 2 \cdot \left( \frac{9-1}{9} \right) = 2 \cdot \frac{8}{9} = \boldsymbol{\frac{16}{9} \ \mu/\text{sec}}
\end{align*}
Ο ρυθμός μεταβολής της τεταγμένης επιλύεται σε $\frac{16}{9} \ \mu/\text{sec}$.

\end{erwthma}
\end{thema}
