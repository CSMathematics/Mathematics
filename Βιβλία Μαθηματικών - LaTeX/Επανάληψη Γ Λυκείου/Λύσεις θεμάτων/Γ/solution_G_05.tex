%=========== ΛΥΣΗ - 5ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για κάθε $x\ge 0$ ισχύει
\[
f'(x)=\frac{1}{x^2+1}>0,
\]
άρα η $f$ είναι γνησίως αύξουσα στο $[0,+\infty)$.

Ακόμη,
\[
f''(x)=\left(\frac{1}{x^2+1}\right)'=-\frac{2x}{(x^2+1)^2}.
\]
Για $x>0$ ισχύει $f''(x)<0$, άρα η $f$ είναι κοίλη στο $(0,+\infty)$.

\item Επειδή $f(0)=0$ και
\[
f'(0)=1,
\]
η εφαπτομένη της $C_f$ στο $x_0=0$ είναι
\[
y-f(0)=f'(0)(x-0)
\]
δηλαδή
\[
y=x.
\]
Εφόσον η $f$ είναι κοίλη στο $(0,+\infty)$, η γραφική της παράσταση βρίσκεται κάτω από κάθε εφαπτομένη της. Άρα
\[
f(x)\le x,\quad x\ge 0.
\]

\item Εφαρμόζουμε το θεώρημα Μέσης Τιμής στο διάστημα $[x,x+1]$. Υπάρχει $c\in(x,x+1)$ τέτοιο ώστε
\[
f(x+1)-f(x)=f'(c).
\]
Επειδή
\[
f''(x)<0,\quad x>0,
\]
η $f'$ είναι γνησίως φθίνουσα στο $(0,+\infty)$. Αφού $c>x$, έχουμε
\[
f'(c)<f'(x)=\frac{1}{x^2+1}.
\]
Άρα
\[
f(x+1)-f(x)<\frac{1}{x^2+1}.
\]

\item Το όριο είναι της μορφής $\dfrac{0}{0}$, αφού $f(0)=0$. Εφαρμόζουμε τον κανόνα \eng{DLH}:
\[
\lim_{x\to 0^+}\frac{f(x)-x}{x^2}
\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}
\lim_{x\to 0^+}\frac{f'(x)-1}{2x}.
\]
Άρα
\[
\lim_{x\to 0^+}\frac{f(x)-x}{x^2}
=
\lim_{x\to 0^+}\frac{\frac{1}{x^2+1}-1}{2x}
=
\lim_{x\to 0^+}\frac{-x}{2(x^2+1)}=0.
\]
\end{erwthma}
\end{thema}
