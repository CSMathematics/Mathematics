\begin{thema}{Γ}
\begin{erwthma}
  \item 
  Η συνάρτηση $f(x) = x^3 - 3x + 1$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
  \[ f'(x) = 3x^2 - 3 = 3(x^2 - 1) \]
  Οι ρίζες της παραγώγου είναι:
  \[ f'(x) = 0 \Leftrightarrow x^2 - 1 = 0 \Leftrightarrow x = -1 \text{ ή } x = 1 \]
  Το πρόσημο της $f'$ (τριώνυμο με θετικό συντελεστή στο $x^2$) είναι αρνητικό εντός των ριζών και θετικό εκτός:
  \begin{center}
  \begin{tikzpicture}
    \tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
    {$x$ / .8, $f'(x)$ / 0.8, $f(x)$ / 1.5}
    { $-\infty$, $-1$, $1$, $+\infty$ }
    \tkzTabLine{ , +, z, -, z, +, }
    \tkzTabVar{ -/,+/,-/,+/}
  \end{tikzpicture}
  \end{center}
  Η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στα διαστήματα $(-\infty, -1]$ και $[1, +\infty)$, ενώ είναι \textbf{γνησίως φθίνουσα} στο $[-1, 1]$.
  Παρουσιάζει:
  \begin{itemize}
    \item Τοπικό μέγιστο για $x_1 = -1$, με τιμή $f(-1) = (-1)^3 - 3(-1) + 1 = 3$.
    \item Τοπικό ελάχιστο για $x_2 = 1$, με τιμή $f(1) = 1^3 - 3(1) + 1 = -1$.
  \end{itemize}

  \item 
  Για να βρούμε το πλήθος των ριζών της εξίσωσης $f(x) = 0$, βρίσκουμε το σύνολο τιμών της $f$ σε κάθε διάστημα μονοτονίας:
  \begin{itemize}
    \item Έστω $\Delta_1 = (-\infty, -1]$. Η $f$ είναι συνεχής και γνησίως αύξουσα στο $\Delta_1$, άρα το σύνολο τιμών της είναι $f(\Delta_1) = \left( \lim\limits_{x \to -\infty} f(x), f(-1) \right] = (-\infty, 3]$. 
    Επειδή το $0 \in f(\Delta_1)$, η εξίσωση $f(x)=0$ έχει ρίζα στο $\Delta_1$, η οποία είναι \textbf{μοναδική} (έστω $\rho_1 \in (-\infty, -1)$) λόγω της γνήσιας μονοτονίας.
    \item Έστω $\Delta_2 = [-1, 1]$. Η $f$ είναι συνεχής και γνησίως φθίνουσα, άρα $f(\Delta_2) = \left[ f(1), f(-1) \right] = [-1, 3]$.
    Επειδή το $0 \in f(\Delta_2)$, υπάρχει \textbf{μοναδική} ρίζα $\rho_2 \in (-1, 1)$.
    \item Έστω $\Delta_3 = [1, +\infty)$. Η $f$ είναι συνεχής και γνησίως αύξουσα, άρα $f(\Delta_3) = \left[ f(1), \lim\limits_{x \to +\infty} f(x) \right) = [-1, +\infty)$.
    Επειδή το $0 \in f(\Delta_3)$, υπάρχει \textbf{μοναδική} ρίζα $\rho_3 \in (1, +\infty)$.
  \end{itemize}
  Άρα η εξίσωση $f(x) = 0$ έχει ακριβώς τρεις πραγματικές ρίζες $\rho_1 < x_1 < \rho_2 < x_2 < \rho_3$.

  \item 
  Υπολογίζουμε τη δεύτερη παράγωγο της $f$:
  \[ f''(x) = (3x^2 - 3)' = 6x \]
  Ρίζες και πρόσημο της $f''$:
  \[ f''(x) = 0 \Leftrightarrow 6x = 0 \Leftrightarrow x = 0 \]
  \[ f''(x) > 0 \Leftrightarrow x > 0 \quad \text{και} \quad f''(x) < 0 \Leftrightarrow x < 0 \]
  Επειδή η $f''$ αλλάζει πρόσημο εκατέρωθεν του $0$, το σημείο με τετμημένη $0$ είναι \textbf{σημείο καμπής}. Η τεταγμένη του είναι $f(0) = 1$, άρα το σημείο καμπής είναι το $K(0, 1)$.
  
  Η κλίση της εφαπτομένης $\varepsilon$ στο σημείο $K$ είναι:
  \[ \lambda = f'(0) = 3(0)^2 - 3 = -3 \]
  Η εξίσωση της εφαπτομένης $\varepsilon$ στο $K(0,1)$ είναι:
  \[ y - f(0) = f'(0)(x - 0) \Leftrightarrow y - 1 = -3x \Leftrightarrow y = -3x + 1 \]

  \item 
  Το ζητούμενο εμβαδόν $E$ περικλείεται από τη γραφική παράσταση $C_f$, την εφαπτομένη $\varepsilon$: $y = -3x+1$ και την κατακόρυφη ευθεία $x = 1$. 
  Το άλλο όριο ολοκλήρωσης προκύπτει από το σημείο επαφής των δύο καμπυλών (καθώς και τομής τους), το οποίο είναι το $x=0$.
  Βρίσκουμε τη διαφορά τους:
  \[ f(x) - y = (x^3 - 3x + 1) - (-3x + 1) = x^3 \]
  Για $x \in [0, 1]$ ισχύει $x^3 \ge 0$, άρα η $C_f$ βρίσκεται πάνω από την εφαπτομένη $\varepsilon$. 
  Το εμβαδόν του χωρίου είναι:
  \[ E = \int_0^1 |f(x) - y| \d x = \int_0^1 x^3 \d x = \left[ \frac{x^4}{4} \right]_0^1 = \frac{1^4}{4} - 0 = \frac{1}{4} \text{ τ.μ.} \]
\end{erwthma}
\end{thema}
