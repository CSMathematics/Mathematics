%=========== ΛΥΣΗ - 10ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Γράφουμε
\[
f(x)=\left(x^2-ax+2\right)e^{-x}.
\]
Τότε
\[
f'(x)=\left(2x-a\right)e^{-x}-\left(x^2-ax+2\right)e^{-x}
\]
δηλαδή
\[
f'(x)=\left(-x^2+(a+2)x-a-2\right)e^{-x}.
\]
Άρα
\begin{align*}
f''(x)
&=
\left(-2x+a+2\right)e^{-x}-\left(-x^2+(a+2)x-a-2\right)e^{-x}\\
&=
\left(x^2-(a+4)x+2a+4\right)e^{-x}.
\end{align*}
Εφόσον η $f$ παρουσιάζει σημείο καμπής για $x_0=1$, ισχύει
\[
f''(1)=0.
\]
Άρα
\[
\left(1-(a+4)+2a+4\right)e^{-1}=0
\]
οπότε
\[
a+1=0
\]
και τελικά
\[
a=-1.
\]

\item Για $a=-1$:
\[
f(x)=\frac{x^2+x+2}{e^x}
\]
και
\[
f'(x)=\left(-x^2+x-1\right)e^{-x}.
\]
Επειδή $e^{-x}>0$, το πρόσημο της $f'(x)$ είναι το ίδιο με το πρόσημο του τριωνύμου
\[
p(x)=-x^2+x-1.
\]
Η διακρίνουσά του είναι
\[
\Delta=1-4=-3<0.
\]
Αφού ο συντελεστής του $x^2$ είναι αρνητικός, έχουμε
\[
p(x)<0,\quad x\in \mathbb{R}.
\]
Άρα
\[
f'(x)<0,\quad x\in \mathbb{R}
\]
και επομένως η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$.
Συνεπώς δεν παρουσιάζει τοπικά ακρότατα.

\item Αφού η $f$ είναι γνησίως φθίνουσα και $2<3$, ισχύει
\[
f(2)>f(3).
\]

\item Έχουμε
\[
\lim_{x\to +\infty}\frac{x^2+x+2}{e^x}=0,
\]
άρα η ευθεία
\[
y=0
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

Επίσης,
\[
\lim_{x\to -\infty}(x^2+x+2)e^{-x}=+\infty,
\]
οπότε στο $-\infty$ δεν υπάρχει ασύμπτωτη.
\end{erwthma}
\end{thema}
