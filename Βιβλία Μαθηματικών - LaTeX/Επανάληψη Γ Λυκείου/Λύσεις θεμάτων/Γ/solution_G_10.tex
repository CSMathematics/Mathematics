%=========== ΛΥΣΗ - 10ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση γράφεται $f(x)=\left(x^2-ax+2\right)e^{-x}$. Είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}$ με:
\begin{align*}
f'(x) &= (2x-a)e^{-x} - \left(x^2-ax+2\right)e^{-x} = \left(-x^2+(a+2)x-a-2\right)e^{-x} \\
f''(x) &= (-2x+a+2)e^{-x} - \left(-x^2+(a+2)x-a-2\right)e^{-x} = \left(x^2-(a+4)x+2a+4\right)e^{-x}
\end{align*}
Εφόσον η γραφική παράσταση της $f$ παρουσιάζει σημείο καμπής στο σημείο με τετμημένη $x_0=1$, αναγκαία συνθήκη είναι:
\[f''(1)=0 \Leftrightarrow \left(1^2-(a+4)\cdot 1+2a+4\right)e^{-1}=0\]
Επειδή $e^{-1}\neq 0$, παίρνουμε:
\[1-a-4+2a+4=0 \Leftrightarrow a+1=0 \Leftrightarrow a=-1\]
Επαλήθευση: Για $a=-1$, είναι $f''(x)=\left(x^2-3x+2\right)e^{-x}$. Οι ρίζες του τριωνύμου είναι $x=1$ και $x=2$. Η $f''$ αλλάζει πρόσημο εκατέρωθεν του $1$, άρα πράγματι το $x_0=1$ είναι θέση σημείου καμπής. Άρα $a=-1$.

\item Για $a=-1$, ο τύπος της $f$ είναι $f(x)=\dfrac{x^2+x+2}{e^x}$ και η πρώτη παράγωγος:
\[f'(x)=\left(-x^2+x-1\right)e^{-x}\]
Επειδή $e^{-x}>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f'$ εξαρτάται αποκλειστικά από το πρόσημο του τριωνύμου $-x^2+x-1$.
Η διακρίνουσα του τριωνύμου είναι:
\[\Delta=1^2-4(-1)(-1)=1-4=-3<0\]
Αφού $\Delta<0$ και ο συντελεστής του $x^2$ είναι αρνητικός ($-1$), το τριώνυμο είναι αρνητικό για κάθε $x\in\mathbb{R}$. Δηλαδή:
\[-x^2+x-1<0 \Rightarrow f'(x)<0 \text{ για κάθε } x\in\mathbb{R}\]
Επομένως, η $f$ είναι \textbf{γνησίως φθίνουσα} στο $\mathbb{R}$. Αφού είναι γνησίως μονότονη στο πεδίο ορισμού της, η συνάρτηση \textbf{δεν παρουσιάζει τοπικά ακρότατα}.

\item Αφού η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$, η διάταξη των τιμών της είναι αντίστροφη από τη διάταξη των $x$. Επειδή $2 < 3$, προκύπτει άμεσα ότι:
\[f(2) > f(3)\]

\item \textbf{Στο $+\infty$:}
Αναζητούμε το όριο $\lim\limits_{x\to+\infty}f(x)=\lim\limits_{x\to+\infty}\dfrac{x^2+x+2}{e^x}$. Η μορφή είναι $\dfrac{+\infty}{+\infty}$, άρα εφαρμόζουμε τον κανόνα \eng{De L'Hospital} (δύο φορές):
\[\lim_{x\to+\infty}\frac{x^2+x+2}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty}\frac{(x^2+x+2)'}{(e^x)'} = \lim_{x\to+\infty}\frac{2x+1}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty}\frac{2}{e^x} = 0\]
Άρα η ευθεία \textbf{$y=0$} (δηλαδή ο άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} της $C_f$ στο $+\infty$.

\textbf{Στο $-\infty$:}
Αναζητούμε το όριο $\lim\limits_{x\to-\infty}f(x)=\lim\limits_{x\to-\infty}\left[(x^2+x+2)e^{-x}\right]$. Καθώς $x\to-\infty$, ισχύει $x^2+x+2\to+\infty$ και $e^{-x}\to+\infty$. Άρα:
\[\lim_{x\to-\infty}f(x)=(+\infty)\cdot(+\infty)=+\infty\]
Επειδή το όριο είναι $+\infty$, δεν υπάρχει οριζόντια ασύμπτωτη στο $-\infty$. Ελέγχουμε για \textbf{πλάγια ασύμπτωτη} της μορφής $y=\lambda x+\beta$:
\[\lambda = \lim_{x\to-\infty}\frac{f(x)}{x} = \lim_{x\to-\infty}\left[\frac{x^2+x+2}{x}\cdot e^{-x}\right] = \lim_{x\to-\infty}\left[\left(x+1+\frac{2}{x}\right)e^{-x}\right]\]
Καθώς $x\to-\infty$, η παρένθεση τείνει στο $-\infty$ και το $e^{-x}$ στο $+\infty$, άρα:
\[\lambda = (-\infty)\cdot(+\infty) = -\infty \notin \mathbb{R}\]
Συνεπώς, η $C_f$ \textbf{δεν έχει} ούτε πλάγια (ούτε οριζόντια) ασύμπτωτη στο $-\infty$.
\end{erwthma}
\end{thema}
