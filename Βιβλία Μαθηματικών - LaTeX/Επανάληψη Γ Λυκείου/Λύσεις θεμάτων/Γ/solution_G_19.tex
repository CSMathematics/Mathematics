%=========== ΛΥΣΗ - 19ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f(x) = \frac{\ln x}{x}$ έχει πεδίο ορισμού το $(0, +\infty)$. Είναι παραγωγίσιμη ως πηλίκο παραγωγίσιμων συναρτήσεων, με:
\[ f'(x) = \frac{(\ln x)' x - \ln x (x)'}{x^2} = \frac{\frac{1}{x} \cdot x - \ln x}{x^2} = \frac{1 - \ln x}{x^2} \]
Το πρόσημο της παραγώγου εξαρτάται μόνο από τον αριθμητή, αφού $x^2 > 0$ για $x > 0$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 1 - \ln x = 0 \Leftrightarrow \ln x = 1 \Leftrightarrow x = e$
    \item $f'(x) > 0 \Leftrightarrow 1 - \ln x > 0 \Leftrightarrow \ln x < 1 \Leftrightarrow 0 < x < e$
    \item $f'(x) < 0 \Leftrightarrow 1 - \ln x < 0 \Leftrightarrow \ln x > 1 \Leftrightarrow x > e$
\end{itemize}

Από τον πίνακα διαπιστώνουμε ότι η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0, e]$ και \textbf{γνησίως φθίνουσα} στο $[e, +\infty)$.
Παρουσιάζει \textbf{ολικό μέγιστο} στη θέση $x_0 = e$, με τιμή:
\[ f(e) = \frac{\ln e}{e} = \frac{1}{e} \]

\item Γνωρίζουμε ότι ο αριθμός $e \approx 2.71$ και το $\pi \approx 3.14$, συνεπώς $e < \pi$.
Αφού $e < \pi$, και τα δύο νούμερα ανήκουν στο διάστημα $[e, +\infty)$ όπου η συνάρτηση $f$ είναι \textbf{γνησίως φθίνουσα}. Επομένως, αντιστρέφεται η φορά της ανισότητας:
\[ e < \pi \overset{f \searrow}{\implies} f(e) > f(\pi) \]
Αντικαθιστώντας τον τύπο της $f$:
\[ \frac{\ln e}{e} > \frac{\ln \pi}{\pi} \implies \frac{1}{e} > \frac{\ln \pi}{\pi} \]
Επειδή τα $e$ και $\pi$ είναι θετικοί αριθμοί, πολλαπλασιάζουμε χιαστί χωρίς να αλλάξει η φορά:
\[ \pi > e \ln \pi \implies \pi \ln e > e \ln \pi \implies \ln(e^\pi) > \ln(\pi^e) \implies e^\pi > \pi^e \]

\item Υπολογίζουμε τη δεύτερη παράγωγο της $f$:
\begin{align*}
f''(x) &= \left( \frac{1 - \ln x}{x^2} \right)' = \frac{(1 - \ln x)' x^2 - (1 - \ln x)(x^2)'}{(x^2)^2} \\
&= \frac{-\frac{1}{x} \cdot x^2 - (1 - \ln x) \cdot 2x}{x^4} = \frac{-x - 2x + 2x \ln x}{x^4} \\
&= \frac{x(-3 + 2 \ln x)}{x^4} = \frac{2\ln x - 3}{x^3}
\end{align*}
Το πρόσημο της $f''(x)$ εξαρτάται μόνο από τον αριθμητή, αφού $x^3 > 0$:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow 2\ln x = 3 \Leftrightarrow \ln x = \frac{3}{2} \Leftrightarrow x = e^{\frac{3}{2}} = e\sqrt{e}$
    \item $f''(x) > 0 \Leftrightarrow 2\ln x > 3 \Leftrightarrow x > e^{\frac{3}{2}}$
    \item $f''(x) < 0 \Leftrightarrow 2\ln x < 3 \Leftrightarrow 0 < x < e^{\frac{3}{2}}$
\end{itemize}

Συνεπώς, η $C_f$ είναι \textbf{κοίλη} στο $(0, e^{\frac{3}{2}}]$ και \textbf{κυρτή} στο $[e^{\frac{3}{2}}, +\infty)$.
Το μοναδικό \textbf{σημείο καμπής} είναι το $K(e^{\frac{3}{2}}, f(e^{\frac{3}{2}}))$, όπου:
\[ f(e^{\frac{3}{2}}) = \frac{\ln e^{\frac{3}{2}}}{e^{\frac{3}{2}}} = \frac{\frac{3}{2}}{e\sqrt{e}} = \frac{3}{2e\sqrt{e}} \]

\item Έστω $F$ μια αρχική της $f$ στο $(0, +\infty)$. Τότε, εξ ορισμού ισχύει $F'(x) = f(x)$.
Παρατηρούμε ότι $\left( \frac{1}{2}\ln^2 x \right)' = \frac{1}{2} \cdot 2\ln x \cdot \frac{1}{x} = \frac{\ln x}{x} = f(x)$.
Άρα η γενική μορφή της $F(x)$ είναι:
\[ F(x) = \frac{1}{2}\ln^2 x + c, \quad c \in \mathbb{R} \]
Καθώς το $x \to +\infty$, η ποσότητα $\ln x \to +\infty$, άρα και το $\ln^2 x \to +\infty$.
Επομένως, $\lim_{x \to +\infty} F(x) = +\infty$. Το ζητούμενο όριο έχει τη μορφή:
\[ \lim_{x \to +\infty} \frac{F(x)}{x^2} \quad \left( \text{Μορφή } \frac{+\infty}{+\infty} \right) \]
Εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to +\infty} \frac{F(x)}{x^2} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to +\infty} \frac{F'(x)}{(x^2)'} = \lim_{x \to +\infty} \frac{f(x)}{2x} \]
Αντικαθιστούμε την $f(x)$:
\[ \lim_{x \to +\infty} \frac{\frac{\ln x}{x}}{2x} = \lim_{x \to +\infty} \frac{\ln x}{2x^2} \quad \left( \text{Μορφή } \frac{+\infty}{+\infty} \right) \]
Εφαρμόζουμε ξανά τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to +\infty} \frac{\ln x}{2x^2} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to +\infty} \frac{(\ln x)'}{(2x^2)'} = \lim_{x \to +\infty} \frac{\frac{1}{x}}{4x} = \lim_{x \to +\infty} \frac{1}{4x^2} = 0 \]
Άρα, το όριο ισούται με $\boldsymbol{0}$.
\end{erwthma}
\end{thema}
