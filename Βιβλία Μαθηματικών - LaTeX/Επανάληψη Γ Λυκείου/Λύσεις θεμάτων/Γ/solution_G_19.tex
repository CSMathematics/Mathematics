%=========== ΛΥΣΗ - ΘΕΜΑ Γ ===========

\begin{thema}{Γ}
\begin{erwthma}
\item 
Η συνάρτηση $f(x) = \dfrac{\ln x}{x}$ είναι παραγωγίσιμη στο $D_f = (0, +\infty)$ και η παράγωγός της είναι:
\[
f'(x) = \frac{\frac{1}{x} \cdot x - \ln x \cdot 1}{x^2} = \frac{1 - \ln x}{x^2}
\]
Για τον προσδιορισμό της μονοτονίας της $f$ εξετάζουμε το πρόσημο της $f'(x)$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 1 - \ln x = 0 \Leftrightarrow \ln x = 1 \Leftrightarrow x = e$
    \item $f'(x) > 0 \Leftrightarrow 1 - \ln x > 0 \Leftrightarrow \ln x < 1 \Leftrightarrow 0 < x < e$
    \item $f'(x) < 0 \Leftrightarrow 1 - \ln x < 0 \Leftrightarrow \ln x > 1 \Leftrightarrow x > e$
\end{itemize}
Συνεπώς, η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0, e]$ και \textbf{γνησίως φθίνουσα} στο $[e, +\infty)$.
Για $x = e$, η συνάρτηση παρουσιάζει \textbf{ολικό μέγιστο} με τιμή:
\[
f(e) = \frac{\ln e}{e} = \frac{1}{e}
\]

\item 
Θέλουμε να συγκρίνουμε τους αριθμούς $e^\pi$ και $\pi^e$. Επειδή $e, \pi > 0$, μπορούμε να συγκρίνουμε τους λογάριθμους τους:
\[
\ln(e^\pi) = \pi \ln e = \pi \quad \text{και} \quad \ln(\pi^e) = e \ln \pi
\]
Συγκρίνουμε τους αριθμούς $\pi$ και $e \ln \pi$ διαιρώντας και τους δύο με το γινόμενο $e\pi > 0$:
\[
\frac{\pi}{e\pi} = \frac{1}{e} \quad \text{και} \quad \frac{e \ln \pi}{e\pi} = \frac{\ln \pi}{\pi}
\]
Από το ερώτημα (α) γνωρίζουμε ότι η συνάρτηση $f(x) = \dfrac{\ln x}{x}$ έχει ολικό μέγιστο στο $x = e$. 
Επειδή $\pi \neq e$, ισχύει:
\[
f(\pi) < f(e) \Rightarrow \frac{\ln \pi}{\pi} < \frac{1}{e}
\]
Πολλαπλασιάζοντας με $e\pi > 0$, έχουμε:
\[
e \ln \pi < \pi \Rightarrow \ln(\pi^e) < \ln(e^\pi)
\]
Επειδή η συνάρτηση $\ln x$ είναι \textbf{γνησίως αύξουσα}, εξ follows ότι:
\[
\pi^e < e^\pi
\]

\item 
Υπολογίζουμε τη δεύτερη παράγωγο της $f$:
\[
f''(x) = \left( \frac{1 - \ln x}{x^2} \right)' = \frac{-\frac{1}{x} \cdot x^2 - (1 - \ln x) \cdot 2x}{x^4} = \frac{-x - 2x + 2x \ln x}{x^4} = \frac{-3 + 2 \ln x}{x^3}
\]
Εξετάζουμε το πρόσημο της $f''(x)$ για $x > 0$:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow -3 + 2 \ln x = 0 \Leftrightarrow \ln x = \frac{3}{2} \Leftrightarrow x = e^{3/2}$
    \item $f''(x) > 0 \Leftrightarrow 2 \ln x > 3 \Leftrightarrow \ln x > \frac{3}{2} \Leftrightarrow x > e^{3/2}$
    \item $f''(x) < 0 \Leftrightarrow 2 \ln x < 3 \Leftrightarrow \ln x < \frac{3}{2} \Leftrightarrow 0 < x < e^{3/2}$
\end{itemize}
Άρα, η $C_f$ στρέφει τα \textbf{κοίλα άνω} στο $(0, e^{3/2})$ και τα \textbf{κοίλα κάτω} στο $(e^{3/2}, +\infty)$.
Το μοναδικό \textbf{σημείο καμπής} είναι το $K(e^{3/2}, f(e^{3/2}))$, όπου:
\[
f(e^{3/2}) = \frac{\ln e^{3/2}}{e^{3/2}} = \frac{3/2}{e^{3/2}} = \frac{3}{2e^{3/2}}
\]

\item 
Έστω $F(x)$ μια αρχική της $f(x)$. Για το όριο $\lim\limits_{x \to +\infty} \dfrac{F(x)}{x^2}$, παρατηρούμε ότι καθώς $x \to +\infty$, το $x^2 \to +\infty$. 
Επίσης, $F(x) = \int \frac{\ln x}{x} \d x = \frac{1}{2}(\ln x)^2 + C$, οπότε $F(x) \to +\infty$.
Εφαρμόζοντας τον κανόνα του \eng{L'Hôpital}:
\[
\lim_{x \to +\infty} \frac{F(x)}{x^2} \xlongequal{\eng{L'Hôpital}} \lim_{x \to +\infty} \frac{F'(x)}{2x} = \lim_{x \to +\infty} \frac{f(x)}{2x}
\]
Αντικαθιστώντας την $f(x)$:
\[
\lim_{x \to +\infty} \frac{\frac{\ln x}{x}}{2x} = \lim_{x \to +\infty} \frac{\ln x}{2x^2}
\]
Εφαρμόζοντας ξανά τον κανόνα του \eng{L'Hôpital}:
\[
\lim_{x \to +\infty} \frac{\ln x}{2x^2} \xlongequal{\eng{L'Hôpital}} \lim_{x \to +\infty} \frac{\frac{1}{x}}{4x} = \lim_{x \to +\infty} \frac{1}{4x^2} = 0
\]
Άρα, $\lim\limits_{x \to +\infty} \dfrac{F(x)}{x^2} = 0$.
\end{erwthma}
\end{thema}
