%=========== ΛΥΣΗ - 8ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $x=0$ δίνεται
\[
e^0-1\le f(0)\le 0\cdot e^0+0
\]
οπότε
\[
0\le f(0)\le 0
\]
και άρα
\[
f(0)=0.
\]

Για $h>0$ ισχύει
\[
e^h-1\le f(h)\le he^h+h.
\]
Διαιρώντας με $h>0$, παίρνουμε
\[
\frac{e^h-1}{h}\le \frac{f(h)-f(0)}{h}\le e^h+1.
\]
Περνώντας στο όριο καθώς $h\to 0^+$:
\[
1\le \lim_{h\to 0^+}\frac{f(h)-f(0)}{h}\le 2.
\]
Επομένως
\[
1\le f'(0)\le 2.
\]

\item Για $f(x)=e^x-1$ έχουμε
\[
\lim_{x\to -\infty}(e^x-1)=-1.
\]
Άρα η γραφική παράσταση της συνάρτησης έχει οριζόντια ασύμπτωτη την ευθεία
\[
y=-1
\]
στο $-\infty$.

\item Θέτουμε
\[
g(x)=xe^x+x-(e^x-1)=xe^x-e^x+x+1.
\]
Τότε
\[
g(0)=0
\]
και
\[
g'(x)=e^x+xe^x-e^x+1=xe^x+1.
\]
Για $x\ge 0$ ισχύει $xe^x\ge 0$, άρα
\[
g'(x)>0.
\]
Επομένως η $g$ είναι γνησίως αύξουσα στο $[0,+\infty)$ και, αφού $g(0)=0$, έχουμε
\[
g(x)\ge 0,\quad x\ge 0.
\]
Άρα
\[
xe^x+x\ge e^x-1,\quad x\ge 0.
\]

\item Στο $[0,1]$ ισχύει
\[
g(x)-h(x)=xe^x+x-(e^x-1)=xe^x-e^x+x+1\ge 0,
\]
οπότε το εμβαδόν του χωρίου είναι
\[
E=\int_0^1 \left(xe^x-e^x+x+1\right)\,\d x.
\]
Επειδή
\[
\int xe^x\,\d x=xe^x-e^x,
\]
παίρνουμε
\begin{align*}
E
&=
\left[xe^x-e^x-e^x+\frac{x^2}{2}+x\right]_0^1\\
&=
\left[e-2e+\frac{1}{2}+1\right]-\left[0-2+0+0\right]\\
&=
-e+\frac{3}{2}+2\\
&=
\frac{7}{2}-e.
\end{align*}
\end{erwthma}
\end{thema}
