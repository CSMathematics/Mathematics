\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $h(x) = \ln(\ln f(x))$. Επειδή η $f$ είναι παραγωγίσιμη και $f(x) > 1$ για κάθε $x \ge 0$, η συνάρτηση $h$ είναι παραγωγίσιμη στο $[0, +\infty)$. Εφαρμόζοντας τον κανόνα της αλυσίδας, έχουμε:
\[
h'(x) = \frac{1}{\ln f(x)} \cdot (\ln f(x))' = \frac{1}{\ln f(x)} \cdot \frac{f'(x)}{f(x)}
\]
Από την εκφώνηση γνωρίζουμε ότι $f'(x) = f(x) \ln(f(x))$, επομένως:
\[
h'(x) = \frac{1}{\ln f(x)} \cdot \frac{f(x) \ln(f(x))}{f(x)} = \frac{\ln f(x)}{\ln f(x)} = 1
\]
Άρα $\big(\ln(\ln f(x))\big)' = 1$.
Συνεπώς, η $\ln(\ln f(x))$ είναι της μορφής $x + C$. Για $x=0$, έχουμε $f(0)=e$, οπότε:
\[
\ln(\ln e) = 0 + C \Rightarrow \ln(1) = C \Rightarrow C = 0
\]
Άρα $\ln(\ln f(x)) = x$, το οποίο συνεπάγεται:
\[
\ln f(x) = e^x \Rightarrow f(x) = e^{e^x}
\]

\item Η συνάρτηση $f(x) = e^{e^x}$ είναι γνησίως αύξουσα στο $[0, +\infty)$ καθώς η παράγωγός της είναι:
\[
f'(x) = e^{e^x} \cdot (e^x)' = e^{e^x} \cdot e^x
\]
Επειδή $e^{e^x} > 0$ και $e^x > 0$ για κάθε $x \in \mathbb{R}$, ισχύει $f'(x) > 0$ για κάθε $x \ge 0$.
Εφόσον η $f$ είναι \textbf{συνεχώς} και \textbf{γνησίως αύξουσα} στο $[0, +\infty)$, τότε είναι μονότονη και άρα διαθέτει αντίστροφη συνάρτηση $g(x)$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (\eng{Lagrange}) στη συνάρτηση $f(x) = e^{e^x}$ στο διάστημα $[x, x+1]$ για $x \ge 0$. Η $f$ είναι \textbf{συνεχώς} στο $[x, x+1]$ και \textbf{παραγωγήσιμη} στο $(x, x+1)$. 
Άρα υπάρχει $\xi \in (x, x+1)$ τέτοια ώστε:
\[
\frac{f(x+1) - f(x)}{(x+1) - x} = f'(\xi) \Rightarrow f(x+1) - f(x) = f'(\xi)
\]
Γνωρίζουμε ότι $f'(t) = e^{e^t} \cdot e^t$. Επειδή η $f'$ είναι γνησίως αύξουσα (καθώς το γινόμενο δύο θετικών γνησίων αυξουσών συναρτήσεων είναι γνησίως αύξουσα), και $\xi > x$, ισχύει $f'(\xi) > f'(x)$.
Έτσι:
\[
e^{e^{x+1}} - e^{e^x} = f'(\xi) > f'(x) = e^{e^x} \cdot e^x = e^{x+e^x}
\]
Άρα $e^{e^{x+1}} - e^{e^x} > e^{x+e^x}$ για κάθε $x \ge 0$.

\item Η συνάρτηση $g^{-1}(x)$ είναι η ίδια η συνάρτηση $f(x) = e^{e^x}$. Το εμβαδόν $E$ του χωρίου μεταξύ της $C_{g^{-1}}$, του άξονα $x'x$ και των κατακόρυφων $x=e, x=e^e$ δίνεται από το ορισμένο ολοκλήρωμα:
\[
E = \int_{e}^{e^e} e^{e^x} \d x
\]
Επειδή η συνάρτηση $e^{e^x}$ δεν είναι στοιχειωδώς ολοκληρώσιμη, το εμβαδόν εκφράζεται από το ανωτέρω ολοκλήρωμα. 
\textit{Σημείωση: Αν η εκφώνηση αναφερόταν στην αντίστροφη συνάρτηση $g(x) = \ln(\ln x)$, τότε το εμβαδόν θα ήταν:}
\[
\int_{e}^{e^e} \ln(\ln x) \d x = \left[ x \ln(\ln x) \right]_{e}^{e^e} - \int_{e}^{e^e} \frac{x}{x \ln x} \d x = e^e \ln(e) - e \ln(1) - \int_{e}^{e^e} \frac{1}{\ln x} \d x = e^e - \int_{e}^{e^e} \frac{1}{\ln x} \d x
\]
\end{erwthma}
\end{thema}
