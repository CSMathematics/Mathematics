%=========== ΛΥΣΗ - 76ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Επειδή $f(x)>1$ για κάθε $x\geq 0$, έχουμε
\[
\ln f(x)>0
\]
οπότε ορίζεται η συνάρτηση
\[
h(x)=\ln(\ln f(x))
\]
Η $h$ είναι παραγωγίσιμη και
\begin{align*}
h'(x)
&=\frac{1}{\ln f(x)}\cdot \frac{f'(x)}{f(x)}\\
&=\frac{1}{\ln f(x)}\cdot \frac{f(x)\ln f(x)}{f(x)}\\
&=1
\end{align*}
Άρα
\[
\left(\ln(\ln f(x))\right)'=(x)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
\ln(\ln f(x))=x+c
\]
Για $x=0$ έχουμε $f(0)=e$, άρα
\[
\ln(\ln f(0))=0+c
\Leftrightarrow
\ln(\ln e)=c
\Leftrightarrow
c=0
\]
Επομένως
\[
\ln(\ln f(x))=x
\]
οπότε
\[
\ln f(x)=e^x
\]
και τελικά
\[
f(x)=e^{e^x}\ ,\quad x\geq 0
\]

\item Από τον τύπο
\[
f(x)=e^{e^x}
\]
έχουμε
\[
f'(x)=e^{e^x}\cdot e^x>0
\]
για κάθε $x\geq 0$.
Άρα η $f$ είναι γνησίως αύξουσα στο $[0,+\infty)$, επομένως είναι $1-1$ και αντιστρέφεται.

Επίσης,
\[
f(0)=e
\quad \text{και}\quad
\lim\limits_{x\to+\infty}f(x)=+\infty
\]
οπότε
\[
f([0,+\infty))=[e,+\infty)
\]
Άρα η αντίστροφη συνάρτηση είναι
\[
g=f^{-1}:[e,+\infty)\to[0,+\infty)
\]
και, αν $y=f(x)$, τότε
\[
y=e^{e^x}
\Leftrightarrow \ln y=e^x
\Leftrightarrow x=\ln(\ln y)
\]
οπότε
\[
g(y)=\ln(\ln y)\ ,\quad y\geq e
\]

\item Έστω $x\geq 0$.
Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στη $f$ στο διάστημα $[x,x+1]$.
Υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
f(x+1)-f(x)=f'(\xi)
\]
Έχουμε
\[
f'(t)=e^{e^t}\cdot e^t=e^{t+e^t}
\]
Η συνάρτηση $f'$ είναι γνησίως αύξουσα, αφού η $t+e^t$ είναι γνησίως αύξουσα.
Επειδή $\xi>x$, έχουμε
\[
f'(\xi)>f'(x)
\]
Άρα
\[
f(x+1)-f(x)>f'(x)
\]
δηλαδή
\[
e^{e^{x+1}}-e^{e^x}>e^{x+e^x}
\]
για κάθε $x\geq 0$.

\item Από το δεύτερο ερώτημα έχουμε
\[
g(x)=\ln(\ln x)\ ,\quad x\geq e
\]
Άρα
\[
g'(x)=\frac{1}{x\ln x}
\]
Για $x\in[e,e^e]$ ισχύει $g'(x)>0$, οπότε το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\dint_e^{e^e}g'(x)\d x\\
&=\dint_e^{e^e}\frac{1}{x\ln x}\d x\\
&=\left[\ln(\ln x)\right]_e^{e^e}\\
&=\ln e-\ln 1\\
&=1
\end{align*}
\end{erwthma}
\end{thema}
