    \item Για να έχουν οι συναρτήσεις $f$ και $g$ κοινή εφαπτομένη σε κοινό σημείο για $x_0 = 1$, πρέπει να ισχύουν τα εξής:
    \begin{itemize}
        \item $f(1) = g(1)$ (κοινό σημείο)
        \item $f'(1) = g'(1)$ (κοινή κλίση εφαπτομένης)
    \end{itemize}
    Από την πρώτη σχέση έχουμε:
    \[ a \ln 1 = e^1 - \beta \implies 0 = e - \beta \implies \beta = e \]
    Υπολογίζουμε τις παραγώγους:
    \[ f'(x) = \frac{a}{x}, \quad g'(x) = e^x \]
    Από τη δεύτερη σχέση έχουμε:
    \[ f'(1) = g'(1) \implies \frac{a}{1} = e^1 \implies a = e \]
    Άρα, οι παράμετροι είναι $a = e$ και $\beta = e$.

    \item Η κοινή εφαπτομένη $\varepsilon$ διέρχεται από το σημείο $(1, f(1)) = (1, 0)$ και έχει κλίση $\lambda = f'(1) = e$. Η εξίσωσή της είναι:
    \[ y - 0 = e(x - 1) \implies y = ex - e \]

    \item Έστω η συνάρτηση $h_f(x) = f(x) - (ex - e) = e \ln x - ex + e$ για $x > 0$.
    \[ h_f'(x) = \frac{e}{x} - e = e \left( \frac{1-x}{x} \right) \]
    Εξετάζουμε το πρόσημο της $h_f'(x)$:
    \begin{itemize}
        \item $h_f'(x) = 0 \Leftrightarrow x = 1$
        \item $h_f'(x) > 0 \Leftrightarrow x \in (0, 1) \implies h_f$ \textbf{γνησίως αύξουσα}
        \item $h_f'(x) < 0 \Leftrightarrow x \in (1, +\infty) \implies h_f$ \textbf{γνησίως φθίνουσα}
    \end{itemize}
    Άρα η $h_f(x)$ παρουσιάζει το απόλυτο μέγιστο στο $x=1$ με $h_f(1) = e \ln 1 - e(1) + e = 0$. Συνεπώς, $h_f(x) \leq 0$ για κάθε $x > 0$ και $h_f(x) = 0$ μόνο για $x=1$, οπότε η $\varepsilon$ δεν τέμνει την $C_f$ σε άλλο σημείο.

    Έστω η συνάρτηση $h_g(x) = g(x) - (ex - e) = (e^x - e) - (ex - e) = e^x - ex$ για $x \in \mathbb{R}$.
    \[ h_g'(x) = e^x - e \]
    Εξετάζουμε το πρόσημο της $h_g'(x)$:
    \begin{itemize}
        \item $h_g'(x) = 0 \Leftrightarrow x = 1$
        \item $h_g'(x) < 0 \Leftrightarrow x \in (-\infty, 1) \implies h_g$ \textbf{γνησίως φθίνουσα}
        \item $h_g'(x) > 0 \Leftrightarrow x \in (1, +\infty) \implies h_g$ \textbf{γνησίως αύξουσα}
    \end{itemize}
    Άρα η $h_g(x)$ παρουσιάζει το απόλυτο ελάχιστο στο $x=1$ με $h_g(1) = e^1 - e(1) = 0$. Συνεπώς, $h_g(x) \geq 0$ για κάθε $x \in \mathbb{R}$ και $h_g(x) = 0$ μόνο για $x=1$, οπότε η $\varepsilon$ δεν τέμνει την $C_g$ σε άλλο σημείο.

    \item Το χωρίο βρίσκεται μεταξύ της $C_g: y = e^x - e$, της $\varepsilon: y = ex - e$ και των αξόνων. Για $x \in [0, 1]$, έχουμε $g(x) \geq \varepsilon(x)$ και και οι δύο συναρτήσεις είναι $\leq 0$. Το εμβαδόν $S$ είναι:
    \[ S = \int_{0}^{1} [g(x) - \varepsilon(x)] \d x = \int_{0}^{1} (e^x - e - (ex - e)) \d x \]
    \[ S = \int_{0}^{1} (e^x - ex) \d x \]
    \[ S = \left[ e^x - \frac{ex^2}{2} \right]_0^1 \]
    \[ S = \left( e^1 - \frac{e(1)^2}{2} \right) - \left( e^0 - \frac{e(0)^2}{2} \right) \]
    \[ S = e - \frac{e}{2} - 1 = \frac{e}{2} - 1 \]
\end{erwthma}
\end{thema}
