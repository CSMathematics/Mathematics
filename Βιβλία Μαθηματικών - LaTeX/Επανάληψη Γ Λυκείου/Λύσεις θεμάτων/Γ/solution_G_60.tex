%=========== ΛΥΣΗ - 60ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για να έχουν οι $C_f$ και $C_g$ κοινή εφαπτομένη σε κοινό σημείο με τετμημένη $x_0=1$, πρέπει
\[f(1)=g(1)\]
και
\[f'(1)=g'(1)\]
Έχουμε
\[f(1)=a\ln 1=0\]
και
\[g(1)=e-\beta\]
Άρα
\[0=e-\beta\Rightarrow \beta=e\]
Επίσης
\[f'(x)=\frac{a}{x}\]
και
\[g'(x)=e^x\]
Οπότε
\[f'(1)=g'(1)\Rightarrow a=e\]
Επομένως
\[a=e\quad\text{και}\quad \beta=e\]

\item Για $a=e$ και $\beta=e$ έχουμε
\[f(x)=e\ln x\]
και
\[g(x)=e^x-e\]
Το κοινό σημείο επαφής είναι
\[A(1,0)\]
και ο συντελεστής διεύθυνσης της κοινής εφαπτομένης είναι
\[\lambda=f'(1)=e\]
Άρα η κοινή εφαπτομένη είναι
\[y-0=e(x-1)\]
δηλαδή
\[\varepsilon:y=ex-e\]

\item Για $x>0$ έχουμε
\[f''(x)=-\frac{e}{x^2}<0\]
άρα η $f$ είναι \textbf{κοίλη} στο $(0,+\infty)$.
Επομένως η γραφική της παράσταση βρίσκεται κάτω από κάθε εφαπτομένη της, άρα
\[f(x)\leq ex-e\ ,\ x>0\]
δηλαδή
\[e\ln x\leq ex-e\ ,\ x>0\]

Επίσης
\[g''(x)=e^x>0\]
άρα η $g$ είναι \textbf{κυρτή} στο $\mathbb{R}$.
Επομένως η γραφική της παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της, άρα
\[ex-e\leq g(x)\ ,\ x>0\]
δηλαδή
\[ex-e\leq e^x-e\ ,\ x>0\]
Συνεπώς
\[f(x)\leq ex-e\leq g(x)\ ,\ x>0\]

\item Από το Γ3 έχουμε, για κάθε $x>0$,
\[e\ln x\leq ex-e\]
και
\[ex-e\leq e^x-e\]
Άρα
\[e\ln x\leq e^x-e\]
οπότε
\[e^x-e\ln x\geq e\ ,\ x>0\]

Θέλουμε να υπολογίσουμε το όριο
\[\lim\limits_{x\to 1}\frac{e^x-e\ln x-e}{(x-1)^2}\]
Για $x\to 1$ ο αριθμητής τείνει στο
\[e-e\ln1-e=0\]
και ο παρονομαστής στο $0$, άρα έχουμε μορφή $\dfrac{0}{0}$.
Με δύο εφαρμογές του κανόνα \eng{L'Hôpital}:
\begin{align*}
\lim\limits_{x\to 1}\frac{e^x-e\ln x-e}{(x-1)^2}
&=\lim\limits_{x\to 1}\frac{e^x-\frac{e}{x}}{2(x-1)}\\
&=\lim\limits_{x\to 1}\frac{e^x+\frac{e}{x^2}}{2}\\
&=\frac{e+e}{2}\\
&=e
\end{align*}
\end{erwthma}
\end{thema}
