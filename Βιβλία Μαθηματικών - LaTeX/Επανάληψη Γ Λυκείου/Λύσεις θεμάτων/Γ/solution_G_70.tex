%=========== ΛΥΣΗ - 70ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τα δεδομένα έχουμε
\[
\lim\limits_{x\to+\infty}\frac{f(x)}{x}=2
\]
και
\[
\lim\limits_{x\to+\infty}\left(f(x)-2x\right)=3
\]
Άρα, στο $+\infty$, η $C_f$ έχει πλάγια ασύμπτωτη την ευθεία
\[
y=2x+3
\]

\item Υπολογίζουμε:
\begin{align*}
\lim\limits_{x\to+\infty}\frac{f^2(x)-x^2}{x^2+f(x)}
&=\lim\limits_{x\to+\infty}
\frac{\frac{f^2(x)}{x^2}-1}{1+\frac{f(x)}{x^2}}
\end{align*}
Επειδή
\[
\lim\limits_{x\to+\infty}\frac{f(x)}{x}=2
\]
έχουμε
\[
\lim\limits_{x\to+\infty}\frac{f^2(x)}{x^2}
=\left(\lim\limits_{x\to+\infty}\frac{f(x)}{x}\right)^2
=4
\]
και
\[
\lim\limits_{x\to+\infty}\frac{f(x)}{x^2}
=\lim\limits_{x\to+\infty}\frac{f(x)}{x}\cdot\frac{1}{x}
=2\cdot 0=0
\]
Επομένως
\[
\lim\limits_{x\to+\infty}\frac{f^2(x)-x^2}{x^2+f(x)}
=\frac{4-1}{1+0}=3
\]

\item Η $f$ είναι συνεχής στο $\mathbb{R}$ και τέμνει τον άξονα $x'x$ ακριβώς μία φορά, στο $x_0=-1$.
Άρα
\[
f(x)=0\Leftrightarrow x=-1
\]

Στο διάστημα $(-\infty,-1)$ η $f$ δεν μηδενίζεται.
Επειδή είναι συνεχής, διατηρεί σταθερό πρόσημο στο διάστημα αυτό.
Ακόμη, από το δεδομένο
\[
\lim\limits_{x\to-\infty}f(x)=-\infty
\]
υπάρχει σημείο του $(-\infty,-1)$ στο οποίο η $f$ είναι αρνητική.
Άρα
\[
f(x)<0\ ,\quad x\in(-\infty,-1)
\]

Στο διάστημα $(-1,+\infty)$ η $f$ επίσης δεν μηδενίζεται, οπότε, λόγω συνέχειας, διατηρεί σταθερό πρόσημο.
Επιπλέον, από
\[
\lim\limits_{x\to+\infty}\frac{f(x)}{x}=2
\]
προκύπτει ότι
\[
\lim\limits_{x\to+\infty}f(x)=+\infty
\]
Άρα υπάρχει σημείο του $(-1,+\infty)$ στο οποίο η $f$ είναι θετική.
Επομένως
\[
f(x)>0\ ,\quad x\in(-1,+\infty)
\]

Συνολικά:
\[
f(x)<0 \text{ για } x<-1,\quad f(-1)=0,\quad f(x)>0 \text{ για } x>-1
\]

\item Για το πρώτο όριο, επειδή
\[
\lim\limits_{x\to-\infty}f(x)=-\infty
\]
θέτουμε
\[
t=-f(x)
\]
Τότε, καθώς $x\to-\infty$, έχουμε $t\to+\infty$ και
\[
\ln(|f(x)|)e^{f(x)}=\ln t\cdot e^{-t}=\frac{\ln t}{e^t}
\]
Άρα
\[
\lim\limits_{x\to-\infty}\ln(|f(x)|)e^{f(x)}
=\lim\limits_{t\to+\infty}\frac{\ln t}{e^t}
\]
Εφαρμόζοντας τον κανόνα \eng{De L' Hospital}, παίρνουμε
\[
\lim\limits_{t\to+\infty}\frac{\ln t}{e^t}
=\lim\limits_{t\to+\infty}\frac{\frac{1}{t}}{e^t}
=\lim\limits_{t\to+\infty}\frac{1}{te^t}=0
\]
Επομένως
\[
\lim\limits_{x\to-\infty}\ln(|f(x)|)e^{f(x)}=0
\]

Για το δεύτερο όριο, επειδή η $f$ είναι συνεχής στο $\mathbb{R}$ και η μοναδική της ρίζα είναι το $-1$, έχουμε
\[
f(0)\neq 0
\]
Μάλιστα, από το προηγούμενο ερώτημα, αφού $0>-1$, είναι
\[
f(0)>0
\]
Άρα η συνάρτηση $\dfrac{1}{f(x)}$ είναι συνεχής σε μια περιοχή του $0$ και
\[
\lim\limits_{x\to 0}\frac{1}{f(x)}=\frac{1}{f(0)}
\]
Η τιμή αυτή δεν μπορεί να προσδιοριστεί αριθμητικά από τα δεδομένα της εκφώνησης.
\end{erwthma}
\end{thema}
