%=========== ΛΥΣΗ - Ϛο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για να εξετάσουμε την ύπαρξη \textbf{πλάγιας ασύμπτωτης} της $f(x)$ στο $+\infty$, εξετάζουμε τα όρια:
\[ \lim_{x \to +\infty} \frac{f(x)}{x} = 2 \quad \text{και} \quad \lim_{x \to +\infty} (f(x) - 2x) = 3 \]
Από τον ορισμό της πλάγιας ασύμπτωτου, η ευθεία $y = \alpha x + \beta$ είναι ασύμπτωτος στο $+\infty$ εάν $\lim\limits_{x \to +\infty} \frac{f(x)}{x} = \alpha$ και $\lim\limits_{x \to +\infty} (f(x) - \alpha x) = \beta$.
Στη συγκεκριμένη περίπτωση, έχουμε $\alpha = 2$ και $\beta = 3$.
Επομένως, η $f(x)$ έχει \textbf{πλάγια ασύμπτωτο} στο $+\infty$ την ευθεία:
\[ y = 2x + 3 \]

\item Υπολογίζουμε το όριο $\lim\limits_{x \to +\infty} \dfrac{f^2(x) - x^2}{x^2 + f(x)}$ διαιρώντας αριθμητή και παρονομαστή με το $x^2$:
\[ \lim_{x \to +\infty} \frac{f^2(x) - x^2}{x^2 + f(x)} = \lim_{x \to +\infty} \frac{\frac{f^2(x)}{x^2} - 1}{1 + \frac{f(x)}{x^2}} \]
Γνωρίζουμε ότι $\lim\limits_{x \to +\infty} \frac{f(x)}{x} = 2$, άρα:
\[ \lim_{x \to +\infty} \frac{f^2(x)}{x^2} = \left( \lim_{x \to +\infty} \frac{f(x)}{x} \right)^2 = 2^2 = 4 \]
Επιπλέον, για το όριο $\lim\limits_{x \to +\infty} \frac{f(x)}{x^2}$, έχουμε:
\[ \lim_{x \to +\infty} \frac{f(x)}{x^2} = \lim_{x \to +\infty} \left( \frac{f(x)}{x} \cdot \frac{1}{x} \right) = 2 \cdot 0 = 0 \]
Συνεπώς:
\[ \lim_{x \to +\infty} \frac{f^2(x) - x^2}{x^2 + f(x)} = \frac{4 - 1}{1 + 0} = 3 \]

\item Η συνάρτηση $f$ είναι \textbf{συνεχής} στο $\mathbb{R}$ και τέμνει τον άξονα $x'x$ ακριβώς μία φορά στο $x_0 = -1$, άρα $f(x) = 0 \Leftrightarrow x = -1$.
Στα διαστήματα $(-\infty, -1)$ και $(-1, +\infty)$, η $f(x)$ διατηρεί \textbf{σταθερό πρόσημο}.
\begin{itemize}
    \item Για $x \in (-\infty, -1)$:
    Δεδομένου ότι $\lim\limits_{x \to -\infty} f(x) = -\infty$, η συνάρτηση είναι \textbf{αρνητική} στο διάστημα αυτό.
    Άρα $f(x) < 0$ για $x \in (-\infty, -1)$.
    \item Για $x \in (-1, +\infty)$:
    Από το ερώτημα (α), η $f$ έχει ασύμπτωτο $y = 2x + 3$ στο $+\infty$, οπότε:
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} (2x + 3) = +\infty \]
    Επομένως, η συνάρτηση είναι \textbf{θετική} στο διάστημα αυτό.
    Άρα $f(x) > 0$ για $x \in (-1, +\infty)$.
\end{itemize}

\item Για το πρώτο όριο $\lim\limits_{x \to -\infty} \ln(|f(x)|) \cdot e^{f(x)}$, θέτουμε $u = f(x)$.
Καθώς $x \to -\infty$, έχουμε $u \to -\infty$. Το όριο γίνεται:
\[ \lim_{u \to -\infty} \ln(|u|) \cdot e^u = \lim_{u \to -\infty} \frac{\ln(-u)}{e^{-u}} \]
Θέτουμε $t = -u$, οπότε καθώς $u \to -\infty \implies t \to +\infty$:
\[ \lim_{t \to +\infty} \frac{\ln t}{e^t} \]
Το όριο είναι της μορφής $\frac{\infty}{\infty}$, οπότε εφαρμόζουμε τον κανόνα \eng{L'Hôpital}:
\[ \lim_{t \to +\infty} \frac{\ln t}{e^t} \xlongequal{\eng{L'Hôpital}} \lim_{t \to +\infty} \frac{\frac{1}{t}}{e^t} = \lim_{t \to +\infty} \frac{1}{t e^t} = 0 \]
Άρα $\lim\limits_{x \to -\infty} \ln(|f(x)|) \cdot e^{f(x)} = 0$.

Για το δεύτερο όριο $\lim\limits_{x \to 0} \dfrac{1}{f(x)}$, επειδή η $f$ είναι \textbf{συνεχής} και η μόνη της ρίζα είναι το $x_0 = -1$, το $f(0)$ είναι μια σταθερά διάφορη του μηδενός.
Από το ερώτημα (γ), εφόσον $0 \in (-1, +\infty)$, ισχύει $f(0) > 0$.
Συνεπώς:
\[ \lim_{x \to 0} \frac{1}{f(x)} = \frac{1}{f(0)} \]
\end{erwthma}
\end{thema}
