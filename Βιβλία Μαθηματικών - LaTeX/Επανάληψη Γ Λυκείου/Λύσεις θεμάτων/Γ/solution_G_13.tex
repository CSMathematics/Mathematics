%=========== ΛΥΣΗ - 13ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το κουτί είναι ορθογώνιο παραλληλεπίπεδο χωρίς καπάκι. Αποτελείται από την τετράγωνη βάση πλευράς $x$ (εμβαδόν $E_{\beta} = x^2$) και 4 πλευρικές ορθογώνιες έδρες με διαστάσεις $x$ και $y$ (εμβαδόν 4 εδρών = $4xy$).
Το συνολικό εμβαδόν επιφάνειας είναι σταθερό και ίσο με $12$:
\[ x^2 + 4xy = 12 \]
Επειδή οι διαστάσεις πρέπει να είναι θετικές, έχουμε $x > 0$ και $y > 0$. Λύνουμε τη σχέση ως προς το ύψος $y$:
\[ 4xy = 12 - x^2 \overset{x>0}{\implies} y = \frac{12 - x^2}{4x} \]
Η συνθήκη $y > 0$ απαιτεί:
\[ \frac{12 - x^2}{4x} > 0 \overset{x>0}{\implies} 12 - x^2 > 0 \implies x^2 < 12 \implies x < \sqrt{12} \]
Συνεπώς, το πεδίο ορισμού για την πλευρά της βάσης είναι $x \in (0, \sqrt{12})$.
Ο όγκος $V$ του κουτιού υπολογίζεται από τον τύπο $V = (\text{Εμβαδόν βάσης}) \cdot (\text{ύψος}) = x^2 \cdot y$. Αντικαθιστώντας το $y$:
\[ V(x) = x^2 \cdot \frac{12 - x^2}{4x} = x \cdot \frac{12 - x^2}{4} = \frac{12x - x^3}{4} = 3x - \frac{x^3}{4} \]
για $x \in (0, \sqrt{12})$.

\item Η συνάρτηση του όγκου $V(x) = 3x - \frac{x^3}{4}$ είναι παραγωγίσιμη στο διάστημα $(0, \sqrt{12})$ ως πολυωνυμική, με παράγωγο:
\[ V'(x) = 3 - \frac{3x^2}{4} \]
Βρίσκουμε τις ρίζες της παραγώγου:
\[ V'(x) = 0 \Leftrightarrow \frac{3x^2}{4} = 3 \Leftrightarrow 3x^2 = 12 \Leftrightarrow x^2 = 4 \overset{x>0}{\implies} x = 2 \]
Το πρόσημο της $V'(x)$ καθορίζεται από την ανισότητα $\frac{3x^2}{4} \lessgtr 3 \Leftrightarrow x^2 \lessgtr 4$:
\begin{itemize}
    \item $V'(x) > 0$ για $0 < x < 2$
    \item $V'(x) < 0$ για $2 < x < \sqrt{12}$
\end{itemize}

Η συνάρτηση είναι γνησίως αύξουσα στο $(0, 2]$ και γνησίως φθίνουσα στο $[2, \sqrt{12})$. Άρα ο όγκος γίνεται μέγιστος (ολικό μέγιστο) όταν ${x = 2 \text{ m}}$. 
Το αντίστοιχο ύψος του κουτιού υπολογίζεται από τον τύπο του $y$:
\[ y = \frac{12 - 2^2}{4 \cdot 2} = \frac{12 - 4}{8} = \frac{8}{8} = {1 \text{ m}} \]
Επομένως, οι διαστάσεις που μεγιστοποιούν τον όγκο είναι $x = 2\ \si{m}$ και $y = 1\ \si{m}$.

\item Στη βέλτιστη κατασκευή ($x=2$ και $y=1$) παρατηρούμε άμεσα ότι:
\[ x = 2y \]
Δηλαδή, για να μεγιστοποιηθεί ο όγκος, \textbf{η πλευρά της βάσης πρέπει να είναι διπλάσια από το ύψος} του κουτιού.

\item Η τιμή που μεγιστοποιεί τον όγκο είναι η $x_0 = 2$.
Το πρώτο ζητούμενο όριο ταυτίζεται με τον ορισμό της παραγώγου της $V$ στο $x_0 = 2$:
\[ \lim_{x \to x_0} \frac{V(x) - V(x_0)}{x-x_0} = \lim_{x \to 2} \frac{V(x) - V(2)}{x-2} = V'(2) \]
Επειδή η θέση $x=2$ είναι ρίζα της $V'$, ισχύει ${V'(2) = 0}$.

Για το δεύτερο όριο, υπολογίζουμε το όριο καθώς το $x$ πλησιάζει το άνω άκρο του πεδίου ορισμού:
\begin{align*}
\lim_{x \to \sqrt{12}^-} \frac{V(x)}{\sqrt{12}-x} &= \lim_{x \to \sqrt{12}^-} \frac{\frac{x(12-x^2)}{4}}{\sqrt{12}-x} \\
&= \lim_{x \to \sqrt{12}^-} \frac{x(\sqrt{12}-x)(\sqrt{12}+x)}{4(\sqrt{12}-x)}
\end{align*}
Απλοποιώντας τον κοινό παράγοντα $(\sqrt{12}-x)$ (αφού $x \neq \sqrt{12}$ στο όριο), προκύπτει:
\begin{align*}
\lim_{x \to \sqrt{12}^-} \frac{x(\sqrt{12}+x)}{4} &= \frac{\sqrt{12}(\sqrt{12}+\sqrt{12})}{4} \\
&= \frac{\sqrt{12} \cdot 2\sqrt{12}}{4} = \frac{2 \cdot 12}{4} = \frac{24}{4} = {6}
\end{align*}
\end{erwthma}
\end{thema}
