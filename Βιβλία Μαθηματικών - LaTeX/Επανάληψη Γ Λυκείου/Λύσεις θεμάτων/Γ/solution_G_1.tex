%=========== ΛΥΣΗ - 1ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για κάθε $x>0$ ισχύει
\[x^2f'(x)+xf(x)=1 \Leftrightarrow x f'(x)+f(x)=\frac{1}{x} \Leftrightarrow (xf(x))'=(\ln x)'\]
Άρα, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c \in \mathbb{R}$ τέτοια ώστε για κάθε $x>0$ να ισχύει
\[xf(x)=\ln x + c\]
Για $x=1$ προκύπτει $1\cdot f(1) = \ln 1 + c \Leftrightarrow c = 0$.

Επομένως, για κάθε $x>0$ είναι
\[xf(x)=\ln x  \Leftrightarrow f(x)=\frac{\ln x}{x}\]

\item Θεωρούμε τη συνάρτηση $g(x)=1-\ln x-2x^2\ ,\ x\in(0,1]$.
Η $g$ είναι συνεχής στο $(0,1]$ ως πράξεις συνεχών συναρτήσεων. Έχουμε:
\begin{itemize}
\item $\lim\limits_{x\to 0^+}{g(x)}=\lim\limits_{x\to 0^+}{(1-\ln x-2x^2)}=+\infty$
\item $g(1)=1-\ln 1-2(1)^2=-1$
\end{itemize}
Επειδή $\lim\limits_{x\to 0^+}{g(x)}=+\infty$, υπάρχει αριθμός $\alpha \in (0,1)$ τέτοιος ώστε $g(\alpha)>0$. 
Η $g$ είναι συνεχής στο κλειστό διάστημα $[\alpha, 1]$ (ως περιορισμός συνεχούς συνάρτησης) και ισχύει
\[g(\alpha)g(1) < 0\]
αφού $g(\alpha)>0$ και $g(1)=-1<0$.
Επομένως, ικανοποιούνται οι προϋποθέσεις του θεωρήματος \eng{Bolzano} στο $[\alpha, 1]$, οπότε υπάρχει τουλάχιστον ένα $x_0\in(\alpha,1) \subseteq (0,1)$ τέτοιο ώστε
\[g(x_0)=0 \Leftrightarrow 1-\ln x_0=2x_0^2\]

\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $(0, +\infty)$ με
\[f'(x) = \left(\frac{\ln x}{x}\right)' = \frac{(\ln x)'x - \ln x(x)'}{x^2} = \frac{1 - \ln x}{x^2}\]
Ο συντελεστής διεύθυνσης της εφαπτομένης της $C_f$ στο σημείο με τετμημένη $x_0$ είναι:
\[\lambda = f'(x_0) = \frac{1 - \ln x_0}{x_0^2}\]
Από το ερώτημα Γ2, γνωρίζουμε ότι ο αριθμός $x_0$ ικανοποιεί τη σχέση $1 - \ln x_0 = 2x_0^2$. Αντικαθιστώντας, έχουμε:
\[\lambda = \frac{2x_0^2}{x_0^2} = 2\]
Ο συντελεστής διεύθυνσης της ευθείας $\varepsilon: y = 2x + 2026$ είναι $\lambda_{\varepsilon} = 2$.
Επειδή $\lambda = \lambda_{\varepsilon}$, η εφαπτομένη της $C_f$ στο συγκεκριμένο σημείο είναι παράλληλη στην ευθεία $\varepsilon$.

\item Έστω $x(t)$ η τετμημένη και $y(t)$ η τεταγμένη του κινητού τη χρονική στιγμή $t$. Γνωρίζουμε ότι
\[y(t)=f(x(t))\]
Παραγωγίζοντας τη σχέση $y(t)=f(x(t))$ ως προς τον χρόνο $t$, με τον κανόνα της αλυσίδας, έχουμε:
\[y'(t)=f'(x(t))\cdot x'(t) \Rightarrow y'(t)=\frac{1-\ln x(t)}{x^2(t)}\cdot x'(t)\]
Έστω $t_0$ η χρονική στιγμή που το κινητό διέρχεται από το σημείο με τετμημένη $x(t_0)=e$. Γνωρίζουμε επίσης ότι ο ρυθμός μεταβολής της τετμημένης είναι $x'(t_0)=2\,\text{m/s}$. Τη χρονική στιγμή $t_0$ ισχύει:
\[y'(t_0)=\frac{1-\ln x(t_0)}{x^2(t_0)}\cdot x'(t_0)=\frac{1-\ln e}{e^2}\cdot 2=\frac{1-1}{e^2}\cdot 2=0\,\text{m/s}\]
\end{erwthma}
\end{thema}
