%=========== ΛΥΣΗ - 34ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το πεδίο ορισμού της συνάρτησης είναι το $A_f = [0, +\infty)$.
Υπολογίζουμε το πλευρικό όριο στο 0 από τα δεξιά:
\[ L = \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} (x \ln x) \]
Καθώς $x \to 0^+$, το $x \to 0$ και $\ln x \to -\infty$, οπότε έχουμε απροσδιοριστία μορφής $0 \cdot (-\infty)$. 
Τη μετατρέπουμε σε $\frac{-\infty}{+\infty}$ γράφοντας το $x$ ως $\frac{1}{1/x}$:
\[ L = \lim_{x \to 0^+} \frac{\ln x}{\frac{1}{x}} \]
Εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ L \xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}} \lim_{x \to 0^+} \frac{(\ln x)'}{\left(\frac{1}{x}\right)'} = \lim_{x \to 0^+} \frac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x \to 0^+} (-x) = \boldsymbol{0} \]
Από τον κλάδο της συνάρτησης για $x=0$, έχουμε $f(0) = 0$.
Επειδή ισχύει $\lim\limits_{x \to 0^+} f(x) = f(0) = 0$, η συνάρτηση $f$ είναι \textbf{συνεχής στο 0} (από τα δεξιά).

\item Ελέγχουμε την παραγωγισιμότητα στο $x_0=0$, υπολογίζοντας το όριο του πηλίκου μεταβολής:
\[ \lim_{x \to 0^+} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0^+} \frac{x \ln x - 0}{x} = \lim_{x \to 0^+} \ln x = -\infty \]
Επειδή το όριο είναι $-\infty$ (μη πραγματικός αριθμός), η συνάρτηση $\boldsymbol{f}$ \textbf{δεν είναι παραγωγίσιμη} στο 0.

\item Για $x > 0$, η $f(x) = x \ln x$ είναι παραγωγίσιμη ως γινόμενο με παράγωγο:
\[ f'(x) = (x)' \ln x + x (\ln x)' = \ln x + x \cdot \frac{1}{x} = \ln x + 1 \]
Ζητείται το όριο:
\begin{align*}
\lim_{x \to 0^+} [ f(x) \cdot f'(x) ] &= \lim_{x \to 0^+} [ (x \ln x) (\ln x + 1) ] \\
&= \lim_{x \to 0^+} ( x (\ln x)^2 + x \ln x )
\end{align*}
Γνωρίζουμε από το (α) ότι $\lim\limits_{x \to 0^+} (x \ln x) = 0$. 
Υπολογίζουμε ξεχωριστά το όριο του πρώτου όρου, το οποίο έχει μορφή $0 \cdot (+\infty)$:
\[ \lim_{x \to 0^+} x(\ln x)^2 = \lim_{x \to 0^+} \frac{(\ln x)^2}{\frac{1}{x}} \]
Εφαρμόζουμε κανόνα \eng{De L'Hospital} ($\frac{+\infty}{+\infty}$):
\[ \lim_{x \to 0^+} \frac{(\ln x)^2}{\frac{1}{x}} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to 0^+} \frac{2 \ln x \cdot \frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x \to 0^+} (-2x \ln x) = -2 \cdot 0 = 0 \]
Επομένως, χρησιμοποιώντας τις ιδιότητες των ορίων, έχουμε:
\[ \lim_{x \to 0^+} [ f(x) \cdot f'(x) ] = 0 + 0 = \boldsymbol{0} \]

\item Το ζητούμενο ολοκλήρωμα θα υπολογιστεί με τη μέθοδο της παραγοντικής ολοκλήρωσης (ή ολοκλήρωσης κατά παράγοντες). Για $x \in [1/e, 1]$, η $f(x) = x \ln x$ είναι συνεχής:
\begin{align*}
I &= \int_{\frac{1}{e}}^{1} x \ln x \d x = \int_{\frac{1}{e}}^{1} \left( \frac{x^2}{2} \right)' \ln x \d x \\
&= \left[ \frac{x^2}{2} \ln x \right]_{\frac{1}{e}}^{1} - \int_{\frac{1}{e}}^{1} \frac{x^2}{2} (\ln x)' \d x \\
&= \left( \frac{1^2}{2} \ln 1 - \frac{\left(\frac{1}{e}\right)^2}{2} \ln\left(\frac{1}{e}\right) \right) - \int_{\frac{1}{e}}^{1} \frac{x^2}{2} \cdot \frac{1}{x} \d x \\
&= \left( 0 - \frac{1}{2e^2} (-1) \right) - \frac{1}{2} \int_{\frac{1}{e}}^{1} x \d x \\
&= \frac{1}{2e^2} - \frac{1}{2} \left[ \frac{x^2}{2} \right]_{\frac{1}{e}}^{1} \\
&= \frac{1}{2e^2} - \frac{1}{2} \left( \frac{1}{2} - \frac{\frac{1}{e^2}}{2} \right) = \frac{1}{2e^2} - \frac{1}{4} + \frac{1}{4e^2}
\end{align*}
Κάνοντας τα κλάσματα ομώνυμα:
\[ I = \frac{2}{4e^2} + \frac{1}{4e^2} - \frac{1}{4} = \frac{3}{4e^2} - \frac{e^2}{4e^2} = \boldsymbol{\frac{3-e^2}{4e^2}} \]
\end{erwthma}
\end{thema}
