%=========== ΛΥΣΗ - 56ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $M(x_0,\ln x_0)$ το σημείο επαφής της ζητούμενης εφαπτομένης, με $x_0>0$.
Είναι
\[f'(x)=\frac{1}{x}\]
οπότε η εφαπτομένη της $C_f$ στο $M$ έχει εξίσωση
\[y-\ln x_0=\frac{1}{x_0}(x-x_0)\]
Αφού διέρχεται από την αρχή των αξόνων $O(0,0)$, έχουμε
\[0-\ln x_0=\frac{1}{x_0}(0-x_0)\Rightarrow -\ln x_0=-1\Rightarrow \ln x_0=1\]
Άρα
\[x_0=e\]
Επομένως υπάρχει ακριβώς μία τέτοια εφαπτομένη και η εξίσωσή της είναι
\[y-1=\frac{1}{e}(x-e)\Rightarrow y=\frac{x}{e}\]

\item Από το προηγούμενο ερώτημα, η ευθεία
\[y=\frac{x}{e}\]
είναι εφαπτομένη της $C_f$ στο σημείο με τετμημένη $e$.
Επειδή η συνάρτηση $f(x)=\ln x$ είναι κοίλη στο $(0,+\infty)$, αφού
\[f''(x)=-\frac{1}{x^2}<0\ ,\ x>0\]
η γραφική της παράσταση βρίσκεται κάτω από κάθε εφαπτομένη της.
Άρα, για κάθε $x>0$,
\[\ln x\leq \frac{x}{e}\]

Λύνουμε τώρα την εξίσωση
\[e^x=x^e\ ,\ x>0\]
Επειδή και τα δύο μέλη είναι θετικά, λογαριθμίζουμε:
\begin{align*}
e^x=x^e
&\Leftrightarrow \ln(e^x)=\ln(x^e)\\
&\Leftrightarrow x=e\ln x\\
&\Leftrightarrow \ln x=\frac{x}{e}
\end{align*}
Από την ανισότητα $\ln x\leq\dfrac{x}{e}$ η ισότητα ισχύει μόνο στο σημείο επαφής, δηλαδή για
\[x=e\]
Άρα η μοναδική λύση της εξίσωσης στο $(0,+\infty)$ είναι
\[x=e\]

\item Η απόσταση ενός σημείου $M(x,\ln x)$ της $C_f$ από την ευθεία
\[\zeta:y=x\Leftrightarrow x-y=0\]
είναι
\[d(x)=\frac{|x-\ln x|}{\sqrt{2}}\ ,\ x>0\]
Για κάθε $x>0$ ισχύει
\[x-\ln x>0\]
αφού η συνάρτηση $\varphi(x)=x-\ln x$ έχει ελάχιστο στο $x=1$ με τιμή $\varphi(1)=1$.
Άρα
\[d(x)=\frac{x-\ln x}{\sqrt{2}}\]
Για να ελαχιστοποιήσουμε την απόσταση, αρκεί να ελαχιστοποιήσουμε τη συνάρτηση
\[\varphi(x)=x-\ln x\ ,\ x>0\]
Έχουμε
\[\varphi'(x)=1-\frac{1}{x}=\frac{x-1}{x}\]
Άρα η $\varphi$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.
Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$.
Το ζητούμενο σημείο είναι
\[A(1,\ln 1)\equiv A(1,0)\]
και η ελάχιστη απόσταση είναι
\[d_{\min}=\frac{1-0}{\sqrt{2}}=\frac{\sqrt{2}}{2}\]

\item Από το Γ1 η εφαπτομένη είναι
\[y=\frac{x}{e}\]
Το χωρίο που περικλείεται από την $C_f$, την ευθεία $y=\dfrac{x}{e}$ και τον άξονα $x'x$ έχει κορυφές στα σημεία:
\begin{itemize}
\item $(1,0)$, όπου η $C_f$ τέμνει τον άξονα $x'x$
\item $(0,0)$, όπου η εφαπτομένη τέμνει τον άξονα $x'x$
\item $(e,1)$, όπου η εφαπτομένη εφάπτεται στην $C_f$
\end{itemize}
Στο διάστημα $[0,1]$ το πάνω όριο του χωρίου είναι η ευθεία $y=\dfrac{x}{e}$, ενώ στο διάστημα $[1,e]$ το πάνω όριο είναι η ευθεία $y=\dfrac{x}{e}$ και το κάτω όριο είναι η $y=\ln x$.
Άρα το εμβαδόν είναι
\begin{align*}
E&=\dint_0^1 \frac{x}{e}\d x+\dint_1^e\left(\frac{x}{e}-\ln x\right)\d x\\
&=\left[\frac{x^2}{2e}\right]_0^1+\left[\frac{x^2}{2e}-x\ln x+x\right]_1^e\\
&=\frac{1}{2e}+\left(\frac{e}{2}-e+e\right)-\left(\frac{1}{2e}-0+1\right)\\
&=\frac{e}{2}-1
\end{align*}

\item Για κάθε $x>0$ έχουμε
\[x f(x)\hm\left(\frac{1}{x}\right)=x\ln x\cdot \hm\left(\frac{1}{x}\right)\]
Επειδή
\[-1\leq \hm\left(\frac{1}{x}\right)\leq 1\]
παίρνουμε
\[-|x\ln x|\leq x\ln x\cdot \hm\left(\frac{1}{x}\right)\leq |x\ln x|\]
Όμως
\[\lim\limits_{x\to 0^+}x\ln x=0\]
άρα, από το κριτήριο παρεμβολής,
\[\lim\limits_{x\to 0^+}\left[x f(x)\hm\left(\frac{1}{x}\right)\right]=0\]
\end{erwthma}
\end{thema}
