\begin{thema}{Γ}
\begin{erwthma}
\item Θέτουμε στην ανισότητα $x=0$ και παίρνουμε:
\[e^0 - 1 \le f(0) \le 0 \cdot e^0 + 0 \Leftrightarrow 0 \le f(0) \le 0\]
Επομένως, αναγκαστικά $f(0) = 0$.

Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, άρα παραγωγίσιμη και στο $x_0=0$. Η παράγωγός της στο $0$ ισούται με την πλευρική παράγωγο στο $0^+$, δηλαδή:
\[f'(0) = \lim_{x \to 0^+} \frac{f(x)-f(0)}{x-0} = \lim_{x \to 0^+} \frac{f(x)}{x}\]
Για κάθε $x > 0$, διαιρούμε την αρχική ανισότητα με $x>0$ και η φορά διατηρείται:
\[\frac{e^x - 1}{x} \le \frac{f(x)}{x} \le \frac{x e^x + x}{x} \Leftrightarrow \frac{e^x - 1}{x} \le \frac{f(x)}{x} \le e^x + 1\]
Υπολογίζουμε τα όρια των ακραίων συναρτήσεων στο $0^+$:
\begin{itemize}
  \item $\lim\limits_{x \to 0^+} (e^x + 1) = e^0 + 1 = 2$
  \item $\lim\limits_{x \to 0^+} \frac{e^x - 1}{x} \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim\limits_{x \to 0^+} \frac{e^x}{1} = 1$
\end{itemize}
Εφόσον τα δύο όρια υπάρχουν και η $\dfrac{f(x)}{x}$ βρίσκεται ανάμεσά τους, περνώντας στο όριο καθώς $x \to 0^+$ παίρνουμε:
\[1 \le \lim_{x \to 0^+} \frac{f(x)}{x} \le 2 \Leftrightarrow 1 \le f'(0) \le 2\]

\item Για $f(x) = e^x - 1$, αναζητούμε οριζόντια ασύμπτωτη στο $-\infty$ και στο $+\infty$:
\begin{itemize}
  \item Στο $-\infty$: $\lim\limits_{x \to -\infty} f(x) = \lim\limits_{x \to -\infty} (e^x - 1) = 0 - 1 = -1$. 
  Άρα η ευθεία $y = -1$ είναι \textbf{οριζόντια ασύμπτωτη} της $C_f$ στο $-\infty$.
  \item Στο $+\infty$: $\lim\limits_{x \to +\infty} f(x) = \lim\limits_{x \to +\infty} (e^x - 1) = +\infty$. 
  Άρα η $C_f$ δεν έχει οριζόντια ασύμπτωτη στο $+\infty$.
\end{itemize}

\item Θεωρούμε τη συνάρτηση διαφοράς $K(x) = (xe^x + x) - (e^x - 1) = xe^x - e^x + x + 1$ για $x \ge 0$. Θέλουμε να δείξουμε ότι $K(x) \ge 0$.
Η $K$ είναι παραγωγίσιμη με:
\[K'(x) = (e^x + xe^x) - e^x + 1 = xe^x + 1\]
Για κάθε $x \ge 0$ ισχύει $xe^x \ge 0$ (και $1>0$), οπότε $K'(x) > 0$. Άρα η $K$ είναι \textbf{γνησίως αύξουσα} στο $[0, +\infty)$.
Επομένως, για κάθε $x \ge 0$ ισχύει:
\[K(x) \ge K(0) \Leftrightarrow K(x) \ge 0\cdot e^0 - e^0 + 0 + 1 \Leftrightarrow K(x) \ge -1 + 1 \Leftrightarrow K(x) \ge 0\]
Δηλαδή $xe^x + x \ge e^x - 1$ για κάθε $x \ge 0$.

\item Από το ερώτημα Γ3 δείξαμε ότι $g(x) \ge h(x) \Leftrightarrow g(x) - h(x) \ge 0$ για κάθε $x \ge 0$. Άρα και στο διάστημα $[0,1]$ η γραφική παράσταση της $g$ βρίσκεται πάνω από αυτή της $h$.
Το ζητούμενο εμβαδόν $E$ δίνεται από το ολοκλήρωμα:

\begin{align*}
E &= \int_0^1 |g(x) - h(x)| \d x = \int_0^1 (g(x) - h(x)) \d x = \int_0^1 (xe^x - e^x + x + 1) \d x \\
&= \int_0^1 xe^x \d x - \int_0^1 e^x \d x + \int_0^1 x \d x + \int_0^1 1 \d x \\
&= \left[(xe^x - e^x) - e^x + \frac{x^2}{2} + x\right]_0^1 \\
&= \left[ xe^x - 2e^x + \frac{x^2}{2} + x \right]_0^1 = \left( 1\cdot e^1 - 2e^1 + \frac{1^2}{2} + 1 \right) - \left( 0\cdot e^0 - 2e^0 + 0 + 0 \right) \\
&= \left( -e + \frac{3}{2} \right) - (-2)= \frac{7}{2} - e \ \text{τ.μ.}
\end{align*}
\end{erwthma}
\end{thema}
