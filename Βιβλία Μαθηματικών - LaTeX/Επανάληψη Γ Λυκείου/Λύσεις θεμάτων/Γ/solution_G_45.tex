%=========== ΛΥΣΗ - 45ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για κάθε $x\in\mathbb{R}$ ισχύει
\[f^3(x)+3f^2(x)\hm x+3f(x)\hm^2 x=x^3-\hm^3 x\]
Προσθέτουμε $\hm^3 x$ και στα δύο μέλη:
\[f^3(x)+3f^2(x)\hm x+3f(x)\hm^2 x+\hm^3 x=x^3\]
Το πρώτο μέλος είναι ανάπτυγμα κύβου, οπότε
\[\left(f(x)+\hm x\right)^3=x^3\]
Επειδή η συνάρτηση $g(t)=t^3$ είναι γνησίως αύξουσα στο $\mathbb{R}$, είναι και ένα προς ένα, οπότε προκύπτει ότι
\[f(x)+\hm x=x\]
Επομένως, για κάθε $x\in\mathbb{R}$ είναι
\[f(x)=x-\hm x\]

\item Από το προηγούμενο ερώτημα έχουμε
\[f(x)=x-\hm x\ ,\ x\in\mathbb{R}\]
Θα αποδείξουμε ότι $f(x)>0$ για κάθε $x>0$.

Αν $0<x\leq 1$, τότε η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με
\[f'(x)=1-\syn x\]
Για κάθε $x\in(0,1]$ ισχύει $\syn x<1$, άρα
\[f'(x)>0\]
οπότε η $f$ είναι γνησίως αύξουσα στο $[0,1]$. Επειδή
\[f(0)=0-\hm 0=0\]
έχουμε
\[f(x)>f(0)=0\ ,\ 0<x\leq 1\]

Αν $x>1$, τότε
\[f(x)=x-\hm x\geq x-1>0\]
αφού $\hm x\leq 1$ για κάθε $x\in\mathbb{R}$.

Άρα για κάθε $x>0$ ισχύει $f(x)>0$.

\item Η $f$ είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}$ με
\[f'(x)=1-\syn x\]
και
\[f''(x)=\hm x\]
Για κάθε $x\in(0,\pi)$ ισχύει
\[\hm x>0\]
άρα
\[f''(x)>0\ ,\ x\in(0,\pi)\]
Επομένως η $f$ είναι \textbf{κυρτή} στο διάστημα $[0,\pi]$.

\item Από το ερώτημα Γ2 γνωρίζουμε ότι
\[f(x)>0\ ,\ x>0\]
και επιπλέον $f(0)=0$. Άρα στο διάστημα $\left[0,\frac{\pi}{2}\right]$ η γραφική παράσταση της $f$ βρίσκεται πάνω από τον άξονα $x'x$.

Το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\dint_0^{\frac{\pi}{2}} f(x)\d x\\
&=\dint_0^{\frac{\pi}{2}}(x-\hm x)\d x\\
&=\left[\frac{x^2}{2}+\syn x\right]_0^{\frac{\pi}{2}}\\
&=\left(\frac{\pi^2}{8}+\syn\frac{\pi}{2}\right)-\left(0+\syn 0\right)\\
&=\frac{\pi^2}{8}-1
\end{align*}
Επομένως το ζητούμενο εμβαδόν είναι
\[E=\frac{\pi^2}{8}-1\]
\end{erwthma}
\end{thema}
