%=========== ΛΥΣΗ - Xο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Παρατηρούμε ότι η αριστερή πλευρά της εξίσωσης είναι σχεδόν η ανάπτυξη του κύβου ενός αθροίσματος. Γνωρίζουμε ότι:
\[(f(x) + \hm x)^3 = f^3(x) + 3f^2(x)\hm x + 3f(x)\hm^2 x + \hm^3 x.\]
Η δεδομένη σχέση είναι:
\[f^3(x) + 3f^2(x)\hm x + 3f(x)\hm^2 x = x^3 - \hm^3 x.\]
Προσθέτοντας το $\hm^3 x$ και στα δύο μέλη, έχουμε:
\[f^3(x) + 3f^2(x)\hm x + 3f(x)\hm^2 x + \hm^3 x = x^3.\]
Άρα:
\[(f(x) + \hm x)^3 = x^3.\]
Επειδή η συνάρτηση $g(t) = t^3$ είναι μια \textbf{αντιστρέψιμη} συνάρτηση στο $\mathbb{R}$, προκύπτει ότι:
\[f(x) + \hm x = x \implies f(x) = x - \hm x.\]

\item Για τη συνάρτηση $f(x) = x - \hm x$ με $x \in [0, +\infty)$, υπολογίζουμε την παράγωγό της:
\[f'(x) = 1 - \syn x.\]
Επειδή $-1 \le \syn x \le 1$ για κάθε $x \in \mathbb{R}$, προκύπτει ότι:
\[f'(x) = 1 - \syn x \ge 0.\]
Η ισότητα $f'(x) = 0$ ισχύει μόνο για τις τιμές $x = 2k\pi$ ($k \in \mathbb{Z}$), οι οποίες είναι μεμονωμένα σημεία. Επομένως, η $f$ είναι \textbf{γνησίως αύξουσα} στο διάστημα $[0, +\infty)$.
Επειδή $f(0) = 0 - \hm 0 = 0$, τότε για κάθε $x > 0$ ισχύει:
\[f(x) > f(0) \implies f(x) > 0.\]

\item Για τη μελέτη της κυρτότητας στο διάστημα $[0, \pi]$, υπολογίζουμε τη δεύτερη παράγωγο της $f$:
\[f''(x) = (1 - \syn x)' = -\syn' x = -(-\hm x) = \hm x.\]
Για κάθε $x \in (0, \pi)$, γνωρίζουμε ότι $\hm x > 0$.
Συνεπώς, $f''(x) > 0$ για κάθε $x \in (0, \pi)$, άρα η συνάρτηση $f$ είναι \textbf{κυρτή} στο διάστημα $[0, \pi]$.

\item Το εμβαδόν $S$ του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα των τετμημένων και την ευθεία $x = \pi/2$ δίνεται από το ορισμένο ολοκλήρωμα:
\[S = \int_{0}^{\pi/2} |f(x)| \d x.\]
Από το ερώτημα (β) γνωρίζουμε ότι $f(x) > 0$ για $x \in (0, \pi/2]$. Άρα:
\begin{align*}
S &= \int_{0}^{\pi/2} (x - \hm x) \d x \\
  &= \left[ \frac{x^2}{2} + \syn x \right]_{0}^{\pi/2} \\
  &= \left( \frac{(\pi/2)^2}{2} + \syn \frac{\pi}{2} \right) - \left( \frac{0^2}{2} + \syn 0 \right) \\
  &= \frac{\pi^2}{8} + 0 - (0 + 1) \\
  &= \frac{\pi^2}{8} - 1.
\end{align*}
Άρα το ζητούμενο εμβαδόν είναι $S = \frac{\pi^2}{8} - 1$.
\end{erwthma}
\end{thema}
