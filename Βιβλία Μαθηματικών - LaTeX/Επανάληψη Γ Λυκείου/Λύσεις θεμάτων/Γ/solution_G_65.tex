%=========== ΛΥΣΗ - 65ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[f'(x)=e^x+1\]
Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, έχουμε
\[f'(x)>0\ ,\quad \text{για κάθε }x\in\mathbb{R}\]
Άρα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, οπότε είναι $1-1$ και επομένως αντιστρέφεται.

Για το σύνολο τιμών της $f$ έχουμε:
\[\lim\limits_{x\to-\infty}f(x)=\lim\limits_{x\to-\infty}(e^x+x-1)=-\infty\]
και
\[\lim\limits_{x\to+\infty}f(x)=\lim\limits_{x\to+\infty}(e^x+x-1)=+\infty\]
Επειδή η $f$ είναι συνεχής και γνησίως αύξουσα στο $\mathbb{R}$, προκύπτει
\[f(\mathbb{R})=\mathbb{R}\]
Άρα το πεδίο ορισμού της αντίστροφης συνάρτησης είναι
\[D_{f^{-1}}=f(\mathbb{R})=\mathbb{R}\]

\item Έχουμε
\[f(0)=e^0+0-1=0\]
άρα
\[f^{-1}(0)=0\]
Επίσης
\[f(1)=e^1+1-1=e\]
άρα
\[f^{-1}(e)=1\]

Επειδή η $f$ είναι γνησίως αύξουσα, και η $f^{-1}$ είναι γνησίως αύξουσα στο $\mathbb{R}$.
Έτσι
\begin{align*}
f^{-1}(x)<1
&\Leftrightarrow f^{-1}(x)<f^{-1}(e)\\
&\Leftrightarrow x<e
\end{align*}
Άρα η λύση της ανίσωσης είναι
\[x\in(-\infty,e)\]

\item Θεωρούμε το ολοκλήρωμα
\[\dint_0^e f^{-1}(x)\d x\]
Κάνουμε την αντικατάσταση
\[x=f(t)\]
Τότε
\[\d x=f'(t)\d t\]
και, επειδή $f^{-1}(0)=0$ και $f^{-1}(e)=1$, τα άκρα γίνονται:
\begin{itemize}
\item για $x=0$, έχουμε $t=f^{-1}(0)=0$
\item για $x=e$, έχουμε $t=f^{-1}(e)=1$
\end{itemize}
Επομένως
\begin{align*}
\dint_0^e f^{-1}(x)\d x
&=\dint_0^1 f^{-1}(f(t))f'(t)\d t\\
&=\dint_0^1 t f'(t)\d t
\end{align*}
Άρα, με αλλαγή συμβόλου της μεταβλητής ολοκλήρωσης,
\[\dint_0^e f^{-1}(x)\d x=\dint_0^1 x f'(x)\d x\]

\item Η $f^{-1}$ είναι γνησίως αύξουσα και $f^{-1}(0)=0$, οπότε για κάθε $x\in[0,e]$ ισχύει
\[f^{-1}(x)\geq 0\]
Άρα το ζητούμενο εμβαδόν είναι
\[E=\dint_0^e f^{-1}(x)\d x\]
Από το προηγούμενο ερώτημα έχουμε
\[E=\dint_0^1 x f'(x)\d x\]
Ολοκληρώνουμε κατά παράγοντες:
\begin{align*}
E&=\dint_0^1 x f'(x)\d x\\
&=\left[xf(x)\right]_0^1-\dint_0^1 f(x)\d x\\
&=f(1)-\dint_0^1(e^x+x-1)\d x\\
&=e-\left[e^x+\frac{x^2}{2}-x\right]_0^1\\
&=e-\left(e+\frac{1}{2}-1-1\right)\\
&=e-\left(e-\frac{3}{2}\right)\\
&=\frac{3}{2}
\end{align*}
Επομένως το ζητούμενο εμβαδόν είναι
\[E=\frac{3}{2}\]
\end{erwthma}
\end{thema}
