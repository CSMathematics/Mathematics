%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.65 ===========
\begin{thema}{Γ}
Δίνεται $f(x) = e^x + x - 1$.
\begin{erwthma}

\item Η παραγωγισιμότητα και τα ακρότατα χαρακτηρίζουν την αντιστρεψιμότητα μιας συνάρτησης. 
Παραγωγίζουμε:
\[ f'(x) = e^x + 1 \]
Ως εκθέτης αναλλοίωτων φυσικών, $e^x > 0 \Rightarrow f'(x) > 1 > 0$ για κάθε $x \in \mathbb{R}$.
Επειδή η $f'(x) > 0$, η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το $\mathbb{R}$. Δεδομένου ότι είναι μονότονη συνάρτηση, δεν επαναλαμβάνει τιμές και είναι αυστηρά \textbf{1-1}. Ως 1-1, \textbf{έχει αναπόφευκτα αντίστροφη} συνάρτηση $f^{-1}$.

\textbf{Πεδία Ορισμού / Τιμών:}
- Το Πεδίο Ορισμού της $D_f = \mathbb{R}$.
- Το Σύνολο Τιμών $f(D_f)$ εξάγεται από τα όριά της (ως αύξουσα):
  - $\lim_{x\to-\infty} f(x) = 0 + (-\infty) - 1 = -\infty$.
  - $\lim_{x\to+\infty} f(x) = (+\infty) + (+\infty) - 1 = +\infty$.
  Άρα $f(D_f) = \mathbb{R}$.

Βάσει του ορισμού αντίστροφης:
- Το \textbf{Πεδίο Ορισμού} της $f^{-1}$ είναι το σύνολο τιμών της $f$, δηλαδή το $\mathbf{\mathbb{R}}$.
- Το \textbf{Σύνολο Τιμών} της $f^{-1}$ είναι το πεδίο ορισμού της $f$, δηλαδή το $\mathbf{\mathbb{R}}$.

\item Γνωρίζουμε εκ της θεωρίας ότι αν $y = f(x)$ τότε αντίστοιχα $x = f^{-1}(y)$.
- Μας δίνεται $f(0) = 0$. Εφαρμόζοντας τον παραπάνω κανόνα (ή εφαρμόζοντας $f^{-1}$ και στα δύο μέλη):
  \[ \mathbf{f^{-1}(0) = 0} \]
- Μας δίνεται $f(1) = e^1 + 1 - 1 = e$. Ομοίως:
  \[ \mathbf{f^{-1}(e) = 1} \]

\item Ζητείται μετασχηματισμός ολοκληρώματος. Έχουμε το $\dint_0^e f^{-1}(y)\,\d y$.
Εφαρμόζουμε την προτεινόμενη αντικατάσταση $\boldsymbol{y = f(x)}$. 
- Το διαφορικό αλλάζει: $\d y = f'(x)\,\d x$.
- Πρέπει να αλλάξουμε και τα επιμέρους άκρα ολοκλήρωσης με τα νέα δεδομένα της $x$:
  - Όταν $y = 0 \Rightarrow f(x) = 0 \Rightarrow x = f^{-1}(0) \Rightarrow x = 0$.
  - Όταν $y = e \Rightarrow f(x) = e \Rightarrow x = f^{-1}(e) \Rightarrow x = 1$.

Με αυτά, το ολοκλήρωμα γράφεται (χρησιμοποιώντας και το ότι $f^{-1}(f(x)) = x$):
\begin{align*}
I &= \int_0^1 f^{-1}(f(x)) \cdot f'(x)\,\d x \\
&= \mathbf{\int_0^1 x \cdot f'(x)\,\d x}
\end{align*}

\item Υπολογίζουμε την ακριβή αριθμητική τιμή αυτού του εμβαδού:
Εφαρμόζουμε ολοκλήρωση κατά παράγοντες στο τελικό εξαγόμενο αποτέλεσμα:
\begin{align*}
I &= \int_0^1 x \cdot (f(x))'\,\d x = \left[ x \cdot f(x) \right]_0^1 - \int_0^1 (x)' f(x)\,\d x \\
&= \left( 1 \cdot f(1) - 0 \cdot f(0) \right) - \int_0^1 f(x)\,\d x \\
&= 1 \cdot e - 0 - \int_0^1 (e^x + x - 1)\,\d x \\
&= e - \left[ e^x + \frac{x^2}{2} - x \right]_0^1 \\
&= e - \left( \left( e^1 + \frac{1}{2} - 1 \right) - \left( e^0 + 0 - 0 \right) \right) \\
&= e - \left( e - \frac{1}{2} - 1 \right) \\
&= e - e + \frac{1}{2} + 1 = \boldsymbol{\frac{3}{2}}
\end{align*}
Η ακριβής αριθμητική τιμή του εμβαδού είναι το $\frac{3}{2}$.
\end{erwthma}
\end{thema}
