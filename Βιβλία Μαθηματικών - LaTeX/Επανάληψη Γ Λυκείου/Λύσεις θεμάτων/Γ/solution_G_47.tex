%=========== ΛΥΣΗ - 47ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $x(t)$ η απόσταση του ανθρώπου από τη βάση του φανοστάτη και $s(t)$ το μήκος της σκιάς του τη χρονική στιγμή $t$.

Από τα όμοια ορθογώνια τρίγωνα που σχηματίζονται έχουμε
\[\frac{s(t)}{2}=\frac{x(t)+s(t)}{6}\]
Άρα
\begin{align*}
6s(t)&=2(x(t)+s(t))\\
6s(t)&=2x(t)+2s(t)\\
4s(t)&=2x(t)\\
s(t)&=\frac{x(t)}{2}
\end{align*}
Επομένως
\[s(t)=\frac{x(t)}{2}\]

\item Έστω $y(t)$ η απόσταση της άκρης της σκιάς από τη βάση του φανοστάτη.
Τότε
\[y(t)=x(t)+s(t)\]
και, από το προηγούμενο ερώτημα,
\[y(t)=x(t)+\frac{x(t)}{2}=\frac{3}{2}x(t)\]
Επειδή ο άνθρωπος πλησιάζει τον φανοστάτη με σταθερή ταχύτητα $1.5\si{m/s}$, η απόστασή του από τον φανοστάτη μειώνεται με ρυθμό
\[x'(t)=-1.5\si{m/s}\]
Παραγωγίζοντας τη σχέση
\[y(t)=\frac{3}{2}x(t)\]
ως προς τον χρόνο $t$, προκύπτει
\[y'(t)=\frac{3}{2}x'(t)=\frac{3}{2}\cdot(-1.5)=-2.25\si{m/s}\]
Άρα η άκρη της σκιάς κινείται με σταθερή ταχύτητα μέτρου
\[2.25\si{m/s}\]
προς τη βάση του φανοστάτη.

\item Η οπτική ακτίνα από την κορυφή του φανοστάτη προς το κεφάλι του ανθρώπου σχηματίζει με τον κατακόρυφο φανοστάτη ορθογώνιο τρίγωνο.
Η οριζόντια πλευρά του τριγώνου είναι $x$, ενώ η κατακόρυφη πλευρά είναι
\[6-2=4\si{m}\]
Επομένως
\[\ef\theta=\frac{x}{4}\]

\item Από το προηγούμενο ερώτημα έχουμε
\[\ef\theta(t)=\frac{x(t)}{4}\]
Παραγωγίζοντας ως προς τον χρόνο $t$, παίρνουμε
\[\frac{1}{\syn^2\theta(t)}\theta'(t)=\frac{x'(t)}{4}\]
Τη χρονική στιγμή $t_0$ κατά την οποία ο άνθρωπος απέχει $4\si{m}$ από τον φανοστάτη, έχουμε
\[x(t_0)=4\]
οπότε
\[\ef\theta(t_0)=\frac{4}{4}=1\]
Άρα
\[\frac{1}{\syn^2\theta(t_0)}=1+\ef^2\theta(t_0)=1+1^2=2\]
Επειδή $x'(t_0)=-1.5\si{m/s}$, από τη σχέση
\[\frac{1}{\syn^2\theta(t)}\theta'(t)=\frac{x'(t)}{4}\]
για $t=t_0$ έχουμε
\begin{align*}
2\theta'(t_0)&=\frac{-1.5}{4}\\
2\theta'(t_0)&=-\frac{3}{8}\\
\theta'(t_0)&=-\frac{3}{16}\si{rad/s}
\end{align*}
Επομένως η γωνία $\theta$ μειώνεται με ρυθμό $\dfrac{3}{16}\si{rad/s}$ τη χρονική στιγμή που ο άνθρωπος απέχει $4\si{m}$ από τον φανοστάτη.
\end{erwthma}
\end{thema}
