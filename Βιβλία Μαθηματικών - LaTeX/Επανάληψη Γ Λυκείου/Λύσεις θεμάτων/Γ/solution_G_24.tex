%=========== ΛΥΣΗ - 24ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $M(x_0, f(x_0))$ το σημείο επαφής της ζητούμενης εφαπτομένης ($\varepsilon$) με τη γραφική παράσταση της $f$. 
Γνωρίζουμε ότι $f(x) = e^x$, οπότε η συνάρτηση είναι παραγωγίσιμη με $f'(x) = e^x$.
Η εξίσωση της εφαπτομένης στο $M$ είναι:
\[ y - f(x_0) = f'(x_0)(x - x_0) \implies y - e^{x_0} = e^{x_0}(x - x_0) \]
Για να διέρχεται η εφαπτομένη από την αρχή των αξόνων $O(0,0)$, πρέπει οι συντεταγμένες $(0,0)$ να επαληθεύουν την εξίσωσή της. 
Αντικαθιστούμε $x = 0$ και $y = 0$:
\begin{align*}
0 - e^{x_0} &= e^{x_0}(0 - x_0) \\
-e^{x_0} &= -x_0 e^{x_0}
\end{align*}
Επειδή $e^{x_0} \neq 0$ για κάθε $x_0 \in \mathbb{R}$, διαιρούμε και τα δύο μέλη με $-e^{x_0}$ και λαμβάνουμε:
\[ x_0 = 1 \]
Για $x_0 = 1$, έχουμε $f(1) = e^1 = e$ και $f'(1) = e^1 = e$. Επομένως, η εξίσωση της εφαπτομένης είναι:
\[ y - e = e(x - 1) \implies y - e = ex - e \implies \boldsymbol{y = ex} \]

\item Για τη μελέτη της κυρτότητας, υπολογίζουμε τη δεύτερη παράγωγο της $f$:
\[ f'(x) = e^x \implies f''(x) = (e^x)' = e^x \]
Επειδή ισχύει $e^x > 0$ για κάθε πραγματικό αριθμό $x$, συμπεραίνουμε ότι $f''(x) > 0$ στο $\mathbb{R}$.
Άρα, η συνάρτηση $f$ είναι \textbf{κυρτή} σε όλο το $\mathbb{R}$.

Γνωρίζουμε ότι η γραφική παράσταση μιας κυρτής συνάρτησης βρίσκεται "πάνω" από κάθε εφαπτομένη της, με εξαίρεση το σημείο επαφής όπου συμπίπτουν.
Από το ερώτημα (α), η ευθεία $y = ex$ είναι η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $x_0 = 1$.
Επομένως, για κάθε $x \in \mathbb{R}$ ισχύει η ανισότητα:
\[ f(x) \ge y_{\varepsilon} \implies \boldsymbol{e^x \ge ex} \]
(Η ισότητα ισχύει προφανώς μόνο για $x = 1$).

\item Ζητείται ο υπολογισμός του ορίου:
\[ L_1 = \lim_{x \to 1} \frac{\ln(f(x)) - 1}{x - 1} \]
Αντικαθιστώντας τον τύπο της συνάρτησης $f(x) = e^x$, έχουμε:
\[ \ln(f(x)) = \ln(e^x) = x \ln e = x \]
Άρα το όριο μετασχηματίζεται ως εξής:
\[ L_1 = \lim_{x \to 1} \frac{x - 1}{x - 1} \]
Εφόσον το $x$ τείνει στο $1$, ισχύει $x \neq 1$, οπότε μπορούμε να απλοποιήσουμε τον αριθμητή με τον παρονομαστή:
\[ L_1 = \lim_{x \to 1} 1 = \boldsymbol{1} \]

\item Ζητείται ο υπολογισμός του ορίου:
\[ L_2 = \lim_{x \to +\infty} \frac{f(x)}{f'(x) + x} = \lim_{x \to +\infty} \frac{e^x}{e^x + x} \]
Καθώς $x \to +\infty$, το όριο παρουσιάζει απροσδιοριστία της μορφής $\frac{+\infty}{+\infty}$.
Βγάζουμε κοινό παράγοντα τον όρο $e^x$ στον παρονομαστή:
\[ L_2 = \lim_{x \to +\infty} \frac{e^x}{e^x \left(1 + \frac{x}{e^x}\right)} = \lim_{x \to +\infty} \frac{1}{1 + \frac{x}{e^x}} \]
Υπολογίζουμε ξεχωριστά το όριο του κλάσματος $\frac{x}{e^x}$. 
Πρόκειται για απροσδιοριστία $\frac{+\infty}{+\infty}$, οπότε εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to +\infty} \frac{x}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to +\infty} \frac{(x)'}{(e^x)'} = \lim_{x \to +\infty} \frac{1}{e^x} = 0 \]
Αντικαθιστώντας αυτή την τιμή στο αρχικό όριο, προκύπτει:
\[ L_2 = \frac{1}{1 + 0} = \boldsymbol{1} \]
\textit{(Σημείωση: Το όριο $L_2$ θα μπορούσε να υπολογιστεί εξίσου σωστά και με δύο διαδοχικές εφαρμογές του κανόνα \eng{De L'Hospital} απευθείας στο αρχικό κλάσμα $\frac{e^x}{e^x+x}$).}
\end{erwthma}
\end{thema}
