\begin{thema}{Γ}
\begin{erwthma}
\item Παρατηρούμε ότι η παράσταση κάτω από τη ρίζα είναι ταυτόσημη με το τετράγωνο ενός τρινομήμου:
\[ x^2 - 4x + 4 = (x-2)^2 \]
Συνεπώς, η συνάρτηση $f$ απλοποιείται ως εξής:
\[ f(x) = \sqrt{(x-2)^2} = |x-2| \]

\item Για να εξετάσουμε την παραγωγισιμότητα της $f$ στο $x_0 = 2$, εξετάζουμε το όριο του διαφορικού συντελεστή:
\[ \lim_{h \to 0} \frac{f(2+h) - f(2)}{h} \]
Γνωρίζουμε ότι $f(2) = |2-2| = 0$ και $f(2+h) = |(2+h)-2| = |h|$. Το όριο γίνεται:
\[ \lim_{h \to 0} \frac{|h|}{h} \]
Υπολογίζουμε τα πλευρικά όρια:
\begin{itemize}
    \item $\lim_{h \to 0^+} \frac{|h|}{h} = \lim_{h \to 0^+} \frac{h}{h} = 1$
    \item $\lim_{h \to 0^-} \frac{|h|}{h} = \lim_{h \to 0^-} \frac{-h}{h} = -1$
\end{itemize}
Επειδή τα πλευρικά όρια είναι άνισα, το όριο $\lim_{h \to 0} \frac{f(2+h) - f(2)}{h}$ δεν υπάρχει. Άρα, η $f$ \textbf{δεν είναι παραγωγίσιμη} στο σημείο $x_0 = 2$.

\item Για $x \to 2^-$, ισχύει $x < 2$, οπότε $f(x) = |x-2| = -(x-2) = 2-x$. Το όριο είναι:
\[ \lim_{x \to 2^-} \frac{f(x)}{x^2 - 4} = \lim_{x \to 2^-} \frac{2-x}{(x-2)(x+2)} \]
\[ \lim_{x \to 2^-} \frac{-(x-2)}{(x-2)(x+2)} = \lim_{x \to 2^-} \frac{-1}{x+2} = \frac{-1}{2+2} = -\frac{1}{4} \]

\item Το εμβαδόν $E$ του χωρίου είναι το ορισμένο ολοκλήρωμα της $f$ στο διάστημα $[0, 4]$:
\[ E = \int_{0}^{4} f(x) \d x = \int_{0}^{4} |x-2| \d x \]
Λόγω της απόλυτης τιμής, χωρίζουμε το ολοκλήρωμα στο σημείο $x=2$:
\[ E = \int_{0}^{2} (2-x) \d x + \int_{2}^{4} (x-2) \d x \]
Υπολογίζουμε τα δύο ολοκληρώματα:
\begin{align*}
\int_{0}^{2} (2-x) \d x &= \left[ 2x - \frac{x^2}{2} \right]_{0}^{2} = \left( 4 - \frac{4}{2} \right) - (0 - 0) = 2 \\
\int_{2}^{4} (x-2) \d x &= \left[ \frac{x^2}{2} - 2x \right]_{2}^{4} = \left( \frac{16}{2} - 8 \right) - \left( \frac{4}{2} - 4 \right) = 0 - (-2) = 2
\end{align*}
Άρα, το εμβαδόν είναι:
\[ E = 2 + 2 = 4 \text{ τ.μ.} \]
\end{erwthma}
\end{thema}
