%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.73 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Δίνεται η $x f'(x) - 2f(x) = x^3 e^x$ για $x > 0$.
Διαιρούμε ολόκληρη την εξίσωση με τον παράγοντα $x^3$ ($>0$) ώστε να δημιουργηθεί ο κανόνας πηλίκου:
\begin{align*}
\frac{x f'(x) - 2f(x)}{x^3} &= \frac{x^3 e^x}{x^3} \\
\frac{x^2 f'(x) - 2x f(x)}{x^4} &= e^x \\
\left( \frac{f(x)}{x^2} \right)' &= (e^x)'
\end{align*}
Από τη συνέπεια του θεωρήματος Μέσης Τιμής, οι αρχικές διαφέρουν κατά σταθερά $c$:
\[ \frac{f(x)}{x^2} = e^x + c \]
Για την εύρεση της $c$, χρησιμοποιούμε την αρχική συνθήκη $f(1) = e$:
\[ \frac{f(1)}{1^2} = e^1 + c \Rightarrow e = e + c \Rightarrow c = 0 \]
Κατά συνέπεια:
\[ \frac{f(x)}{x^2} = e^x \Rightarrow \mathbf{f(x) = x^2 e^x} \]

\item \textbf{Μονοτονία και Ακρότατα:}
Παραγωγίζουμε την εξαχθείσα συνάρτηση:
\[ f'(x) = (x^2)' e^x + x^2(e^x)' = 2x e^x + x^2 e^x = x(x+2) e^x \]
Αφού δεδομένα εργαζόμαστε στο $x \in (0, +\infty)$, όλοι οι παράγοντες $x$, $x+2$ και $e^x$ είναι αυστηρά θετικοί.
Άρα $f'(x) > 0$ για κάθε $x > 0$. 
Η συνάρτηση είναι \textbf{γνησίως αύξουσα} και επειδή αυξάνεται μονότονα στο ανοιχτό πεδίο, \textbf{δε διαθέτει κανένα ακρότατο}.

\textbf{Κυρτότητα:}
Παραγωγίζουμε εκ νέου:
\begin{align*}
f''(x) &= \left( (x^2+2x) e^x \right)' = (x^2+2x)' e^x + (x^2+2x)(e^x)' \\
&= (2x+2)e^x + (x^2+2x)e^x = (x^2 + 4x + 2) e^x
\end{align*}
Ομοίως, για $x > 0$, οι όροι του τριωνύμου είναι θετικοί $\Rightarrow x^2+4x+2 > 0$.
Επομένως $\boldsymbol{f''(x) > 0} \ \forall x > 0$, πράγμα που επισφραγίζει ότι η $f$ είναι \textbf{απολύτως κυρτή (κοίλα άνω)} στο πεδίο της.

\item Διαμορφώνουμε τα διαστήματα $[x, x+1]$ και $[x+1, x+2]$. Η συνάρτηση $f$ ικανοποιεί τις συνθήκες του Θεωρήματος Μέσης Τιμής σε έκαστο από αυτά.
- Υπάρχει $\xi_1 \in (x, x+1)$ τέτοιο ώστε: $f'(\xi_1) = \frac{f(x+1)-f(x)}{1} = f(x+1)-f(x)$.
- Υπάρχει $\xi_2 \in (x+1, x+2)$ τέτοιο ώστε: $f'(\xi_2) = \frac{f(x+2)-f(x+1)}{1} = f(x+2)-f(x+1)$.

Λόγω του ότι τα διαστήματα είναι διαδοχικά, επιβάλλεται αυστηρά $\xi_1 < \xi_2$.
Επιπροσθέτως, στο προηγούμενο ερώτημα αποδείξαμε ότι η $f$ είναι κυρτή, γεγονός που σημαίνει ισοδύναμα ότι η $f'$ είναι \textbf{γνησίως αύξουσα}.
Εφαρμόζοντας τη μονοτονία της $f'$:
\[ \xi_1 < \xi_2 \xRightarrow{f' \uparrow} f'(\xi_1) < f'(\xi_2) \]
Αντικαθιστούμε τα κλάσματα των Θ.Μ.Τ.:
\[ f(x+1) - f(x) < f(x+2) - f(x+1) \]
Περνώντας το $-f(x+1)$ στο δεύτερο μέλος και το $-f(x)$ στο πρώτο (εναλλαγή):
\[ \mathbf{2f(x+1) < f(x+2) + f(x)} \Rightarrow \boldsymbol{f(x+2) + f(x) > 2f(x+1)} \]

\item Ζητείται σύγκριση με την εφαπτομένη στο $x_0 = 1$.
$f(1) = 1^2 e^1 = e$.
$f'(1) = 1(1+2)e^1 = 3e$.
Η εξίσωση της εφαπτομένης ευθείας $\varepsilon$ είναι:
\[ y - e = 3e(x - 1) \Rightarrow y = 3ex - 3e + e \Rightarrow y = 3ex - 2e \]
Εφόσον η $f$ αποδείχθηκε \textbf{κυρτή}, η γραφική της παράσταση βρίσκεται πάντα («ακουμπώντας» στο 1) \textbf{πάνω από οποιαδήποτε εφαπτομένη}:
\[ f(x) \geq y_{\varepsilon} \Rightarrow x^2 e^x \geq 3ex - 2e \]
Διαιρούμε την ανισότητα με τον θετικό αριθμό $e$ (αφαιρείται εκθέτης):
\[ \frac{x^2 e^x}{e} \geq \frac{e(3x - 2)}{e} \Rightarrow \mathbf{x^2 e^{x-1} \geq 3x - 2} \]
Η ανισότητα παράχθηκε επιτυχώς για $x>0$.
\end{erwthma}
\end{thema}
