%=========== ΛΥΣΗ - 73ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
xf'(x)-2f(x)=x^3e^x
\]
διαιρούμε με $x^3$, αφού $x>0$, και παίρνουμε
\[
\frac{xf'(x)-2f(x)}{x^3}=e^x
\]
Όμως
\[
\left(\frac{f(x)}{x^2}\right)'
=\frac{x^2f'(x)-2xf(x)}{x^4}
=\frac{xf'(x)-2f(x)}{x^3}
\]
Άρα
\[
\left(\frac{f(x)}{x^2}\right)'=e^x=(e^x)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
\frac{f(x)}{x^2}=e^x+c
\]
Για $x=1$ έχουμε $f(1)=e$, οπότε
\[
\frac{f(1)}{1^2}=e^1+c\Leftrightarrow e=e+c\Leftrightarrow c=0
\]
Άρα
\[
\frac{f(x)}{x^2}=e^x
\]
και τελικά
\[
f(x)=x^2e^x\ ,\quad x>0
\]

\item Από το προηγούμενο ερώτημα
\[
f(x)=x^2e^x
\]
οπότε
\[
f'(x)=2xe^x+x^2e^x=x(x+2)e^x
\]
Για κάθε $x>0$ ισχύει
\[
x>0,\quad x+2>0,\quad e^x>0
\]
άρα
\[
f'(x)>0\ ,\quad \text{για κάθε }x>0
\]
Επομένως η $f$ είναι γνησίως αύξουσα στο $(0,+\infty)$ και δεν παρουσιάζει ακρότατο στο πεδίο ορισμού της.

Επιπλέον,
\begin{align*}
f''(x)
&=\left(x(x+2)e^x\right)'\\
&=(2x+2)e^x+(x^2+2x)e^x\\
&=(x^2+4x+2)e^x
\end{align*}
Για κάθε $x>0$ έχουμε
\[
x^2+4x+2>0
\quad \text{και}\quad
e^x>0
\]
άρα
\[
f''(x)>0\ ,\quad \text{για κάθε }x>0
\]
Συνεπώς η $f$ είναι κυρτή στο $(0,+\infty)$.

\item Έστω $x>0$.
Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[x,x+1]$.
Υπάρχει $\xi_1\in(x,x+1)$ τέτοιο ώστε
\[
f(x+1)-f(x)=f'(\xi_1)
\]
Επίσης, εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[x+1,x+2]$.
Υπάρχει $\xi_2\in(x+1,x+2)$ τέτοιο ώστε
\[
f(x+2)-f(x+1)=f'(\xi_2)
\]
Επειδή
\[
\xi_1<\xi_2
\]
και, από την κυρτότητα της $f$, η $f'$ είναι γνησίως αύξουσα, έχουμε
\[
f'(\xi_1)<f'(\xi_2)
\]
Άρα
\[
f(x+1)-f(x)<f(x+2)-f(x+1)
\]
οπότε
\[
f(x+2)+f(x)>2f(x+1)
\]

\item Από το προηγούμενο ερώτημα γνωρίζουμε ότι η $f$ είναι κυρτή στο $(0,+\infty)$.
Άρα η γραφική της παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της.

Στο σημείο με τετμημένη $1$ έχουμε
\[
f(1)=e
\]
και
\[
f'(1)=1(1+2)e=3e
\]
Η εφαπτομένη της $C_f$ στο $x=1$ είναι
\[
y-e=3e(x-1)
\]
δηλαδή
\[
y=3ex-2e
\]
Επειδή η $f$ είναι κυρτή, για κάθε $x>0$ ισχύει
\[
f(x)\geq 3ex-2e
\]
Άρα
\[
x^2e^x\geq 3ex-2e
\]
Διαιρώντας με τον θετικό αριθμό $e$, παίρνουμε
\[
x^2e^{x-1}\geq 3x-2
\]
για κάθε $x>0$.
\end{erwthma}
\end{thema}
