%=========== ΛΥΣΗ - 26ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Γνωρίζουμε ότι η σχέση $|f(x)| \le x^2$ ισχύει για κάθε πραγματικό αριθμό $x$.
Για $x = 0$, αντικαθιστούμε στη σχέση και παίρνουμε:
\[ |f(0)| \le 0^2 \implies |f(0)| \le 0 \]
Όμως, η απόλυτη τιμή είναι εξ ορισμού μη αρνητική ποσότητα, δηλαδή $|f(0)| \ge 0$.
Για να ισχύουν και τα δύο ταυτόχρονα, πρέπει υποχρεωτικά:
\[ |f(0)| = 0 \implies \boldsymbol{f(0) = 0} \]

\item Για να ελέγξουμε αν η $f$ είναι παραγωγίσιμη στο $x_0 = 0$, θα υπολογίσουμε, αν υπάρχει, το όριο του πηλίκου μεταβολής της:
\[ L = \lim_{x \to 0} \frac{f(x) - f(0)}{x - 0} = \lim_{x \to 0} \frac{f(x)}{x} \]
Από τη δοσμένη σχέση έχουμε $|f(x)| \le x^2$, που ισοδυναμεί με:
\[ -x^2 \le f(x) \le x^2 \]
Για $x \neq 0$, διαιρούμε την ανισότητα με το θετικό αριθμό $|x|$ (αφού $|x|^2 = x^2$):
\[ -|x|^2 \le f(x) \le |x|^2 \implies -|x| \le \frac{f(x)}{|x|} \le |x| \]
Επειδή $\left| \frac{f(x)}{x} \right| = \frac{|f(x)|}{|x|}$, η σχέση ισοδυναμεί με:
\[ \left| \frac{f(x)}{x} \right| \le |x| \implies -|x| \le \frac{f(x)}{x} \le |x| \]
Υπολογίζουμε τα όρια των ακραίων συναρτήσεων:
\[ \lim_{x \to 0} (-|x|) = 0 \quad \text{και} \quad \lim_{x \to 0} |x| = 0 \]
Σύμφωνα με το \textbf{Κριτήριο Παρεμβολής}, συμπεραίνουμε ότι:
\[ \lim_{x \to 0} \frac{f(x)}{x} = 0 \]
Το όριο υπάρχει και είναι πραγματικός αριθμός, άρα η $f$ είναι \textbf{παραγωγίσιμη στο 0} με $\boldsymbol{f'(0) = 0}$.

\item Ζητείται ο υπολογισμός του ορίου:
\[ \lim_{x \to 0} \frac{f(x) + \hm x}{x + \ef x} \]
Όταν το $x \to 0$, ο αριθμητής τείνει στο $f(0) + \hm 0 = 0+0=0$ και ο παρονομαστής στο $0 + \ef 0 = 0+0=0$. 
Πρόκειται για απροσδιόριστη μορφή $\frac{0}{0}$. Δεν μπορούμε να εφαρμόσουμε τον κανόνα \eng{De L'Hospital} διότι δεν γνωρίζουμε αν η $f$ είναι παραγωγίσιμη σε περιοχή του 0, παρά μόνο στο 0.
Διαιρούμε αριθμητή και παρονομαστή με το $x \neq 0$:
\[ \lim_{x \to 0} \frac{\frac{f(x)}{x} + \frac{\hm x}{x}}{\frac{x}{x} + \frac{\ef x}{x}} = \lim_{x \to 0} \frac{\frac{f(x)}{x} + \frac{\hm x}{x}}{1 + \frac{\hm x}{x} \cdot \frac{1}{\syn x}} \]
Από τα προηγούμενα γνωρίζουμε ότι $\lim\limits_{x \to 0} \dfrac{f(x)}{x} = f'(0) = 0$. 
Επίσης, γνωρίζουμε το βασικό τριγωνομετρικό όριο $\lim\limits_{x \to 0} \dfrac{\hm x}{x} = 1$.
Για τη συνάρτηση $\syn x$, έχουμε $\lim\limits_{x \to 0} \dfrac{1}{\syn x} = \dfrac{1}{1} = 1$.
Εφαρμόζοντας τις ιδιότητες των ορίων, προκύπτει:
\[ \frac{0 + 1}{1 + 1 \cdot 1} = \boldsymbol{\frac{1}{2}} \]

\item Η συνάρτηση $f$ είναι κυρτή στο $(0, +\infty)$ και παραγωγίσιμη σε αυτό. Επομένως, η παράγωγός της $f'$ είναι \textbf{γνησίως αύξουσα} στο $(0, +\infty)$.
Θα αποδείξουμε ότι $f(x) \ge 0$ για κάθε $x > 0$ με τη \textbf{μέθοδο της απαγωγής σε άτοπο}.

Έστω ότι υπάρχει κάποιο $x_0 > 0$ τέτοιο ώστε $f(x_0) < 0$.
Εφαρμόζουμε το \textbf{Θεώρημα Μέσης Τιμής} για την $f$ στο διάστημα $[0, x_0]$. Η $f$ είναι συνεχής στο $[0, x_0]$ και παραγωγίσιμη στο $(0, x_0)$. Άρα υπάρχει ένα $\xi_1 \in (0, x_0)$ τέτοιο ώστε:
\[ f'(\xi_1) = \frac{f(x_0) - f(0)}{x_0 - 0} = \frac{f(x_0)}{x_0} \]
Επειδή $x_0 > 0$ και υποθέσαμε $f(x_0) < 0$, έπεται ότι $f'(\xi_1) < 0$.
Επιλέγουμε τώρα ένα τυχαίο $x \in (0, \xi_1)$.
Εφαρμόζουμε ξανά το \textbf{Θεώρημα Μέσης Τιμής} για την $f$ στο διάστημα $[0, x]$. 
Υπάρχει ένα $\xi_2 \in (0, x)$ τέτοιο ώστε:
\[ f'(\xi_2) = \frac{f(x) - f(0)}{x - 0} = \frac{f(x)}{x} \]
Παρατηρούμε ότι $\xi_2 \in (0, x) \subset (0, \xi_1)$, άρα $\xi_2 < \xi_1$.
Επειδή η $f'$ είναι γνησίως αύξουσα στο $(0, +\infty)$, ισχύει:
\[ \xi_2 < \xi_1 \implies f'(\xi_2) < f'(\xi_1) \implies \frac{f(x)}{x} < f'(\xi_1) \]
Η ανισότητα αυτή ισχύει για κάθε $x \in (0, \xi_1)$. 
Παίρνουμε το όριο καθώς το $x \to 0^+$ και στα δύο μέλη. Αφού γνωρίζουμε από το (β) ότι το όριο υπάρχει:
\[ \lim_{x \to 0^+} \frac{f(x)}{x} \le \lim_{x \to 0^+} f'(\xi_1) \implies f'(0) \le f'(\xi_1) \]
Όμως, γνωρίζουμε ότι $f'(0) = 0$. Συνεπώς, καταλήγουμε στην ανισότητα:
\[ 0 \le f'(\xi_1) \]
Αυτό όμως είναι \textbf{άτοπο}, διότι είχαμε αποδείξει νωρίτερα ότι $f'(\xi_1) < 0$.
Η αρχική μας υπόθεση ότι υπάρχει $x_0 > 0$ με $f(x_0) < 0$ οδήγησε σε άτοπο, επομένως απορρίπτεται.
Κατά συνέπεια, ισχύει $f(x) \ge 0$ για κάθε $x > 0$.
\end{erwthma}
\end{thema}
