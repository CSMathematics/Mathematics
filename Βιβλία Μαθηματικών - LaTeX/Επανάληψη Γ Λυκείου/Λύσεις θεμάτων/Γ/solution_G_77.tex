%=========== ΛΥΣΗ - 77ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
f^2(x)-4f(x)=x^2-4
\]
προσθέτουμε $4$ και στα δύο μέλη:
\[
f^2(x)-4f(x)+4=x^2
\]
Άρα
\[
\left(f(x)-2\right)^2=x^2
\]
οπότε
\[
|f(x)-2|=|x|
\]
Επειδή από την εκφώνηση έχουμε $f(x)\geq 2$, είναι
\[
f(x)-2\geq 0
\]
Άρα
\[
f(x)-2=|x|
\]
και επομένως
\[
f(x)=|x|+2\ ,\quad x\in\mathbb{R}
\]

\item Από το προηγούμενο ερώτημα
\[
f(x)=|x|+2
\]
και
\[
f(0)=2
\]
Εξετάζουμε τις πλευρικές παραγώγους στο $0$.

Για $x>0$ είναι $f(x)=x+2$, οπότε
\[
\lim\limits_{x\to 0^+}\frac{f(x)-f(0)}{x-0}
=\lim\limits_{x\to 0^+}\frac{x+2-2}{x}
=1
\]
Για $x<0$ είναι $f(x)=-x+2$, οπότε
\[
\lim\limits_{x\to 0^-}\frac{f(x)-f(0)}{x-0}
=\lim\limits_{x\to 0^-}\frac{-x+2-2}{x}
=-1
\]
Οι δύο πλευρικές παράγωγοι είναι διαφορετικές.
Άρα η $f$ δεν είναι παραγωγίσιμη στο $x_0=0$.

\item Ένα σημείο της $C_f$ με $x\geq 0$ έχει συντεταγμένες
\[
M(x,f(x))=M(x,x+2)
\]
Το τετράγωνο της απόστασής του από την αρχή των αξόνων είναι
\[
d^2(x)=x^2+(x+2)^2
\]
Θεωρούμε
\[
\varphi(x)=d^2(x)=x^2+(x+2)^2=2x^2+4x+4\ ,\quad x\geq 0
\]
Τότε
\[
\varphi'(x)=4x+4>0\ ,\quad \text{για κάθε }x\geq 0
\]
Άρα η $\varphi$ είναι γνησίως αύξουσα στο $[0,+\infty)$.

Επομένως η απόσταση από το $O(0,0)$ ελαχιστοποιείται για $x=0$, δηλαδή στο σημείο
\[
(0,2)
\]

\item Για να εφαρμόζεται το Θεώρημα \eng{Rolle} στο διάστημα $[-1,1]$, πρέπει η $f$ να είναι:
\begin{itemize}
\item συνεχής στο $[-1,1]$
\item παραγωγίσιμη στο $(-1,1)$
\item με $f(-1)=f(1)$
\end{itemize}
Η $f(x)=|x|+2$ είναι συνεχής στο $[-1,1]$ και
\[
f(-1)=|-1|+2=3
\]
ενώ
\[
f(1)=|1|+2=3
\]
Άρα
\[
f(-1)=f(1)
\]
Όμως, από το δεύτερο ερώτημα, η $f$ δεν είναι παραγωγίσιμη στο $0$, και το $0$ ανήκει στο εσωτερικό του διαστήματος $[-1,1]$.

Επομένως δεν ικανοποιούνται όλες οι προϋποθέσεις του Θεωρήματος \eng{Rolle}, άρα το θεώρημα δεν εφαρμόζεται στο $[-1,1]$.
\end{erwthma}
\end{thema}
