%=========== ΛΥΣΗ - 51ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $x\to 0^+$ έχουμε
\[\ln\left(1+\frac{1}{x}\right)=\ln\left(\frac{x+1}{x}\right)=\ln(x+1)-\ln x\]
οπότε
\[x^2\ln\left(1+\frac{1}{x}\right)=x^2\ln(x+1)-x^2\ln x\]
Επειδή
\[\lim\limits_{x\to 0^+}x^2\ln(x+1)=0\]
και
\[\lim\limits_{x\to 0^+}x^2\ln x=0\]
προκύπτει ότι
\[\lim\limits_{x\to 0^+}x^2\ln\left(1+\frac{1}{x}\right)=0\]
Άρα
\[\lim\limits_{x\to 0^+}h(x)=0\]

Για το όριο στο $+\infty$ θέτουμε
\[t=\frac{1}{x}\]
Τότε, καθώς $x\to+\infty$, έχουμε $t\to 0^+$ και
\begin{align*}
h(x)&=x^2\ln\left(1+\frac{1}{x}\right)-x\\
&=\frac{\ln(1+t)}{t^2}-\frac{1}{t}\\
&=\frac{\ln(1+t)-t}{t^2}
\end{align*}
Άρα, με τον κανόνα \eng{L'Hôpital},
\begin{align*}
\lim\limits_{x\to+\infty}h(x)
&=\lim\limits_{t\to 0^+}\frac{\ln(1+t)-t}{t^2}\\
&=\lim\limits_{t\to 0^+}\frac{\frac{1}{1+t}-1}{2t}\\
&=\lim\limits_{t\to 0^+}\frac{-t}{2t(1+t)}\\
&=-\frac{1}{2}
\end{align*}

\item Η $h$ είναι δύο φορές παραγωγίσιμη στο $(0,+\infty)$ και
\begin{align*}
h'(x)&=\left[x^2\ln\left(1+\frac{1}{x}\right)-x\right]'\\
&=2x\ln\left(1+\frac{1}{x}\right)+x^2\cdot\frac{-\frac{1}{x^2}}{1+\frac{1}{x}}-1\\
&=2x\ln\left(1+\frac{1}{x}\right)-\frac{x}{x+1}-1
\end{align*}
Επομένως
\[h''(x)=2\ln\left(1+\frac{1}{x}\right)-\frac{2}{x+1}-\frac{1}{(x+1)^2}\]
Θέτουμε
\[t=\frac{1}{x}>0\]
Τότε
\[h''(x)=2\ln(1+t)-\frac{2t}{1+t}-\frac{t^2}{(1+t)^2}\]
Θεωρούμε τη συνάρτηση
\[\phi(t)=2\ln(1+t)-\frac{2t}{1+t}-\frac{t^2}{(1+t)^2}\ ,\ t>0\]
Έχουμε
\[\lim\limits_{t\to 0^+}\phi(t)=0\]
και
\begin{align*}
\phi'(t)&=\frac{2}{1+t}-\frac{2}{(1+t)^2}-\frac{2t}{(1+t)^3}\\
&=\frac{2t^2}{(1+t)^3}>0\ ,\ t>0
\end{align*}
Άρα η $\phi$ είναι γνησίως αύξουσα στο $(0,+\infty)$ και, επειδή
\[\lim\limits_{t\to 0^+}\phi(t)=0\]
ισχύει
\[\phi(t)>0\ ,\ t>0\]
Επομένως
\[h''(x)>0\ ,\ x>0\]
και η $h$ είναι \textbf{κυρτή} στο $(0,+\infty)$.

\item Από το προηγούμενο ερώτημα έχουμε
\[h'(x)=2x\ln\left(1+\frac{1}{x}\right)-\frac{x}{x+1}-1\]
Θέτουμε πάλι $t=\dfrac{1}{x}>0$. Τότε
\begin{align*}
h'(x)<0
&\Leftrightarrow \frac{2}{t}\ln(1+t)-\frac{1}{1+t}-1<0\\
&\Leftrightarrow 2\ln(1+t)-\frac{t}{1+t}-t<0\\
&\Leftrightarrow 2\ln(1+t)-\frac{t(t+2)}{1+t}<0
\end{align*}
Θεωρούμε τη συνάρτηση
\[\psi(t)=\frac{t(t+2)}{1+t}-2\ln(1+t)\ ,\ t>0\]
Έχουμε
\[\lim\limits_{t\to 0^+}\psi(t)=0\]
και
\begin{align*}
\psi'(t)&=\left(t+1-\frac{1}{t+1}\right)'-\frac{2}{1+t}\\
&=1+\frac{1}{(t+1)^2}-\frac{2}{1+t}\\
&=\frac{t^2}{(1+t)^2}>0\ ,\ t>0
\end{align*}
Άρα η $\psi$ είναι γνησίως αύξουσα στο $(0,+\infty)$ και, επειδή
\[\lim\limits_{t\to 0^+}\psi(t)=0\]
ισχύει
\[\psi(t)>0\ ,\ t>0\]
Δηλαδή
\[2\ln(1+t)-\frac{t(t+2)}{1+t}<0\ ,\ t>0\]
οπότε
\[h'(x)<0\ ,\ x>0\]
Επομένως η $h$ είναι \textbf{γνησίως φθίνουσα} στο $(0,+\infty)$.

Από το Γ1 έχουμε
\[\lim\limits_{x\to 0^+}h(x)=0\]
και
\[\lim\limits_{x\to+\infty}h(x)=-\frac{1}{2}\]
Επειδή η $h$ είναι συνεχής και γνησίως φθίνουσα στο $(0,+\infty)$, το σύνολο τιμών της είναι
\[h((0,+\infty))=\left(-\frac{1}{2},0\right)\]

\item Από το προηγούμενο ερώτημα, για κάθε $x>0$ ισχύει
\[-\frac{1}{2}<h(x)<0\]
Άρα, για κάθε $x\in[1,2]$,
\[-\frac{1}{2}<h(x)<0\]
Ολοκληρώνοντας στο διάστημα $[1,2]$, παίρνουμε
\[\dint_1^2 -\frac{1}{2}\d x<\dint_1^2 h(x)\d x<\dint_1^2 0\d x\]
δηλαδή
\[-\frac{1}{2}<\dint_1^2 h(x)\d x<0\]
\end{erwthma}
\end{thema}
