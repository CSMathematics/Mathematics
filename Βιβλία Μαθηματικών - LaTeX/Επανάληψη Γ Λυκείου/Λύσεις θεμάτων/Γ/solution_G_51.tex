\item Για το όριο $\lim\limits_{x \to +\infty} h(x)$, θέτουμε $t = \dfrac{1}{x}$. Όταν $x \to +\infty$, τότε $t \to 0^+$.
Η συνάρτηση $h(x)$ γίνεται:
\[ h(x) = \frac{1}{t^2} \ln(1+t) - \frac{1}{t} = \frac{\ln(1+t) - t}{t^2} \]
Το όριο είναι της μορφής $\frac{0}{0}$.

\item Για την απόδειξη του ορίου, εφαρμόζουμε τον κανόνα του $\eng{L'Hôpital}$:
\[ \lim_{t \to 0^+} \frac{\ln(1+t) - t}{t^2} \xlongequal{\eng{L'Hôpital}} \lim_{t \to 0^+} \frac{\dfrac{1}{1+t} - 1}{2t} \]
Απλοποιούμε το κλάσμα στον αριθμητή:
\[ \begin{align*} \lim_{t \to 0^+} \frac{\dfrac{1 - (1+t)}{1+t}}{2t} &= \lim_{t \to 0^+} \frac{-t}{2t(1+t)} \\ &= \lim_{t \to 0^+} \frac{-1}{2(1+t)} \\ &= -\frac{1}{2} \end{align*} \]
Άρα, $\lim\limits_{x \to +\infty} h(x) = -\dfrac{1}{2}$.

\item Για το όριο στο $0^+$, έχουμε:
\[ \lim_{x \to 0^+} h(x) = \lim_{x \to 0^+} \left[ x^2 \ln\left(1 + \frac{1}{x}\right) - x \right] \]
Αναλύουμε τον λογάριθμο:
\[ \ln\left(1 + \frac{1}{x}\right) = \ln\left(\frac{x+1}{x}\right) = \ln(x+1) - \ln x \]
Έτσι, το όριο γίνεται:
\[ \lim_{x \to 0^+} \left[ x^2 \ln(x+1) - x^2 \ln x - x \right] \]
Γνωρίζουμε ότι:
\begin{itemize}
    \item $\lim\limits_{x \to 0^+} x^2 \ln(x+1) = 0 \cdot \ln(1) = 0$.
    \item $\lim\limits_{x \to 0^+} x^2 \ln x = 0$ (από το γνωστό όριο $\lim\limits_{u \to 0^+} u^k \ln u = 0$ για $k > 0$).
    \item $\lim\limits_{x \to 0^+} x = 0$.
\end{itemize}
Συνεπώς, $\lim\limits_{x \to 0^+} h(x) = 0 - 0 - 0 = 0$.

\item Θεωρούμε τη συνάρτηση $g(t) = \ln(1+t) - t$ για $t > 0$.

\[ g'(t) = \frac{1}{1+t} - 1 = \frac{1 - (1+t)}{1+t} = \frac{-t}{1+t} \]
Επειδή $t > 0$, έχουμε $g'(t) < 0$, άρα η $g$ είναι \textbf{γνησίως φθίνουσα} στο $(0, +\infty)$.

\[ x^2 \left[ \ln\left(1 + \frac{1}{x}\right) - \frac{1}{x} \right] < 0 \implies x^2 \ln\left(1 + \frac{1}{x}\right) - x < 0 \]
