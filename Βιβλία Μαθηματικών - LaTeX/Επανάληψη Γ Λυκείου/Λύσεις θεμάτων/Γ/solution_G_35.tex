%=========== ΛΥΣΗ - 35ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Παραγωγίζουμε και τα δύο μέλη της σχέσης $e^{f(x)} + f(x) = x + 2$ ως προς $x$:
\[ \left( e^{f(x)} \right)' + f'(x) = 1 \]
\[ e^{f(x)} \cdot f'(x) + f'(x) = 1 \]
\[ f'(x) \left( e^{f(x)} + 1 \right) = 1 \]
\[ f'(x) = \frac{1}{e^{f(x)} + 1} \]
Αφού $e^{f(x)} > 0$ για κάθε $x \in \mathbb{R}$, ισχύει $e^{f(x)} + 1 > 1 > 0$.
Επομένως $f'(x) > 0$ για κάθε $x \in \mathbb{R}$, άρα η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Αφού η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, είναι \textbf{1-1}, συνεπώς αντιστρέφεται.
Για να βρούμε το πεδίο ορισμού της $f^{-1}$, χρειαζόμαστε το σύνολο τιμών της $f$.
Η $f$ είναι συνεχής και γνησίως αύξουσα, και από τη δοσμένη σχέση μπορούμε να ελέγξουμε τα όρια:
\begin{itemize}
    \item Για $x \to +\infty$: Αν $f(x) \to \ell$ (πεπερασμένο), τότε $e^{\ell} + \ell = +\infty$, άτοπο. Άρα $f(x) \to +\infty$.
    \item Για $x \to -\infty$: Αν $f(x) \to m$ (πεπερασμένο), τότε $e^{m} + m = -\infty$, άτοπο. Άρα $f(x) \to -\infty$.
\end{itemize}
Επομένως το σύνολο τιμών της $f$ είναι το $\mathbb{R}$, άρα:
\[ \boldsymbol{A_{f^{-1}} = \mathbb{R}} \]

\item Για να βρούμε τον τύπο της $f^{-1}$, υπολογίζουμε μερικές τιμές της $f$ από τη σχέση $e^{f(x)} + f(x) = x + 2$.
Δοκιμάζουμε ορισμένες τιμές:
\begin{itemize}
    \item Για $f(x) = 0$: $e^0 + 0 = x + 2 \Rightarrow 1 = x + 2 \Rightarrow x = -1$, άρα $f(-1) = 0$.
    \item Για $f(x) = 1$: $e^1 + 1 = x + 2 \Rightarrow x = e - 1$, άρα $f(e-1) = 1$.
\end{itemize}
Για να βρούμε αναλυτική έκφραση του τύπου της $f^{-1}$, λύνουμε $y = f(x)$, δηλαδή γράφουμε $x = f^{-1}(y)$.
Η εξίσωση $e^y + y = x + 2$ δίνει απευθείας:
\[ x = e^y + y - 2 \]
Άρα $f^{-1}(y) = e^y + y - 2$, ή ισοδύναμα:
\[ \boldsymbol{f^{-1}(x) = e^x + x - 2, \quad x \in \mathbb{R}} \]

\item Ζητείται το εμβαδόν που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις ευθείες $x=-1$ και $x=e-1$.
Από το ερώτημα (γ), γνωρίζουμε ότι $f(-1) = 0$ και $f(e-1) = 1$.
Δεδομένου ότι η $f$ είναι γνησίως αύξουσα και $f(-1) = 0$:
\begin{itemize}
    \item Για $x \in [-1, e-1]$: $f(x) \ge f(-1) = 0$, άρα η γραφική παράσταση βρίσκεται πάνω από τον $x'x$.
\end{itemize}
Επομένως:
\[ E = \int_{-1}^{e-1} f(x) \d x \]
Χρησιμοποιούμε τον τύπο για το εμβαδόν μέσω της αντίστροφης συνάρτησης. Η αντίστροφη $f^{-1}(x) = e^x + x - 2$ είναι πιο εύκολα ολοκληρώσιμη.
Εφαρμόζουμε τον τύπο: $\int_a^b f(x)\d x = b\beta - a\alpha - \int_\alpha^\beta f^{-1}(y)\d y$, όπου $f(a) = \alpha$ και $f(b) = \beta$.
Εδώ $a=-1, b=e-1$ και $f(-1) = 0, f(e-1) = 1$:
\begin{align*}
E &= (e-1) \cdot 1 - (-1) \cdot 0 - \int_0^1 f^{-1}(y) \d y \\
&= (e-1) - 0 - \int_0^1 (e^y + y - 2) \d y \\
&= (e-1) - \left[ e^y + \frac{y^2}{2} - 2y \right]_0^1 \\
&= (e-1) - \left[ \left(e^1 + \frac{1}{2} - 2\right) - \left(e^0 + 0 - 0\right) \right] \\
&= (e-1) - \left[ \left(e - \frac{3}{2}\right) - 1 \right] \\
&= (e-1) - \left( e - \frac{5}{2} \right) \\
&= e - 1 - e + \frac{5}{2} = \boldsymbol{\frac{3}{2} \ \text{τ.μ.}}
\end{align*}
\end{erwthma}
\end{thema}
