%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.69 ===========
\begin{thema}{Γ}
Δίνεται η $f(x) = x^3 - 6x^2 + 9x$.
\begin{erwthma}

\item \textbf{Εύρεση εφαπτομένης και σημείου τομής:}
Η παράγωγος είναι $f'(x) = 3x^2 - 12x + 9$.
Υπολογίζουμε τα στοιχεία στο $x_1 = 1$:
$f(1) = 1 - 6 + 9 = 4$. Το σημείο επαφής είναι το $A(1, 4)$.
$f'(1) = 3(1)^2 - 12(1) + 9 = 0$. Η κλίση είναι μηδέν (οριζόντια εφαπτομένη).
Εξίσωση εφαπτομένης $(\varepsilon): y - 4 = 0 \cdot (x - 1) \Rightarrow \mathbf{y = 4}$.

Για να βρούμε τα σημεία τομής με τη γραφική παράσταση, λύνουμε το σύστημα:
\[ f(x) = 4 \Rightarrow x^3 - 6x^2 + 9x - 4 = 0 \]
Γνωρίζουμε εκ των προτέρων ότι το $x=1$ είναι διπλή ρίζα λόγω της εφαπτομένης. Παραγοντοποιούμε με το $(x-1)^2 = x^2 - 2x + 1$:
Εκτελώντας τη διαίρεση ή ομαδοποίηση, προκύπτει:
\[ x^3 - 6x^2 + 9x - 4 = (x-1)^2 \cdot (x-4) = 0 \]
Οι ρίζες είναι $x=1$ (διπλή, το αναμενόμενο σημείο επαφής) και $x=4$. 
Επομένως, όντως η εφαπτομένη \textbf{τέμνει ξανά} την $C_f$ ακριβώς στο σημείο με τετμημένη $\mathbf{x_2 = 4}$.

\item \textbf{Εμβαδόν Χωρίου:}
Το ζητούμενο εμβαδόν περικλείεται μεταξύ της $C_f$, της $(\varepsilon)$ και των κατακόρυφων ορίων διασταύρωσής τους στο $[1, 4]$.
\[ E = \int_1^4 |f(x) - 4|\,\d x = \int_1^4 | (x-1)^2(x-4) |\,\d x \]
Στο διάστημα $[1, 4]$, ο παράγοντας $(x-4) \leq 0$, ενώ το $(x-1)^2 \geq 0$. Συνεπώς, η παράσταση είναι μη θετική (αρνητική), η $\varepsilon$ είναι «πάνω» από την $C_f$. Βγάζουμε την απόλυτη τιμή με αντίθετο πρόσημο:
\begin{align*}
E &= \int_1^4 -(x^3 - 6x^2 + 9x - 4)\,\d x = \left[ -\frac{x^4}{4} + 2x^3 - \frac{9x^2}{2} + 4x \right]_1^4
\end{align*}
Υπολογίζουμε τις τιμές στα άκρα:
- Στο $4$: $-\frac{256}{4} + 2(64) - \frac{9(16)}{2} + 16 = -64 + 128 - 72 + 16 = 8$.
- Στο $1$: $-\frac{1}{4} + 2 - \frac{9}{2} + 4 = 6 - \frac{19}{4} = \frac{24 - 19}{4} = \frac{5}{4}$.
Άρα το εμβαδόν είναι:
\[ E = 8 - \frac{5}{4} = \frac{32 - 5}{4} = \boldsymbol{\frac{27}{4} \ \text{τ.μ.}} \]

\item \textbf{Σημείο Καμπής $K$ και Γεωμετρική Ιδιότητα:}
Υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x) = (3x^2 - 12x + 9)' = 6x - 12 = 6(x-2) \]
- $f''(x) = 0 \Leftrightarrow x = 2$.
- Εκατέρωθεν του $2$, η $f''$ αλλάζει αποδεδειγμένα πρόσημο (αρνητική για $x<2$, θετική για $x>2$). Άρα το σημείο τετμημένης $\mathbf{x_K = 2}$ είναι \textbf{σημείο καμπής}.

Υπολογίζουμε τις αποστάσεις:
- Απόσταση καμπής από επαφή: $x_K - x_1 = 2 - 1 = 1$.
- Συνολική απόσταση τομής: $\Delta x = x_2 - x_1 = 4 - 1 = 3$.
Αντιπαραθέτοντας τα μήκη:
\[ \frac{x_K - x_1}{x_2 - x_1} = \frac{1}{3} \Rightarrow \mathbf{x_K - x_1 = \frac{1}{3}(x_2 - x_1)} \]
Η ιδιότητα επαληθεύτηκε εμφατικά.

\item \textbf{Υπολογισμός Ορίου:}
Έχουμε προσδιορισμένη απροσδιοριστία $\frac{0}{0}$ καθώς το $h \to 0$. Χρησιμοποιούμε τον κανόνα \eng{De L'Hospital} παραγωγίζοντας ως προς $h$:
\begin{align*}
L &= \lim_{h\to 0} \frac{f(1+3h) - f(1-h)}{h} \xlongequal[\text{\eng{DLH}}]{ \frac{0}{0} } \lim_{h\to 0} \frac{ f'(1+3h)\cdot 3 - f'(1-h)\cdot (-1) }{1} \\
&= \lim_{h\to 0} \left[ 3f'(1+3h) + f'(1-h) \right]
\end{align*}
Αφού η $f'$ είναι συνεχής συνάρτηση (ως πολυώνυμο), αντικαθιστούμε απευθείας το $h=0$:
\[ L = 3f'(1) + f'(1) = 4f'(1) \]
Ωστόσο, είχαμε υπολογίσει στο Ερώτημα 1 ότι η εφαπτομένη στο $x=1$ είναι επίπεδη, δηλαδή $f'(1) = 0$. Συνεπώς:
\[ L = 4 \cdot 0 = \boldsymbol{0} \]

\end{erwthma}
\end{thema}
