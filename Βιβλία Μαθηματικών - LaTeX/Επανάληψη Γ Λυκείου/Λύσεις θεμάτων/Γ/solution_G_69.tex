%=========== ΛΥΣΗ - 69ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=3x^2-12x+9
\]
Στο σημείο με τετμημένη $x_1=1$ έχουμε
\[
f(1)=1-6+9=4
\]
και
\[
f'(1)=3-12+9=0
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο $A(1,4)$ είναι
\[
y-4=0(x-1)\Leftrightarrow y=4
\]

Για να βρούμε τα κοινά σημεία της ευθείας $y=4$ με την $C_f$, λύνουμε:
\begin{align*}
f(x)&=4\\
x^3-6x^2+9x&=4\\
x^3-6x^2+9x-4&=0
\end{align*}
Παραγοντοποιούμε:
\[
x^3-6x^2+9x-4=(x-1)^2(x-4)
\]
Άρα
\[
(x-1)^2(x-4)=0\Leftrightarrow x=1 \ \text{ή}\ x=4
\]
Το $x=1$ αντιστοιχεί στο σημείο επαφής, ενώ η εφαπτομένη τέμνει ξανά την καμπύλη στο σημείο με τετμημένη
\[
x_2=4
\]

\item Από το προηγούμενο ερώτημα, η εφαπτομένη είναι η ευθεία
\[
y=4
\]
και τα σημεία τομής έχουν τετμημένες $1$ και $4$.
Για $x\in[1,4]$ έχουμε
\[
f(x)-4=(x-1)^2(x-4)\leq 0
\]
άρα η ευθεία $y=4$ βρίσκεται πάνω από την $C_f$.

Το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\dint_1^4\left[4-f(x)\right]\d x\\
&=\dint_1^4\left(-x^3+6x^2-9x+4\right)\d x\\
&=\left[-\frac{x^4}{4}+2x^3-\frac{9x^2}{2}+4x\right]_1^4\\
&=\left(-64+128-72+16\right)
-\left(-\frac{1}{4}+2-\frac{9}{2}+4\right)\\
&=8-\frac{5}{4}\\
&=\frac{27}{4}
\end{align*}

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\[
f''(x)=6x-12=6(x-2)
\]
Έχουμε
\[
f''(x)=0\Leftrightarrow x=2
\]
Επίσης, η $f''$ είναι αρνητική για $x<2$ και θετική για $x>2$, άρα αλλάζει πρόσημο στο $x=2$.
Επομένως η $C_f$ έχει σημείο καμπής το
\[
K(2,f(2))
\]
Επειδή
\[
f(2)=8-24+18=2
\]
είναι
\[
K(2,2)
\]

Τώρα
\[
x_K-x_1=2-1=1
\]
και
\[
x_2-x_1=4-1=3
\]
Άρα
\[
x_K-x_1=\frac{1}{3}(x_2-x_1)
\]

\item Θέλουμε το όριο
\[
L=\lim\limits_{h\to 0}\frac{f(1+3h)-f(1-h)}{h}
\]
Για $h\to 0$ έχουμε απροσδιόριστη μορφή $\dfrac{0}{0}$.
Εφαρμόζοντας τον κανόνα \eng{De L' Hospital} ως προς $h$, παίρνουμε:
\begin{align*}
L
&=\lim\limits_{h\to 0}\frac{3f'(1+3h)+f'(1-h)}{1}\\
&=3f'(1)+f'(1)\\
&=4f'(1)
\end{align*}
Από το πρώτο ερώτημα γνωρίζουμε ότι $f'(1)=0$.
Επομένως
\[
L=0
\]
\end{erwthma}
\end{thema}
