%=========== ΛΥΣΗ - 86ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη γραφική παράσταση παρατηρούμε ότι η $f$ είναι:
\begin{itemize}
\item γνησίως αύξουσα στο $[0,2]$
\item γνησίως φθίνουσα στο $[2,6]$
\end{itemize}
Επομένως η $f$ παρουσιάζει τοπικό μέγιστο στο $x=2$, με τιμή
\[
f(2)=3
\]
Στο κλειστό διάστημα $[0,6]$ το ολικό μέγιστο είναι
\[
f(2)=3
\]
και το ολικό ελάχιστο είναι
\[
f(6)=-2
\]
Άρα το σύνολο τιμών της $f$ είναι
\[
f([0,6])=[-2,3]
\]

Για την εξίσωση
\[
f(x)=1
\]
φέρνουμε νοητά την οριζόντια ευθεία $y=1$.
Από το σχήμα, η ευθεία αυτή τέμνει την $C_f$ δύο φορές: μία στο διάστημα $(0,2)$ και μία στο διάστημα $(2,6)$.
Άρα η εξίσωση $f(x)=1$ έχει δύο ρίζες.

\item Στο διάστημα $[0,4]$ η $f$ είναι συνεχής και στο $(0,4)$ είναι παραγωγίσιμη, αφού είναι δύο φορές παραγωγίσιμη στο $[0,6]$.
Επιπλέον, από το σχήμα έχουμε
\[
f(0)=0
\quad \text{και}\quad
f(4)=0
\]
Άρα
\[
f(0)=f(4)
\]
Επομένως ικανοποιούνται οι προϋποθέσεις του Θεωρήματος \eng{Rolle} στο $[0,4]$.
Συνεπώς υπάρχει τουλάχιστον ένα
\[
\xi\in(0,4)
\]
τέτοιο ώστε
\[
f'(\xi)=0
\]
Γεωμετρικά, αυτό σημαίνει ότι υπάρχει σημείο της $C_f$ στο οποίο η εφαπτομένη είναι οριζόντια.
Από το σχήμα αυτό συμβαίνει στο σημείο
\[
(2,3)
\]

\item Στο διάστημα $[0,4]$ η $C_f$ βρίσκεται πάνω από τον άξονα $x'x$.
Το ολοκλήρωμα
\[
\dint_0^4 f(x)\d x
\]
εκφράζει το εμβαδόν του χωρίου κάτω από την $C_f$ και πάνω από τον άξονα $x'x$.

Από το σχήμα, η καμπύλη βρίσκεται πάνω από τα ευθύγραμμα τμήματα που ενώνουν τα σημεία
\[
(0,0),\ (2,3),\ (4,0)
\]
Άρα το ζητούμενο εμβαδόν είναι μεγαλύτερο από το εμβαδόν του τριγώνου με βάση $4$ και ύψος $3$.
Το εμβαδόν αυτού του τριγώνου είναι
\[
E_{\triangle}=\frac{1}{2}\cdot 4\cdot 3=6
\]
Επομένως
\[
\dint_0^4 f(x)\d x>6
\]

\item Η εφαπτομένη της $C_f$ στο $O(0,0)$ έχει εξίσωση
\[
y=2x
\]
άρα
\[
f(0)=0
\quad \text{και}\quad
f'(0)=2
\]
Θέλουμε το όριο
\[
L=\lim\limits_{x\to 0}\frac{f(x)-\hm x}{x^2+x}
\]
Επειδή η $f$ ορίζεται στο $[0,6]$, το όριο στο άκρο $0$ νοείται ως δεξιό.
Γράφουμε:
\begin{align*}
L
&=\lim\limits_{x\to 0^+}\frac{f(x)-\hm x}{x(x+1)}\\
&=\lim\limits_{x\to 0^+}\frac{\frac{f(x)-f(0)}{x}-\frac{\hm x}{x}}{x+1}
\end{align*}
Επειδή
\[
\lim\limits_{x\to 0^+}\frac{f(x)-f(0)}{x}=f'(0)=2
\]
και
\[
\lim\limits_{x\to 0}\frac{\hm x}{x}=1
\]
παίρνουμε
\[
L=\frac{2-1}{1}=1
\]
\end{erwthma}
\end{thema}
