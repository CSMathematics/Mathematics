\begin{erwthma}
\item Από τη γραφική παράσταση της συνάρτησης $f$ στο διάστημα $[0, 6]$, παρατηρούμε ότι:
\begin{itemize}
    \item Για $x \in (0, 2)$, η $C_f$ είναι αυξανόμενη, άρα η $f$ είναι \textbf{γνησίως αύξουσα}.
    \item Για $x \in (2, 6)$, η $C_f$ είναι φθίνουσα, άρα η $f$ είναι \textbf{γνησίως φθίνουσα}.
\end{itemize}
Επομένως, η συνάρτηση παρουσιάζει τοπικό μέγιστο στο $x=2$ με τιμή $f(2)=3$. 
\begin{itemize}
    \item Το \textbf{ολικό μέγιστο} της $f$ στο $[0, 6]$ είναι το $f(2)=3$.
    \item Το \textbf{ολικό ελάχιστο} της $f$ στο $[0, 6]$ είναι η τιμή στο ακροτελές σημείο $x=6$, δηλαδή $f(6)=-2$.
\end{itemize}

\item Η συνάρτηση $f$ είναι δύο φορές παραγωγίσιμη στο $[0, 6]$, άρα είναι \textbf{συνεχώς} και \textbf{παραγωγήσιμη} στο διάστημα $[0, 4]$. 
Επιπλέον, από τα δεδομένα έχουμε $f(0) = 0$ και $f(4) = 0$.
Σύμφωνα με το θεώρημα \eng{Rolle}, υπάρχει τουλάχιστον ένα $\xi \in (0, 4)$ ώστε 
\[ f'(\xi) = 0 \]
Από τη γραφική παράσταση, βλέπουμε ότι υπάρχει μόνο ένα τέτοιο σημείο, το οποίο είναι το $\xi = 2$ (το σημείο του μέγιστου).

\item Το ολοκλήρωμα $\int_0^6 f(x) \d x$ αντιπροσωπεύει την代 signed area (υπογεγραμμένο εμβαδόν) μεταξύ της $C_f$ και του άξονα $x'x$. 
Διαχωρίζουμε το ολοκλήρωμα στα δύο σκιασμένα χωρία:
\[ \int_0^6 f(x) \d x = \int_0^4 f(x) \d x + \int_4^6 f(x) \d x \]
Έστω $S_1$ το εμβαδόν του πρώτου σκιασμένου χωρίου (πράσινο) και $S_2$ το εμβαδόν του δεύτερου (κόκκινο). Τότε:
\begin{itemize}
    \item $\int_0^4 f(x) \d x = S_1 > 0$
    \item $\int_4^6 f(x) \d x = -S_2 < 0$
\end{itemize}
Από τη γραφική παράσταση παρατηρούμε ότι το εμβαδόν $S_1$ είναιconsiderably μεγαλύτερο από το εμβαδόν $S_2$, δηλαδή $S_1 > S_2$. 
Συνεπώς:
\[ \int_0^6 f(x) \d x = S_1 - S_2 > 0 \]

\item Η τεταγμένη του σημείου $M$ είναι $y(t) = f(x(t))$. 
Ο ρυθμός μεταβολής της τεταγμένης $y'(t)$ δίνεται από τον κανόνα της αλυσίδας:
\[ y'(t) = \frac{d}{dt} f(x(t)) = f'(x(t)) \cdot x'(t) \]
Τη στιγμή που $x=5$, έχουμε από τα δεδομένα:
\begin{itemize}
    \item $x'(t) = 1 \text{ μον}/\sec$
    \item $f'(5) \approx -1$
\end{itemize}
Αντικαθιστώντας τις τιμές, προκύπτει:
\[ y'(t) \approx (-1) \cdot 1 = -1 \text{ μον}/\sec \]
Άρα, η τεταγμένη $y$ μεταβάλλεται με ρυθμό περίπου $-1 \text{ μον}/\sec$.
\end{erwthma}
\end{thema}
