%=========== ΛΥΣΗ - 3ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Γράφουμε
\[
f(x)=x^{-ax}=e^{-ax\ln x}, \quad x>0.
\]
Τότε
\[
f'(x)=e^{-ax\ln x}\cdot(-a(\ln x+1))=-ax^{-ax}(\ln x+1).
\]
Για $x=\dfrac{1}{e}$:
\[
f'\left(\frac{1}{e}\right)
=
-a\left(\frac{1}{e}\right)^{-a/e}\left(\ln \frac{1}{e}+1\right)=0.
\]
Για τη συγκεκριμένη συνάρτηση προκύπτει
\[
a=1.
\]

\item Για $a=1$ έχουμε
\[
f(x)=x^{-x}
\]
και
\[
f'(x)=-x^{-x}(\ln x+1).
\]
Επειδή $x^{-x}>0$ για κάθε $x>0$, το πρόσημο της $f'(x)$ είναι αντίθετο από το πρόσημο του $\ln x+1$:
\begin{itemize}
\item $f'(x)=0 \Leftrightarrow \ln x=-1 \Leftrightarrow x=\dfrac{1}{e}$
\item $f'(x)>0 \Leftrightarrow 0<x<\dfrac{1}{e}$
\item $f'(x)<0 \Leftrightarrow x>\dfrac{1}{e}$
\end{itemize}
Άρα η $f$ παρουσιάζει ολικό μέγιστο στη θέση $x=\dfrac{1}{e}$ και
\[
f\left(\frac{1}{e}\right)=\left(\frac{1}{e}\right)^{-1/e}=e^{1/e}.
\]

Θέτουμε
\[
h(x)=ex\ln x+1.
\]
Τότε
\[
h'(x)=e(\ln x+1).
\]
Άρα:
\begin{itemize}
\item $h'(x)=0 \Leftrightarrow x=\dfrac{1}{e}$
\item $h'(x)<0 \Leftrightarrow 0<x<\dfrac{1}{e}$
\item $h'(x)>0 \Leftrightarrow x>\dfrac{1}{e}$
\end{itemize}
οπότε η $h$ παρουσιάζει ολικό ελάχιστο στο $x=\dfrac{1}{e}$. Επειδή
\[
h\left(\frac{1}{e}\right)=e\cdot \frac{1}{e}\cdot \ln \frac{1}{e}+1=0,
\]
έχουμε
\[
h(x)\ge 0,\quad x>0
\]
δηλαδή
\[
ex\ln x+1\ge 0,\quad x>0.
\]

\item Εφόσον $f(x)=x^{-x}$,
\[
f(x)\cdot \ln x=\frac{\ln x}{x^x}.
\]
Καθώς $x\to +\infty$, ο παρονομαστής αυξάνεται ταχύτερα από τον αριθμητή, οπότε
\[
\lim_{x\to +\infty}\left(f(x)\cdot \ln x\right)=0.
\]

\item Από τον κανόνα της αλυσίδας,
\[
\frac{\d y}{\d t}=f'(x)\frac{\d x}{\d t}.
\]
Για $x=e$:
\[
f'(e)=-e^{-e}(\ln e+1)=-2e^{-e}.
\]
Αφού $\dfrac{\d x}{\d t}=3\mu/s$, παίρνουμε
\[
\frac{\d y}{\d t}=-2e^{-e}\cdot 3=-6e^{-e}\mu/s.
\]
\end{erwthma}
\end{thema}
