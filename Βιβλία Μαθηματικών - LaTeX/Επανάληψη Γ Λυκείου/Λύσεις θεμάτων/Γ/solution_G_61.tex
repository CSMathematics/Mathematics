%=========== ΛΥΣΗ - 61ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $t\geq 0$ ο χρόνος σε δευτερόλεπτα.
Το σημείο $M$ κινείται στον θετικό ημιάξονα $Ox$ με σταθερή ταχύτητα $3\si{m/s}$ και ξεκινά από το $O(0,0)$.
Άρα
\[M(t)=(3t,0)\]
Το σημείο $N$ κινείται στον θετικό ημιάξονα $Oy$, ξεκινά από το σημείο $(0,10)$ και απομακρύνεται με σταθερή ταχύτητα $4\si{m/s}$.
Άρα
\[N(t)=(0,10+4t)\]

\item Η απόσταση των σημείων $M(t)$ και $N(t)$ είναι
\begin{align*}
d(t)&=\sqrt{(0-3t)^2+(10+4t-0)^2}\\
&=\sqrt{9t^2+(10+4t)^2}\\
&=\sqrt{9t^2+100+80t+16t^2}\\
&=\sqrt{25t^2+80t+100}
\end{align*}
Επομένως
\[d(t)=\sqrt{25t^2+80t+100}\ ,\ t\geq 0\]

\item Για $t\geq 0$ έχουμε
\[d(t)=(25t^2+80t+100)^{\frac{1}{2}}\]
οπότε
\begin{align*}
d'(t)&=\frac{1}{2}(25t^2+80t+100)^{-\frac{1}{2}}(50t+80)\\
&=\frac{25t+40}{\sqrt{25t^2+80t+100}}
\end{align*}
Για $t=2$ παίρνουμε
\begin{align*}
d'(2)&=\frac{25\cdot2+40}{\sqrt{25\cdot2^2+80\cdot2+100}}\\
&=\frac{90}{\sqrt{360}}\\
&=\frac{90}{6\sqrt{10}}\\
&=\frac{15}{\sqrt{10}}\\
&=\frac{3\sqrt{10}}{2}\si{m/s}
\end{align*}
Άρα τη χρονική στιγμή $t=2\si{s}$ η απόσταση αυξάνεται με ρυθμό
\[\frac{3\sqrt{10}}{2}\si{m/s}\]

\item Για $t\geq 0$ είναι
\[25t+40>0\]
και
\[\sqrt{25t^2+80t+100}>0\]
άρα
\[d'(t)>0\ ,\ t\geq 0\]
Επομένως η $d$ είναι \textbf{γνησίως αύξουσα} στο $[0,+\infty)$.
Άρα η ελάχιστη απόσταση επιτυγχάνεται στην αρχική χρονική στιγμή
\[t=0\]
και είναι
\[d(0)=\sqrt{100}=10\si{m}\]
\end{erwthma}
\end{thema}
