\item Για $x > 0$, η συνάρτηση $g(x) = \dfrac{e^x}{x}$ είναι συνεχής και παραγώγιμη στο $(0, +\infty)$. Η παράγωγός της είναι:
\[
g'(x) = \frac{(e^x)' \cdot x - e^x \cdot (x)'}{x^2} = \frac{e^x \cdot x - e^x \cdot 1}{x^2} = \frac{e^x(x-1)}{x^2}
\]
Εξετάζουμε το πρόσημο της $g'(x)$:
\begin{itemize}
    \item $g'(x) = 0 \Leftrightarrow x-1 = 0 \Leftrightarrow x = 1$
    \item $g'(x) > 0 \Leftrightarrow x-1 > 0 \Leftrightarrow x > 1$
    \item $g'(x) < 0 \Leftrightarrow x-1 < 0 \Leftrightarrow 0 < x < 1$
\end{itemize}
Συνεπώς, η συνάρτηση $g$ είναι \textbf{γνησίως φθίνουσα} στο $(0, 1]$ και \textbf{γνησίως αύξουσα} στο $[1, +\infty)$.
Άρα, η συνάρτηση $g$ παρουσιάζει \textbf{ολίστατο} στο $x=1$ και η ελάχιστη τιμή της είναι:
\[
g(1) = \frac{e^1}{1} = e
\]

\item Η εξίσωση $e^x = ax$ για $x > 0$ είναι ισοδύναμη με την εξίσωση:
\[
\frac{e^x}{x} = a \Leftrightarrow g(x) = a
\]
Γνωρίζουμε από το προηγούμενο ερώτημα ότι η $g$ έχει ελάχιστο $g(1) = e$.
Επιπλέον, υπολογίζουμε τα όρια της $g$ στα άκρα του ορισμού της:
\begin{align*}
\lim_{x \to 0^+} g(x) &= \lim_{x \to 0^+} \frac{e^x}{x} = \frac{1}{0^+} = +\infty \\
\lim_{x \to +\infty} g(x) &= \lim_{x \to +\infty} \frac{e^x}{x} = +\infty \quad (\text{αφού } e^x \text{ αυξάνει ταχύτερα από το } x)
\end{align*}
Δεδομένου ότι $a > e$, η τιμή $a$ ανήκει στο λίμνο της εικόνας της $g$ σε δύο διαφορετικά διαστήματα:
\begin{itemize}
    \item Στο διάστημα $(0, 1)$, η $g$ είναι συνεχής και $\text{ελάχιστη τιμή } e < a < \lim_{x \to 0^+} g(x) = +\infty$. Άρα, από το \eng{Bolzano}, υπάρχει μία ρίζα $x_1 \in (0, 1)$.
    \item Στο διάστημα $(1, +\infty)$, η $g$ είναι συνεχής και $e = g(1) < a < \lim_{x \to +\infty} g(x) = +\infty$. Άρα, από το \eng{Bolzano}, υπάρχει μία ρίζα $x_2 \in (1, +\infty)$.
\end{itemize}
Επειδή η $g$ είναι γνησίως μονότονη σε κάθε ένα από αυτά τα διαστήματα, οι ρίζες είναι μοναδικές. Άρα η ευθεία $y=ax$ τέμνει την καμπύλη $y=e^x$ σε ακριβώς δύο σημεία στο πρώτο τεταρτημόριο.

\item Εξετάζουμε τον αριθμό των λύσεων της εξίσωσης $e^x = ax$:
\begin{itemize}
    \item Για $a = e$: Η εξίσωση $\frac{e^x}{x} = e$ έχει μία και μοναδική λύση $x=1$ (το σημείο του ολίστατου).
    \item Για $0 < a < e$: Εφόσον το ελάχιστο της $g$ είναι $e$, η εξίσωση $g(x) = a$ δεν έχει λύσεις για $x > 0$. Για $x \le 0$, έχουμε $e^x > 0$ και $ax \le 0$, οπότε η εξίσωση δεν έχει λύσεις. Άρα, η εξίσωση έχει \textbf{μηδέν λύσεις}.
    \item Για $a \le 0$:
    \begin{itemize}
        \item Αν $a = 0$, η εξίσωση $e^x = 0$ είναι αδύνατη.
        \item Αν $a < 0$, θεωρούμε τη συνάρτηση $h(x) = e^x - ax$. Η παράγωγός της είναι $h'(x) = e^x - a$. Επειδή $a < 0$, τότε $h'(x) = e^x + |a| > 0$ για κάθε $x \in \mathbb{R}$, άρα η $h$ είναι \textbf{γνησίως αύξουσα}.
        Υπολογίζουμε τα όρια:
        \[
        \lim_{x \to -\infty} (e^x - ax) = 0 - (-\infty \cdot \text{αρνητικό}) = -\infty, \quad h(0) = e^0 - 0 = 1
        \]
        Από το \eng{Bolzano}, υπάρχει μία μοναδική λύση στο $(-\infty, 0)$.
    \end{itemize}
\end{itemize}

\item Έστω $x_1 \in (0, 1)$ και $x_2 \in (1, +\infty)$ οι δύο ρίζες της εξίσωσης $g(x) = a$.
\begin{itemize}
    \item Για τη ρίζα $x_1$: Έχουμε $a = \dfrac{e^{x_1}}{x_1}$. Καθώς $a \to +\infty$, και εφόσον $e^{x_1} \in (1, e)$, πρέπει αναγκαστικά $x_1 \to 0^+$.
    \item Για τη ρίζα $x_2$: Έχουμε $a = \dfrac{e^{x_2}}{x_2}$. Καθώς $a \to +\infty$, και εφόσον η $g$ είναι γνησίως αύξουσα στο $[1, +\infty)$ με $\lim_{x \to +\infty} g(x) = +\infty$, πρέπει αναγκαστικά $x_2 \to +\infty$.
\end{itemize}
Άρα, η οριακή θέση των ριζών είναι $x_1 \to 0$ και $x_2 \to +\infty$.
\end{erwthma}
