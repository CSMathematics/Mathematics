%=========== ΛΥΣΗ - 66ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $x>0$ έχουμε
\[g(x)=\frac{e^x}{x}\]
Η $g$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[g'(x)=\frac{(e^x)'x-e^x(x)'}{x^2}=\frac{xe^x-e^x}{x^2}=\frac{e^x(x-1)}{x^2}\]
Επειδή $e^x>0$ και $x^2>0$ για κάθε $x>0$, το πρόσημο της $g'$ καθορίζεται από το $x-1$.
Έχουμε:
\begin{itemize}
\item $g'(x)=0\Leftrightarrow x=1$
\item $g'(x)<0\Leftrightarrow 0<x<1$
\item $g'(x)>0\Leftrightarrow x>1$
\end{itemize}
Άρα η $g$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.
Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με ελάχιστη τιμή
\[g(1)=\frac{e^1}{1}=e\]

\item Για $x>0$ η εξίσωση
\[e^x=ax\]
είναι ισοδύναμη με
\[\frac{e^x}{x}=a\Leftrightarrow g(x)=a\]
Από το προηγούμενο ερώτημα γνωρίζουμε ότι η $g$ έχει ελάχιστη τιμή το $e$ στο $x=1$.
Επίσης
\[\lim\limits_{x\to 0^+}g(x)=\lim\limits_{x\to 0^+}\frac{e^x}{x}=+\infty\]
και
\[\lim\limits_{x\to+\infty}g(x)=\lim\limits_{x\to+\infty}\frac{e^x}{x}=+\infty\]

Έστω $a>e$.
Στο διάστημα $(0,1]$ η $g$ είναι συνεχής και γνησίως φθίνουσα, με
\[\lim\limits_{x\to 0^+}g(x)=+\infty\quad \text{και}\quad g(1)=e<a\]
Άρα υπάρχει μοναδικό $x_1\in(0,1)$ τέτοιο ώστε
\[g(x_1)=a\]

Στο διάστημα $[1,+\infty)$ η $g$ είναι συνεχής και γνησίως αύξουσα, με
\[g(1)=e<a\quad \text{και}\quad \lim\limits_{x\to+\infty}g(x)=+\infty\]
Άρα υπάρχει μοναδικό $x_2\in(1,+\infty)$ τέτοιο ώστε
\[g(x_2)=a\]

Επομένως, για $a>e$, η εξίσωση $e^x=ax$ έχει ακριβώς δύο λύσεις στο $(0,+\infty)$.
Άρα η ευθεία $y=ax$ τέμνει την καμπύλη $y=e^x$ σε ακριβώς δύο σημεία στο πρώτο τεταρτημόριο.

\item Για $x>0$ η εξίσωση $e^x=ax$ είναι ισοδύναμη με $g(x)=a$.
Επειδή, από το πρώτο ερώτημα,
\[g(x)\geq e\ ,\quad \text{για κάθε }x>0\]
έχουμε:
\begin{itemize}
\item Αν $a=e$, τότε η εξίσωση $g(x)=e$ έχει μοναδική λύση το $x=1$.
\item Αν $a<e$, τότε η εξίσωση $g(x)=a$ δεν έχει καμία λύση στο $(0,+\infty)$.
\end{itemize}
Άρα, για $a\leq e$, η εξίσωση έχει μία λύση στο $(0,+\infty)$ όταν $a=e$, ενώ δεν έχει καμία λύση όταν $a<e$.

\item Έχουμε
\[g'(x)=\frac{e^x(x-1)}{x^2}\]
και
\begin{align*}
g''(x)
&=\left(e^x\cdot\frac{x-1}{x^2}\right)'\\
&=e^x\cdot\frac{x-1}{x^2}+e^x\left(\frac{x-1}{x^2}\right)'\\
&=e^x\left(\frac{x-1}{x^2}+\frac{2-x}{x^3}\right)\\
&=e^x\cdot\frac{x(x-1)+2-x}{x^3}\\
&=e^x\cdot\frac{x^2-2x+2}{x^3}
\end{align*}
Άρα
\[g''(1)=e\]
Επειδή $g(1)=e$ και $g'(1)=0$, έχουμε απροσδιόριστη μορφή $\dfrac{0}{0}$.
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De L' Hospital}, παίρνουμε
\begin{align*}
\lim\limits_{x\to 1}\frac{g(x)-e}{(x-1)^2}
&=\lim\limits_{x\to 1}\frac{g'(x)}{2(x-1)}\\
&=\lim\limits_{x\to 1}\frac{g''(x)}{2}\\
&=\frac{g''(1)}{2}\\
&=\frac{e}{2}
\end{align*}

\item Από το προηγούμενο ερώτημα έχουμε
\[g''(x)=e^x\cdot\frac{x^2-2x+2}{x^3}\]
Για κάθε $x>0$ ισχύει
\[e^x>0,\quad x^3>0\quad \text{και}\quad x^2-2x+2=(x-1)^2+1>0\]
Άρα
\[g''(x)>0\ ,\quad \text{για κάθε }x>0\]
Επομένως η $g$ είναι κυρτή στο $(0,+\infty)$.

Έστω $1<\alpha<\beta$.
Η $g$ είναι συνεχής στο $[\alpha,\beta]$ και παραγωγίσιμη στο $(\alpha,\beta)$.
Σύμφωνα με το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(\alpha,\beta)$ τέτοιο ώστε
\[\frac{g(\beta)-g(\alpha)}{\beta-\alpha}=g'(\xi)\]
Επειδή $g''(x)>0$ στο $(0,+\infty)$, η $g'$ είναι γνησίως αύξουσα στο $(0,+\infty)$.
Άρα, αφού
\[\alpha<\xi<\beta\]
παίρνουμε
\[g'(\alpha)<g'(\xi)<g'(\beta)\]
δηλαδή
\[g'(\alpha)<\frac{g(\beta)-g(\alpha)}{\beta-\alpha}<g'(\beta)\]
\end{erwthma}
\end{thema}
