\begin{thema}{Γ}

\begin{center}

  \foreach \x in {-2,-1,1,2,3,4}
    \draw (\x,0.1) -- (\x,-0.1) node[below,font=\footnotesize] {$\x$};
  \foreach \y in {-1,1,2,3}
    \draw (0.1,\y) -- (-0.1,\y) node[left,font=\footnotesize] {$\y$};

  \draw[very thick, maincolor, smooth, tension=0.6]
    plot coordinates {(-2,0) (-1,1.6) (0,2.6) (0.5,2.9) (1,3) (1.5,2.4) (2,1) (2.5,-0.4) (3,-1) (3.5,-0.2) (4,1)};

\begin{erwthma}
\item Από το σχήμα παρατηρούμε ότι η $f$ είναι:
\[
\text{γνησίως αύξουσα στο }[-2,1],
\]
\[
\text{γνησίως φθίνουσα στο }[1,3],
\]
και
\[
\text{γνησίως αύξουσα στο }[3,4].
\]

Άρα η $f$ παρουσιάζει τοπικό μέγιστο στο $x=1$, με
\[
f(1)=3,
\]
και τοπικό ελάχιστο στο $x=3$, με
\[
f(3)=-1.
\]

Για την εξίσωση $f(x)=0$, από το σχήμα έχουμε αρχικά
\[
f(-2)=0,
\]
άρα το $x=-2$ είναι μία ρίζα.

Στο διάστημα $(1,3)$ η $f$ είναι συνεχής και
\[
f(1)=3>0,\qquad f(3)=-1<0.
\]
Άρα, από το Θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον μία ρίζα στο $(1,3)$. Επειδή η $f$ είναι γνησίως φθίνουσα στο $[1,3]$, η ρίζα αυτή είναι μοναδική.

Στο διάστημα $(3,4)$ η $f$ είναι συνεχής και
\[
f(3)=-1<0,\qquad f(4)=1>0.
\]
Άρα, από το Θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον μία ρίζα στο $(3,4)$. Επειδή η $f$ είναι γνησίως αύξουσα στο $[3,4]$, η ρίζα αυτή είναι μοναδική.

Στο διάστημα $(-2,1]$ δεν υπάρχει άλλη ρίζα, αφού η $f$ είναι γνησίως αύξουσα και
\[
f(x)>f(-2)=0,\quad x\in(-2,1].
\]

Επομένως, η εξίσωση $f(x)=0$ έχει ακριβώς τρεις ρίζες στο πεδίο ορισμού της.

\item Η $f$ είναι παραγωγίσιμη στο $[-2,4]$, άρα είναι συνεχής στα κλειστά διαστήματα $[-2,1]$, $[1,3]$ και παραγωγίσιμη στα αντίστοιχα εσωτερικά τους.

Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[-2,1]$. Υπάρχει $\xi_1\in(-2,1)$ τέτοιο ώστε
\[
f'(\xi_1)=\frac{f(1)-f(-2)}{1-(-2)}
=\frac{3-0}{3}=1.
\]

Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[1,3]$. Υπάρχει $\xi_2\in(1,3)$ τέτοιο ώστε
\[
f'(\xi_2)=\frac{f(3)-f(1)}{3-1}
=\frac{-1-3}{2}=-2.
\]

Άρα
\[
2f'(\xi_1)+f'(\xi_2)=2\cdot 1+(-2)=0.
\]

\item Επειδή η $f$ είναι παραγωγίσιμη και παρουσιάζει τοπικό μέγιστο στο εσωτερικό σημείο $x=1$, από το Θεώρημα \eng{Fermat} έχουμε
\[
f'(1)=0.
\]

Επίσης, επειδή $f(1)=3$, έχουμε
\[
\lim_{x\to 1}\bigl(f(x)-3\bigr)=0.
\]

Για $x\neq 1$ γράφουμε
\[
\frac{\hm(f(x)-3)}{x^2-1}
=
\frac{\hm(f(x)-3)}{f(x)-3}\cdot
\frac{f(x)-3}{x-1}\cdot
\frac{1}{x+1}.
\]

Επομένως,
\[
\lim_{x\to 1}\frac{\hm(f(x)-3)}{f(x)-3}=1,
\]
\[
\lim_{x\to 1}\frac{f(x)-3}{x-1}=f'(1)=0
\]
και
\[
\lim_{x\to 1}\frac{1}{x+1}=\frac{1}{2}.
\]

Άρα
\[
\lim_{x\to 1}\frac{\hm(f(x)-3)}{x^2-1}
=1\cdot 0\cdot \frac{1}{2}=0.
\]

\item Το ευθύγραμμο τμήμα που ενώνει τα σημεία $(-2,0)$ και $(1,3)$ σχηματίζει με τον άξονα $x'x$ ορθογώνιο τρίγωνο με βάση
\[
1-(-2)=3
\]
και ύψος
\[
3.
\]
Άρα το εμβαδόν του τριγώνου είναι
\[
E=\frac{1}{2}\cdot 3\cdot 3=\frac{9}{2}.
\]

Από το σχήμα, στο διάστημα $[-2,1]$ η γραφική παράσταση της $f$ βρίσκεται πάνω από το ευθύγραμμο τμήμα που ενώνει τα σημεία $(-2,0)$ και $(1,3)$, χωρίς να ταυτίζεται με αυτό στο εσωτερικό του διαστήματος.

Επομένως, το εμβαδόν κάτω από την $C_f$ στο διάστημα $[-2,1]$ είναι μεγαλύτερο από το εμβαδόν του παραπάνω τριγώνου. Δηλαδή
\[
\int_{-2}^1 f(x)\,\d x>\frac{9}{2}.
\]
\end{erwthma}
\end{thema}
