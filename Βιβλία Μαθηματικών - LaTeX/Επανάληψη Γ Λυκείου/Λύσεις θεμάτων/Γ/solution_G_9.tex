%=========== ΛΥΣΗ - 9ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f(x)=e^x-\dfrac{x^2}{2}-x-1$ είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}$ με:
\[f'(x)=\left(e^x-\dfrac{x^2}{2}-x-1\right)=e^x-x-1\ \ \text{και}\]
\[f''(x)=\left(e^x-x-1\right)=e^x-1\]
Βρίσκουμε ρίζες και πρόσημα της $f''$:
\begin{itemize}
\item $f''(x)=0\Leftrightarrow e^x=1\Leftrightarrow x=0$
\item $f''(x)>0\Leftrightarrow e^x>1\Leftrightarrow x>0$
\item $f''(x)<0\Leftrightarrow e^x<1\Leftrightarrow x<0$
\end{itemize}

Η $f$ είναι \textbf{κοίλη} στο $(-\infty,0]$ και \textbf{κυρτή} στο $[0,+\infty)$. Επειδή η $f''$ μηδενίζεται και αλλάζει πρόσημο στο $0$, το σημείο $A(0,f(0))=A(0,0)$ είναι \textbf{σημείο καμπής} της $C_f$.

\item Το πρόσημο της $f''$ μας δίνει τη μονοτονία της $f'$. Επειδή $f''(x)<0$ για $x<0$, η $f'$ είναι γνησίως φθίνουσα στο $(-\infty,0]$. Επειδή $f''(x)>0$ για $x>0$, η $f'$ είναι γνησίως αύξουσα στο $[0,+\infty)$.
Άρα η $f'$ παρουσιάζει \textbf{ολικό ελάχιστο} στο $x=0$, με τιμή:
\[f'(0)=e^0-0-1=0\]
\begin{center}

Επομένως, για κάθε $x\in\mathbb{R}$ ισχύει $f'(x)\geq f'(0)\Leftrightarrow f'(x)\geq 0$. Η ισότητα $f'(x)=0$ ισχύει \textbf{μόνο} για $x=0$.

\item Εφόσον $f'(x)\geq 0$ για κάθε $x\in\mathbb{R}$ και η $f'$ μηδενίζεται μόνο σε ένα μεμονωμένο σημείο (στο $x=0$), η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.
Για κάθε $x\geq 0$, επειδή η $f$ είναι γνησίως αύξουσα, ισχύει:
\[x\geq 0 \Rightarrow f(x)\geq f(0)\]
Υπολογίζουμε $f(0)=e^0-0-0-1=0$. Επομένως, για $x\geq 0$:
\[e^x-\frac{x^2}{2}-x-1\geq 0 \Leftrightarrow e^x\geq \frac{x^2}{2}+x+1\]
(Σημείωση: Αν $x<0$, τότε $f(x)<f(0)\Rightarrow f(x)<0$, οπότε η ανισότητα δεν ισχύει στα αρνητικά).

\item Από το προηγούμενο ερώτημα, για κάθε $x\in[0,1]$ ισχύει η ανισότητα:
\[e^x\geq \frac{x^2}{2}+x+1\]
Επειδή η ισότητα ισχύει μόνο για $x=0$, οι δύο συναρτήσεις δεν ταυτίζονται στο $[0,1]$. Επομένως, ολοκληρώνοντας και τα δύο μέλη, η ανισότητα γίνεται γνήσια:
\[\int_0^1 e^x\,\d x > \int_0^1 \left(\frac{x^2}{2}+x+1\right)\,\d x\]
Υπολογίζουμε το δεύτερο ολοκλήρωμα:
\[\int_0^1 \left(\frac{x^2}{2}+x+1\right)\,\d x = \left[\frac{x^3}{6}+\frac{x^2}{2}+x\right]_0^1 = \left(\frac{1}{6}+\frac{1}{2}+1\right) - 0 = \frac{1+3+6}{6} = \frac{10}{6} = \frac{5}{3}\]
Άρα αποδεικνύεται ότι:
\[\int_0^1 e^x\,\d x > \frac{5}{3}\]
\end{erwthma}
\end{thema}
