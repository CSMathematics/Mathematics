\begin{thema}{Γ}
\begin{erwthma}
\item Για $x>0$ έχουμε
\[f'(x)=\frac{|x|}{x}e^{|x|}=\frac{x}{x}e^x=e^x\]
Άρα, για κάθε $x>0$,
\begin{align*}
f(x)-f(0)&=\dint_0^x f'(t)\d t\\
&=\dint_0^x e^t\d t\\
&=\left[e^t\right]_0^x\\
&=e^x-1
\end{align*}
Επειδή $f(0)=2$, προκύπτει
\[f(x)=e^x+1\ ,\ x>0\]

Για $x<0$ έχουμε
\[f'(x)=\frac{|x|}{x}e^{|x|}=\frac{-x}{x}e^{-x}=-e^{-x}\]
Άρα, για κάθε $x<0$,
\begin{align*}
f(0)-f(x)&=\dint_x^0 f'(t)\d t\\
&=\dint_x^0 -e^{-t}\d t\\
&=\left[e^{-t}\right]_x^0\\
&=1-e^{-x}
\end{align*}
Επειδή $f(0)=2$, έχουμε
\[2-f(x)=1-e^{-x}\Rightarrow f(x)=e^{-x}+1\ ,\ x<0\]
Επομένως, λαμβάνοντας υπόψη και ότι $f(0)=2=e^0+1$, για κάθε $x\in\mathbb{R}$ ισχύει
\[f(x)=e^{|x|}+1\]

\item Εξετάζουμε την παραγωγισιμότητα της $f$ στο $x_0=0$ με τον ορισμό.
Για $h\neq 0$ είναι
\[\frac{f(h)-f(0)}{h}=\frac{e^{|h|}+1-2}{h}=\frac{e^{|h|}-1}{h}\]
Έχουμε
\[\lim\limits_{h\to 0^+}\frac{e^{|h|}-1}{h}=\lim\limits_{h\to 0^+}\frac{e^h-1}{h}=1\]
ενώ, θέτοντας $u=-h$, για $h\to 0^-$ έχουμε $u\to 0^+$ και
\[\lim\limits_{h\to 0^-}\frac{e^{|h|}-1}{h}=\lim\limits_{u\to 0^+}\frac{e^u-1}{-u}=-1\]
Επειδή τα πλευρικά όρια είναι διαφορετικά, η $f$ \textbf{δεν είναι παραγωγίσιμη} στο $x_0=0$.

\item Για $x\neq 0$ ισχύει
\[f'(x)=\frac{|x|}{x}e^{|x|}\]
Επειδή $e^{|x|}>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f'(x)$ είναι το πρόσημο του $\dfrac{|x|}{x}$.
Άρα
\begin{itemize}
\item αν $x<0$, τότε $f'(x)<0$
\item αν $x>0$, τότε $f'(x)>0$
\end{itemize}
Επομένως η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$.

Στον παρακάτω πίνακα βλέπουμε τη μονοτονία της $f$:

Άρα η $f$ παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[f(0)=2\]

\item Για $x<0$ είναι
\[f(x)=e^{-x}+1\]
Άρα
\begin{align*}
\lim\limits_{x\to-\infty}x^2f(x)
&=\lim\limits_{x\to-\infty}x^2(e^{-x}+1)\\
&=\lim\limits_{x\to-\infty}\left(x^2e^{-x}+x^2\right)
\end{align*}
Θέτουμε $u=-x$. Τότε, καθώς $x\to-\infty$, έχουμε $u\to+\infty$ και
\begin{align*}
\lim\limits_{x\to-\infty}x^2f(x)
&=\lim\limits_{u\to+\infty}\left(u^2e^u+u^2\right)\\
&=+\infty
\end{align*}
Επομένως
\[\lim\limits_{x\to-\infty}x^2f(x)=+\infty\]
\end{erwthma}
\end{thema}
