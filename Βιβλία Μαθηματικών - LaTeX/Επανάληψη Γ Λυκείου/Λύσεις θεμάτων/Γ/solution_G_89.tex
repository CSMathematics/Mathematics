\begin{erwthma}
\item Είναι
\[
h(x)=f(g(x)).
\]

Από το διάγραμμα της $g$ έχουμε
\[
g(0)=0,\quad g(1)=2,\quad g(3)=1.
\]

Άρα
\[
h(0)=f(g(0))=f(0)=1,
\]
\[
h(1)=f(g(1))=f(2)=3
\]
και
\[
h(3)=f(g(3))=f(1)=2{,}2.
\]

\item Η $h$ είναι παραγωγίσιμη ως σύνθεση παραγωγίσιμων συναρτήσεων και, από τον κανόνα της αλυσίδας,
\[
h'(x)=f'(g(x))\cdot g'(x).
\]

Για $x=1$ παίρνουμε
\[
h'(1)=f'(g(1))\cdot g'(1)=f'(2)\cdot g'(1).
\]

Από το διάγραμμα, η $f$ παρουσιάζει τοπικό μέγιστο στο σημείο με τετμημένη $2$, άρα η εφαπτομένη της $C_f$ στο σημείο $(2,3)$ είναι οριζόντια. Επομένως,
\[
f'(2)=0.
\]

Η $g'(1)$ είναι θετική, όπως φαίνεται γραφικά από την αύξηση της $g$ κοντά στο $x=1$. Δεν χρειάζεται όμως η ακριβής τιμή της, αφού
\[
h'(1)=f'(2)\cdot g'(1)=0\cdot g'(1)=0.
\]

\item Από το σχήμα, η $f$ παρουσιάζει τοπικό μέγιστο στο $u=2$. Δηλαδή, για τιμές του $u$ κοντά στο $2$ ισχύει
\[
f(u)\leq f(2).
\]

Επειδή η $g$ είναι συνεχής και $g(1)=2$, για τιμές του $x$ κοντά στο $1$ οι τιμές $g(x)$ είναι κοντά στο $2$. Άρα, για $x$ κοντά στο $1$,
\[
h(x)=f(g(x))\leq f(2)=h(1).
\]

Συνεπώς, η $h$ παρουσιάζει τοπικό μέγιστο στο $x=1$.

\item Η $h$ είναι συνεχής στο $[0,4]$, ως σύνθεση συνεχών συναρτήσεων.

Από το πρώτο ερώτημα έχουμε
\[
h(0)=1
\]
και
\[
h(1)=3.
\]

Επειδή
\[
h(0)=1<2<3=h(1),
\]
από το Θεώρημα \eng{Bolzano} για τη συνεχή συνάρτηση $h$ στο διάστημα $[0,1]$, υπάρχει $\xi\in(0,1)$ τέτοιο ώστε
\[
h(\xi)=2.
\]

Επειδή $(0,1)\subset(0,4)$, τελικά υπάρχει $\xi\in(0,4)$ τέτοιο ώστε
\[
h(\xi)=2.
\]
\end{erwthma}
\end{thema}
