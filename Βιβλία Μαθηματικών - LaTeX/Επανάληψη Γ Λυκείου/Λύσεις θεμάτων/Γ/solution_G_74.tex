%=========== ΛΥΣΗ - 74ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
xf'(x)+2f(x)=x\ln x
\]
πολλαπλασιάζουμε με $x$, αφού $x\geq 1$, και παίρνουμε
\[
x^2f'(x)+2xf(x)=x^2\ln x
\]
Όμως
\[
\left(x^2f(x)\right)'=2xf(x)+x^2f'(x)
\]
Άρα
\[
\left(x^2f(x)\right)'=x^2\ln x
\]
Ολοκληρώνουμε:
\[
x^2f(x)=\dint x^2\ln x\d x+c
\]
Υπολογίζουμε το ολοκλήρωμα κατά παράγοντες:
\begin{align*}
\dint x^2\ln x\d x
&=\frac{x^3}{3}\ln x-\dint \frac{x^3}{3}\cdot\frac{1}{x}\d x\\
&=\frac{x^3}{3}\ln x-\frac{1}{3}\dint x^2\d x\\
&=\frac{x^3}{3}\ln x-\frac{x^3}{9}
\end{align*}
Άρα
\[
x^2f(x)=\frac{x^3}{3}\ln x-\frac{x^3}{9}+c
\]
Από τη συνθήκη $f(1)=-\dfrac{1}{9}$ έχουμε
\[
-\frac{1}{9}=0-\frac{1}{9}+c
\]
οπότε
\[
c=0
\]
Επομένως
\[
x^2f(x)=\frac{x^3}{3}\ln x-\frac{x^3}{9}
\]
και, διαιρώντας με $x^2$,
\[
f(x)=\frac{x}{3}\ln x-\frac{x}{9}\ ,\quad x\geq 1
\]

\item Από το προηγούμενο ερώτημα
\[
f(x)=\frac{x}{3}\ln x-\frac{x}{9}
\]
οπότε
\[
f'(x)=\frac{1}{3}\ln x+\frac{1}{3}-\frac{1}{9}
=\frac{1}{3}\ln x+\frac{2}{9}
\]
Για $x=e$ έχουμε
\[
f(e)=\frac{e}{3}\ln e-\frac{e}{9}
=\frac{e}{3}-\frac{e}{9}
=\frac{2e}{9}
\]
και
\[
f'(e)=\frac{1}{3}\ln e+\frac{2}{9}
=\frac{1}{3}+\frac{2}{9}
=\frac{5}{9}
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $e$ είναι
\begin{align*}
y-\frac{2e}{9}&=\frac{5}{9}(x-e)\\
y&=\frac{5}{9}x-\frac{e}{3}
\end{align*}

\item Έχουμε
\[
\frac{f(x)}{x}
=\frac{\frac{x}{3}\ln x-\frac{x}{9}}{x}
=\frac{1}{3}\ln x-\frac{1}{9}
\]
Άρα
\[
\lim\limits_{x\to+\infty}\frac{f(x)}{x}
=\lim\limits_{x\to+\infty}\left(\frac{1}{3}\ln x-\frac{1}{9}\right)
=+\infty
\]

\item Θεωρούμε τη συνάρτηση
\[
g(t)=\ln t
\]
στο διάστημα $[x,x+1]$, όπου $x\geq 1$.
Η $g$ είναι συνεχής στο $[x,x+1]$ και παραγωγίσιμη στο $(x,x+1)$.
Σύμφωνα με το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
\frac{g(x+1)-g(x)}{x+1-x}=g'(\xi)
\]
δηλαδή
\[
\ln(x+1)-\ln x=\frac{1}{\xi}
\]
Επειδή
\[
\xi>x>0
\]
έχουμε
\[
\frac{1}{\xi}<\frac{1}{x}
\]
Άρα
\[
\ln(x+1)-\ln x<\frac{1}{x}
\]
για κάθε $x\geq 1$.
\end{erwthma}
\end{thema}
