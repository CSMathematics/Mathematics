%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.12 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $[1, e]$, άρα είναι και συνεχής στο $[1,e]$. 
Ικανοποιούνται οι προϋποθέσεις του \textbf{Θεωρήματος Μέσης Τιμής (\eng{T.M.T.})} στο διάστημα $[1, e]$.
Επομένως, υπάρχει τουλάχιστον ένα σημείο $\xi \in (1, e)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(e) - f(1)}{e - 1} \]
Αντικαθιστώντας τις δεδομένες τιμές $f(e) = e$ και $f(1) = 0$, προκύπτει:
\[ f'(\xi) = \frac{e - 0}{e - 1} = \frac{e}{e - 1} \]

\item Γνωρίζουμε ότι ο αριθμός $e \approx 2.71 > 1$. Αφού $e > 1$, ο παρονομαστής $e-1$ είναι θετικός. 
Ισχύει προφανώς ότι:
\[ e > e - 1 \overset{e-1>0}{\implies} \frac{e}{e - 1} > \frac{e - 1}{e - 1} \implies \frac{e}{e - 1} > 1 \]
Επομένως, $f'(\xi) > 1$.

Μας δίνεται η υπόθεση ότι η $f'$ παρουσιάζει ελάχιστο στο $\xi$, δηλαδή $\min f'(x) = f'(\xi)$. 
Αυτό σημαίνει ότι για κάθε $x \in (1, e)$ ισχύει:
\[ f'(x) \ge f'(\xi) \implies f'(x) \ge \frac{e}{e-1} > 1 \]
Άρα, $f'(x) > 1$ για κάθε $x \in (1, e)$. 
Γεωμετρικά, η $f'(x)$ εκφράζει την κλίση $\lambda$ της εφαπτομένης της $C_f$ στο αντίστοιχο σημείο. Συμπεραίνουμε λοιπόν ότι \textbf{οποιαδήποτε εφαπτομένη της γραφικής παράστασης $C_f$ στα εσωτερικά σημεία του διαστήματος $[1, e]$ έχει κλίση αυστηρά μεγαλύτερη της μονάδας ($\lambda > 1$)} (δηλαδή σχηματίζει γωνία μεγαλύτερη των $45^\circ$ με τον θετικό ημιάξονα $Ox$).

\item Δίνεται ότι $f(x) \geq \ln x$ για κάθε $x \in [1, e]$.
Θεωρούμε τη βοηθητική συνάρτηση $g(x) = f(x) - \ln x$, με $x \in [1, e]$. 
Από τη δοσμένη ανισότητα προκύπτει ότι $g(x) \geq 0$ για κάθε $x \in [1, e]$.
Υπολογίζουμε την τιμή της $g$ στο αριστερό άκρο $x=1$:
\[ g(1) = f(1) - \ln 1 = 0 - 0 = 0 \]
Κατά συνέπεια, ισχύει $g(x) \geq g(1)$ για κάθε $x \in [1, e]$. Αυτό σημαίνει ότι η συνάρτηση $g$ παρουσιάζει \textbf{ολικό ελάχιστο} στη θέση $x=1$.
Επειδή το $1$ είναι άκρο του πεδίου ορισμού, δεν μπορούμε να εφαρμόσουμε το Θεώρημα Fermat. Ωστόσο, μπορούμε να χρησιμοποιήσουμε τον ορισμό της παραγώγου:
Για $x \in (1, e]$ ισχύει $x - 1 > 0$, και επειδή $g(x) \ge g(1)$, έχουμε:
\[ \frac{g(x) - g(1)}{x - 1} \ge 0 \]
Παίρνοντας το όριο καθώς $x \to 1^+$, επειδή η $g$ είναι παραγωγίσιμη (άρα και παραγωγίσιμη από τα δεξιά στο 1), προκύπτει:
\[ \lim_{x \to 1^+} \frac{g(x) - g(1)}{x - 1} \ge 0 \implies g'(1) \ge 0 \]
Παραγωγίζοντας την $g(x)$, βρίσκουμε $g'(x) = f'(x) - \frac{1}{x}$. Επομένως, για $x=1$:
\[ g'(1) \ge 0 \implies f'(1) - \frac{1}{1} \ge 0 \implies f'(1) \ge 1 \]

\item Αναζητούμε ρίζα της εξίσωσης $f(x) = e - x \Leftrightarrow f(x) + x - e = 0$ στο $(1, e)$.
Θεωρούμε τη συνάρτηση $H(x) = f(x) + x - e$, ορισμένη στο $[1, e]$.
\begin{itemize}
    \item Η $H$ είναι συνεχής στο $[1, e]$ ως άθροισμα συνεχών συναρτήσεων.
    \item Υπολογίζουμε τις τιμές στα άκρα:
    \[ H(1) = f(1) + 1 - e = 0 + 1 - e = 1 - e \]
    Επειδή $e > 1$, έχουμε $1 - e < 0$, άρα $H(1) < 0$.
    \[ H(e) = f(e) + e - e = e + 0 = e > 0 \]
    Επομένως, ισχύει $H(1) \cdot H(e) = e(1 - e) < 0$.
\end{itemize}
Σύμφωνα με το \textbf{Θεώρημα \eng{Bolzano}}, υπάρχει τουλάχιστον ένα $x_0 \in (1, e)$ τέτοιο ώστε $H(x_0) = 0 \Leftrightarrow f(x_0) = e - x_0$.
\end{erwthma}
\end{thema}
