%=========== ΛΥΣΗ - 64ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $M$ το σημείο της ακτής στο οποίο αποβιβάζεται ο οδηγός και έστω
\[OM=x\ ,\quad x\in[0,4]\]
Τότε, από το ορθογώνιο τρίγωνο $AOM$, έχουμε
\[AM=\sqrt{AO^2+OM^2}=\sqrt{3^2+x^2}=\sqrt{x^2+9}\]
Άρα ο χρόνος κίνησης στη θάλασσα είναι
\[t_1=\frac{AM}{15}=\frac{\sqrt{x^2+9}}{15}\]

Επίσης, επειδή $OB=4$ και $OM=x$, είναι
\[MB=4-x\]
οπότε ο χρόνος κίνησης στην παραλία είναι
\[t_2=\frac{MB}{25}=\frac{4-x}{25}\]

Επομένως ο συνολικός χρόνος της διαδρομής $AMB$ είναι
\[T(x)=\frac{\sqrt{x^2+9}}{15}+\frac{4-x}{25}\ ,\quad x\in[0,4]\]

\item Η $T$ είναι παραγωγίσιμη στο $(0,4)$ και
\begin{align*}
T'(x)&=\frac{1}{15}\cdot\frac{2x}{2\sqrt{x^2+9}}-\frac{1}{25}\\
&=\frac{x}{15\sqrt{x^2+9}}-\frac{1}{25}
\end{align*}
Λύνουμε την εξίσωση $T'(x)=0$:
\begin{align*}
\frac{x}{15\sqrt{x^2+9}}-\frac{1}{25}&=0\\
\frac{x}{15\sqrt{x^2+9}}&=\frac{1}{25}\\
25x&=15\sqrt{x^2+9}\\
5x&=3\sqrt{x^2+9}
\end{align*}
Επειδή $x\in[0,4]$, τα δύο μέλη είναι μη αρνητικά, οπότε μπορούμε να υψώσουμε στο τετράγωνο:
\begin{align*}
25x^2&=9(x^2+9)\\
25x^2&=9x^2+81\\
16x^2&=81\\
x^2&=\frac{81}{16}
\end{align*}
Άρα, αφού $x\geq 0$,
\[x=\frac{9}{4}\]

\item Για $x\in(0,4)$ έχουμε
\[T'(x)=\frac{x}{15\sqrt{x^2+9}}-\frac{1}{25}\]
και
\[T'(x)<0 \Leftrightarrow 25x<15\sqrt{x^2+9}\Leftrightarrow 5x<3\sqrt{x^2+9}\]
Επειδή τα μέλη είναι μη αρνητικά, ισοδύναμα:
\begin{align*}
25x^2&<9(x^2+9)\\
16x^2&<81\\
x&<\frac{9}{4}
\end{align*}
Άρα
\begin{itemize}
\item $T'(x)<0$ για $0<x<\dfrac{9}{4}$
\item $T'(x)>0$ για $\dfrac{9}{4}<x<4$
\end{itemize}
Συνεπώς η $T$ είναι γνησίως φθίνουσα στο $\left[0,\dfrac{9}{4}\right]$ και γνησίως αύξουσα στο $\left[\dfrac{9}{4},4\right]$.

Επομένως ο χρόνος γίνεται ελάχιστος όταν
\[x=\frac{9}{4}\ \si{km}\]
δηλαδή ο οδηγός πρέπει να αποβιβαστεί σε σημείο της ακτής που απέχει $\dfrac{9}{4}\ \si{km}$ από το $O$.

\item Αν ο οδηγός κινηθεί απευθείας με το σκάφος από το $A$ στο $B$, τότε, από το ορθογώνιο τρίγωνο $AOB$,
\[AB=\sqrt{AO^2+OB^2}=\sqrt{3^2+4^2}=5\ \si{km}\]
Άρα ο χρόνος της απευθείας μετάβασης είναι
\[T_{\text{απ}}=\frac{5}{15}=\frac{1}{3}\ \si{h}\]

Ο ελάχιστος χρόνος που βρήκαμε είναι
\begin{align*}
T\left(\frac{9}{4}\right)
&=\frac{\sqrt{\left(\frac{9}{4}\right)^2+9}}{15}+\frac{4-\frac{9}{4}}{25}\\
&=\frac{\sqrt{\frac{81}{16}+\frac{144}{16}}}{15}+\frac{\frac{7}{4}}{25}\\
&=\frac{\frac{15}{4}}{15}+\frac{7}{100}\\
&=\frac{1}{4}+\frac{7}{100}\\
&=\frac{32}{100}\\
&=\frac{8}{25}\ \si{h}
\end{align*}
Επειδή
\[\frac{1}{3}>\frac{8}{25}\]
η απευθείας μετάβαση με σκάφος διαρκεί περισσότερο από τη βέλτιστη διαδρομή.
\end{erwthma}
\end{thema}
