%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.64 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Στο σχήμα διαμορφώνονται δύο βασικά τμήματα κίνησης. 
Έστω ότι ο οδηγός διασχίζει τη θάλασσα (μήκους $AM$) και ακολούθως τρέχει στην ακτή (μήκος $MB$).

\textbf{Τμήμα 1 (Θάλασσα):} Στο ορθογώνιο τρίγωνο $AOM$ έχουμε κάθετες πλευρές $AO = 3$ και $OM = x$. Από το Πυθαγόρειο Θεώρημα, η υποτείνουσα της θαλάσσιας διαδρομής είναι διαδοχικά:
\[ (AM) = \sqrt{(AO)^2 + (OM)^2} = \sqrt{3^2 + x^2} = \sqrt{x^2 + 9} \quad \text{(\si{km})} \]
Ο χρόνος στη θάλασσα με ταχύτητα $v_w = 15$ \si{km/h} είναι:
\[ t_w = \frac{AM}{v_w} = \frac{\sqrt{x^2+9}}{15} \]

\textbf{Τμήμα 2 (Παραλία):} Αφού η απόσταση $OB = 4$ και το σημείο $M$ είναι στα $x$ \si{km}, το υπολειπόμενο τμήμα καθορίζεται:
\[ (MB) = (OB) - (OM) = 4 - x \quad \text{(\si{km})} \]
Ο χρόνος τρεξίματος στην παραλία με ταχύτητα $v_b = 25$ \si{km/h} είναι:
\[ t_b = \frac{MB}{v_b} = \frac{4-x}{25} \]

Ο \textbf{συνολικός χρόνος} $T(x)$ ως άθροισμα αποτελείται:
\[ \mathbf{T(x) = \frac{\sqrt{x^2+9}}{15} + \frac{4-x}{25}}, \quad x \in [0, 4] \]

\item Παραγωγίζουμε τη συνάρτηση (χρησιμοποιώντας κανόνα παραγώγισης ρίζας):
\begin{align*}
T'(x) &= \frac{1}{15} \cdot \frac{1}{2\sqrt{x^2+9}} \cdot (x^2+9)' + \frac{1}{25}(4-x)' \\
&= \frac{2x}{30\sqrt{x^2+9}} - \frac{1}{25} = \frac{x}{15\sqrt{x^2+9}} - \frac{1}{25}
\end{align*}
Λύνουμε την εξίσωση $T'(x) = 0$:
\[ \frac{x}{15\sqrt{x^2+9}} = \frac{1}{25} \Rightarrow 25x = 15\sqrt{x^2+9} \Rightarrow 5x = 3\sqrt{x^2+9} \]
Επειδή $x \geq 0$, όλα τα μέλη είναι μη αρνητικά. Υψώνουμε στο τετράγωνο:
\[ (5x)^2 = (3\sqrt{x^2+9})^2 \Rightarrow 25x^2 = 9(x^2+9) \Rightarrow 25x^2 = 9x^2 + 81 \]
\[ 16x^2 = 81 \Rightarrow x^2 = \frac{81}{16} \]
Καθώς $x$ αφορά θετικό μήκος, κρατάμε την ομόσημη ρίζα: $\mathbf{x = \frac{9}{4} = 2{,}25 \ \si{km}}$.

\item Μελετάμε το πρόσημο γύρω από τη ρίζα $2{,}25$:
- Αν $x < \frac{9}{4} \Rightarrow 5x < 3\sqrt{x^2+9} \Rightarrow T'(x) < 0$.
- Αν $x > \frac{9}{4} \Rightarrow 5x > 3\sqrt{x^2+9} \Rightarrow T'(x) > 0$.
Οπότε ο χρόνος $T(x)$ είναι γνησίως φθίνων στο $[0, 2{,}25]$ και γνησίως αύξων στο $[2{,}25, 4]$. Παρουσιάζει ολικό ελάχιστο.
Η βέλτιστη απόσταση για να ελαχιστοποιηθεί ο χρόνος είναι συνεπώς \textbf{$\mathbf{x = 2{,}25 \ \si{km}}$ (ή $\frac{9}{4}$ \si{km})}.

\item Αν κινηθεί αποκλειστικά με το σκάφος απευθείας στο $B$, το $x$ θα πάρει την τιμή $4$. Όλη η διαδρομή είναι θαλάσσια:
Το μήκος της υδάτινης απευθείας υποτείνουσας είναι:
\[ AB = \sqrt{3^2 + 4^2} = \sqrt{9+16} = \sqrt{25} = 5 \ \si{km} \]
Ο χρόνος απευθείας διέλευσης ($T_a$) είναι:
\[ T_a = \frac{5}{15} = \frac{1}{3} \ \si{h} \approx 0{,}333 \ \si{h} \ (\text{ή } 20 \ \text{λεπτά}) \]

Υπολογίζουμε συγκριτικά τον ελάχιστο χρόνο της βέλτιστης διαδρομής $T(2{,}25)$:
\begin{align*}
T(9/4) &= \frac{\sqrt{(9/4)^2+9}}{15} + \frac{4-9/4}{25} \\
&= \frac{\sqrt{\frac{81}{16} + \frac{144}{16}}}{15} + \frac{\frac{7}{4}}{25} \\
&= \frac{\sqrt{\frac{225}{16}}}{15} + \frac{7}{100} = \frac{\frac{15}{4}}{15} + \frac{7}{100} \\
&= \frac{1}{4} + \frac{7}{100} = \frac{25}{100} + \frac{7}{100} = \frac{32}{100} \ \si{h} = 0{,}32 \ \si{h} \ (\text{ή } 19{,}2 \ \text{λεπτά})
\end{align*}
Προφανέστατα, το  $0{,}333$ (απευθείας) είναι μεγαλύτερο του $0{,}32$ (της υπολογισμένης βέλτιστης κοίτης). Έτσι \textbf{επιβεβαιώνεται ότι η απευθείας μετάβαση υπολείπεται} της ιδανικής υβριδικής χάραξης.
\end{erwthma}
\end{thema}
