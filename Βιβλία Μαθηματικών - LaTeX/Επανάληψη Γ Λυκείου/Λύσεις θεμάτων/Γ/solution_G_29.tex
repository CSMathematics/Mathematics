\begin{thema}{Γ}
\begin{erwthma}
\item 
Θεωρούμε τη συνάρτηση $g(x) = f(x) e^{-x}$. 
Εφόσον η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, τότε και η $g$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως γινόμενο παραγωγίσιμων συναρτήσεων.
Η παράγωγός της είναι:
\[
g'(x) = f'(x) e^{-x} + f(x) (-e^{-x}) = e^{-x} [f'(x) - f(x)]
\]
Από την εκφώνηση γνωρίζουμε ότι $f'(x) - f(x) = e^x$, επομένως:
\[
g'(x) = e^{-x} \cdot e^x = e^0 = 1
\]
Για κάθε $x \in \mathbb{R}$, χρησιμοποιώντας το θεμελιώδες θεώρημα του ολοκληρωτικού λογισμού:
\[
g(x) - g(0) = \int_{0}^{x} g'(t) \d t = \int_{0}^{x} 1 \d t = [t]_{0}^{x} = x
\]
Από τη συνθήκη $f(0) = 1$ έχουμε $g(0) = f(0) e^{0} = 1 \cdot 1 = 1$.
Συνεπώς:
\[
g(x) - 1 = x \Rightarrow g(x) = x + 1
\]
Εφόσον $g(x) = f(x) e^{-x}$, προκύπτει:
\[
f(x) e^{-x} = x + 1 \Rightarrow f(x) = (x+1) e^x
\]

\item 
Η συνάρτηση $f(x) = (x+1) e^x$ είναι παραγωγίσιμη στο $\mathbb{R}$ και η παράγωγός της είναι:
\[
f'(x) = 1 \cdot e^x + (x+1) \cdot e^x = (1 + x + 1) e^x = (x+2) e^x
\]
Για τον προσδιορισμό των ακροτάτων εξετάζουμε το πρόσημο της $f'(x)$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow x+2 = 0 \Leftrightarrow x = -2$
    \item $f'(x) > 0 \Leftrightarrow x+2 > 0 \Leftrightarrow x > -2$
    \item $f'(x) < 0 \Leftrightarrow x+2 < 0 \Leftrightarrow x < -2$
\end{itemize}

Από τον πίνακα παραγωγίσιμης προκύπτει ότι η συνάρτηση $f$ είναι $\textbf{γνησίως φθίνουσα}$ στο $(-\infty, -2)$ και $\textbf{γνησίως αύξουσα}$ στο $(-2, +\infty)$. 
Επομένως, η συνάρτηση παρουσιάζει ολικό ελάχιστο στο $x = -2$ με τιμή:
\[
f(-2) = (-2+1) e^{-2} = -e^{-2} = -\frac{1}{e^2}
\]

\item 
Για τον έλεγχο οριζόντιας ασύμπτωτου στο $-\infty$ υπολογίζουμε το όριο:
\[
\lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} (x+1) e^x
\]
Το όριο είναι της μορφής $(-\infty \cdot 0)$. Το γράφουμε στη μορφή κλασμάτων:
\[
\lim_{x \to -\infty} \frac{x+1}{e^{-x}}
\]
Το όριο είναι της μορφής $\frac{-\infty}{+\infty}$, οπότε εφαρμόζουμε το θεώρημα \eng{L'Hôpital}:
\[
\lim_{x \to -\infty} \frac{x+1}{e^{-x}} \xlongequal{\eng{L'Hôpital}} \lim_{x \to -\infty} \frac{1}{-e^{-x}} = \lim_{x \to -\infty} -e^x = 0
\]
Επειδή το όριο είναι πεπερασμένος αριθμός, η ευθεία $y = 0$ (ο άξονας $x'$) είναι οριζόντια ασύμπτωτος της γραφικής παράστασης της $f$ στο $-\infty$.

\item 
Υπολογίζουμε την έκφραση $f(x+1) - e \cdot f(x)$:
\begin{align*}
f(x+1) - e \cdot f(x) &= ((x+1)+1) e^{x+1} - e (x+1) e^x \\
&= (x+2) e \cdot e^x - (x+1) e \cdot e^x \\
&= e \cdot e^x [ (x+2) - (x+1) ] \\
&= e \cdot e^x [ x + 2 - x - 1 ] \\
&= e \cdot e^x \cdot 1 = e^{x+1}
\end{align*}
Τότε το ζητούμενο όριο είναι:
\[
\lim_{x \to +\infty} [ f(x+1) - e \cdot f(x) ] = \lim_{x \to +\infty} e^{x+1} = +\infty
\]
\end{erwthma}
\end{thema}
