%=========== ΛΥΣΗ - 29ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για κάθε πραγματικό αριθμό $x$, η δοσμένη σχέση γράφεται:
\[ f'(x) - f(x) = e^x \]
Πολλαπλασιάζουμε και τα δύο μέλη της εξίσωσης με τον "παράγοντα ολοκλήρωσης" $e^{-x} > 0$:
\begin{align*}
e^{-x} f'(x) - e^{-x} f(x) &= e^x \cdot e^{-x} \\
e^{-x} f'(x) + (e^{-x})' f(x) &= e^0 \\
\left( e^{-x} f(x) \right)' &= (x)'
\end{align*}
Σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, εφόσον οι δύο συναρτήσεις έχουν ίσες παραγώγους σε διάστημα (στο $\mathbb{R}$), διαφέρουν κατά μια σταθερά $c \in \mathbb{R}$:
\[ e^{-x} f(x) = x + c \]
Για να βρούμε την τιμή της σταθεράς $c$, χρησιμοποιούμε την αρχική συνθήκη $f(0) = 1$. Αντικαθιστώντας $x = 0$:
\[ e^0 f(0) = 0 + c \implies 1 \cdot 1 = c \implies c = 1 \]
Επομένως, έχουμε:
\[ e^{-x} f(x) = x + 1 \implies f(x) = (x + 1) e^x \]

\item Η συνάρτηση $f(x) = (x+1) e^x$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως γινόμενο παραγωγίσιμων, με παράγωγο:
\[ f'(x) = (x+1)' e^x + (x+1) (e^x)' = 1 \cdot e^x + (x+1) e^x = (x+2) e^x \]
Επειδή $e^x > 0$ για κάθε $x \in \mathbb{R}$, το πρόσημο της παραγώγου εξαρτάται μόνο από το πρόσημο του παράγοντα $(x+2)$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow x+2 = 0 \Leftrightarrow x = -2$
    \item $f'(x) > 0 \Leftrightarrow x+2 > 0 \Leftrightarrow x > -2$
    \item $f'(x) < 0 \Leftrightarrow x+2 < 0 \Leftrightarrow x < -2$
\end{itemize}
Κατασκευάζουμε τον πίνακα προσήμου της $f'$ και μονοτονίας της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[lgt=2, espcl=3]{$x$ / 1, $f'(x)$ / 1, $f(x)$ / 2.5}{$-\infty$, $-2$, $+\infty$}
\tkzTabLine{, -, z, +, }
\tkzTabVar{+ / $0$, - / $-\dfrac{1}{e^2}$, + / $+\infty$}
\end{tikzpicture}
\end{center}
*(Σημείωση: Tο όριο $\lim\limits_{x \to -\infty} f(x) = 0$ αποδεικνύεται στο επόμενο ερώτημα).*

Όπως φαίνεται από τον πίνακα, η συνάρτηση $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty, -2]$ και \textbf{γνησίως αύξουσα} στο $[-2, +\infty)$.
Συνεπώς, παρουσιάζει \textbf{μοναδικό ολικό ελάχιστο} (ακρότατο) στη θέση $x_0 = -2$, με τιμή:
\[ f(-2) = (-2+1) e^{-2} = -e^{-2} = \boldsymbol{-\frac{1}{e^2}} \]

\item Για να ελέγξουμε αν η γραφική παράσταση της $f$ δέχεται οριζόντια ασύμπτωτη στο $-\infty$, υπολογίζουμε το όριο:
\[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} (x+1) e^x \]
Όταν το $x \to -\infty$, έχουμε $(x+1) \to -\infty$ και $e^x \to 0$, επομένως πρόκειται για απροσδιοριστία μορφής $(-\infty) \cdot 0$.
Μετασχηματίζουμε την παράσταση ώστε να πάρει μορφή $\frac{-\infty}{+\infty}$:
\[ \lim_{x \to -\infty} (x+1) e^x = \lim_{x \to -\infty} \frac{x+1}{e^{-x}} \]
Επειδή πληρούνται οι προϋποθέσεις, εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to -\infty} \frac{x+1}{e^{-x}} \xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}} \lim_{x \to -\infty} \frac{(x+1)'}{(e^{-x})'} = \lim_{x \to -\infty} \frac{1}{-e^{-x}} = \lim_{x \to -\infty} (-e^x) = 0 \]
Εφόσον το όριο είναι πραγματικός αριθμός, η ευθεία $\boldsymbol{y = 0}$ (δηλαδή ο άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} της $C_f$ στο $-\infty$.

\item Υπολογίζουμε πρώτα την αλγεβρική παράσταση μέσα στο όριο:
\begin{align*}
f(x+1) - e \cdot f(x) &= ( (x+1) + 1 ) e^{x+1} - e \cdot (x+1) e^x \\
&= (x+2) e^{x+1} - (x+1) e^{x+1} \\
&= e^{x+1} \left( (x+2) - (x+1) \right) \\
&= e^{x+1} (x + 2 - x - 1) \\
&= e^{x+1} \cdot 1 = e^{x+1}
\end{align*}
Άρα το ζητούμενο όριο είναι:
\[ \lim_{x \to +\infty} \left[ f(x+1) - e \cdot f(x) \right] = \lim_{x \to +\infty} e^{x+1} = \boldsymbol{+\infty} \]
(αφού όταν $x \to +\infty$ το $x+1 \to +\infty$, οπότε το όριο της εκθετικής συνάρτησης είναι $+\infty$).
\end{erwthma}
\end{thema}
