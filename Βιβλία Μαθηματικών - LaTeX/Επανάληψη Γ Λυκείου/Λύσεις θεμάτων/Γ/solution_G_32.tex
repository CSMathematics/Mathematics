%=========== ΛΥΣΗ - 32ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $f(x) = e^x - ax$, το ζητούμενο όριο γράφεται:
\[ L_1 = \lim_{x \to 0} \frac{f(x) - 1}{x} = \lim_{x \to 0} \frac{e^x - ax - 1}{x} \]
Διαχωρίζουμε το κλάσμα:
\[ L_1 = \lim_{x \to 0} \left( \frac{e^x - 1}{x} - \frac{ax}{x} \right) \]
Για $x \neq 0$, το $x$ απλοποιείται στον δεύτερο όρο:
\[ L_1 = \lim_{x \to 0} \left( \frac{e^x - 1}{x} - a \right) \]
Υπολογίζουμε ξεχωριστά το όριο $\lim\limits_{x \to 0} \frac{e^x - 1}{x}$, το οποίο είναι μορφής $\frac{0}{0}$, με τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to 0} \frac{e^x - 1}{x} \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim_{x \to 0} \frac{(e^x - 1)'}{(x)'} = \lim_{x \to 0} \frac{e^x}{1} = e^0 = 1 \]
*(Σημείωση: Το όριο αυτό υπολογίζεται εξίσου εύκολα και με τον ορισμό της παραγώγου της συνάρτησης $h(x)=e^x$ στο $x_0=0$).*

Αντικαθιστώντας, έχουμε:
\[ L_1 = 1 - a \]
Απαιτούμε από την εκφώνηση το όριο να ισούται με 2:
\[ 1 - a = 2 \implies \boldsymbol{a = -1} \]

\item Για $a = 1$, ο τύπος της συνάρτησης είναι $f(x) = e^x - x$.
Το ζητούμενο όριο είναι:
\[ L_2 = \lim_{x \to 0} \frac{f(x) - 1}{x^2} = \lim_{x \to 0} \frac{e^x - x - 1}{x^2} \]
Για $x \to 0$, ο αριθμητής τείνει στο $e^0 - 0 - 1 = 0$ και ο παρονομαστής στο $0^2 = 0$. 
Έχουμε απροσδιοριστία $\frac{0}{0}$, οπότε εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ L_2 \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim_{x \to 0} \frac{(e^x - x - 1)'}{(x^2)'} = \lim_{x \to 0} \frac{e^x - 1}{2x} \]
Το νέο όριο είναι πάλι μορφής $\frac{0}{0}$. Εφαρμόζουμε για δεύτερη φορά τον κανόνα \eng{De L'Hospital} (ή χρησιμοποιούμε το όριο που βρήκαμε στο Γ1):
\[ L_2 \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim_{x \to 0} \frac{(e^x - 1)'}{(2x)'} = \lim_{x \to 0} \frac{e^x}{2} \]
Η συνάρτηση είναι πλέον συνεχής στο $0$, οπότε:
\[ L_2 = \frac{e^0}{2} = \boldsymbol{\frac{1}{2}} \]

\item Η συνάρτηση $f(x) = e^x - x$ είναι παραγωγίσιμη σε όλο το $\mathbb{R}$, με παράγωγο:
\[ f'(x) = e^x - 1 \]
Βρίσκουμε τις ρίζες και το πρόσημο της $f'(x)$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow e^x = 1 \Leftrightarrow x = 0$
    \item $f'(x) > 0 \Leftrightarrow e^x > 1 \Leftrightarrow x > 0$
    \item $f'(x) < 0 \Leftrightarrow e^x < 1 \Leftrightarrow x < 0$
\end{itemize}
Κατασκευάζουμε τον πίνακα μεταβολών της συνάρτησης:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[lgt=2, espcl=3]{$x$ / 1, $f'(x)$ / 1, $f(x)$ / 2.5}{$-\infty$, $0$, $+\infty$}
\tkzTabLine{, -, z, +, }
\tkzTabVar{+ / $+\infty$, - / $1$, + / $+\infty$}
\end{tikzpicture}
\end{center}
Από τη μελέτη προκύπτει ότι η συνάρτηση $f$ είναι \textbf{γνησίως φθίνουσα} στο διάστημα $(-\infty, 0]$ και \textbf{γνησίως αύξουσα} στο διάστημα $[0, +\infty)$.
Επομένως, παρουσιάζει \textbf{ολικό ελάχιστο} στη θέση $x_0 = 0$, με ελάχιστη τιμή:
\[ f(0) = e^0 - 0 = 1 \]
Επειδή η ελάχιστη τιμή της συνάρτησης είναι το 1, ισχύει $f(x) \ge 1$ για κάθε πραγματικό αριθμό $x$. 
Κατά συνέπεια αποδεικνύεται ότι $\boldsymbol{f(x) > 0}$ για κάθε $x \in \mathbb{R}$.

\item Ζητείται ο υπολογισμός του ορίου:
\[ L_3 = \lim_{x \to +\infty} \left[ \frac{f(x)}{e^x} \cdot \ln x \right] = \lim_{x \to +\infty} \left[ \frac{e^x - x}{e^x} \cdot \ln x \right] \]
Σπάμε το κλάσμα:
\[ L_3 = \lim_{x \to +\infty} \left[ \left( 1 - \frac{x}{e^x} \right) \cdot \ln x \right] \]
Υπολογίζουμε ξεχωριστά το όριο του κλάσματος $\frac{x}{e^x}$ στο $+\infty$, το οποίο είναι μορφής $\frac{+\infty}{+\infty}$. Με κανόνα \eng{De L'Hospital} έχουμε:
\[ \lim_{x \to +\infty} \frac{x}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to +\infty} \frac{(x)'}{(e^x)'} = \lim_{x \to +\infty} \frac{1}{e^x} = 0 \]
Επιπλέον, γνωρίζουμε ότι $\lim\limits_{x \to +\infty} \ln x = +\infty$.
Αντικαθιστώντας τα επιμέρους όρια στο $L_3$, προκύπτει:
\[ L_3 = (1 - 0) \cdot (+\infty) = 1 \cdot (+\infty) = \boldsymbol{+\infty} \]
\end{erwthma}
\end{thema}
