%=========== ΛΥΣΗ - 31ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι όλο το $\mathbb{R}$, καθώς το υπόρριζο είναι πάντα θετικό ($x^2+1 \ge 1 > 0$).
Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως πηλίκο παραγωγίσιμων συναρτήσεων, με:
\begin{align*}
f'(x) &= \left( \frac{x}{\sqrt{x^2+1}} \right)' = \frac{(x)' \sqrt{x^2+1} - x (\sqrt{x^2+1})'}{(\sqrt{x^2+1})^2} \\
&= \frac{1 \cdot \sqrt{x^2+1} - x \frac{2x}{2\sqrt{x^2+1}}}{x^2+1} \\
&= \frac{\sqrt{x^2+1} - \frac{x^2}{\sqrt{x^2+1}}}{x^2+1} = \frac{\frac{x^2+1-x^2}{\sqrt{x^2+1}}}{x^2+1} \\
&= \frac{1}{(x^2+1)\sqrt{x^2+1}} = \frac{1}{(x^2+1)^{\frac{3}{2}}}
\end{align*}
Επειδή $x^2+1 > 0$ για κάθε $x \in \mathbb{R}$, ισχύει $f'(x) > 0$ για κάθε πραγματικό αριθμό.
Επομένως, η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Εφόσον η $f$ είναι συνεχής και γνησίως αύξουσα, το σύνολο τιμών της είναι:
\[ f(\mathbb{R}) = \left( \lim_{x \to -\infty} f(x), \lim_{x \to +\infty} f(x) \right) \]
Για $x > 0$, γνωρίζουμε ότι $\sqrt{x^2} = |x| = x$. Άρα:
\[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{x}{\sqrt{x^2\left(1+\frac{1}{x^2}\right)}} = \lim_{x \to +\infty} \frac{x}{x\sqrt{1+\frac{1}{x^2}}} = \lim_{x \to +\infty} \frac{1}{\sqrt{1+\frac{1}{x^2}}} = \frac{1}{\sqrt{1+0}} = 1 \]
Αντίστοιχα, για $x < 0$, έχουμε $\sqrt{x^2} = |x| = -x$. Άρα:
\[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} \frac{x}{\sqrt{x^2\left(1+\frac{1}{x^2}\right)}} = \lim_{x \to -\infty} \frac{x}{-x\sqrt{1+\frac{1}{x^2}}} = \lim_{x \to -\infty} \frac{-1}{\sqrt{1+\frac{1}{x^2}}} = \frac{-1}{\sqrt{1+0}} = -1 \]
Επομένως, το \textbf{σύνολο τιμών} της $f$ είναι το διάστημα $\boldsymbol{(-1, 1)}$.

\item Αφού η συνάρτηση $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, είναι συνάρτηση \textbf{1-1} (ένα προς ένα). 
Κατά συνέπεια, \textbf{αντιστρέφεται}.
Το πεδίο ορισμού της αντίστροφης συνάρτησης είναι το σύνολο τιμών της $f$, δηλαδή το $A_{f^{-1}} = (-1, 1)$.
Για την εύρεση του τύπου της, λύνουμε την εξίσωση $y = f(x)$ ως προς $x$, για $y \in (-1, 1)$:
\begin{align*}
y &= \frac{x}{\sqrt{x^2+1}} \\
y^2 &= \frac{x^2}{x^2+1} \\
y^2(x^2+1) &= x^2 \\
x^2y^2 + y^2 &= x^2 \\
x^2(1 - y^2) &= y^2
\end{align*}
Επειδή $y \in (-1, 1)$, ισχύει $y^2 < 1 \implies 1-y^2 > 0$, άρα μπορούμε να διαιρέσουμε:
\[ x^2 = \frac{y^2}{1-y^2} \implies |x| = \frac{|y|}{\sqrt{1-y^2}} \]
Από την αρχική σχέση $y = \frac{x}{\sqrt{x^2+1}}$, βλέπουμε ότι ο παρονομαστής είναι πάντα θετικός, συνεπώς τα $x$ και $y$ είναι \textbf{ομόσημα}. Επομένως, τα απόλυτα αφαιρούνται χωρίς αλλαγή προσήμου:
\[ x = \frac{y}{\sqrt{1-y^2}} \]
Άρα ο \textbf{τύπος της αντίστροφης} είναι:
\[ \boldsymbol{f^{-1}(x) = \frac{x}{\sqrt{1-x^2}}, \quad x \in (-1, 1)} \]

\item Ζητείται η εφαπτομένη της $C_{f^{-1}}$ στο σημείο με τετμημένη $x_0 = \frac{\sqrt{2}}{2} \in (-1, 1)$.
Η τεταγμένη του σημείου επαφής είναι:
\[ f^{-1}\left(\frac{\sqrt{2}}{2}\right) = \frac{\frac{\sqrt{2}}{2}}{\sqrt{1-\left(\frac{\sqrt{2}}{2}\right)^2}} = \frac{\frac{\sqrt{2}}{2}}{\sqrt{1-\frac{1}{2}}} = \frac{\frac{\sqrt{2}}{2}}{\sqrt{\frac{1}{2}}} = \frac{\frac{\sqrt{2}}{2}}{\frac{\sqrt{2}}{2}} = 1 \]
Παραγωγίζουμε την αντίστροφη συνάρτηση στο $(-1, 1)$:
\begin{align*}
(f^{-1})'(x) &= \left( \frac{x}{\sqrt{1-x^2}} \right)' = \frac{1\cdot \sqrt{1-x^2} - x\frac{-2x}{2\sqrt{1-x^2}}}{1-x^2} \\
&= \frac{\sqrt{1-x^2} + \frac{x^2}{\sqrt{1-x^2}}}{1-x^2} = \frac{\frac{1-x^2+x^2}{\sqrt{1-x^2}}}{1-x^2} = \frac{1}{(1-x^2)^{\frac{3}{2}}}
\end{align*}
Υπολογίζουμε την κλίση στο $x_0 = \frac{\sqrt{2}}{2}$:
\[ \lambda = (f^{-1})'\left(\frac{\sqrt{2}}{2}\right) = \frac{1}{\left(1-\frac{1}{2}\right)^{\frac{3}{2}}} = \frac{1}{\left(\frac{1}{2}\right)^{\frac{3}{2}}} = \frac{1}{\frac{1}{2\sqrt{2}}} = 2\sqrt{2} \]
Η εξίσωση της εφαπτομένης είναι:
\[ y - 1 = 2\sqrt{2}\left(x - \frac{\sqrt{2}}{2}\right) \implies y - 1 = 2\sqrt{2}x - 2 \implies \boldsymbol{y = 2\sqrt{2}x - 1} \]

\item Αντικαθιστούμε τον τύπο της αντίστροφης στο ζητούμενο όριο:
\[ L = \lim_{x \to 0} \frac{f^{-1}(x) - x}{x^3} = \lim_{x \to 0} \frac{\frac{x}{\sqrt{1-x^2}} - x}{x^3} \]
Βγάζουμε κοινό παράγοντα το $x$ στον αριθμητή:
\[ L = \lim_{x \to 0} \frac{x \left( \frac{1}{\sqrt{1-x^2}} - 1 \right)}{x^3} = \lim_{x \to 0} \frac{1 - \sqrt{1-x^2}}{x^2 \sqrt{1-x^2}} \]
Πολλαπλασιάζουμε αριθμητή και παρονομαστή με τη \textbf{συζυγή παράσταση} του αριθμητή ($1 + \sqrt{1-x^2}$):
\begin{align*}
L &= \lim_{x \to 0} \frac{(1 - \sqrt{1-x^2})(1 + \sqrt{1-x^2})}{x^2 \sqrt{1-x^2} (1 + \sqrt{1-x^2})} \\
&= \lim_{x \to 0} \frac{1 - (1-x^2)}{x^2 \sqrt{1-x^2} (1 + \sqrt{1-x^2})} \\
&= \lim_{x \to 0} \frac{x^2}{x^2 \sqrt{1-x^2} (1 + \sqrt{1-x^2})}
\end{align*}
Απλοποιούμε το $x^2$ (αφού $x \to 0$, είναι $x \neq 0$):
\[ L = \lim_{x \to 0} \frac{1}{\sqrt{1-x^2} (1 + \sqrt{1-x^2})} \]
Η συνάρτηση είναι πλέον συνεχής στο 0, οπότε αντικαθιστούμε απλά $x = 0$:
\[ L = \frac{1}{\sqrt{1} (1 + \sqrt{1})} = \frac{1}{1 \cdot 2} = \boldsymbol{\frac{1}{2}} \]
\end{erwthma}
\end{thema}
