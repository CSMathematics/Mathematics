%=========== ΛΥΣΗ - 44ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $O$ το κέντρο της σφαίρας, $K$ το κέντρο της βάσης του κώνου και $A$ ένα σημείο της περιφέρειας της βάσης.
Τότε το τρίγωνο $OKA$ είναι ορθογώνιο στο $K$, αφού η $OK$ είναι κάθετη στο επίπεδο της βάσης του κώνου, ενώ η $KA$ ανήκει στο επίπεδο αυτό.

Έχουμε:
\begin{itemize}
\item $OA=R=3$, ως ακτίνα της σφαίρας
\item $OK=x$, από τον ορισμό του $x$
\item $KA=\rho$, ως ακτίνα της βάσης του κώνου
\end{itemize}
Από το Πυθαγόρειο Θεώρημα στο ορθογώνιο τρίγωνο $OKA$ προκύπτει
\[OK^2+KA^2=OA^2\Rightarrow x^2+\rho^2=3^2\Rightarrow \rho^2=9-x^2\]

\item Το ύψος $h$ του κώνου είναι η απόσταση της κορυφής του από το επίπεδο της βάσης.
Αφού η κορυφή του κώνου βρίσκεται στο ανώτερο σημείο της σφαίρας και η βάση του βρίσκεται κάτω από το κέντρο $O$ σε απόσταση $x$, είναι
\[h=3+x\]
Ο όγκος του κώνου δίνεται από τον τύπο
\[V=\frac{1}{3}\pi\rho^2h\]
Άρα, χρησιμοποιώντας το προηγούμενο ερώτημα και τη σχέση $h=3+x$, έχουμε
\[V(x)=\frac{\pi}{3}(9-x^2)(3+x)\ ,\ x\in(0,3)\]

\item Για κάθε $x\in(0,3)$ είναι
\begin{align*}
V(x)&=\frac{\pi}{3}(9-x^2)(3+x)\\
&=\frac{\pi}{3}(27+9x-3x^2-x^3)
\end{align*}
οπότε
\[V'(x)=\frac{\pi}{3}(9-6x-3x^2)=\pi(3-2x-x^2)\]
Βρίσκουμε ρίζες και πρόσημα της $V'$ στο $(0,3)$:
\begin{align*}
V'(x)=0
&\Leftrightarrow 3-2x-x^2=0\\
&\Leftrightarrow x^2+2x-3=0\\
&\Leftrightarrow (x-1)(x+3)=0\\
&\Leftrightarrow x=1 \text{ ή } x=-3
\end{align*}
Επειδή $x\in(0,3)$, δεκτή είναι μόνο η ρίζα $x=1$.
Επιπλέον
\begin{itemize}
\item $V'(x)>0\Leftrightarrow 3-2x-x^2>0\Leftrightarrow -3<x<1$, άρα $V'(x)>0$ για $x\in(0,1)$
\item $V'(x)<0\Leftrightarrow 3-2x-x^2<0\Leftrightarrow x<-3$ ή $x>1$, άρα $V'(x)<0$ για $x\in(1,3)$
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $V'$ και τη μονοτονία της $V$:
\begin{center}

\end{tikzpicture}
\end{center}
Επομένως ο όγκος του κώνου μεγιστοποιείται για
\[x=1\]
Για $x=1$ έχουμε
\[h=3+1=4\si{cm}\]
και
\[\rho^2=9-1^2=8\Rightarrow \rho=2\sqrt{2}\si{cm}\]
Άρα ο μέγιστος κώνος έχει ύψος $4\si{cm}$ και ακτίνα βάσης $2\sqrt{2}\si{cm}$.

\item Έστω $x(t)$ η απόσταση του κέντρου της σφαίρας από τη βάση του κώνου τη χρονική στιγμή $t$.
Αφού το $x$ αυξάνεται με ρυθμό $0.5\si{cm/s}$, έχουμε
\[x'(t)=0.5\si{cm/s}\]
Αν $t_0$ είναι η χρονική στιγμή κατά την οποία το ύψος του κώνου είναι $4\si{cm}$, τότε
\[h=3+x(t_0)=4\Rightarrow x(t_0)=1\]
Θέτουμε $W(t)=V(x(t))$, όπου $W(t)$ είναι ο όγκος του κώνου τη χρονική στιγμή $t$.
Με τον κανόνα της αλυσίδας έχουμε
\[W'(t)=V'(x(t))\cdot x'(t)\]
Άρα
\[W'(t_0)=V'(1)\cdot x'(t_0)=0\cdot 0.5=0\si{cm^3/s}\]
Επομένως τη χρονική στιγμή που το ύψος του κώνου είναι $4\si{cm}$, ο ρυθμός μεταβολής του όγκου είναι $0\si{cm^3/s}$.
\end{erwthma}
\end{thema}
