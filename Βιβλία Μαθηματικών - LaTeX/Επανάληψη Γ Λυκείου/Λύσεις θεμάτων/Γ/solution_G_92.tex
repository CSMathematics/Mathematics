\begin{erwthma}
\item Από τη γραφική παράσταση $C_{f'}$ παρατηρούμε ότι:
\begin{itemize}
    \item $f'(x) > 0 \Leftrightarrow x \in [-1, 1) \cup (4, 5]$
    \item $f'(x) < 0 \Leftrightarrow x \in (1, 4)$
    \item $f'(x) = 0 \Leftrightarrow x = 1$ ή $x = 4$
\end{itemize}
Συνεπώς:
\begin{itemize}
    \item Η $f$ είναι \textbf{γνησίως αύξουσα} στα διαστήματα $[-1, 1]$ και $[4, 5]$.
    \item Η $f$ είναι \textbf{γνησίως φθίνουσα} στο διάστημα $[1, 4]$.
    \item Η $f$ παρουσιάζει \textbf{τοπικό μέγιστο} στο $x = 1$ και \textbf{τοπικό ελάχιστο} στο $x = 4$.
\end{itemize}

\item Η γραφική παράσταση $C_f$ παρουσιάζει σημείο καμπής εκεί που η παράγωγός της, $f'$, παρουσιάζει ακρότατο. 
Από το σχήμα, η $f'$ είναι παραβολή με κορυφή στο σημείο $(2{,}5, -2{,}25)$. 
Επομένως, η $f'$ έχει τοποπικό ελάχιστο στο $x = 2{,}5$.
Συνεπώς, η $C_f$ παρουσιάζει \textbf{σημείο καμπής} στο $x = 2{,}5$.

\item Από το θεμελιώδες θεώρημα του ολοκληρωτικού λογισμού έχουμε:
\[ f(4) - f(1) = \int_{1}^{4} f'(x) \d x \]
Γνωρίζουμε ότι $f(1) = 4$, οπότε:
\[ f(4) = 4 + \int_{1}^{4} f'(x) \d x \]
Για $x \in (1, 4)$, ισχύει $f'(x) < 0$. Το ολοκλήρωμα $\int_{1}^{4} f'(x) \d x$ αντιπροσωπεύει το πρόσημο του εμβαδού που περικλείεται από την $C_{f'}$, τον άξονα $x'$ και τις κάθετες $x=1, x=4$. Επειδή η $C_{f'}$ βρίσκεται κάτω από τον άξονα $x'$ στο διάστημα $(1, 4)$, ισχύει:
\[ \int_{1}^{4} f'(x) \d x < 0 \]
Άρα:
\[ f(4) = 4 + (\text{αρνητικός αριθμός}) < 4 \]

\item Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $x_0 = 2{,}5$ δίνεται από τον τύπο:
\[ y - f(x_0) = f'(x_0)(x - x_0) \]
Από τα δεδομένα και το σχήμα έχουμε:
\begin{itemize}
    \item $f(2{,}5) = 1$
    \item $f'(2{,}5) = -2{,}25 = -\frac{9}{4}$
\end{itemize}
Αντικαθιστώντας, η εξίσωση είναι:
\begin{align*}
    y - 1 &= -\frac{9}{4}(x - 2{,}5) \\
    y - 1 &= -\frac{9}{4}x + \frac{9 \cdot 2{,}5}{4} \\
    y &= -2{,}25x + \frac{22{,}5}{4} + 1 \\
    y &= -2{,}25x + 5{,}625 + 1 \\
    y &= -2{,}25x + 6{,}625
\end{align*}
Ή σε κλασματική μορφή:
\[ y = -\frac{9}{4}x + \frac{53}{8} \]
\end{itemize}
\end{erwthma}
\end{thema}
