\begin{erwthma}
\item Από το διάγραμμα της $f'$ παρατηρούμε ότι:
\[
f'(x)>0,\quad x\in[-1,1)\cup(4,5],
\]
\[
f'(x)<0,\quad x\in(1,4),
\]
και
\[
f'(x)=0 \Leftrightarrow x=1 \text{ ή } x=4.
\]

Άρα η $f$ είναι γνησίως αύξουσα στα διαστήματα $[-1,1]$, $[4,5]$ και γνησίως φθίνουσα στο $[1,4]$.

Στο $x=1$ η $f'$ αλλάζει πρόσημο από θετικό σε αρνητικό, οπότε η $f$ παρουσιάζει τοπικό μέγιστο.

Στο $x=4$ η $f'$ αλλάζει πρόσημο από αρνητικό σε θετικό, οπότε η $f$ παρουσιάζει τοπικό ελάχιστο.

\item Τα σημεία καμπής της $C_f$ αντιστοιχούν σε σημεία όπου η $f'$ αλλάζει μονοτονία.

Από το διάγραμμα, η $f'$ είναι γνησίως φθίνουσα στο $[-1,2{,}5]$ και γνησίως αύξουσα στο $[2{,}5,5]$. Άρα η $f$ είναι κοίλη στο $[-1,2{,}5]$ και κυρτή στο $[2{,}5,5]$.

Επομένως, η $C_f$ παρουσιάζει σημείο καμπής για
\[
x=2{,}5.
\]

\item Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού έχουμε
\[
f(4)-f(1)=\int_1^4 f'(x)\,\d x.
\]

Επειδή $f(1)=4$, προκύπτει
\[
f(4)=4+\int_1^4 f'(x)\,\d x.
\]

Στο διάστημα $(1,4)$ η γραφική παράσταση της $f'$ βρίσκεται κάτω από τον άξονα $x'x$, δηλαδή
\[
f'(x)<0,\quad x\in(1,4).
\]
Άρα
\[
\int_1^4 f'(x)\,\d x<0.
\]

Επομένως,
\[
f(4)=4+\int_1^4 f'(x)\,\d x<4.
\]

\item Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο με τετμημένη $x_0$ είναι
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]

Δίνεται ότι
\[
x_0=2{,}5=\frac{5}{2}
\]
και
\[
f(2{,}5)=1.
\]
Από το διάγραμμα της $f'$ διαβάζουμε
\[
f'(2{,}5)=-2{,}25=-\frac{9}{4}.
\]

Άρα
\[
y-1=-\frac{9}{4}\left(x-\frac{5}{2}\right).
\]
Επομένως,
\[
y-1=-\frac{9}{4}x+\frac{45}{8}
\]
και τελικά
\[
y=-\frac{9}{4}x+\frac{53}{8}.
\]
\end{erwthma}
\end{thema}
