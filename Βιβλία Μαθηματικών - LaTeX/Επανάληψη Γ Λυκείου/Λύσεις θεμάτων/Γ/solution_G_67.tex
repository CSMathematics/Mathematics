%=========== ΛΥΣΗ - 67ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\begin{align*}
f'(x)
&=\left(\frac{\ln x}{x^2}\right)'\\
&=\frac{(\ln x)'x^2-\ln x(x^2)'}{x^4}\\
&=\frac{x-2x\ln x}{x^4}\\
&=\frac{1-2\ln x}{x^3}
\end{align*}
Επειδή $x^3>0$ για κάθε $x>0$, το πρόσημο της $f'$ καθορίζεται από το $1-2\ln x$.
Έχουμε:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow 1-2\ln x=0\Leftrightarrow \ln x=\dfrac{1}{2}\Leftrightarrow x=\sqrt{e}$
\item $f'(x)>0\Leftrightarrow 0<x<\sqrt{e}$
\item $f'(x)<0\Leftrightarrow x>\sqrt{e}$
\end{itemize}
Άρα η $f$ είναι γνησίως αύξουσα στο $(0,\sqrt{e}]$ και γνησίως φθίνουσα στο $[\sqrt{e},+\infty)$.

Επομένως η $f$ παρουσιάζει ολικό μέγιστο για
\[x=\sqrt{e}\]
με μέγιστη τιμή
\[f(\sqrt{e})=\frac{\ln\sqrt{e}}{(\sqrt{e})^2}=\frac{\frac{1}{2}}{e}=\frac{1}{2e}\]

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\begin{align*}
f''(x)
&=\left(\frac{1-2\ln x}{x^3}\right)'\\
&=\frac{\left(1-2\ln x\right)'x^3-\left(1-2\ln x\right)(x^3)'}{x^6}\\
&=\frac{-2x^2-3x^2(1-2\ln x)}{x^6}\\
&=\frac{6\ln x-5}{x^4}
\end{align*}
Επειδή $x^4>0$ για κάθε $x>0$, το πρόσημο της $f''$ καθορίζεται από το $6\ln x-5$.
Έχουμε:
\begin{itemize}
\item $f''(x)=0\Leftrightarrow 6\ln x-5=0\Leftrightarrow x=e^{\frac{5}{6}}$
\item $f''(x)<0\Leftrightarrow 0<x<e^{\frac{5}{6}}$
\item $f''(x)>0\Leftrightarrow x>e^{\frac{5}{6}}$
\end{itemize}
Άρα η $f$ είναι κοίλη στο $\left(0,e^{\frac{5}{6}}\right]$ και κυρτή στο $\left[e^{\frac{5}{6}},+\infty\right)$.

Επειδή η $f''$ αλλάζει πρόσημο στο $x=e^{\frac{5}{6}}$, η $C_f$ έχει σημείο καμπής το
\[
K\left(e^{\frac{5}{6}},f\left(e^{\frac{5}{6}}\right)\right)
\]
όπου
\[
f\left(e^{\frac{5}{6}}\right)
=\frac{\ln e^{\frac{5}{6}}}{\left(e^{\frac{5}{6}}\right)^2}
=\frac{\frac{5}{6}}{e^{\frac{5}{3}}}
=\frac{5}{6e^{\frac{5}{3}}}
\]
Άρα
\[
K\left(e^{\frac{5}{6}},\frac{5}{6e^{\frac{5}{3}}}\right)
\]

\item Για $a>1$ έχουμε
\[E(a)=\dint_1^a \frac{\ln x}{x^2}\d x\]
Εφαρμόζουμε ολοκλήρωση κατά παράγοντες με
\[u=\ln x\quad \text{και}\quad \d v=\frac{1}{x^2}\d x\]
οπότε
\[\d u=\frac{1}{x}\d x\quad \text{και}\quad v=-\frac{1}{x}\]
Άρα
\begin{align*}
E(a)
&=\left[-\frac{\ln x}{x}\right]_1^a-\dint_1^a\left(-\frac{1}{x}\right)\frac{1}{x}\d x\\
&=-\frac{\ln a}{a}+\dint_1^a\frac{1}{x^2}\d x\\
&=-\frac{\ln a}{a}+\left[-\frac{1}{x}\right]_1^a\\
&=-\frac{\ln a}{a}-\frac{1}{a}+1\\
&=1-\frac{\ln a+1}{a}
\end{align*}

\item Από το προηγούμενο ερώτημα
\[E(a)=1-\frac{\ln a+1}{a}\]
Άρα
\begin{align*}
\lim\limits_{a\to+\infty}E(a)
&=\lim\limits_{a\to+\infty}\left(1-\frac{\ln a+1}{a}\right)\\
&=1-\lim\limits_{a\to+\infty}\frac{\ln a+1}{a}\\
&=1
\end{align*}
διότι
\[\lim\limits_{a\to+\infty}\frac{\ln a}{a}=0\quad \text{και}\quad \lim\limits_{a\to+\infty}\frac{1}{a}=0\]
\end{erwthma}
\end{thema}
