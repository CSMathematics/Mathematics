\item Η συνάρτηση $f$ είναι συνεχής και παραγωγίσιμη στο $D_f = (0, +\infty)$. Η παράγωγός της είναι:
\[ f'(x) = \frac{\left(\dfrac{1}{x}\right) \cdot x^2 - (\ln x) \cdot 2x}{(x^2)^2} = \frac{x - 2x \ln x}{x^4} = \frac{1 - 2 \ln x}{x^3} \]
Για τον προσδιορισμό των ακροτάτων μελετάμε το πρόσημο της $f'(x)$ για $x > 0$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 1 - 2 \ln x = 0 \Leftrightarrow \ln x = \frac{1}{2} \Leftrightarrow x = e^{1/2} = \sqrt{e}$
    \item $f'(x) > 0 \Leftrightarrow 1 - 2 \ln x > 0 \Leftrightarrow \ln x < \frac{1}{2} \Leftrightarrow 0 < x < \sqrt{e}$
    \item $f'(x) < 0 \Leftrightarrow 1 - 2 \ln x < 0 \Leftrightarrow \ln x > \frac{1}{2} \Leftrightarrow x > \sqrt{e}$
\end{itemize}
Συνεπώς, η $f$ είναι $\textbf{γνησίως αύξουσα}$ στο $(0, \sqrt{e}]$ και $\textbf{γνησίως φθίνουσα}$ στο $[\sqrt{e}, +\infty)$. Άρα η συνάρτηση παρουσιάζει \textbf{ολικό μέγιστο} για $x = \sqrt{e}$.

\item Υπολογίζουμε το ορισμένο ολοκλήρωμα $E(a) = \int_{1}^{a} \frac{\ln x}{x^2} \d x$ εφαρμόζοντας \eng{Integration by parts}. 
Θέτουμε $u = \ln x \Rightarrow \d u = \frac{1}{x} \d x$ και $\d v = \frac{1}{x^2} \d x \Rightarrow v = -\frac{1}{x}$.
\begin{align*}
E(a) &= \left[ -\frac{\ln x}{x} \right]_1^a - \int_{1}^{a} \left( -\frac{1}{x} \right) \frac{1}{x} \d x \\
&= \left( -\frac{\ln a}{a} - \left( -\frac{\ln 1}{1} \right) \right) + \int_{1}^{a} \frac{1}{x^2} \d x \\
&= -\frac{\ln a}{a} + \left[ -\frac{1}{x} \right]_1^a \\
&= -\frac{\ln a}{a} + \left( -\frac{1}{a} - (-1) \right) \\
&= 1 - \frac{\ln a + 1}{a}
\end{align*}

\item Υπολογίζουμε το όριο:
\[ \lim_{a \to +\infty} E(a) = \lim_{a \to +\infty} \left( 1 - \frac{\ln a + 1}{a} \right) \]
Γνωρίζουμε ότι $\lim_{a \to +\infty} \frac{\ln a}{a} = 0$ (ή εφαρμόζοντας τον κανόνα $\eng{L'Hôpital}$), επομένως:
\[ \lim_{a \to +\infty} \frac{\ln a + 1}{a} = \lim_{a \to +\infty} \frac{\ln a}{a} + \lim_{a \to +\infty} \frac{1}{a} = 0 + 0 = 0 \]
Άρα:
\[ \lim_{a \to +\infty} E(a) = 1 - 0 = 1 \]
\end{erwthma}
