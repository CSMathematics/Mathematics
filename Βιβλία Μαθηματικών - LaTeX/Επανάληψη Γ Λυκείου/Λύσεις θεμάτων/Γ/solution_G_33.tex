\begin{erwthma}
\item Για να βρούμε τα κοινά σημεία των γραφικών παραστάσεων $C_f$ και $C_g$, λύνουμε την εξίσωση $f(x) = g(x)$:
\[ x^2 + 1 = -x^2 + 4x - 1 \Leftrightarrow 2x^2 - 4x + 2 = 0 \Leftrightarrow x^2 - 2x + 1 = 0 \Leftrightarrow (x-1)^2 = 0 \Leftrightarrow x = 1 \]
Για $x=1$, έχουμε $f(1) = 1^2 + 1 = 2$, άρα το μοναδικό κοινό σημείο είναι το $A(1, 2)$.
Υπολογίζουμε τις παραγώγους των συναρτήσεων:
\[ f'(x) = 2x, \quad g'(x) = -2x + 4 \]
Για $x=1$, έχουμε:
\[ f'(1) = 2(1) = 2 \quad \text{και} \quad g'(1) = -2(1) + 4 = 2 \]
Επειδή $f(1) = g(1)$ και $f'(1) = g'(1)$, το σημείο $A(1, 2)$ είναι \textbf{κοινό σημείο επαφής} των δύο καμπυλών.

\item Εφόσον $f'(1) = g'(1) = 2$, οι δύο γραφικές παραστάσεις δέχονται \textbf{κοινή εφαπτόμενη} στο σημείο $A(1, 2)$. Η εξίσωση της εφαπτόμενης είναι:
\[ y - 2 = 2(x - 1) \Leftrightarrow y = 2x - 2 + 2 \Leftrightarrow y = 2x \]

\item Υπολογίζουμε το όριο χρησιμοποιώντας το συζυγές του αριθμητή:
\begin{align*}
\lim_{x \to 1} \frac{\sqrt{f(x)} - \sqrt{2}}{x - 1} &= \lim_{x \to 1} \frac{\sqrt{x^2 + 1} - \sqrt{2}}{x - 1} \\
&= \lim_{x \to 1} \frac{(\sqrt{x^2 + 1} - \sqrt{2})(\sqrt{x^2 + 1} + \sqrt{2})}{(x - 1)(\sqrt{x^2 + 1} + \sqrt{2})} \\
&= \lim_{x \to 1} \frac{x^2 + 1 - 2}{(x - 1)(\sqrt{x^2 + 1} + \sqrt{2})} \\
&= \lim_{x \to 1} \frac{(x - 1)(x + 1)}{(x - 1)(\sqrt{x^2 + 1} + \sqrt{2})} \\
&= \lim_{x \to 1} \frac{x + 1}{\sqrt{x^2 + 1} + \sqrt{2}} = \frac{1 + 1}{\sqrt{1^2 + 1} + \sqrt{2}} = \frac{2}{2\sqrt{2}} = \frac{1}{\sqrt{2}} = \frac{\sqrt{2}}{2}
\end{align*}

\item Το εμβαδόν $E$ του χωρίου που περικλείεται από τις δύο καμπύλες και τον άξονα $y'y$ (δηλαδή στο διάστημα $[0, 1]$) δίνεται από το ολοκλήρωμα:
\[ E = \int_{0}^{1} |f(x) - g(x)| \d x \]
Παρατηρούμε ότι $f(x) - g(x) = 2x^2 - 4x + 2 = 2(x-1)^2 \geq 0$ για κάθε $x \in \mathbb{R}$.
Άρα στο διάστημα $[0, 1]$, η $C_f$ βρίσκεται πάνω από την $C_g$.
\begin{align*}
E &= \int_{0}^{1} (2x^2 - 4x + 2) \d x \\
&= \left[ \frac{2}{3}x^3 - 2x^2 + 2x \right]_{0}^{1} \\
&= \left( \frac{2}{3} \cdot 1^3 - 2 \cdot 1^2 + 2 \cdot 1 \right) - 0 \\
&= \frac{2}{3} - 2 + 2 = \frac{2}{3}
\end{align*}
Επομένως, το εμβαδόν του χωρίου είναι $E = \frac{2}{3} \text{ τ.μ.}$.
\end{erwthma}
