\begin{thema}{Γ}
\begin{erwthma}
\item Για να βρούμε τα κοινά σημεία των γραφικών παραστάσεων $C_f$ και $C_g$, εξισώνουμε τους τύπους των δύο συναρτήσεων:
\begin{align*}
f(x) &= g(x) \\
x^2 + 1 &= -x^2 + 4x - 1 \\
2x^2 - 4x + 2 &= 0 \\
2(x^2 - 2x + 1) &= 0 \\
2(x-1)^2 &= 0 \implies x = 1
\end{align*}
Για $x=1$, η κοινή τεταγμένη είναι $f(1) = 1^2 + 1 = 2$. Άρα το \textbf{μοναδικό κοινό σημείο} είναι το $A(1, 2)$.

Υπολογίζουμε τις παραγώγους των δύο συναρτήσεων:
\[ f'(x) = 2x \quad \text{και} \quad g'(x) = -2x + 4 \]
Υπολογίζουμε την τιμή των παραγώγων στο $x_0 = 1$:
\[ f'(1) = 2 \cdot 1 = 2 \quad \text{και} \quad g'(1) = -2 \cdot 1 + 4 = 2 \]
Επειδή $f(1) = g(1) = 2$ και $f'(1) = g'(1) = 2$, συμπεραίνουμε ότι οι δύο καμπύλες εφάπτονται στο σημείο $A(1, 2)$. 
Άρα το $A(1, 2)$ είναι το \textbf{κοινό σημείο επαφής}.

\item Εφόσον $f(1) = g(1)$ και $f'(1) = g'(1) = 2$, οι δύο γραφικές παραστάσεις δέχονται \textbf{κοινή εφαπτόμενη} στο σημείο $A(1, 2)$ με κλίση $\lambda = 2$.
Η εξίσωση της κοινής εφαπτομένης ($\varepsilon$) είναι:
\begin{align*}
y - f(1) &= f'(1)(x - 1) \\
y - 2 &= 2(x - 1) \\
y &= 2x - 2 + 2 \\
\boldsymbol{y} &\boldsymbol{= 2x}
\end{align*}

\item Αντικαθιστούμε τον τύπο της $f(x)$ στο όριο:
\[ L = \lim_{x \to 1} \frac{\sqrt{f(x)} - \sqrt{2}}{x - 1} = \lim_{x \to 1} \frac{\sqrt{x^2 + 1} - \sqrt{2}}{x - 1} \]
Πρόκειται για απροσδιοριστία $\frac{0}{0}$. Πολλαπλασιάζουμε αριθμητή και παρονομαστή με τη συζυγή παράσταση του αριθμητή ($\sqrt{x^2+1} + \sqrt{2}$):
\begin{align*}
L &= \lim_{x \to 1} \frac{\left( \sqrt{x^2 + 1} - \sqrt{2} \right) \left( \sqrt{x^2 + 1} + \sqrt{2} \right)}{(x - 1) \left( \sqrt{x^2 + 1} + \sqrt{2} \right)} \\
&= \lim_{x \to 1} \frac{(x^2 + 1) - 2}{(x - 1) \left( \sqrt{x^2 + 1} + \sqrt{2} \right)} \\
&= \lim_{x \to 1} \frac{x^2 - 1}{(x - 1) \left( \sqrt{x^2 + 1} + \sqrt{2} \right)}
\end{align*}
Παραγοντοποιούμε τη διαφορά τετραγώνων στον αριθμητή:
\[ L = \lim_{x \to 1} \frac{(x - 1)(x + 1)}{(x - 1) \left( \sqrt{x^2 + 1} + \sqrt{2} \right)} = \lim_{x \to 1} \frac{x + 1}{\sqrt{x^2 + 1} + \sqrt{2}} \]
Αντικαθιστούμε $x = 1$:
\[ L = \frac{1 + 1}{\sqrt{1^2 + 1} + \sqrt{2}} = \frac{2}{\sqrt{2} + \sqrt{2}} = \frac{2}{2\sqrt{2}} = \frac{1}{\sqrt{2}} = \boldsymbol{\frac{\sqrt{2}}{2}} \]

\item Ζητείται το εμβαδόν του χωρίου που περικλείεται από τις δύο καμπύλες ($C_f, C_g$) και τον άξονα $y'y$ ($x=0$).
Το μοναδικό κοινό σημείο των καμπυλών είναι στο $x=1$. Επομένως, το χωρίο ολοκλήρωσης είναι το διάστημα $[0, 1]$.
Βρίσκουμε το πρόσημο της διαφοράς $f(x) - g(x)$ στο διάστημα αυτό:
\[ f(x) - g(x) = (x^2 + 1) - (-x^2 + 4x - 1) = 2x^2 - 4x + 2 = 2(x^2 - 2x + 1) = 2(x - 1)^2 \]
Επειδή $2(x-1)^2 \ge 0$ για κάθε $x \in \mathbb{R}$, η γραφική παράσταση $C_f$ βρίσκεται συνεχώς "πάνω" από την $C_g$.
Το ζητούμενο εμβαδόν είναι:
\begin{align*}
E &= \int_{0}^{1} |f(x) - g(x)| \d x = \int_{0}^{1} 2(x - 1)^2 \d x
\end{align*}
Υπολογίζουμε το ολοκλήρωμα άμεσα με τον κανόνα της δύναμης:
\begin{align*}
E &= 2 \left[ \frac{(x - 1)^3}{3} \right]_0^1 \\
&= \frac{2}{3} \left( (1 - 1)^3 - (0 - 1)^3 \right) \\
&= \frac{2}{3} (0 - (-1)) = \boldsymbol{\frac{2}{3} \ \text{τ.μ.}}
\end{align*}
\end{erwthma}
\end{thema}
