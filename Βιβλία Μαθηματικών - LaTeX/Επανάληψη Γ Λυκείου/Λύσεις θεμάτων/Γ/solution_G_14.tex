\begin{thema}{Γ}
\begin{erwthma}
\item Για τη συνάρτηση $f(x) = x \ln x - x + 1$ με $x > 0$, η παράγωγός της είναι:
\[ f'(x) = (1 \cdot \ln x + x \cdot \frac{1}{x}) - 1 = \ln x + 1 - 1 = \ln x \]
Μελετάμε το πρόσημο της $f'(x)$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow \ln x = 0 \Leftrightarrow x = 1$
    \item $f'(x) > 0 \Leftrightarrow \ln x > 0 \Leftrightarrow x > 1$
    \item $f'(x) < 0 \Leftrightarrow \ln x < 0 \Leftrightarrow 0 < x < 1$
\end{itemize}
Συνεπώς, η $f$ είναι \textbf{γνησίως φθίνουσα} στο διάστημα $(0, 1)$ και \textbf{γνησίως αύξουσα} στο διάστημα $(1, +\infty)$.
Η συνάρτηση παρουσιάζει \textbf{τοπικό ελάχιστο} στο $x = 1$ με τιμή:
\[ f(1) = 1 \cdot \ln 1 - 1 + 1 = 0 \]

\item Για τη μελέτη της εφαπτομένης στο σημείο με τετμημένη $x = 1$, έχουμε:
\[ f(1) = 0 \quad \text{και} \quad f'(1) = \ln 1 = 0 \]
Η εξίσωση της εφαπτομένης είναι:
\[ y - f(1) = f'(1)(x - 1) \Rightarrow y - 0 = 0(x - 1) \Rightarrow y = 0 \]
Άρα η εφαπτομένη είναι ο άξονας των τετμημένων.

\item Υπολογίζουμε τη δεύτερη παράγωγο της $f$:
\[ f''(x) = \frac{1}{x} \]
Επειδή $x > 0$, ισχύει $f''(x) > 0$ για κάθε $x \in D_f$, επομένως η $f$ είναι \textbf{κυρτή} στο $(0, +\infty)$.
Δεδομένου ότι η $f$ είναι κυρτή και παρουσιάζει ελάχιστο στο $x = 1$ με $f(1) = 0$, ισχύει για κάθε $x > 0$:
\[ f(x) \ge f(1) \Rightarrow x \ln x - x + 1 \ge 0 \Rightarrow x \ln x \ge x - 1 \]

\item Θεωρούμε το διάστημα $[x, x+1]$ για $x > 0$. Η συνάρτηση $f$ είναι \textbf{συνεχώς} και \textbf{παραγωγήσιμη} στο διάστημα αυτό.
Εφαρμόζοντας το θεωρημα \eng{Lagrange}, υπάρχει $\xi \in (x, x+1)$ τέτοιο ώστε:
\[ f(x+1) - f(x) = f'(\xi)(x+1 - x) = f'(\xi) \]
Υπολογίζουμε τη διαφορά:
\begin{align*}
f(x+1) - f(x) &= [(x+1)\ln(x+1) - (x+1) + 1] - [x \ln x - x + 1] \\
&= (x+1)\ln(x+1) - x - x \ln x + x - 1 + 1 \\
&= (x+1)\ln(x+1) - x \ln x - 1
\end{align*}
Εφόσον $f'(\xi) = \ln \xi$ και $x < \xi < x+1$, και γνωρίζοντας ότι η συνάρτηση $\ln t$ είναι \textbf{γνησίως αύξουσα}, ισχύει:
\[ \ln x < \ln \xi < \ln(x+1) \]
Αντικαθιστώντας το $\ln \xi$ με τη διαφορά $f(x+1) - f(x)$, καταλήγουμε στο:
\[ \ln x < (x+1)\ln(x+1) - x\ln x - 1 < \ln(x+1) \]
\end{erwthma}
\end{thema}
