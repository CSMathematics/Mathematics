%=========== ΛΥΣΗ - 27ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f$ είναι πολυωνυμική, συνεπώς είναι παραγωγίσιμη σε όλο το $\mathbb{R}$, με παράγωγο:
\[ f'(x) = \left( \frac{1}{3}x^3 - \lambda x^2 + 4x - 2 \right)' = x^2 - 2\lambda x + 4 \]
Για να είναι η συνάρτηση $f$ \textbf{γνησίως αύξουσα} στο $\mathbb{R}$, πρέπει η παράγωγός της να είναι μη αρνητική ($f'(x) \ge 0$) για κάθε $x \in \mathbb{R}$, μηδενιζόμενη το πολύ σε μεμονωμένα σημεία.
Επειδή η παράγωγος $f'(x)$ είναι τριώνυμο 2ου βαθμού με θετικό συντελεστή στον δευτεροβάθμιο όρο ($a=1>0$), η συνθήκη $f'(x) \ge 0$ ικανοποιείται αν και μόνο αν η διακρίνουσα $\Delta$ είναι μικρότερη ή ίση του μηδενός:
\[ \Delta \le 0 \implies (-2\lambda)^2 - 4 \cdot 1 \cdot 4 \le 0 \]
\[ 4\lambda^2 - 16 \le 0 \implies 4\lambda^2 \le 16 \implies \lambda^2 \le 4 \]
\[ |\lambda| \le 2 \implies -2 \le \lambda \le 2 \]
Άρα, το ζητούμενο σύνολο τιμών της παραμέτρου $\lambda$ είναι το κλειστό διάστημα $[-2, 2]$.
(Παρατηρούμε ότι όταν $\Delta=0$, η $f'$ μηδενίζεται σε ένα μόνο σημείο, οπότε η $f$ παραμένει γνησίως αύξουσα παντού).

\item Για $\lambda = 2$, ο τύπος της συνάρτησης γίνεται:
\[ f(x) = \frac{1}{3}x^3 - 2x^2 + 4x - 2 \]
Η πρώτη παράγωγος είναι:
\[ f'(x) = x^2 - 4x + 4 = (x-2)^2 \]
Η δεύτερη παράγωγος της $f$ είναι:
\[ f''(x) = (x^2 - 4x + 4)' = 2x - 4 \]
Για την εύρεση των σημείων καμπής, εξετάζουμε τις ρίζες και το πρόσημο της δεύτερης παραγώγου:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow 2x - 4 = 0 \Leftrightarrow x = 2$
    \item $f''(x) > 0 \Leftrightarrow 2x - 4 > 0 \Leftrightarrow x > 2$
    \item $f''(x) < 0 \Leftrightarrow 2x - 4 < 0 \Leftrightarrow x < 2$
\end{itemize}
Εφόσον η δεύτερη παράγωγος μηδενίζεται στο $x_0 = 2$ και εκατέρωθεν αυτού αλλάζει πρόσημο, το σημείο με τετμημένη $2$ είναι \textbf{σημείο καμπής} της γραφικής παράστασης.
Υπολογίζουμε την τεταγμένη του σημείου καμπής:
\[ f(2) = \frac{1}{3} \cdot 2^3 - 2 \cdot 2^2 + 4 \cdot 2 - 2 = \frac{8}{3} - 8 + 8 - 2 = \frac{8}{3} - \frac{6}{3} = \frac{2}{3} \]
Άρα, το μοναδικό σημείο καμπής είναι το $\boldsymbol{K\left(2, \frac{2}{3}\right)}$.

\item Έχουμε την παράσταση:
\begin{align*}
f(x) - \frac{2}{3} &= \left( \frac{1}{3}x^3 - 2x^2 + 4x - 2 \right) - \frac{2}{3} \\
&= \frac{1}{3}x^3 - 2x^2 + 4x - \frac{8}{3}
\end{align*}
Βγάζοντας κοινό παράγοντα το $\frac{1}{3}$ έχουμε:
\begin{align*}
f(x) - \frac{2}{3} &= \frac{1}{3} (x^3 - 6x^2 + 12x - 8) \\
&= \frac{1}{3} (x^3 - 3 \cdot x^2 \cdot 2 + 3 \cdot x \cdot 2^2 - 2^3) \\
&= \boldsymbol{\frac{1}{3}(x-2)^3}
\end{align*}
Λύνουμε τώρα τη δοσμένη ανίσωση:
\begin{align*}
f(|x| - 1) &< \frac{2}{3} \\
f(|x| - 1) - \frac{2}{3} &< 0
\end{align*}
Εφαρμόζοντας την ταυτότητα που μόλις αποδείξαμε για τη μεταβλητή $(|x|-1)$:
\begin{align*}
\frac{1}{3}(|x| - 1 - 2)^3 &< 0 \\
(|x| - 3)^3 &< 0
\end{align*}
Αφού ο κύβος ενός πραγματικού αριθμού είναι αρνητικός αν και μόνο αν ο ίδιος ο αριθμός είναι αρνητικός, έχουμε:
\begin{align*}
|x| - 3 &< 0 \\
|x| &< 3 \\
\boldsymbol{-3} &\boldsymbol{< x < 3}
\end{align*}

\item Ζητείται το εμβαδόν μεταξύ της $C_f$, της εφαπτομένης της στο σημείο καμπής $K\left(2, \frac{2}{3}\right)$ και του άξονα $y'y$ ($x=0$).
Υπολογίζουμε πρώτα την εξίσωση της εφαπτομένης ($\varepsilon$) στο σημείο $K$:
Ο συντελεστής διεύθυνσης είναι $f'(2) = (2-2)^2 = 0$.
Επομένως, η εφαπτομένη είναι η οριζόντια ευθεία που διέρχεται από το $K$:
\[ \varepsilon: y = \frac{2}{3} \]
Για τον υπολογισμό του εμβαδού στο διάστημα $[0, 2]$, χρειαζόμαστε το πρόσημο της διαφοράς $f(x) - y_{\varepsilon} = f(x) - \frac{2}{3}$.
Από το ερώτημα (γ), γνωρίζουμε ότι:
\[ f(x) - \frac{2}{3} = \frac{1}{3}(x-2)^3 \]
Για $x \in [0, 2]$, έχουμε $x \le 2 \implies x-2 \le 0 \implies (x-2)^3 \le 0$, άρα η $C_f$ βρίσκεται "κάτω" από την εφαπτομένη.
Το ζητούμενο εμβαδόν είναι:
\begin{align*}
E &= \int_0^2 \left| f(x) - \frac{2}{3} \right| \d x = \int_0^2 - \left( f(x) - \frac{2}{3} \right) \d x \\
&= \int_0^2 - \frac{1}{3}(x-2)^3 \d x = - \frac{1}{3} \int_0^2 (x-2)^3 (x-2)' \d x \\
&= - \frac{1}{3} \left[ \frac{(x-2)^4}{4} \right]_0^2 \\
&= - \frac{1}{3} \left( \frac{(2-2)^4}{4} - \frac{(0-2)^4}{4} \right) \\
&= - \frac{1}{3} \left( 0 - \frac{16}{4} \right) = - \frac{1}{3} (-4) = \boldsymbol{\frac{4}{3} \ \text{τ.μ.}}
\end{align*}
\end{erwthma}
\end{thema}
