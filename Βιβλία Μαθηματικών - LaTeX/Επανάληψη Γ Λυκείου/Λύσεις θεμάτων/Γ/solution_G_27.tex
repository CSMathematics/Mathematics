%=========== ΛΥΣΗ - ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f$ είναι πολυωνυμική, άρα είναι συνεχής και παραγωγίσιμη στο $\mathbb{R}$.
\[ f'(x) = x^2 - 2\lambda x + 4 \]
Η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$ αν και μόνο αν $f'(x) \ge 0$ για κάθε $x \in \mathbb{R}$ και $f'(x) = 0$ μόνο για πεπερασμένο αριθμό σημείων.
Επειδή η $f'(x)$ είναι τριγωνική με θετικό συντελεστή του $x^2$, η συνθήκη $f'(x) \ge 0$ για κάθε $x \in \mathbb{R}$ ισχύει αν και μόνο αν η διακρίνουσα $\Delta$ του τριωνύμου είναι μη θετική:
\[ \Delta = (-2\lambda)^2 - 4 \cdot 1 \cdot 4 = 4\lambda^2 - 16 \]
\[ \Delta \le 0 \Leftrightarrow 4\lambda^2 \le 16 \Leftrightarrow \lambda^2 \le 4 \Leftrightarrow \lambda \in [-2, 2] \]
Άρα, το σύνολο των τιμών της παραμέτρου $\lambda$ είναι το διάστημα $[-2, 2]$.

\item Για $\lambda = 2$, η συνάρτηση είναι $f(x) = \dfrac{1}{3}x^3 - 2x^2 + 4x - 2$.
Η πρώτη παράγωγος είναι $f'(x) = x^2 - 4x + 4 = (x-2)^2$.
Η δεύτερη παράγωγος είναι:
\[ f''(x) = 2x - 4 \]
Για τον προσδιορισμό του σημείου καμπής εξετάζουμε το πρόσημο της $f''(x)$:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow 2x - 4 = 0 \Leftrightarrow x = 2$
    \item $f''(x) > 0 \Leftrightarrow x > 2$
    \item $f''(x) < 0 \Leftrightarrow x < 2$
\end{itemize}
Επειδή η $f''(x)$ αλλάζει πρόσημο στο $x=2$, η συνάρτηση έχει σημείο καμπής στο $x=2$.
Η αντίστοιχη τιμή της συνάρτησης είναι:
\[ f(2) = \frac{1}{3}(2)^3 - 2(2)^2 + 4(2) - 2 = \frac{8}{3} - 8 + 8 - 2 = \frac{8}{3} - \frac{6}{3} = \frac{2}{3} \]
Άρα, το μοναδικό σημείο καμπής της γραφικής παράστασης είναι το σημείο $(2, 2/3)$.

\item Για $\lambda = 2$, έχουμε $f'(x) = (x-2)^2$. Το ζητούμενο όριο είναι:
\[ \lim_{x \to 2} \frac{f'(x)}{x-2} = \lim_{x \to 2} \frac{(x-2)^2}{x-2} = \lim_{x \to 2} (x-2) = 2 - 2 = 0 \]

\item Η συνάρτηση $f$ είναι πολυωνυμική, οπότε το όριο στο $-\infty$ καθορίζεται από τον όρο με τον μεγαλύτερο βαθμό:
\[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} \left( \frac{1}{3}x^3 - \lambda x^2 + 4x - 2 \right) = \lim_{x \to -\infty} \frac{1}{3}x^3 = -\infty \]
Θέτοντας $u = f(x)$, όταν $x \to -\infty$ τότε $u \to -\infty$.
Συνεπώς:
\[ \lim_{x \to -\infty} f(f(x)) = \lim_{u \to -\infty} f(u) = \lim_{u \to -\infty} \left( \frac{1}{3}u^3 - \lambda u^2 + 4u - 2 \right) = -\infty \]
\end{erwthma}
\end{thema}
