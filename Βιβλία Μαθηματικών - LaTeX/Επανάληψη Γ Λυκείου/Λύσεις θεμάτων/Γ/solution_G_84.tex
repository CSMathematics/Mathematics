%=========== ΛΥΣΗ - 84ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη γραφική παράσταση παρατηρούμε ότι:
\begin{itemize}
\item στο διάστημα $[0,2]$ η καμπύλη στρέφει τα κοίλα προς τα πάνω, άρα η $f$ είναι κυρτή
\item στο διάστημα $[2,4]$ η καμπύλη στρέφει τα κοίλα προς τα κάτω, άρα η $f$ είναι κοίλη
\end{itemize}
Επομένως η κυρτότητα αλλάζει στο σημείο
\[
K(2,2)
\]
Άρα το $K(2,2)$ είναι σημείο καμπής της $C_f$.

\item Η εφαπτομένη της $C_f$ στο $K(2,2)$ έχει συντελεστή διεύθυνσης
\[
f'(2)=2
\]
Άρα η εξίσωσή της είναι
\begin{align*}
y-2&=2(x-2)\\
y&=2x-2
\end{align*}

Από το σχήμα έχουμε
\[
f(0)=0
\quad \text{και}\quad
f(4)=4
\]
Η $f$ είναι συνεχής στο $[0,4]$ και παραγωγίσιμη στο $(0,4)$.
Σύμφωνα με το Θεώρημα Μέσης Τιμής, υπάρχει $c\in(0,4)$ τέτοιο ώστε
\[
f'(c)=\frac{f(4)-f(0)}{4-0}
\]
Άρα
\[
f'(c)=\frac{4-0}{4}=1
\]

\item Επειδή η $f$ είναι μη αρνητική στο $[0,4]$ και από το σχήμα είναι θετική στο εσωτερικό του διαστήματος $(0,2)$, έχουμε
\[
\dint_0^2 f(x)\d x>0
\]

Επίσης, από το σχήμα, για κάθε $x\in[2,4)$ ισχύει
\[
f(x)<4
\]
και $f(4)=4$.
Άρα
\[
\dint_2^4 f(x)\d x<\dint_2^4 4\d x=8
\]
οπότε
\[
\dint_2^4 f(x)\d x<8
\]

\item Η χορδή $A\Delta$ ενώνει τα σημεία
\[
A(0,0)
\quad \text{και}\quad
\Delta(4,4)
\]
άρα έχει εξίσωση
\[
y=x
\]
Από το σχήμα παρατηρούμε ότι η $C_f$ βρίσκεται κάτω από τη χορδή στο διάστημα $(0,2)$.
Άρα
\[
f(x)<x\ ,\quad x\in(0,2)
\]
Ολοκληρώνοντας στο $[0,2]$, παίρνουμε
\[
\dint_0^2 f(x)\d x<\dint_0^2 x\d x=2
\]
δηλαδή
\[
\dint_0^2 f(x)\d x<2
\]

Αντίστοιχα, από το σχήμα η $C_f$ βρίσκεται πάνω από τη χορδή στο διάστημα $(2,4)$.
Άρα
\[
f(x)>x\ ,\quad x\in(2,4)
\]
Ολοκληρώνοντας στο $[2,4]$, παίρνουμε
\[
\dint_2^4 f(x)\d x>\dint_2^4 x\d x
\]
και
\[
\dint_2^4 x\d x=\left[\frac{x^2}{2}\right]_2^4=8-2=6
\]
Επομένως
\[
\dint_2^4 f(x)\d x>6
\]
\end{erwthma}
\end{thema}
