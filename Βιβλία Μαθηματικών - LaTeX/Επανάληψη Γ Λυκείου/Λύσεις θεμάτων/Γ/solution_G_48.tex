\begin{thema}{Γ}
\begin{erwthma}
\item Από την περιγραφή της γραφικής παράστασης της $f''$ προκύπτει ότι
\[f''(x)=
\begin{cases}
1, & x<2\\
0, & x>2
\end{cases}\]
Στο $x=2$ δεν δίνεται τιμή της $f''$, αφού η $f$ είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}\setminus\{2\}$.
Εργαζόμαστε λοιπόν στα διαστήματα $(-\infty,2)$ και $(2,+\infty)$ και στη συνέχεια χρησιμοποιούμε τη συνέχεια της $f'$ στο $x=2$, επειδή η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$.

\item Για $x<2$ έχουμε $f''(x)=1$, άρα
\[f'(x)=x+c_1\]
Επειδή $f'(0)=0$ και $0<2$, προκύπτει
\[0=f'(0)=0+c_1\Rightarrow c_1=0\]
οπότε
\[f'(x)=x\ ,\ x<2\]
Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, άρα η $f'$ είναι συνεχής στο $x=2$. Επομένως
\[f'(2)=\lim\limits_{x\to 2^-}f'(x)=\lim\limits_{x\to 2^-}x=2\]
Για $x>2$ έχουμε $f''(x)=0$, άρα η $f'$ είναι σταθερή στο $(2,+\infty)$. Αφού πρέπει να συμφωνεί με την τιμή $f'(2)=2$, παίρνουμε
\[f'(x)=2\ ,\ x>2\]
Άρα
\[f'(x)=
\begin{cases}
x, & x<2\\
2, & x\geq 2
\end{cases}\]
Η $f'$ είναι γνησίως αύξουσα στο $(-\infty,2)$ και σταθερή στο $[2,+\infty)$.
Επειδή
\[\lim\limits_{x\to-\infty}f'(x)=-\infty\]
η $f'$ δεν παρουσιάζει ελάχιστο στο $\mathbb{R}$.

\item Για $x<2$ έχουμε
\[f'(x)=x\]
άρα
\[f(x)=\frac{x^2}{2}+c_2\]
Από τη σχέση $f(0)=0$ προκύπτει
\[0=f(0)=0+c_2\Rightarrow c_2=0\]
οπότε
\[f(x)=\frac{x^2}{2}\ ,\ x<2\]
Επειδή η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, είναι και συνεχής στο $x=2$. Άρα
\[f(2)=\lim\limits_{x\to 2^-}f(x)=\frac{2^2}{2}=2\]
Για $x>2$ έχουμε
\[f'(x)=2\]
άρα
\[f(x)=2x+c_3\]
Από τη συνέχεια στο $x=2$ παίρνουμε
\[2=f(2)=2\cdot 2+c_3\Rightarrow c_3=-2\]
οπότε
\[f(x)=2x-2\ ,\ x>2\]
Επομένως
\[f(x)=
\begin{cases}
\dfrac{x^2}{2}, & x<2\\
2, & x=2\\
2x-2, & x>2
\end{cases}\]
και η $f$ είναι συνεχής στο $x=2$, αφού
\[\lim\limits_{x\to 2^-}f(x)=2=\lim\limits_{x\to 2^+}f(x)=f(2)\]

Παρατηρούμε ότι η συνάρτηση που προκύπτει δεν είναι δύο φορές παραγωγίσιμη στο $x=2$, επειδή η $f'$ δεν είναι παραγωγίσιμη στο $2$, όπως επιτρέπεται από τη διορθωμένη υπόθεση.

\item Από τον τύπο της $f'$ έχουμε
\begin{itemize}
\item $f'(x)<0\Leftrightarrow x<0$
\item $f'(x)=0\Leftrightarrow x=0$
\item $f'(x)>0\Leftrightarrow x>0$
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{ , -, z, +, }
\tkzTabLine{ , \searrow,t ,\nearrow,}
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(-\infty,0]$ και \textbf{γνησίως αύξουσα} στο $[0,+\infty)$.
Επομένως παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[f(0)=0\]
\end{erwthma}
\end{thema}
