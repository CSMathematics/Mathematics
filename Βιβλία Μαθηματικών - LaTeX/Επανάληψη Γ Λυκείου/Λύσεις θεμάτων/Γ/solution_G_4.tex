\begin{thema}{Γ}
\begin{erwthma}
\item Παρατηρούμε ότι $\left(f^2(x)\right)'=2f(x)f'(x)=2e^{2x}=\left(e^{2x}\right)'$ για κάθε $x\in\mathbb{R}$. Σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε:
\[f^2(x)=e^{2x}+c\]
Για $x=0$ προκύπτει $f^2(0)=e^0+c\Leftrightarrow \left(\sqrt{2}\right)^2=1+c\Leftrightarrow c=1$. Επομένως:
\[f^2(x)=e^{2x}+1\]
Τώρα, επειδή $f$ είναι παραγωγίσιμη (άρα συνεχής) με $f(0)=\sqrt{2}>0$, και επειδή $f(x)\neq 0$ (αφού $f^2(x)=e^{2x}+1>0$ παντού), από το Θεώρημα Τιμών Τόπου η $f$ δεν αλλάζει πρόσημο. Άρα $f(x)>0$ για κάθε $x\in\mathbb{R}$, οπότε:
\[f(x)=\sqrt{e^{2x}+1}\]

\item Παραγωγίζουμε τον τύπο της $f$:
\[f'(x)=\frac{\left(e^{2x}+1\right)'}{2\sqrt{e^{2x}+1}}=\frac{2e^{2x}}{2\sqrt{e^{2x}+1}}=\frac{e^{2x}}{\sqrt{e^{2x}+1}}\]
Επειδή $e^{2x}>0$ και $\sqrt{e^{2x}+1}>0$ για κάθε $x\in\mathbb{R}$, ισχύει $f'(x)>0$ για κάθε $x\in\mathbb{R}$, άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Για το σύνολο τιμών, η $f$ είναι συνεχής και γνησίως αύξουσα, άρα $f(\mathbb{R})=\left(\lim\limits_{x\to-\infty}f(x),\,\lim\limits_{x\to+\infty}f(x)\right)$. Υπολογίζουμε:
\begin{itemize}
\item $\lim\limits_{x\to-\infty}f(x)=\lim\limits_{x\to-\infty}\sqrt{e^{2x}+1}=\sqrt{0+1}=1$ \quad (αφού $e^{2x}\to 0$)
\item $\lim\limits_{x\to+\infty}f(x)=\lim\limits_{x\to+\infty}\sqrt{e^{2x}+1}=+\infty$
\end{itemize}
Επομένως το σύνολο τιμών της $f$ είναι $(1,+\infty)$.

\item Από το ερώτημα Γ2, η $f$ είναι γνησίως αύξουσα άρα και \textbf{αντιστρέψιμη}. Η αντίστροφη $f^{-1}:(1,+\infty)\to\mathbb{R}$ βρίσκεται λύνοντας ως προς $x$ την εξίσωση $y=\sqrt{e^{2x}+1}$, με $y>1$:
\[y^2=e^{2x}+1 \Leftrightarrow e^{2x}=y^2-1 \Leftrightarrow 2x=\ln(y^2-1)\Leftrightarrow x=\frac{1}{2}\ln(y^2-1)\]
(Σημειώνουμε ότι $y>1\Rightarrow y^2>1\Rightarrow y^2-1>0$, οπότε ο λογάριθμος ορίζεται.) Επομένως:
\[f^{-1}(x)=\frac{1}{2}\ln(x^2-1),\quad x\in(1,+\infty)\]

\item Έστω $x(t)$ η τετμημένη του $M$ τη χρονική στιγμή $t$. Το τρίγωνο $OAM$ έχει κορυφές $O(0,0)$, $A(x(t),0)$ και $M(x(t),f(x(t)))$. Η βάση του είναι $|OA|=|x(t)|$ και το ύψος του $f(x(t))>0$, άρα το εμβαδόν του είναι:
\[E(t)=\frac{1}{2}\,|x(t)|\cdot f(x(t))\]
Επειδή $x'(t)=2>0$, η τετμημένη αυξάνεται, οπότε η $f$ στη γειτονιά του $t_0$ κινείται από αρνητικές σε θετικές τιμές. Για $x>0$ είναι $|x|=x$, άρα:
\[E(t)=\frac{1}{2}\,x(t)\cdot f(x(t))=\frac{1}{2}\,x(t)\cdot\sqrt{e^{2x(t)}+1}\]
Παραγωγίζουμε ως προς $t$ με τον κανόνα της αλυσίδας:
\[E'(t)=\frac{1}{2}\left(x'(t)\cdot\sqrt{e^{2x}+1}+x\cdot\frac{e^{2x}}{\sqrt{e^{2x}+1}}\cdot x'(t)\right)=\frac{1}{2}\cdot x'(t)\cdot\left(\sqrt{e^{2x}+1}+\frac{xe^{2x}}{\sqrt{e^{2x}+1}}\right)\]
Έστω $t_0$ η χρονική στιγμή που το $M$ διέρχεται από τον άξονα $y'y$, οπότε $x(t_0)=0$ και $x'(t)\big|_{t_0}=2\,\mu/\si{sec}$. Τη χρονική στιγμή $t_0$:
\[E'(t_0)=\frac{1}{2}\cdot 2\cdot\left(\sqrt{e^{0}+1}+0\right)=1\cdot\sqrt{2}=\sqrt{2}\,\mu^2/\si{sec}\]
\end{erwthma}
\end{thema}
