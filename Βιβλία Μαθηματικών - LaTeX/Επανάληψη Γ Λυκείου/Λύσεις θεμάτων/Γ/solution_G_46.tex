\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι δύο φορές παραγωγίσιμη στο $\mathbb{R}$ με
\[f'(x)=e^{x-1}+2x\]
και
\[f''(x)=e^{x-1}+2\]
Για κάθε $x\in\mathbb{R}$ ισχύει $e^{x-1}>0$, άρα
\[f''(x)=e^{x-1}+2>0\]
Επομένως η $f$ είναι \textbf{κυρτή} στο $\mathbb{R}$.

\item Για $x_0=1$ έχουμε
\[f(1)=e^{1-1}+1^2-2=1+1-2=0\]
και
\[f'(1)=e^{1-1}+2\cdot 1=1+2=3\]
Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο $A(1,f(1))\equiv A(1,0)$ είναι
\begin{align*}
y-f(1)&=f'(1)(x-1)\\
y&=3(x-1)\\
y&=3x-3
\end{align*}

\item Έχουμε
\begin{align*}
f(x)-3x+3&=e^{x-1}+x^2-2-3x+3\\
&=e^{x-1}+x^2-3x+1
\end{align*}
Για $x\to 1$ έχουμε
\[e^{x-1}+x^2-3x+1\to 0\]
και
\[\ln^2 x\to 0\]
Άρα το ζητούμενο όριο είναι της μορφής $\dfrac{0}{0}$.
Με δύο εφαρμογές του κανόνα \eng{L'Hôpital} προκύπτει:
\begin{align*}
\lim\limits_{x\to 1}\frac{f(x)-3x+3}{\ln^2 x}
&=\lim\limits_{x\to 1}\frac{e^{x-1}+2x-3}{\frac{2\ln x}{x}}\\
&=\lim\limits_{x\to 1}\frac{e^{x-1}+2}{\frac{2(1-\ln x)}{x^2}}\\
&=\frac{1+2}{2}\\
&=\frac{3}{2}
\end{align*}

\item Θεωρούμε τη συνάρτηση
\[g(x)=f(x)+xf'(x)\ ,\ x\in[0,1]\]
Η $g$ είναι συνεχής στο $[0,1]$ ως άθροισμα και γινόμενο συνεχών συναρτήσεων.
Υπολογίζουμε τις τιμές της στα άκρα:
\[g(0)=f(0)+0\cdot f'(0)=e^{-1}-2<0\]
και
\[g(1)=f(1)+1\cdot f'(1)=0+3=3>0\]
Επομένως
\[g(0)g(1)<0\]
Άρα, από το θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα $\xi\in(0,1)$ τέτοιο ώστε
\[g(\xi)=0\]
δηλαδή
\[f(\xi)+\xi f'(\xi)=0\]
Επομένως η εξίσωση $f(x)+xf'(x)=0$ έχει τουλάχιστον μία ρίζα στο $(0,1)$.
\end{erwthma}
\end{thema}
