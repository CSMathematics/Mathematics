%=========== ΛΥΣΗ - 9ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έχουμε
\[
f'(x)=e^x-x-1
\]
και
\[
f''(x)=e^x-1.
\]
Άρα:
\begin{itemize}
\item $f''(x)=0 \Leftrightarrow x=0$
\item $f''(x)<0 \Leftrightarrow x<0$
\item $f''(x)>0 \Leftrightarrow x>0$
\end{itemize}
Επομένως η $f$ είναι κοίλη στο $(-\infty,0)$ και κυρτή στο $(0,+\infty)$.

\item Εφόσον
\[
f''(x)<0 \text{ για } x<0,
\]
η $f'$ είναι γνησίως φθίνουσα στο $(-\infty,0)$, ενώ επειδή
\[
f''(x)>0 \text{ για } x>0,
\]
η $f'$ είναι γνησίως αύξουσα στο $(0,+\infty)$.
Ακόμη,
\[
f'(0)=e^0-0-1=0.
\]
Άρα η $f'$ παρουσιάζει ελάχιστο στο $x=0$ και επομένως
\[
f'(x)\ge 0,\quad x\in \mathbb{R},
\]
με ισότητα μόνο για $x=0$.

\item Αφού
\[
f'(x)\ge 0,\quad x\in \mathbb{R},
\]
η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$.
Ειδικότερα, για $x\ge 0$ έχουμε
\[
f(x)\ge f(0).
\]
Επειδή
\[
f(0)=e^0-\frac{0^2}{2}-0-1=0,
\]
παίρνουμε
\[
e^x-\frac{x^2}{2}-x-1\ge 0,\quad x\ge 0
\]
δηλαδή
\[
e^x\ge \frac{x^2}{2}+x+1,\quad x\ge 0.
\]

\item Από το προηγούμενο ερώτημα, για κάθε $x\in[0,1]$ ισχύει
\[
e^x\ge \frac{x^2}{2}+x+1.
\]
Ολοκληρώνοντας στο $[0,1]$, παίρνουμε
\[
\int_0^1 e^x\,\d x
\ge
\int_0^1\left(\frac{x^2}{2}+x+1\right)\,\d x.
\]
Άρα
\begin{align*}
\int_0^1 e^x\,\d x
&\ge
\left[\frac{x^3}{6}+\frac{x^2}{2}+x\right]_0^1\\
&=
\frac{1}{6}+\frac{1}{2}+1\\
&=
\frac{5}{3}.
\end{align*}
Επομένως
\[
\int_0^1 e^x\,\d x>\frac{5}{3}.
\]
\end{erwthma}
\end{thema}
