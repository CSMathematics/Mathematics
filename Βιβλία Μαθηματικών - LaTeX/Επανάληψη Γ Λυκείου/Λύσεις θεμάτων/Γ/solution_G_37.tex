%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.37 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Δίνεται η συνάρτηση $f(x) = \frac{\sqrt{x^2+1} - ax}{x}$ για $x > 0$.
Ζητείται το όριο στο $+\infty$. Μοιράζουμε τον παρονομαστή:
\[ f(x) = \frac{\sqrt{x^2+1}}{x} - a \]
Επειδή $x > 0$, μπορούμε να εισάγουμε το $x$ μέσα στη ρίζα υψώνοντάς το στο τετράγωνο:
\[ f(x) = \sqrt{\frac{x^2+1}{x^2}} - a = \sqrt{1 + \frac{1}{x^2}} - a \]
Όταν το $x \to +\infty$, το κλάσμα $\frac{1}{x^2} \to 0$. Επομένως:
\[ \lim_{x\to +\infty} f(x) = \lim_{x\to +\infty} \left( \sqrt{1 + \frac{1}{x^2}} - a \right) = \sqrt{1 + 0} - a = \boldsymbol{1 - a} \]

\item Για να αποτελεί ο άξονας $x'x$ (ήτοι η ευθεία $y = 0$) οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$, πρέπει εξ ορισμού να ισχύει:
\[ \lim_{x\to +\infty} f(x) = 0 \]
Χρησιμοποιώντας το αποτέλεσμα του πρώτου ερωτήματος, έχουμε άμεσα:
\[ 1 - a = 0 \Rightarrow \boldsymbol{a = 1} \]

\item Αντικαθιστούμε $a = 1$. Ο τύπος της συνάρτησης διαμορφώνεται σε $f(x) = \frac{\sqrt{x^2+1} - x}{x}$.
Ζητείται το όριο $\lim_{x\to +\infty} [x f(x)]$.
Πολλαπλασιάζουμε με το $x$:
\[ x f(x) = x \frac{\sqrt{x^2+1} - x}{x} = \sqrt{x^2+1} - x \]
Καθώς $x \to +\infty$, η παράσταση οδηγεί σε απροσδιοριστία μορφής $(+\infty - \infty)$. Πολλαπλασιάζουμε και διαιρούμε με τη συζυγή παράσταση:
\begin{align*}
\sqrt{x^2+1} - x &= \frac{\left( \sqrt{x^2+1} - x \right)\left( \sqrt{x^2+1} + x \right)}{\sqrt{x^2+1} + x} \\
&= \frac{(\sqrt{x^2+1})^2 - x^2}{\sqrt{x^2+1} + x} \\
&= \frac{x^2 + 1 - x^2}{\sqrt{x^2+1} + x} = \frac{1}{\sqrt{x^2+1} + x}
\end{align*}
Το νέο όριο είναι άμεσα υπολογίσιμο:
\[ \lim_{x\to +\infty} \frac{1}{\sqrt{x^2+1} + x} = \frac{1}{+\infty} = \boldsymbol{0} \]

\item Με $a = 1$, ζητείται το όριο $\lim_{x\to +\infty} \ln(f(x))$.
Από τα προηγούμενα βήματα ξέρουμε ότι διαιρώντας το ρητοποιημένο $x f(x)$ με το $x$, η συνάρτηση είναι αυστηρά:
\[ f(x) = \frac{\sqrt{x^2+1} - x}{x} = \frac{1}{x \left(\sqrt{x^2+1} + x\right)} \]
Από αυτή τη μορφή προκύπτει καταφανέστατα ότι καθώς $x \to +\infty$, ο παρονομαστής τείνει στο $+\infty$, άρα:
\[ \lim_{x\to +\infty} f(x) = \frac{1}{+\infty} = 0 \]
Και κυρίως επειδή $x > 0$, ολόκληρο το κλάσμα είναι σταθερά θετικό, οπότε το όριο είναι $0$ «από τα θετικά» ($0^+$).
Χάρη στον κανόνα σύνθεσης εσωτερικού ορίου της λογαριθμικής συνάρτησης, έχουμε:
\[ \lim_{u\to 0^+} \ln u = -\infty \]
Άρα εν τέλει:
\[ \lim_{x\to +\infty} \ln(f(x)) = \boldsymbol{-\infty} \]

\end{erwthma}
\end{thema}
