%=========== ΛΥΣΗ - 37ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $x > 0$, μπορούμε να εισάγουμε το $x$ μέσα στη ρίζα (αφού $x > 0 \Rightarrow x = \sqrt{x^2}$):
\[ f(x) = \frac{\sqrt{x^2+1} - ax}{x} = \frac{\sqrt{x^2+1}}{x} - a = \sqrt{\frac{x^2+1}{x^2}} - a = \sqrt{1 + \frac{1}{x^2}} - a \]
Καθώς $x \to +\infty$, το $\dfrac{1}{x^2} \to 0$, επομένως:
\[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \left( \sqrt{1 + \frac{1}{x^2}} - a \right) = \sqrt{1+0} - a = \boldsymbol{1 - a} \]

\item Για να αποτελεί ο άξονας $x'x$ (ευθεία $y=0$) οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$, πρέπει:
\[ \lim_{x \to +\infty} f(x) = 0 \]
Από το προηγούμενο ερώτημα, $\lim_{x \to +\infty} f(x) = 1 - a$. Άρα:
\[ 1 - a = 0 \implies \boldsymbol{a = 1} \]

\item Για $a = 1$, ο τύπος της συνάρτησης γίνεται $f(x) = \dfrac{\sqrt{x^2+1} - x}{x}$, $x > 0$. Ζητείται το $\lim\limits_{x \to +\infty} [x f(x)]$.
\[ x f(x) = x \cdot \frac{\sqrt{x^2+1} - x}{x} = \sqrt{x^2+1} - x \]
Η παράσταση $\sqrt{x^2+1} - x$ έχει μορφή $+\infty - \infty$ για $x \to +\infty$.
Πολλαπλασιάζουμε με τη \textbf{συζυγή παράσταση} $(\sqrt{x^2+1} + x)$:
\begin{align*}
\sqrt{x^2+1} - x &= \frac{(\sqrt{x^2+1} - x)(\sqrt{x^2+1} + x)}{\sqrt{x^2+1} + x} \\
&= \frac{(x^2+1) - x^2}{\sqrt{x^2+1} + x} = \frac{1}{\sqrt{x^2+1} + x}
\end{align*}
Τώρα το όριο είναι εύκολα υπολογίσιμο:
\[ \lim_{x \to +\infty} [x f(x)] = \lim_{x \to +\infty} \frac{1}{\sqrt{x^2+1} + x} = \frac{1}{+\infty} = \boldsymbol{0} \]

\item Για $a = 1$, έχουμε $f(x) = \dfrac{\sqrt{x^2+1}-x}{x}$ με $x > 0$.
Από το Γ1, γνωρίζουμε ότι $f(x) = \sqrt{1+\frac{1}{x^2}} - 1$.

Παραγωγίζουμε για να μελετήσουμε τη μονοτονία:
\[ f'(x) = \left( \sqrt{1+\frac{1}{x^2}} \right)' = \frac{1}{2\sqrt{1+\frac{1}{x^2}}} \cdot \left( -\frac{2}{x^3} \right) = \frac{-\frac{1}{x^3}}{\sqrt{1+\frac{1}{x^2}}} \]
Εφόσον $x > 0$, ο αριθμητής $-\frac{1}{x^3} < 0$ και ο παρονομαστής $> 0$, άρα $f'(x) < 0$ για κάθε $x > 0$.
Επομένως, η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0, +\infty)$, άρα είναι \textbf{1-1} και \textbf{αντιστρέφεται}.

Για το πεδίο ορισμού της $f^{-1}$, χρειαζόμαστε το σύνολο τιμών της $f$:
\begin{itemize}
    \item $\lim\limits_{x \to 0^+} f(x) = \lim\limits_{x \to 0^+} \left( \sqrt{1+\frac{1}{x^2}} - 1 \right) = +\infty$ \quad (αφού $\frac{1}{x^2} \to +\infty$)
    \item $\lim\limits_{x \to +\infty} f(x) = 1 - 1 = 0$ \quad (από το Γ1 με $a=1$)
\end{itemize}
Επομένως, το σύνολο τιμών της $f$ είναι το $(0, +\infty)$ και άρα:
\[ \boldsymbol{A_{f^{-1}} = (0, +\infty)} \]
\end{erwthma}
\end{thema}
