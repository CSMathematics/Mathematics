%=========== ΛΥΣΗ - 2ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για να είναι η $f$ παραγωγίσιμη στο $0$, πρέπει πρώτα να είναι συνεχής στο $0$.
Άρα
\[
\lim_{x\to 0^-}f(x)=\lim_{x\to 0^+}f(x)
\]
δηλαδή
\[
\beta=a.
\]
Ακόμη,
\[
f'(x)=
\begin{cases}
2x,& x>0\\
e^x+xe^x-1,& x<0
\end{cases}
\]
οπότε
\[
\lim_{x\to 0^+}f'(x)=0
\]
και
\[
\lim_{x\to 0^-}f'(x)=1+0-1=0.
\]
Άρα η συνάρτηση είναι παραγωγίσιμη στο $0$ όταν $a=\beta$.
Για τη συγκεκριμένη συνάρτηση προκύπτει
\[
a=\beta=-1.
\]

\item Για $x<0$ έχουμε
\[
f(x)=xe^x-x+\beta,\qquad f'(x)=e^x+xe^x-1.
\]
Άρα
\[
\frac{f(x)}{xf'(x)}
=
\frac{xe^x-x+\beta}{x(e^x+xe^x-1)}
=
\frac{e^x-1+\frac{\beta}{x}}{e^x+xe^x-1}.
\]
Επειδή
\[
e^x\to 0,\qquad xe^x\to 0,\qquad \frac{\beta}{x}\to 0 \quad \text{καθώς } x\to -\infty,
\]
παίρνουμε
\[
\lim_{x\to -\infty}\frac{f(x)}{xf'(x)}
=
\frac{-1}{-1}=1.
\]

\item Για $x>0$ ισχύει
\[
f''(x)=2>0.
\]
Για $x<0$ ισχύει
\[
f''(x)=e^x+e^x+xe^x=e^x(x+2).
\]
Επειδή $e^x>0$, το πρόσημο της $f''(x)$ ταυτίζεται με το πρόσημο του $x+2$:
\begin{itemize}
\item $f''(x)=0 \Leftrightarrow x=-2$
\item $f''(x)<0 \Leftrightarrow x<-2$
\item $f''(x)>0 \Leftrightarrow x>-2$
\end{itemize}
Άρα η $f$ είναι κοίλη στο $(-\infty,-2)$ και κυρτή στο $(-2,+\infty)$, οπότε παρουσιάζει σημείο καμπής για $x=-2$.

Με $\beta=-1$:
\[
f(-2)=-2e^{-2}-(-2)-1=1-2e^{-2}.
\]
Επομένως το σημείο καμπής είναι το
\[
M\left(-2,1-2e^{-2}\right).
\]

\item Η ευθεία $\zeta:y=x+1$ έχει συντελεστή διεύθυνσης $1$, άρα η ζητούμενη εφαπτομένη έχει συντελεστή διεύθυνσης $-1$.
Λύνουμε λοιπόν την εξίσωση
\[
f'(x)=-1.
\]
Για $x\ge 0$:
\[
2x=-1
\]
αδύνατο.

Για $x<0$:
\[
e^x+xe^x-1=-1 \Leftrightarrow e^x(1+x)=0 \Leftrightarrow x=-1.
\]
Άρα το σημείο επαφής είναι το
\[
A\left(-1,f(-1)\right).
\]
Με $\beta=-1$:
\[
f(-1)=-e^{-1}+1-1=-\frac{1}{e}.
\]
Η εξίσωση της εφαπτομένης είναι
\[
y+\frac{1}{e}=-1(x+1)
\]
οπότε
\[
y=-x-1-\frac{1}{e}.
\]
\end{erwthma}
\end{thema}
