%=========== ΛΥΣΗ - 75ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
f(x)f'(x)=\frac{\ln x}{x}
\]
πολλαπλασιάζουμε με $2$ και παίρνουμε
\[
2f(x)f'(x)=2\frac{\ln x}{x}
\]
Όμως
\[
\left(f^2(x)\right)'=2f(x)f'(x)
\]
και
\[
\left((\ln x)^2\right)'=2\ln x\cdot\frac{1}{x}=2\frac{\ln x}{x}
\]
Άρα
\[
\left(f^2(x)\right)'=\left((\ln x)^2\right)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
f^2(x)=(\ln x)^2+c
\]
Από τη συνθήκη $f(e)=1$ έχουμε
\[
f^2(e)=(\ln e)^2+c\Leftrightarrow 1=1+c\Leftrightarrow c=0
\]
Άρα
\[
f^2(x)=(\ln x)^2
\]
Επειδή $x>1$, είναι $\ln x>0$, ενώ από την εκφώνηση γνωρίζουμε ότι $f(x)>0$.
Επομένως
\[
f(x)=\ln x\ ,\quad x\in(1,+\infty)
\]

\item Θεωρούμε τη συνάρτηση
\[
\varphi(x)=\frac{\ln x}{x}\ ,\quad x>1
\]
Η $\varphi$ είναι παραγωγίσιμη στο $(1,+\infty)$ και
\[
\varphi'(x)=\frac{(\ln x)'x-\ln x(x)'}{x^2}
=\frac{1-\ln x}{x^2}
\]
Επειδή $x^2>0$, το πρόσημο της $\varphi'$ καθορίζεται από το $1-\ln x$.
Έχουμε:
\begin{itemize}
\item $\varphi'(x)=0\Leftrightarrow \ln x=1\Leftrightarrow x=e$
\item $\varphi'(x)>0\Leftrightarrow 1<x<e$
\item $\varphi'(x)<0\Leftrightarrow x>e$
\end{itemize}
Άρα η $\varphi$ είναι γνησίως αύξουσα στο $(1,e]$ και γνησίως φθίνουσα στο $[e,+\infty)$.
Συνεπώς παρουσιάζει ολικό μέγιστο στο $x=e$, με τιμή
\[
\varphi(e)=\frac{1}{e}
\]

\item Επειδή
\[
e<\pi
\]
και η $\varphi$ είναι γνησίως φθίνουσα στο $[e,+\infty)$, έχουμε
\[
\varphi(e)>\varphi(\pi)
\]
δηλαδή
\[
\frac{1}{e}>\frac{\ln\pi}{\pi}
\]
Άρα
\[
\pi>e\ln\pi
\]
και, επειδή $\ln e=1$,
\[
\pi\ln e>e\ln\pi
\]
οπότε
\[
\ln(e^\pi)>\ln(\pi^e)
\]
Επειδή η συνάρτηση $\ln x$ είναι γνησίως αύξουσα στο $(0,+\infty)$, προκύπτει
\[
e^\pi>\pi^e
\]

\item Από το πρώτο ερώτημα έχουμε
\[
f(x)=\ln x
\]
Η $C_f$ τέμνει τον άξονα $x'x$ στο σημείο με τετμημένη $1$, αφού
\[
\ln x=0\Leftrightarrow x=1
\]
Επιπλέον, για $x\in[1,e]$ ισχύει
\[
\ln x\geq 0
\]
Άρα το ζητούμενο εμβαδόν είναι
\[
E=\dint_1^e \ln x\d x
\]
Ολοκληρώνουμε κατά παράγοντες:
\begin{align*}
E
&=\dint_1^e \ln x\d x\\
&=\left[x\ln x\right]_1^e-\dint_1^e x\cdot\frac{1}{x}\d x\\
&=e-\dint_1^e 1\d x\\
&=e-(e-1)\\
&=1
\end{align*}
Επομένως το εμβαδόν του χωρίου είναι $1$ τ.μ.
\end{erwthma}
\end{thema}
