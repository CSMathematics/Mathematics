%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.75 ===========
\begin{thema}{Γ}
\begin{erwthma}

\item Έχουμε την εξίσωση $f(x) f'(x) = \frac{\ln x}{x}$ για $x > 1$, με $f(x) > 0$.
Πολλαπλασιάζουμε και τα δύο μέλη επι 2 για να εμφανιστεί ο κανόνας της παραγώγου σύνθετης δύναμης:
\begin{align*}
2 f(x) f'(x) &= 2 \frac{\ln x}{x} \\
\left( [f(x)]^2 \right)' &= 2 \ln x \cdot (\ln x)' \\
\left( [f(x)]^2 \right)' &= \left( [\ln x]^2 \right)'
\end{align*}
Από το χώρο ολοκλήρωσης, συλλαμβάνουμε τη σταθερά:
\[ [f(x)]^2 = (\ln x)^2 + c \]
Για την εύρεση της $c$ χρησιμοποιούμε το αρχικό δεδομένο $f(e) = 1$:
\[ [f(e)]^2 = (\ln e)^2 + c \Rightarrow 1^2 = 1^2 + c \Rightarrow c = 0 \]
Άρα $[f(x)]^2 = (\ln x)^2$. Καθώς η εξίσωση αφορά την περιοχή $x > 1$ οπότε το $\ln x > 0$, και το δεδομένο αναφέρει ρητά ότι $f(x) > 0$, τα δύο μέλη είναι θετικά. Μπορούμε να εξάγουμε τετραγωνική ρίζα προσημασμένα:
\[ \mathbf{f(x) = \ln x} \quad \text{για κάθε } x \in (1, +\infty) \]

\item Μελέτη της συνάρτησης $\varphi(x) = \frac{\ln x}{x}$.
Παραγωγίζουμε με τον κανόνα πηλίκου:
\[ \varphi'(x) = \frac{(\ln x)' x - \ln x (x)'}{x^2} = \frac{\frac{1}{x}x - \ln x}{x^2} = \frac{1 - \ln x}{x^2} \]
Μηδενίζουμε την παράγωγο:
\[ \varphi'(x) = 0 \Leftrightarrow 1 - \ln x = 0 \Leftrightarrow \ln x = 1 \Leftrightarrow x = e \]
Το πρόσημο εξαρτάται από τον αριθμητή (αφού $x^2 > 0$):
- Αν $1 < x < e \Rightarrow \ln x < 1 \Rightarrow \varphi'(x) > 0$. Συνεπώς η $\varphi$ είναι \textbf{γνησίως αύξουσα} στο $(1, e]$.
- Αν $x > e \Rightarrow \ln x > 1 \Rightarrow \varphi'(x) < 0$. Έτσι η $\varphi$ είναι \textbf{γνησίως φθίνουσα} στο $[e, +\infty)$.
(Παρουσιάζει ολικό μέγιστο στο σημείο $x=e$, με τιμή $\varphi(e) = \frac{1}{e}$).

\item Εφαρμόζουμε την παραπάνω μελέτη μονοτονίας για να συγκρίνουμε τις εκθετικές ποσότητες.
Γνωρίζουμε την αρχή σχέσης αριθμών: $e \approx 2{,}71$ και $\pi \approx 3{,}14$. 
Είναι ξεκάθαρο ότι $\pi > e$. 
Και οι δύο αυτοί αριθμοί ανήκουν στο διάστημα $(e, +\infty)$ όπου η συνάρτηση $\varphi(x)$ είναι \textbf{γνησίως φθίνουσα}.
Λόγω της φθίνουσας μονοτονίας, η ανισότητα αντιστρέφεται αναφορικά με τη συνάρτηση:
\begin{align*}
\pi &> e \\
\varphi(\pi) &< \varphi(e) \\
\frac{\ln \pi}{\pi} &< \frac{\ln e}{e}
\end{align*}
Πολλαπλασιάζουμε χιαστί (αφού $\pi, e > 0$) για να απαλείψουμε τους παρονομαστές:
\begin{align*}
e \ln \pi &< \pi \ln e \\
\ln(\pi^e) &< \ln(e^\pi)
\end{align*}
Επειδή και η $\ln x$ είναι γνησίως αύξουσα συνάρτηση, οι ποσότητες εντός του λογαρίθμου διατηρούν τη σχέση ανισότητας ακέραιη:
\[ \mathbf{\pi^e < e^\pi} \]

\item Ζητείται το εμβαδόν του χωρίου μεταξύ της $C_f$ (την $f(x) = \ln x$), τον άξονα $x'x$ και την ευθεία $x = e$.
Άξονας $x'x$ σημαίνει ταύτιση με το κρυφό $y=0$, που τέμνει την $C_f$ στο $x=1$ (αφού $\ln 1 = 0$). Και το πάνω φράγμα δίνεται ως $x=e$. 
Επειδή στο $[1, e]$ η $f(x) \geq 0$, το εμβαδόν ταυτίζεται επακριβώς με το απλό ορισμένο ολοκλήρωμα:
\begin{align*}
E &= \int_1^e \ln x\,\d x = \int_1^e (x)' \ln x\,\d x \\
&= \left[ x \ln x \right]_1^e - \int_1^e x (\ln x)'\,\d x \\
&= (e \ln e - 1 \ln 1) - \int_1^e x \frac{1}{x}\,\d x \\
&= (e \cdot 1 - 0) - \int_1^e 1\,\d x \\
&= e - \left[ x \right]_1^e = e - (e - 1) = \boldsymbol{1 \ \text{τ.μ.}}
\end{align*}
Αποδείχθηκε το ζητούμενο.
\end{erwthma}
\end{thema}
