\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $x,y\in\mathbb{R}$ με $x\geq y$.
Από τη δοσμένη ανισότητα έχουμε
\[f(x)-f(y)\geq e^x-e^y\]
οπότε
\[f(x)-e^x\geq f(y)-e^y\]
Δηλαδή
\[g(x)\geq g(y)\]
Επομένως η συνάρτηση
\[g(x)=f(x)-e^x\]
είναι αύξουσα στο $\mathbb{R}$.

\item Αν η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, τότε και η
\[g(x)=f(x)-e^x\]
είναι παραγωγίσιμη στο $\mathbb{R}$.
Από το Γ1 η $g$ είναι αύξουσα, άρα
\[g'(x)\geq 0\ ,\ x\in\mathbb{R}\]
Επειδή
\[g'(x)=f'(x)-e^x\]
προκύπτει
\[f'(x)-e^x\geq 0\Rightarrow f'(x)\geq e^x\ ,\ x\in\mathbb{R}\]

\item Αν θέσουμε $y=0$ στη δοσμένη ανισότητα, πρέπει να ισχύει $x\geq 0$.
Για κάθε $x\geq 0$ λοιπόν, έχουμε
\[f(x)-f(0)\geq e^x-e^0\]
δηλαδή
\[f(x)-f(0)\geq e^x-1\]

\item Αν $f(0)=1$, τότε από το Γ3, για κάθε $x\geq 0$,
\[f(x)-1\geq e^x-1\]
άρα
\[f(x)\geq e^x\]
Για $x>0$ προκύπτει
\[\frac{f(x)}{x}\geq \frac{e^x}{x}\]
Όμως
\[\lim\limits_{x\to+\infty}\frac{e^x}{x}=+\infty\]
Επομένως, από σύγκριση,
\[\lim\limits_{x\to+\infty}\frac{f(x)}{x}=+\infty\]
\end{erwthma}
\end{thema}
