%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.62 ===========
\begin{thema}{Γ}
Δίνεται $f(x) - f(y) \geq e^x - e^y$ για κάθε $x \geq y$.
\begin{erwthma}

\item Έστω $x, y \in \mathbb{R}$ με $x \geq y$. Η δοθείσα ανισότητα αναδιατάσσεται διαχωρίζοντας τις μεταβλητές:
\begin{align*}
f(x) - f(y) &\geq e^x - e^y \\
f(x) - e^x &\geq f(y) - e^y
\end{align*}
Αντικαθιστώντας την έκφραση με τη γνωριμία της $g(t) = f(t) - e^t$, η παραπάνω ανισότητα ανάγεται σε:
\[ g(x) \geq g(y) \]
Ολοκληρώντας, αποδείξαμε ότι για κάθε $x \geq y$ συνεπάγεται $g(x) \geq g(y)$. Το γεγονός αυτό αποτελεί τον αυστηρό ορισμό της \textbf{αύξουσας} (μη φθίνουσας) συνάρτησης στο $\mathbb{R}$.

\item Εάν δίνεται ότι η $f$ είναι παραγωγίσιμη, τότε αυτόματα και η $g(x) = f(x) - e^x$ είναι παραγωγίσιμη, ως διαφορά παραγωγίσιμων συναρτήσεων.
Γνωρίζουμε από το (1) ότι η $g$ είναι αύξουσα. Από απόρροια του Θεωρήματος Μέσης Τιμής, εάν μια παραγωγίσιμη συνάρτηση είναι αύξουσα, η παράγωγός της δεν μπορεί να είναι αρνητική οπουδήποτε:
\[ g'(x) \geq 0 \quad \text{για κάθε } x \in \mathbb{R} \]
Παραγωγίζουμε την $g$:
\[ g'(x) = f'(x) - e^x \]
Αντικαθιστώντας:
\[ f'(x) - e^x \geq 0 \Rightarrow \mathbf{f'(x) \geq e^x} \]

GIA\_ENALLAKTIKH \textbf{Εναλλακτική με όριο:} Η παράγωγος είναι $\lim_{h\to 0^+} \frac{g(x+h) - g(x)}{h}$. Καθώς $h>0 \Rightarrow x+h > x$, κι αφού $g$ αύξουσα έχουμε $g(x+h) \ge g(x)$, οπότε το κλάσμα $\ge 0$. Το όριο λοιπόν είναι $\ge 0$.

\item Ζητείται $f(x) - f(0) \geq e^x - 1$.
Επειδή η αρχική δομή $f(x) - f(y) \geq e^x - e^y$ δόθηκε \textbf{συγκεκριμένα για $x \geq y$}, ο περιορισμός αυτός πρέπει να λαμβάνεται υπ'όψιν.
Για να θέσουμε $y = 0 \Rightarrow$ θα πρέπει $x \geq 0$.
αντικαθιστούμε $y=0$ στη δεδομένη:
\[ f(x) - f(0) \geq e^x - e^0 \Rightarrow \mathbf{f(x) - f(0) \geq e^x - 1} \quad \text{για κάθε } \mathbf{x \geq 0} \]

\begin{parat}{\linewidth}
\textbf{Σημείωση - Επέκταση αρνητικών:} Αν $x < 0$, τότε αναστρέφοντας τη σειρά ($0 \geq x$), βάζουμε $x_{\text{new}}=0$ και $y_{\text{new}}=x$, οπότε $f(0) - f(x) \geq 1 - e^x \Rightarrow f(x)-f(0) \leq e^x-1$. Άρα το αιτούμενο της άσκησης ισχύει μόνο στο δεξιό ημιάξονα! (Η άσκηση παρέλειψε να αναφέρει τον περιορισμό).
\end{parat}

\item Δίνεται επιπρόσθετα $f(0) = 1$. Αναζητούμε το $\lim_{x\to +\infty} \frac{f(x)}{x}$.
Από το ερώτημα 3 (αφού μιλάμε για όριο στο $+\infty$ το $x$ είναι θετικό):
\[ f(x) - f(0) \geq e^x - 1 \Rightarrow f(x) - 1 \geq e^x - 1 \Rightarrow f(x) \geq e^x \]
Διαιρούμε και τα δύο μέλη με το θετικό $x > 0$:
\[ \frac{f(x)}{x} \geq \frac{e^x}{x} \]
Υπολογίζουμε το συνοδεύον όριο:
\[ \lim_{x\to +\infty} \frac{e^x}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to +\infty} \frac{e^x}{1} = +\infty \]
Επειδή η συνάρτησή μας είναι μεγαλύτερη από μία άλλη που απειρίζεται, προκύπτει άμεσα μέσω του θεωρήματος \textbf{Σύγκρισης Άπειρων Ορίων (κριτήριο μονοτονίας στο άπειρο)}:
\[ \lim_{x\to +\infty} \frac{f(x)}{x} = \mathbf{+\infty} \]

\end{erwthma}
\end{thema}
