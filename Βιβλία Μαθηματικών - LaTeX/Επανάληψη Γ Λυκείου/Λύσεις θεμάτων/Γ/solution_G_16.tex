%=========== ΛΥΣΗ - 16ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για κάθε $x > 0$, η δοσμένη σχέση διαιρείται με $x^2 \neq 0$:
\[ \frac{x f'(x) - f(x)}{x^2} = \frac{x^2 e^x}{x^2}\Leftrightarrow \]
\[ \frac{f'(x) \cdot x - f(x) \cdot (x)'}{x^2} = e^x\Leftrightarrow \]
\[ \left( \frac{f(x)}{x} \right)' = (e^x)' \]
Από τις συνέπειες του Θεωρήματος Μέσης Τιμής, προκύπτει ότι:
\[ \frac{f(x)}{x} = e^x + c, \quad \text{όπου } c \in \mathbb{R} \]
Για $x = 1$, έχουμε $f(1) = e$, οπότε:
\[ \frac{f(1)}{1} = e^1 + c \implies e = e + c \implies c = 0 \]
Άρα, $\dfrac{f(x)}{x} = e^x \implies f(x) = x e^x$ για κάθε $x > 0$.

\item \textbf{Ρίζες και Πρόσημο:}
Η συνάρτηση $f(x) = x e^x$ ορίζεται στο $(0, +\infty)$. 
Επειδή $x > 0$ και $e^x > 0$ για κάθε $x \in (0, +\infty)$, ισχύει:
\[ f(x) > 0 \text{ για κάθε } x > 0 \]
Άρα, η $f$ διατηρεί \textbf{θετικό πρόσημο} σε όλο το πεδίο ορισμού της και \textbf{δεν έχει καμία ρίζα}.

\textbf{Ασύμπτωτες:}
\begin{itemize}
    \item Στο $x_0 = 0$ (δεξιό άκρο του πεδίου ορισμού):
    Υπολογίζουμε το όριο: $\lim\limits_{x \to 0^+} f(x) = \lim\limits_{x \to 0^+} (x e^x) = 0 \cdot e^0 = 0$.
    Επειδή το όριο είναι πραγματικός αριθμός, η γραφική παράσταση της $f$ \textbf{δεν έχει κατακόρυφη ασύμπτωτη} στο $0$.
    \item Στο $+\infty$:
    Υπολογίζουμε το όριο: $\lim\limits_{x \to +\infty} f(x) = \lim\limits_{x \to +\infty} (x e^x) = (+\infty) \cdot (+\infty) = +\infty$.
    Άρα, δεν υπάρχει οριζόντια ασύμπτωτη. 
    Ελέγχουμε για πλάγια ασύμπτωτη $y = \lambda x + \beta$:
    \[ \lambda = \lim_{x \to +\infty} \frac{f(x)}{x} = \lim_{x \to +\infty} \frac{x e^x}{x} = \lim_{x \to +\infty} e^x = +\infty \]
    Εφόσον $\lambda \notin \mathbb{R}$, η συνάρτηση \textbf{δεν έχει ούτε πλάγια (ούτε οριζόντια) ασύμπτωτη} στο $+\infty$.
\end{itemize}

\item Έχουμε το όριο:
\[ L = \lim_{x \to 0^+} \frac{f(x) - x}{x^2} = \lim_{x \to 0^+} \frac{x e^x - x}{x^2} = \lim_{x \to 0^+} \frac{x(e^x - 1)}{x^2} = \lim_{x \to 0^+} \frac{e^x - 1}{x} \]
Η μορφή του νέου ορίου καθώς $x \to 0^+$ είναι $\dfrac{e^0 - 1}{0} = \dfrac{0}{0}$.
Εφαρμόζοντας τον κανόνα \eng{De L'Hospital}, έχουμε:
\[ L \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim_{x \to 0^+} \frac{(e^x - 1)'}{(x)'} = \lim_{x \to 0^+} \frac{e^x}{1} = e^0 = 1 \]

\item Έστω $M(x_0, f(x_0))$ ένα τυχαίο σημείο της γραφικής παράστασης $C_f$, με $x_0 > 0$.
Η παράγωγος της $f$ είναι $f'(x) = (x e^x)' = e^x + x e^x = (x+1)e^x$.
Ο συντελεστής διεύθυνσης της εφαπτομένης στο $M$ είναι $\lambda = f'(x_0) = (x_0+1)e^{x_0}$.
Η εξίσωση της εφαπτομένης ευθείας $\varepsilon$ είναι:
\[ y - f(x_0) = f'(x_0)(x - x_0) \implies y - x_0 e^{x_0} = (x_0 + 1)e^{x_0}(x - x_0) \]
Για να διέρχεται η εφαπτομένη από την αρχή των αξόνων $O(0,0)$, πρέπει οι συντεταγμένες $(x,y)=(0,0)$ να επαληθεύουν την εξίσωσή της:
\begin{align*}
0 - x_0 e^{x_0} &= (x_0 + 1)e^{x_0}(0 - x_0) \\
-x_0 e^{x_0} &= -x_0(x_0 + 1)e^{x_0}
\end{align*}
Επειδή $x_0 > 0$, γνωρίζουμε ότι $x_0 \neq 0$ και προφανώς $e^{x_0} \neq 0$. 
Άρα μπορούμε να διαιρέσουμε και τα δύο μέλη με την ποσότητα $-x_0 e^{x_0}$, λαμβάνοντας:
\[ 1 = x_0 + 1 \implies x_0 = 0 \]
Η τιμή αυτή όμως απορρίπτεται, διότι το $0$ δεν ανήκει στο πεδίο ορισμού $(0, +\infty)$ της συνάρτησης.
Συμπεραίνουμε ότι \textbf{δεν υπάρχει} καμία εφαπτομένη της $C_f$ η οποία να διέρχεται από την αρχή των αξόνων.
\end{erwthma}
\end{thema}
