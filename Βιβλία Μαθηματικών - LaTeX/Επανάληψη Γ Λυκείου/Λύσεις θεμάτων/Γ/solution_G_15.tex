%=========== ΛΥΣΗ - 15ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
  \item 
  Στη δοσμένη σχέση $f(x+y) = f(x) + f(y)$ θέτουμε $x=0$ και $y=0$:
  \[ f(0+0) = f(0) + f(0) \implies f(0) = 2f(0) \implies f(0) = 0 \]
  Εφόσον η συνάρτηση είναι παραγωγίσιμη στο $x_0=0$ με $f'(0)=2$, από τον ορισμό της παραγώγου με τη βοήθεια ορίου έχουμε:
  \[ f'(0) = \lim_{h \to 0} \frac{f(0+h) - f(0)}{h}=\lim_{h \to 0} \frac{f(h)}{h} = 2 \]
  \item 
  Για να εξετάσουμε την παραγωγισιμότητα της $f$ σε ένα τυχαίο σημείο $x \in \mathbb{R}$, υπολογίζουμε το όριο:
  \[ \lim_{h \to 0} \frac{f(x+h) - f(x)}{h} \]
  Από τη δοσμένη σχέση, ισχύει $f(x+h) = f(x) + f(h)$. Αντικαθιστούμε στον αριθμητή:
  \[ \lim_{h \to 0} \frac{f(x) + f(h) - f(x)}{h} = \lim_{h \to 0} \frac{f(h)}{h} \]
  Από το πρώτο ερώτημα, το όριο αυτό υπάρχει και ισούται με 2.
  Επομένως, η $f$ είναι παραγωγίσιμη σε όλο το $\mathbb{R}$ με:
  \[ f'(x) = 2 \text{ για κάθε } x \in \mathbb{R} \]
  
  Για την εύρεση του τύπου της συνάρτησης, αξιοποιούμε τις συνέπειες του Θεωρήματος Μέσης Τιμής:
  \[ f'(x) = 2 \implies f'(x) = (2x)' \implies f(x) = 2x + c, \text{ όπου } c \in \mathbb{R} \text{ σταθερά.} \]
  Για να προσδιορίσουμε τη σταθερά $c$, χρησιμοποιούμε την τιμή $f(0) = 0$:
  \[ f(0) = 2(0) + c \implies 0 = 0 + c \implies c = 0 \]
  Άρα ο τύπος της συνάρτησης είναι $\boldsymbol{f(x) = 2x}$ για κάθε $x \in \mathbb{R}$.

  \item 
  Επειδή $f'(x) = 2 > 0$ για κάθε $x \in \mathbb{R}$, η συνάρτηση $f$ είναι γνησίως αύξουσα, άρα είναι συνάρτηση 1-1 (\eng{«ένα προς ένα»}) και επομένως αντιστρέφεται.
  Για να βρούμε τον τύπο της αντίστροφης $f^{-1}$, λύνουμε την εξίσωση $y = f(x)$ ως προς $x$:
  \[ y = 2x \Leftrightarrow x = \frac{y}{2} \]
  Εναλλάσσοντας τις μεταβλητές, προκύπτει ο τύπος της αντίστροφης:
  \[ f^{-1}(x) = \frac{x}{2}, \quad x \in \mathbb{R} \]
  Το ζητούμενο εμβαδόν $E$ περικλείεται από τις $C_f, C_{f^{-1}}$ και τις κατακόρυφες ευθείες $x=2$ και $x=4$.
  Βρίσκουμε τη διαφορά των δύο συναρτήσεων:
  \[ f(x) - f^{-1}(x) = 2x - \frac{x}{2} = \frac{3x}{2} \]
  Για $x \in [2, 4]$, ισχύει $\dfrac{3x}{2} > 0$, οπότε η $C_f$ βρίσκεται "πάνω" από τη $C_{f^{-1}}$. Το εμβαδόν υπολογίζεται από το ολοκλήρωμα:
  \[ E = \int_{2}^{4} \left| f(x) - f^{-1}(x) \right| \d x = \int_{2}^{4} \frac{3x}{2} \d x \]
  Υπολογίζουμε το ορισμένο ολοκλήρωμα:
  \[ E = \left[ \frac{3x^2}{4} \right]_{2}^{4} = \frac{3(4^2)}{4} - \frac{3(2^2)}{4} = \frac{48}{4} - \frac{12}{4} = 12 - 3 = \boldsymbol{9 \text{ τ.μ.}} \]
\end{erwthma}
\end{thema}
