%=========== ΛΥΣΗ - ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για την εύρεση της εφαπτομένης στο σημείο με τετμημένη $x_0=0$, υπολογίζουμε την τιμή της συνάρτησης και της παραγώγου της στο σημείο αυτό.
Η τιμή της συνάρτησης είναι:
\[ f(0) = \sqrt{0+1} = 1 \]
Η παράγωγος της συνάρτησης είναι:
\[ f'(x) = \frac{1}{2\sqrt{x+1}} \]
Ο συντελεστής διεύθυνσης της εφαπτομένης είναι:
\[ f'(0) = \frac{1}{2\sqrt{0+1}} = \frac{1}{2} \]
Η εξίσωση της εφαπτομένης είναι:
\begin{align*}
y - f(x_0) &= f'(x_0)(x - x_0) \\
y - 1 &= \frac{1}{2}(x - 0) \\
y &= \frac{1}{2}x + 1
\end{align*}

\item Για τη μελέτη της κυρτότητας, υπολογίζουμε τη δεύτερη παράγογο της συνάρτησης:
\[ f'(x) = \frac{1}{2}(x+1)^{-\frac{1}{2}} \implies f''(x) = \frac{1}{2} \cdot \left(-\frac{1}{2}\right)(x+1)^{-\frac{3}{2}} \cdot 1 = -\frac{1}{4(x+1)^{\frac{3}{2}}} \]
Για κάθε $x > -1$, έχουμε $(x+1)^{\frac{3}{2}} > 0$, επομένως:
\[ f''(x) < 0 \]
Άρα η συνάρτηση $f$ είναι \textbf{συνεχώς κυρτή} στο διάστημα $(-1, +\infty)$.
Επειδή η συνάρτηση είναι \textbf{κυρτή}, το γράφημά της βρίσκεται κάτω από κάθε εφαπτομένη του. Από το ερώτημα (α), η εφαπτομένη στο $x_0=0$ είναι η ευθεία $y = \frac{x}{2} + 1$. Επομένως, για κάθε $x \ge -1$, ισχύει:
\[ \sqrt{x+1} \le 1 + \frac{x}{2} \]

\item Χρησιμοποιώντας την ανισότητα από το ερώτημα (β), ολοκληρώνουμε και τα δύο μέλη στο διάστημα $[0, 3]$:
\[ \int_{0}^{3} \sqrt{x+1} \d x \le \int_{0}^{3} \left( 1 + \frac{x}{2} \right) \d x \]
Υπολογίζουμε το ολοκλήρωμα του δεξιού μέλους:
\begin{align*}
\int_{0}^{3} \left( 1 + \frac{x}{2} \right) \d x &= \left[ x + \frac{x^2}{4} \right]_{0}^{3} \\
&= \left( 3 + \frac{3^2}{4} \right) - (0 + 0) \\
&= 3 + \frac{9}{4} = \frac{12+9}{4} = \frac{21}{4}
\end{align*}
Άρα αποδείξαμε ότι $\int_{0}^{3} \sqrt{x+1} \d x \le \frac{21}{4}$.
\textit{(Σημείωση: Το όριο $\frac{15}{4}$ της εκφώνησης είναι μαθηματικά αδύνατο καθώς η τιμή του ολοκληρώματος είναι μεγαλύτερη από αυτό το κλάσμα).}

\item Υπολογίζουμε αναλυτικά το ορισμένο ολοκλήρωμα:
\begin{align*}
\int_{0}^{3} \sqrt{x+1} \d x &= \left[ \frac{2}{3}(x+1)^{\frac{3}{2}} \right]_{0}^{3} \\
&= \frac{2}{3}(3+1)^{\frac{3}{2}} - \frac{2}{3}(0+1)^{\frac{3}{2}} \\
&= \frac{2}{3}(4)^{\frac{3}{2}} - \frac{2}{3}(1)^{\frac{3}{2}} \\
&= \frac{2}{3}(8) - \frac{2}{3}(1) \\
&= \frac{16-2}{3} = \frac{14}{3}
\end{align*}
Συγκρίνουμε την ακρίβεια της προσέγγισης:
Η πραγματική τιμή είναι $\frac{14}{3} \approx 4,67$ και η ανισοτική προσέγγιση είναι $\frac{21}{4} = 5,25$.
Η διαφορά είναι:
\[ \frac{21}{4} - \frac{14}{3} = \frac{63 - 56}{12} = \frac{7}{12} \approx 0,58 \]
\end{erwthma}
\end{thema}
