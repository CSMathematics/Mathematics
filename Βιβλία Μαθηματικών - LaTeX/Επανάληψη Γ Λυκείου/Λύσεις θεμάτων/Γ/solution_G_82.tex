\begin{thema}{Γ}
\begin{erwthma}
\item Η μονοτονία της $f$ καθορίζεται από το πρόσημο της $f'$.
Από το σχήμα της $C_{f'}$ παρατηρούμε ότι:
\begin{itemize}
\item $f'(x)>0$ για $x\in[0,1)$
\item $f'(x)<0$ για $x\in(1,5)$
\item $f'(x)>0$ για $x\in(5,6]$
\end{itemize}
Άρα η $f$ είναι γνησίως αύξουσα στο $[0,1]$, γνησίως φθίνουσα στο $[1,5]$ και γνησίως αύξουσα στο $[5,6]$.

Επομένως:
\begin{itemize}
\item στο $x=1$ η $f$ παρουσιάζει τοπικό μέγιστο
\item στο $x=5$ η $f$ παρουσιάζει τοπικό ελάχιστο
\end{itemize}

Από το σχήμα έχουμε
\[
f'(x)=(x-1)(x-5)=x^2-6x+5
\]
οπότε
\[
f(1)-f(0)=\dint_0^1 f'(x)\d x=\dint_0^1(x^2-6x+5)\d x=\frac{7}{3}>0
\]
Επίσης
\[
f(5)-f(0)=\dint_0^5 f'(x)\d x=-\frac{25}{3}<0
\]
και
\[
f(6)-f(5)=\dint_5^6 f'(x)\d x=\frac{7}{3}>0
\]
άρα
\[
f(6)=f(5)+\frac{7}{3}<f(0)<f(1)
\]
Συνεπώς, στο $[0,6]$, η $f$ παρουσιάζει ολικό μέγιστο στο $x=1$ και ολικό ελάχιστο στο $x=5$.

\item Η κυρτότητα της $f$ συνδέεται με τη μονοτονία της $f'$.
Από το σχήμα βλέπουμε ότι η $f'$:
\begin{itemize}
\item είναι γνησίως φθίνουσα στο $[0,3]$
\item είναι γνησίως αύξουσα στο $[3,6]$
\end{itemize}
Άρα η $f$ είναι κοίλη στο $[0,3]$ και κυρτή στο $[3,6]$.

Επειδή η μονοτονία της $f'$ αλλάζει στο $x=3$, η $f$ έχει σημείο καμπής στη θέση
\[
x=3
\]
δηλαδή στο σημείο
\[
(3,f(3))
\]

\item Από το σχήμα έχουμε
\[
f'(x)=(x-1)(x-5)=x^2-6x+5
\]
Άρα
\begin{align*}
f(5)-f(0)
&=\dint_0^5 f'(x)\d x\\
&=\dint_0^5(x^2-6x+5)\d x\\
&=\left[\frac{x^3}{3}-3x^2+5x\right]_0^5\\
&=\frac{125}{3}-75+25\\
&=-\frac{25}{3}
\end{align*}
Επομένως
\[
f(5)-f(0)<0
\]
άρα
\[
f(5)<f(0)
\]

\item Η κλίση της εφαπτομένης της $C_f$ στο σημείο με τετμημένη $x_0=3$ είναι
\[
\lambda=f'(3)
\]
Από το σχήμα της $C_{f'}$ βλέπουμε ότι
\[
f'(3)=-4
\]
Άρα
\[
\lambda=-4
\]
Αν $f(3)=1$, τότε η εφαπτομένη διέρχεται από το σημείο $(3,1)$ και έχει εξίσωση
\begin{align*}
y-1&=-4(x-3)\\
y&=-4x+13
\end{align*}
\end{erwthma}
\end{thema}
