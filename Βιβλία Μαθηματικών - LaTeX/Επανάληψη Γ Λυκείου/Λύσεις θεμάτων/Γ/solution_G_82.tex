%=========== ΛΥΣΗ - ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η μονοτονία της $f$ στο $[0, 6]$ εξαρτάται από το πρόσημο της παραγώγου της $f'$. Από το γράφημα της $C_{f'}$ παρατηρούμε ότι:
\begin{itemize}
    \item Για $x \in [0, 1)$, $f'(x) > 0$, οπότε η $f$ είναι \textbf{γνησίως αύξουσα}.
    \item Για $x \in (1, 5)$, $f'(x) < 0$, οπότε η $f$ είναι \textbf{γνησίως φθίνουσα}.
    \item Για $x \in (5, 6]$, $f'(x) > 0$, οπότε η $f$ είναι \textbf{γνησίως αύξουσα}.
\end{itemize}
Σύμφωνα με το κριτήριο των ακροτάτων:
\begin{itemize}
    \item Στο $x = 1$, η $f'$ αλλάζει πρόσημο από θετικό σε αρνητικό, επομένως η $f$ παρουσιάζει \textbf{τοπικό μέγιστο}.
    \item Στο $x = 5$, η $f'$ αλλάζει πρόσημο από αρνητικό σε θετικό, επομένως η $f$ παρουσιάζει \textbf{τοπικό ελάχιστο}.
\end{itemize}

\item Ένα σημείο $(x_0, f(x_0))$ είναι σημείο καμπής της $f$ αν η $f$ είναι δύο φορές παραγωγίσιμη σε μια γειτονιά του $x_0$ και η $f''$ αλλάζει πρόσημο στο $x_0$.
Εφόσον $f''$ είναι η παράγωγος της $f'$, η $f''$ αλλάζει πρόσημο εκεί που η $f'$ παρουσιάζει τοπικό ακρότατο.
Από το γράφημα, η $f'$ έχει τοπικό ελάχιστο στο $x = 3$ (κορυφή της παραβολής). Επομένως, η $f$ παρουσιάζει \textbf{σημείο καμπής} στο $x = 3$.

\item Η διαφορά $f(5) - f(0)$ δίνεται από το ορισμένο ολοκλήρωμα της παραγώγου στο διάστημα $[0, 5]$:
\[ f(5) - f(0) = \int_{0}^{5} f'(x) \d x \]
Από τη μορφή της $C_{f'}$ (παραβολή με κορυφή $(3, -4)$ και ρίζες $1, 5$), η εξίσωσή της είναι $f'(x) = (x-1)(x-5) = x^2 - 6x + 5$.
Υπολογίζουμε το ολοκλήρωμα:
\begin{align*}
f(5) - f(0) &= \int_{0}^{5} (x^2 - 6x + 5) \d x \\
&= \left[ \frac{x^3}{3} - 3x^2 + 5x \right]_{0}^{5} \\
&= \left( \frac{125}{3} - 3(25) + 5(5) \right) - 0 \\
&= \frac{125}{3} - 75 + 25 \\
&= \frac{125}{3} - 50 = \frac{125 - 150}{3} = -\frac{25}{3}
\end{align*}
Επειδή $f(5) - f(0) = -\frac{25}{3} < 0$, συμπεραίνουμε ότι $f(5) < f(0)$.

\item Η κλίση της εφαπτομένης της $C_f$ στο σημείο με τετμημένη $x_0 = 3$ είναι η τιμή της παραγώγου στο σημείο αυτό:
\[ \lambda = f'(3) = -4 \]
Αν $f(3) = 1$, τότε η εφαπτόμενη διέρχεται από το σημείο $(3, 1)$ με κλίση $\lambda = -4$. Η εξίσωσή της είναι:
\begin{align*}
y - f(3) &= f'(3)(x - 3) \\
y - 1 &= -4(x - 3) \\
y &= -4x + 12 + 1 \\
y &= -4x + 13
\end{align*}
\end{erwthma}
\end{thema}
