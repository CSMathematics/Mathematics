%=========== ΛΥΣΗ - 57ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με
\[f'(x)=3x^2+2>0\ ,\ x\in\mathbb{R}\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$, οπότε είναι $1-1$ και αντιστρέφεται.

Επιπλέον
\[\lim\limits_{x\to-\infty}f(x)=-\infty\]
και
\[\lim\limits_{x\to+\infty}f(x)=+\infty\]
Άρα το σύνολο τιμών της $f$ είναι το $\mathbb{R}$.
Επομένως το πεδίο ορισμού της αντίστροφης συνάρτησης $f^{-1}$ είναι
\[D_{f^{-1}}=\mathbb{R}\]

\item Επειδή η $f$ είναι γνησίως αύξουσα και
\[f(1)=1^3+2\cdot1-3=0\]
έχουμε
\[f(u)<0\Leftrightarrow u<1\]
Η ανίσωση
\[f\left(f^{-1}(x^2-4x)+1\right)<0\]
είναι ισοδύναμη με
\[f^{-1}(x^2-4x)+1<1\]
δηλαδή
\[f^{-1}(x^2-4x)<0\]
Επειδή η $f^{-1}$ είναι γνησίως αύξουσα, προκύπτει
\[x^2-4x<f(0)\]
Υπολογίζουμε
\[f(0)=-3\]
Άρα
\[x^2-4x<-3\Leftrightarrow x^2-4x+3<0\]
δηλαδή
\[(x-1)(x-3)<0\]
Επομένως οι λύσεις της ανίσωσης είναι
\[x\in(1,3)\]

\item Το σημείο $M(x,y)$ κινείται πάνω στη γραφική παράσταση της $f$, άρα
\[y=f(x)=x^3+2x-3\]
και
\[y'(t)=f'(x(t))x'(t)\]
Δίνεται ότι
\[x'(t)=2\]
Τη χρονική στιγμή $t_0$ που το σημείο διέρχεται από το $M_0(2,9)$, έχουμε
\[x(t_0)=2\]
και
\[f'(2)=3\cdot2^2+2=14\]
Άρα
\[y'(t_0)=14\cdot2=28\]

Το τρίγωνο $OAM$ έχει βάση $OA=3$ πάνω στον άξονα $x'x$ και ύψος $|y|$.
Τη χρονική στιγμή $t_0$ είναι $y=9>0$, άρα το εμβαδόν του είναι
\[E(t)=\frac{1}{2}\cdot3\cdot y(t)=\frac{3}{2}y(t)\]
στη γειτονιά της χρονικής στιγμής $t_0$.
Επομένως
\[E'(t_0)=\frac{3}{2}y'(t_0)=\frac{3}{2}\cdot28=42\]
Άρα ο ρυθμός μεταβολής του εμβαδού τη χρονική στιγμή $t_0$ είναι
\[42\ \text{τετρ. μονάδες}/\sec\]

\item Θέλουμε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f^{-1}$, τους άξονες και την ευθεία $x=9$.
Επειδή
\[f(2)=8+4-3=9\]
έχουμε
\[f^{-1}(9)=2\]
Επίσης
\[f(1)=1+2-3=0\Rightarrow f^{-1}(0)=1\]
Στο διάστημα $[0,9]$ ισχύει $f^{-1}(x)>0$, άρα το ζητούμενο εμβαδόν είναι
\[E=\dint_0^9 f^{-1}(x)\d x\]
Χρησιμοποιούμε τη σχέση για το ολοκλήρωμα αντίστροφης συνάρτησης:
\[\dint_{f(a)}^{f(b)} f^{-1}(y)\d y=\left[xf(x)\right]_a^b-\dint_a^b f(x)\d x\]
Εδώ παίρνουμε $a=1$ και $b=2$, αφού $f(1)=0$ και $f(2)=9$.
Άρα
\begin{align*}
E&=\left[xf(x)\right]_1^2-\dint_1^2 (x^3+2x-3)\d x\\
&=2f(2)-1f(1)-\left[\frac{x^4}{4}+x^2-3x\right]_1^2\\
&=18-\left[\left(4+4-6\right)-\left(\frac{1}{4}+1-3\right)\right]\\
&=18-\left[2-\left(-\frac{7}{4}\right)\right]\\
&=18-\frac{15}{4}\\
&=\frac{57}{4}
\end{align*}
Επομένως το ζητούμενο εμβαδόν είναι
\[E=\frac{57}{4}\]
\end{erwthma}
\end{thema}
