%=========== ΛΥΣΗ - 21ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Θα αποδείξουμε τη μονοτονία της $f$ με τη μέθοδο της απαγωγής σε άτοπο.
Έστω $x_1, x_2 \in \mathbb{R}$ με $x_1 < x_2$.
Υποθέτουμε ότι $f(x_1) \ge f(x_2)$. Τότε, επειδή η συνάρτηση $g(t) = t^3$ είναι γνησίως αύξουσα και οι αριθμοί είναι πραγματικοί, προκύπτουν οι ανισότητες:
\[ f^3(x_1) \ge f^3(x_2) \]
\[ 3f(x_1) \ge 3f(x_2) \]
Προσθέτοντας τις παραπάνω ανισότητες κατά μέλη, έχουμε:
\[ f^3(x_1) + 3f(x_1) \ge f^3(x_2) + 3f(x_2) \]
Βάσει της αρχικής σχέσης που ικανοποιεί η $f$, η ανισότητα αυτή γράφεται:
\[ x_1 \ge x_2 \]
Αυτό όμως είναι \textbf{άτοπο}, καθώς εξ υποθέσεως έχουμε $x_1 < x_2$.
Άρα, η υπόθεσή μας είναι λανθασμένη και ισχύει υποχρεωτικά $f(x_1) < f(x_2)$.
Συμπεραίνουμε ότι η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\textit{(Εναλλακτικός τρόπος: Έστω $x_1, x_2 \in \mathbb{R}$ με $x_1 < x_2 \implies x_2 - x_1 > 0$. Αφαιρώντας κατά μέλη: $x_2 - x_1 = f^3(x_2) - f^3(x_1) + 3(f(x_2) - f(x_1)) = (f(x_2) - f(x_1)) \cdot [f^2(x_2) + f(x_2)f(x_1) + f^2(x_1) + 3]$. Επειδή η αγκύλη είναι $\ge 3 > 0$ (άθροισμα τετραγώνων), πρέπει υποχρεωτικά $f(x_2) - f(x_1) > 0$).}

\item Αναζητούμε τις λύσεις της εξίσωσης $f(x) = 0$.
Αν στην αρχική σχέση $f^3(x) + 3f(x) = x$ θέσουμε όπου υπάρχει $f(x)$ το $0$, παίρνουμε:
\[ 0^3 + 3 \cdot 0 = x \implies x = 0 \]
Άρα η εξίσωση έχει μοναδική λύση (αφού η $f$ είναι γνησίως αύξουσα, δηλαδή '1-1') την $x = 0$. Ισχύει δηλαδή $f(0) = 0$.

Εξετάζουμε τώρα το πρόσημο της $f(x)$ αξιοποιώντας τη γνήσια μονοτονία της:
\begin{itemize}
    \item Για $x > 0 \overset{f \nearrow}{\implies} f(x) > f(0) \implies f(x) > 0$.
    \item Για $x < 0 \overset{f \nearrow}{\implies} f(x) < f(0) \implies f(x) < 0$.
\end{itemize}
Παρατηρούμε ότι όταν το $x$ είναι θετικό, η $f(x)$ είναι θετική. Όταν το $x$ είναι αρνητικό, η $f(x)$ είναι αρνητική. Άρα η συνάρτηση $f(x)$ διατηρεί πάντοτε \textbf{το ίδιο πρόσημο με το $x$}.

\item Επειδή η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$, είναι συνάρτηση \textbf{1-1} (\eng{«ένα προς ένα»}). Επομένως, ορίζεται η αντίστροφη συνάρτησή της $f^{-1}$.
Για να βρούμε το πεδίο ορισμού και τον τύπο της $f^{-1}$, θέτουμε $y = f(x)$. 
Η αρχική σχέση γίνεται:
\[ y^3 + 3y = x \]
Από την εξίσωση αυτή παρατηρούμε ότι για κάθε $y \in \mathbb{R}$, υπάρχει ένα μοναδικό $x \in \mathbb{R}$ (το $x = y^3+3y$) τέτοιο ώστε $f(x) = y$.
Άρα το σύνολο τιμών της $f$, το οποίο αποτελεί το πεδίο ορισμού της $f^{-1}$, είναι όλο το $\mathbb{R}$.
Λύνοντας ως προς $x$ βρήκαμε $x = y^3 + 3y$, δηλαδή $f^{-1}(y) = y^3 + 3y$.
Εναλλάσσοντας τις μεταβλητές, ο τύπος της αντίστροφης είναι:
\[ f^{-1}(x) = x^3 + 3x, \quad x \in \mathbb{R} \]

\item Ζητείται ο υπολογισμός του ορισμένου ολοκληρώματος:
\[ I = \int_0^4 f(x) \d x \]
Δεν γνωρίζουμε τον τύπο της $f$, γνωρίζουμε όμως τον τύπο της αντίστροφης.
Εφαρμόζουμε τη μέθοδο της \textbf{αντικατάστασης}. Θέτουμε:
\[ x = f^{-1}(u) \implies x = u^3 + 3u \]
Τότε το διαφορικό $\d x$ είναι:
\[ \d x = (u^3 + 3u)' \d u \implies \d x = (3u^2 + 3) \d u \]
Βρίσκουμε τα νέα όρια ολοκλήρωσης ως προς $u$:
\begin{itemize}
    \item Για $x = 0$, λύνουμε την εξίσωση $u^3 + 3u = 0 \implies u(u^2 + 3) = 0 \implies u = 0$ (αφού $u^2+3 > 0$).
    \item Για $x = 4$, λύνουμε την εξίσωση $u^3 + 3u = 4 \implies u^3 + 3u - 4 = 0$.
    Παρατηρούμε ότι το $u=1$ αποτελεί προφανή ρίζα ($1^3 + 3\cdot 1 - 4 = 0$). Επειδή η συνάρτηση $g(u) = u^3 + 3u$ είναι γνησίως αύξουσα (παράγωγος $3u^2+3>0$), η ρίζα αυτή είναι μοναδική. Άρα $u=1$.
\end{itemize}
Το ολοκλήρωμα εκφράζεται πλέον ως προς τη μεταβλητή $u$:
\begin{align*}
I &= \int_0^1 u \cdot (3u^2 + 3) \d u = \int_0^1 (3u^3 + 3u) \d u \\
&= \left[ \frac{3u^4}{4} + \frac{3u^2}{2} \right]_0^1 = \left( \frac{3 \cdot 1^4}{4} + \frac{3 \cdot 1^2}{2} \right) - 0 \\
&= \frac{3}{4} + \frac{3}{2} = \frac{3}{4} + \frac{6}{4} = \boldsymbol{\frac{9}{4}}
\end{align*}
\end{erwthma}
\end{thema}
