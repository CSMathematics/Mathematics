%=========== ΛΥΣΗ - ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Έστω $x_1, x_2 \in \mathbb{R}$ με $x_1 < x_2$. Από τη σχέση που δίνεται έχουμε:
\[ f^3(x_1) + 3f(x_1) = x_1 \quad \text{και} \quad f^3(x_2) + 3f(x_2) = x_2 \]
Αφαιρώντας κατά članους τις δύο ισότητες, προκύπτει:
\[ x_2 - x_1 = (f^3(x_2) - f^3(x_1)) + 3(f(x_2) - f(x_1)) \]
Χρησιμοποιώντας την ταυτότητα $a^3 - b^3 = (a-b)(a^2 + ab + b^2)$, έχουμε:
\[ x_2 - x_1 = (f(x_2) - f(x_1))(f^2(x_2) + f(x_2)f(x_1) + f^2(x_1) + 3) \]
Παρατηρούμε ότι η παράσταση $f^2(x_2) + f(x_2)f(x_1) + f^2(x_1)$ είναι μια quadratic form της μορφής $a^2 + ab + b^2$, η οποία είναι πάντα μη αρνητική, καθώς:
\[ f^2(x_2) + f(x_2)f(x_1) + f^2(x_1) = \left( f(x_2) + \frac{f(x_1)}{2} \right)^2 + \frac{3f^2(x_1)}{4} \geq 0 \]
Συνεπώς, $f^2(x_2) + f(x_2)f(x_1) + f^2(x_1) + 3 \geq 3 > 0$.
Εφόσον $x_2 - x_1 > 0$ και ο δεύτερος παράγοντας είναι θετικός, πρέπει αναγκαστικά:
\[ f(x_2) - f(x_1) > 0 \implies f(x_2) > f(x_1) \]
Άρα η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα}.

\item Για την εξίσωση $f(x) = 0$, αντικαθιστούμε στη σχέση $f^3(x) + 3f(x) = x$:
\[ 0^3 + 3(0) = x \implies x = 0 \]
Άρα $f(x) = 0 \Leftrightarrow x = 0$.
Επειδή η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} και $f(0) = 0$:
\begin{itemize}
    \item Αν $x > 0$, τότε $f(x) > f(0) = 0$.
    \item Αν $x < 0$, τότε $f(x) < f(0) = 0$.
\end{itemize}
Συνεπώς, η συνάρτηση $f(x)$ διατηρεί το ίδιο πρόσημο με την ανεξάρτητη μεταβλητή $x$.

\item Επειδή η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$, είναι አንድ-ένας. Επίσης, από τη σχέση $f^3(x) + 3f(x) = x$, καθώς το $x$ καλύπτει όλο το $\mathbb{R}$, η συνάρτηση $f$ είναι και επί του $\mathbb{R}$. Άρα η $f$ είναι αντιστρέψιμη.
Για να βρούμε τον τύπο της $f^{-1}$, θέτουμε $y = f(x)$, οπότε $f^{-1}(y) = x$.
Από τη σχέση $f^3(x) + 3f(x) = x$ έχουμε:
\[ y^3 + 3y = x \implies f^{-1}(y) = y^3 + 3y \]
Αλλάζοντας τη μεταβλητή σε $x$, ο τύπος της αντίστροφης συνάρτησης είναι:
\[ f^{-1}(x) = x^3 + 3x \]

\item Για τον υπολογισμό του ολοκληρώματος $I = \int_0^4 f(x) \d x$, πραγματοποιούμε την αντικατάσταση $x = f^{-1}(y) = y^3 + 3y$.
Τότε $\d x = (3y^2 + 3) \d y$.
Για τα όρια ολοκλήρωσης:
\begin{itemize}
    \item Αν $x = 0$, τότε $y^3 + 3y = 0 \implies y(y^2 + 3) = 0 \implies y = 0$.
    \item Αν $x = 4$, τότε $y^3 + 3y = 4$. Παρατηρούμε ότι για $y = 1$ ισχύει $1^3 + 3(1) = 4$, και επειδή η $f^{-1}$ είναι γνησίως αύξουσα, το $y = 1$ είναι η μοναδική πραγματική λύση.
\end{itemize}
Το ολοκλήρωμα γίνεται:
\begin{align*}
I &= \int_0^1 y \cdot (3y^2 + 3) \d y \\
&= \int_0^1 (3y^3 + 3y) \d y \\
&= \left[ \frac{3y^4}{4} + \frac{3y^2}{2} \right]_0^1 \\
&= \frac{3}{4} + \frac{3}{2} - 0 \\
&= \frac{3+6}{4} = \frac{9}{4}
\end{align*}
Άρα $\int_0^4 f(x) \d x = \frac{9}{4} = 2,25$.
\end{erwthma}
\end{thema}
