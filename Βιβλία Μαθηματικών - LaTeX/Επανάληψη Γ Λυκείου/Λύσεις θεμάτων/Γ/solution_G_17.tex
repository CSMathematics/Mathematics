\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $[0, 1]$, άρα είναι και \textbf{συνεχής} σε αυτό.
Ικανοποιούνται οι προϋποθέσεις του \textbf{Θεωρήματος Μέσης Τιμής} στο διάστημα $[0, 1]$.
Επομένως, υπάρχει τουλάχιστον ένα $\xi \in (0, 1)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(1) - f(0)}{1 - 0} \]
Αντικαθιστώντας τις δεδομένες τιμές $f(0) = 0$ και $f(1) = 1$, προκύπτει:
\[ f'(\xi) = \frac{1 - 0}{1} = 1 \]

\item Δεδομένου ότι η πρώτη παράγωγος $f'$ είναι \textbf{γνησίως αύξουσα} στο $(0, 1)$, η εξίσωση $f'(x) = 1$ μπορεί να έχει \textbf{το πολύ μία} λύση στο διάστημα αυτό. Αφού από το προηγούμενο ερώτημα αποδείξαμε την ύπαρξη του $\xi$ με $f'(\xi)=1$, το $\xi$ αυτό είναι \textbf{μοναδικό}.

Η συνάρτηση $h(x) = f(x) - x$ έχει παράγωγο $h'(x) = f'(x) - 1$.
Εξετάζουμε το πρόσημό της εκατέρωθεν του $\xi$, αξιοποιώντας τη γνήσια μονοτονία της $f'$:
\begin{itemize}
    \item Για $0 < x < \xi \overset{f' \nearrow}{\implies} f'(x) < f'(\xi) \implies f'(x) < 1 \implies f'(x) - 1 < 0 \implies \boldsymbol{h'(x) < 0}$
    \item Για $\xi < x < 1 \overset{f' \nearrow}{\implies} f'(x) > f'(\xi) \implies f'(x) > 1 \implies f'(x) - 1 > 0 \implies \boldsymbol{h'(x) > 0}$
\end{itemize}

\item Η συνάρτηση $h(x) = f(x) - x$ είναι συνεχής στο $[0, 1]$ και παραγωγίσιμη στο $(0, 1)$.
Με βάση τα πρόσημα της $h'$ που βρήκαμε στο Γ2:
\begin{center}
\begin{tikzpicture}

{$x$ / .8, $h'(x)$ / 0.8, $h(x)$ / 1.5}
{ $0$, $\xi$, $1$ }
\tkzTabLine{ , -, z, +,  }
\tkzTabVar{ +/ $0$, -/ $h(\xi)$, +/ $0$  }
\end{tikzpicture}
\end{center}
Παρατηρούμε ότι:
\begin{itemize}
    \item Στο διάστημα $[0, \xi]$ η $h$ είναι \textbf{γνησίως φθίνουσα}. 
    Άρα για κάθε $x \in (0, \xi]$ ισχύει: $0 < x \implies h(0) > h(x) \implies 0 > h(x) \implies h(x) < 0$.
    \item Στο διάστημα $[\xi, 1]$ η $h$ είναι \textbf{γνησίως αύξουσα}. 
    Άρα για κάθε $x \in [\xi, 1)$ ισχύει: $x < 1 \implies h(x) < h(1) \implies h(x) < 0$.
\end{itemize}
Ενώνοντας τα δύο διαστήματα, προκύπτει ότι για κάθε $x \in (0, 1)$ ισχύει $h(x) < 0$.
\[ h(x) < 0 \implies f(x) - x < 0 \implies \boldsymbol{f(x) < x} \]
(Σημείωση: Στα άκρα $0$ και $1$ προφανώς ισχύει η ισότητα).

\item Από το Γ3 γνωρίζουμε ότι $f(x) \le x$ για κάθε $x \in [0, 1]$, με την ανισότητα να είναι γνήσια για όλα τα εσωτερικά σημεία $x \in (0,1)$. Εφόσον οι συναρτήσεις δεν ταυτίζονται στο $[0,1]$, ολοκληρώνοντας και τα δύο μέλη η ανισότητα γίνεται γνήσια:
\[ \int_0^1 f(x) \d x < \int_0^1 x \d x \]
Υπολογίζουμε το ολοκλήρωμα του δεύτερου μέλους:
\[ \int_0^1 x \d x = \left[ \frac{x^2}{2} \right]_0^1 = \frac{1^2}{2} - \frac{0^2}{2} = \frac{1}{2} \]
Κατά συνέπεια, αποδείχθηκε ότι:
\[ \int_0^1 f(x) \d x < \frac{1}{2} \]
\end{erwthma}
\end{thema}
