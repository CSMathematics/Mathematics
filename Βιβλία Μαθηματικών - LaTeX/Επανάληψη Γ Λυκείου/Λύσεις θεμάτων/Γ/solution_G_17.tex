%=========== ΛΥΣΗ - {ΘΕΜΑ Γ} ===========
\begin{thema}{Γ}
\begin{erwthma}
    \item Η συνάρτηση $f$ είναι \textbf{συνεχώς} στο $[0, 1]$ και \textbf{παραγωγήσιμη} στο $(0, 1)$.
    Εφαρμόζοντας το θεωρήμα του \eng{Lagrange} στο διάστημα $[0, 1]$, υπάρχει $\xi \in (0, 1)$ τέτοιο ώστε
    \[ f'(\xi) = \frac{f(1) - f(0)}{1 - 0} \]
    Δεδομένου ότι $f(0) = 0$ και $f(1) = 1$, έχουμε
    \[ f'(\xi) = \frac{1 - 0}{1} = 1 \]

    \item Επειδή η $f'$ είναι \textbf{γνησίως αύξουσα} στο $(0, 1)$, η εξίσωση $f'(x) = 1$ μπορεί να έχει το πολύ μία λύση. Από το προηγούμενο ερώτημα γνωρίζουμε ότι η $\xi$ υπάρχει, άρα η τιμή αυτή είναι \textbf{μοναδική}.
    
    Θεωρούμε τη συνάρτηση $h'(x) = f'(x) - 1$.
    \begin{itemize}
        \item Για $x < \xi \Rightarrow f'(x) < f'(\xi) \Rightarrow f'(x) < 1 \Rightarrow h'(x) < 0$
        \item Για $x > \xi \Rightarrow f'(x) > f'(\xi) \Rightarrow f'(x) > 1 \Rightarrow h'(x) > 0$
    \end{itemize}

    \item Θεωρούμε τη συνάρτηση $h(x) = f(x) - x$ στο διάστημα $[0, 1]$.
    Η $h$ είναι \textbf{συνεχώς} στο $[0, 1]$ και \textbf{παραγωγήσιμη} στο $(0, 1)$ με $h'(x) = f'(x) - 1$.
    Από το προηγούμενο ερώτημα ισχύει:
    \begin{itemize}
        \item $h'(x) < 0$ για $x \in (0, \xi) \Rightarrow$ η $h$ είναι \textbf{γνησίως φθίνουσα} στο $[0, \xi]$.
        \item $h'(x) > 0$ για $x \in (\xi, 1) \Rightarrow$ η $h$ είναι \textbf{γνησίως αύξουσα} στο $[\xi, 1]$.
    \end{itemize}
    Επίσης, $h(0) = f(0) - 0 = 0$ και $h(1) = f(1) - 1 = 1 - 1 = 0$.
    \begin{itemize}
        \item Για $x \in (0, \xi]$, επειδή η $h$ είναι \textbf{γνησίως φθίνουσα}, ισχύει $h(x) < h(0) = 0$.
        \item Για $x \in [\xi, 1)$, επειδή η $h$ είναι \textbf{γνησίως αύξουσα}, ισχύει $h(x) < h(1) = 0$.
    \end{itemize}
    Συνεπώς, για κάθε $x \in (0, 1)$ ισχύει $h(x) < 0$, το οποίο σημαίνει ότι $f(x) < x$.

    \item Από το προηγούμενο ερώτημα, για κάθε $x \in (0, 1)$ ισχύει $f(x) < x$.
    Ολοκληρώνοντας τη συγκεκριμένη ανισότητα στο διάστημα $[0, 1]$, έχουμε
    \[ \int_0^1 f(x) \d x < \int_0^1 x \d x \]
    Υπολογίζοντας το ολοκλήρωμα του δεξιού μέλους:
    \[ \int_0^1 x \d x = \left[ \frac{x^2}{2} \right]_0^1 = \frac{1^2}{2} - \frac{0^2}{2} = \frac{1}{2} \]
    Άρα, $\int_0^1 f(x) \d x < \frac{1}{2}$.
\end{erwthma}
\end{thema}
