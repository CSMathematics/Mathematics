\begin{thema}{Γ}
\begin{erwthma}
    \item Η συνάρτηση $f(x) = e^x + x - 2$ είναι \textbf{συνεχώς} και \textbf{παραγωγήσιμη} στο $\mathbb{R}$, και άρα στο διάστημα $[0, 1]$.
    Υπολογίζουμε τις τιμές της συνάρτησης στα άκρα του διαστήματος:
    \[ f(0) = e^0 + 0 - 2 = 1 - 2 = -1 \]
    \[ f(1) = e^1 + 1 - 2 = e - 1 \]
    Εφόσον $f(0) = -1 < 0$ και $f(1) = e - 1 > 0$, τότε $f(0) \cdot f(1) < 0$. 
    Σύμφωνα με το θεώρημα \eng{Bolzano}, η εξίσωση $f(x) = 0$ έχει τουλάχιστον μία ρίζα $x_0 \in (0, 1)$.
    
    Για τη μοναδικότητα της ρίζας, εξετάζουμε την παράγωγο της $f$:
    \[ f'(x) = e^x + 1 \]
    Δεδομένου ότι $e^x > 0$ για κάθε $x \in \mathbb{R}$, προκύπτει ότι $f'(x) > 1 > 0$ για κάθε $x \in \mathbb{R}$.
    Άρα η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το πεδίο ορισμού της.
    Επειδή η $f$ είναι γνησίως μονότονη, η ρίζα $x_0 \in (0, 1)$ είναι μοναδική.

    \item Η συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ είναι \textbf{γνησίως αύξουσα} και συνεχής.
    Εξετάζουμε τα όρια στα άκρα του πεδίου ορισμού:
    \[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} (e^x + x - 2) = 0 + (-\infty) - 2 = -\infty \]
    \[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} (e^x + x - 2) = +\infty + \infty - 2 = +\infty \]
    Συνεπώς, η $f$ είναι bijection (αντιστρόφιστη) από το $\mathbb{R}$ στο $\mathbb{R}$ και άρα η αντίστροφη συνάρτηση $f^{-1}: \mathbb{R} \to \mathbb{R}$ υπάρχει.
    Η $f^{-1}$ δεν μπορεί να εκφραστεί με στοιχειώδεις συναρτήσεις, καθώς η εξίσωση $y = e^x + x - 2$ δεν λύνεται ως προς $x$ με απλούς τρόπους.

    \item Υπολογίζουμε το όριο:
    \[ \lim_{x \to 0} \frac{f(x) + 1}{x} = \lim_{x \to 0} \frac{e^x + x - 2 + 1}{x} = \lim_{x \to 0} \frac{e^x + x - 1}{x} \]
    Το όριο είναι της μορφής $\frac{0}{0}$, οπότε εφαρμόζουμε τον κανόνα \eng{L'Hôpital}:
    \[ \lim_{x \to 0} \frac{e^x + x - 1}{x} \xlongequal{\eng{L'Hôpital}} \lim_{x \to 0} \frac{e^x + 1}{1} = e^0 + 1 = 2 \]

    \item Για την εφαπτομένη της $C_f$ στο σημείο $(0, -1)$, έχουμε:
    $x_1 = 0$ και $y_1 = f(0) = -1$. 
    Η κλίση της εφαπτομένης είναι $f'(0) = e^0 + 1 = 2$.
    Η εξίσωση της εφαπτομένης είναι:
    \[ y - (-1) = 2(x - 0) \implies y + 1 = 2x \implies y = 2x - 1 \]
    
    Η γραφική παράσταση $C_{f^{-1}}$ της αντίστροφης συνάρτησης είναι η συμμετρική της $C_f$ ως προς την ευθεία $y = x$.
    Εφόσον το σημείο $(0, -1) \in C_f$, τότε το σημείο $(-1, 0) \in C_{f^{-1}}$.
    Η εφαπτομένη της $C_{f^{-1}}$ στο σημείο $(-1, 0)$ είναι η συμμετρική της εφαπτομένης της $C_f$ στο $(0, -1)$ ως προς την ευθεία $y = x$.
    Αν η εφαπτομένη της $C_f$ είναι $y = 2x - 1$, τότε η εφαπτομένη της $C_{f^{-1}}$ προκύπτει αντικαθιστώντας το $x$ με $y$ και το $y$ με $x$:
    \[ x = 2y - 1 \implies 2y = x + 1 \implies y = \frac{1}{2}x + \frac{1}{2} \]
    Η κλίση της εφαπτομένης της $C_{f^{-1}}$ είναι $\frac{1}{f'(0)} = \frac{1}{2}$.
\end{erwthma}
\end{thema}
