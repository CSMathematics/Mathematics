%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Εφαρμόζουμε το θεώρημα \eng{Lagrange} στη συνάρτηση $f(x)=\ln x$ στο διάστημα $[x,x+1]$, όπου $x>0$.
Υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
\frac{f(x+1)-f(x)}{(x+1)-x}=f'(\xi).
\]
Επομένως
\[
\ln(x+1)-\ln x=\frac{1}{\xi}.
\]

\item Από το προηγούμενο ερώτημα υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
\ln\left(1+\frac{1}{x}\right)=\frac{1}{\xi}.
\]
Επειδή $x<\xi<x+1$, έχουμε
\[
\frac{1}{x+1}<\frac{1}{\xi}<\frac{1}{x}
\]
και άρα
\[
\frac{1}{x+1}<\ln\left(1+\frac{1}{x}\right)<\frac{1}{x}.
\]

\item Πολλαπλασιάζουμε την ανισότητα του προηγούμενου ερωτήματος με $x>0$:
\[
\frac{x}{x+1}<x\ln\left(1+\frac{1}{x}\right)<1.
\]
Περνώντας στο όριο καθώς $x\to +\infty$:
\[
\lim_{x\to +\infty}\frac{x}{x+1}=1
\]
και
\[
\lim_{x\to +\infty}1=1.
\]
Άρα, από το κριτήριο παρεμβολής,
\[
\lim_{x\to +\infty}\left[x\ln\left(1+\frac{1}{x}\right)\right]=1.
\]

\item Από το δεύτερο ερώτημα,
\[
\frac{1}{x+1}<\ln\left(1+\frac{1}{x}\right)<\frac{1}{x},\qquad x\in[1,e].
\]
Ολοκληρώνουμε στο $[1,e]$:
\[
\int_1^e \frac{1}{x+1}\,\d x
<
\int_1^e \ln\left(1+\frac{1}{x}\right)\,\d x
<
\int_1^e \frac{1}{x}\,\d x.
\]
Επειδή
\[
f\left(1+\frac{1}{x}\right)=\ln\left(1+\frac{1}{x}\right),
\]
παίρνουμε
\[
\int_1^e \frac{1}{x+1}\,\d x
<
\int_1^e f\left(1+\frac{1}{x}\right)\,\d x
<
\int_1^e \frac{1}{x}\,\d x.
\]
Υπολογίζουμε:
\[
\int_1^e \frac{1}{x+1}\,\d x
=
\left[\ln(x+1)\right]_1^e
=
\ln\frac{e+1}{2}
\]
και
\[
\int_1^e \frac{1}{x}\,\d x
=
\left[\ln x\right]_1^e
=1.
\]
Άρα
\[
\ln\frac{e+1}{2}
<
\int_1^e f\left(1+\frac{1}{x}\right)\,\d x
<
1.
\]
\end{erwthma}
\end{thema}
