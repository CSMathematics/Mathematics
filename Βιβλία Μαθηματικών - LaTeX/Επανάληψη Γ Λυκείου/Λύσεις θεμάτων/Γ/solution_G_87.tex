%=========== ΛΥΣΗ - 87ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από το σχήμα της $C_f$ έχουμε
\[
f(3)=2
\]
άρα
\[
f^{-1}(2)=3
\]
Επίσης
\[
f(5)=4
\]
άρα
\[
f^{-1}(4)=5
\]
Τέλος, επειδή το $3$ ανήκει στο σύνολο τιμών της $f$, έχουμε
\[
(f\circ f^{-1})(3)=3
\]

\item Για μια γνησίως αύξουσα συνάρτηση $f:[a,b]\to[f(a),f(b)]$ ισχύει
\[
\dint_a^b f(x)\d x+\dint_{f(a)}^{f(b)} f^{-1}(y)\d y=bf(b)-af(a)
\]
Στο σχήμα βλέπουμε ότι
\[
f(0)=0
\quad \text{και}\quad
f(5)=4
\]
Άρα, για $a=0$ και $b=5$,
\[
\dint_0^5 f(x)\d x+\dint_0^4 f^{-1}(y)\d y=5\cdot 4-0\cdot 0=20
\]
Επειδή
\[
\dint_0^5 f(x)\d x=9
\]
παίρνουμε
\[
9+\dint_0^4 f^{-1}(y)\d y=20
\]
οπότε
\[
\dint_0^4 f^{-1}(y)\d y=11
\]

\item Γραφικά, η $C_{f^{-1}}$ προκύπτει από την $C_f$ με συμμετρία ως προς την ευθεία $y=x$.
Αφού η $f$ είναι κοίλη στο $[0,5]$, η αντίστροφη συνάρτηση $f^{-1}$ φαίνεται κυρτή στο $[0,4]$.

Αλγεβρικά, θέτουμε
\[
g=f^{-1}
\]
Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε $f'(x)>0$ στα σημεία όπου παραγωγίζεται.
Ισχύει
\[
g'(y)=\frac{1}{f'(g(y))}
\]
και
\[
g''(y)=-\frac{f''(g(y))}{\left[f'(g(y))\right]^3}
\]
Εφόσον η $f$ είναι κοίλη, έχουμε
\[
f''(x)\leq 0
\]
ενώ
\[
\left[f'(g(y))\right]^3>0
\]
Άρα
\[
g''(y)\geq 0
\]
Επομένως η $f^{-1}$ είναι κυρτή.

\item Έστω $M(X(t),Y(t))$ σημείο της $C_{f^{-1}}$.
Τότε
\[
Y(t)=f^{-1}(X(t))
\]
ή, ισοδύναμα,
\[
X(t)=f(Y(t))
\]
Η τεταγμένη αυξάνεται με ρυθμό
\[
Y'(t)=3\ \mu/\sec
\]
Παραγωγίζοντας τη σχέση $X(t)=f(Y(t))$ ως προς τον χρόνο, παίρνουμε
\[
X'(t)=f'(Y(t))Y'(t)
\]
Τη στιγμή που η τεταγμένη είναι
\[
Y(t_0)=3
\]
έχουμε, από το σχήμα, ότι γύρω από το σημείο $(3,2)$ η $C_f$ έχει κλίση $1$, δηλαδή
\[
f'(3)=1
\]
Άρα
\[
X'(t_0)=f'(3)\cdot Y'(t_0)=1\cdot 3=3
\]
Επομένως η τετμημένη μεταβάλλεται με ρυθμό
\[
3\ \mu/\sec
\]
\end{erwthma}
\end{thema}
