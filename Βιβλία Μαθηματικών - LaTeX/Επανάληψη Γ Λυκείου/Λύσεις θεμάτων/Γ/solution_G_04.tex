%=========== ΛΥΣΗ - 4ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Ισχύει
\[
\left(f^2(x)\right)'=2f(x)f'(x)=2e^{2x}.
\]
Ολοκληρώνουμε από $0$ έως $x$:
\[
f^2(x)-f^2(0)=\int_0^x 2e^{2t}\,\d t.
\]
Επειδή $f(0)=\sqrt{2}$, έχουμε $f^2(0)=2$, άρα
\[
f^2(x)-2=\left[e^{2t}\right]_0^x=e^{2x}-1
\]
και επομένως
\[
f^2(x)=e^{2x}+1.
\]
Άρα
\[
f(x)=\sqrt{e^{2x}+1}
\]
αφού $f(0)=\sqrt{2}>0$.

\item Παραγωγίζοντας, βρίσκουμε
\[
f'(x)=\frac{e^{2x}}{\sqrt{e^{2x}+1}}.
\]
Επειδή $e^{2x}>0$ και $\sqrt{e^{2x}+1}>0$, ισχύει
\[
f'(x)>0,\quad x\in \mathbb{R}.
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$.

Ακόμη,
\[
\lim_{x\to -\infty}f(x)=\lim_{x\to -\infty}\sqrt{e^{2x}+1}=1
\]
και
\[
\lim_{x\to +\infty}f(x)=+\infty.
\]
Επομένως το σύνολο τιμών της είναι
\[
(1,+\infty).
\]

\item Θέτουμε
\[
y=\sqrt{e^{2x}+1}, \quad y>1.
\]
Τότε
\[
y^2=e^{2x}+1 \Leftrightarrow e^{2x}=y^2-1
\]
και άρα
\[
2x=\ln(y^2-1).
\]
Επομένως
\[
f^{-1}(x)=\frac{1}{2}\ln(x^2-1), \quad x>1.
\]

\item Το εμβαδόν του τριγώνου $OAM$ είναι
\[
S=\frac{1}{2}xy=\frac{1}{2}x\sqrt{e^{2x}+1}.
\]
Με τον κανόνα της αλυσίδας:
\[
\frac{\d S}{\d t}=\frac{\d S}{\d x}\frac{\d x}{\d t}.
\]
Υπολογίζουμε
\[
\frac{\d S}{\d x}
=
\frac{1}{2}\left(\sqrt{e^{2x}+1}+x\frac{e^{2x}}{\sqrt{e^{2x}+1}}\right).
\]
Τη στιγμή που το $M$ διέρχεται από τον άξονα $y'y$ είναι $x=0$, άρα
\[
\left.\frac{\d S}{\d x}\right|_{x=0}
=
\frac{1}{2}.
\]
Εφόσον $\dfrac{\d x}{\d t}=2\mu/\si{sec}$, παίρνουμε
\[
\frac{\d S}{\d t}=\frac{1}{2}\cdot 2=1\mu^2/\si{sec}.
\]
\end{erwthma}
\end{thema}
