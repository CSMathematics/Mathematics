%=========== ΛΥΣΗ - 23ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το πεδίο ορισμού της $f(x) = \frac{e^x - e^{-x}}{2}$ είναι το $\mathbb{R}$.
Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ ως πράξεις παραγωγίσιμων, με παράγωγο:
\[ f'(x) = \frac{e^x - e^{-x}(-x)'}{2} = \frac{e^x + e^{-x}}{2} \]
Επειδή $e^x > 0$ και $e^{-x} > 0$ για κάθε $x \in \mathbb{R}$, ισχύει $f'(x) > 0$ στο $\mathbb{R}$.
Συνεπώς, η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το $\mathbb{R}$.

Η συνάρτηση $f$ είναι συνεχής και γνησίως αύξουσα, άρα το σύνολο τιμών της είναι:
\[ f(\mathbb{R}) = \left( \lim_{x \to -\infty} f(x), \lim_{x \to +\infty} f(x) \right) \]
Υπολογίζουμε τα όρια:
\begin{align*}
\lim_{x \to +\infty} f(x) &= \lim_{x \to +\infty} \frac{e^x - e^{-x}}{2} = \frac{(+\infty) - 0}{2} = +\infty \\
\lim_{x \to -\infty} f(x) &= \lim_{x \to -\infty} \frac{e^x - e^{-x}}{2} = \frac{0 - (+\infty)}{2} = -\infty
\end{align*}
Άρα το σύνολο τιμών είναι το $f(\mathbb{R}) = (-\infty, +\infty) = \mathbb{R}$.
Εφόσον η $f$ είναι γνησίως αύξουσα, είναι συνάρτηση \textbf{1-1} (\eng{«ένα προς ένα»}) και κατά συνέπεια \textbf{αντιστρέφεται}. Το πεδίο ορισμού της $f^{-1}$ είναι το $f(\mathbb{R}) = \mathbb{R}$.

\item Η εξίσωση $f(x) = 1$ μπορεί να λυθεί μέσω της αντίστροφης συνάρτησης, εφόσον το $1 \in f(\mathbb{R})$.
Λόγω της ιδιότητας $f(x) = y \Leftrightarrow x = f^{-1}(y)$, η εξίσωση ισοδυναμεί με:
\[ x = f^{-1}(1) \]
Επειδή η $f^{-1}$ είναι συνάρτηση, η τιμή $f^{-1}(1)$ είναι \textbf{μοναδική}. Άρα η εξίσωση $f(x) = 1$ έχει ακριβώς μία λύση.
Αντικαθιστώντας στον τύπο της αντίστροφης, η μοναδική λύση είναι:
\[ x = \ln(1 + \sqrt{1^2+1}) = \ln(1 + \sqrt{2}) \]

\item Από τους υπολογισμούς στο Γ1, έχουμε βρει τα όρια της $f$ στο άπειρο:
\[ \lim_{x \to -\infty} f(x) = -\infty \quad \text{και} \quad \lim_{x \to +\infty} f(x) = +\infty \]
Επειδή κανένα από τα δύο όρια δεν ισούται με πραγματικό αριθμό, η συνάρτηση $f$ \textbf{δεν έχει} οριζόντιες ασύμπτωτες.

\item Υπολογίζουμε πρώτα το ορισμένο ολοκλήρωμα της $f(x)$ στο $[0, 1]$:
\begin{align*}
\int_0^1 f(x) \d x &= \int_0^1 \frac{e^x - e^{-x}}{2} \d x = \frac{1}{2} \left[ e^x + e^{-x} \right]_0^1 \\
&= \frac{1}{2} \left( (e^1 + e^{-1}) - (e^0 + e^0) \right) = \frac{1}{2} \left( e + \frac{1}{e} - 2 \right) \\
&= \frac{1}{2} \left( \frac{e^2 - 2e + 1}{e} \right) = \frac{(e-1)^2}{2e}
\end{align*}

Ζητείται το εμβαδόν του χωρίου μεταξύ της $C_{f^{-1}}$, του άξονα $x'x$, της ευθείας $x=0$ (άξονας $y'y$) και της κατακόρυφης ευθείας $x = \frac{e^2-1}{2e}$.
Παρατηρούμε ότι το δεξιό άκρο ολοκλήρωσης ταυτίζεται με το $f(1)$:
\[ f(1) = \frac{e - e^{-1}}{2} = \frac{e - \frac{1}{e}}{2} = \frac{\frac{e^2-1}{e}}{2} = \frac{e^2-1}{2e} \]
Επίσης, η $f^{-1}(x)$ για $x \in \left[0, \frac{e^2-1}{2e}\right]$ είναι θετική (καθώς $x \ge 0 \Rightarrow f^{-1}(x) \ge f^{-1}(0) \Rightarrow f^{-1}(x) \ge 0$).
Άρα το ζητούμενο εμβαδόν είναι:
\[ E = \int_0^{f(1)} f^{-1}(x) \d x \]
Για τον υπολογισμό αυτού του ολοκληρώματος κάνουμε αλλαγή μεταβλητής θέτοντας $x = f(u)$, οπότε $\d x = f'(u) \d u$.
Για $x = 0 \Rightarrow f(u) = 0 \implies u = 0$.
Για $x = f(1) \Rightarrow f(u) = f(1) \implies u = 1$ (λόγω του 1-1).
Έτσι το εμβαδόν γράφεται:
\[ E = \int_0^1 u \cdot f'(u) \d u \]
Εφαρμόζουμε παραγοντική ολοκλήρωση:
\begin{align*}
E &= \int_0^1 u (f(u))' \d u = \left[ u f(u) \right]_0^1 - \int_0^1 (u)' f(u) \d u \\
&= (1 \cdot f(1) - 0 \cdot f(0)) - \int_0^1 f(u) \d u \\
&= f(1) - \int_0^1 f(x) \d x
\end{align*}
Αντικαθιστώντας τις τιμές που έχουμε βρει:
\[ E = \frac{e^2-1}{2e} - \frac{e^2 - 2e + 1}{2e} = \frac{e^2 - 1 - e^2 + 2e - 1}{2e} = \frac{2e - 2}{2e} = \frac{e-1}{e} \ \text{τ.μ.} \]
\end{erwthma}
\end{thema}
