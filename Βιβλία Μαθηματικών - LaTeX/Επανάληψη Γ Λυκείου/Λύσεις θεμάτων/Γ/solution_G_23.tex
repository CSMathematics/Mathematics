\begin{erwthma}
\item Η συνάρτηση $f(x) = \dfrac{e^x - e^{-x}}{2}$ έχει πεδίο ορισμού $D_f = \mathbb{R}$.
Παραγωγίζοντας τη συνάρτηση, έχουμε:
\[ f'(x) = \frac{1}{2} (e^x - (-e^{-x})) = \frac{e^x + e^{-x}}{2} \]
Επειδή $e^x > 0$ και $e^{-x} > 0$ για κάθε $x \in \mathbb{R}$, folgt ότι $f'(x) > 0$ για κάθε $x \in \mathbb{R}$.
Συνεπώς, η $f$ είναι $\textbf{γνησίως αύξουσα}$ σε όλο το $\mathbb{R}$.

Για τον προσδιορισμό του συνόλου τιμών $S_f$, υπολογίζουμε τα όρια στα άκρα του πεδίου ορισμού:
\[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} \frac{e^x - e^{-x}}{2} = \frac{+\infty - 0}{2} = +\infty \]
\[ \lim_{x \to -\infty} f(x) = \lim_{x \to -\infty} \frac{e^x - e^{-x}}{2} = \frac{0 - (+\infty)}{2} = -\infty \]
Επειδή η $f$ είναι $\textbf{συνεχώς}$ και $\textbf{γνησίως αύξουσα}$ στο $\mathbb{R}$, το σύνολο τιμών της είναι $S_f = \mathbb{R}$.
Επιπλέον, καθώς η $f$ είναι $\textbf{αντιστρέφουσα}$ (είναι $\eng{bijection}$ από το $\mathbb{R}$ στο $\mathbb{R}$), η συνάρτηση αντιστρέφεται.

\item Η εξίσωση $f(x) = 1$ είναι ισοδύναμη με την εξίσωση $x = f^{-1}(1)$.
Δεδομένου ότι η αντίστροφη συνάρτηση είναι $\eng{inverse} f^{-1}(x) = \ln(x + \sqrt{x^2+1})$, έχουμε:
\[ x = \ln(1 + \sqrt{1^2+1}) = \ln(1 + \sqrt{2}) \]
Επειδή η $f$ είναι $\textbf{γνησίως αύξουσα}$ στο $\mathbb{R}$ και το $1 \in S_f$, η εξίσωση $f(x) = 1$ έχει $\textbf{ακριβώς μία}$ λύση, η οποία είναι $x = \ln(1 + \sqrt{2})$.

\item Τα όρια της συνάρτησης έχουν ήδη υπολογιστεί στο πρώτο ερώτημα:
\[ \lim_{x \to +\infty} f(x) = +\infty \]
\[ \lim_{x \to -\infty} f(x) = -\infty \]
Επειδή κανένα από τα δύο όρια δεν είναι πεπερασμένος αριθμός, η συνάρτηση $f(x)$ $\textbf{δεν έχει}$ οριζόντιες ασύμπτωτες.

\item Υπολογίζουμε το ορισμένο ολοκλήρωμα:
\begin{align*}
\int_0^1 f(x) \d x &= \int_0^1 \frac{e^x - e^{-x}}{2} \d x \\
&= \frac{1}{2} \int_0^1 (e^x - e^{-x}) \d x \\
&= \frac{1}{2} \left[ e^x - (-e^{-x}) \right]_0^1 \\
&= \frac{1}{2} \left[ e^x + e^{-x} \right]_0^1 \\
&= \frac{1}{2} \left( (e^1 + e^{-1}) - (e^0 + e^0) \right) \\
&= \frac{1}{2} \left( e + \frac{1}{e} - 2 \right)
\end{align*}
Άρα, $\int_0^1 f(x) \d x = \frac{e + e^{-1} - 2}{2}$.
\end{erwthma}
\end{thema}
