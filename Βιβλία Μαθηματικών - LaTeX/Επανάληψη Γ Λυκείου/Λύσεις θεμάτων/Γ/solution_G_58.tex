%=========== ΛΥΣΗ - 58ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Το χωρίο $\Omega$ βρίσκεται κάτω από την παραβολή $y=x^2$, πάνω από τον άξονα $x'x$ και ανάμεσα στις ευθείες $x=0$ και $x=4$.
Άρα το συνολικό εμβαδόν του είναι
\[E=\dint_0^4 x^2\d x=\left[\frac{x^3}{3}\right]_0^4=\frac{64}{3}\]

\item Η ευθεία $y=ax$, με $a>0$, τέμνει την παραβολή $y=x^2$ στα σημεία που ικανοποιούν
\[x^2=ax\Leftrightarrow x(x-a)=0\]
δηλαδή στα σημεία με τετμημένες
\[x=0\quad\text{ή}\quad x=a\]
Για να υπάρχει μη μηδενικό πάνω μέρος μέσα στο δοσμένο χωρίο, παίρνουμε
\[0<a<4\]
Στο διάστημα $(a,4)$ η παραβολή βρίσκεται πάνω από την ευθεία $y=ax$.
Άρα το εμβαδόν του πάνω μέρους είναι
\begin{align*}
E_1(a)&=\dint_a^4 (x^2-ax)\d x\\
&=\left[\frac{x^3}{3}-\frac{ax^2}{2}\right]_a^4\\
&=\frac{64}{3}-8a-\left(\frac{a^3}{3}-\frac{a^3}{2}\right)\\
&=\frac{a^3}{6}-8a+\frac{64}{3}\ ,\ 0<a<4
\end{align*}

\item Για να διχοτομεί η ευθεία το συνολικό εμβαδόν, πρέπει
\[E_1(a)=\frac{E}{2}=\frac{32}{3}\]
Άρα
\begin{align*}
\frac{a^3}{6}-8a+\frac{64}{3}&=\frac{32}{3}\\
\frac{a^3}{6}-8a+\frac{32}{3}&=0\\
a^3-48a+64&=0
\end{align*}
Θεωρούμε τη συνάρτηση
\[p(a)=a^3-48a+64\ ,\ a\in(0,4)\]
Έχουμε
\[p'(a)=3a^2-48=3(a^2-16)<0\ ,\ 0<a<4\]
άρα η $p$ είναι γνησίως φθίνουσα στο $(0,4)$.
Επίσης
\[p(1)=1-48+64=17>0\]
και
\[p(2)=8-96+64=-24<0\]
Επομένως, από το θεώρημα \eng{Bolzano} και τη γνησία μονοτονία της $p$, υπάρχει μοναδική λύση
\[a_0\in(1,2)\]
της εξίσωσης
\[a^3-48a+64=0\]
Αυτή είναι η μοναδική κλίση που διχοτομεί το ολικό εμβαδόν.

\item Έστω $a=a(t)$ η κλίση της ευθείας τη χρονική στιγμή $t$.
Δίνεται ότι
\[a'(t)=1\]
Το εμβαδόν του άνω χωρίου είναι
\[E_1(a)=\frac{a^3}{6}-8a+\frac{64}{3}\]
οπότε
\[E_1'(a)=\frac{a^2}{2}-8\]
Με τον κανόνα της αλυσίδας,
\[\frac{\d E_1}{\d t}=E_1'(a)\cdot a'(t)\]
Τη στιγμή που $a=2$ έχουμε
\[\frac{\d E_1}{\d t}=\left(\frac{2^2}{2}-8\right)\cdot1=-6\]
Άρα το εμβαδόν του άνω χωρίου μειώνεται με ρυθμό
\[6\ \text{τετρ. μονάδες}/\sec\]
\end{erwthma}
\end{thema}
