\begin{thema}{Γ}
\begin{erwthma}
    \item Το συνολικό εμβαδόν του χωρίου $\Omega$ είναι:
    \[ E = \int_{0}^{4} x^2 \d x = \left[ \frac{x^3}{3} \right]_{0}^{4} = \frac{64}{3} - 0 = \frac{64}{3} \text{ τ.μ.} \]

    \item Η ευθεία $y = ax$ τέμνει την παραβολή $y = x^2$ στο σημείο όπου $x^2 = ax \Leftrightarrow x(x-a) = 0$. Εφόσον $a > 0$, τα σημεία τομής είναι το $O(0,0)$ και το $A(a, a^2)$. 
    Το χωρίο $\Omega$ ορίζεται για $x \in [0, 4]$ και $0 \le y \le x^2$. Το «πάνω μέρος» $E_1$ είναι το τμήμα του χωρίου $\Omega$ που βρίσκεται πάνω από την ευθεία $y=ax$, δηλαδή όπου $ax < y \le x^2$. Αυτό συμβαίνει για $x \in (a, 4)$ (με την προϋπόθεση $a < 4$).
    Το εμβαδόν του πάνω μέρους $E_1$ συναρτήσει του $a$ είναι:
    \begin{align*}
        E_1(a) &= \int_{a}^{4} (x^2 - ax) \d x \\
        &= \left[ \frac{x^3}{3} - \frac{ax^2}{2} \right]_{a}^{4} \\
        &= \left( \frac{64}{3} - \frac{16a}{2} \right) - \left( \frac{a^3}{3} - \frac{a^3}{2} \right) \\
        &= \frac{64}{3} - 8a - \left( -\frac{a^3}{6} \right) \\
        &= \frac{a^3}{6} - 8a + \frac{64}{3} \text{ τ.μ.}
    \end{align*}

    \item Για να διχοτομήσει η ευθεία το συνολικό εμβαδόν $E$, πρέπει $E_1(a) = \frac{1}{2}E = \frac{32}{3}$.
    \[ \frac{a^3}{6} - 8a + \frac{64}{3} = \frac{32}{3} \Leftrightarrow \frac{a^3}{6} - 8a + \frac{32}{3} = 0 \]
    Πολλαπλασιάζοντας όλα τα μέλη με $6$, προκύπτει η εξίσωση:
    \[ a^3 - 48a + 64 = 0 \]
    Ελέγχοντας πιθανές ρίζες, παρατηρούμε ότι η συνάρτηση $f(a) = a^3 - 48a + 64$ αλλάζει πρόσημο στο διάστημα $(1, 2)$ καθώς $f(1) = 17 > 0$ και $f(2) = 8 - 96 + 64 = -24 < 0$. Η τιμή της κλίσης $a$ είναι η μοναδική ρίζα αυτής της εξίσωσης στο διάστημα $(0, 4)$.

    \item Ο ρυθμός μεταβολής του εμβαδού $E(a)$ ως προς τον χρόνο $t$ δίνεται από τον κανόνα της αλυσίδας (\eng{Chain Rule}):
    \[ \frac{dE}{dt} = \frac{dE}{da} \cdot \frac{da}{dt} \]
    Υπολογίζουμε την παράγωγο του $E_1(a)$ ως προς $a$:
    \[ \frac{dE_1}{da} = \frac{3a^2}{6} - 8 = \frac{a^2}{2} - 8 \]
    Για $a = 2$ και $\frac{da}{dt} = 1\ \text{μον}/\text{sec}$, έχουμε:
    \[ \frac{dE}{dt} = \left( \frac{2^2}{2} - 8 \right) \cdot 1 = (2 - 8) \cdot 1 = -6\ \text{τ.μ.}/\text{sec} \]
    Άρα, το εμβαδόν του άνω χωρίου μειώνεται με ρυθμό $6\ \text{τ.μ.}/\text{sec}$.
\end{erwthma}
