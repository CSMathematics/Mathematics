\begin{erwthma}
\item Η $f$ είναι άρτια, οπότε
\[
f(-x)=f(x),\quad x\in[-3,3].
\]
Για παράδειγμα, από το σχήμα έχουμε
\[
f(-1)=f(1)=2.
\]
Άρα δύο διαφορετικές τιμές του $x$ έχουν την ίδια εικόνα. Επομένως, η $f$ δεν είναι $1-1$ και άρα δεν αντιστρέφεται στο $[-3,3]$.

Από το σχήμα, η $f$ μηδενίζεται στα $x=-2$ και $x=2$ και είναι αρνητική για
\[
x\in[-3,-2)\cup(2,3].
\]

Θέλουμε
\[
f(f(x))<0.
\]
Αυτό σημαίνει ότι η τιμή $f(x)$ πρέπει να ανήκει στο σύνολο όπου η $f$ είναι αρνητική, δηλαδή
\[
f(x)\in[-3,-2)\cup(2,3].
\]

Για $x\in[0,3]$, από το σχήμα η $f$ είναι γνησίως φθίνουσα και οι τιμές της ανήκουν στο διάστημα $[-1,3]$. Επομένως δεν μπορεί να ισχύει
\[
f(x)\in[-3,-2).
\]
Άρα αρκεί να λύσουμε
\[
f(x)\in(2,3].
\]

Στο διάστημα $[0,3]$ ισχύει
\[
f(0)=3,\qquad f(1)=2,
\]
και η $f$ είναι γνησίως φθίνουσα. Επομένως
\[
f(x)>2 \Leftrightarrow x\in[0,1).
\]

Άρα η λύση της ανίσωσης είναι
\[
x\in[0,1).
\]

\item Επειδή η $f$ είναι άρτια, έχουμε
\[
f(-x)=f(x).
\]
Άρα το ορθογώνιο έχει βάση
\[
AB=2x
\]
και ύψος
\[
f(x).
\]
Επομένως, το εμβαδόν του είναι
\[
E(x)=2x f(x).
\]

Παραγωγίζοντας ως προς τον χρόνο $t$, έχουμε
\[
E'(t)=2x'(t)f(x(t))+2x(t)f'(x(t))x'(t).
\]

Δίνεται ότι
\[
x'(t)=1\,\mu/\si{sec}.
\]
Τη στιγμή που $x=1$, από το σχήμα έχουμε
\[
f(1)=2,
\]
ενώ δίνεται
\[
f'(1)=-2.
\]

Άρα
\[
E'(t)=2\cdot 1\cdot 2+2\cdot 1\cdot (-2)\cdot 1=4-4=0.
\]

Συνεπώς, τη στιγμή που $x=1$, το εμβαδόν του ορθογωνίου έχει ρυθμό μεταβολής
\[
0\,\mu^2/\si{sec}.
\]

\item Επειδή η $f$ είναι άρτια, για κάθε $x$ κοντά στο $0$ ισχύει
\[
f(-x)=f(x).
\]
Αν η $f$ είναι παραγωγίσιμη στο $0$, τότε
\[
f'(0)=\lim_{x\to 0}\frac{f(x)-f(0)}{x}.
\]
Θέτοντας $x=-h$, έχουμε
\[
\frac{f(-h)-f(0)}{-h}
=-\frac{f(h)-f(0)}{h}.
\]
Άρα, αν υπάρχει το όριο, είναι ίσο με το αντίθετό του. Επομένως
\[
f'(0)=0.
\]

Από το σχήμα έχουμε
\[
f(0)=3.
\]
Δίνεται ότι
\[
\lim_{x\to 0}\frac{f(x)-3}{x^2}=-1.
\]

Επειδή η $f$ είναι δύο φορές παραγωγίσιμη στο $0$ και $f'(0)=0$, ισχύει
\[
f''(0)=\lim_{x\to 0}2\frac{f(x)-f(0)-f'(0)x}{x^2}.
\]
Άρα
\[
f''(0)=2\lim_{x\to 0}\frac{f(x)-3}{x^2}=2\cdot(-1)=-2.
\]

\item Επειδή η $f$ είναι άρτια,
\[
\int_{-2}^{2} f(t)\,\d t=2\int_0^2 f(t)\,\d t.
\]
Δίνεται ότι
\[
\int_0^2 f(t)\,\d t=\frac{7}{2}.
\]
Άρα
\[
\int_{-2}^{2} f(t)\,\d t=2\cdot \frac{7}{2}=7.
\]

Επομένως,
\[
g(x)=x^2+7x.
\]
Άρα
\[
g(1)=1^2+7\cdot 1=8
\]
και
\[
g'(x)=2x+7,
\]
οπότε
\[
g'(1)=9.
\]

Η εφαπτομένη της $C_g$ στο σημείο με τετμημένη $x_0=1$ είναι
\[
y-g(1)=g'(1)(x-1).
\]
Άρα
\[
y-8=9(x-1)
\]
και τελικά
\[
y=9x-1.
\]
\end{erwthma}
\end{thema}
