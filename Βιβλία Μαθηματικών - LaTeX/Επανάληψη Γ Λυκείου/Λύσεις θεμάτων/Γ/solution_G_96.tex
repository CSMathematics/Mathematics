\begin{erwthma}
\item Η $f$ είναι γνησίως αύξουσα στο $[0,4]$. Επομένως είναι $1-1$, άρα αντιστρέφεται.

Το πεδίο ορισμού της $f^{-1}$ είναι το σύνολο τιμών της $f$. Επειδή
\[
f(0)=0
\quad \text{και}\quad
f(4)=3,
\]
και η $f$ είναι γνησίως αύξουσα, το σύνολο τιμών της είναι
\[
f([0,4])=[0,3].
\]

Άρα το πεδίο ορισμού της $f^{-1}$ είναι
\[
[0,3].
\]

\item Για μια συνεχή και γνησίως μονότονη συνάρτηση ισχύει η γνωστή σχέση εμβαδών:
\[
\int_0^4 f(x)\,\d x+\int_0^3 f^{-1}(y)\,\d y=4\cdot 3.
\]

Δίνεται ότι
\[
\int_0^4 f(x)\,\d x=4.
\]
Άρα
\[
4+\int_0^3 f^{-1}(y)\,\d y=12.
\]
Επομένως,
\[
\int_0^3 f^{-1}(y)\,\d y=8.
\]

\item Επειδή η $f$ είναι γνησίως αύξουσα και $f(0)=0$, για κάθε $x\in(0,4]$ ισχύει
\[
f(x)>0.
\]
Άρα η παράσταση είναι ορισμένη για $x\to 0^+$.

Για κάθε $x>0$ ισχύει
\[
-1\leq \hm\left(\frac{1}{f(x)}\right)\leq 1.
\]
Πολλαπλασιάζοντας με $x>0$, παίρνουμε
\[
-x\leq x\hm\left(\frac{1}{f(x)}\right)\leq x.
\]

Επειδή
\[
\lim_{x\to 0^+}(-x)=0
\quad \text{και}\quad
\lim_{x\to 0^+}x=0,
\]
από το Κριτήριο Παρεμβολής προκύπτει
\[
\lim_{x\to 0^+}x\hm\left(\frac{1}{f(x)}\right)=0.
\]

\item Από τα δεδομένα έχουμε
\[
\int_0^4 f(x)\,\d x=4.
\]
Άρα η εξίσωση
\[
4f(\xi)=\int_0^4 f(x)\,\d x
\]
είναι ισοδύναμη με
\[
4f(\xi)=4
\]
δηλαδή
\[
f(\xi)=1.
\]

Η $f$ είναι συνεχής στο $[0,4]$ και
\[
f(0)=0<1<3=f(4).
\]
Άρα, από το Θεώρημα \eng{Bolzano}, υπάρχει τουλάχιστον ένα $\xi\in(0,4)$ τέτοιο ώστε
\[
f(\xi)=1.
\]

Επειδή η $f$ είναι γνησίως αύξουσα στο $[0,4]$, η εξίσωση $f(x)=1$ έχει το πολύ μία λύση.

Συνεπώς, υπάρχει ακριβώς ένα $\xi\in(0,4)$ τέτοιο ώστε
\[
4f(\xi)=\int_0^4 f(x)\,\d x.
\]
\end{erwthma}
\end{thema}
