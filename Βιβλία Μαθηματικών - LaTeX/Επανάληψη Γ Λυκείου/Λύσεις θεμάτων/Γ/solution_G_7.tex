%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f(x)=\sqrt{x}$ είναι παραγωγίσιμη στο $(0,+\infty)$ με παράγωγο:
\[f'(x)=\frac{1}{2\sqrt{x}}\]
Η εφαπτομένη της $C_f$ στο σημείο $M(x_0,\sqrt{x_0})$ (με $x_0>0$) έχει εξίσωση:
\[y-f(x_0)=f'(x_0)(x-x_0) \Leftrightarrow y-\sqrt{x_0}=\frac{1}{2\sqrt{x_0}}(x-x_0)\]
Για να βρούμε το σημείο τομής $B$ με τον άξονα $x'x$, θέτουμε $y=0$:
\[-\sqrt{x_0}=\frac{1}{2\sqrt{x_0}}(x-x_0) \Leftrightarrow -2x_0=x-x_0 \Leftrightarrow x=-x_0\]
Επομένως, η εφαπτομένη τέμνει τον άξονα $x'x$ στο σημείο $B(-x_0,0)$.

\item Το τρίγωνο $OBM$ έχει κορυφές την αρχή των αξόνων $O(0,0)$, το σημείο $B(-x_0,0)$ πάνω στον άξονα $x'x$ και το σημείο $M(x_0,\sqrt{x_0})$.
Η βάση του πάνω στον άξονα $x'x$ είναι το ευθύγραμμο τμήμα $OB$, το οποίο έχει μήκος:
\[|OB| = |-x_0 - 0| = |-x_0| = x_0 \quad (\text{αφού } x_0>0)\]
Το αντίστοιχο ύψος από την κορυφή $M$ στον άξονα $x'x$ είναι η τεταγμένη του $M$, δηλαδή $h = \sqrt{x_0}$.
Άρα το εμβαδόν του τριγώνου $OBM$ είναι:
\[E(x_0)=\frac{1}{2}\cdot|OB|\cdot h = \frac{1}{2}x_0\sqrt{x_0} = \frac{x_0\sqrt{x_0}}{2}\]

\item Έστω $\theta(t)$ η γωνία που σχηματίζει η εφαπτομένη με τον άξονα $x'x$ και $x(t)$ η τετμημένη του οχήματος τη χρονική στιγμή $t$. Ο συντελεστής διεύθυνσης της εφαπτομένης είναι $\ef\theta(t)$, άρα:
\[\ef\theta(t)=f'(x(t))=\frac{1}{2\sqrt{x(t)}}=\frac{1}{2}x(t)^{-1/2}\]
Παραγωγίζουμε και τα δύο μέλη ως προς $t$ (με τον κανόνα της αλυσίδας):
\[\frac{1}{\syn^2\theta(t)}\cdot\theta'(t)=-\frac{1}{4}x(t)^{-3/2}\cdot x'(t) \Leftrightarrow \frac{1}{\syn^2\theta(t)}\cdot\theta'(t)=-\frac{1}{4x(t)\sqrt{x(t)}}\cdot x'(t)\]
Γνωρίζουμε ότι τη χρονική στιγμή $t_0$ είναι $x(t_0)=4$ και $x'(t_0)=1\,\mu/\si{sec}$. Υπολογίζουμε την $\ef\theta(t_0)$:
\[\ef\theta(t_0)=\frac{1}{2\sqrt{4}}=\frac{1}{4}\]
Από τη βασική τριγωνομετρική ταυτότητα $\dfrac{1}{\syn^2\theta}=1+\ef^2\theta$, έχουμε:
\[\frac{1}{\syn^2\theta(t_0)}=1+\left(\frac{1}{4}\right)^2=1+\frac{1}{16}=\frac{17}{16}\]
Αντικαθιστώντας στη σχέση των παραγώγων για $t=t_0$:
\[\frac{17}{16}\cdot\theta'(t_0)=-\frac{1}{4\cdot 4\sqrt{4}}\cdot 1 \Leftrightarrow \frac{17}{16}\cdot\theta'(t_0)=-\frac{1}{32} \Leftrightarrow \theta'(t_0)=-\frac{1}{32}\cdot\frac{16}{17}=-\frac{1}{34}\,\si{rad/sec}\]

\item Το εμβαδόν του τριγώνου ως συνάρτηση του χρόνου είναι:
\[E(t)=\frac{x(t)\sqrt{x(t)}}{2}=\frac{1}{2}x(t)^{3/2}\]
Παραγωγίζουμε ως προς $t$:
\[E'(t)=\frac{1}{2}\cdot\frac{3}{2}x(t)^{1/2}\cdot x'(t)=\frac{3}{4}\sqrt{x(t)}\cdot x'(t)\]
Τη χρονική στιγμή $t_0$, με $x(t_0)=4$ και $x'(t_0)=1\,\mu/\si{sec}$, ο ρυθμός μεταβολής του εμβαδού είναι:
\[E'(t_0)=\frac{3}{4}\sqrt{4}\cdot 1=\frac{3}{4}\cdot 2=\frac{3}{2}\,\mu^2/\si{sec}\]
\end{erwthma}
\end{thema}
