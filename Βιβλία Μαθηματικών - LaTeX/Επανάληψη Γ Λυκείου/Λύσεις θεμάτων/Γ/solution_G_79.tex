%=========== ΛΥΣΗ - 79ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
(x-1)f'(x)-f(x)=(x-1)^2e^x
\]
διαιρούμε με $(x-1)^2$, αφού $x>1$, και παίρνουμε
\[
\frac{(x-1)f'(x)-f(x)}{(x-1)^2}=e^x
\]
Όμως
\[
\left(\frac{f(x)}{x-1}\right)'
=\frac{(x-1)f'(x)-f(x)}{(x-1)^2}
\]
Άρα
\[
\left(\frac{f(x)}{x-1}\right)'=e^x=(e^x)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
\frac{f(x)}{x-1}=e^x+c
\]
Για $x=2$ έχουμε $f(2)=e^2$, οπότε
\[
\frac{f(2)}{2-1}=e^2+c\Leftrightarrow e^2=e^2+c\Leftrightarrow c=0
\]
Άρα
\[
\frac{f(x)}{x-1}=e^x
\]
και τελικά
\[
f(x)=(x-1)e^x\ ,\quad x>1
\]

\item Από τον τύπο
\[
f(x)=(x-1)e^x
\]
έχουμε
\[
f'(x)=e^x+(x-1)e^x=xe^x
\]
Για κάθε $x>1$ ισχύει
\[
x>0\quad \text{και}\quad e^x>0
\]
άρα
\[
f'(x)>0
\]
Επομένως η $f$ είναι γνησίως αύξουσα στο $(1,+\infty)$.

Επιπλέον,
\[
\lim\limits_{x\to 1^+}f(x)
=\lim\limits_{x\to 1^+}(x-1)e^x=0
\]
και
\[
\lim\limits_{x\to+\infty}f(x)=+\infty
\]
Επειδή η $f$ είναι συνεχής και γνησίως αύξουσα στο $(1,+\infty)$, το σύνολο τιμών της είναι
\[
f((1,+\infty))=(0,+\infty)
\]

\item Από το πρώτο ερώτημα έχουμε
\[
f(x)=(x-1)e^x
\]
Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στη $f$ στο διάστημα $[x,x+1]$, όπου $x>1$.
Υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
f(x+1)-f(x)=f'(\xi)
\]
Δηλαδή
\[
xe^{x+1}-(x-1)e^x=\xi e^\xi
\]
Η συνάρτηση
\[
h(t)=te^t
\]
είναι γνησίως αύξουσα για $t>0$, αφού
\[
h'(t)=e^t+te^t=(t+1)e^t>0
\]
Επειδή $\xi>x>1$, έχουμε
\[
\xi e^\xi>xe^x
\]
Άρα
\[
xe^{x+1}-(x-1)e^x>xe^x
\]

\item Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $2$ είναι
\[
y-f(2)=f'(2)(x-2)
\]
Έχουμε
\[
f(2)=e^2
\]
και
\[
f'(2)=2e^2
\]
Άρα
\begin{align*}
y-e^2&=2e^2(x-2)\\
y&=e^2(2x-3)
\end{align*}

Επίσης
\[
f''(x)=\left(xe^x\right)'=(x+1)e^x>0
\]
για κάθε $x>1$.
Άρα η $f$ είναι κυρτή στο $(1,+\infty)$, οπότε η $C_f$ βρίσκεται πάνω από την εφαπτομένη της.

Το ζητούμενο εμβαδόν είναι
\[
E=\dint_1^2\left[(x-1)e^x-e^2(2x-3)\right]\d x
\]
Το ολοκλήρωμα στο άκρο $x=1$ νοείται με τη συνεχή επέκταση της $f$, αφού
\[
\lim\limits_{x\to1^+}f(x)=0
\]
Υπολογίζουμε:
\begin{align*}
\dint_1^2 (x-1)e^x\d x
&=\left[(x-2)e^x\right]_1^2\\
&=0-(-e)\\
&=e
\end{align*}
και
\begin{align*}
\dint_1^2 e^2(2x-3)\d x
&=e^2\left[x^2-3x\right]_1^2\\
&=e^2\left((-2)-(-2)\right)\\
&=0
\end{align*}
Άρα
\[
E=e
\]
\end{erwthma}
\end{thema}
