\begin{thema}{Γ}
\begin{erwthma}
\item Για $x>0$ έχουμε
\[f(x)=\frac{x^2+x+1}{x}=x+1+\frac{1}{x}\]
Άρα
\[f'(x)=1-\frac{1}{x^2}=\frac{x^2-1}{x^2}\]
Επειδή $x>0$, είναι $x^2>0$, οπότε το πρόσημο της $f'$ καθορίζεται από το $x^2-1$.
Έχουμε:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow x^2-1=0\Leftrightarrow x=1$
\item $f'(x)<0\Leftrightarrow 0<x<1$
\item $f'(x)>0\Leftrightarrow x>1$
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{ d, -, z, +, }
\tkzTabLine{ d, \searrow,t ,\nearrow,}
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$.
Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[f(1)=1+1+1=3\]

\item Επειδή
\[f(x)=x+1+\frac{1}{x}\]
έχουμε
\[\lim\limits_{x\to 0^+}f(x)=+\infty\]
άρα η ευθεία
\[x=0\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Επίσης
\[\lim\limits_{x\to+\infty}\left[f(x)-(x+1)\right]=\lim\limits_{x\to+\infty}\frac{1}{x}=0\]
άρα η ευθεία
\[y=x+1\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$.

\item Η ευθεία
\[y=\frac{3}{4}x+2024\]
έχει συντελεστή διεύθυνσης $\lambda=\dfrac{3}{4}$.
Ζητούμε εφαπτομένη της $C_f$ παράλληλη σε αυτήν, άρα πρέπει
\[f'(x_0)=\frac{3}{4}\]
Έχουμε
\begin{align*}
1-\frac{1}{x_0^2}&=\frac{3}{4}\\
\frac{1}{x_0^2}&=\frac{1}{4}\\
x_0^2&=4
\end{align*}
Επειδή $x_0>0$, παίρνουμε
\[x_0=2\]
Τότε
\[f(2)=2+1+\frac{1}{2}=\frac{7}{2}\]
Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $2$ είναι
\begin{align*}
y-f(2)&=f'(2)(x-2)\\
y-\frac{7}{2}&=\frac{3}{4}(x-2)\\
y&=\frac{3}{4}x+2
\end{align*}

\item Η πλάγια ασύμπτωτη της $C_f$ είναι
\[y=x+1\]
Για κάθε $x>0$ έχουμε
\[f(x)-(x+1)=\frac{1}{x}>0\]
άρα η $C_f$ βρίσκεται πάνω από την πλάγια ασύμπτωτή της.
Το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\dint_1^e\left[f(x)-(x+1)\right]\d x\\
&=\dint_1^e \frac{1}{x}\d x\\
&=\left[\ln x\right]_1^e\\
&=1
\end{align*}
\end{erwthma}
\end{thema}
