%=========== ΛΥΣΗ - 49ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Είναι
\[f'(x)=2x+a\]
Η εφαπτομένη της $C_f$ στο σημείο $M(x_0,f(x_0))$ έχει εξίσωση
\[y-f(x_0)=f'(x_0)(x-x_0)\]
δηλαδή
\[y-\left(x_0^2+ax_0+\beta\right)=(2x_0+a)(x-x_0)\]
Επειδή η εφαπτομένη διέρχεται από το σημείο $A(0,-1)$, έχουμε
\begin{align*}
-1-\left(x_0^2+ax_0+\beta\right)&=(2x_0+a)(0-x_0)\\
-1-x_0^2-ax_0-\beta&=-2x_0^2-ax_0\\
x_0^2&=\beta+1
\end{align*}
Άρα οι τετμημένες των σημείων επαφής των εφαπτομένων που διέρχονται από το $A(0,-1)$ είναι οι ρίζες της εξίσωσης
\[x_0^2-(\beta+1)=0\]
και η αντίστοιχη εφαπτομένη γράφεται, επειδή περνά από το $A(0,-1)$,
\[\varepsilon_{x_0}:y=(2x_0+a)x-1\]
όπου $x_0^2=\beta+1$.

\item Έστω $x_1,x_2$ οι τετμημένες των σημείων επαφής των δύο εφαπτομένων.
Από την εξίσωση
\[x_0^2-(\beta+1)=0\]
έχουμε, από τους τύπους του \eng{Vieta},
\[x_1+x_2=0\]
και
\[x_1x_2=-(\beta+1)\]
Οι συντελεστές διεύθυνσης των δύο εφαπτομένων είναι
\[\lambda_1=f'(x_1)=2x_1+a\]
και
\[\lambda_2=f'(x_2)=2x_2+a\]
Αφού οι εφαπτόμενες είναι κάθετες, ισχύει
\[\lambda_1\lambda_2=-1\]
οπότε
\begin{align*}
(2x_1+a)(2x_2+a)&=-1\\
4x_1x_2+2a(x_1+x_2)+a^2&=-1\\
4[-(\beta+1)]+2a\cdot 0+a^2&=-1\\
-4\beta-4+a^2&=-1\\
a^2-4\beta&=3
\end{align*}
Άρα
\[\beta=\frac{a^2-3}{4}\]
Αν επιπλέον $a=0$, τότε
\[\beta=-\frac{3}{4}\]
όπως ζητείται.

\item Για $a=0$ και $\beta=-\dfrac{3}{4}$ έχουμε
\[f(x)=x^2-\frac{3}{4}\]
Από το Γ1 οι τετμημένες των σημείων επαφής ικανοποιούν
\[x_0^2=\beta+1=-\frac{3}{4}+1=\frac{1}{4}\]
άρα
\[x_1=-\frac{1}{2}\quad\text{και}\quad x_2=\frac{1}{2}\]
Είναι
\[f\left(-\frac{1}{2}\right)=f\left(\frac{1}{2}\right)=-\frac{1}{2}\]
και
\[f'(x)=2x\]
Άρα οι εφαπτομένες είναι:
\begin{itemize}
\item στο $x_1=-\dfrac{1}{2}$:
\[y+\frac{1}{2}=-1\left(x+\frac{1}{2}\right)\Rightarrow y=-x-1\]
\item στο $x_2=\dfrac{1}{2}$:
\[y+\frac{1}{2}=1\left(x-\frac{1}{2}\right)\Rightarrow y=x-1\]
\end{itemize}
Στο διάστημα $\left[-\dfrac{1}{2},0\right]$ η παραβολή βρίσκεται πάνω από την ευθεία $y=-x-1$, ενώ στο διάστημα $\left[0,\dfrac{1}{2}\right]$ βρίσκεται πάνω από την ευθεία $y=x-1$.

Λόγω συμμετρίας ως προς τον άξονα $y'y$, το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=2\dint_0^{\frac{1}{2}}\left[\left(x^2-\frac{3}{4}\right)-(x-1)\right]\d x\\
&=2\dint_0^{\frac{1}{2}}\left(x^2-x+\frac{1}{4}\right)\d x\\
&=2\left[\frac{x^3}{3}-\frac{x^2}{2}+\frac{x}{4}\right]_0^{\frac{1}{2}}\\
&=2\left(\frac{1}{24}-\frac{1}{8}+\frac{1}{8}\right)\\
&=\frac{1}{12}
\end{align*}
Επομένως το ζητούμενο εμβαδόν είναι
\[E=\frac{1}{12}\]
\end{erwthma}
\end{thema}
