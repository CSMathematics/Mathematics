\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f(x) = \ln(x^2+1)$ είναι παραγωγίσιμη σε όλο το $\mathbb{R}$ (αφού $x^2+1 \ge 1 > 0$).
\textbf{Πρώτη παράγωγος:}
\[ f'(x) = \frac{2x}{x^2+1} \]

\textbf{Μονοτονία και ακρότατα:}
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow x = 0$
    \item $f'(x) < 0 \Leftrightarrow x < 0$ (γνησίως φθίνουσα στο $(-\infty, 0]$)
    \item $f'(x) > 0 \Leftrightarrow x > 0$ (γνησίως αύξουσα στο $[0, +\infty)$)
\end{itemize}
Η $f$ παρουσιάζει \textbf{ολικό ελάχιστο} στο $x = 0$ με τιμή $f(0) = \ln(1) = 0$.

\textbf{Δεύτερη παράγωγος:}
\begin{align*}
f''(x) &= \frac{(2x)'(x^2+1) - 2x(x^2+1)'}{(x^2+1)^2} = \frac{2(x^2+1) - 2x(2x)}{(x^2+1)^2} = \frac{2-2x^2}{(x^2+1)^2}
\end{align*}

\textbf{Κυρτότητα και σημεία καμπής:}
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow 2 - 2x^2 = 0 \Leftrightarrow x = \pm 1$
    \item $f''(x) > 0 \Leftrightarrow 2-2x^2 > 0 \Leftrightarrow x \in (-1, 1)$: \textbf{Κυρτή} (η $C_f$ κυρτή)
    \item $f''(x) < 0 \Leftrightarrow x \in (-\infty,-1) \cup (1, +\infty)$: \textbf{Κοίλη}
\end{itemize}
Η δεύτερη παράγωγος αλλάζει πρόσημο και στα δύο $x = \pm 1$, άρα υπάρχουν δύο σημεία καμπής:
\[ K_1(-1, f(-1)) = K_1\left(-1, \ln 2\right) \quad \text{και} \quad K_2(1, \ln 2) \]

\item Θεωρούμε τη βοηθητική συνάρτηση $h(x) = x^2 - \ln(x^2+1) = x^2 - f(x)$.
\[ h'(x) = 2x - f'(x) = 2x - \frac{2x}{x^2+1} = 2x\left(1 - \frac{1}{x^2+1}\right) = 2x \cdot \frac{x^2}{x^2+1} = \frac{2x^3}{x^2+1} \]
Εξετάζουμε το πρόσημο:
\begin{itemize}
    \item $h'(x) = 0 \Leftrightarrow x = 0$
    \item $h'(x) < 0 \Leftrightarrow x < 0$ (γνησίως φθίνουσα)
    \item $h'(x) > 0 \Leftrightarrow x > 0$ (γνησίως αύξουσα)
\end{itemize}
Άρα η $h$ παρουσιάζει ολικό ελάχιστο στο $x=0$ με τιμή $h(0) = 0 - \ln 1 = 0$.
Επομένως $h(x) \ge h(0) = 0$ για κάθε $x \in \mathbb{R}$, δηλαδή:
\[ x^2 - f(x) \ge 0 \implies \boldsymbol{f(x) \le x^2} \]

\item Το ζητούμενο όριο για $x \to +\infty$ έχει μορφή $\frac{+\infty}{+\infty}$.
Εφαρμόζουμε τον κανόνα \eng{De L'Hospital}:
\[ \lim_{x \to +\infty} \frac{f(x)}{x} = \lim_{x \to +\infty} \frac{\ln(x^2+1)}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x \to +\infty} \frac{f'(x)}{1} = \lim_{x \to +\infty} \frac{2x}{x^2+1} \]
Διαιρούμε αριθμητή και παρονομαστή με $x$:
\[ \lim_{x \to +\infty} \frac{2}{x + \frac{1}{x}} = \frac{2}{+\infty} = \boldsymbol{0} \]

\item \textbf{Απόδειξη ότι η $f'$ είναι περιττή:}
Έχουμε $f'(x) = \dfrac{2x}{x^2+1}$. Εξετάζουμε $f'(-x)$:
\[ f'(-x) = \frac{2(-x)}{(-x)^2+1} = \frac{-2x}{x^2+1} = -f'(x) \]
Εφόσον $f'(-x) = -f'(x)$ για κάθε $x \in \mathbb{R}$, η $f'$ είναι \textbf{περιττή συνάρτηση}.

\textbf{Εμβαδόν:} Ζητείται το εμβαδόν που περικλείεται από την $C_{f'}$, τον $x'x$ και τις ευθείες $x = \pm 1$.
Από τη μονοτονία, $f'(x) > 0$ για $x \in (0,1]$ και $f'(x) < 0$ για $x \in [-1, 0)$.
Λόγω της περιττής συμμετρίας της $f'$, το εμβαδόν στο $[-1, 0]$ ισούται με αυτό στο $[0, 1]$.
\begin{align*}
E &= \int_{-1}^{1} |f'(x)| \d x = 2 \int_0^1 f'(x) \d x = 2 \left[ f(x) \right]_0^1 \\
&= 2(f(1) - f(0)) = 2(\ln 2 - \ln 1) = 2\ln 2 = \boldsymbol{\ln 4 \ \text{τ.μ.}}
\end{align*}
\end{erwthma}
\end{thema}
