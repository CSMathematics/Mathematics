\item Η συνάρτηση $f$ είναι της μορφής $\ln(u(x))$ με $u(x) = x^2+1$. Η πρώτη παράγωγος είναι:
\[ f'(x) = \frac{u'(x)}{u(x)} = \frac{2x}{x^2+1} \]
Για τη δεύτερη παράγωγο, εφαρμόζουμε τον κανόνα του πηλίκου:
\begin{align*}
f''(x) &= \frac{(2x)'(x^2+1) - (2x)(x^2+1)'}{(x^2+1)^2} \\
&= \frac{2(x^2+1) - (2x)(2x)}{(x^2+1)^2} \\
&= \frac{2x^2+2-4x^2}{(x^2+1)^2} \\
&= \frac{2-2x^2}{(x^2+1)^2}
\end{align*}

\item Υπολογίζουμε το όριο:
\[ \lim_{x \to 0} \frac{f(x)}{f'(x)} = \lim_{x \to 0} \frac{\ln(x^2+1)}{\frac{2x}{x^2+1}} = \lim_{x \to 0} \frac{(x^2+1)\ln(x^2+1)}{2x} \]
Χρησιμοποιώντας το γνωστό όριο $\lim_{u \to 0} \frac{\ln(1+u)}{u} = 1$ με $u=x^2$:
\begin{align*}
\lim_{x \to 0} \frac{(x^2+1)\ln(x^2+1)}{2x} &= \lim_{x \to 0} \left( \frac{\ln(x^2+1)}{x^2} \cdot \frac{x^2(x^2+1)}{2x} \right) \\
&= \lim_{x \to 0} \left( \frac{\ln(x^2+1)}{x^2} \cdot \frac{x(x^2+1)}{2} \right) \\
&= 1 \cdot \frac{0(0+1)}{2} = 0
\end{align*}

\item Το όριο είναι της μορφής $\frac{\infty}{\infty}$, επομένως εφαρμόζουμε τον κανόνα του \eng{L'Hôpital}:
\[ \lim_{x \to +\infty} \frac{f(x)}{x} \xlongequal{\eng{L'Hôpital}} \lim_{x \to +\infty} \frac{f'(x)}{1} = \lim_{x \to +\infty} \frac{2x}{x^2+1} \]
