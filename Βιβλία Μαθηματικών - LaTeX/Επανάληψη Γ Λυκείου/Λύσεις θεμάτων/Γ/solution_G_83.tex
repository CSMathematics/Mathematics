%=========== ΛΥΣΗ - 83ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Αφού η $C_f$ έχει πλάγια ασύμπτωτη την ευθεία
\[
y=x-1
\]
στο $+\infty$, ισχύει
\[
\lim\limits_{x\to+\infty}\left[f(x)-(x-1)\right]=0
\]
Επίσης, επειδή
\[
\lim\limits_{x\to+\infty}(x-1)=+\infty
\]
και η διαφορά $f(x)-(x-1)$ τείνει στο $0$, παίρνουμε
\[
\lim\limits_{x\to+\infty}f(x)=+\infty
\]

Από το σχήμα φαίνεται ότι, καθώς $x\to 0^+$, η γραφική παράσταση ανεβαίνει απεριόριστα.
Άρα
\[
\lim\limits_{x\to0^+}f(x)=+\infty
\]

\item Από το σχήμα, στο διάστημα $(0,+\infty)$ η $f$ είναι:
\begin{itemize}
\item γνησίως φθίνουσα στο $(0,1]$
\item γνησίως αύξουσα στο $[1,+\infty)$
\end{itemize}
Επομένως η $f$ παρουσιάζει ολικό ελάχιστο στο σημείο
\[
E(1,1)
\]
δηλαδή
\[
f_{\min}=f(1)=1
\]

\item Από τον τύπο
\[
f(x)=x-1+\frac{1}{x}\ ,\quad x\neq 0
\]
ο δεξιός κλάδος έχει ολικό ελάχιστο στο $E(1,1)$.
Επιπλέον, στον αριστερό κλάδο η $f$ έχει τοπικό μέγιστο στο
\[
x=-1,\quad f(-1)=-3
\]
Άρα το πλήθος των λύσεων της εξίσωσης $f(x)=c$ είναι:
\begin{itemize}
\item $2$ λύσεις για $c>1$
\item $1$ λύση για $c=1$
\item καμία λύση για $-3<c<1$
\item $1$ λύση για $c=-3$
\item $2$ λύσεις για $c<-3$
\end{itemize}

\item Από τον τύπο
\[
f(x)=x-1+\frac{1}{x}
\]
έχουμε
\[
f'(x)=1-\frac{1}{x^2}
\]
Έστω
\[
y(t)=f(x(t))
\]
Με τον κανόνα της αλυσίδας,
\[
y'(t)=f'(x(t))x'(t)
\]
Τη στιγμή $t_0$ έχουμε
\[
x(t_0)=2
\quad \text{και}\quad
x'(t_0)=2
\]
Άρα
\begin{align*}
y'(t_0)
&=f'(2)\cdot 2\\
&=\left(1-\frac{1}{2^2}\right)\cdot 2\\
&=\frac{3}{4}\cdot 2\\
&=\frac{3}{2}
\end{align*}
Επομένως ο ρυθμός μεταβολής της τεταγμένης είναι
\[
\frac{3}{2}\ \mu/\sec
\]
\end{erwthma}
\end{thema}
