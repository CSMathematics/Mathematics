\begin{thema}{Γ}
\begin{erwthma}
\item Για να ορίζεται η συνάρτηση $f$, πρέπει το όρισμα του λογαρίθμου να είναι θετικό:
\[ \frac{x+1}{x} > 0 \Leftrightarrow x(x+1) > 0 \]
Οι ρίζες του τριωνύμου είναι $x=0$ και $x=-1$. Το τριώνυμο είναι θετικό (ομόσημο του $a=1$) εκτός του διαστήματος των ριζών. 
Άρα το πεδίο ορισμού είναι $D_f = (-\infty, -1) \cup (0, +\infty)$.

\textbf{Κατακόρυφες Ασύμπτωτες:} Ελέγχουμε τα ανοιχτά άκρα του πεδίου ορισμού, $x=0$ και $x=-1$.
\begin{itemize}
    \item Στο $x_0 = 0$: 
    \[ \lim_{x \to 0^+} f(x) = \lim_{x \to 0^+} x \ln\left(\frac{x+1}{x}\right) = \lim_{x \to 0^+} \left( x \ln(x+1) - x \ln x \right) \]
    Γνωρίζουμε ότι $\lim_{x \to 0^+} x \ln(x+1) = 0 \cdot \ln 1 = 0$. 
    Για το $\lim_{x \to 0^+} x \ln x$, έχουμε μορφή $0 \cdot (-\infty)$, οπότε γράφουμε:
    \[ \lim_{x \to 0^+} \frac{\ln x}{\frac{1}{x}} \xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}} \lim_{x \to 0^+} \frac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x \to 0^+} (-x) = 0 \]
    Άρα $\lim_{x \to 0^+} f(x) = 0 - 0 = 0$. Αφού το όριο είναι πραγματικός αριθμός, \textbf{δεν υπάρχει} κατακόρυφη ασύμπτωτη στο $0$.
    \item Στο $x_0 = -1$:
    \[ \lim_{x \to -1^-} f(x) = \lim_{x \to -1^-} x \ln\left(\frac{x+1}{x}\right) \]
    Καθώς $x \to -1^-$, έχουμε $x+1 \to 0^-$, άρα $\frac{x+1}{x} \to \frac{0^-}{-1} = 0^+$. Επομένως $\ln\left(\frac{x+1}{x}\right) \to -\infty$.
    Άρα, το όριο γίνεται $(-1) \cdot (-\infty) = +\infty$. 
    Άρα η ευθεία $\boldsymbol{x = -1}$ είναι \textbf{κατακόρυφη ασύμπτωτη} της $C_f$.
\end{itemize}

\textbf{Οριζόντιες Ασύμπτωτες:}
Υπολογίζουμε το όριο στο $\pm\infty$. Θέτουμε $u = \frac{1}{x}$. Όταν $x \to \pm\infty$, τότε $u \to 0$.
\[ \lim_{x \to \pm\infty} f(x) = \lim_{x \to \pm\infty} x \ln\left(1 + \frac{1}{x}\right) = \lim_{u \to 0} \frac{\ln(1+u)}{u} \]
Το όριο αυτό έχει μορφή $\frac{0}{0}$. Με τον κανόνα \eng{De L'Hospital}:
\[ \lim_{u \to 0} \frac{\ln(1+u)}{u} \xlongequal[\text{\eng{DLH}}]{\frac{0}{0}} \lim_{u \to 0} \frac{\frac{1}{1+u}}{1} = 1 \]
Άρα η ευθεία $\boldsymbol{y = 1}$ είναι \textbf{οριζόντια ασύμπτωτη} της $C_f$ τόσο στο $+\infty$ όσο και στο $-\infty$.

\item Η $f$ είναι παραγωγίσιμη στο πεδίο ορισμού της με:
\[ f'(x) = \ln\left(\frac{x+1}{x}\right) + x \left( \ln\left(\frac{x+1}{x}\right) \right)' = \ln\left(\frac{x+1}{x}\right) + x \cdot \frac{x}{x+1} \cdot \left( \frac{x+1}{x} \right)' \]
\[ f'(x) = \ln\left(\frac{x+1}{x}\right) + \frac{x^2}{x+1} \cdot \frac{1 \cdot x - (x+1) \cdot 1}{x^2} = \ln\left(\frac{x+1}{x}\right) - \frac{1}{x+1} \]
Το πρόσημο της $f'$ δεν είναι προφανές, οπότε υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x) = \frac{x}{x+1} \cdot \left( \frac{-1}{x^2} \right) - \left( \frac{-1}{(x+1)^2} \right) = \frac{-1}{x(x+1)} + \frac{1}{(x+1)^2} = \frac{-(x+1) + x}{x(x+1)^2} = \frac{-1}{x(x+1)^2} \]
\begin{itemize}
    \item Για $x > 0$: $f''(x) < 0$, άρα η $f'$ είναι \textbf{γνησίως φθίνουσα} στο $(0, +\infty)$.
    Το όριό της στο $+\infty$ είναι:
    \[ \lim_{x \to +\infty} f'(x) = \lim_{x \to +\infty} \left( \ln\left(1+\frac{1}{x}\right) - \frac{1}{x+1} \right) = \ln 1 - 0 = 0 \]
    Άρα, για κάθε $x \in (0, +\infty)$ ισχύει $x < +\infty \implies f'(x) > \lim\limits_{x \to +\infty} f'(x) \implies f'(x) > 0$.
    \item Για $x < -1$: $f''(x) > 0$, άρα η $f'$ είναι \textbf{γνησίως αύξουσα} στο $(-\infty, -1)$.
    Το όριό της στο $-\infty$ είναι:
    \[ \lim_{x \to -\infty} f'(x) = \lim_{x \to -\infty} \left( \ln\left(1+\frac{1}{x}\right) - \frac{1}{x+1} \right) = \ln 1 - 0 = 0 \]
    Άρα, για κάθε $x \in (-\infty, -1)$ ισχύει $x > -\infty \implies f'(x) > \lim\limits_{x \to -\infty} f'(x) \implies f'(x) > 0$.
\end{itemize}
Εφόσον αποδείξαμε ότι $f'(x) > 0$ σε κάθε ένα από τα διαστήματα του πεδίου ορισμού, η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $(-\infty, -1)$ και \textbf{γνησίως αύξουσα} στο $(0, +\infty)$.

Επειδή η $f$ είναι γνησίως αύξουσα στο διάστημα $(0, +\infty)$, για κάθε $x > 0$ έχουμε:
\[ x < +\infty \implies f(x) < \lim_{t \to +\infty} f(t) \]
Από το ερώτημα (α) γνωρίζουμε ότι $\lim_{t \to +\infty} f(t) = 1$. Επομένως αποδεικνύεται ότι:
\[ f(x) < 1 \text{ για κάθε } x > 0 \]

\item Οι συντεταγμένες του σημείου $M(x(t), y(t))$ είναι συναρτήσεις του χρόνου $t$, με $y(t) = f(x(t))$.
Παραγωγίζοντας τη σχέση ως προς τον χρόνο $t$, από τον κανόνα της αλυσίδας προκύπτει:
\[ y'(t) = f'(x(t)) \cdot x'(t) \]
Δίνεται ότι ο ρυθμός μεταβολής της τετμημένης είναι σταθερός: $x'(t) = 2 \ \mu/\si{sec}$.
Έστω $t_0$ η χρονική στιγμή που το σημείο διέρχεται από το $x=1$, δηλαδή $x(t_0) = 1$.
Ζητείται ο ρυθμός μεταβολής της τεταγμένης $y'(t_0)$.
Αντικαθιστούμε $t = t_0$:
\[ y'(t_0) = f'(x(t_0)) \cdot x'(t_0) = f'(1) \cdot 2 \]
Υπολογίζουμε την τιμή της παραγώγου στο $1$:
\[ f'(1) = \ln\left(\frac{1+1}{1}\right) - \frac{1}{1+1} = \ln 2 - \frac{1}{2} \]
Άρα ο ρυθμός μεταβολής της τεταγμένης είναι:
\[ y'(t_0) = 2 \left( \ln 2 - \frac{1}{2} \right) = \boldsymbol{2 \ln 2 - 1 \ \mu/\si{sec}} \]
\end{erwthma}
\end{thema}
