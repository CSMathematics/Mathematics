\begin{thema}{Γ}
\begin{erwthma}
\item Για να ορίζεται η συνάρτηση $f(x) = x \ln\left(\dfrac{x+1}{x}\right)$, θα πρέπει το όρισμα του λογαρίθμου να είναι θετικό, δηλαδή:
\[ \frac{x+1}{x} > 0 \]
Η ανισότητα αυτή ισχύει όταν ο αριθμητής και ο παρονομαστή έχουν το ίδιο πρόσημο:
\begin{itemize}
    \item $x+1 > 0$ και $x > 0 \Leftrightarrow x > -1$ και $x > 0 \Leftrightarrow x > 0$
    \item $x+1 < 0$ και $x < 0 \Leftrightarrow x < -1$ και $x < 0 \Leftrightarrow x < -1$
\end{itemize}
Συνεπώς, το πεδίο ορισμού είναι $D_f = (-\infty, -1) \cup (0, +\infty)$.

\item Υπολογίζουμε το όριο στο $+\infty$:
\[ \lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} x \ln\left(\frac{x+1}{x}\right) = \lim_{x \to +\infty} x \ln\left(1 + \frac{1}{x}\right) \]
Πρόκειται για απροσδιόριστη μορφή του τύπου $\infty \cdot 0$. Αναδιαμορφώνουμε το όριο ως εξής:
\[ \lim_{x \to +\infty} \frac{\ln\left(1 + \frac{1}{x}\right)}{\frac{1}{x}} \]
Θέτοντας $t = \frac{1}{x}$, καθώς $x \to +\infty$ έχουμε $t \to 0^+$. Το όριο γίνεται:
\[ \lim_{t \to 0^+} \frac{\ln(1+t)}{t} \]
Από το γνωστό θεμελιώδες όριο $\lim_{u \to 0} \frac{\ln(1+u)}{u} = 1$ (ή εφαρμόζοντας τον κανόνα \eng{L'Hôpital}), προκύπτει:
\[ \lim_{x \to +\infty} f(x) = 1 \]

\item Εξετάζουμε τις κατακόρυφες ασύμπτωτες στα άκρα του πεδίου ορισμού:
\begin{itemize}
    \item Για $x \to 0^+$:
    \[ \lim_{x \to 0^+} x \ln\left(\frac{x+1}{x}\right) = \lim_{x \to 0^+} [x \ln(x+1) - x \ln x] \]
    Γνωρίζουμε ότι $\lim_{x \to 0^+} x \ln(x+1) = 0 \cdot \ln(1) = 0$ και $\lim_{x \to 0^+} x \ln x = 0$.
    Άρα $\lim_{x \to 0^+} f(x) = 0$, οπότε η ευθεία $x = 0$ \textbf{δεν είναι} κατακόρυφη ασύμπτωτη.
    \item Για $x \to -1^-$:
    \[ \lim_{x \to -1^-} x \ln\left(\frac{x+1}{x}\right) \]
    Καθώς $x \to -1^-$, έχουμε $x \to -1$ και $\frac{x+1}{x} \to \frac{0^+}{-1} = 0^+$.
    Επομένως, $\ln\left(\frac{x+1}{x}\right) \to -\infty$.
    Συνεπώς:
    \[ \lim_{x \to -1^-} f(x) = (-1) \cdot (-\infty) = +\infty \]
    Άρα η ευθεία $x = -1$ είναι \textbf{κατακόρυφη ασύμπτωτη} της γραφικής παράστασης.
\end{itemize}

\item Από το ερώτημα (β), έχουμε ότι $\lim\limits_{x \to +\infty} f(x) = 1$.
Συνεπώς, η ευθεία $y = 1$ είναι \textbf{οριζόντια ασύμπτωτη} της γραφικής παράστασης της συνάρτησης στο $+\infty$.
\end{erwthma}
\end{thema}
