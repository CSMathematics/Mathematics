%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Για $f(x)=\sqrt{x}$ ισχύει
\[
f'(x)=\frac{1}{2\sqrt{x}}, \quad x>0.
\]
Η εφαπτομένη της $C_f$ στο σημείο $M(x_0,\sqrt{x_0})$ είναι
\[
y-\sqrt{x_0}=\frac{1}{2\sqrt{x_0}}(x-x_0).
\]
Για να βρούμε το σημείο τομής της με τον άξονα $x'x$, θέτουμε $y=0$:
\[
-\sqrt{x_0}=\frac{1}{2\sqrt{x_0}}(x-x_0)
\]
οπότε
\[
-2x_0=x-x_0
\]
και τελικά
\[
x=-x_0.
\]
Άρα η εφαπτομένη τέμνει τον άξονα $x'x$ στο σημείο
\[
B(-x_0,0).
\]

\item Το τρίγωνο $OBM$ έχει βάση
\[
OB=x_0
\]
και ύψος ίσο με την απόσταση του $M$ από τον άξονα $x'x$, δηλαδή
\[
\sqrt{x_0}.
\]
Άρα το εμβαδόν του είναι
\[
E(x_0)=\frac{1}{2}\cdot x_0\cdot \sqrt{x_0}
=
\frac{x_0\sqrt{x_0}}{2}.
\]

\item Αν $\theta$ είναι η γωνία που σχηματίζει η εφαπτομένη με τον άξονα $x'x$, τότε
\[
\ef \theta=f'(x)=\frac{1}{2\sqrt{x}}.
\]
Παραγωγίζουμε ως προς $t$:
\[
\frac{1}{\syf^2\theta}\frac{\d \theta}{\d t}
=
-\frac{1}{4x^{3/2}}\frac{\d x}{\d t}.
\]
Για $x=4$ και $\dfrac{\d x}{\d t}=1\mu/\si{sec}$:
\[
\ef\theta=\frac{1}{4}
\]
και
\[
\frac{1}{\syf^2\theta}=1+\ef^2\theta=1+\frac{1}{16}=\frac{17}{16}.
\]
Άρα
\[
\frac{17}{16}\frac{\d \theta}{\d t}
=
-\frac{1}{4\cdot 4^{3/2}}
=
-\frac{1}{32}
\]
οπότε
\[
\frac{\d \theta}{\d t}
=
-\frac{1}{32}\cdot \frac{16}{17}
=
-\frac{1}{34}\,\si{rad/sec}.
\]

\item Από το δεύτερο ερώτημα
\[
E(x)=\frac{1}{2}x^{3/2}.
\]
Άρα
\[
\frac{\d E}{\d t}=\frac{\d E}{\d x}\frac{\d x}{\d t}
\]
και
\[
\frac{\d E}{\d x}=\frac{1}{2}\cdot \frac{3}{2}x^{1/2}=\frac{3\sqrt{x}}{4}.
\]
Για $x=4$ και $\dfrac{\d x}{\d t}=1\mu/\si{sec}$:
\[
\frac{\d E}{\d t}
=
\frac{3\sqrt{4}}{4}\cdot 1
=
\frac{3}{2}\mu^2/\si{sec}.
\]
\end{erwthma}
\end{thema}
