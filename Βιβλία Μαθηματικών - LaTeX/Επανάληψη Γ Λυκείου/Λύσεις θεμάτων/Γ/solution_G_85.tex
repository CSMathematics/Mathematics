%=========== ΛΥΣΗ - 85ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη γραφική παράσταση της $f'$ βλέπουμε ότι η $C_{f'}$ τέμνει τον άξονα $x'x$ στα σημεία με τετμημένες
\[
x=-3,\quad x=1,\quad x=4
\]
Άρα οι ρίζες της $f'$ είναι
\[
-3,\ 1,\ 4
\]

Επίσης, από το πρόσημο της $f'$ στο σχήμα έχουμε:
\begin{itemize}
\item $f'(x)>0$ για $x\in(-3,1)$
\item $f'(x)<0$ για $x\in(1,4)$
\item $f'(x)>0$ για $x\in(4,5)$
\end{itemize}
Επομένως η $f$ είναι γνησίως αύξουσα στο $[-3,1]$, γνησίως φθίνουσα στο $[1,4]$ και γνησίως αύξουσα στο $[4,5]$.

Άρα η $f$ παρουσιάζει τοπικό μέγιστο στο $x=1$ και τοπικό ελάχιστο στο $x=4$.

\item Ισχύει
\[
f(b)-f(a)=\dint_a^b f'(x)\d x
\]
Από το σχήμα, στο διάστημα $[-3,-1]$ το εμβαδόν κάτω από την $C_{f'}$ είναι τρίγωνο με βάση $2$ και ύψος $4$.
Άρα
\[
\dint_{-3}^{-1} f'(x)\d x=\frac{1}{2}\cdot 2\cdot 4=4
\]
Επειδή $f(-3)=1$, παίρνουμε
\[
f(-1)=f(-3)+\dint_{-3}^{-1}f'(x)\d x=1+4=5
\]

Για το $f(2)$ έχουμε
\[
f(2)=f(-1)+\dint_{-1}^{2}f'(x)\d x
\]
Στο διάστημα $[-1,1]$ το εμβαδόν είναι θετικό τρίγωνο με βάση $2$ και ύψος $4$, ενώ στο $[1,2]$ είναι αρνητικό τρίγωνο με βάση $1$ και ύψος $2$.
Άρα
\[
\dint_{-1}^{2}f'(x)\d x=4-1=3
\]
οπότε
\[
f(2)=5+3=8
\]

\item Υπολογίζουμε πρώτα το $f(5)$.
Έχουμε
\[
f(5)=f(2)+\dint_2^5 f'(x)\d x
\]
Στο διάστημα $[2,4]$ το εμβαδόν είναι αρνητικό τρίγωνο με βάση $2$ και ύψος $2$, ενώ στο $[4,5]$ είναι θετικό τρίγωνο με βάση $1$ και ύψος $1$.
Άρα
\[
\dint_2^5 f'(x)\d x=-2+\frac{1}{2}=-\frac{3}{2}
\]
και επομένως
\[
f(5)=8-\frac{3}{2}=\frac{13}{2}
\]
Επειδή
\[
f(-1)=5
\quad \text{και}\quad
f(5)=\frac{13}{2}
\]
προκύπτει
\[
f(5)>f(-1)
\]

Για την εξίσωση $f(x)=f(-1)$ στο $(2,5)$, παρατηρούμε ότι στο $[2,4]$ η $f$ είναι γνησίως φθίνουσα και στο $[4,5]$ γνησίως αύξουσα.
Άρα στο $[2,5]$ η ελάχιστη τιμή της είναι η $f(4)$.
Είναι
\[
f(4)=f(2)+\dint_2^4 f'(x)\d x=8-2=6
\]
Άρα για κάθε $x\in[2,5]$ ισχύει
\[
f(x)\geq 6
\]
ενώ
\[
f(-1)=5
\]
Επομένως η εξίσωση
\[
f(x)=f(-1)
\]
δεν έχει λύση στο $(2,5)$.

\item Η κλίση της χορδής που ενώνει τα σημεία
\[
(-3,f(-3))
\quad \text{και}\quad
(5,f(5))
\]
είναι
\[
\lambda=\frac{f(5)-f(-3)}{5-(-3)}
=\frac{\frac{13}{2}-1}{8}
=\frac{\frac{11}{2}}{8}
=\frac{11}{16}
\]
Ζητούμε τα σημεία $\xi\in(-3,5)$ για τα οποία
\[
f'(\xi)=\frac{11}{16}
\]
Από το σχήμα οι κλάδοι της $f'$ είναι ευθύγραμμα τμήματα:
\[
f'(x)=2x+6,\quad x\in[-3,-1]
\]
\[
f'(x)=-2x+2,\quad x\in[-1,2]
\]
\[
f'(x)=x-4,\quad x\in[2,5]
\]
Λύνουμε σε κάθε τμήμα:
\begin{itemize}
\item $2x+6=\dfrac{11}{16}\Rightarrow x=-\dfrac{85}{32}$
\item $-2x+2=\dfrac{11}{16}\Rightarrow x=\dfrac{21}{32}$
\item $x-4=\dfrac{11}{16}\Rightarrow x=\dfrac{75}{16}$
\end{itemize}
Επομένως υπάρχουν τρία σημεία στα οποία η εφαπτομένη της $C_f$ είναι παράλληλη στη χορδή:
\[
\xi_1=-\frac{85}{32},\quad
\xi_2=\frac{21}{32},\quad
\xi_3=\frac{75}{16}
\]
\end{erwthma}
\end{thema}
