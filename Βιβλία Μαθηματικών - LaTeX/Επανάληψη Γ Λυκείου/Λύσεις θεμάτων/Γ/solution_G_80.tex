%=========== ΛΥΣΗ - 80ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
f'(x)-f(x)=e^x
\]
πολλαπλασιάζουμε με $e^{-x}$ και παίρνουμε
\[
e^{-x}f'(x)-e^{-x}f(x)=1
\]
Όμως
\[
\left(f(x)e^{-x}\right)'=f'(x)e^{-x}-f(x)e^{-x}
\]
Άρα
\[
\left(f(x)e^{-x}\right)'=1=(x)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
f(x)e^{-x}=x+c
\]
Για $x=0$ έχουμε $f(0)=0$, οπότε
\[
f(0)e^0=0+c\Leftrightarrow c=0
\]
Άρα
\[
f(x)e^{-x}=x
\]
και τελικά
\[
f(x)=xe^x\ ,\quad x\in\mathbb{R}
\]

\item Από τον τύπο
\[
f(x)=xe^x
\]
έχουμε
\[
f'(x)=(x+1)e^x
\]
και
\[
f''(x)=(x+2)e^x
\]
Επειδή $e^x>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f''$ καθορίζεται από το $x+2$.
Έχουμε:
\begin{itemize}
\item $f''(x)=0\Leftrightarrow x=-2$
\item $f''(x)<0\Leftrightarrow x<-2$
\item $f''(x)>0\Leftrightarrow x>-2$
\end{itemize}
Άρα η $f$ είναι κοίλη στο $(-\infty,-2]$ και κυρτή στο $[-2,+\infty)$.

Επειδή η $f''$ αλλάζει πρόσημο στο $x=-2$, η $C_f$ έχει μοναδικό σημείο καμπής το
\[
K(-2,f(-2))
\]
Επειδή
\[
f(-2)=-2e^{-2}=-\frac{2}{e^2}
\]
το σημείο καμπής είναι
\[
K\left(-2,-\frac{2}{e^2}\right)
\]

\item Η $C_f$ τέμνει τον άξονα $x'x$ όταν
\[
f(x)=0\Leftrightarrow xe^x=0\Leftrightarrow x=0
\]
Στο διάστημα $[0,1]$ ισχύει
\[
f(x)=xe^x\geq 0
\]
Άρα το ζητούμενο εμβαδόν είναι
\[
E=\dint_0^1 xe^x\d x
\]
Ολοκληρώνουμε κατά παράγοντες:
\begin{align*}
E
&=\dint_0^1 xe^x\d x\\
&=\left[(x-1)e^x\right]_0^1\\
&=0-(-1)\\
&=1
\end{align*}

\item Θέλουμε το όριο
\[
L=\lim\limits_{x\to 0}\frac{f(x)-\hm x}{x^2}
=\lim\limits_{x\to 0}\frac{xe^x-\hm x}{x^2}
\]
Για $x\to 0$ έχουμε απροσδιόριστη μορφή $\dfrac{0}{0}$.
Εφαρμόζοντας δύο φορές τον κανόνα \eng{De L' Hospital}, παίρνουμε
\begin{align*}
L
&=\lim\limits_{x\to 0}\frac{(x+1)e^x-\syn x}{2x}\\
&=\lim\limits_{x\to 0}\frac{(x+2)e^x+\hm x}{2}\\
&=\frac{2}{2}\\
&=1
\end{align*}
\end{erwthma}
\end{thema}
