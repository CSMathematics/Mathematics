\begin{thema}{Γ}
\begin{erwthma}
\item Η παράγωγος της συνάρτησης $f(x) = x^4 - 2x^2 + \lambda$ είναι:
\[ f'(x) = 4x^3 - 4x = 4x(x^2 - 1) = 4x(x-1)(x+1) \]
Οι ρίζες της παραγώγου είναι $x_1 = -1$, $x_2 = 0$, $x_3 = 1$ και το πρόσημό της διαμορφώνεται ως:
\begin{itemize}
    \item $f'(x) < 0$ για $x \in (-\infty, -1) \cup (0, 1)$
    \item $f'(x) > 0$ για $x \in (-1, 0) \cup (1, +\infty)$
\end{itemize}
Επομένως, η $f$ παρουσιάζει \textbf{τοπικά ελάχιστα} στα $x = \pm 1$ (αλλαγή από $f'<0$ σε $f'>0$) και \textbf{τοπικό μέγιστο} στο $x=0$.

Δεδομένου ότι ο $x'x$ εφάπτεται στη γραφική παράσταση ακριβώς στα τοπικά ελάχιστα, σημαίνει ότι τα τοπικά ελάχιστα βρίσκονται πάνω στον $x'x$, δηλαδή $f(\pm 1) = 0$.
Από τη συνθήκη $f(1) = 0$:
\[ 1^4 - 2(1)^2 + \lambda = 0 \implies 1 - 2 + \lambda = 0 \implies \boldsymbol{\lambda = 1} \]
Επαλήθευση για $x = -1$: $f(-1) = 1 - 2 + 1 = 0$

\item Με $\lambda = 1$, η συνάρτηση γράφεται:
\[ f(x) = x^4 - 2x^2 + 1 = (x^2)^2 - 2(x^2)(1) + 1^2 = (x^2 - 1)^2 \]
Εφόσον είναι τέλειο τετράγωνο, ισχύει $f(x) = (x^2-1)^2 \ge 0$ για κάθε $x \in \mathbb{R}$.

Υπολογίζουμε αναλυτικά το ολοκλήρωμα:
\[ G(a) = \int_0^a f(x)\d x = \int_0^a (x^4 - 2x^2 + 1)\d x = \left[ \frac{x^5}{5} - \frac{2x^3}{3} + x \right]_0^a = \frac{a^5}{5} - \frac{2a^3}{3} + a \]
Παραγοντοποιούμε:
\[ G(a) = a \left( \frac{a^4}{5} - \frac{2a^2}{3} + 1 \right) = \frac{a(3a^4 - 10a^2 + 15)}{15} \]
Εξετάζουμε το πρόσημο του τριωνύμου $p(u) = 3u^2 - 10u + 15$ (για $u = a^2 \ge 0$):
\[ \Delta = (-10)^2 - 4 \cdot 3 \cdot 15 = 100 - 180 = -80 < 0 \]
Εφόσον $\Delta < 0$ και ο συντελεστής του $u^2$ είναι θετικός, ισχύει $p(u) > 0$ για κάθε $u \ge 0$.
Άρα $3a^4 - 10a^2 + 15 > 0$ για κάθε $a \in \mathbb{R}$, και το πρόσημο του $G(a)$ ταυτίζεται με το πρόσημο του $a$:
\begin{itemize}
    \item Για $a > 0$: $G(a) > 0$
    \item Για $a = 0$: $G(0) = 0$
    \item Για $a < 0$: $G(a) < 0$
\end{itemize}
Άρα $\boldsymbol{\int_0^a f(x)\d x \ge 0}$ \textbf{για κάθε} $\boldsymbol{a \ge 0}$.

\textit{(Σημείωση: Για $a < 0$ το ολοκλήρωμα $\int_0^a f(x)\d x$ είναι αρνητικό, κάτι που δεν αντιφάσκει — απλά η εκφώνηση ζητά να αποδειχθεί η ανισότητα για $a \ge 0$, όπου έχει νόημα γεωμετρικά ως εμβαδόν).}

\item Η εξίσωση είναι $f(x) = (x^2-1)^2 = k$, με $k \in (0, 1)$.
Μελετάμε τη μονοτονία: $f$ φθίνουσα στα $(-\infty,-1), (0,1)$ και αύξουσα στα $(-1,0), (1,+\infty)$.
Η $f$ είναι άρτια, οπότε η γραφική παράσταση είναι συμμετρική ως προς τον $y'y$.
Για $x \ge 0$:
\begin{itemize}
    \item Στο $[0,1]$: $f$ φθίνουσα από $f(0)=1$ ως $f(1)=0$. Επειδή $k \in (0,1)$, υπάρχει μία ρίζα εδώ.
    \item Στο $[1,+\infty)$: $f$ αύξουσα από $f(1)=0$ ως $+\infty$. Υπάρχει μία ρίζα εδώ.
\end{itemize}
Από τη συμμετρία, υπάρχουν άλλες 2 αντίστοιχες ρίζες για $x \le 0$.
Άρα, η εξίσωση $f(x) = k$ έχει συνολικά $\boldsymbol{4}$ ρίζες για $k \in (0,1)$.

\item Από το Γ2, $\sqrt{f(x)} = \sqrt{(x^2-1)^2} = |x^2-1|$.
Εξετάζουμε τα πλευρικά όρια:
\begin{itemize}
    \item Για $x \to 1^+$: $x > 1 \Rightarrow x^2 > 1 \Rightarrow x^2-1 > 0$, άρα $|x^2-1| = x^2-1$:
    \[ \lim_{x \to 1^+} \frac{x^2-1}{x-1} = \lim_{x \to 1^+} \frac{(x-1)(x+1)}{x-1} = \lim_{x \to 1^+} (x+1) = 2 \]
    \item Για $x \to 1^-$: $x < 1 \Rightarrow x^2 < 1 \Rightarrow x^2-1 < 0$, άρα $|x^2-1| = -(x^2-1)$:
    \[ \lim_{x \to 1^-} \frac{-(x^2-1)}{x-1} = \lim_{x \to 1^-} \frac{-(x-1)(x+1)}{x-1} = \lim_{x \to 1^-} (-(x+1)) = -2 \]
\end{itemize}
Επειδή $\lim_{x \to 1^+} \dfrac{\sqrt{f(x)}}{x-1} = 2 \neq -2 = \lim_{x \to 1^-} \dfrac{\sqrt{f(x)}}{x-1}$, τα δύο πλευρικά όρια είναι άνισα.
Συνεπώς, το ζητούμενο όριο $\lim\limits_{x \to 1} \dfrac{\sqrt{f(x)}}{x-1}$ \textbf{δεν υπάρχει}.
\end{erwthma}
\end{thema}
