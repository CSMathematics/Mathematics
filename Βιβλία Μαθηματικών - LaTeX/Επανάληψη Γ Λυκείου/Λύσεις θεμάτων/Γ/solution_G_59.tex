%=========== ΛΥΣΗ - 59ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση
\[\phi(u)=ue^u\ ,\ u\geq 0\]
Έχουμε
\[\phi'(u)=e^u+ue^u=e^u(1+u)>0\ ,\ u\geq 0\]
άρα η $\phi$ είναι γνησίως αύξουσα στο $[0,+\infty)$.
Επίσης
\[\phi(0)=0\]
και
\[\lim\limits_{u\to+\infty}\phi(u)=+\infty\]
οπότε
\[\phi([0,+\infty))=[0,+\infty)\]

Η δοσμένη σχέση
\[f(x)e^{f(x)}=x\]
σημαίνει ότι
\[\phi(f(x))=x\]
Άρα
\[f=\phi^{-1}\]
στο $[0,+\infty)$. Επειδή η $\phi$ είναι γνησίως αύξουσα, και η αντίστροφή της $f$ είναι γνησίως αύξουσα στο $[0,+\infty)$.

\item Για $x=0$ έχουμε
\[f(0)e^{f(0)}=0\]
Επειδή $e^{f(0)}>0$, προκύπτει
\[f(0)=0\]

Για $x=e$ έχουμε
\[f(e)e^{f(e)}=e\]
Παρατηρούμε ότι $u=1$ ικανοποιεί
\[ue^u=1\cdot e=e\]
Επειδή η $\phi(u)=ue^u$ είναι γνησίως αύξουσα στο $[0,+\infty)$, η λύση είναι μοναδική.
Άρα
\[f(e)=1\]

\item Παραγωγίζουμε τη σχέση
\[f(x)e^{f(x)}=x\]
ως προς $x$:
\[f'(x)e^{f(x)}+f(x)e^{f(x)}f'(x)=1\]
Άρα
\[f'(x)e^{f(x)}(1+f(x))=1\]
και επομένως
\[f'(x)=\frac{1}{e^{f(x)}(1+f(x))}\]
Στο $x_0=e$ έχουμε, από το Γ2,
\[f(e)=1\]
οπότε
\[f'(e)=\frac{1}{e(1+1)}=\frac{1}{2e}\]
Η εφαπτομένη της $C_f$ στο σημείο $(e,f(e))\equiv(e,1)$ είναι
\begin{align*}
y-1&=\frac{1}{2e}(x-e)\\
y&=\frac{x}{2e}+\frac{1}{2}
\end{align*}

\item Θα αποδείξουμε πρώτα ότι
\[\lim\limits_{x\to+\infty}f(x)=+\infty\]
Πράγματι, αν το $f(x)$ δεν έτεινε στο $+\infty$, τότε, επειδή είναι γνησίως αύξουσα και μη αρνητική, θα ήταν άνω φραγμένη.
Τότε το γινόμενο $f(x)e^{f(x)}$ θα ήταν επίσης άνω φραγμένο, κάτι που είναι αδύνατο, αφού
\[f(x)e^{f(x)}=x\to+\infty\]
Άρα
\[\lim\limits_{x\to+\infty}f(x)=+\infty\]

Για $x>0$ και $f(x)>0$, από τη σχέση
\[f(x)e^{f(x)}=x\]
λογαριθμίζουμε:
\[\ln x=\ln f(x)+f(x)\]
Επομένως
\begin{align*}
\frac{f(x)}{\ln x}
&=\frac{f(x)}{f(x)+\ln f(x)}\\
&=\frac{1}{1+\frac{\ln f(x)}{f(x)}}
\end{align*}
Καθώς $x\to+\infty$, έχουμε $f(x)\to+\infty$, άρα
\[\frac{\ln f(x)}{f(x)}\to 0\]
Επομένως
\[\lim\limits_{x\to+\infty}\frac{f(x)}{\ln x}=1\]
\end{erwthma}
\end{thema}
