%=========== ΛΥΣΗ ΘΕΜΑΤΟΣ Γ.59 ===========
\begin{thema}{Γ}
Δίνεται η εξίσωση $f(x) e^{f(x)} = x$ για $x \ge 0$, με $f(x) \ge 0$.
\begin{erwthma}

\item Παραγωγίζουμε και τα δύο μέλη της δοσμένης εξίσωσης ως προς $x$.
Εφαρμόζουμε τον κανόνα γινομένου (και σύνθετης συνάρτησης για το εκθετικό) στο αριστερό μέλος:
\[ \left( f(x) e^{f(x)} \right)' = (x)' \]
\[ f'(x) e^{f(x)} + f(x) \left( e^{f(x)} \right)' = 1 \]
\[ f'(x) e^{f(x)} + f(x) e^{f(x)} f'(x) = 1 \]
Εξάγουμε το $f'(x) e^{f(x)}$ ως κοινό παράγοντα:
\[ f'(x) e^{f(x)} \left( 1 + f(x) \right) = 1 \]
Καθώς δίνεται ότι $f(x) \geq 0$, προκύπτει $1+f(x) \geq 1 > 0$, οπότε η παρένθεση δεν μηδενίζεται. Διαιρούμε:
\[ f'(x) = \frac{1}{e^{f(x)} (1 + f(x))} \]

\item Για να δείξουμε ότι είναι γνησίως αύξουσα, μελετάμε το πρόσημο της παραγώγου $f'(x)$ στο $[0, +\infty)$.
Από τα δεδομένα της άσκησης εξασφαλίζεται ότι $f(x) \ge 0$.
Επομένως:
- Το $e^{f(x)} > 0$ ως εκθετική.
- Η παρένθεση $(1 + f(x)) \ge 1 > 0$.
Συνεπώς, ο παρονομαστής είναι αυστηρά θετικός: $e^{f(x)} (1 + f(x)) > 0$.
Ως εκ τούτου:
\[ f'(x) = \frac{1}{e^{f(x)} (1 + f(x))} > 0 \quad \text{για κάθε } x \geq 0 \]
Καταλήγουμε ότι η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $[0, +\infty)$.

\item Για να βρούμε τις τιμές αναφοράς αντικαθιστούμε στην αρχική εξίσωση: $f(x) e^{f(x)} = x$.
- \textbf{Για $\boldsymbol{x = 0}$:}
  \[ f(0) e^{f(0)} = 0 \]
  Επειδή πάντα $e^{f(0)} > 0$, για να εκμηδενιστεί το γινόμενο είναι υποχρεωτικό να ισχύει $\mathbf{f(0) = 0}$.

- \textbf{Για $\boldsymbol{x = e}$:}
  \[ f(e) e^{f(e)} = e \]
  Παρατηρούμε ως προφανή λύση ότι αν η $f(e) = 1$, τότε το αριστερό μέλος γίνεται $1 \cdot e^1 = e$, επαληθεύοντας απόλυτα την ταυτότητα. 
  Επειδή αποδείξαμε στο Ερώτημα 2 ότι η $f$ είναι γνησίως αύξουσα (οπότε και $1-1$), η ρίζα αυτή είναι και \textbf{μοναδική}.
  Επομένως $\mathbf{f(e) = 1}$.

\item Η εξίσωση εφαπτομένης της $C_f$ στο $x_0 = e$:
Χρειαζόμαστε το σημείο επαφής (υπολογίστηκε στο τρίτο ερώτημα):
\[ y_0 = f(e) = 1 \Rightarrow M(e, 1) \]
Υπολογίζουμε την κλίση $f'(e)$ αντικαθιστώντας στον τύπο της παραγώγου από το πρώτο ερώτημα:
\[ f'(e) = \frac{1}{e^{f(e)} (1 + f(e))} = \frac{1}{e^1 (1 + 1)} = \frac{1}{2e} \]
Η εξίσωση εφαπτομένης δίνεται από δομή: $y - f(x_0) = f'(x_0)(x - x_0)$
\[ y - 1 = \frac{1}{2e}(x - e) \Rightarrow y = \frac{1}{2e} x - \frac{e}{2e} + 1 \Rightarrow y = \frac{1}{2e} x - \frac{1}{2} + 1 \]
Συνοψίζοντας, η εφαπτομένη είναι η ευθεία: $\mathbf{y = \frac{1}{2e} x + \frac{1}{2}}$.

\end{erwthma}
\end{thema}
