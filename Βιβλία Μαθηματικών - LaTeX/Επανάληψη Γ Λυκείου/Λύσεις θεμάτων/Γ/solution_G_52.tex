\begin{thema}{Γ}
\begin{erwthma}
\item Επειδή η $f$ είναι $\textbf{γνησίως αύξουσα}$ στο $\mathbb{R}$, είναι $\textbf{έπεται}$ ότι η $f$ είναι $\textbf{ένα-προς-ένα}$ (injective). Κάθε γνησίως μονοτονική συνάρτηση είναι $\eng{bijective}$ ως προς το σύνολο τιμών της, επομένως η $f$ αντιστρέφεται.
Από τα δεδομένα της άσκησης έχουμε:
\[ f(1) = 2 \Rightarrow f^{-1}(2) = 1 \]
\[ f(2) = 3 \Rightarrow f^{-1}(3) = 2 \]

\item Για να λύσουμε την εξίσωση $f^{-1}\big(f(x^2)-1\big) = 1$, εφαρμόζουμε τη συνάρτηση $f$ και στα δύο μέλη:
\begin{align*}
f^{-1}\big(f(x^2)-1\big) &= 1 \\
f\big(f^{-1}(f(x^2)-1)\big) &= f(1) \\
f(x^2) - 1 &= 2 \\
f(x^2) &= 3
\end{align*}
Επειδή η $f$ είναι $\textbf{γνησίως αύξουσα}$ (και άρα αντιστρέψιμη), η εξίσωση $f(x^2) = 3$ είναι ισοδύναμη με:
\begin{align*}
x^2 &= f^{-1}(3) \\
x^2 &= 2 \\
x &= \pm\sqrt{2}
\end{align*}
Άρα, το σύνολο λύσεων είναι $\{\pm\sqrt{2}\}$.

\item Λύνουμε την ανίσωση $f\big(f^{-1}(2x) - 1\big) > 3$:
Επειδή η $f$ είναι $\textbf{γνησίως αύξουσα}$, η ανίσωση είναι ισοδύναμη με:
\begin{align*}
f^{-1}(2x) - 1 &> f^{-1}(3) \\
f^{-1}(2x) - 1 &> 2 \\
f^{-1}(2x) &> 3
\end{align*}
Εφόσον η $f$ είναι $\textbf{γνησίως αύξουσα}$, τότε και η αντίστροφός της $f^{-1}$ είναι $\textbf{γνησίως αύξουσα}$. Επομένως:
\begin{align*}
2x &> f(3) \\
x &> \frac{f(3)}{2}
\end{align*}

\[ \int_{f(a)}^{f(b)} f^{-1}(y) \d y = \left[ x \cdot f(x) \right]_a^b - \int_a^b f(x) \d x \]
