\begin{thema}{Γ}
\begin{erwthma}
\item Επειδή η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, είναι $1-1$.
Άρα αντιστρέφεται στο σύνολο τιμών της.

Από τα δεδομένα έχουμε:
\[f(1)=2\Rightarrow f^{-1}(2)=1\]
και
\[f(2)=3\Rightarrow f^{-1}(3)=2\]

\item Έχουμε
\begin{align*}
f^{-1}(f(x^2)-1)=1
&\Rightarrow f(x^2)-1=f(1)\\
&\Rightarrow f(x^2)-1=2\\
&\Rightarrow f(x^2)=3\\
&\Rightarrow x^2=f^{-1}(3)\\
&\Rightarrow x^2=2\\
&\Rightarrow x=\pm\sqrt{2}
\end{align*}
Επομένως οι λύσεις της εξίσωσης είναι
\[x=\pm\sqrt{2}\]

\item Για να ορίζεται η παράσταση $f^{-1}(2x)$, πρέπει $2x\in f(\mathbb{R})$.
Για αυτές τις τιμές του $x$, επειδή η $f$ είναι γνησίως αύξουσα, έχουμε:
\begin{align*}
f(f^{-1}(2x)-1)>3
&\Leftrightarrow f(f^{-1}(2x)-1)>f(2)\\
&\Leftrightarrow f^{-1}(2x)-1>2\\
&\Leftrightarrow f^{-1}(2x)>3
\end{align*}
Επειδή και η $f^{-1}$ είναι γνησίως αύξουσα, προκύπτει
\[f^{-1}(2x)>3\Leftrightarrow 2x>f(3)\Leftrightarrow x>\frac{f(3)}{2}\]
Άρα οι λύσεις είναι οι τιμές $x$ για τις οποίες
\[2x\in f(\mathbb{R})\quad\text{και}\quad x>\frac{f(3)}{2}\]
Αν επιπλέον θεωρήσουμε ότι $f(\mathbb{R})=\mathbb{R}$, τότε το σύνολο λύσεων είναι
\[\left(\frac{f(3)}{2},+\infty\right)\]

\item Αφού η γραφική παράσταση της $f$ διέρχεται από το σημείο $(0,0)$, έχουμε
\[f(0)=0\]
Επίσης, από τα δεδομένα,
\[f(2)=3\]
Χρησιμοποιούμε τη γνωστή σχέση για ολοκλήρωμα αντίστροφης συνάρτησης:
\[\dint_{f(a)}^{f(b)}f^{-1}(y)\d y=\left[xf(x)\right]_a^b-\dint_a^b f(x)\d x\]
Εδώ παίρνουμε $a=0$ και $b=2$, αφού
\[f(0)=0\quad\text{και}\quad f(2)=3\]
Άρα
\begin{align*}
\dint_0^3 f^{-1}(y)\d y
&=\left[xf(x)\right]_0^2-\dint_0^2 f(x)\d x\\
&=2f(2)-0\cdot f(0)-4\\
&=2\cdot 3-4\\
&=2
\end{align*}
Επομένως
\[\dint_0^3 f^{-1}(y)\d y=2\]
\end{erwthma}
\end{thema}
