%=========== ΛΥΣΗ - 72ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση έχουμε
\[
f'(x)e^{f(x)}=2x
\]
Όμως
\[
\left(e^{f(x)}\right)'=f'(x)e^{f(x)}
\]
Άρα
\[
\left(e^{f(x)}\right)'=2x=(x^2)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
e^{f(x)}=x^2+c
\]
Για $x=0$ έχουμε $f(0)=0$, οπότε
\[
e^{f(0)}=0^2+c\Leftrightarrow e^0=c\Leftrightarrow c=1
\]
Άρα
\[
e^{f(x)}=x^2+1
\]
και, επειδή $x^2+1>0$ για κάθε $x\in\mathbb{R}$,
\[
f(x)=\ln(x^2+1)
\]

\item Από το προηγούμενο ερώτημα
\[
f(x)=\ln(x^2+1)
\]
οπότε
\[
f'(x)=\frac{2x}{x^2+1}
\]
και
\begin{align*}
f''(x)
&=\left(\frac{2x}{x^2+1}\right)'\\
&=\frac{2(x^2+1)-2x\cdot 2x}{(x^2+1)^2}\\
&=\frac{2-2x^2}{(x^2+1)^2}\\
&=\frac{2(1-x^2)}{(x^2+1)^2}
\end{align*}
Επειδή $(x^2+1)^2>0$ για κάθε $x\in\mathbb{R}$, το πρόσημο της $f''$ καθορίζεται από το $1-x^2$.
Έχουμε:
\begin{itemize}
\item $f''(x)=0\Leftrightarrow x=-1$ ή $x=1$
\item $f''(x)>0\Leftrightarrow -1<x<1$
\item $f''(x)<0\Leftrightarrow x<-1$ ή $x>1$
\end{itemize}
Άρα η $f$ είναι κυρτή στο $[-1,1]$ και κοίλη στα διαστήματα $(-\infty,-1]$ και $[1,+\infty)$.

Επειδή η $f''$ αλλάζει πρόσημο στα $x=-1$ και $x=1$, η $C_f$ έχει σημεία καμπής:
\[
K_1(-1,\ln 2)
\quad \text{και}\quad
K_2(1,\ln 2)
\]

\item Αποδεικνύουμε την ανισότητα για $x>0$.
Θεωρούμε τη συνάρτηση
\[
g(x)=x-\ln(x^2+1)\ ,\quad x\geq 0
\]
Έχουμε
\begin{align*}
g'(x)
&=1-\frac{2x}{x^2+1}\\
&=\frac{x^2+1-2x}{x^2+1}\\
&=\frac{(x-1)^2}{x^2+1}
\end{align*}
Άρα
\[
g'(x)\geq 0\ ,\quad \text{για κάθε }x\geq 0
\]
και η $g$ είναι γνησίως αύξουσα στο $[0,+\infty)$.
Επειδή
\[
g(0)=0-\ln 1=0
\]
για κάθε $x>0$ έχουμε
\[
g(x)>g(0)=0
\]
δηλαδή
\[
x-\ln(x^2+1)>0
\]
Άρα
\[
\ln(x^2+1)<x\ ,\quad x>0
\]

\item Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $0$ είναι
\[
y-f(0)=f'(0)(x-0)
\]
Επειδή
\[
f(0)=0
\quad \text{και}\quad
f'(0)=0
\]
η εφαπτομένη είναι
\[
y=0
\]
Για $x\in[0,1]$ έχουμε
\[
f(x)=\ln(x^2+1)\geq 0
\]
Άρα το ζητούμενο εμβαδόν είναι
\[
E=\dint_0^1\ln(x^2+1)\d x
\]
Ολοκληρώνουμε κατά παράγοντες με
\[
u=\ln(x^2+1)\quad \text{και}\quad \d v=\d x
\]
οπότε
\[
\d u=\frac{2x}{x^2+1}\d x
\quad \text{και}\quad
v=x
\]
Έτσι
\begin{align*}
E
&=\left[x\ln(x^2+1)\right]_0^1-\dint_0^1 x\cdot\frac{2x}{x^2+1}\d x\\
&=\ln 2-2\dint_0^1\frac{x^2}{x^2+1}\d x\\
&=\ln 2-2\dint_0^1\left(1-\frac{1}{x^2+1}\right)\d x\\
&=\ln 2-2\left[x-\arctan x\right]_0^1\\
&=\ln 2-2\left(1-\frac{\pi}{4}\right)\\
&=\ln 2-2+\frac{\pi}{2}
\end{align*}
Επομένως
\[
E=\ln 2+\frac{\pi}{2}-2
\]
\end{erwthma}
\end{thema}
