%=========== ΛΥΣΗ - 78ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Από τη δοσμένη σχέση
\[
(x^2+1)f'(x)+2xf(x)=2x
\]
παρατηρούμε ότι
\[
\left((x^2+1)f(x)\right)'=(x^2+1)f'(x)+2xf(x)
\]
Άρα
\[
\left((x^2+1)f(x)\right)'=2x=(x^2)'
\]
Επομένως, σύμφωνα με τις συνέπειες του Θεωρήματος Μέσης Τιμής, υπάρχει σταθερά $c\in\mathbb{R}$ τέτοια ώστε
\[
(x^2+1)f(x)=x^2+c
\]
Για $x=0$ έχουμε $f(0)=0$, οπότε
\[
(0^2+1)f(0)=0^2+c\Leftrightarrow c=0
\]
Άρα
\[
(x^2+1)f(x)=x^2
\]
και τελικά
\[
f(x)=\frac{x^2}{x^2+1}\ ,\quad x\in\mathbb{R}
\]

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=\frac{2x(x^2+1)-x^2\cdot 2x}{(x^2+1)^2}
=\frac{2x}{(x^2+1)^2}
\]
Επειδή
\[
(x^2+1)^2>0
\]
το πρόσημο της $f'$ καθορίζεται από το $x$.
Άρα
\begin{itemize}
\item $f'(x)<0$ για $x<0$
\item $f'(0)=0$
\item $f'(x)>0$ για $x>0$
\end{itemize}
Συνεπώς η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$.
Επομένως παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
f(0)=0
\]

Για τις ασύμπτωτες έχουμε
\[
\lim\limits_{x\to+\infty}f(x)
=\lim\limits_{x\to+\infty}\frac{x^2}{x^2+1}=1
\]
και
\[
\lim\limits_{x\to-\infty}f(x)
=\lim\limits_{x\to-\infty}\frac{x^2}{x^2+1}=1
\]
Άρα η ευθεία
\[
y=1
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$ και στο $-\infty$.
Η $f$ δεν έχει κατακόρυφες ασύμπτωτες, αφού είναι ορισμένη και συνεχής σε όλο το $\mathbb{R}$.

\item Για $x>0$ έχουμε
\begin{align*}
x-f(x)
&=x-\frac{x^2}{x^2+1}\\
&=\frac{x(x^2+1)-x^2}{x^2+1}\\
&=\frac{x(x^2-x+1)}{x^2+1}
\end{align*}
Επειδή $x>0$, $x^2+1>0$ και
\[
x^2-x+1=\left(x-\frac{1}{2}\right)^2+\frac{3}{4}>0
\]
παίρνουμε
\[
x-f(x)>0
\]
Άρα
\[
f(x)<x\ ,\quad x>0
\]
δηλαδή η ευθεία $y=x$ βρίσκεται πάνω από την $C_f$ για κάθε $x>0$.

\item Έστω $k\in(0,1)$.
Λύνουμε την εξίσωση
\[
f(x)=k
\]
Έχουμε
\begin{align*}
\frac{x^2}{x^2+1}&=k\\
x^2&=k(x^2+1)\\
x^2-kx^2&=k\\
(1-k)x^2&=k\\
x^2&=\frac{k}{1-k}
\end{align*}
Επειδή $k\in(0,1)$, είναι
\[
\frac{k}{1-k}>0
\]
Άρα η εξίσωση έχει ακριβώς δύο πραγματικές ρίζες:
\[
x_1=\sqrt{\frac{k}{1-k}}
\quad \text{και}\quad
x_2=-\sqrt{\frac{k}{1-k}}
\]
\end{erwthma}
\end{thema}
