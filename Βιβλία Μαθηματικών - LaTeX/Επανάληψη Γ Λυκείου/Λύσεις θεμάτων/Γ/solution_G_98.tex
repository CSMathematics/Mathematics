\begin{erwthma}
\item Από το διάγραμμα της $f'$ παρατηρούμε ότι
\[
f'(x)>0,\quad x\in[0,2),
\]
και
\[
f'(x)<0,\quad x\in(2,6).
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $[0,2]$ και γνησίως φθίνουσα στο $[2,6]$.

Για την κυρτότητα εξετάζουμε τη μονοτονία της $f'$. Από το σχήμα, η $f'$ είναι γνησίως φθίνουσα στο $[0,4]$ και γνησίως αύξουσα στο $[4,6]$.

Άρα η $f$ είναι κοίλη στο $[0,4]$ και κυρτή στο $[4,6]$. Στο $x=4$ η $f'$ αλλάζει μονοτονία, επομένως η $C_f$ παρουσιάζει σημείο καμπής στο
\[
\bigl(4,f(4)\bigr).
\]

\item Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού έχουμε
\[
f(2)-f(0)=\int_0^2 f'(x)\,\d x.
\]
Το ολοκλήρωμα ισούται με το εμβαδόν ορθογωνίου τριγώνου με βάση $2$ και ύψος $2$, άρα
\[
\int_0^2 f'(x)\,\d x=\frac{1}{2}\cdot 2\cdot 2=2.
\]
Επειδή $f(0)=0$, παίρνουμε
\[
f(2)=2.
\]

Επίσης,
\[
f(4)-f(2)=\int_2^4 f'(x)\,\d x.
\]
Στο $[2,4]$ η $C_{f'}$ βρίσκεται κάτω από τον άξονα και σχηματίζει τρίγωνο με βάση $2$ και ύψος $2$. Άρα
\[
\int_2^4 f'(x)\,\d x=-\frac{1}{2}\cdot 2\cdot 2=-2.
\]
Επομένως,
\[
f(4)=f(2)-2=0.
\]

Τέλος,
\[
f(6)-f(4)=\int_4^6 f'(x)\,\d x.
\]
Στο $[4,6]$ το αντίστοιχο προσημασμένο εμβαδόν είναι
\[
\int_4^6 f'(x)\,\d x=-\frac{1}{2}\cdot 2\cdot 2=-2.
\]
Άρα
\[
f(6)=f(4)-2=-2.
\]

Η $f$ είναι αύξουσα στο $[0,2]$ και φθίνουσα στο $[2,6]$, άρα παρουσιάζει ολικό μέγιστο στο $x=2$, με τιμή
\[
f(2)=2.
\]
Συγκρίνοντας τις τιμές στα άκρα και στο σημείο αλλαγής μονοτονίας,
\[
f(0)=0,\quad f(2)=2,\quad f(6)=-2,
\]
προκύπτει ότι η $f$ παρουσιάζει ολικό ελάχιστο στο $x=6$, με τιμή
\[
f(6)=-2.
\]

\item Από το σχήμα, για $0\leq x\leq 4$ η $f'$ είναι η ευθεία που περνά από τα σημεία $(0,2)$ και $(4,-2)$. Άρα
\[
f'(x)=2-x,\quad 0\leq x\leq 4.
\]
Επειδή $f(0)=0$, για $0\leq x\leq 4$ έχουμε
\[
f(x)=\int_0^x (2-t)\,\d t
=2x-\frac{x^2}{2}.
\]
Δηλαδή
\[
f(x)=-\frac{1}{2}x^2+2x,\quad 0\leq x\leq 4.
\]

Για $4<x\leq 6$, η $f'$ είναι η ευθεία που περνά από τα σημεία $(4,-2)$ και $(6,0)$, άρα
\[
f'(x)=x-6,\quad 4<x\leq 6.
\]
Επειδή $f(4)=0$, για $4<x\leq 6$ έχουμε
\[
f(x)=f(4)+\int_4^x (t-6)\,\d t
=\left[\frac{t^2}{2}-6t\right]_4^x.
\]
Άρα
\[
f(x)=\frac{x^2}{2}-6x-\left(8-24\right)
=\frac{x^2}{2}-6x+16.
\]

Επομένως,
\[
f(x)=
\begin{cases}
-\dfrac{1}{2}x^2+2x, & 0\leq x\leq 4,\\[2ex]
\dfrac{1}{2}x^2-6x+16, & 4<x\leq 6.
\end{cases}
\]

\item Από τον τύπο της $f$ έχουμε
\[
f(x)=-\frac{1}{2}x^2+2x=\frac{x(4-x)}{2},\quad 0\leq x\leq 4.
\]
Άρα η $f$ είναι θετική στο $(0,4)$ και μηδενίζεται στα $x=0$, $x=4$.

Για $4<x\leq 6$ έχουμε
\[
f(x)=\frac{1}{2}x^2-6x+16=\frac{1}{2}(x-4)(x-8).
\]
Στο διάστημα $(4,6]$ ισχύει $f(x)<0$.

Επομένως, το ζητούμενο εμβαδόν είναι
\[
E=\int_0^4 f(x)\,\d x-\int_4^6 f(x)\,\d x.
\]
Υπολογίζουμε
\[
\int_0^4\left(-\frac{1}{2}x^2+2x\right)\,\d x
=\left[-\frac{x^3}{6}+x^2\right]_0^4
=-\frac{64}{6}+16
=\frac{16}{3}.
\]
Επίσης,
\[
\int_4^6\left(\frac{1}{2}x^2-6x+16\right)\,\d x
=\left[\frac{x^3}{6}-3x^2+16x\right]_4^6
=-\frac{8}{3}.
\]
Άρα
\[
E=\frac{16}{3}-\left(-\frac{8}{3}\right)=8.
\]

Για το όριο, κοντά στο $x=2$ ισχύει
\[
f(x)=-\frac{1}{2}x^2+2x
\]
και
\[
f'(x)=2-x.
\]
Επομένως
\[
f(x)-2=-\frac{1}{2}x^2+2x-2
=-\frac{1}{2}(x-2)^2.
\]
Άρα, για $x\neq 2$ κοντά στο $2$,
\[
\frac{f(x)-2}{(x-2)f'(x)}
=\frac{-\frac{1}{2}(x-2)^2}{(x-2)(2-x)}
=\frac{-\frac{1}{2}(x-2)^2}{-(x-2)^2}
=\frac{1}{2}.
\]
Συνεπώς,
\[
\lim_{x\to 2}\frac{f(x)-2}{(x-2)f'(x)}=\frac{1}{2}.
\]
\end{erwthma}
\end{thema}
