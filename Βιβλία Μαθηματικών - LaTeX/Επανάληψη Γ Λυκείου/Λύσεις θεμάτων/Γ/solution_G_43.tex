\begin{thema}{Γ}
\begin{erwthma}
\item Θέτουμε
\[I=\dint_0^1 f(t)\d t\]
Το $I$ είναι πραγματικός αριθμός, αφού η $f$ είναι συνεχής στο $[0,1]$. Η δοσμένη σχέση γράφεται
\[f(x)=e^x-xI\ ,\ x\in\mathbb{R}\]
Ολοκληρώνοντας στο διάστημα $[0,1]$, έχουμε:
\begin{align*}
\dint_0^1 f(x)\d x&=\dint_0^1(e^x-xI)\d x\\
I&=\left[e^x\right]_0^1-I\left[\frac{x^2}{2}\right]_0^1\\
I&=e-1-\frac{I}{2}\\
\frac{3I}{2}&=e-1\\
I&=\frac{2(e-1)}{3}
\end{align*}
Άρα, για κάθε $x\in\mathbb{R}$ είναι
\[f(x)=e^x-\frac{2(e-1)}{3}x\]

\item Από το προηγούμενο ερώτημα έχουμε
\[f'(x)=e^x-\frac{2(e-1)}{3}\ ,\ x\in\mathbb{R}\]
Βρίσκουμε ρίζες και πρόσημα της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow e^x=\dfrac{2(e-1)}{3}\Leftrightarrow x=\ln\left(\dfrac{2(e-1)}{3}\right)$
\item $f'(x)>0\Leftrightarrow e^x>\dfrac{2(e-1)}{3}\Leftrightarrow x>\ln\left(\dfrac{2(e-1)}{3}\right)$
\item $f'(x)<0\Leftrightarrow e^x<\dfrac{2(e-1)}{3}\Leftrightarrow x<\ln\left(\dfrac{2(e-1)}{3}\right)$
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}

\end{center}
Επομένως η $f$ είναι \textbf{γνησίως φθίνουσα} στο
\[\left(-\infty,\ln\left(\frac{2(e-1)}{3}\right)\right]\]
και \textbf{γνησίως αύξουσα} στο
\[\left[\ln\left(\frac{2(e-1)}{3}\right),+\infty\right)\]
Άρα η $f$ παρουσιάζει ολικό ελάχιστο για
\[x_0=\ln\left(\frac{2(e-1)}{3}\right)\]
με τιμή
\[f(x_0)=\frac{2(e-1)}{3}-\frac{2(e-1)}{3}\ln\left(\frac{2(e-1)}{3}\right)\]

\item Θεωρούμε τη συνάρτηση
\[g(x)=f(x)-1=e^x-\frac{2(e-1)}{3}x-1\ ,\ x\in[0,1]\]
Η $g$ είναι συνεχής στο $[0,1]$ και παραγωγίσιμη στο $(0,1)$, ως πράξεις συνεχών και παραγωγίσιμων συναρτήσεων.
Επιπλέον,
\[g'(x)=e^x-\frac{2(e-1)}{3}=f'(x)\]
Από το προηγούμενο ερώτημα η $g$ είναι γνησίως φθίνουσα στο
\[\left[0,\ln\left(\frac{2(e-1)}{3}\right)\right]\]
και γνησίως αύξουσα στο
\[\left[\ln\left(\frac{2(e-1)}{3}\right),1\right]\]
αφού
\[e>\frac{5}{2}\Rightarrow 2(e-1)>3\Rightarrow \frac{2(e-1)}{3}>1\]
και
\[\frac{2(e-1)}{3}<e\]
Επομένως
\[1<\frac{2(e-1)}{3}<e \Rightarrow 0<\ln\left(\frac{2(e-1)}{3}\right)<1\]
Έχουμε
\[g(0)=f(0)-1=1-1=0\]
Επειδή η $g$ είναι γνησίως φθίνουσα στο
\[\left[0,\ln\left(\frac{2(e-1)}{3}\right)\right]\]
προκύπτει ότι
\[g(x)<g(0)=0\ ,\ 0<x\leq \ln\left(\frac{2(e-1)}{3}\right)\]
Άρα η εξίσωση $g(x)=0$ δεν έχει ρίζα στο
\[\left(0,\ln\left(\frac{2(e-1)}{3}\right)\right]\]
Ακόμη,
\[g(1)=e-\frac{2(e-1)}{3}-1=\frac{e-1}{3}>0\]
Επειδή
\[g\left(\ln\left(\frac{2(e-1)}{3}\right)\right)<0<g(1)\]
και η $g$ είναι συνεχής στο
\[\left[\ln\left(\frac{2(e-1)}{3}\right),1\right]\]
από το θεώρημα \eng{Bolzano} υπάρχει τουλάχιστον μία ρίζα της εξίσωσης $g(x)=0$ στο
\[\left(\ln\left(\frac{2(e-1)}{3}\right),1\right)\subseteq(0,1)\]
Η ρίζα αυτή είναι μοναδική, αφού η $g$ είναι γνησίως αύξουσα στο
\[\left[\ln\left(\frac{2(e-1)}{3}\right),1\right]\]
Επομένως η εξίσωση $f(x)=1$ έχει ακριβώς μία ρίζα στο $(0,1)$.

\item Η ευθεία
\[\varepsilon:(1-e)x+3y-2026=0\]
γράφεται
\[3y=(e-1)x+2026 \Leftrightarrow y=\frac{e-1}{3}x+\frac{2026}{3}\]
οπότε ο συντελεστής διεύθυνσής της είναι
\[\lambda_{\varepsilon}=\frac{e-1}{3}\]
Θέλουμε να αποδείξουμε ότι υπάρχει $\xi\in(0,1)$ τέτοιο ώστε
\[f'(\xi)=\frac{e-1}{3}\]
Έχουμε
\begin{align*}
f'(\xi)=\frac{e-1}{3}
&\Leftrightarrow e^\xi-\frac{2(e-1)}{3}=\frac{e-1}{3}\\
&\Leftrightarrow e^\xi=e-1\\
&\Leftrightarrow \xi=\ln(e-1)
\end{align*}
Επειδή $1<e-1<e$, είναι
\[0<\ln(e-1)<1\]
άρα $\xi=\ln(e-1)\in(0,1)$.
Για αυτό το $\xi$ ισχύει $f'(\xi)=\lambda_{\varepsilon}$, επομένως η εφαπτομένη της $C_f$ στο σημείο $M(\xi,f(\xi))$ είναι παράλληλη στην ευθεία $\varepsilon$.
\end{erwthma}
\end{thema}
