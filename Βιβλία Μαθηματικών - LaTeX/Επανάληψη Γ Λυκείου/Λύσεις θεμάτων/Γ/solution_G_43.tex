\begin{erwthma}
\item Η συνάρτηση $f(x) = 1 - \dfrac{1}{1 + x^2}$ είναι παραγωγίσιμη στο $(0, 1)$ και η παράγωγός της είναι:
\[ f'(x) = 0 - \left( - \dfrac{1}{(1 + x^2)^2} \cdot 2x \right) = \dfrac{2x}{(1 + x^2)^2} \]
Για κάθε $x \in (0, 1)$, ισχύει $x > 0$ και $(1 + x^2)^2 > 0$, επομένως $f'(x) > 0$.
Συνεπώς, η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο διάστημα $[0, 1]$.

\item Επειδή η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $[0, 1]$, το σύνολο τιμών της είναι το διάστημα $[f(0), f(1)]$.
Υπολογίζουμε τις τιμές στα άκρα:
\[ f(0) = 1 - \dfrac{1}{1 + 0^2} = 1 - 1 = 0 \]
\[ f(1) = 1 - \dfrac{1}{1 + 1^2} = 1 - \dfrac{1}{2} = \dfrac{1}{2} \]
Άρα, το σύνολο τιμών της συνάρτησης στο $[0, 1]$ είναι το $[0, \dfrac{1}{2}]$.

\item Από το προηγούμενο ερώτημα, γνωρίζουμε ότι για κάθε $x \in [0, 1]$ ισχύει:
\[ 0 \le f(x) \le \dfrac{1}{2} \]
Ολοκληρώνοντας τα μέλη της ανισότητας στο διάστημα $[0, 1]$, έχουμε:
\[ \int_{0}^{1} 0 \d x \le \int_{0}^{1} f(x) \d x \le \int_{0}^{1} \dfrac{1}{2} \d x \]
\[ 0 \le \int_{0}^{1} f(x) \d x \le \left[ \dfrac{1}{2}x \right]_{0}^{1} \]
\[ 0 \le \int_{0}^{1} f(x) \d x \le \dfrac{1}{2} \]

\item Για τη συνάρτηση $f(x)$, ισχύει:
\[ f(\hm x) = 1 - \dfrac{1}{1 + (\hm x)^2} = \dfrac{1 + (\hm x)^2 - 1}{1 + (\hm x)^2} = \dfrac{(\hm x)^2}{1 + (\hm x)^2} \]
Το ζητούμενο όριο είναι:
\begin{align*}
\lim_{x \to 0} \dfrac{f(\hm x)}{x^2} &= \lim_{x \to 0} \dfrac{(\hm x)^2}{x^2(1 + (\hm x)^2)} \\
&= \lim_{x \to 0} \left( \dfrac{\hm x}{x} \right)^2 \cdot \dfrac{1}{1 + (\hm x)^2}
\end{align*}
Γνωρίζουμε ότι $\lim\limits_{x \to 0} \dfrac{\hm x}{x} = 1$, επομένως:
\[ \lim_{x \to 0} \dfrac{f(\hm x)}{x^2} = 1^2 \cdot \dfrac{1}{1 + 0^2} = 1 \cdot 1 = 1 \]
\end{erwthma}
\end{thema}
