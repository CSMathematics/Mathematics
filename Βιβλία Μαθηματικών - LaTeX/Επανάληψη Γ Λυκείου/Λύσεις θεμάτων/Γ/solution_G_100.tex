\begin{thema}{Γ}

\begin{center}

  \addplot[very thick, maincolor, domain=-1:1] {-0.75*x^2 + 1.5*x + 2.25};
  \addplot[very thick, maincolor, domain=1:4] {(2/27)*(x-1)^3 - (2/3)*(x-1)^2 + 3}
    node[pos=0.8, above right] {$C_1$};

  \addplot[very thick, secondcolor, domain=-1:1] {-1.5*x + 1.5};
  \addplot[very thick, secondcolor, domain=1:4] {(2/9)*(x-1)^2 - (4/3)*(x-1)}

  \addplot[mark=*, maincolor, only marks, mark size=1.5pt] coordinates {(-1,0) (1,3) (4,-1)};

  \node[secondcolor, right, font=\footnotesize] at (axis cs:-1,3) {$(-1,3)$};

\begin{erwthma}
\item Η καμπύλη $C_1$ περνά από τα σημεία
\[
(-1,0),\quad (1,3),\quad (4,-1),
\]
ενώ η $C_2$ περνά από τα σημεία
\[
(-1,3),\quad (1,0),\quad (4,-2).
\]

Παρατηρούμε ότι η $C_1$ είναι γνησίως αύξουσα μέχρι το $x=1$ και γνησίως φθίνουσα μετά το $x=1$. Άρα η παράγωγός της πρέπει να είναι θετική πριν από το $1$, μηδενική στο $1$ και αρνητική μετά το $1$.

Αυτό ακριβώς δείχνει η $C_2$. Επομένως,
\[
C_1=C_f
\quad \text{και}\quad
C_2=C_{f'}.
\]

Από το πρόσημο της $f'$ έχουμε:
\[
f'(x)>0,\quad x\in[-1,1),
\]
και
\[
f'(x)<0,\quad x\in(1,4].
\]
Άρα η $f$ είναι γνησίως αύξουσα στο $[-1,1]$ και γνησίως φθίνουσα στο $[1,4]$.

Συνεπώς, η $f$ παρουσιάζει ολικό μέγιστο στο $x=1$, με τιμή
\[
f(1)=3.
\]

Συγκρίνοντας τις τιμές στα άκρα,
\[
f(-1)=0
\quad \text{και}\quad
f(4)=-1,
\]
προκύπτει ότι η $f$ παρουσιάζει ολικό ελάχιστο στο $x=4$, με τιμή
\[
f(4)=-1.
\]

Με τη σχετική έννοια στο πεδίο ορισμού, το $x=-1$ είναι τοπικό ελάχιστο.

\item Η κυρτότητα της $f$ προκύπτει από τη μονοτονία της $f'$.

Από το σχήμα, η $C_{f'}$, δηλαδή η $C_2$, είναι γνησίως φθίνουσα στο $[-1,4]$. Άρα η $f'$ είναι γνησίως φθίνουσα στο $[-1,4]$.

Επομένως, η $f$ είναι κοίλη στο $[-1,4]$.

Για την εξίσωση $f(x)=0$ στο $(1,4)$, έχουμε
\[
f(1)=3>0
\]
και
\[
f(4)=-1<0.
\]
Η $f$ είναι συνεχής στο $[1,4]$, άρα από το Θεώρημα \eng{Bolzano} υπάρχει τουλάχιστον μία ρίζα στο $(1,4)$.

Επειδή όμως η $f$ είναι γνησίως φθίνουσα στο $[1,4]$, η ρίζα αυτή είναι μοναδική.

Άρα η εξίσωση
\[
f(x)=0
\]
έχει ακριβώς μία ρίζα στο $(1,4)$.

\item Στο διάστημα $[-1,1]$ η $f$ είναι κοίλη. Άρα η γραφική της παράσταση βρίσκεται πάνω από τη χορδή που ενώνει τα σημεία
\[
(-1,f(-1))=(-1,0)
\]
και
\[
(1,f(1))=(1,3).
\]

Το εμβαδόν κάτω από τη χορδή αυτή είναι το εμβαδόν τραπεζίου με βάσεις $0$, $3$ και ύψος $2$:
\[
E=\frac{0+3}{2}\cdot 2=3.
\]

Επειδή η $C_f$ δεν ταυτίζεται με τη χορδή στο εσωτερικό του διαστήματος, έχουμε
\[
\int_{-1}^1 f(x)\,\d x>3.
\]

Το ζητούμενο εμβαδόν που περικλείεται από την $C_2$, τον άξονα $x'x$ και τις ευθείες $x=-1$, $x=4$ είναι το εμβαδόν ανάμεσα στην $C_{f'}$ και τον άξονα.

Στο $[-1,1]$ η $f'$ είναι θετική, οπότε
\[
\int_{-1}^1 f'(x)\,\d x=f(1)-f(-1)=3-0=3.
\]

Στο $[1,4]$ η $f'$ είναι αρνητική, οπότε το αντίστοιχο εμβαδόν είναι
\[
-\int_1^4 f'(x)\,\d x
=-\bigl(f(4)-f(1)\bigr)
=-\bigl(-1-3\bigr)=4.
\]

Άρα το συνολικό εμβαδόν είναι
\[
3+4=7.
\]

\item Με ολοκλήρωση κατά μέρη έχουμε
\[
\int_1^4 x f''(x)\,\d x
=\bigl[x f'(x)\bigr]_1^4-\int_1^4 f'(x)\,\d x.
\]

Από το σχήμα της $f'$ διαβάζουμε
\[
f'(1)=0
\quad \text{και}\quad
f'(4)=-2.
\]
Άρα
\[
\bigl[x f'(x)\bigr]_1^4
=4f'(4)-1f'(1)
=4\cdot(-2)-0=-8.
\]

Επίσης,
\[
\int_1^4 f'(x)\,\d x=f(4)-f(1)=-1-3=-4.
\]

Επομένως,
\[
\int_1^4 x f''(x)\,\d x
=-8-(-4)=-4.
\]
\end{erwthma}
\end{thema}
