%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f(x)=\ln x$ είναι συνεχής στο $[x,x+1]$ και παραγωγίσιμη στο $(x,x+1)$ (ως γνωστός τύπος για $x>0$). Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο διάστημα $[x,x+1]$: υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε
\[
\frac{f(x+1)-f(x)}{(x+1)-x}=f'(\xi)=\frac{1}{\xi}.
\]
Επομένως:
\[
\ln(x+1)-\ln x=\frac{1}{\xi}\]

\item Παρατηρούμε ότι $\ln\left(1+\dfrac{1}{x}\right)=\ln\dfrac{x+1}{x}=\ln(x+1)-\ln x$. Από το Γ1, υπάρχει $\xi\in(x,x+1)$ τέτοιο ώστε:
\[
\ln\left(1+\frac{1}{x}\right)=\ln(x+1)-\ln x=\frac{1}{\xi}
\]
Επειδή $x<\xi<x+1$ και όλοι οι αριθμοί είναι θετικοί, αντιστρέφεται η φορά της ανισότητας:
\[
\frac{1}{x+1}<\frac{1}{\xi}<\frac{1}{x}
\]
Αντικαθιστώντας $\dfrac{1}{\xi}=\ln\left(1+\dfrac{1}{x}\right)$, προκύπτει:
\[
\frac{1}{x+1}<\ln\left(1+\frac{1}{x}\right)<\frac{1}{x}
\]

\item Επειδή $x>0$, πολλαπλασιάζουμε και τα τρία μέλη της ανισότητας του Γ2 με $x$ χωρίς αλλαγή φοράς:
\[
\frac{x}{x+1}<x\ln\left(1+\frac{1}{x}\right)<1
\]
Επειδή $\lim\limits_{x\to +\infty}\dfrac{x}{x+1}=1$ και $\lim\limits_{x\to +\infty}1=1$, από το Θεώρημα Παρεμβολής:
\[
\lim_{x\to +\infty}\left[x\ln\left(1+\frac{1}{x}\right)\right]=1
\]

\item Από το Γ2, για κάθε $x\in[1,e]$ ισχύει:
\[
\frac{1}{x+1}<\ln\left(1+\frac{1}{x}\right)<\frac{1}{x}
\]
Εφόσον η ανισότητα ισχύει παντού στο $[1,e]$, από τη μονοτονία του ολοκληρώματος ολοκληρώνουμε και τα τρία μέλη στο $[1,e]$:
\[
\int_1^e \frac{1}{x+1}\,\d x
<
\int_1^e \ln\left(1+\frac{1}{x}\right)\,\d x
<
\int_1^e \frac{1}{x}\,\d x.
\]
Επειδή
\[
f\left(1+\frac{1}{x}\right)=\ln\left(1+\frac{1}{x}\right),
\]
παίρνουμε
\[
\int_1^e \frac{1}{x+1}\,\d x
<
\int_1^e f\left(1+\frac{1}{x}\right)\,\d x
<
\int_1^e \frac{1}{x}\,\d x.
\]
Υπολογίζουμε τα ολοκληρώματα στα άκρα της ανισότητας:
\[
\int_1^e \frac{1}{x+1}\,\d x
=
\left[\ln(x+1)\right]_1^e
=
\ln\frac{e+1}{2}
\]
και
\[
\int_1^e \frac{1}{x}\,\d x
=
\left[\ln x\right]_1^e
=1.
\]
Άρα η ανισότητα γράφεται
\[
\ln\frac{e+1}{2}
<
\int_1^e f\left(1+\frac{1}{x}\right)\,\d x
<
1.
\]
\end{erwthma}
\end{thema}
