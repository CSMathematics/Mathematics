\begin{thema}{Γ}
\begin{erwthma}
\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[f'(x)=3x^2-3a^2=3(x-a)(x+a)\]
Επειδή $a>0$, οι ρίζες της $f'$ είναι
\[x=-a\quad \text{και}\quad x=a\]
Μελετάμε το πρόσημο της $f'$:
\begin{itemize}
\item $f'(x)>0$ για $x\in(-\infty,-a)\cup(a,+\infty)$
\item $f'(x)<0$ για $x\in(-a,a)$
\end{itemize}
Άρα η $f$ είναι γνησίως αύξουσα στο $(-\infty,-a]$, γνησίως φθίνουσα στο $[-a,a]$ και γνησίως αύξουσα στο $[a,+\infty)$.

Επομένως η $f$ παρουσιάζει τοπικό μέγιστο στο $x=-a$ και τοπικό ελάχιστο στο $x=a$.
Οι αντίστοιχες τιμές είναι:
\[
f(-a)=(-a)^3-3a^2(-a)+2a^3=4a^3
\]
και
\[
f(a)=a^3-3a^3+2a^3=0
\]

\item Υπολογίζουμε:
\begin{align*}
\dint_{-1}^{1}f(x)\d x
&=\dint_{-1}^{1}(x^3-3a^2x+2a^3)\d x\\
&=\dint_{-1}^{1}x^3\d x-3a^2\dint_{-1}^{1}x\d x+2a^3\dint_{-1}^{1}1\d x
\end{align*}
Οι συναρτήσεις $x^3$ και $x$ είναι περιττές, άρα
\[
\dint_{-1}^{1}x^3\d x=0
\quad \text{και}\quad
\dint_{-1}^{1}x\d x=0
\]
Επομένως
\[
\dint_{-1}^{1}f(x)\d x=2a^3\cdot 2=4a^3
\]
Για να ισχύει
\[\dint_{-1}^{1}f(x)\d x=0\]
πρέπει
\[4a^3=0\Leftrightarrow a=0\]
Όμως από την εκφώνηση είναι $a>0$.
Άρα δεν υπάρχει τιμή του $a>0$ για την οποία
\[\dint_{-1}^{1}f(x)\d x=0\]

\item Από το πρώτο ερώτημα, το τοπικό μέγιστο της $f$ είναι
\[
M(a)=f(-a)=4a^3\ ,\quad a>0
\]
Η $M$ είναι παραγωγίσιμη στο $(0,+\infty)$ και
\[
M'(a)=12a^2>0\ ,\quad \text{για κάθε }a>0
\]
Άρα η $M$ είναι γνησίως αύξουσα στο $(0,+\infty)$.

Επομένως η $M$ δεν παρουσιάζει ελάχιστη τιμή στο $(0,+\infty)$.
Έχει όμως
\[
\lim\limits_{a\to 0^+}M(a)=\lim\limits_{a\to 0^+}4a^3=0
\]
οπότε το κάτω φράγμα της είναι το $0$, χωρίς να λαμβάνεται για $a>0$.

\item Παραγοντοποιούμε την $f$:
\[
f(x)=x^3-3a^2x+2a^3=(x-a)^2(x+2a)
\]
Άρα οι ρίζες της είναι
\[
x=-2a\quad \text{και}\quad x=a
\]
με το $x=a$ διπλή ρίζα.
Επειδή $(x-a)^2\geq 0$, το πρόσημο της $f$ καθορίζεται από το $x+2a$.
Έτσι
\[
f(x)\geq 0\quad \text{για κάθε }x\in[-2a,a]
\]
Το χωρίο που περικλείεται από τη $C_f$ και τον άξονα $x'x$ είναι το χωρίο πάνω από το διάστημα $[-2a,a]$.
Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E(a)
&=\dint_{-2a}^{a}f(x)\d x\\
&=\dint_{-2a}^{a}(x^3-3a^2x+2a^3)\d x\\
&=\left[\frac{x^4}{4}-\frac{3a^2x^2}{2}+2a^3x\right]_{-2a}^{a}\\
&=\left(\frac{a^4}{4}-\frac{3a^4}{2}+2a^4\right)
-\left(\frac{16a^4}{4}-\frac{3a^2\cdot 4a^2}{2}-4a^4\right)\\
&=\frac{3a^4}{4}-\left(4a^4-6a^4-4a^4\right)\\
&=\frac{3a^4}{4}+6a^4\\
&=\frac{27a^4}{4}
\end{align*}
Επομένως
\[
E(a)=\frac{27a^4}{4}
\]
\end{erwthma}
\end{thema}
