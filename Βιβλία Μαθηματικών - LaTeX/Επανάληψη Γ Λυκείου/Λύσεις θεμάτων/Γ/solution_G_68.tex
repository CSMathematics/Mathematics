%=========== ΛΥΣΗ - {ΘΕΜΑ Γ} ===========
\begin{thema}{Γ}
\begin{erwthma}
\item 
Η παράγωγος της συνάρτησης $f$ είναι:
\[ f'(x) = 3x^2 - 3a^2 = 3(x^2 - a^2) = 3(x - a)(x + a) \]
Οι κρίσιμες τιμές είναι οι ρίζες της $f'(x) = 0$, δηλαδή:
\[ 3(x - a)(x + a) = 0 \Leftrightarrow x = a \text{ ή } x = -a \]
Με βάση το πρόσημο της $f'(x)$ για $a > 0$:
\begin{itemize}
    \item $f'(x) > 0 \Leftrightarrow x \in (-\infty, -a) \cup (a, +\infty)$
    \item $f'(x) < 0 \Leftrightarrow x \in (-a, a)$
\end{itemize}
Άρα, η συνάρτηση $f$ παρουσιάζει \textbf{τοπικό μέγιστο} στο $x = -a$ και \textbf{τοπικό ελάχιστο} στο $x = a$.
Οι τιμές των ακροτάτων είναι:
\[ f(-a) = (-a)^3 - 3a^2(-a) + 2a^3 = -a^3 + 3a^3 + 2a^3 = 4a^3 \]
\[ f(a) = a^3 - 3a^2(a) + 2a^3 = a^3 - 3a^3 + 2a^3 = 0 \]

\item 
Για να είναι το $a$ \textbf{διπλή ρίζα} της $f$, πρέπει να ισχύει $f(a) = 0$ και $f'(a) = 0$, ενώ $f''(a) \neq 0$.
Από το προηγούμενο ερώτημα έχουμε:
\[ f(a) = a^3 - 3a^3 + 2a^3 = 0 \]
\[ f'(a) = 3(a^2 - a^2) = 0 \]
Η δεύτερη παράγωγος είναι $f''(x) = 6x$, οπότε:
\[ f''(a) = 6a \]
Εφόσον $a > 0$, τότε $f''(a) \neq 0$. Επομένως, το $a$ είναι \textbf{διπλή ρίζα} της $f$.

\item 
Το τοπικό μέγιστο της $f$ ως συνάρτηση του $a$ είναι η τιμή $g(a) = f(-a) = 4a^3$ για $a > 0$.
Η συνάρτηση $g(a) = 4a^3$ έχει παράγωγο $g'(a) = 12a^2$.
Επειδή $g'(a) > 0$ για κάθε $a > 0$, η συνάρτηση $g$ είναι \textbf{γνησίως αύξουσα} στο διάστημα $(0, +\infty)$.
Συνεπώς, η συνάρτηση $g(a)$ δεν παρουσιάζει ελάχιστη τιμή στο ανοικτό διάστημα $(0, +\infty)$, καθώς $\lim_{a \to 0^+} g(a) = 0$.

\item 
Οι ρίζες της $f(x)$ είναι το $x = a$ (διπλή) και το $x = -2a$, καθώς:
\[ f(-2a) = (-2a)^3 - 3a^2(-2a) + 2a^3 = -8a^3 + 6a^3 + 2a^3 = 0 \]
Για $x \in [-2a, a]$, η συνάρτηση $f(x)$ είναι μη αρνητική, καθώς $f(x) = (x-a)^2(x+2a)$.
Το εμβαδόν $E$ είναι:
\begin{align*}
E &= \int_{-2a}^{a} (x^3 - 3a^2 x + 2a^3) \d x \\
  &= \left[ \frac{x^4}{4} - \frac{3a^2 x^2}{2} + 2a^3 x \right]_{-2a}^{a} \\
  &= \left( \frac{a^4}{4} - \frac{3a^4}{2} + 2a^4 \right) - \left( \frac{(-2a)^4}{4} - \frac{3a^2 (-2a)^2}{2} + 2a^3 (-2a) \right) \\
  &= \left( \frac{a^4 - 6a^4 + 8a^4}{4} \right) - \left( \frac{16a^4}{4} - \frac{12a^4}{2} - 4a^4 \right) \\
  &= \frac{3}{4}a^4 - (4a^4 - 6a^4 - 4a^4) \\
  &= \frac{3}{4}a^4 - (-6a^4) \\
  &= \frac{3a^4 + 24a^4}{4} = \frac{27}{4}a^4
\end{align*}
\end{erwthma}
\end{thema}
