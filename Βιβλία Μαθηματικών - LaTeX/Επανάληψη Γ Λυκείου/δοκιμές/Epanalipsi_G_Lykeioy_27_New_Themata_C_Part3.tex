\end{erwthma}
\end{askhsh}

\begin{askhsh}{Γ}

  \item Να μελετήσετε την $f(x)$ ως προς τη μονοτονία και να εντοπίσετε τα τοπικά ακρότατα, αν υπάρχουν.
\end{erwthma}
\end{askhsh}

\begin{askhsh}{Γ}
Δίνεται η παραβολή $C_f: f(x) = x^2 + a x + \beta, a, \beta \in \mathbb{R}$. Φέρνουμε δύο εφαπτόμενες $\varepsilon_1,\varepsilon_2$ την $C_f$ από το εξωτερικό σημείο $A(0, -1)$, οι οποίες εφάπτονται της $f$ στα σημεία $M(x_1, f(x_1))$ και $N(x_2, f(x_2))$.
\begin{erwthma}

Έστω συνεχής συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ με $\lim\limits_{x \to -\infty} f(x) = -\infty$, $\lim\limits_{x \to +\infty} \dfrac{f(x)}{x} = 2$ και $\lim\limits_{x \to +\infty} (f(x) - 2x) = 3$. Επιπλέον η $C_f$ τέμνει τον άξονα $x'x$ ακριβώς μια φορά στο $x_0 = -1$.
\begin{erwthma}
  \item Να βρείτε, εάν υπάρχει, την ασύμπτωτη της $f(x)$ στο $+\infty$. 
  \item Με βάση το δοσμένο όριο πηλίκου, υπολογίστε το $\lim\limits_{x \to +\infty} \dfrac{f^2(x) - x^2}{x^2 + f(x)}$.
  \item Εξετάστε το πρόσημο της συνάρτησης αξιοποιώντας το όριο στο $-\infty$ και το όριο στο $+\infty$.
  \item Να βρείτε το όριο της σύνθεσης $\lim\limits_{x \to -\infty} \ln(|f(x)|) \cdot e^{f(x)}$ και $\lim\limits_{x \to 0} \dfrac{1}{f(x)}$ αν η ρίζα είναι $x_0=-1$. 
\end{erwthma}
