\section*{ΚΕΦΑΛΑΙΟ 2: Διαφορικός Λογισμός (100 Ερωτήσεις)}
\begin{enumerate}[label=\textbf{2.\arabic*.}]
    \item Στιγμιαία ταχύτητα κινητού ισούται με την παράγωγο της θέσης του.
    \item Η έννοια του ρυθμού μεταβολής $y$ ως προς $x$ είναι το $f'(x_0)$.
    \item Μια συνάρτηση παραγωγίσιμη στο $x_0$ είναι πάντα συνεχής στο $x_0$.
    \item Κάθε συνεχής συνάρτηση σε ένα σημείο $x_0$, είναι και παραγωγίσιμη σε αυτό.
    \item Αν μια συνάρτηση δεν είναι συνεχής στο $x_0$, τότε ΔΕΝ είναι παραγωγίσιμη σε αυτό.
    \item Η παράγωγος της $|x|$ στο 0 είναι 0.
    \item H $f(x)=|x|$ είναι συνεχής αλλά όχι παραγωγίσιμη στο 0.
    \item Η συνάρτηση $f(x) = \sqrt{x}$ είναι παραγωγίσιμη στο $x=0$.
    \item Η εφαπτομένη της γραφικής παράστασης μιας συνάρτησης απαγορεύεται να τέμνει την καμπύλη σε άλλα σημεία.
    \item Ο συντελεστής διεύθυνσης της εφαπτομένης στο σημείο $(x_0, f(x_0))$ είναι η $f'(x_0)$.
    \item Αν η εφαπτομένη είναι παράλληλη στον $x'x$, τότε $f'(x_0) = 0$.
    \item Δύο καμπύλες εφάπτονται, αν έχουν κοινό σημείο με κοινή εφαπτομένη (δηλ. $f(x_0)=g(x_0)$ και $f'(x_0)=g'(x_0)$).
    \item Κάθε σημείο $x_0$ με $f'(x_0)=0$ λέγεται κρίσιμο σημείο.
    \item Κρίσιμα σημεία είναι μόνο όσα μηδενίζουν την παράγωγο.
    \item Τα εσωτερικά σημεία όπου δεν υπάρχει η παράγωγος, συμπεριλαμβάνονται στα κρίσιμα σημεία.
    \item Σε κάθε τοπικό ακρότατο η παράγωγος πρέπει να μηδενίζεται.
    \item Θεώρημα \eng{Fermat}: Αν η $f$ έχει τοπικό ακρότατο στο εσωτερικό σημείο $x_0$ και είναι παραγωγίσιμη σε αυτό, τότε $f'(x_0) = 0$.
    \item Αν $f'(x_0) = 0$, τότε υποχρεωτικά το $x_0$ είναι τοπικό ακρότατο.
    \item Αν $x_0$ είναι ακρότατο σε κλειστό άκρο διαστήματος, τότε ισχύει το \eng{Fermat}.
    \item Αν $x_0$ εσωτερικό όπου η $f$ παραγωγίζεται και $f'(x_0) \ne 0$, τότε το $x_0$ δεν είναι ακρότατο.
    \item Αν $f$ συνεχής στο $[a, \beta]$ και παραγωγίσιμη στο $(a, \beta)$ με $f(a)=f(\beta)$, υπάρχει μιά τουλάχιστον ρίζα της $f'$ στο $(a, \beta)$.
    \item Αν υπάρχει εξακριβωμένα $\xi \in (a, \beta)$ με $f'(\xi)=0$, τότε υποχρεωτικά $f(a)=f(\beta)$.
    \item Το Θεώρημα \eng{Rolle} εξασφαλίζει τουλάχιστον μια οριζόντια εφαπτομένη.
    \item Το Θ.Μ.Τ. προϋποθέτει ότι η συνάρτηση είναι παραγωγίσιμη και στα άκρα του διαστήματος.
    \item Σύμφωνα με το Θ.Μ.Τ., υπάρχει τουλάχιστον ένα $\xi \in (a, \beta)$ ώστε η εφαπτομένη στο $\xi$ να είναι παράλληλη στη χορδή $AB$.
    \item Το Θ.Μ.Τ. είναι ειδική περίπτωση του \eng{Rolle}.
    \item Αν $f'(x) = 0$ σε ένα διάστημα $\Delta$, τότε η $f$ είναι υποχρεωτικά σταθερή στο $\Delta$.
    \item Αν $f'(x) = 0$ σε ένωση ξένων διαστημάτων, τότε η $f$ μπορεί να παίρνει διαφορετικές σταθερές τιμές σε κάθε διάστημα.
    \item Αν $f'(x) = g'(x)$ στο $\Delta$, τότε $f(x) = g(x) + c$ στο $\Delta$.
    \item Αν $f'(x) > 0$ για κάθε $x \in \Delta$, τότε η $f$ είναι γνησίως αύξουσα στο $\Delta$.
    \item Αν η συνάρτηση $f$ είναι γνησίως αύξουσα, τότε υποχρεωτικά $f'(x) > 0$ για κάθε εσωτερικό σημείο $x$.
    \item Αν μια παραγωγίσιμη συνάρτηση δεν είναι γνησίως αύξουσα, τότε η παράγωγός της δεν μπορεί να είναι παντού θετική.
    \item Αν η παράγωγος μηδενίζεται σε διακριτά μόνο σημεία (όχι σε ολόκληρο διάστημα) και παραμένει π.χ. θετική στα υπόλοιπα, η συνάρτηση εξακολουθεί να είναι γνησίως αύξουσα. 
    \item Αν $f$ είναι 1-1, η παράγωγος της δεν έχει ρίζες.
    \item Ένα σημείο καμπής είναι σημείο όπου η γραφική παράσταση αλλάζει κυρτότητα.
    \item Αν στο σημείο καμπής η συνάρτηση παραγωγίζεται, τότε η εφαπτομένη διαπερνά την καμπύλη.
    \item Εάν η εφαπτομένη σε ένα σημείο $M$ της γραφικής παράστασης διαπερνά την καμπύλη, τότε το $M$ ονομάζεται σημείο καμπής της.
    \item Αν η $f''(x)$ αλλάζει πρόσημο εκατέρωθεν του $x_0$ (όπου ορίζεται η εφαπτομένη), το $x_0$ είναι σημείο καμπής.
    \item Αν το $x_0$ είναι σημείο καμπής και υπάρχει η δεύτερη παράγωγος, τότε υποχρεωτικά $f''(x_0) = 0$.
    \item Αν $f''(x_0) = 0$, τότε το $x_0$ είναι απαραίτητα σημείο καμπής.
    \item Αν μια συνάρτηση είναι κυρτή στο $\Delta$, η εφαπτομένη της βρίσκεται πάντοτε "κάτω" από τη γραφική παράσταση, με εξαίρεση το σημείο επαφής.
    \item Αν η $f''(x) > 0$, τότε υποχρεωτικά η $f(x)$ είναι γνησίως αύξουσα.
    \item Η παράγωγος του αθροίσματος $(f(x)+g(x))'$ ισούται με $f'(x)+g'(x)$.
    \item O κανόνας γινομένου ορίζει $(f(x)g(x))' = f'(x)g'(x)$.
    \item Παράγωγος σταθερής συνάρτησης είναι 0.
    \item Η παράγωγος της $\hm x$ είναι $\syn x$, αλλά σε μοίρες; (Όχι, αποκλειστικά σε ακτίνια).
    \item Ισχύει ότι $(f(g(x)))' = f'(g(x)) \cdot g'(x)$.
    \item Η παράγωγος της $e^x$ είναι $e^x$, ενώ του $a^x$ είναι $a^x \ln a$.
    \item Η παράγωγος της $\ln x$ είναι $1/x$ για $x>0$, και της $\ln |x|$ είναι επίσης $1/x$ για $x \ne 0$.
    \item Ένα τοπικό μέγιστο μπορεί να είναι μικρότερο από ένα τοπικό ελάχιστο.
    \item Κάθε ολικό μέγιστο είναι σίγουρα μεγαλύτερο από οποιοδήποτε τοπικό ελάχιστο.
    \item Ο κανόνας \eng{De L'Hospital} εφαρμόζεται μόνο σε μορφές $\dfrac{0}{0}$ ή $\dfrac{\pm \infty}{\pm \infty}$.
    \item Για μορφές $0 \cdot \infty$, μπορούμε να εφαρμόσουμε κανόνα \eng{De L'Hospital}
    \item Η συνάρτηση $\dfrac{|x|}{x}$ έχει παράγωγο παντού στο $\mathbb{R}^*$ ίση με μηδέν.
    \item Υπάρχει συνάρτηση που η παράγωγός της είναι η $f'(x) = |x|$.
    \item Η παράγωγος μιας άρτιας είναι περιττή.
    \item Η παράγωγος πολυωνύμου 3ου βαθμού έχει άπειρο πεδίο ορισμού.
    \item Η συνάρτηση $f(x) = x^3$ είναι γνησίως αύξουσα στο $\mathbb{R}$, παρόλο που η παράγωγός της μηδενίζεται στο $x=0$.
    \item Αν $f$ συνεχή στο κλειστό $[a, \beta]$ και δεν έχει κρίσιμα σημεία, τα ακρότατά της εντοπίζονται υποχρεωτικά στα άκρα.
    \item Αν η μεταβολή ενός μεγέθους $N$ δίνεται από παραγωγίσιμη συνάρτηση του χρόνου $t$, ο ρυθμός μεταβολής του μεγέθους υπολογίζεται ως η παράγωγος $N'(t)$.
    \item Κάθε συνεχής συνάρτηση σε ανοιχτό διάστημα έχει τουλάχιστον ένα κρίσιμο σημείο.
    \item Το θεώρημα \eng{\eng{Fermat}} αναφέρεται αποκλειστικά σε ολικά ακρότατα.
    \item Αν $f'(x_0) = 0$ και $f''(x_0) > 0$, τότε η $f$ έχει τοπικό ελάχιστο στο $x_0$.
    \item Αν η συνάρτηση $f$ έχει κατακόρυφη ασύμπτωτη την ευθεία $x = x_0$, τότε σίγουρα η $f$ δεν είναι συνεχής στο $x_0$.
    \item Στην κίνηση κινητού σε άξονα, αν ταχύτητα και επιτάχυνση είναι ομόσημες, η κίνηση επιταχύνεται.
    \item Η κατακόρυφη ασύμπτωτη μιας συνάρτησης μελετάται στο $\pm \infty$.
    \item Μια πολυωνυμική συνάρτηση βαθμού $\ge 2$ μπορεί να έχει πλάγια ασύμπτωτη.
    \item Αν $\lim \dfrac{f(x)}{x} = \lambda$ και $\lim (f(x)-\lambda x) = \mu$, τότε η $y=\lambda x + \mu$ είναι πλάγια (ή οριζόντια) ασύμπτωτη.
    \item Ρυθμός μεταβολής 0 υποδηλώνει ότι το μέγεθος (στιγμιαία) σταματά να μεταβάλλεται.
    \item Αν το $\lim_{x\to +\infty} (f(x) - (x+2)) = 0$, τότε η ευθεία $y = x+2$ είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$.
    \item Υπάρχει πιθανότητα το Θεώρημα Μέσης Τιμής να δίνει περισσότερα από ένα $\xi$.
    \item Για κάθε πολυωνυμική συνάρτηση βαθμού μεγαλύτερου του 2, αν έχει τοπικά ακρότατα σε σημεία $x_1<x_2$, τότε θα έχει και σημείο καμπής.
    \item Μια συνάρτηση μπορεί να έχει μέγιστο σε σημείο μη παραγωγισιμότητας.
    \item Αν $f$ 1-1 και παραγωγίσιμη, η αντίστροφη είναι επίσης παραγωγίσιμη στο πεδίο αντίστοιχων τιμών, αρκεί $f'(x) \ne 0$.
    \item Η παράγωγος του $x^x$ είναι $x \cdot x^{x-1}$.
    \item Αν $f, g$ κυρτές, το άθροισμά τους $f+g$ είναι κυρτή.
    \item Το πεδίο ορισμού της συνάρτησης παράγωγος $f'$ είναι πάντα υποσύνολο του πεδίου ορισμού της $f$.
    \item Το σημείο καμπής είναι το σημείο όπου ο ρυθμός μεταβολής της κλίσης αλλάζει από αύξηση σε μείωση (ή αντίστροφα).
    \item Αν η εξίσωση $f'(x)=0$ έχει μια μόνο ρίζα σε ένα διάστημα, η $f$ μπορεί να μην παρουσιάζει κανένα ακρότατο στο διάστημα αυτό.
    \item Οι συντελεστές $\lambda, \mu$ της ασύμπτωτης στο $+\infty$ ταυτίζονται υποχρεωτικά με αυτούς στο $-\infty$.
    \item Αν $f$ περιττή, έχει $f'(0)=0$ ή $f(0)=0$. (Μόνο το $f(0)=0$ αν ορίζεται).

    \item Αν η συνάρτηση $f$ έχει τοπικό μέγιστο στο σημείο $x_0$, τότε απαραιτήτως ισχύει $f'(x_0) = 0$.
    \item Η ύπαρξη δεύτερης παραγώγου στο $x_0$ εξασφαλίζει ότι το $x_0$ είναι σημείο καμπής, εάν προηγουμένως γνωρίζουμε ότι $f''(x_0)=0$.
    \item Υπάρχει συνάρτηση που είναι συνεχής στο $\mathbb{R}$ αλλά παρουσιάζει σημείο όπου η εφαπτομένη της δεν ορίζεται.
    \item Αν $f''(x)>0$ στο $\Delta$, τότε το τμήμα της $C_f$ στο $\Delta$ βρίσκεται εξολοκλήρου ``πάνω'' από την εφαπτομένη της, με εξαίρεση το σημείο επαφής.
    \item $f(x) \ge f(x_0) + f'(x_0)(x-x_0)$ για κάθε κυρτή $f$.
    \item Αν δύο συναρτήσεις $f, g$ έχουν την ίδια παράγωγο στο $\Delta$, τότε οι εφαπτομένες των $C_f, C_g$ στα σημεία με μια κοινή τετμημένη, είναι είτε παράλληλες είτε ταυτίζονται.
    \item Τα κρίσιμα σημεία εντοπίζονται και στα άκρα του πεδίου ορισμού κλειστού διαστήματος.
    \item Οι εκθετικές εξισώσεις επιλύονται συχνά χρησιμοποιώντας θεωρήματα μονοτονίας.
    \item Κάθε τοπικό ελάχιστο είναι πάντοτε κρίσιμο σημείο.
    \item Αν $f''(x) > 0$, η $f'$ είναι 1-1.
    \item Η σχέση $f'(x)>0 \implies f$ αύξουσα, αποδεικνύεται με το Θ.Μ.Τ.
    \item Αν η συνάρτηση $s(t)$ παριστάνει τη θέση κινητού την χρονική στιγμή $t$, τότε ο ρυθμός μεταβολής της ταχύτητας $v(t)$ δίνεται από τον τύπο $a(t)=s''(t)$.
    \item Η εφαπτομένη της $e^x$ στο 0 είναι η ευθεία $y=x+1$.
    \item Η $f(x)=x^3$ έχει εξίσωση εφαπτομένης στο $x=0$ τον άξονα $x'x$.
    \item H ασύμπτωτη της $f(x)=\dfrac{1}{x}$ στο άπειρο είναι ο άξονας $x'x$.
    \item Κάθε διάστημα που η γραφική παράσταση είναι "κοίλη", η παράγωγος είναι γνησίως φθίνουσα.
    \item Η ακρότατη τιμή μιας συνεχούς $f$ σε κλειστό $[a,b]$ λαμβάνεται είτε σε κρίσιμο σημείο είτε στα άκρα $a, b$.
    \item Κάθε εσωτερικό σημείο $x_0$ στο οποίο μηδενίζεται η παράγωγος, δεν είναι υποχρεωτικά θέση τοπικού ακροτάτου.
\end{enumerate}

\section*{ΚΕΦΑΛΑΙΟ 3: Ολοκληρωτικός Λογισμός (60 Ερωτήσεις)}
\begin{enumerate}[label=\textbf{3.\arabic*.}]
    \item Κάθε συνεχής συνάρτηση σε ένα κλειστό διάστημα $[\alpha, \beta]$ είναι ολοκληρώσιμη σε αυτό.
    \item Αν $F, G$ είναι δύο παράγουσες της $f$ στο $\Delta$, τότε διαφέρουν κατά μια σταθερά $c$, δηλαδή $F(x) = G(x) + c$.
    \item Αν ισχύει $F(x) = G(x) + c$, τότε η $F$ και η $G$ είναι παράγουσες της ίδιας συνάρτησης.
    \item Αν $F, G$ είναι παράγουσες της συνάρτησης $f$ σε σύνολο που αποτελεί ένωση δύο ξένων διαστημάτων, τότε διαφέρουν κατά μία υποχρεωτικά σταθερά παντού.
    \item Το $\dint_a^\beta f(x) dx$ είναι ένας πραγματικός αριθμός, εξαρτώμενος από τα όρια $a, \beta$ και τη συνάρτηση.
    \item Το ολοκλήρωμα $\dint_{\alpha}^{\beta} f(x) dx$ εκφράζει υποχρεωτικά εμβαδόν κάποιου χωρίου.
    \item Αν $f(x) \ge 0$ για $x \in [a, \beta]$, τότε $\dint_a^\beta f(x) dx \ge 0$.
    \item Αν $\dint_a^\beta f(x) dx \ge 0$, τότε υποχρεωτικά η $f$ είναι μη αρνητική παντού στο $[a, \beta]$.
    \item Αν $\dint_a^\beta f(x) dx < 0$, τότε απαγορεύεται η $f(x)$ να είναι $\ge 0$ παντού στο $[a, \beta]$.
    \item Αν η $f$ είναι συνεχής με $\dint_a^\beta f(x) dx = 0$ και $f(x) \ge 0$, τότε $f(x) = 0$ για κάθε $x \in [a, \beta]$.
    \item Αν μια συνεχής μη-αρνητική συνάρτηση δεν είναι η μηδενική, τότε το ολοκλήρωμά της σε διάστημα $a<\beta$ είναι αυστηρά $> 0$.
    \item Ισχύει ότι $\dint_a^\beta f(x) dx = F(\beta) - F(a)$, όπου $F$ οποιαδήποτε παράγουσα της $f$.
    \item Το εμβαδόν ενός χωρίου που περικλείεται από την $C_f$, τον άξονα $x'x$ και τις ευθείες $x=\alpha$, $x=\beta$, είναι ένας μη αρνητικός αριθμός.
    \item Υπάρχει παράγουσα για κάθε συνεχής συνάρτηση στο $\mathbb{R}$.
    \item Ισχύει $\dint_a^\beta f(t) dt + \dint_\beta^a f(x) dx = 0$.
    \item Ο συμβολισμός της μεταβλητής ολοκλήρωσης $x$ ή $t$ δεν προκαλεί καμία απολύτως αλλαγή στην αριθμητική τιμή ενός ορισμένου ολοκληρώματος.
    \item Η παράγωγος $\left(\dint_a^b f(x) dx \right)'$ ισούται με $ f(x)$.
    \item Η τιμή του $\dint_{\alpha}^{\beta} f(x) dx$ (εφόσον η $f$ είναι συνεχής), είναι πάντοτε ένας θετικός αριθμός.
    \item Το εμβαδόν μεταξύ της συνεχούς $f$ και του $x'x$ από το $a$ στο $\beta$ δίνεται πάντα από τον τύπο $E = \dint_a^\beta f(x) dx$. 
    \item Αν $f(x) \le 0$ στο $[a, \beta]$, τότε το εμβαδόν του χωρίου δίνεται από $-\dint_a^\beta f(x) dx$.
    \item Το εμβαδόν μεταξύ δύο καμπυλών είναι $\dint_a^\beta |f(x) - g(x)| dx$.
    \item Ο τύπος ολοκλήρωσης κατά παράγοντες είναι $\dint_a^\beta f(x)g'(x) dx = [f(x)g(x)]_a^\beta - \dint_a^\beta f'(x)g'(x) dx$. (Λάθος ο 2ος όρος).
    \item Ισχύει ότι $\dint_{\alpha}^{\beta} (f(x) + g(x)) dx = \dint_{\alpha}^{\beta} f(x) dx + \dint_{\alpha}^{\beta} g(x) dx$, όταν οι $f, g$ είναι συνεχείς στο $[\alpha, \beta]$.
    \item Αν $F$ είναι μια παράγουσα της καλώς ορισμένης και συνεχούς $f$, τότε ισχύει $F'(x) = f(x)$ στο πεδίο ορισμού της.
    \item Αν $f$ περιττή και ολοκληρώσιμη, τότε $\dint_{-a}^a f(x) dx = 0$.
    \item Αν $f$ άρτια, τότε $\dint_{-a}^a f(x) dx = 2 \dint_0^a f(x) dx$.
    \item Η σταθερά μέσα σε ένα ολοκλήρωμα μπορεί να "εξαχθεί" εκτός: $\dint_a^\beta c f(x) dx = c \dint_a^\beta f(x) dx$.
    \item Το ολοκλήρωμα του αθροίσματος ισούται με το άθροισμα των ολοκληρωμάτων.
    \item Το ολοκλήρωμα του γινομένου ισούται πάντοτε με το γινόμενο των ολοκληρωμάτων.
    \item Το $\dint_{\alpha}^{\beta} f(x) dx$ με $\alpha, \beta$ σταθερές, αποτελεί μια συνάρτηση της μεταβλητής $x$.
    \item Αν $f(x) < g(x)$ για κάθε $x \in [a, \beta]$ (και συνεχείς), τότε $\dint_a^\beta f(x) dx < \dint_a^\beta g(x) dx$.
    \item Υπάρχουν συναρτήσεις συνεχείς στο $[\alpha, \beta]$ για τις οποίες δεν μπορεί να υπολογιστεί η τιμή $\dint_{\alpha}^{\beta} f(x) dx$ προσεγγιστικά.
    \item Αν $\dint_a^\beta f(x) dx = \dint_a^\beta g(x) dx$, τότε υποχρεωτικά συνεπάγεται ότι $f(x) = g(x)$ σε όλο το $[a, \beta]$.
    \item Αν $\dint_a^\beta f(x) dx = 0$ και η $f$ παίρνει και θετικές και αρνητικές τιμές, το εμβαδόν του αντίστοιχου χωρίου είναι στο σύνολο 0. (Το εμβαδόν είναι $>0$).
    \item Το εμβαδόν μεταξύ δύο ευθειών που τέμνονται και του άξονα ολοκληρώνεται χωρίς να χωριστεί το διάστημα στην προβολή της τομής τους.
    \item Το ορισμένο ολοκλήρωμα μιας συνεχούς συνάρτησης, γεωμετρικά προκύπτει ως το όριο αθροισμάτων εμβαδών απείρων ορθογωνίων.
    \item Το ολοκλήρωμα $\dint_1^2 e^x dx $ ισούται με $ e^2 - e$.
    \item Αν η $f'$ είναι συνεχής, τότε $\dint_a^\beta f'(x) dx = f(\beta) - f(a)$.
    \item Αν $\dint_a^\beta f(x) dx = 0$ και $a \ne \beta$, συνεπάγεται ότι η $f$ έχει τουλάχιστον μια ρίζα στο $[a, \beta]$ (προυποθέτοντας ότι η $f$ είναι συνεχής).
    \item Κάθε συνεχής συνάρτηση είναι παράγουσα κάποιας άλλης συνάρτησης.
    \item Αν $\dint_{\alpha}^{\beta} f(x) dx > 0$ με $\alpha < \beta$, τότε η συνεχής συνάρτηση $f$ θα παίρνει απαραιτήτως και θετικές τιμές μέσα στο $[\alpha, \beta]$.
    \item Η $f(x) = \ln x$ έχει παράγουσα την $F(x) = x\ln x - x + c$.
    \item Αν μια συνάρτηση $f$ είναι περιττή και συνεχής, το εμβαδόν του χωρίου μεταξύ της $C_f$, του $x'x$ και των $x=-\alpha, x=\alpha$ ($\alpha>0$) ισούται με $2\dint_0^\alpha |f(x)| dx$.
    \item Αν μετατοπίσουμε μια θετική συνάρτηση κατακόρυφα κατά $c>0$, το εμβαδόν μεταξύ της καμπύλης, του $x'x$ και των $x=\alpha, x=\beta$ (εφόσον $\alpha<\beta$) αυξάνεται κατά $c(\beta-\alpha)$.
    \item Αν η $f'$ είναι συνεχής συνάρτηση, τότε $\dint_{\alpha}^{\beta} f'(x)g(x)dx = [f(x)g(x)]_{\alpha}^{\beta} - \dint_{\alpha}^{\beta} f(x)g'(x)dx$.
    \item Για τον υπολογισμό του εμβαδού του χωρίου μεταξύ των $C_f, C_g$, είναι απαραίτητο να εντοπίσουμε πρώτα τις τετμημένες των σημείων τομής τους.
    \item Αν $f(x) \geq g(x)$ για $x \in [\alpha, \beta]$, τότε το εμβαδόν ανάμεσά τους στο εν λόγω διάστημα ισούται με $\dint_{\alpha}^{\beta} (f(x)-g(x)) dx$.
    \item $\dint_a^b c dt = c(b-a)$ για άγνωστο σταθερό $c$.
    \item Αν $f'(x) = g'(x)$ τότε $\dint_a^\beta f'(x)dx = \dint_a^\beta g'(x)dx \Rightarrow f(\beta)-f(a) = g(\beta)-g(a)$.
    \item Ισχύει ότι $\dint_{\alpha}^{\alpha} f(x) dx = 0$, ανεξάρτητα από τον τύπο της συνεχούς συνάρτησης $f$.
    \item Ισχύει ότι $\dint_{a}^{\beta} f(a) dx = f(a)(\beta - a)$.
    \item Αν $f(x) \ge 2$, τότε $\dint_1^3 f(x) dx \ge 4$.
    \item Αν $\dint_\alpha^\beta f(x)dx > 0$ με $\alpha < \beta$, τότε αναγκαστικά $f(x) > 0$ για κάθε $x \in [\alpha, \beta]$.
    \item Αν η $f$ είναι συνεχής και $f(x) \ge 0$ για κάθε $x \in [\alpha, \beta]$ με $\dint_\alpha^\beta f(x) dx = 0$, τότε υποχρεωτικά $f(x) = 0$ σε όλο το διάστημα.
    \item Για κάθε συνεχή συνάρτηση $f$, ισχύει ότι $\left| \dint_\alpha^\beta f(x)dx \right| = \dint_\alpha^\beta |f(x)|dx$ αν και μόνο αν η $f$ διατηρεί σταθερό πρόσημο στο $[\alpha, \beta]$.
    \item Το ολοκλήρωμα $\dint_1^1 f(x)dx$ είναι μηδέν για κάθε $f$.
    \item Αν $\dint_a^\beta f(x)dx = 0$, τότε αναγκαστικά $a = \beta$.
    \item Αν $f(x)<g(x)$ για κάθε $x \in [\alpha, \beta]$ (οπού $f,g$ συνεχείς), τότε το εμβαδόν μεταξύ των $C_f, C_g$ δίνεται από το $E = \dint_\alpha^\beta (g(x)-f(x))dx$, ανεξάρτητα από το αν οι $f, g$ λαμβάνουν αρνητικές τιμές.
    \item Αν $\dint_\alpha^\beta f(x) dx = \dint_\alpha^\beta g(x) dx$, με $\alpha \ne \beta$ και $f,g$ συνεχείς, τότε απαραίτητα οι γραφικές τους παραστάσεις έχουν τουλάχιστον ένα κοινό σημείο στο $[\alpha, \beta]$.
    \item Το ολοκλήρωμα $\dint_a^b |f(x)| dx$ αντιπροσωπεύει πάντοτε το εμβαδόν του χωρίου μεταξύ της $C_{f}$, του άξονα $x'x$ και των $x=a$ και $x=b$, ακόμα και αν δεν γνωρίζουμε ποιο είναι μεγαλύτερο ($a$ ή $b$).
\end{enumerate}
