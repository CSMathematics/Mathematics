\subsection*{Παραγοντική Ολοκλήρωση: Η Μέθοδος Tabular (DI)}
Η μέθοδος του παραγοντικού πίνακα χρησιμοποιείται όταν έχουμε γινόμενο Πολυωνύμου με Εκθετική/Τριγωνομετρική συνάρτηση, ειδικά όταν το πολυώνυμο έχει μεγάλο βαθμό, άρα θα χρειαζόντουσαν απανωτές κλασικές παραγοντικές. 

\textbf{Βήματα αλγορίθμου:}
1. Ορίζουμε στήλη Προσήμων (ενάλλασσονται $+ \ \ - \ \ + \dots$).
2. Στη στήλη Διαφόρισης (D) γράφουμε το πολυώνυμο και το παραγωγίζουμε μέχρι μηδενισμού.
3. Στη στήλη Ολοκλήρωσης (I) γράφουμε και ολοκληρώνουμε σε κάθε γραμμή την εκάστοτε δύσκολη συνάρτηση.
4. Το αποτέλεσμα είναι τα χιαστί γινόμενα (Διαγώνια).

\begin{center}
\textbf{Περίπτωση 1:} $P(x) \cdot e^{ax}$
\begin{mytblr}{colspec={|c|X[c]|X[c]|}, row{1}={purple!80!black, fg=white, font=\bfseries}, row{even}={purple!10!white}}
Πρόσημο (+ / -) & Παραγώγιση (D) - \textit{Πολυώνυμο} & Ολοκλήρωση (I) - \textit{Εκθετική} \\
$(+)$ & $P(x)$ & $e^{ax+\beta}$ \vspace{1.5mm} \\
$(-)$ & $\searrow$ \ $P'(x)$ & $\dfrac{1}{a} e^{ax+\beta}$ \vspace{1.5mm} \\
$(+)$ & $\searrow$ \ $P''(x)$ & $\dfrac{1}{a^2} e^{ax+\beta}$ \vspace{1.5mm} \\
$\vdots$ & $\vdots$ & $\vdots$ \\
& $0$ & $\dfrac{1}{a^\nu} e^{ax+\beta}$
\end{mytblr}
\end{center}

\begin{center}
\textbf{Περίπτωση 2:} $P(x) \cdot \hm{(ax)}$
\begin{mytblr}{colspec={|c|X[c]|X[c]|}, row{1}={purple!80!black, fg=white, font=\bfseries}, row{even}={purple!10!white}}
Πρόσημο (+ / -) & Παραγώγιση (D) - \textit{Πολυώνυμο} & Ολοκλήρωση (I) - \textit{Τριγωνομετρική} \\
$(+)$ & $P(x)$ & $\hm{(ax)}$ \vspace{1.5mm} \\
$(-)$ & $\searrow$ \ $P'(x)$ & $-\dfrac{1}{a} \syn{(ax)}$ \vspace{1.5mm} \\
$(+)$ & $\searrow$ \ $P''(x)$ & $-\dfrac{1}{a^2} \hm{(ax)}$ \vspace{1.5mm} \\
$\vdots$ & $\vdots$ & $\vdots$ \\
& $0$ & $\pm \dfrac{1}{a^\nu} \syn{(ax)} \text{ ή } \hm{(ax)}$
\end{mytblr}
\end{center}
