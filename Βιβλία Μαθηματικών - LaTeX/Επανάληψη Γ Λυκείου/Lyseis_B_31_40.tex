\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = e^x - 1$ και $g(x) = \ln x$. 
\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι το $D_f = \mathbb{R}$ και της $g$ είναι το $A_g = (0, +\infty)$.
- Για την $h = \frac{f}{g}$, απαιτείται $x \in D_f \cap A_g$ και $g(x) \ne 0 \Leftrightarrow \ln x \ne 0 \Leftrightarrow x \ne 1$.
Άρα το πεδίο ορισμού είναι το $A_h = (0, 1) \cup (1, +\infty)$. 
Τύπος: $h(x) = \frac{e^x - 1}{\ln x}$.
- Για την $k = f \cdot g$, απαιτείται $x \in D_f \cap A_g \Rightarrow x \in (0, +\infty)$.
Άρα $A_k = (0, +\infty)$. Τύπος: $k(x) = (e^x - 1) \ln x$.

\item Υπολογίζουμε τα πλευρικά όρια στο $x_0 = 1$ για τη συνάρτηση $h(x)$:
Ο αριθμητής τείνει στο $e^1 - 1 = e - 1 > 0$.
Ο παρονομαστής $\ln x$ τείνει στο $0$. Συγκεκριμένα:
- $\lim_{x\to1^+} \ln x = 0^+$ (αφού $x>1 \Rightarrow \ln x > 0$), άρα $\lim_{x\to1^+} \frac{e^x-1}{\ln x} = \frac{e-1}{0^+} = +\infty$.
- $\lim_{x\to1^-} \ln x = 0^-$ (αφού $x<1 \Rightarrow \ln x < 0$), άρα $\lim_{x\to1^-} \frac{e^x-1}{\ln x} = \frac{e-1}{0^-} = -\infty$.
Εφόσον τα πλευρικά όρια είναι διαφορετικά, το όριο $\lim_{x\to1} \frac{f(x)}{g(x)}$ \textbf{δεν υπάρχει}.

\item Εξετάζουμε την εξίσωση $k(x) = 1 \Leftrightarrow k(x) - 1 = 0$ στο $[1, e]$.
Θεωρούμε τη συνάρτηση $K(x) = k(x) - 1 = (e^x-1)\ln x - 1$, συνεχής στο $[1, e]$.
- $K(1) = (e-1)\ln 1 - 1 = 0 - 1 = -1 < 0$.
- $K(e) = (e^e-1)\ln e - 1 = e^e - 1 - 1 = e^e - 2$. Επειδή $e > 2 \Rightarrow e^e > 2^2 = 4 \Rightarrow e^e-2 > 0$. 
Οπότε $K(e) > 0$.
Αφού $K(1)K(e) < 0$, σύμφωνα με το Θεώρημα Bolzano έχει τουλάχιστον μία πραγματική ρίζα της εξίσωσης στο $(1, e)$.

\item Εξετάζουμε τα όρια στο $\pm\infty$:
- Στο $-\infty$: $\lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} (e^x - 1) = 0 - 1 = -1$.
Άρα η ευθεία $y = -1$ είναι οριζόντια ασύμπτωτη στο $-\infty$.
- Στο $+\infty$: $\lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} (e^x - 1) = +\infty$. 
Δεν έχει οριζόντια ασύμπτωτη. Για πλάγια ελέγχουμε το $\lim_{x\to+\infty} \frac{f(x)}{x} = \lim_{x\to+\infty} \frac{e^x-1}{x} \xlongequal{\text{DLH}} \lim_{x\to+\infty} e^x = +\infty$. Οπότε δεν υπάρχει ούτε πλάγια ασύμπτωτη στο $+\infty$. 
Κατακόρυφες δεν υπάρχουν διότι είναι συνεχής στο $\mathbb{R}$.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x \ln x$ με πεδίο ορισμού $(0, +\infty)$.
\begin{erwthma}
\item Το πλευρικό όριο αναφορικά στο 0 υπολογίζεται ως έξης:
\[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} (x \ln x) = \lim_{x\to0^+} \frac{\ln x}{1/x} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to0^+} \frac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x\to0^+} (-x) = 0 \]
Επειδή το όριο είναι πραγματικός αριθμός, η γραφική παράσταση δεν έχει κατακόρυφη ασύμπτωτη στον άξονα $y'y$ (στο $x=0$).

\item Η $f$ είναι δύο φορές παραγωγίσιμη στο $(0, +\infty)$ με:
\[ f'(x) = x' \ln x + x (\ln x)' = \ln x + 1 \]
Εξισώνοντας την παράγωγο με μηδέν: $f'(x) = 0 \Leftrightarrow \ln x = -1 \Leftrightarrow x = e^{-1} = \frac{1}{e}$.
Είναι $f'(x) > 0 \Leftrightarrow \ln x > -1 \Leftrightarrow x > \frac{1}{e}$ και αντίστοιχα $f'(x) < 0 \Leftrightarrow 0 < x < \frac{1}{e}$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για τη συνάρτηση $f$ στο διάστημα $[1, e]$:
- Η $f$ είναι συνεχής στο $[1, e]$.
- Η $f$ είναι παραγωγίσιμη στο $(1, e)$.
Άρα υπάρχει τουλάχιστον ένα $\xi \in (1, e)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(e) - f(1)}{e - 1} = \frac{e\ln e - 1\ln 1}{e - 1} = \frac{e\cdot 1 - 0}{e-1} = \frac{e}{e-1} \]
Το ζητούμενο αποδείχθηκε.

\item Στο διάστημα $[1, e]$, αφού $x \ge 1$ ισχύει $\ln x \ge 0$, άρα $f(x) \ge 0$. Το εμβαδόν ισούται με:
\[ E = \int_1^e x \ln x dx = \int_1^e \left(\frac{x^2}{2}\right)' \ln x dx = \left[ \frac{x^2}{2} \ln x \right]_1^e - \int_1^e \frac{x^2}{2} \cdot \frac{1}{x} dx \]
\[ = \frac{e^2}{2} \ln e - 0 - \frac{1}{2} \int_1^e x dx = \frac{e^2}{2} - \frac{1}{2} \left[ \frac{x^2}{2} \right]_1^e = \frac{e^2}{2} - \frac{1}{2} \left( \frac{e^2}{2} - \frac{1}{2} \right) = \frac{e^2}{2} - \frac{e^2}{4} + \frac{1}{4} \]
\[ = \frac{2e^2 - e^2 + 1}{4} = \frac{e^2+1}{4} \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x(x-2)e^x$ με πεδίο ορισμού το $\mathbb{R}$.
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση $g(x) = x^2 - 2$, συνεχής στο κλειστό $[0, 2]$.
Υπολογίζουμε τις τιμές στα άκρα:
- $g(0) = 0^2 - 2 = -2 < 0$.
- $g(2) = 2^2 - 2 = 2 > 0$.
Επειδή $g(0)g(2) < 0$, το Θεώρημα Bolzano εξασφαλίζει ότι η εξίσωση $g(x)=0 \Leftrightarrow x^2-2=0$ έχει τουλάχιστον μία πραγματική ρίζα στο διάστημα $(0, 2)$.

\item Γράφουμε την $f(x)$ ως πηλίκο για να υπολογίσουμε το όριο στο $-\infty$:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} (x^2-2x)e^x = \lim_{x\to-\infty} \frac{x^2-2x}{e^{-x}} \]
Επειδή έχουμε απροσδιοριστία μορφής $\frac{+\infty}{+\infty}$, εφαρμόζουμε τον κανόνα De L' Hospital:
\[ \lim_{x\to-\infty} \frac{(x^2-2x)'}{(e^{-x})'} = \lim_{x\to-\infty} \frac{2x-2}{-e^{-x}} \xlongequal[\text{DLH}]{\frac{-\infty}{-\infty}} \lim_{x\to-\infty} \frac{(2x-2)'}{(-e^{-x})'} = \lim_{x\to-\infty} \frac{2}{e^{-x}} = 0 \]

\item Σύμφωνα με το προηγούμενο ερώτημα, επειδή $\lim\limits_{x\to-\infty} f(x) = 0$, συνάγεται ότι η ευθεία $y = 0$ (που ταυτίζεται με τον άξονα $x'x$) αποτελεί την οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $-\infty$.

\item Η $f(x)$ είναι παραγωγίσιμη με:
\[ f'(x) = (x^2-2x)'e^x + (x^2-2x)(e^x)' = (2x-2)e^x + (x^2-2x)e^x = e^x(x^2-2) \]
Οι ρίζες της παραγώγου ταυτίζονται με τις ρίζες του $x^2-2=0$, δηλαδή $x_1 = \sqrt{2}$ και $x_2 = -\sqrt{2}$.
Επειδή $e^x > 0$, το πρόσημο της $f'$ εξαρτάται από το τριώνυμο $x^2-2$.

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{e^x - x - 1}{x^2}$ με $x \ne 0$.
\begin{erwthma}
\item Υπολογίζουμε το όριο χρησιμοποιώντας τον κανόνα De L' Hospital δύο φορές, καθώς έχουμε απροσδιοριστία $\frac{0}{0}$:
\[ \lim_{x\to0} f(x) = \lim_{x\to0} \frac{e^x - x - 1}{x^2} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to0} \frac{(e^x - x - 1)'}{(x^2)'} = \lim_{x\to0} \frac{e^x - 1}{2x} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to0} \frac{(e^x - 1)'}{(2x)'} = \lim_{x\to0} \frac{e^x}{2} = \frac{1}{2} \]

\item Προκειμένου η συνάρτηση:
\[ g(x) = \begin{cases} f(x), & x \ne 0 \\ a, & x = 0 \end{cases} \]
να είναι συνεχής στο $\mathbb{R}$, πρέπει να είναι συνεχής στο σημείο $x_0 = 0$. Αυτό απαιτεί:
\[ \lim_{x\to0} g(x) = g(0) \Leftrightarrow \lim_{x\to0} f(x) = a \Leftrightarrow \frac{1}{2} = a \]
Επομένως, το $a$ πρέπει να είναι ίσο με $0,5$.

\item Εξετάζουμε τη συνεχή συνάρτηση $g(x)$ (με $a=0,5$) στο κλειστό διάστημα $[0, 1]$.
- Υπολογίζουμε $g(0) = a = 0,5$.
- Υπολογίζουμε $g(1) = f(1) = \frac{e^1 - 1 - 1}{1^2} = e - 2$. 
Όμως δίνεται ότι $e \approx 2,71$, οπότε $g(1) \approx 2,71 - 2 = 0,71$.
Παρατηρούμε ότι $0,5 < 0,7 < 0,71$, οπότε $g(0) < 0,7 < g(1)$.
Επειδή η $g$ είναι συνεχής στο $[0, 1]$, σύμφωνα με το Θεώρημα Ενδιάμεσων Τιμών υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $g(x_0) = 0,7$.

\item Θέλουμε να αποδείξουμε ότι $\lim_{x\to+\infty} \frac{\ln x}{f(x)} = 0$.
Αρχικά εξάγουμε το $\lim_{x\to+\infty} f(x)$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{e^x - x - 1}{x^2} \xlongequal{\text{DLH}} \lim_{x\to+\infty} \frac{e^x - 1}{2x} \xlongequal{\text{DLH}} \lim_{x\to+\infty} \frac{e^x}{2} = +\infty \]
Οπότε έχουμε απροσδιοριστία μορφής $\frac{+\infty}{+\infty}$. 
Ο συνολικός λόγος γράφεται:
\[ K(x) = \frac{\ln x}{\frac{e^x-x-1}{x^2}} = \frac{x^2 \ln x}{e^x - x - 1} = \frac{\frac{x^2 \ln x}{e^x}}{\frac{e^x - x - 1}{e^x}} = \frac{\frac{x^2}{e^{x/2}} \cdot \frac{\ln x}{e^{x/2}}}{1 - \frac{x}{e^x} - \frac{1}{e^x}} \]
Όμως γνωρίζουμε από βασικά όρια ότι οι εκθετικές συναρτήσεις κυριαρχούν τόσο των πολυωνύμων όσο και των λογαρίθμων στο $+\infty$. 
Πράγματι, $\lim \frac{x^2}{e^{x/2}} = 0$ (με δύο φορές DLH) και $\lim \frac{\ln x}{e^{x/2}} = 0$. 
Επίσης $\lim \frac{x}{e^x} = 0$ και $\lim \frac{1}{e^x} = 0$. 
Άρα:
\[ \lim_{x\to+\infty} \frac{\ln x}{f(x)} = \frac{0 \cdot 0}{1 - 0 - 0} = \frac{0}{1} = 0 \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^3 - 3x + 1$ με $x \in \mathbb{R}$.
\begin{erwthma}
\item Η $f$ είναι συνεχής στο κλειστό $[0, 1]$ ως πολυωνυμική.
Επίσης:
- $f(0) = 0^3 - 3 \cdot 0 + 1 = 1 > 0$.
- $f(1) = 1^3 - 3 \cdot 1 + 1 = -1 < 0$.
Αφού $f(0)f(1) < 0$, από το Θεώρημα Bolzano υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $f(x_0)=0$.

\[ f'(x) = 3x^2 - 3 = 3(x^2 - 1) \]
Οι ρίζες είναι $x_1=1$ και $x_2=-1$.
$f'(x) > 0 \Leftrightarrow x^2 > 1 \Leftrightarrow x \in (-\infty, -1) \cup (1, +\infty)$.
$f'(x) < 0 \Leftrightarrow x \in (-1, 1)$.

\end{erwthma}

\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = \ln(x^2+1)$ και $g(x) = x$, με πεδίο ορισμού το $\mathbb{R}$.
\begin{erwthma}
\item Η συνάρτηση ορίζεται ως $h(x) = f(x) - g(x) = \ln(x^2+1) - x$ με $A_h = \mathbb{R}$.
Ζητείται να δειχθεί ότι $\lim_{x\to0} \frac{h(x)}{x^2} = 1$. Ωστόσο, υπάρχει ένα προφανές τυπογραφικό λάθος στην εκφώνηση του αντίστοιχου διαγωνίσματος, διότι:
\[ \lim_{x\to0} \frac{h(x)}{x^2} = \lim_{x\to0} \frac{\ln(x^2+1) - x}{x^2} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to0} \frac{\frac{2x}{x^2+1} - 1}{2x} = \lim_{x\to0} \frac{2x - x^2 - 1}{2x(x^2+1)} = \frac{-1}{0} \]
Το οποίο οδηγεί στο $\pm\infty$. Το σωστό όριο που ισούται με 1 είναι αυτό του κλάσματος $\frac{f(x)}{x^2}$ (ή δηλαδή $\frac{h(x)+g(x)}{x^2}$), το οποίο και θα αποδείξουμε:
\[ \lim_{x \to 0} \frac{f(x)}{x^2} = \lim_{x \to 0} \frac{\ln(x^2+1)}{x^2} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x \to 0} \frac{\frac{2x}{x^2+1}}{2x} = \lim_{x \to 0} \frac{1}{x^2+1} = \frac{1}{1+0} = 1 \]

\item Θέλουμε να αποδείξουμε ότι $\ln(x^2+1) \le x \Leftrightarrow \ln(x^2+1) - x \le 0 \Leftrightarrow h(x) \le 0$.
Η $h'(x)$ ισούται με:
\[ h'(x) = \frac{2x}{x^2+1} - 1 = \frac{2x - (x^2+1)}{x^2+1} = \frac{-x^2+2x-1}{x^2+1} = -\frac{(x-1)^2}{x^2+1} \]
Είναι προφανές ότι $(x-1)^2 \ge 0$ και $x^2+1 > 0$, επομένως $h'(x) \le 0$ για κάθε $x \in \mathbb{R}$. Η $h$ μηδενίζεται ως προς την παράγωγό της μόνο στο μεμονωμένο σημείο $x=1$. Επομένως, σε όλο το $\mathbb{R}$ η $h$ είναι γνησίως φθίνουσα.
Έτσι, για κάθε $x \ge 0 \Rightarrow h(x) \le h(0) \Rightarrow h(x) \le \ln(1)-0 \Rightarrow h(x) \le 0$.
Η ισότητα ισχύει προφανώς μόνο όταν $x = 0$.

\item Όπως αποδείξαμε στο προηγούμενο ερώτημα, $h'(x) = -\frac{(x-1)^2}{x^2+1} \le 0$ για κάθε $x \in \mathbb{R}$ με την ισότητα να ισχύει μόνο στο μεμονωμένο σημείο $x=1$. Κατά συνέπεια, η συνεχής συνάρτηση $h(x)$ είναι γνησίως φθίνουσα σε όλο το $\mathbb{R}$.

\item Αναζητούμε την πλάγια ασύμπτωτη της μορφής $y = \lambda x + \beta$ στο $+\infty$:
\[ \lambda = \lim_{x\to+\infty} \frac{h(x)}{x} = \lim_{x\to+\infty} \left( \frac{\ln(x^2+1)}{x} - 1 \right) \]
Το όριο $\lim_{x\to+\infty} \frac{\ln(x^2+1)}{x}$ υπολογίζεται με De L' Hospital:
\[ \lim_{x\to+\infty} \frac{\ln(x^2+1)}{x} \xlongequal{\text{DLH}} \lim_{x\to+\infty} \frac{\frac{2x}{x^2+1}}{1} = \lim_{x\to+\infty} \frac{2x}{x^2+1} = 0 \]
Άρα $\lambda = 0 - 1 = -1$.
Στη συνέχεια υπολογίζουμε το $\beta$:
\[ \beta = \lim_{x\to+\infty} (h(x) - \lambda x) = \lim_{x\to+\infty} (\ln(x^2+1) - x - (-x)) = \lim_{x\to+\infty} \ln(x^2+1) = +\infty \]
Επειδή το $\beta$ ειναι $+\infty$ και όχι πραγματικός αριθμός, συμπεραίνουμε πως \textbf{δεν υπάρχει} πλάγια (ή οριζόντια) ασύμπτωτη στο $+\infty$.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{\ln x}{x-1}$ για $x > 0$ και $x \ne 1$.
\begin{erwthma}
\item Για να είναι η $g(x)$ συνεχής, πρέπει να είναι συνεχής στο σημείο $x_0 = 1$ (στα υπόλοιπα σημεία είναι συνεχής ως πηλίκο συνεχών). Πρέπει δηλαδή να ισχύει:
\[ \lim_{x\to1} g(x) = g(1) \Leftrightarrow \lim_{x\to1} \frac{\ln x}{x-1} = 1 \]
Υπολογίζουμε το παραπάνω όριο χρησιμοποιώντας τον κανόνα De L' Hospital, λόγω της απροσδιοριστίας $\frac{0}{0}$:
\[ \lim_{x\to1} \frac{\ln x}{x-1} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to1} \frac{(\ln x)'}{(x-1)'} = \lim_{x\to1} \frac{1/x}{1} = 1 \]
Πράγματι, η σχέση ισχύει. Επομένως η $g(x)$ είναι συνεχής στο $\mathbb{R}^*_+$.

\item Έχουμε $h(x) = (x-1)^2 f(x) = (x-1)^2 \frac{\ln x}{x-1} = (x-1)\ln x$.
Για την εξίσωση $h(x) = 1 \Leftrightarrow h(x) - 1 = 0$, θεωρούμε τη συνάρτηση $H(x) = (x-1)\ln x - 1$ στο κλειστό $[1, e]$.
- $H(1) = (1-1)\ln 1 - 1 = 0 - 1 = -1 < 0$.
- $H(e) = (e-1)\ln e - 1 = e - 1 - 1 = e - 2 > 0$ (αφού $e \approx 2,71 > 2$).
Η $H$ είναι συνεχής στο $[1, e]$ και $H(1)H(e) < 0$. Σύμφωνα με το Θεώρημα Bolzano έχει τουλάχιστον μία ρίζα στο $(1, e)$.

\item \textbf{Κατακόρυφες ασύμπτωτες:} Ελέγχουμε τα άκρα του πεδίου ορισμού, που είναι το $x=0$ και το $x=1$ (σημείο τομής ορισμού της $f$).
- Στο $x=1$, βρήκαμε ήδη ότι $\lim_{x\to1} f(x) = 1 \in \mathbb{R}$. Οπότε δεν υπάρχει ασύμπτωτη εκεί.
- Στο $x=0$, έχουμε $\lim_{x\to0^+} \frac{\ln x}{x-1} = \frac{-\infty}{-1} = +\infty$. Άρα η ευθεία $x=0$ (ο άξονας $y'y$) είναι κατακόρυφη ασύμπτωτη.
\textbf{Οριζόντιες ασύμπτωτες:} Ελέγχουμε στο $+\infty$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{\ln x}{x-1} \xlongequal[\text{DLH}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{1/x}{1} = \lim_{x\to+\infty} \frac{1}{x} = 0 \]
Άρα ο άξονας $y=0$ (ευθεία $x'x$) είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^x - x - 1$ με $x \in \mathbb{R}$.
\begin{erwthma}
\item Υπολογίζουμε το όριο χρησιμοποιώντας τον κανόνα De L' Hospital (απροσδιοριστία $\frac{0}{0}$):
\[ \lim_{x\to0} \frac{f(x)}{x^2} = \lim_{x\to0} \frac{e^x - x - 1}{x^2} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to0} \frac{e^x - 1}{2x} \xlongequal[\text{DLH}]{\frac{0}{0}} \lim_{x\to0} \frac{e^x}{2} = \frac{1}{2} \]

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με:
\[ f'(x) = e^x - 1 \]
$\Rightarrow f'(x) = 0 \Leftrightarrow e^x = 1 \Leftrightarrow x = 0$.
Για $x > 0 \Rightarrow e^x > 1 \Rightarrow f'(x) > 0$, οπότε η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$.
Για $x < 0 \Rightarrow e^x < 1 \Rightarrow f'(x) < 0$, οπότε η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$.
Παρουσιάζει ολικό ελάχιστο στο $x=0$, την τιμή $f(0) = e^0 - 0 - 1 = 0$.
Επειδή το ολικό ελάχιστο της συνάρτησης είναι το 0, συνεπάγεται ότι για κάθε $x \in \mathbb{R}$ ισχύει $f(x) \ge f(0) \Rightarrow f(x) \ge 0$.

\item Η $f'$ είναι ξανά παραγωγίσιμη με:
\[ f''(x) = (e^x - 1)' = e^x \]
Επειδή $f''(x) = e^x > 0$ για κάθε $x \in \mathbb{R}$, η συνάρτηση $f$ είναι κυρτή στο $\mathbb{R}$. Δεν παρουσιάζει σημεία καμπής.

\item Θεωρούμε τη συνάρτηση $H(x) = f(x) - 2$ στο διάστημα $[1, 2]$.
Η $H$ είναι συνεχής στο $[1, 2]$.
Υπολογίζουμε τις τιμές της στα άκρα:
- $H(1) = f(1) - 2 = e^1 - 1 - 1 - 2 = e - 4$. Γνωρίζοντας ότι $e \approx 2,7$, $H(1) \approx -1,3 < 0$.
- $H(2) = f(2) - 2 = e^2 - 2 - 1 - 2 = e^2 - 5$. Γνωρίζοντας ότι $e^2 \approx 7,4$, $H(2) \approx 2,4 > 0$.
Επειδή $H(1)H(2) < 0$, σύμφωνα με το Θεώρημα Bolzano υπάρχει τουλάχιστον μία ρίζα $x_0 \in (1, 2)$.
Επιπροσθέτως, η συνάρτηση $H'(x) = f'(x) = e^x - 1 > 0$ για $x \in (1, 2)$. Άρα η $H$ είναι γνησίως αύξουσα στο διάστημα αυτό, που σημαίνει ότι η ρίζα είναι μοναδική.
\end{erwthma}
