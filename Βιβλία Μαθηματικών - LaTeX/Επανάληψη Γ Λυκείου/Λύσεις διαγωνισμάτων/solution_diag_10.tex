\begin{thema}{Α}
\begin{erwthma}
\item Μια συνάρτηση $f$ με πεδίο ορισμού $A$ παρουσιάζει \textbf{ολικό μέγιστο} στο $x_0\in A$, όταν
\[
f(x)\leq f(x_0)
\]
για κάθε $x\in A$. Η τιμή $f(x_0)$ λέγεται ολικό μέγιστο της $f$.

Αντίστοιχα, παρουσιάζει \textbf{ολικό ελάχιστο} στο $x_0\in A$, όταν
\[
f(x)\geq f(x_0)
\]
για κάθε $x\in A$. Η τιμή $f(x_0)$ λέγεται ολικό ελάχιστο της $f$.

\item Έστω συναρτήσεις $f,g$ παραγωγίσιμες σε ένα διάστημα γύρω από το $x_0$, εκτός ίσως από το $x_0$, με
\[
\lim_{x\to x_0}f(x)=0
\quad\text{και}\quad
\lim_{x\to x_0}g(x)=0.
\]
Αν $g'(x)\neq0$ κοντά στο $x_0$ και υπάρχει το όριο
\[
\lim_{x\to x_0}\frac{f'(x)}{g'(x)}=L,
\]
τότε υπάρχει και το όριο
\[
\lim_{x\to x_0}\frac{f(x)}{g(x)}
\]
και ισούται με $L$.

\item Αν μια συνάρτηση $f$ είναι παραγωγίσιμη σε ένα σημείο $x_0$, τότε είναι και συνεχής στο $x_0$.

Πράγματι,
\[
f(x)-f(x_0)=\frac{f(x)-f(x_0)}{x-x_0}(x-x_0),
\]
οπότε, καθώς $x\to x_0$,
\[
\lim_{x\to x_0}\bigl(f(x)-f(x_0)\bigr)=f'(x_0)\cdot0=0.
\]
Άρα
\[
\lim_{x\to x_0}f(x)=f(x_0).
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Το Θεώρημα Μέγιστης-Ελάχιστης Τιμής εφαρμόζεται για συνεχείς συναρτήσεις σε κλειστά διαστήματα $[a,\beta]$.
\item \textbf{Σωστό}. Για παράδειγμα, η συνάρτηση Dirichlet
\[
f(x)=
\begin{cases}
1,&x\in\mathbb{Q},\\
0,&x\notin\mathbb{Q},
\end{cases}
\]
δεν είναι συνεχής σε κανένα σημείο, αλλά έχει ολικό μέγιστο $1$ και ολικό ελάχιστο $0$.
\item \textbf{Λάθος}. Μια κοίλη συνάρτηση βρίσκεται κάτω από κάθε εφαπτομένη της, όμως δεν είναι υποχρεωτικό η εφαπτομένη να έχει μοναδικό κοινό σημείο με τη γραφική παράσταση. Για παράδειγμα, η $f(x)=x$ είναι κοίλη και κάθε εφαπτομένη της ταυτίζεται με τη γραφική της παράσταση.
\item \textbf{Σωστό}. Αν η $f$ είναι συνεχής και
\[
F(x)=\dint_a^x f(t)\,\d t,
\]
τότε
\[
F'(x)=f(x).
\]
\item \textbf{Λάθος}. Η γνήσια μονοτονία συνεπάγεται ότι μια συνάρτηση είναι $1-1$, αλλά το αντίστροφο δεν ισχύει γενικά σε οποιοδήποτε πεδίο ορισμού.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=x^2e^{-x},\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=2xe^{-x}-x^2e^{-x}=x(2-x)e^{-x}.
\]
Επειδή $e^{-x}>0$, το πρόσημο της $f'$ καθορίζεται από το $x(2-x)$.

Έχουμε
\[
f'(x)=0\Leftrightarrow x=0\ \text{ή}\ x=2.
\]
Στον παρακάτω πίνακα φαίνονται τα πρόσημα της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$2$,$+\infty$}%
\tkzTabLine{,-,z,+,z,-,}
\tkzTabLine{,\searrow,t,\nearrow,t,\searrow,}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$, γνησίως αύξουσα στο $[0,2]$ και γνησίως φθίνουσα στο $[2,+\infty)$.

Επομένως παρουσιάζει τοπικό ελάχιστο στο $x=0$, με
\[
f(0)=0,
\]
και τοπικό μέγιστο στο $x=2$, με
\[
f(2)=4e^{-2}=\frac{4}{e^2}.
\]

\item Έχουμε
\[
f'(x)=e^{-x}(2x-x^2).
\]
Άρα
\begin{align*}
f''(x)
&=e^{-x}(2-2x)-e^{-x}(2x-x^2)\\
&=e^{-x}(x^2-4x+2).
\end{align*}
Επειδή $e^{-x}>0$, το πρόσημο της $f''$ καθορίζεται από το τριώνυμο
\[
x^2-4x+2.
\]
Οι ρίζες του είναι
\[
x=2-\sqrt2
\quad\text{και}\quad
x=2+\sqrt2.
\]
Άρα
\[
f''(x)>0
\]
για
\[
x\in(-\infty,2-\sqrt2)\cup(2+\sqrt2,+\infty),
\]
οπότε η $f$ στρέφει τα κοίλα άνω στα διαστήματα αυτά.

Επίσης
\[
f''(x)<0
\]
για
\[
x\in(2-\sqrt2,2+\sqrt2),
\]
οπότε η $f$ στρέφει τα κοίλα κάτω στο διάστημα αυτό.

Επειδή η $f''$ αλλάζει πρόσημο στις θέσεις $2-\sqrt2$ και $2+\sqrt2$, η $C_f$ έχει σημεία καμπής:
\[
\left(2-\sqrt2,(2-\sqrt2)^2e^{-(2-\sqrt2)}\right)
\]
και
\[
\left(2+\sqrt2,(2+\sqrt2)^2e^{-(2+\sqrt2)}\right).
\]

\item Η εξίσωση
\[
x^2=\lambda e^x
\]
γράφεται ισοδύναμα
\[
x^2e^{-x}=\lambda
\Leftrightarrow
f(x)=\lambda.
\]
Από τη μονοτονία του πρώτου ερωτήματος και τα όρια
\[
\lim_{x\to-\infty}x^2e^{-x}=+\infty,
\quad
\lim_{x\to+\infty}x^2e^{-x}=0
\]
έχουμε:
\begin{itemize}
\item στο $(-\infty,0]$, η $f$ παίρνει όλες τις τιμές του $[0,+\infty)$,
\item στο $[0,2]$, η $f$ παίρνει όλες τις τιμές του $\left[0,\frac{4}{e^2}\right]$,
\item στο $[2,+\infty)$, η $f$ παίρνει όλες τις τιμές του $\left(0,\frac{4}{e^2}\right]$.
\end{itemize}

Επειδή $\lambda>0$, προκύπτει:
\begin{itemize}
\item Αν
\[
0<\lambda<\frac{4}{e^2},
\]
η εξίσωση έχει τρεις ρίζες.
\item Αν
\[
\lambda=\frac{4}{e^2},
\]
η εξίσωση έχει δύο ρίζες.
\item Αν
\[
\lambda>\frac{4}{e^2},
\]
η εξίσωση έχει μία ρίζα.
\end{itemize}

\item Έχουμε
\[
\frac{f(x)}{x}=\frac{x^2e^{-x}}{x}=xe^{-x}.
\]
Καθώς $x\to-\infty$, ισχύει
\[
x\to-\infty
\quad\text{και}\quad
e^{-x}\to+\infty.
\]
Άρα
\[
\lim_{x\to-\infty}xe^{-x}=-\infty.
\]
Επομένως
\[
{\lim_{x\to-\infty}\frac{f(x)}{x}=-\infty}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Η κάψουλα κινείται με εξισώσεις θέσης
\[
X(t)=t^2+1,\quad Y(t)=3t.
\]
\begin{erwthma}
\item Από τη σχέση
\[
Y=3t
\]
έχουμε
\[
t=\frac{Y}{3}.
\]
Άρα
\[
X=t^2+1=\frac{Y^2}{9}+1.
\]
Επομένως
\[
Y^2=9(X-1).
\]
Με τις συνήθεις συντεταγμένες $(x,y)$, η τροχιά έχει εξίσωση
\[
{y^2=9(x-1)}.
\]

\item Από την εξίσωση της τροχιάς έχουμε
\[
x=\frac{y^2}{9}+1.
\]
Παραγωγίζοντας ως προς τον χρόνο $t$, παίρνουμε
\[
\frac{\d x}{\d t}=\frac{2y}{9}\frac{\d y}{\d t}.
\]
Τη στιγμή που
\[
y=6
\quad\text{και}\quad
\frac{\d y}{\d t}=0{,}5=\frac12,
\]
έχουμε
\[
\frac{\d x}{\d t}
=\frac{2\cdot6}{9}\cdot\frac12
=\frac{2}{3}.
\]
Άρα η τετμημένη αυξάνεται με ρυθμό
\[
{\frac{2}{3}\ \si{m}/\sec}.
\]

\item Στον άνω κλάδο της παραβολής έχουμε
\[
y=3\sqrt{x-1},\quad x\geq1.
\]
Αν $d$ είναι η απόσταση ενός σημείου $(x,y)$ από την αρχή των αξόνων, τότε
\[
d^2=x^2+y^2=x^2+9(x-1).
\]
Θεωρούμε
\[
\Phi(x)=d^2=x^2+9x-9,\quad x\geq1.
\]
Τότε
\[
\Phi'(x)=2x+9>0
\]
για κάθε $x\geq1$. Άρα η $\Phi$ είναι γνησίως αύξουσα στο $[1,+\infty)$, οπότε η απόσταση ελαχιστοποιείται για
\[
x=1.
\]
Τότε
\[
y=3\sqrt{1-1}=0.
\]
Άρα το ζητούμενο σημείο είναι
\[
{A(1,0)}.
\]

\item Η καμπύλη γράφεται
\[
x=\frac{y^2}{9}+1.
\]
Για $0\leq y\leq3$ βρίσκεται δεξιά από τον άξονα $y'y$, δηλαδή από την ευθεία $x=0$.

Άρα το ζητούμενο εμβαδόν είναι
\begin{align*}
E
&=\dint_0^3\left(\frac{y^2}{9}+1\right)\,\d y\\
&=\left[\frac{y^3}{27}+y\right]_0^3\\
&=1+3\\
&=4.
\end{align*}
Επομένως
\[
{E=4}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται συνεχώς παραγωγίσιμη συνάρτηση
\[
f:(0,+\infty)\to\mathbb{R}
\]
με
\[
f(1)=0
\]
και
\[
xf'(x)-f(x)=x^2
\]
για κάθε $x>0$.
\begin{erwthma}
\item Επειδή $x>0$, διαιρούμε με $x^2$:
\[
\frac{xf'(x)-f(x)}{x^2}=1.
\]
Όμως
\[
\left(\frac{f(x)}{x}\right)'
=\frac{xf'(x)-f(x)}{x^2}.
\]
Άρα
\[
\left(\frac{f(x)}{x}\right)'=1.
\]
Επομένως
\[
\frac{f(x)}{x}=x+c.
\]
Από τη συνθήκη $f(1)=0$ έχουμε
\[
0=1+c
\]
και άρα
\[
c=-1.
\]
Συνεπώς
\[
\frac{f(x)}{x}=x-1
\]
και
\[
{f(x)=x^2-x}.
\]

\item Για
\[
f(x)=x^2-x
\]
έχουμε
\[
f'(x)=2x-1.
\]
Η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $a>0$ είναι
\[
y-f(a)=f'(a)(x-a).
\]
Δηλαδή
\[
y-(a^2-a)=(2a-1)(x-a).
\]
Για να διέρχεται από το $P(0,-1)$, πρέπει
\[
-1-(a^2-a)=(2a-1)(0-a).
\]
Άρα
\[
-1-a^2+a=-2a^2+a
\]
και επομένως
\[
a^2=1.
\]
Επειδή $a>0$, έχουμε
\[
a=1.
\]
Για $a=1$ η εφαπτομένη είναι
\[
y-f(1)=f'(1)(x-1)
\]
δηλαδή
\[
y=x-1.
\]
Πράγματι, για $x=0$ δίνει $y=-1$. Άρα υπάρχει μοναδική τέτοια εφαπτομένη:
\[
{y=x-1}.
\]

\item Στο διάστημα $[1,2]$ έχουμε
\[
f(x)=x^2-x=x(x-1)\geq0.
\]
Άρα το εμβαδόν είναι
\begin{align*}
E
&=\dint_1^2(x^2-x)\,\d x\\
&=\left[\frac{x^3}{3}-\frac{x^2}{2}\right]_1^2\\
&=\left(\frac83-2\right)-\left(\frac13-\frac12\right)\\
&=\frac23+\frac16\\
&=\frac56.
\end{align*}
Επομένως
\[
{E=\frac56}.
\]

\item Για $x>1$ έχουμε $x-1>0$, οπότε η εξίσωση
\[
(x^2-1)\ln x=x-1
\]
γράφεται
\[
(x-1)(x+1)\ln x=x-1.
\]
Διαιρώντας με $x-1$, παίρνουμε
\[
(x+1)\ln x=1.
\]
Θεωρούμε τη συνάρτηση
\[
K(x)=(x+1)\ln x-1,\quad x>1.
\]
Η $K$ είναι συνεχής στο $(1,+\infty)$ και
\[
K'(x)=\ln x+\frac{x+1}{x}
=\ln x+1+\frac1x.
\]
Για $x>1$ έχουμε
\[
K'(x)>0.
\]
Άρα η $K$ είναι γνησίως αύξουσα στο $(1,+\infty)$, οπότε έχει το πολύ μία ρίζα.

Επιπλέον,
\[
\lim_{x\to1^+}K(x)=-1<0,
\]
ενώ
\[
K(e)=(e+1)\ln e-1=e>0.
\]
Άρα, από το θεώρημα Bolzano, η $K$ έχει μία ρίζα στο $(1,e)$. Επειδή είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

Επομένως η εξίσωση έχει
\[
{\text{μοναδική ρίζα στο }(1,+\infty)}.
\]
\end{erwthma}
\end{thema}
