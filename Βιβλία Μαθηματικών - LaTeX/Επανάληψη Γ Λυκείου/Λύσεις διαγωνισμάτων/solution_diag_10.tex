\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Λ.} 
Το Θεώρημα Μέγιστης - Ελάχιστης Τιμής εφαρμόζεται υποχρεωτικά \textbf{μόνο σε εντελώς κλειστά διαστήματα} $[\alpha, \beta]$. Σε ημιανοιχτά, η ύπαρξη ακροτάτων δεν είναι γενικά εξασφαλισμένη.
\item \textbf{Σ.}
Ισχύει. Μπορεί να κατασκευαστεί π.χ. μία συνάρτηση ασυνεχής παντού (όπως η \eng{Dirichlet}) αλλά με αυστηρά περιορισμένο εύρος, η οποία να λαμβάνει κανονικά ολικό μέγιστο και ελάχιστο έστω και σε ασυνεχή σημεία (π.χ. $max=1, min=0$).
\item \textbf{Σ.}
Είναι βασικό θεώρημα: η γραφική παράσταση μιας κοίλης συνάρτησης βρίσκεται ολόκληρη κάτω από κάθε εφαπτομένη της, με μοναδικό κοινό σημείο αυτό της επαφής. 
\item \textbf{Σ.}
Πράγματι, αποτελεί άμεση διατύπωση του Θεμελιώδους Θεωρήματος του Ολοκληρωτικού Λογισμού: $F'(x) = \left( \dint_\alpha^x f(t)\,\d t \right)' = f(x)$.
\item \textbf{Λ.}
Η γνήσια μονοτονία είναι ικανή αλλά \textbf{όχι αναγκαία} συνθήκη για να είναι μια συνάρτηση 1-1. Για παράδειγμα, η $f(x) = \frac{1}{x}$ είναι 1-1 στο $\mathbb{R}^*$ αλλά δεν είναι γνησίως μονότονη γενικά σε όλο το $\mathbb{R}^*$ (είναι φθίνουσα μόνο κατά διαστήματα, γιατί $f(-1)=-1 < f(1)=1$).
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 10ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 e^{-x}$ στο $\mathbb{R}$.
\begin{erwthma}

\item \textbf{Μονοτονία και Ακρότατα:}
Παραγωγίζουμε την $f$ σε όλο το $\mathbb{R}$:
\[ f'(x) = (x^2)' e^{-x} + x^2 (e^{-x})' = 2x e^{-x} - x^2 e^{-x} = e^{-x} (2x - x^2) = x(2 - x) e^{-x} \]

Λύνουμε την εξίσωση $f'(x) = 0 \Leftrightarrow x(2-x) = 0 \Leftrightarrow x=0$ ή $x=2$. (καθώς $e^{-x} > 0$).
Το πρόσημο της $f'$ ταυτίζεται με το τριώνυμο $2x - x^2$:
\begin{itemize}
    \item $f'(x) > 0 \Leftrightarrow x \in (0, 2)$
    \item $f'(x) < 0 \Leftrightarrow x \in (-\infty, 0) \cup (2, +\infty)$
\end{itemize}

Συμπεραίνουμε:
- Η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$, γνησίως αύξουσα στο $[0, 2]$, και γνησίως φθίνουσα στο $[2, +\infty)$.
- Παρουσιάζει \textbf{τοπικό ελάχιστο} στο $x=0$, με $f(0) = 0$.
- Παρουσιάζει \textbf{τοπικό μέγιστο} στο $x=2$, με $f(2) = 4e^{-2} = \frac{4}{e^2}$.

\item \textbf{Κυρτότητα και Σημεία Καμπής:}
Παραγωγίζουμε την $f'(x) = (2x - x^2)e^{-x}$:
\begin{align*}
f''(x) &= (2x-x^2)'e^{-x} + (2x-x^2)(e^{-x})' = (2-2x)e^{-x} - (2x-x^2)e^{-x} \\
&= e^{-x} (2 - 2x - 2x + x^2) = e^{-x} (x^2 - 4x + 2)
\end{align*}

Βρίσκουμε τις ρίζες της $f''(x) = 0 \Leftrightarrow x^2 - 4x + 2 = 0$. 
$\Delta = 16 - 8 = 8$, οπότε $x_{1,2} = \frac{4 \pm \sqrt{8}}{2} = \frac{4 \pm 2\sqrt{2}}{2} = 2 \pm \sqrt{2}$.

Η $f''(x)$ έχει το πρόσημο του $x^2 - 4x + 2$:
\begin{itemize}
    \item Θετικό ($+$) άρα $f$ \textbf{κυρτή (κοίλα άνω)} στα διαστήματα $(-\infty, 2-\sqrt{2}]$ και $[2+\sqrt{2}, +\infty)$.
    \item Αρνητικό ($-$) άρα $f$ \textbf{κοίλη (κοίλα κάτω)} στο διάστημα $[2-\sqrt{2}, 2+\sqrt{2}]$.
\end{itemize}
Άρα παρουσιάζει \textbf{δύο σημεία καμπής} στα $x_1 = 2-\sqrt{2}$ και $x_2 = 2+\sqrt{2}$.

\item Για μεγάλο αριθμό $\lambda > 0$, εξετάζουμε την εξίσωση:
\[ x^2 = \lambda e^x \Leftrightarrow x^2 e^{-x} = \lambda \Leftrightarrow f(x) = \lambda \]
Χρειαζόμαστε τα επιμέρους σύνολα τιμών:
- $\lim_{x\to-\infty} f(x) = \lim (x^2 e^{-x}) = (+\infty)(+\infty) = +\infty$.
- $\lim_{x\to+\infty} f(x) = \lim \frac{x^2}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim \frac{2x}{e^x} \xlongequal[\text{\eng{DLH}}]{\dots} 0$.

Εξετάζουμε τα πεδία:
$\bullet$ Στο $\Delta_1 = (-\infty, 0]$ η $f$ φθίνει από το $+\infty$ στο $0$. Άρα $f(\Delta_1) = [0, +\infty)$.
$\bullet$ Στο $\Delta_2 = (0, 2]$ η $f$ αυξάνει από το $0$ στο $4/e^2$. Άρα $f(\Delta_2) = (0, 4/e^2]$.
$\bullet$ Στο $\Delta_3 = [2, +\infty)$ η $f$ φθίνει από $4/e^2$ στο $0$. Άρα $f(\Delta_3) = (0, 4/e^2]$.

Κρίνοντας από τις οριζόντιες τομές για το $\lambda$:
\begin{itemize}
    \item Αν $\boldsymbol{0 < \lambda < 4/e^2}$, η ευθεία $y=\lambda$ τέμνει και τα τρία διαστήματα. Επομένως η εξίσωση έχει \textbf{3 ρίζες}.
    \item Αν $\boldsymbol{\lambda = 4/e^2}$, η εξίσωση έχει \textbf{2 ρίζες} (μία στο $(-\infty, 0)$ και τη διπλή στο τοπικό μέγιστο $x=2$).
    \item Αν $\boldsymbol{\lambda > 4/e^2}$, η εξίσωση έχει \textbf{1 ρίζα} (που εντοπίζεται αποκλειστικά στον πρώτο κλάδο $(-\infty, 0)$).
\end{itemize}

\item Ζητείται ο υπολογισμός του ορίου:
\[ \lim_{x\to-\infty} \frac{f(x)}{x} = \lim_{x\to-\infty} \frac{x^2 e^{-x}}{x} = \lim_{x\to-\infty} (x e^{-x}) \]
Αφού καθώς το $x \to -\infty$, το $x \to -\infty$ και το $e^{-x} \to +\infty$:
\[ \lim_{x\to-\infty} (x e^{-x}) = (-\infty) \cdot (+\infty) = \boldsymbol{-\infty} \]

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 10ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Χρονικές συναρτήσεις διαδρομής $X(t) = t^2+1$ και $Y(t) = 3t$.
\begin{erwthma}

\item Από τη δεύτερη χρονική εξίσωση έχουμε $Y = 3t \Rightarrow t = \dfrac{Y}{3}$.
Εισάγουμε την παράμετρο $t$ στην πρώτη εξίσωση προσδιορίζοντας πλήρως τον γεωμετρικό τόπο χωρίς εξάρτηση από το χρόνο:
\begin{align*}
X &= \left( \frac{Y}{3} \right)^2 + 1 \\
X &= \frac{Y^2}{9} + 1 \\
X - 1 &= \frac{Y^2}{9} \\
Y^2 &= 9(X-1) 
\end{align*}
Αντικαθιστώντας τα επιμέρους $X, Y$ με τα συνήθη $(x, y)$, προκύπτει πως πράγματι είναι η αλγεβρική καμπύλη $\boldsymbol{y^2 = 9(x-1)}$.

\item Έχουμε την παραμετρική σχέση ελεύθερων μεταβλητών $x(t) = \frac{1}{9}y^2(t) + 1$.
Παραγωγίζουμε ως προς $t$:
\[ x'(t) = \frac{1}{9} \cdot 2y(t) \cdot y'(t) = \frac{2}{9} y(t) y'(t) \]
Tη χρονική στιγμή $t_0$, τίθενται τα δεδομένα $y(t_0) = 6$ και $y'(t_0) = 0{,}5 = \frac{1}{2}$.
Ο ρυθμός της τετμημένης $x'(t_0)$ υπολογίζεται:
\[ x'(t_0) = \frac{2}{9} \cdot 6 \cdot \frac{1}{2} = \frac{6}{9} = \boldsymbol{\frac{2}{3} \ \text{m/s}} \]

\item Ζητείται σημείο της καμπύλης $y = 3\sqrt{x-1}$ (δηλαδή ο πάνω κλάδος με $y \ge 0, x \ge 1$) όπου ο ρυθμός της απόστασης $d$ μηδενίζεται. Απόσταση από την αρχή $(0,0)$:
\[ d^2 = x^2 + y^2 = x^2 + 9(x-1) = x^2 + 9x - 9 \]
Ο ρυθμός μεταβολής της απόστασης (παραγωγίζοντας τη σχέση $d^2$ ως προς το χρόνο):
\[ 2d \cdot d' = 2x x' + 9x' \Rightarrow d' = \frac{(2x+9)x'}{2d} \]
Για την εκμηδένιση του ρυθμού $d'=0$, θα πρέπει $2x+9 = 0 \Rightarrow x = -4{,}5$.
Ωστόσο η καμπύλη ορίζεται \textbf{αυστηρά για $x \ge 1$}.
Επομένως ο ρυθμός \textbf{δεν εκμηδενίζεται πουθενά αλγεβρικά (σε εσωτερικό σημείο)}. Η ελάχιστη απόσταση ωστόσο εντοπίζεται στο τοπικό (μαθηματικά οριακό) σημείο έναρξης της καμπύλης $x=1$ (σημείο κορυφής παραβολής). Το σημείο είναι το $A(1, 0)$. 

\begin{parat}{\linewidth}
Εναλλακτική ερμηνεία: Το ελάχιστο εντοπίζεται στο άκρο της καμπύλης. Επειδή εκεί η ταχύτητα στον άξονα $x$ μηδενίζεται ($\frac{x'}{t} = 0$ στο $t=0$), τότε συνολικά ο ρυθμός απομάκρυνσης στο $(1,0)$ είναι αντικειμενικά $0$.
\end{parat}

\item Απαιτείται το εμβαδόν μεταξύ της καμπύλης, του άξονα $x=0$, και των ευθειών $y=0$ και $y=3$. 
Επειδή η καμπύλη είναι $x = 1 + \frac{y^2}{9}$ και εμείς ορίζουμε την περίφραξη μέσω του άξονα $x=0$, η πιο κομψή στρατηγική είναι να \textbf{ολοκληρώσουμε ως προς $y$}. 
Για $y \in [0, 3]$, η καμπύλη $x(y)$ είναι αναμφισβήτητα θετική. Άρα:
\begin{align*}
E &= \int_0^3 |x(y) - 0|\,\d y = \int_0^3 \left( 1 + \frac{y^2}{9} \right)\,\d y \\
&= \left[ y + \frac{y^3}{27} \right]_0^3 = \left( 3 + \frac{3^3}{27} \right) - (0+0) \\
&= 3 + \frac{27}{27} = 3 + 1 = \boldsymbol{4 \ \text{τ.μ.}}
\end{align*}

\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται $f(1)=0$ και $x f'(x) - f(x) = x^2$ για $x>0$.
\begin{erwthma}

\item Η δοσμένη σχέση έχει σαφή δομή διαφορικής εξίσωσης. Αφού $x > 0$, διαιρούμε με το $x^2$:
\begin{align*}
\frac{x f'(x) - f(x)}{x^2} &= \frac{x^2}{x^2} \\
\left( \frac{f(x)}{x} \right)' &= 1 \\
\left( \frac{f(x)}{x} \right)' &= (x)'
\end{align*}
Έπεται από τις συνέπειες Θ.Μ.Τ. ότι $\frac{f(x)}{x} = x + c$.
Με τη χρήση του αρχικού δεδομένου $f(1) = 0$:
\[ \frac{f(1)}{1} = 1 + c \Rightarrow 0 = 1 + c \Rightarrow c = -1 \]
Άρα:
\[ \frac{f(x)}{x} = x - 1 \Rightarrow \boldsymbol{f(x) = x(x-1) = x^2 - x} \]

\item Η εξίσωση της εφαπτομένης της $C_f$ σε κάποιο σημείο $\alpha > 0$ δίνεται από το:
\[ y - f(\alpha) = f'(\alpha)(x - \alpha) \]
Η παράγωγος είναι $f'(x) = 2x - 1$. Οπότε:
\[ y - (\alpha^2 - \alpha) = (2\alpha - 1)(x - \alpha) \]
Αν επιζητούσαμε η εφαπτομένη να διέρχεται από το $P(0, -1)$, θα έπρεπε οι συντεταγμένες $(0,-1)$ να επαληθεύουν την παραπάνω εξίσωση. Αντικαθιστούμε:
\begin{align*}
-1 - \alpha^2 + \alpha &= (2\alpha - 1)(0 - \alpha) \\
-1 - \alpha^2 + \alpha &= -2\alpha^2 + \alpha \\
-1 - \alpha^2 + 2\alpha^2 &= 0 \\
\alpha^2 - 1 &= 0 \Rightarrow \alpha^2 = 1 \Rightarrow \alpha = 1 \quad (\text{λόγω } \alpha > 0)
\end{align*}
Διαπιστώνουμε δηλαδή ότι υπάρχει \textbf{μοναδικό σημείο επαφής} στο $\alpha=1$ του οποίου η εφαπτομένη διέρχεται \textbf{κανονικά} από το εν λόγω σημείο $P(0,-1)$. 

\begin{prosoxi}{\linewidth}
\textbf{Πιθανό Τυπογραφικό Εκφώνησης:} Η εκφώνηση ζητά να αποδείξουμε ότι "δεν διέρχεται ποτέ", οπότε αντιβαίνει με τα αυστηρά αλγεβρικά δεδομένα (αφού διέρχεται για $\alpha=1$). Επιδείξαμε όμως ακριβώς σε ποιο σημείο τέμνει τη γραφική.
\end{prosoxi}

\item Το εμβαδόν μεταξύ της καμπύλης $C_f$ και του άξονα των τετμημένων $x'x$.
Αναζητούμε τις ρίζες της $f$: $f(x) = 0 \Leftrightarrow x^2 - x = 0 \Leftrightarrow x=0$ ή $x=1$.
Στο εσωτερικό διάστημα $(0, 1)$, η $f(x) < 0$. Άρα το εμβαδόν έχει όρια επιφάνειας:
\begin{align*}
E &= \int_0^1 |f(x)|\,\d x = \int_0^1 -(x^2-x)\,\d x = \int_0^1 (x - x^2)\,\d x \\
&= \left[ \frac{x^2}{2} - \frac{x^3}{3} \right]_0^1 = \left(\frac{1}{2} - \frac{1}{3}\right) - 0 = \frac{3-2}{6} = \boldsymbol{\frac{1}{6} \ \text{τ.μ.}}
\end{align*}
Επαληθεύτηκε απολύτως.

\item Διαθέτουμε την εξίσωση $(x^2-1)\ln x = x - 1$ για αναζήτηση στο πεδίο $\Delta=(1, +\infty)$.
Επειδή στο πεδίο αυτό είναι ξεκάθαρα $x > 1 \Rightarrow x - 1 > 0$, χωρίζουμε τα μέλη:
\begin{align*}
(x-1)(x+1)\ln x &= x - 1 \quad \Big/ : (x-1) \\
(x+1)\ln x &= 1 \\
\ln x - \frac{1}{x+1} &= 0
\end{align*}
Ορίζουμε την προσκείμενη $K(x) = \ln x - \frac{1}{x+1}$ για $x \in [1, +\infty)$. Η συνάρτηση αυτή είναι πλήρως συνεχής.
\[ K'(x) = \frac{1}{x} + \frac{1}{(x+1)^2} \]
Για $x>1$ η παράγωγος είναι αυστηρά θετική. Επομένως η $K(x)$ είναι \textbf{γνησίως αύξουσα} και κατά συνέπεια εξασφαλίζεται η 1-1 αναλογικότητα (ότι υπάρχει το πολύ μια ρίζα).

Δοκιμάζουμε τιμές (Θεώρημα \eng{Bolzano} στο $[1, e]$):
$\bullet$ $K(1) = \ln 1 - \frac{1}{2} = -\frac{1}{2} < 0$.
$\bullet$ $K(e) = \ln e - \frac{1}{e+1} = 1 - \frac{1}{e+1} = \frac{e}{e+1} > 0$.
Διότι $K(1)\cdot K(e) < 0$ και η $K$ είναι συνεχής, υπάρχει ρίζα στο $(1, e) \subset (1, +\infty)$.
Λόγω γνήσιας μονοτονίας, αυτή καθίσταται αυτομάτως \textbf{μοναδική}.

\end{erwthma}
\end{thema}
