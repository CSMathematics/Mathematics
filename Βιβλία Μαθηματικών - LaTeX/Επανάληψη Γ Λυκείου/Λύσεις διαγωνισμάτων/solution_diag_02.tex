%=========== ΛΥΣΗ - 2ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}

\item \textbf{Σ.}
Αυτή είναι η ιδιότητα \textbf{διατήρησης προσήμου} συνεχούς συνάρτησης. Αν η $f$ συνεχής στο $x_0$ και $f(x_0)<0$, τότε υπάρχει περιοχή $(x_0-\delta,x_0+\delta)$ στην οποία $f(x)<0$.

\begin{parat}{\linewidth}
Αυτό ισχύει γιατί η συνέχεια εγγυάται ότι η $f$ «μένει κοντά» στο $f(x_0)$. Αν $f(x_0)<0$ τότε διαλέγοντας $\varepsilon = |f(x_0)|/2$ βρίσκουμε $\delta$ ώστε $|f(x)-f(x_0)|<\varepsilon$ δηλαδή η $f(x)$ παραμένει αρνητική.
\end{parat}

\item \textbf{Σ.}
Από το Θεμελιώδες Θεώρημα (ΘΘΟΛ):
\[ \int_\alpha^\beta f'(x)\,\d x = f(\beta)-f(\alpha) \stackrel{f(\alpha)=f(\beta)}{=} 0 \]

\begin{prosoxi}{\linewidth}
Δεν μας ενδιαφέρει το πρόσημο της $f'$! Η ιδιότητα $\dint_\alpha^\beta f'(x)\d x = f(\beta)-f(\alpha)$ ισχύει πάντα για συνεχή $f'$.
\end{prosoxi}

\item \textbf{Σ.}
Κλασικό παράδειγμα: $f(x)=\dfrac{1}{x}$ στο $\mathbb{R}^*$.

Η $f$ είναι $1$-$1$: αν $f(x_1) = f(x_2)$ τότε $x_1=x_2$.

Η $f$ \textbf{δεν} είναι γνησίως μονότονη στο $\mathbb{R}^*$: είναι φθίνουσα σε κάθε κλάδο, αλλά $f(-1)<f(1)$ ενώ $-1<1$, άρα δεν είναι ούτε γνησίως αύξουσα ούτε φθίνουσα σε ολόκληρο το $D_f$.

\begin{parat}{\linewidth}
Ισχύει: γν. μονότονη $\Rightarrow$ $1$-$1$. Το αντίστροφο \textbf{δεν ισχύει}. Η $1$-$1$ εξασφαλίζεται και σε ένωση μη συνδεδεμένων διαστημάτων.
\end{parat}

\item \textbf{Λ.}
Μια κατακόρυφη ευθεία $x=a$ τέμνει τη $C_f$ \textbf{μόνο αν} $a\in D_f$.

Αντιπαράδειγμα: η $f(x)=\frac{1}{x}$ — η ευθεία $x=0$ \textbf{δεν} τέμνει τη $C_f$.

\begin{prosoxi}{\linewidth}
Μην μπερδεύετε: «κάθε κατακόρυφη τέμνει τη $C_f$ σε \textbf{το πολύ} ένα σημείο» — αυτό ΝΑΙ, είναι ο ορισμός συνάρτησης. «Τουλάχιστον ένα» ισχύει μόνο αν $D_f=\mathbb{R}$.
\end{prosoxi}

\item \textbf{Σ.}
Σημείο καμπής εμφανίζεται μόνο όπου η $f''$ αλλάζει πρόσημο. Αφού $f''(x)>0$ $\forall\, x\in\mathbb{R}$, η $f''$ δεν αλλάζει \textbf{ποτέ} πρόσημο, άρα δεν υπάρχει σημείο καμπής.

\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 2ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνονται $f(x)=\dfrac{e^x}{e^x+1}$ και $g(x)=\ln x$.
\begin{erwthma}

\item Η $h(x) = (f\circ g)(x) = f(\ln x)$. Πεδίο ορισμού: $x\in D_g = (0,+\infty)$.
\[ h(x) = \frac{e^{\ln x}}{e^{\ln x}+1} = \frac{x}{x+1} \]

Αφού $x>0$, ο παρονομαστής $x+1>1>0$, άρα $D_h = (0,+\infty)$ και $\boxed{h(x)=\dfrac{x}{x+1}}$.

\item \underline{Η $h$ αντιστρέφεται:}
\[ h'(x) = \frac{(x+1)-x}{(x+1)^2} = \frac{1}{(x+1)^2}>0\quad\forall\, x>0 \]
Η $h$ γνησίως αύξουσα $\Rightarrow$ $1$-$1$ $\Rightarrow$ αντιστρέψιμη.

\underline{Εύρεση $h^{-1}$:} Θέτω $y = \frac{x}{x+1}$ και λύνω ως προς $x$:
\[ y(x+1) = x \Rightarrow yx + y = x \Rightarrow y = x(1-y) \Rightarrow x = \frac{y}{1-y} \]

Σύνολο τιμών: $h(x) = 1-\frac{1}{x+1}$ με $x>0 \Rightarrow x+1>1 \Rightarrow \frac{1}{x+1}\in(0,1) \Rightarrow h(x)\in(0,1)$.

Άρα $\boxed{h^{-1}(x) = \dfrac{x}{1-x},\quad x\in(0,1)}$.

\begin{parat}{\linewidth}
Ο τύπος $h^{-1}$ βγαίνει αλλάζοντας $x\leftrightarrow y$ στη σχέση $y = \frac{x}{x+1}$. Πάντα ελέγχετε το πεδίο ορισμού της $h^{-1}$ (= σύνολο τιμών της $h$).
\end{parat}

\item Τα ζητούμενα όρια:
\[ \lim_{x\to 0^+} h(x) = \lim_{x\to 0^+}\frac{x}{x+1} = \frac{0}{1} = 0 \]
\[ \lim_{x\to+\infty} h(x) = \lim_{x\to+\infty}\frac{x}{x+1} = \lim_{x\to+\infty}\frac{1}{1+\frac{1}{x}} = 1 \]

\item Θέτουμε $\varphi(x) = h(x)+e^{-x}-1 = \frac{x}{x+1}+e^{-x}-1$ στο $[1,e]$.

Η $\varphi$ είναι \textbf{συνεχής} στο $[1,e]$ ως άθροισμα συνεχών.

Υπολογίζουμε στα άκρα:
\[ \varphi(1) = \frac{1}{2}+e^{-1}-1 = \frac{1}{2}+\frac{1}{e}-1 = \frac{e+2-2e}{2e} = \frac{2-e}{2e} < 0 \quad (\text{αφού }e>2) \]

Γράφουμε $\varphi(x) = \frac{x}{x+1}-1+e^{-x} = -\frac{1}{x+1}+e^{-x}$, οπότε:
\[ \varphi(x) = e^{-x}-\frac{1}{x+1} \]

$\varphi(1) = \frac{1}{e}-\frac{1}{2} \approx 0{,}368-0{,}5 < 0$.

$\varphi'(x) = -e^{-x}+\frac{1}{(x+1)^2}$.

$\varphi'(1) = -e^{-1}+\frac{1}{4}\approx -0{,}368+0{,}25<0$.
$\varphi'(2) = -e^{-2}+\frac{1}{9}\approx -0{,}135+0{,}111<0$.
$\varphi'(3) = -e^{-3}+\frac{1}{16}\approx -0{,}050+0{,}063>0$.

Άρα η $\varphi$ σε κάποιο σημείο μεταξύ $2$ και $3$ αρχίζει να αυξάνεται.

$\varphi(e) = e^{-e}-\frac{1}{e+1}\approx 0{,}066 - 0{,}269 < 0$.

Και τα δύο άκρα αρνητικά, άρα η $\varphi$ δεν αλλάζει πρόσημο στο $[1,e]$!

\begin{prosoxi}{\linewidth}
Η εξίσωση $h(x)+e^{-x}=1$ ισοδυναμεί με $e^{-x}=\frac{1}{x+1}$. Η μοναδική ρίζα είναι $x=0$, εκτός $(1,e)$. Σε εξεταστικό πλαίσιο η μεθοδολογία παραμένει η ίδια: δείχνουμε \textbf{αλλαγή πρόσήμου} + \textbf{συνέχεια} $\Rightarrow$ \eng{Bolzano}.
\end{prosoxi}

\textbf{Μεθοδολογία \eng{Bolzano}:}
Θέτουμε $\varphi(x) = h(x)+e^{-x}-1$, $x\in[1,e]$. Η $\varphi$ συνεχής στο $[1,e]$. Αν βρεθούν $\varphi(a)\cdot\varphi(b)<0$, τότε υπάρχει ρίζα στο $(a,b)\subseteq(1,e)$.

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 2ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Δίνεται $f:(0,+\infty)\to\mathbb{R}$ συνεχής, $f(e) = \frac{2}{e}$, και
\[ x^2 f^2(x) - 2x\ln x\cdot f(x) + \ln^2 x - 1 = 0 \]
\begin{erwthma}

\item Η σχέση είναι \textbf{τριώνυμο ως προς $xf(x)$}. Θέτουμε $u = xf(x)$:
\[ u^2 - 2\ln x\cdot u + \ln^2 x - 1 = 0 \]
Διακρίνουσα: $\Delta = 4\ln^2 x - 4(\ln^2 x - 1) = 4 > 0$.

Ρίζες:
\[ u = \frac{2\ln x \pm 2}{2} = \ln x \pm 1 \]

Δηλαδή: $xf(x) = \ln x + 1$ ή $xf(x) = \ln x - 1$.

Άρα: $f(x) = \frac{\ln x + 1}{x}$ ή $f(x) = \frac{\ln x - 1}{x}$.

Ελέγχουμε $f(e) = \frac{2}{e}$:
\begin{itemize}
\item $\frac{\ln e + 1}{e} = \frac{2}{e}$ \quad \textcolor{check}{\faCheck}
\item $\frac{\ln e - 1}{e} = \frac{0}{e} = 0 \ne \frac{2}{e}$ \quad \textcolor{error}{\faTimes}
\end{itemize}

Αφού $f$ συνεχής, η επιλογή είναι σταθερή:
\[ \boxed{f(x) = \frac{\ln x + 1}{x}} \]

\begin{parat}{\linewidth}
Η τεχνική «τριώνυμο ως προς $xf(x)$» εμφανίζεται πολύ συχνά στα Θέματα Γ. Πάντα ελέγχετε \textbf{και τις δύο ρίζες} και αποκλείστε τη μία με αρχική τιμή.
\end{parat}

\item Παράγωγος:
\[ f'(x) = \frac{\frac{1}{x}\cdot x - (\ln x+1)}{x^2} = \frac{1-\ln x-1}{x^2} = \frac{-\ln x}{x^2} \]

Πρόσημο:
\begin{itemize}
\item $f'(x) = 0 \Leftrightarrow \ln x = 0 \Leftrightarrow x = 1$
\item $f'(x)>0 \Leftrightarrow -\ln x > 0 \Leftrightarrow \ln x < 0 \Leftrightarrow 0<x<1$
\item $f'(x)<0 \Leftrightarrow x>1$
\end{itemize}

\item Οριζόντια ασύμπτωτη στο $+\infty$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty}\frac{\ln x + 1}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty}\frac{\frac{1}{x}}{1} = 0 \]

Άρα η ευθεία $\boxed{y=0}$ (ο άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} στο $+\infty$.

\item Έχουμε $E(t) = x(t)\cdot y(t) = x\cdot f(x) = \ln x + 1$.

Παραγωγίζοντας ως προς $t$:
\[ E'(t) = \frac{\d}{\d t}(\ln x + 1) = \frac{1}{x}\cdot x'(t) \]

Στη στιγμή $t_0$: $x(t_0)=e$ και $x'(t_0)=2e^2$.
\[ E'(t_0) = \frac{1}{e}\cdot 2e^2 = 2e \]

Άρα ο ρυθμός μεταβολής του εμβαδού είναι $\boxed{E'(t_0) = 2e\ \text{τ.μ./sec}}$.

\begin{parat}{\linewidth}
Το κλειδί ήταν ότι $E = xf(x) = \ln x + 1$ (απλοποίηση!). Πάντα απλοποιήστε πριν παραγωγίσετε.
\end{parat}

%=========== ΛΥΣΗ - 2ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Δίνεται $f:(0,+\infty)\to\mathbb{R}$ παραγωγίσιμη:
\[ f(x) + e^{f(x)} = e^{x-1} - \ln x + \frac{e^{e^{x-1}}}{x} \]
\begin{erwthma}

\item Θέτουμε $\varphi(x) = e^{x-1}-\ln x$. Τότε:
\[ \varphi(x) + e^{\varphi(x)} = e^{x-1}-\ln x + e^{e^{x-1}-\ln x} =  e^{x-1}-\ln x + \frac{e^{e^{x-1}}}{e^{\ln x}} = e^{x-1}-\ln x + \frac{e^{e^{x-1}}}{x} \]

Αυτό ταυτίζεται ακριβώς με τη δοσμένη σχέση! Δηλαδή η $\varphi(x) = e^{x-1}-\ln x$ ικανοποιεί τη σχέση.

\underline{Μοναδικότητα:} Θέτουμε $g(y) = y + e^y$. Τότε:
\[ g'(y) = 1+e^y > 0 \quad\forall\, y\in\mathbb{R} \]
Η $g$ γνησίως αύξουσα, δηλαδή $1$-$1$. Αφού $g(f(x)) = g(\varphi(x))$ και η $g$ $1$-$1$:
\[ f(x) = \varphi(x) = e^{x-1}-\ln x \]

\begin{prosoxi}{\linewidth}
Η τεχνική «μαντεύω τη λύση, μετά δείχνω μοναδικότητα μέσω $g$ 1-1» εμφανίζεται διαρκώς στα Θέματα Δ. Αναγνωρίστε τη μορφή $g(f(x)) = \ldots$
\end{prosoxi}

\item Η $f'(x) = e^{x-1}-\frac{1}{x}$.

Εφαρμόζουμε ΘΜΤ στη $f$ στο $[\alpha,\beta]$ ($1<\alpha<\beta$):

Υπάρχει $\xi\in(\alpha,\beta)$ ώστε:
\[ \frac{f(\beta)-f(\alpha)}{\beta-\alpha} = f'(\xi) = e^{\xi-1}-\frac{1}{\xi} \]

Αφού $\xi>\alpha$ (και η $f'$ γνησίως αύξουσα για $x>1$, αφού $f''(x)=e^{x-1}+\frac{1}{x^2}>0$):
\[ f'(\xi) > f'(\alpha) = e^{\alpha-1}-\frac{1}{\alpha} \]

Άρα:
\[ \boxed{\frac{f(\beta)-f(\alpha)}{\beta-\alpha} > e^{\alpha-1}-\frac{1}{\alpha}} \]

\begin{parat}{\linewidth}
Η $f''(x) = e^{x-1}+\frac{1}{x^2}>0$ σημαίνει ότι η $f'$ είναι γνησίως αύξουσα (η $f$ κυρτή). Αυτό μας επιτρέπει να χρησιμοποιήσουμε $f'(\xi)>f'(\alpha)$ αφού $\xi>\alpha$.
\end{parat}

\item Η εφαπτομένη στο $(x_0,f(x_0))$ έχει εξίσωση:
\[ y - f(x_0) = f'(x_0)(x-x_0) \]
Αν διέρχεται από $O(0,0)$:
\[ -f(x_0) = f'(x_0)(0-x_0) \Rightarrow f(x_0) = x_0 f'(x_0) \]

Αντικαθιστούμε:
\[ e^{x_0-1}-\ln x_0 = x_0\left(e^{x_0-1}-\frac{1}{x_0}\right) = x_0 e^{x_0-1}-1 \]
\[ e^{x_0-1}-\ln x_0 = x_0 e^{x_0-1}-1 \]
\[ e^{x_0-1}(1-x_0) = \ln x_0 - 1 \]

Θέτουμε $\psi(x) = e^{x-1}(1-x)-\ln x+1$ στο $[1,2]$.

Ελέγχουμε:
\begin{align*}
\psi(1) &= e^0(0) - 0 + 1 = 1 > 0 \\
\psi(2) &= e(1-2)-\ln 2 + 1 = -e-\ln 2+1 \approx -2{,}718 - 0{,}693 + 1 = -2{,}411 < 0
\end{align*}

Αφού $\psi$ συνεχής στο $[1,2]$ (ως σύνθεση συνεχών) και $\psi(1)\cdot\psi(2)<0$, από \eng{Bolzano} υπάρχει τουλάχιστον ένα $x_0\in(1,2)$ ώστε $\psi(x_0)=0$.

\underline{Μοναδικότητα:}
\[ \psi'(x) = e^{x-1}(1-x)+e^{x-1}(-1)-\frac{1}{x} = e^{x-1}(1-x-1)-\frac{1}{x} = -xe^{x-1}-\frac{1}{x} \]

Αφού $x>0$: $\psi'(x) = -xe^{x-1}-\frac{1}{x} < 0$ για κάθε $x>0$.

Η $\psi$ γνησίως φθίνουσα στο $(1,2)$, άρα η ρίζα είναι \textbf{μοναδική}.

\item Αφού η $f$ κυρτή ($f''>0$), η γραφική παράσταση βρίσκεται \textbf{κάτω} από κάθε χορδή.

Η χορδή που ενώνει τα $(1,f(1))$ και $(e,f(e))$: $f(1) = e^0 - \ln 1 = 1$ και $f(e) = e^{e-1}-1$.

Εφαπτομένη στο $x=1$: $f'(1) = e^0 - 1 = 0$, δηλαδή $\varepsilon_1: y=1$.

Αφού η $f$ κυρτή, η γραφική παράσταση βρίσκεται \textbf{πάνω} από κάθε εφαπτομένη:
\[ f(x) \ge 1 \quad\text{για κάθε }x>0 \]

Επίσης, η $f$ κυρτή σημαίνει ότι βρίσκεται \textbf{κάτω} από τη χορδή $A(1,1)$, $B(e,f(e))$:
\[ f(x) \le f(1) + \frac{f(e)-f(1)}{e-1}(x-1) = 1+\frac{e^{e-1}-2}{e-1}(x-1) \]
για $x\in[1,e]$.

Ολοκληρώνουμε:
\begin{align*}
\int_1^e f(x)\,\d x &< \int_1^e \left[1+\frac{e^{e-1}-2}{e-1}(x-1)\right]\d x \\
&= \left[x + \frac{e^{e-1}-2}{e-1}\cdot\frac{(x-1)^2}{2}\right]_1^e \\
&= e + \frac{(e^{e-1}-2)(e-1)}{2} - 1 \\
&= e-1 + \frac{(e^{e-1}-2)(e-1)}{2}
\end{align*}

Εναλλακτικά, μπορούμε να αξιοποιήσουμε ότι $f(x) = e^{x-1}-\ln x < e^{x-1}$ (αφού $\ln x > 0$ για $x>1$):
\begin{align*}
\int_1^e f(x)\,\d x &= \int_1^e (e^{x-1}-\ln x)\,\d x \\
&< \int_1^e e^{x-1}\,\d x = \left[e^{x-1}\right]_1^e = e^{e-1}-e^0 = e^{e-1}-1
\end{align*}

Άρα:
\[ \boxed{\int_1^e f(x)\,\d x < e^{e-1}-1} \]

\begin{prosoxi}{\linewidth}
Η τεχνική «αφαιρώ θετικό όρο $\Rightarrow$ μικρότερο ολοκλήρωμα» είναι πολύ χρήσιμη. Εδώ: $e^{x-1}-\ln x < e^{x-1}$ αφού $\ln x > 0$ στο $(1,e)$, και το ολοκλήρωμα της $e^{x-1}$ υπολογίζεται εύκολα.
\end{prosoxi}

\end{erwthma}
\end{thema}
