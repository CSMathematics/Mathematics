%=========== ΛΥΣΗ - 3ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}

\item \textbf{Σ.}
Αν $f(x)<g(x)$ $\forall x\in[\alpha,\beta]$, τότε $f(x)-g(x)<0$ παντού, οπότε:
\[ \int_\alpha^\beta [f(x)-g(x)]\,\d x < 0 \Rightarrow \int_\alpha^\beta f(x)\,\d x < \int_\alpha^\beta g(x)\,\d x \]

\item \textbf{Σ.}
Αυτή είναι η πρόταση του θεωρήματος \eng{Fermat}: αν η $f$ παραγωγίσιμη στο $x_0$ και παρουσιάζει τοπικό ακρότατο, τότε $f'(x_0)=0$.

\item \textbf{Λ.}
Αντιπαράδειγμα: $f(x) = \hm{x}$ στο $[0,2\pi]$:
\[ \int_0^{2\pi}\hm{x}\,\d x = [-\syn{x}]_0^{2\pi} = -1+1 = 0 \]
ενώ $\alpha = 0 \ne 2\pi = \beta$.

\item \textbf{Λ.}
Το πεδίο ορισμού της $g\circ f$ \textbf{δεν} είναι $D_f\cap D_g$. Είναι:
\[ D_{g\circ f} = \{x\in D_f : f(x)\in D_g\} \]
Παράδειγμα: $f(x) = x^2$, $g(x) = \ln x$. Τότε $D_f=\mathbb{R}$, $D_g=(0,+\infty)$, $D_f\cap D_g = (0,+\infty)$, αλλά $(g\circ f)(x) = \ln(x^2)$ ορίζεται σε $\mathbb{R}^*$.

\begin{prosoxi}{\linewidth}
Κλασική παγίδα: πεδίο ορισμού σύνθεσης $\ne$ τομή πεδίων ορισμού.
\end{prosoxi}

\item \textbf{Λ.}
Η πρόταση αφορά \textbf{αόριστο ολοκλήρωμα} (παράγουσα), το οποίο είναι \textbf{εκτός ύλης} Γ' Λυκείου. Στο πλαίσιο του μαθήματος, ξέρουμε μόνο ότι $\dint_\alpha^\beta f'(x)\d x = f(\beta)-f(\alpha)$ (ορισμένο, όχι αόριστο).

Αλλά ακόμα κι αν δεχτούμε αόριστο ολοκλήρωμα: η πρόταση λέει «πάντοτε $f(x)+c$» — αυτό ισχύει μόνο σε \textbf{διαστήματα}, όχι σε ενώσεις (η σταθερά μπορεί να διαφέρει σε κάθε κλάδο).

\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 3ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
$f(x)=\begin{cases} \frac{\sqrt{x^2+3}-2}{x-1}, & x<1 \\ \alpha x+\beta, & x\ge 1 \end{cases}$
\begin{erwthma}

\item Υπολογίζουμε $\lim\limits_{x\to 1^-} f(x) = \lim\limits_{x\to 1^-}\dfrac{\sqrt{x^2+3}-2}{x-1}$.

Πολλαπλασιάζουμε με τον συζυγή:
\begin{align*}
\lim_{x\to 1^-}\frac{\sqrt{x^2+3}-2}{x-1}\cdot\frac{\sqrt{x^2+3}+2}{\sqrt{x^2+3}+2}
&= \lim_{x\to 1^-}\frac{x^2+3-4}{(x-1)(\sqrt{x^2+3}+2)} \\
&= \lim_{x\to 1^-}\frac{x^2-1}{(x-1)(\sqrt{x^2+3}+2)} \\
&= \lim_{x\to 1^-}\frac{(x-1)(x+1)}{(x-1)(\sqrt{x^2+3}+2)} \\
&= \lim_{x\to 1^-}\frac{x+1}{\sqrt{x^2+3}+2} = \frac{2}{2+2} = \frac{1}{2}
\end{align*}

\item Η $f$ διέρχεται από $(2,\frac{3}{4})$ (με $2\ge 1$, άρα $f(2)=2\alpha+\beta$):
\[ 2\alpha+\beta = \frac{3}{4} \quad (1) \]

Για συνέχεια στο $x_0=1$: $\lim\limits_{x\to 1^-} f(x) = f(1)$:
\[ \frac{1}{2} = \alpha+\beta \quad (2) \]

Από $(1)-(2)$: $\alpha = \frac{3}{4}-\frac{1}{2}=\frac{1}{4}$.

Από $(2)$: $\beta = \frac{1}{2}-\frac{1}{4} = \frac{1}{4}$.

Άρα $\boxed{\alpha=\frac{1}{4},\quad\beta=\frac{1}{4}}$.

\item Για $\alpha=\beta=\frac{1}{4}$: $f(x) = \frac{x+1}{4}$ για $x\ge 1$.

\underline{Στο $+\infty$:}
\[ \lim_{x\to+\infty}f(x)=\lim_{x\to+\infty}\frac{x+1}{4}=+\infty \]

\underline{Στο $-\infty$:} Εδώ $x<1$, οπότε $f(x)=\frac{\sqrt{x^2+3}-2}{x-1}$.

Για $x\to-\infty$: $\sqrt{x^2+3}=|x|\sqrt{1+\frac{3}{x^2}}\stackrel{x<0}{=}-x\sqrt{1+\frac{3}{x^2}}$.

\begin{align*}
f(x)&=\frac{-x\sqrt{1+\frac{3}{x^2}}-2}{x-1}=\frac{x\left(-\sqrt{1+\frac{3}{x^2}}-\frac{2}{x}\right)}{x\left(1-\frac{1}{x}\right)}
=\frac{-\sqrt{1+\frac{3}{x^2}}-\frac{2}{x}}{1-\frac{1}{x}}
\end{align*}
\[ \lim_{x\to-\infty}f(x) = \frac{-1-0}{1-0} = -1 \]

\item Για $x\in[1,3]$: $f(x)=\frac{x+1}{4}$.

$f(x)=0 \Leftrightarrow x=-1\notin[1,3]$. Αφού $f(1)=\frac{1}{2}>0$, η $f(x)>0$ στο $[1,3]$.

\begin{align*}
E &= \int_1^3 f(x)\,\d x = \int_1^3 \frac{x+1}{4}\,\d x
= \frac{1}{4}\left[\frac{x^2}{2}+x\right]_1^3 \\
&= \frac{1}{4}\left[\left(\frac{9}{2}+3\right)-\left(\frac{1}{2}+1\right)\right]
= \frac{1}{4}\left[\frac{15}{2}-\frac{3}{2}\right]
= \frac{1}{4}\cdot 6 = \frac{3}{2} \text{ τ.μ.}
\end{align*}

\end{erwthma}
\end{thema}

\begin{thema}{Γ}
$f(x)=x-\ln x$, $x\in(0,+\infty)$.
\begin{erwthma}

\item \underline{Κυρτότητα:}
\[ f'(x) = 1-\frac{1}{x},\qquad f''(x) = \frac{1}{x^2} > 0 \quad\forall\, x>0 \]
Η $f$ είναι \textbf{κυρτή} (στρέφει τα κοίλα άνω) σε όλο το $(0,+\infty)$. Δεν υπάρχει σημείο καμπής.

\underline{Εφαπτομένη στο $x_0=1$:}
\[ f(1) = 1-\ln 1 = 1,\qquad f'(1) = 1-1 = 0 \]
\[ \varepsilon:\ y = 0\cdot(x-1)+1 \Rightarrow \boxed{y = 1} \]

\item \underline{Ολικό ελάχιστο:}
\[ f'(x)=0 \Leftrightarrow 1-\frac{1}{x}=0 \Leftrightarrow x=1 \]

Η $f$ φθίνουσα στο $(0,1]$, αύξουσα στο $[1,+\infty)$. Ολικό ελάχιστο: $f(1)=1$.

\underline{Ανισότητα $e^{x-1}\ge x$:}

Αφού $f(x)\ge f(1) = 1$ $\forall\, x>0$:
\[ x - \ln x \ge 1 \Rightarrow x-1 \ge \ln x \Rightarrow e^{x-1}\ge e^{\ln x} = x \]

\begin{prosoxi}{\linewidth}
Η τεχνική «ολικό ελάχιστο $\Rightarrow$ ανισότητα $\Rightarrow$ εκθετικοποίηση» εμφανίζεται πολύ συχνά στις Πανελλαδικές. Θυμηθείτε: $a\ge b \Rightarrow e^a\ge e^b$.
\end{prosoxi}

\item Εφαπτομένη στο $A(x_1,f(x_1))$ που περνά από $O(0,0)$:
\[ \varepsilon:\ y - f(x_1) = f'(x_1)(x-x_1) \]
Στο $O(0,0)$:
\[ -f(x_1) = -x_1 f'(x_1) \Rightarrow f(x_1) = x_1 f'(x_1) \]
\[ x_1-\ln x_1 = x_1\left(1-\frac{1}{x_1}\right) = x_1-1 \]
\[ -\ln x_1 = -1 \Rightarrow \ln x_1 = 1 \Rightarrow x_1 = e \]

Σημείο επαφής: $A(e,\, e-1)$. Κλίση: $f'(e) = 1-\frac{1}{e} = \frac{e-1}{e}$.

Εξίσωση εφαπτομένης:
\[ \boxed{\varepsilon:\ y = \frac{e-1}{e}\,x} \]

Επαλήθευση: $\varepsilon$ στο $O$: $y=0$ \checkmark. $\varepsilon$ στο $A$: $y=\frac{e-1}{e}\cdot e = e-1$ \checkmark.

\item Σημείο $M(x,f(x))$ με $x>e$. Η γωνία $\theta$ της επιβατικής ακτίνας $OM$:
\[ \ef{\theta} = \frac{f(x)}{x} = \frac{x-\ln x}{x} = 1-\frac{\ln x}{x} \]

Παραγωγίζουμε ως προς $t$:
\[ \frac{1}{\syn^2{\theta}}\cdot\theta'(t) = -\frac{\frac{1}{x}\cdot x - \ln x}{x^2}\cdot x'(t) = -\frac{1-\ln x}{x^2}\cdot x'(t) \]

Στο $x=e^2$, $x'(t)=e$:
\[ \ef{\theta}\bigg|_{x=e^2} = 1-\frac{2}{e^2} = \frac{e^2-2}{e^2} \]

\[ \syn^2{\theta}=\frac{1}{1+\ef^2{\theta}} = \frac{1}{1+\left(\frac{e^2-2}{e^2}\right)^2}=\frac{e^4}{e^4+(e^2-2)^2} = \frac{e^4}{2e^4-4e^2+4}\]

Ο αριθμητής του ρυθμού:
\[ -\frac{1-\ln(e^2)}{(e^2)^2}\cdot e = -\frac{1-2}{e^4}\cdot e = \frac{e}{e^4} = \frac{1}{e^3} \]

Άρα:
\[ \theta'(t_0) = \frac{1}{e^3}\cdot\syn^2{\theta} = \frac{1}{e^3}\cdot\frac{e^4}{2e^4-4e^2+4} = \frac{e}{2e^4-4e^2+4} = \frac{e}{2(e^4-2e^2+2)} \]

$\boxed{\theta'(t_0) = \dfrac{e}{2(e^4-2e^2+2)}\ \text{rad/sec}}$

\end{erwthma}
\end{thema}

\begin{thema}{Δ}
$f:(0,+\infty)\to(0,+\infty)$ παραγωγίσιμη: $f(x)\cdot e^{f(x)} = e^{x-1}\cdot e^{e^{x-1}}$.
\begin{erwthma}

\item Η δοσμένη σχέση γράφεται: $f(x)\cdot e^{f(x)} = e^{x-1}\cdot e^{e^{x-1}}$.

Θέτουμε $g(y) = ye^y$ για $y>0$. Τότε $g'(y) = e^y+ye^y = (1+y)e^y > 0$ για $y>0$.

Η $g$ γνησίως αύξουσα (στο $(0,+\infty)$), δηλαδή $1$-$1$.

Η σχέση γίνεται: $g(f(x)) = g(e^{x-1})$.

Αφού η $g$ $1$-$1$: $\boxed{f(x) = e^{x-1}}$.

\item $f'(x) = e^{x-1} > 0$ $\forall\, x>0$, άρα η $f$ \textbf{γνησίως αύξουσα}.

$f''(x) = e^{x-1} > 0$ $\forall\, x>0$, άρα η $f$ \textbf{κυρτή}.

\underline{Εφαπτομένη στο $x=1$:}
\[ f(1)=1,\quad f'(1)=1 \qquad\Rightarrow\qquad \varepsilon:\ y = x \]

Αφού η $f$ κυρτή, η $C_f$ βρίσκεται \textbf{πάνω} από κάθε εφαπτομένη:
\[ f(x) \ge x \quad\forall\, x>0 \]

Ολοκληρώνουμε στο $[0,1]$:
\[ \int_0^1 f(x)\,\d x \ge \int_0^1 x\,\d x = \frac{1}{2} \]

Η ισότητα ισχύει μόνο στο $x=1$, άρα (αφού $f(x)>x$ σε ολόκληρο το $(0,1)$):
\[ \boxed{\int_0^1 f(x)\,\d x > \frac{1}{2}} \]

\item Αξιοποιούμε πάλι $f(x)\ge x$ $\forall\, x\ge 0$ (κυρτότητα):
\[ \int_0^2 f(x)\,\d x > \int_0^2 x\,\d x = \left[\frac{x^2}{2}\right]_0^2 = 2 \]

\underline{Ακριβής τιμή:}
\begin{align*}
\int_0^2 f(x)\,\d x &= \int_0^2 e^{x-1}\,\d x = \left[e^{x-1}\right]_0^2 = e^1-e^{-1} = e-\frac{1}{e} = \frac{e^2-1}{e}
\end{align*}

$\boxed{\dint_0^2 e^{x-1}\d x = \dfrac{e^2-1}{e}}$

Επαλήθευση: $\frac{e^2-1}{e}\approx\frac{6{,}39}{2{,}72}\approx 2{,}35 > 2$ \checkmark.

\item Αφού η $f = e^{x-1}$ γνησίως αύξουσα:
\[ f(x^2+1)>f(x+1) \Leftrightarrow x^2+1>x+1 \Leftrightarrow x^2-x>0 \Leftrightarrow x(x-1)>0 \]

\[ \boxed{x\in(-\infty,0)\cup(1,+\infty)} \]

\end{erwthma}
\end{thema}
