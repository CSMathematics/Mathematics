\begin{thema}{Α}
\begin{erwthma}
\item Έστω ότι το $f(x_0)$ είναι τοπικό μέγιστο της $f$. Τότε υπάρχει $\delta>0$ τέτοιο ώστε, για κάθε $x\in(x_0-\delta,x_0+\delta)$, να ισχύει
\[
f(x)\leq f(x_0).
\]
Για $h>0$ αρκετά μικρό έχουμε
\[
\frac{f(x_0+h)-f(x_0)}{h}\leq0.
\]
Άρα
\[
\lim_{h\to0^+}\frac{f(x_0+h)-f(x_0)}{h}\leq0.
\]
Για $h<0$ αρκετά μικρό, επειδή $f(x_0+h)-f(x_0)\leq0$ και $h<0$, έχουμε
\[
\frac{f(x_0+h)-f(x_0)}{h}\geq0.
\]
Άρα
\[
\lim_{h\to0^-}\frac{f(x_0+h)-f(x_0)}{h}\geq0.
\]
Επειδή η $f$ είναι παραγωγίσιμη στο $x_0$, τα δύο πλευρικά όρια είναι ίσα με $f'(x_0)$. Επομένως
\[
f'(x_0)\leq0
\quad\text{και}\quad
f'(x_0)\geq0,
\]
οπότε
\[
f'(x_0)=0.
\]
Αν το $f(x_0)$ είναι τοπικό ελάχιστο, η απόδειξη είναι ανάλογη, με τις ανισότητες αντίστροφες. Άρα, σε κάθε περίπτωση τοπικού ακροτάτου,
\[
{f'(x_0)=0}.
\]

\item Μια συνάρτηση $f$ λέγεται κυρτή σε ένα διάστημα $\varDelta$, όταν για κάθε $x_1,x_2\in\varDelta$ και κάθε $\lambda\in[0,1]$ ισχύει
\[
f\bigl(\lambda x_1+(1-\lambda)x_2\bigr)
\leq
\lambda f(x_1)+(1-\lambda)f(x_2).
\]
Γεωμετρικά, η γραφική της παράσταση βρίσκεται κάτω από κάθε χορδή της.

\item Αν η $f$ είναι συνεχής στο $[a,\beta]$ και $k$ είναι ένας αριθμός μεταξύ των $f(a)$ και $f(\beta)$, τότε υπάρχει τουλάχιστον ένα $x_0\in[a,\beta]$ τέτοιο ώστε
\[
f(x_0)=k.
\]
Αν επιπλέον $k$ βρίσκεται αυστηρά μεταξύ των $f(a)$ και $f(\beta)$, τότε το $x_0$ ανήκει στο $(a,\beta)$.

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Η $f(x)=|x|$ παρουσιάζει τοπικό ελάχιστο στο $x_0=0$, αλλά δεν είναι παραγωγίσιμη στο $0$.
\item \textbf{Σωστό}. Τα τοπικά ακρότατα συγκρίνουν τιμές μόνο σε μια περιοχή του σημείου, όχι σε όλο το πεδίο ορισμού.
\item \textbf{Σωστό}. Αν $f''(x)>0$ για κάθε $x\in\mathbb{R}$, τότε η $f$ είναι κυρτή και η γραφική της παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της, εκτός από το σημείο επαφής.
\item \textbf{Σωστό}. Ισχύει
\[
\dint_a^\beta 1\,\d x=\beta-a,
\]
που είναι το εμβαδόν ορθογωνίου με πλευρές $1$ και $\beta-a$.
\item \textbf{Σωστό}. Για $f(x)=\ln x$ έχουμε $f(1)=0$ και $f'(1)=1$, άρα η εφαπτομένη στο $x_0=1$ είναι
\[
y=x-1.
\]
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Θεωρούμε τη συνάρτηση
\[
f(x)=x\ln x-x+1,\quad x>0.
\]
\begin{erwthma}
\item Για κάθε $x>0$ είναι
\[
f'(x)=\ln x+x\cdot\frac{1}{x}-1=\ln x.
\]
Βρίσκουμε ρίζες και πρόσημα της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow \ln x=0\Leftrightarrow x=1$.
\item $f'(x)>0\Leftrightarrow x>1$.
\item $f'(x)<0\Leftrightarrow 0<x<1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{d,-,z,+,}
\tkzTabLine{d,\searrow,t,\nearrow,}
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στο $(0,1]$ και \textbf{γνησίως αύξουσα} στο $[1,+\infty)$.

Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
f(1)=1\cdot\ln1-1+1=0.
\]

\item Για κάθε $x>0$ είναι
\[
f''(x)=\frac{1}{x}>0.
\]
Άρα η $f$ είναι \textbf{κυρτή} σε όλο το $(0,+\infty)$.

Επειδή η $f''$ δεν μηδενίζεται και δεν αλλάζει πρόσημο στο πεδίο ορισμού, η γραφική παράσταση της $f$ δεν έχει σημεία καμπής.

\item Από το πρώτο ερώτημα έχουμε
\[
f(x)\geq f(1)=0
\]
για κάθε $x>0$. Άρα
\[
x\ln x-x+1\geq0
\Leftrightarrow
x\ln x\geq x-1.
\]
Η ισότητα ισχύει μόνο στη θέση του ολικού ελαχίστου, δηλαδή μόνο για
\[
{x=1}.
\]

\item Για $x\to0^+$ έχουμε
\[
\lim_{x\to0^+}x\ln x=0,
\]
οπότε
\[
\lim_{x\to0^+}f(x)=1.
\]
Άρα δεν υπάρχει κατακόρυφη ασύμπτωτη στο $x=0$.

Στο $+\infty$ έχουμε
\[
\lim_{x\to+\infty}f(x)=+\infty,
\]
άρα δεν υπάρχει οριζόντια ασύμπτωτη.

Για πλάγια ασύμπτωτη εξετάζουμε
\[
\lim_{x\to+\infty}\frac{f(x)}{x}
=\lim_{x\to+\infty}\left(\ln x-1+\frac{1}{x}\right)=+\infty.
\]
Άρα δεν υπάρχει πλάγια ασύμπτωτη.

Τέλος,
\begin{align*}
\lim_{x\to+\infty}\frac{f(x)}{x^2}
&=\lim_{x\to+\infty}\left(\frac{\ln x}{x}-\frac{1}{x}+\frac{1}{x^2}\right)\\
&=0.
\end{align*}
Επομένως
\[
{\lim_{x\to+\infty}\frac{f(x)}{x^2}=0}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Έστω $x$ το μήκος των δύο κάθετων πλευρών του ορθογωνίου και $y$ το μήκος της πλευράς που περιφράσσεται απέναντι από το ποτάμι.
\begin{erwthma}
\item Επειδή η πλευρά $AB$ δεν περιφράσσεται και υπάρχει άνοιγμα $2$ μέτρων, το σύρμα μήκους $40$ μέτρων καλύπτει μήκος
\[
2x+y-2.
\]
Άρα
\[
2x+y-2=40
\Leftrightarrow y=42-2x.
\]
Το εμβαδόν είναι
\[
E=x y=x(42-2x)=42x-2x^2.
\]
Οι φυσικοί περιορισμοί δίνουν
\[
x>0
\quad\text{και}\quad
y=42-2x>0\Leftrightarrow x<21.
\]
Άρα
\[
{E(x)=42x-2x^2,\quad x\in(0,21)}.
\]

\item Παραγωγίζουμε:
\[
E'(x)=42-4x.
\]
Έχουμε
\[
E'(x)=0\Leftrightarrow 42-4x=0\Leftrightarrow x=\frac{21}{2}.
\]
Επίσης
\[
E'(x)>0\quad\text{για }0<x<\frac{21}{2}
\]
και
\[
E'(x)<0\quad\text{για }\frac{21}{2}<x<21.
\]
Άρα το εμβαδόν μεγιστοποιείται μοναδικά για
\[
{x=\frac{21}{2}\ \si{m}}.
\]

\item Για τη βέλτιστη κατασκευή έχουμε
\[
x=\frac{21}{2}
\]
και
\[
y=42-2\cdot\frac{21}{2}=21.
\]
Για να εγγράφεται κυκλικό σιντριβάνι στο ορθογώνιο, η διάμετρός του δεν μπορεί να ξεπερνά τη μικρότερη πλευρά του ορθογωνίου. Άρα
\[
2\rho\leq \frac{21}{2}.
\]
Η μέγιστη αποδεκτή ακτίνα είναι
\[
{\rho_{\max}=\frac{21}{4}\ \si{m}}.
\]

\item Έχουμε
\[
E(t)=42x(t)-2x^2(t).
\]
Παραγωγίζοντας ως προς $t$, παίρνουμε
\[
E'(t)=\bigl(42-4x(t)\bigr)x'(t).
\]
Τη χρονική στιγμή που $x=12$ και $x'(t)=1$, έχουμε
\[
E'(t)=\bigl(42-4\cdot12\bigr)\cdot1=-6.
\]
Άρα το εμβαδόν μεταβάλλεται με ρυθμό
\[
{-6\ \si{m^2}/\sec}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
f:\mathbb{R}\to(0,+\infty)
\]
τέτοια ώστε
\[
f(x)+\ln(f(x))=x
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Παραγωγίζουμε τη δοσμένη σχέση:
\[
f'(x)+\frac{f'(x)}{f(x)}=1.
\]
Άρα
\[
f'(x)\left(1+\frac{1}{f(x)}\right)=1
\]
και, επειδή $f(x)>0$,
\[
f'(x)=\frac{f(x)}{f(x)+1}>0.
\]
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Από το προηγούμενο ερώτημα έχουμε
\[
f'(x)=\frac{f(x)}{f(x)+1}.
\]
Παραγωγίζοντας,
\begin{align*}
f''(x)
&=\left(\frac{f(x)}{f(x)+1}\right)'\\
&=\frac{f'(x)(f(x)+1)-f(x)f'(x)}{(f(x)+1)^2}\\
&=\frac{f'(x)}{(f(x)+1)^2}.
\end{align*}
Με αντικατάσταση του $f'(x)=\frac{f(x)}{f(x)+1}$ παίρνουμε
\[
f''(x)=\frac{f(x)}{(f(x)+1)^3}.
\]
Επειδή $f(x)>0$, έχουμε
\[
f''(x)>0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $f$ είναι \textbf{κυρτή} στο $\mathbb{R}$.

\item Πρώτα βρίσκουμε την τιμή $f(1)$. Αν $f(1)=y$, τότε
\[
y+\ln y=1.
\]
Η συνάρτηση
\[
\varphi(y)=y+\ln y,\quad y>0,
\]
είναι γνησίως αύξουσα, αφού
\[
\varphi'(y)=1+\frac{1}{y}>0.
\]
Επειδή
\[
\varphi(1)=1,
\]
προκύπτει
\[
f(1)=1.
\]
Για $x>1$, επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\[
f(x)>f(1)=1.
\]
Άρα
\[
\ln(f(x))>0.
\]
Από τη δοσμένη σχέση
\[
x=f(x)+\ln(f(x))
\]
παίρνουμε
\[
x>f(x).
\]
Επομένως
\[
{f(x)<x,\quad x>1}.
\]

\item Από τη σχέση
\[
x=f(x)+\ln(f(x))
\]
και επειδή $x\to+\infty$, προκύπτει ότι
\[
f(x)\to+\infty.
\]
Πράγματι, αν η $f(x)$ δεν έτεινε στο $+\infty$, τότε το $f(x)+\ln(f(x))$ δεν θα μπορούσε να τείνει στο $+\infty$.

Θέτουμε $y=f(x)$. Τότε, καθώς $x\to+\infty$, έχουμε $y\to+\infty$ και
\[
x=y+\ln y.
\]
Άρα
\[
\frac{f(x)}{x}=\frac{y}{y+\ln y}
=\frac{1}{1+\frac{\ln y}{y}}.
\]
Επειδή
\[
\lim_{y\to+\infty}\frac{\ln y}{y}=0,
\]
παίρνουμε
\[
{\lim_{x\to+\infty}\frac{f(x)}{x}=1}.
\]
\end{erwthma}
\end{thema}
