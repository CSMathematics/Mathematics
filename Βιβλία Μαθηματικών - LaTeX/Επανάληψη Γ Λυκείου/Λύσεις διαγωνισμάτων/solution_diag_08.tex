%=========== ΛΥΣΗ - 8ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Λ.} 
Μία συνάρτηση μπορεί κάλλιστα να παρουσιάζει ακρότατο σε σημείο που \textbf{δεν είναι παραγωγίσιμη} (π.χ. «γωνιακό» σημείο). Το θεώρημα \eng{Fermat} είναι μονόδρομο: Αν είναι παραγωγίσιμη στο εσωτερικό ακρότατο, τότε $f'=0$. Το αντίστροφο δεν ισχύει. 
Αντιπαράδειγμα: η $f(x) = |x|$ έχει τοπικό (και ολικό) ελάχιστο στο $x_0=0$ όπου δεν είναι παραγωγίσιμη.
\item \textbf{Σ.}
Πράγματι, ένα τοπικό μέγιστο μπορεί να είναι μικρότερο από ένα τοπικό ελάχιστο διαφορετικής «περιοχής» του πεδίου ορισμού μιας συνάρτησης. (Δεν γίνεται σύγκριση απόλυτων, αλλά τοπικών).
\item \textbf{Σ.}
Αφού $f''(x) > 0$, η $f$ είναι κυρτή. Το θεώρημα αναφέρει ότι η γραφική παράσταση μιας κυρτής συνάρτησης βρίσκεται \textbf{πάνω} από οποιαδήποτε εφαπτομένη της, με εξαίρεση το κοινό σημείο επαφής τους.
\item \textbf{Σ.}
Ισχύει. Το $\dint_\alpha^\beta 1\,\d x = \beta-\alpha > 0$ υπολογίζει το εμβαδόν μεταξύ της σταθερής ευθείας $y=1$ και του άξονα των τετμημένων στο $[\alpha, \beta]$, που συνιστά γεωμετρικά ένα ορθογώνιο διαστάσεων $1 \times (\beta-\alpha)$.
\item \textbf{Σ.}
Για την $f(x) = \ln x$, παραγωγίζοντας παίρνουμε $f'(x) = \frac{1}{x}$.
Στο $x_0 = 1$: $f(1) = 0$ και $f'(1) = 1$.
Η εξίσωση της εφαπτομένης είναι: $y - f(1) = f'(1)(x-1) \Rightarrow y - 0 = 1(x-1) \Rightarrow y = x - 1$.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 8ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η $f(x) = x \ln x - x + 1, \quad x \in (0, +\infty)$.
\begin{erwthma}

\item \textbf{Μονοτονία και Ακρότατα:}
Παραγωγίζουμε την $f$ στο εσωτερικό πεδίο $(0, +\infty)$:
\[ f'(x) = (x)' \ln x + x (\ln x)' - (x)' + 0 = 1 \cdot \ln x + x \frac{1}{x} - 1 = \ln x + 1 - 1 = \ln x \]

Λύνουμε τις ανισότητες προσήμου:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow \ln x = 0 \Leftrightarrow x = 1$
    \item $f'(x) > 0 \Leftrightarrow \ln x > 0 \Leftrightarrow x > 1$
    \item $f'(x) < 0 \Leftrightarrow \ln x < 0 \Leftrightarrow 0 < x < 1$
\end{itemize}

Συμπεραίνουμε ότι η συνάρτηση $f$ είναι \textbf{γνησίως φθίνουσα} στο ανοιχτό-κλειστό $x \in (0, 1]$ και \textbf{γνησίως αύξουσα} στο $x \in [1, +\infty)$.
Παρουσιάζει συνεπώς \textbf{ολικό ελάχιστο} στο $x = 1$, με τιμή εκτιμώμενη:
\[ f(1) = 1 \cdot \ln 1 - 1 + 1 = 0 \cdot 0 = 0 \]
Άρα Ο.Ε. $\to (1,0)$.

\item \textbf{Κυρτότητα:}
Παραγωγίζουμε εκ νέου την πρώτη παράγωγο $(f'(x) = \ln x)$:
\[ f''(x) = (\ln x)' = \frac{1}{x} \]
Δεδομένου ότι το πεδίο ορισμού είναι $x > 0$, έπεται ολόφωνα πως $f''(x) > 0$ για κάθε $x \in (0, +\infty)$.
Επομένως, η $f$ είναι \textbf{γνησίως κυρτή} $\left( \cup \right)$ σε ολόκληρο το πεδίο ορισμού της.
Κρίνοντας πώς η $f''$ δεν μηδενίζεται σε κανένα σημείο και δεν αλλάζει πρόσημο, βεβαιώνεται ότι η γραφική της παράσταση \textbf{δεν διαθέτει κανένα απολύτως σημείο καμπής}.

\item Ζητείται αλγεβρικά η ανισότητα $x \ln x \ge x - 1 \Leftrightarrow x \ln x - x + 1 \ge 0 \Leftrightarrow f(x) \ge 0$.
Όπως έχει ήδη επιβεβαιωθεί στο $1$\textsuperscript{ο} ερώτημα, η $f$ διαθέτει ένα και μοναδικό ολικό ελάχιστο, το $0$ στη θέση $x=1$.
Από τον ακριβή ορισμό του ελαχίστου αναδύεται απευθείας η εξής συμπεριφορά:
\[ f(x) \ge f(1) \quad \text{για κάθε } x \in (0, +\infty) \]
Αντικαθιστώντας, καταλήγουμε:
\[ x \ln x - x + 1 \ge 0 \Rightarrow x \ln x \ge x - 1 \]
Ως προς την ισότητα: Η σχέση λαμβάνει την ισότητα $f(x)=0$ \textbf{μόνο και αποκλειστικά} στο σημείο ολικού ελαχίστου. Άρα επαληθεύεται μόνο στο $\boldsymbol{x = 1}$.

\item \textbf{Ασύμπτωτες:}
$\bullet$ \underline{Κατακόρυφη στο $0^+$:} (δεν ανήκει στο Π.Ο. και δίνει ανοιχτό άκρο)
Οφείλουμε να ελέγξουμε το όριο $\lim_{x\to 0^+} f(x)$.
Χρειαζόμαστε πρώτα το $\lim_{x\to 0^+} (x \ln x)$:
\[ \lim_{x\to 0^+} (x \ln x) = \lim_{x\to 0^+} \frac{\ln x}{\frac{1}{x}} \xlongequal[\text{\eng{DLH}}]{\frac{-\infty}{+\infty}} \lim_{x\to 0^+} \frac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim_{x\to 0^+} (-x) = 0 \]
Συνολικά:
\[ \lim_{x\to 0^+} f(x) = \lim_{x\to 0^+} (x \ln x - x + 1) = 0 - 0 + 1 = 1 \]
Καθώς το όριο είναι πεπερασμένος αριθμός, η συνάρτηση δεν παραλύει στο άπειρο. Οπότε \textbf{δεν διαθέτει κατακόρυφη ασύμπτωτη}.

$\bullet$ \underline{Πλάγια / Οριζόντια στο $+\infty$:}
Ελέγχουμε τον συντελεστή διεύθυνσης $\lambda = \lim_{x\to+\infty} \frac{f(x)}{x}$:
\[ \lim_{x\to +\infty} \frac{x \ln x - x + 1}{x} = \lim_{x\to +\infty} \left( \ln x - 1 + \frac{1}{x} \right) \]
Δεδομένου ότι το $\lim_{x\to+\infty} \ln x = +\infty$ και $\lim_{x\to+\infty} \frac{1}{x} = 0$, το σύνολο τείνει στο $+\infty$. 
Εφόσον δεν υφίσταται πεπερασμένος συντελεστής κατεύθυνσης $\lambda$, \textbf{δεν παραδέχεεται πλάγια (ούτε οριζόντια) ασύμπτωτη στο $+\infty$}.

\textbf{Υπολογισμός τελικού ορίου:}
\[ \lim_{x\to +\infty} \frac{f(x)}{x^2} = \lim_{x\to +\infty} \frac{x \ln x - x + 1}{x^2} = \lim_{x\to +\infty} \left( \frac{\ln x}{x} - \frac{1}{x} + \frac{1}{x^2} \right) \]
όπου $\lim_{x\to+\infty} \frac{\ln x}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{\frac{1}{x}}{1} = 0$.
Επομένως: $0 - 0 + 0 = \boldsymbol{0}$.

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 8ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Ορθογώνιο οικόπεδο με μήκος σύρματος $40$ μέτρα και συνοριακό ποτάμι (μία ανοιχτή επιμήκης πλευρά). Άνοιγμα πύλης: 2 μέτρα.
\begin{erwthma}

\item Έστω $x > 0$ το μήκος των δύο κάθετων πλευρών (στις όχθες του ποταμού) και $y > 0$ το μήκος της μίας και μόνης παράλληλης πλευράς που θα περιφραχτεί. Αναφέρεται ρητά ότι θα αφεθεί ένα άνοιγμα $2$ μέτρων σε μία γωνία (δηλαδή στην πλευρά $y$ χωρίς να χαλάσει η γεωμετρία). Το σύρμα που θα χρησιμοποιηθεί αποτελείται από:
Δύο κάθετες πλευρές + Το υπόλοιπο της παράλληλης πλευράς που αφαιρείται η πύλη.
Δηλαδή:
\[ 2x + (y - 2) = 40 \]
Λύνουμε ως προς $y$:
\[ 2x + y - 2 = 40 \Rightarrow 2x + y = 42 \Rightarrow y = 42 - 2x \]

Το εμβαδόν του ορθογωνίου δίνεται από $E = x \cdot y$. Αντικαθιστώντας το $y$:
\[ E(x) = x(42 - 2x) = 42x - 2x^2 \]
Για τους φυσικούς περιορισμούς, ισχύει $x > 0$ και $y > 0 \Rightarrow 42 - 2x > 0 \Rightarrow 2x < 42 \Rightarrow x < 21$.
Επομένως, το πεδίο ορισμού είναι αναμενόμενα το $\Delta = (0, 21)$.

\item Παραγωγίζουμε την αντίστοιχη συνάρτηση εμβαδού:
\[ E'(x) = (42x - 2x^2)' = 42 - 4x \]
Μηδενίζουμε την παράγωγο για να καθορίσουμε το εύρος της μεγιστοποίησης:
\[ E'(x) = 0 \Leftrightarrow 42 - 4x = 0 \Leftrightarrow 4x = 42 \Leftrightarrow x = \frac{42}{4} = 10{,}5 \]
Έχουμε:
\begin{itemize}
    \item $E'(x) > 0$ για $x < 10{,}5$
    \item $E'(x) < 0$ για $x > 10{,}5$
\end{itemize}
Η συνάρτηση εμβαδού $E(x)$ είναι γνησίως αύξουσα στο $(0, 10{,}5]$ και γνησίως φθίνουσα στο $[10{,}5, 21)$.
Άρα πράγματι, μεγιστοποιείται με ολικό (και μοναδικό) μέγιστο για $\boldsymbol{x = 10{,}5 \ \text{μέτρα}}$.

\item Στη βέλτιστη διάταξη έχουμε τις εξής οριακές διαστάσεις:
\[ x_{\text{max}} = 10{,}5 \ \text{m} \]
\[ y_{\text{max}} = 42 - 2(10{,}5) = 42 - 21 = 21 \ \text{m} \]
Ζητείται να τοποθετηθεί ένα κυκλικό σιντριβάνι ακτίνας $\rho$ εντός του εσωτερικού. Προκειμένου ο κύκλος να χωράει οριακά (να εγγράφεται χωρίς να τέμνει τα σύνορα του οικοπέδου), κρίνεται απαραίτητο η διάμετρός του $2\rho$ να είναι μικρότερη ή το πολύ ίση με την οικεία \textbf{μικρότερη διάσταση} του ορθογωνίου.
Σήμερα η μικρότερη διάσταση του χώρου είναι $x = 10{,}5$. Κατά συνέπεια:
\[ 2\rho \le 10{,}5 \Rightarrow \rho \le \frac{10{,}5}{2} = 5{,}25 \]
Η μέγιστη αποδεκτή ακτίνα $\rho$ αντιστοιχεί στα $\boldsymbol{5{,}25 \ \text{μέτρα}}$.

\item Το δυναμικό εμβαδόν συναρτήσει του χρόνου $t$ εξάγεται:
\[ E(t) = 42x(t) - 2x^2(t) \]
Παραγωγίζουμε ως προς τον χρόνο αναζητώντας τον ζητούμενο ρυθμό (χρήση κανόνα αλυσίδας):
\[ E'(t) = 42 \cdot x'(t) - 4x(t) \cdot x'(t) = (42 - 4x(t)) x'(t) \]

Τη χρονική στιγμή $t_0$, μας παρέχεται ότι $x(t_0) = 12 \ \text{m}$ και η ταχύτητα (ρυθμός μεταβολής θέσης) είναι $x'(t_0) = 1 \ \text{m/s}$. Εισάγουμε:
\[ E'(t_0) = (42 - 4 \cdot 12) \cdot 1 = 42 - 48 = -6 \]
Επομένως, το εμβαδόν το συγκεκριμένο χρονικό στιγμιότυπο \textbf{ελαττώνεται} γραμμικά με ρυθμό μεταβολής $\boldsymbol{-6 \ \text{m}^2\text{/sec}}$.

\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται $f: \mathbb{R} \to (0, +\infty)$ με $f(x) + \ln(f(x)) = x$.
\begin{erwthma}

\item Παραγωγίζουμε τα μέλη της ταυτότητας (η $f$ έχει δοθεί παραγωγίσιμη, άρα ισχύει ο κανόνας αλυσίδας):
\[ f'(x) + \frac{1}{f(x)} f'(x) = 1 \Rightarrow f'(x) \left( 1 + \frac{1}{f(x)} \right) = 1 \Rightarrow f'(x) \frac{f(x)+1}{f(x)} = 1 \]

Αφού εδραιώνεται πως $f(x) > 0 \Rightarrow f(x)+1 > 0$, το κλάσμα $\frac{f(x)+1}{f(x)}$ είναι πάντοτε αυστηρά θετικό. Επομένως:
\[ f'(x) = \frac{f(x)}{f(x)+1} \]
Προκύπτει ξεκάθαρα πως $f'(x) > 0$ για κάθε $x \in \mathbb{R}$. Η συνάρτηση $f(x)$ καθίσταται \textbf{γνησίως αύξουσα} στο σύνολο του $\mathbb{R}$.

\item Παραγωγίζουμε την πρώτη παράγωγο, εφαρμόζοντας τον κανόνα πηλίκου:
\[ f''(x) = \left( \frac{f(x)}{f(x)+1} \right)' = \frac{f'(x)(f(x)+1) - f(x) f'(x)}{(f(x)+1)^2} \]
Κοινός παράγοντας ο αριθμητής:
\[ f''(x) = \frac{f'(x) (f(x)+1 - f(x))}{(f(x)+1)^2} = \frac{f'(x)}{(f(x)+1)^2} \]

Αντικαθιστούμε τον τύπο της $f'(x) = \frac{f(x)}{f(x)+1}$:
\[ f''(x) = \frac{\frac{f(x)}{f(x)+1}}{(f(x)+1)^2} = \frac{f(x)}{(f(x)+1)^3} \]
Δεδομένου ότι το πεδίο τιμών επιβάλλει $f(x) > 0$, ο αριθμητής είναι θετικός και ο παρανομαστής $(f(x)+1)^3 > 1 > 0$.
Συμπεραίνουμε ότι $f''(x) > 0$ για κάθε $x$, οπότε η $f$ είναι πράγματι \textbf{γνησίως κυρτή ($\cup$)}.

\item Η δεδομένη συνθήκη ισοδυναμεί με $f(x) = x - \ln(f(x))$.
Ζητείται να αποδείξουμε ότι για $x > 1$ ισχύει αποκλειστικά $f(x) < x$.
Υποθέτουμε προς άτοπο ότι για κάποιο $x_0 > 1$, θα ίσχυε $f(x_0) \ge x_0$.
Αφού η συνάρτηση του λογαρίθμου $\ln(x)$ είναι γνησίως αύξουσα:
\[ f(x_0) \ge x_0 \Rightarrow \ln(f(x_0)) \ge \ln(x_0) \]

Όμως δοθέντος ότι $x_0 > 1$, το θεμελιώδες όφελος λέει πως $\ln(x_0) > \ln 1 = 0$.
Άρα $\ln(f(x_0)) \ge \ln(x_0) > 0$.
Προσθέτοντας κατά μέλη:
\[ f(x_0) + \ln(f(x_0)) \ge x_0 + \ln(x_0) \]
Αποφαινόμαστε ωστόσο ότι $x_0 + \ln(x_0) > x_0 + 0 = x_0$.
Συνεπώς:
\[ f(x_0) + \ln(f(x_0)) > x_0 \]
Αυτό έρχεται σε μετωπική και ξεκάθαρη αντίφαση με τη δεδομένη ταυτότητα $f(x) + \ln(f(x)) = x$. Εν κατακλείδι, η υπόθεσή μας καταρρίπτεται, και επαληθεύεται αυστηρά πως $\boldsymbol{f(x) < x}$ για κάθε $x > 1$.

\item Διαμορφώνουμε το προς αναζήτηση όριο $\lim_{x\to+\infty} \frac{f(x)}{x}$.
Από την αρχική σχέση διαιρούμε με το $x$ (για $x > 0$):
\[ f(x) + \ln(f(x)) = x \Rightarrow \frac{f(x)}{x} + \frac{\ln(f(x))}{x} = 1 \Rightarrow \frac{f(x)}{x} = 1 - \frac{\ln(f(x))}{x} \]

(Ασφαλώς, επειδή η $f(x)$ είναι γνησίως αύξουσα, εάν έχει συνέπεια πεπερασμένου ορίου $\ell > 0$ στο άπειρο, τότε $\lim (f(x)+\ln f(x)) = \ell + \ln\ell \neq +\infty = \lim x$. Επομένως $\lim_{x\to+\infty} f(x) = +\infty$).
Υπολογίζουμε το συνοδό όριο του κλάσματος με τον αξιόπιστο κανόνα \eng{De l'Hospital}:
\[ \lim_{x\to+\infty} \frac{\ln(f(x))}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{(\ln(f(x)))'}{(x)'} = \lim_{x\to+\infty} \frac{\frac{f'(x)}{f(x)}}{1} = \lim_{x\to+\infty} \frac{f'(x)}{f(x)} \]

Χρησιμοποιούμε την παράγωγο που διαγνώσαμε στο (1): $f'(x) = \frac{f(x)}{f(x)+1}$:
\[ \lim_{x\to+\infty} \frac{\frac{f(x)}{f(x)+1}}{f(x)} = \lim_{x\to+\infty} \frac{1}{f(x)+1} \]
Επειδή $\lim_{x\to+\infty} f(x) = +\infty$, προκύπτει $\lim_{x\to+\infty} \frac{1}{f(x)+1} = \frac{1}{+\infty} = 0$.

Αντικαθιστώντας στη μετασχηματισμένη διαίρεση, το ολοκληρωτικό όριο γίνεται επιτυχώς:
\[ \lim_{x\to+\infty} \frac{f(x)}{x} = 1 - 0 = \boldsymbol{1} \]

\end{erwthma}
\end{thema}
