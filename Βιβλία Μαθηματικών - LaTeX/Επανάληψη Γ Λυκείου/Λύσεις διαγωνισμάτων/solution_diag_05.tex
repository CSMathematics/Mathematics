%=========== ΛΥΣΗ - 5ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Λ.} 
Αντιπαράδειγμα: Η \textbf{εκθετική συνάρτηση} $f(x) = e^x$ είναι 1-1 (ως γνησίως αύξουσα) αλλά $f''(x) = e^x > 0$ για κάθε $x$, οπότε \textbf{δεν έχει κανένα σημείο καμπής}. Μία συνάρτηση που είναι 1-1 δεν είναι απαραίτητο να αλλάζει κυρτότητα.
\item \textbf{Σ.}
Πρόκειται για την κλασική βασική ιδιότητα των ολοκληρωμάτων (γραμμικότητα): 
$\dint_a^b [f(x)+g(x)]\,\d x = \dint_a^b f(x)\,\d x + \dint_a^b g(x)\,\d x$.
\item \textbf{Λ.}
Η ύπαρξη απόλυτων ακροτάτων εξασφαλίζεται με το θεώρημα Μεγίστου - Ελαχίστου \textbf{μόνο σε κλειστά διαστήματα} $[\alpha,\beta]$. Σε ανοικτό διάστημα τα ακρότατα δεν είναι εγγυημένα.
Αντιπαράδειγμα: η $f(x)=x$ στο $(0,1)$ δεν έχει ούτε μέγιστο ούτε ελάχιστο.
\item \textbf{Λ.}
Μία ρίζα της παραγώγου \textbf{δεν} αποτελεί εγγυημένα θέση τοπικού ακροτάτου. Απαιτείται επιπλέον η παράγωγος \textbf{να αλλάζει πρόσημο} εκατέρωθεν της ρίζας.
Αντιπαράδειγμα: Στην $f(x)=x^3$, η παράγωγος είναι $f'(x)=3x^2$. Έχει ρίζα το $x_0=0$ ($f'(0)=0$), αλλά στο $0$ δεν υπάρχει ακρότατο.
\item \textbf{Σ.}
Ισχύει από τον ορισμό. Αν τα όρια ολοκλήρωσης ταυτίζονται, το εμβαδόν μεταξύ $\alpha$ και $\alpha$ είναι μηδενικό, ανεξάρτητα από τον τύπο της $f$. $\dint_a^a f(x)\,\d x = 0$.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 5ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Η συνάρτηση $f(x) = x^3 + x - 2$, $x \in \mathbb{R}$.
\begin{erwthma}

\item Παραγωγίζουμε:
\[ f'(x) = 3x^2 + 1 \]
Παρατηρούμε ότι $f'(x) > 0$ για κάθε $x \in \mathbb{R}$. Επομένως, η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το $\mathbb{R}$.

Για το \textbf{πεδίο τιμών}:
Η $f$ είναι συνεχής και γνησίως αύξουσα στο $\varDelta = \mathbb{R}$, άρα:
\[ f(\varDelta) = (\lim_{x\to-\infty} f(x), \lim_{x\to+\infty} f(x)) \]
Όπου:
\[ \lim_{x\to-\infty} (x^3 + x - 2) = \lim_{x\to-\infty} x^3 = -\infty \]
\[ \lim_{x\to+\infty} (x^3 + x - 2) = \lim_{x\to+\infty} x^3 = +\infty \]
Άρα το πεδίο τιμών είναι το $f(\mathbb{R}) = \mathbb{R}$.

\textbf{Πλήθος ριζών:}
Επειδή $0 \in f(\mathbb{R})$, η εξίσωση $f(x)=0$ έχει \textbf{τουλάχιστον} μία ρίζα.
Επειδή η $f$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι \textbf{ακριβώς μία}. 
Με προφανή δοκιμή στο $x=1$: $f(1) = 1^3 + 1 - 2 = 0$. Άρα η μοναδική ρίζα είναι το $x=1$.

\item Αφού η $f$ είναι γνησίως αύξουσα στο πεδίο ορισμού της, είναι 1-1. Άρα η συνάρτηση $f$ \textbf{αντιστρέφεται με καθολικό τρόπο}.
Το \textbf{πεδίο ορισμού} της αντίστροφης $f^{-1}$ ταυτίζεται με το σύνολο τιμών της $f$, το οποίο αποδείξαμε ότι είναι το \textbf{$\mathbb{R}$}.

Για τον υπολογισμό των τιμών της αντίστροφης ακολουθούμε τον εξής συλλογισμό:
\begin{itemize}
    \item Έστω $f^{-1}(-2) = y \Leftrightarrow f(y) = -2 \Leftrightarrow y^3 + y - 2 = -2 \Leftrightarrow y^3 + y = 0 \Leftrightarrow y(y^2+1) = 0$.
    Επειδή $y^2+1>0$, προκύπτει υποχρεωτικά $y=0$. Άρα $f^{-1}(-2) = 0$.
    \item Έστω $f^{-1}(8) = z \Leftrightarrow f(z) = 8 \Leftrightarrow z^3 + z - 2 = 8 \Leftrightarrow z^3 + z - 10 = 0$.
    Παρατηρούμε προφανή ρίζα το $z=2$ (αφού $8+2-10=0$). Αφού η $f$ είναι 1-1, αποτελεί τη μοναδική. Άρα $f^{-1}(8) = 2$.
\end{itemize}

\item Θέλουμε να υπολογίσουμε το $\displaystyle\lim_{x \to -2} \frac{f^{-1}(x)}{x + 2}$.
Θέτουμε $x = f(y)$.
Όταν $x \to -2$, τότε $y = f^{-1}(x) \to f^{-1}(-2) = 0$ (έχουμε αξιοποιήσει το γεγονός πως η αντίστροφη είναι συνεχής).
Το όριο μετασχηματίζεται ως κάτωθι:
\[ \lim_{y \to 0} \frac{f^{-1}(f(y))}{f(y) + 2} = \lim_{y \to 0} \frac{y}{y^3 + y - 2 + 2} = \lim_{y \to 0} \frac{y}{y^3 + y} = \lim_{y \to 0} \frac{y}{y(y^2 + 1)} = \lim_{y \to 0} \frac{1}{y^2 + 1} = \frac{1}{0 + 1} = 1 \]
Άρα το όριο είναι ίσο με 1.

\item Το εμβαδόν μεταξύ της καμπύλης $C_{f^{-1}}$, του άξονα των τετμημένων (άξονας $x'x$) και των κατακόρυφων $x=-2$ και $x=8$ δίνεται από το ολοκλήρωμα:
\[ E = \int_{-2}^8 |f^{-1}(x)|\, \d x \]

Παρατηρούμε ότι η συνάρτηση $f^{-1}(x)$ για $x \in [-2,8]$ είναι θετική, διότι η $f$ είναι αύξουσα οπότε η $f^{-1}$ είναι επίσης αύξουσα. Αφού $f^{-1}(-2) = 0$ και $x > -2 \Rightarrow f^{-1}(x) > f^{-1}(-2) = 0$, οπότε $f^{-1}(x) \ge 0$.
Αρκεί λοιπόν να υπολογίσουμε:
\[ E = \int_{-2}^8 f^{-1}(x)\, \d x \]

Θέτουμε $x = f(u)$, τότε $\d x = f'(u)\,\d u$.
Τα άκρα του ολοκληρώματος γίνονται:  
Όταν $x = -2 \Rightarrow f(u) = -2 \Rightarrow u = 0$.
Όταν $x = 8 \Rightarrow f(u) = 8 \Rightarrow u = 2$.

\[ E = \int_0^2 u \cdot f'(u)\, \d u \]
Με ολοκλήρωση κατά παράγοντες (παραγοντική) το ολοκλήρωμα γίνεται:
\begin{align*}
\int_0^2 u \cdot (f(u))'\, \d u &= \left[ u \cdot f(u) \right]_0^2 - \int_0^2 f(u) \cdot (u)'\, \d u \\
&= (2 \cdot f(2) - 0 \cdot f(0)) - \int_0^2 (u^3 + u - 2)\, \d u \\
&= 2 \cdot 8 - \left[ \frac{u^4}{4} + \frac{u^2}{2} - 2u \right]_0^2 \\
&= 16 - \left( \frac{16}{4} + \frac{4}{2} - 4 \right) \\
&= 16 - (4 + 2 - 4) = 16 - 2 = 14 \ \text{τ.μ.}
\end{align*}

\begin{parat}{\linewidth}
Ο \textbf{γενικός κανόνας ολοκλήρωσης αντίστροφης συνάρτησης} $f^{-1}$ απαιτεί πάντοτε σχεδόν αλλαγή μεταβλητής $x = f(u)$ ώστε να επιστρέψουμε στον γνωστό τύπο της $f$. 
\end{parat}

\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Θεωρούμε την συνάρτηση $f(x) = (x-1)\ln x$ ορισμένη στο πεδίο $(0, +\infty)$.
\begin{erwthma}

\item \textbf{Μονοτονία:}
Παραγωγίζουμε στο $(0,+\infty)$:
\[ f'(x) = (x-1)'\ln x + (x-1)(\ln x)' = \ln x + \frac{x-1}{x} = \ln x + 1 - \frac{1}{x} \]

Παρατηρούμε ότι για κάθε $x \in (0, 1)$: $\ln x < 0$ και $\frac{1}{x} > 1 \Rightarrow 1 - \frac{1}{x} < 0$. Άρα $f'(x) < 0$.
Ομοίως, για $x > 1$: $\ln x > 0$ και $\frac{1}{x} < 1 \Rightarrow 1 - \frac{1}{x} > 0$. Άρα $f'(x) > 0$.
Στο σημείο $x = 1$, έχουμε $f'(1) = 0 + 1 - 1 = 0$.

Επομένως, το πρόσημο της $f'$ φαίνεται στον επόμενο πίνακα:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = red7,colorL = red9,colorV = red7]%
{$x$ / .8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$1$,$+\infty$}%
\tkzTabLine{ d, -, z, +, }
\tkzTabLine{ d, \searrow, t, \nearrow, }
\end{tikzpicture}
\end{center}

Συμπεραίνουμε ότι η συνάρτηση είναι γνησίως φθίνουσα στο $(0, 1]$ και γνησίως αύξουσα στο $[1, +\infty)$.
Η $f$ παρουσιάζει \textbf{ολικό ελάχιστο} στο $x=1$ με τιμή $f(1) = 0 \cdot \ln 1 = 0$.

\textbf{Κυρτότητα:}
Υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x) = \left( \ln x + 1 - \frac{1}{x} \right)' = \frac{1}{x} + \frac{1}{x^2} \]
Προφανώς, για κάθε $x > 0$ ισχύει ότι $f''(x) > 0$. Άρα, η συνάρτηση $f$ είναι \textbf{γνησίως κυρτή} σε ολόκληρο το πεδίο ορισμού της.

\item Αφού αποδείξαμε στο (1) ότι στο $x=1$ η $f$ έχει \textbf{ολικό ελάχιστο} ίσο με $0$, κατά συνέπεια συμπεραίνουμε άμεσα ότι για κάθε $x \in D_f = (0,+\infty)$ θα ισχύει:
\[ f(x) \ge f(1) \Rightarrow f(x) \ge 0 \Rightarrow (x-1)\ln x \ge 0 \]
Η ισότητα ισχύει αποκλειστικά για $x=1$, διότι η $f$ είναι 1-1 στους εκατέρωθεν κλάδους μονοτονίας της (κανένα άλλο ακρότατο ούτε σε σταθερό διάστημα).

\item Το χωρίο μεταξύ της καμπύλης $C_f$, του οριζόντιου άξονα $(x'x)$, και των ευθειών $x=1, x=e$ έχει εμβαδόν (εξαιτίας του ότι $f(x) \ge 0$ όπως βρέθηκε στο (2)):
\[ E = \int_1^e f(x)\, \d x = \int_1^e (x-1)\ln x\, \d x \]

Θα εφαρμόσουμε ολοκλήρωση κατά παράγοντες αξιοποιώντας την αρχική του πολυωνύμου:
\begin{align*}
E &= \int_1^e \left( \frac{x^2}{2} - x \right)' \ln x\, \d x \\
&= \left[ \left( \frac{x^2}{2} - x \right) \ln x \right]_1^e - \int_1^e \left( \frac{x^2}{2} - x \right) \frac{1}{x}\, \d x \\
&= \left( \frac{e^2}{2} - e \right)\ln e - \left( \frac{1}{2} - 1 \right)\ln 1 - \int_1^e \left( \frac{x}{2} - 1 \right)\, \d x \\
&= \left( \frac{e^2}{2} - e \right) - 0 - \left[ \frac{x^2}{4} - x \right]_1^e \\
&= \frac{e^2}{2} - e - \left( \left( \frac{e^2}{4} - e \right) - \left( \frac{1}{4} - 1 \right) \right) \\
&= \frac{e^2}{2} - e - \frac{e^2}{4} + e - \frac{3}{4} \\
&= \frac{e^2}{4} - \frac{3}{4} = \frac{e^2 - 3}{4} \ \text{τ.μ.}
\end{align*}

\item Αναζητούμε σημείο $M(x_0, f(x_0))$ με $x_0>1$ τέτοιο ώστε η εφαπτομένη να είναι παράλληλη προς την $\varepsilon: y = x - 2026$. Η κλίση της $\varepsilon$ είναι $\lambda = 1$.
Επομένως, οφείλει να ισχύει $f'(x_0) = \lambda \Rightarrow f'(x_0) = 1$.

Λύνουμε την εξίσωση:
\[ \ln x_0 + 1 - \frac{1}{x_0} = 1 \Leftrightarrow \ln x_0 - \frac{1}{x_0} = 0 \Leftrightarrow x_0 \ln x_0 - 1 = 0 \quad (\text{αφού } x_0 > 0) \]
Θέτουμε τη βοηθητική συνάρτηση $h(x) = x \ln x - 1$.
Παραγωγίζουμε την $h$: $h'(x) = \ln x + x \frac{1}{x} = \ln x + 1$.
Για $x>1$: $\ln x > 0 \Rightarrow h'(x) > 1 > 0$, επομένως η $h$ είναι γνησίως αύξουσα στο $[1,+\infty)$.
Η εξίσωση $h(x_0) = 0$ δίνει ρίζα.
Ελέγχουμε στο διάστημα $[1,e]$:
\begin{itemize}
    \item $h(1) = 1\cdot 0 - 1 = -1 < 0$
    \item $h(e) = e \cdot 1 - 1 = e - 1 > 0$ (καθώς $e > 2,71$)
\end{itemize}
Εφόσον η $h$ είναι συνεχής στο $[1,e]$, κατόπιν εφαρμογής του \textbf{θεωρήματος \eng{Bolzano}}, υπάρχει $x_0 \in (1,e)$ τέτοιο ώστε $h(x_0) = 0 \Leftrightarrow f'(x_0) = 1$.
Η ρίζα είναι \textbf{μοναδική} διότι η $h$ είναι γνησίως αύξουσα.

\begin{prosoxi}{\linewidth}
Είναι σύνηθες σε θέματα να μη λύνεται μεταβατική ή λογαριθμική εξίσωση αναλυτικά (όπως η $x \ln x = 1$), οπότε εμπόδίζετε επιτυχώς την αλγεβρική επίλυση καταφεύγοντας στο θεώρημα \eng{Bolzano} προκειμένου να επικυρώσετε την ύπαρξη λύσης.
\end{prosoxi}

\end{erwthma}
\end{thema}

\begin{thema}{Δ}
$f(x) + \hm(f(x)) = x$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}

\item Η δοσμένη σχέση έχει τον τύπο $y + \hm y = x$ (όπου $y = f(x)$).
Θεωρούμε τη βοηθητική συνάρτηση $g(y) = y + \hm y$, ορισμένη για κάθε $y \in \mathbb{R}$.
\[ g'(y) = 1 + \syn y \]
Γνωρίζουμε από τον τριγωνομετρικό κύκλο ότι $\syn y \ge -1 \Rightarrow 1 + \syn y \ge 0$.
Η παράγωγος μηδενίζεται αποκλειστικά στα απομονωμένα σημεία $y = (2k+1)\pi, k \in \mathbb{Z}$ και είναι $g'(y) > 0$ στο ενδιάμεσο. Ένεκα αυτού, η συνάρτηση $g$ δεν παραμένει σταθερή σε απολύτως κανένα υποδιάστημα, οπότε είναι \textbf{γνησίως αύξουσα} σε ολόκληρο το $\mathbb{R}$.

Αφού η $g$ είναι γνησίως αύξουσα, είναι 1-1 και κατά συνέπεια \textbf{αντιστρέψιμη}.
Η αρχική σχέση γράφεται $g(f(x)) = x$.
Αφού προκύπτει $f(x) = g^{-1}(x)$, η $f$ ορίζεται \textbf{μονοσήμαντα} διότι αποτελεί την αντίστροφη μίας γνησίως αύξουσας και πλήρως ορισμένης κατασκευής.

\textbf{Υπολογισμός $f(\pi)$:}
Αναζητούμε το $y_0 = f(\pi)$. Με βάση τη λειτουργία της συνάρτησης θα ισχύει: $g(y_0) = \pi \Rightarrow y_0 + \hm y_0 = \pi$.
Με προφανή ρίζα βλέπουμε $y_0 = \pi$, διότι $\pi + \hm\pi = \pi + 0 = \pi$. Εξαιτίας του ότι η $g$ είναι 1-1, η ρίζα είναι αναγκαστικά μοναδική. Άρα $\boxed{f(\pi) = \pi}$.

\item \underline{$1$\textsuperscript{ος} τρόπος (Μέσω $g^{-1}$):}
Αποδείξαμε πριν ότι η συνάρτηση $g(y) = y+\hm{y}$ είναι γνησίως αύξουσα και $f(x) = g^{-1}(x)$. Γνωρίζουμε μαθηματικά πως η αντίστροφη μίας γνησίως αύξουσας συνάρτησης είναι ομοίως \textbf{γνησίως αύξουσα}. Άρα η $f$ αυξάνεται γνήσια σε όλο το $\mathbb{R}$.

\underline{$2$\textsuperscript{ος} τρόπος (Μέσω παραγώγισης):}
Παραγωγίζουμε τα μέλη της σχέσης $f(x) + \hm(f(x)) = x$:
\begin{align*}
f'(x) + \syn(f(x)) f'(x) &= 1 \\
f'(x)(1 + \syn(f(x))) &= 1
\end{align*}
Έχουμε $1+\syn(f(x)) \ge 0$. Αν ποτέ μηδενιζόταν, το 1ο μέλος θα ήταν $0$, ενώ το 2ο μέλος είναι $1$, άτοπο. Προκύπτει $1+\syn(f(x)) > 0$ για κάθε $x \in \mathbb{R}$.
Οπότε:
\[ f'(x) = \frac{1}{1+\syn(f(x))} > 0 \]
Καταλήγουμε επομένως πως η $f$ είναι γνησίως αύξουσα.

\item Θέλουμε να αποδείξουμε την ανισότητα διπλής προσέγγισης, δηλαδή $0 < f(x) < x$ για $x \in (0,\pi)$.
Καταρχάς, έχουμε υπολογίσει:
\begin{itemize}
    \item $f(0)$: $g(f(0)) = 0 \Leftrightarrow f(0) + \hm(f(0)) = 0$. Η σχέση έχει προφανή μοναδική (λόγω 1-1) ρίζα το $f(0) = 0$.
    \item $f(\pi) = \pi$ όπως αποδείχθηκε αναλυτικά από το 1ο ερώτημα.
\end{itemize}

Επειδή η $f(x)$ είναι γνησίως αύξουσα στο πεδίο $\mathbb{R}$:
Για κάθε $x \in (0,\pi)$ ισχύει: $0 < x < \pi \stackrel{f \nearrow}{\Longrightarrow} f(0) < f(x) < f(\pi) \Rightarrow 0 < f(x) < \pi$.
Άρα αποδείξαμε το πρώτο (αριστερό) σκέλος, δηλαδή ότι \textbf{$f(x) > 0$}.

Για το δεύτερο (δεξί) σκέλος, $f(x) < x$:
Ανακαλούμε την αρχική σχέση, $f(x) = x - \hm(f(x))$.
Με βάση τα δείγματα προηγουμένως, για κάθε $x \in (0,\pi)$, μάθαμε ότι $0 < f(x) < \pi$.
Στο εσωτερικό του 1ου και 2ου τεταρτημορίου (μεταξύ $0$ και $\pi$), το ημίτονο είναι πάντοτε αυστηρά θετικό: 
Άρα $\hm(f(x)) > 0$.
Από αυτό, έπεται:
\[ x - f(x) = \hm(f(x)) > 0 \Rightarrow x > f(x) \Rightarrow f(x) < x \]
Η ανισότητα επομένως αποδείχθηκε αυστηρά: $0 < f(x) < x$.

\item Από το προηγούμενο ερώτημα $\Delta_3$ προέκυψε ότι $f(x) < x$ για ολόκληρο το ανοικτό διάστημα $x \in (0,\pi)$.
Σε συνδυασμό με την επιβεβαιωμένη συνέχεια της $f$, εάν ολοκληρώσουμε κατά μέλη σε αυτό το ορισμένο πεδίο:
\begin{align*}
\int_0^\pi f(x)\, \d x &< \int_0^\pi x\, \d x \\
\int_0^\pi f(x)\, \d x &< \left[ \frac{x^2}{2} \right]_0^\pi \\
\int_0^\pi f(x)\, \d x &< \frac{\pi^2}{2} - \frac{0}{2} \\
\int_0^\pi f(x)\, \d x &< \frac{\pi^2}{2}
\end{align*}
Το ζητούμενο κατά συνέπεια αποδείχθηκε επιτυχώς.

\end{erwthma}
\end{thema}
