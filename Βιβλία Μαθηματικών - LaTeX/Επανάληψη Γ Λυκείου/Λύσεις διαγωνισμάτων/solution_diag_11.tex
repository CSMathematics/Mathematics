%=========== ΛΥΣΗ - 11ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Σ.} 
Είναι βασικό θεώρημα. Η παραγωγισιμότητα (ύπαρξη ορίου λόγου μεταβολής) σε ένα σημείο εξασφαλίζει απαρέγκλιτα και τη συνέχεια της συνάρτησης στο ίδιο σημείο. (Το αντίστροφο είδαμε ότι δεν ισχύει).
\item \textbf{Σ.}
Πρόκειται για την ιδιότητα διατήρησης της διάταξης στα \textbf{ορισμένα ολοκληρώματα}: $f(x) \leq g(x) \Rightarrow \dint_\alpha^\beta f(x)\,\d x \leq \dint_\alpha^\beta g(x)\,\d x$.
\item \textbf{Λ.}
Μια τμηματικά (πολλαπλού τύπου) συνάρτηση \textbf{μπορεί} κάλλιστα να είναι παραγωγίσιμη στα σημεία αλλαγής, υπό την προϋπόθεση ότι οι πλευρικές παράγωγοι / πλευρικά όρια του λόγου μεταβολής συμπίπτουν.
\item \textbf{Λ.}
Η συνθήκη $f'(x_0) = 0$ δίνει \textbf{πιθανή θέση} ακροτάτου, όχι σίγουρη. Αν η παράγωγος δεν αλλάζει πρόσημο εκατέρωθεν (όπως η $f(x)=x^3$ στο $0$), ακρότατο δεν υφίσταται.
\item \textbf{Σ.}
Αποτελεί την \textbf{προσθετική ιδιότητα} του ορισμένου ολοκληρώματος ως προς τα διαδοχικά διαστήματα ολοκλήρωσης (για το ίδιο ακριβώς ολοκλήρωμα).
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 11ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Η συνάρτηση $f(x)$ είναι τριπλού τύπου $f:[-3, 4]\to\mathbb{R}$:
\[
f(x) = \begin{cases} x+1, & -3\leq x \leq 0 \\ -x^2+2x+1, & 0 < x \leq 2 \\ -1, & 2 < x \leq 4 \end{cases}
\]
\begin{erwthma}

\item \textbf{Συνέχεια στο $x=0$:}
- $f(0) = 0+1 = 1$.
- Αριστερό όριο: $\lim_{x\to 0^-} f(x) = \lim_{x\to 0^-} (x+1) = 1$.
- Δεξιό όριο: $\lim_{x\to 0^+} f(x) = \lim_{x\to 0^+} (-x^2+2x+1) = 1$.
Λόγω ταύτισης, η $f$ είναι \textbf{συνεχής} στο $x=0$.

\textbf{Συνέχεια στο $x=2$:}
- $f(2) = -2^2 + 2\cdot2 + 1 = -4 + 4 + 1 = 1$.
- Αριστερό όριο: $\lim_{x\to 2^-} f(x) = \lim_{x\to 2^-} (-x^2+2x+1) = 1$.
- Δεξιό όριο: $\lim_{x\to 2^+} f(x) = \lim_{x\to 2^+} (-1) = -1$.
Αφού $1 \neq -1$, τα πλευρικά όρια δε συμπίπτουν, οπότε η $f$ \textbf{δεν είναι συνεχής} στο $x=2$. 

\begin{prosoxi}{\linewidth}
Παρότι η εκφώνηση αναφέρει εσφαλμένα τη φράση "συνεχούς συνάρτησης f", η καθαρή αλγεβρική πράξη αποδεικνύει ότι φέρει \textbf{σημείο ασυνέχειας (άλμα)} στο $x=2$. Η εξέταση υπακούει αυστηρά στους υπολογισμούς των ορίων.
\end{prosoxi}

\item \textbf{Ολικά Ακρότατα:}
Αναζητούμε τα σύνολα τιμών στους επιμέρους κλάδους:
\begin{itemize}
    \item $[ -3, 0]$: Είναι ευθύγραμμο αύξον, άρα πεδίο τιμών $[ -3+1, 0+1 ] = [ -2, 1]$.
    \item $(0, 2]$: Παραβολή που κοιτάζει κάτω. Κορυφή: παράγωγος $-2x+2 = 0 \Rightarrow x=1$. Τιμή στο $x=1$: $f(1) = -1+2+1 = 2$.
    Στο όριο από $0$ (τιμή 1) αυξάνεται στο $2$ και φθίνει στο $2$ (τιμή 1). Άρα πεδίο τιμών το $(1, 2]$.
    \item $(2, 4]$: Είναι σταθερή στο $-1 \Rightarrow \{-1\}$.
\end{itemize}

Οι παραγόμενες τιμές συναθροίζονται στο συνολικό διάστημα $[-2, 2]$. 
Εφόσον η μέγιστη τιμή που λαμβάνει είναι το $2$ και η ελάχιστη το $-2$, καταλήγουμε:
- \textbf{Ολικό μέγιστο:} $\boldsymbol{2}$, το οποίο παρατηρείται στο $x = 1$.
- \textbf{Ολικό ελάχιστο:} $\boldsymbol{-2}$, το οποίο παρατηρείται στο $x = -3$.

\item Υπολογίζουμε το ολοκλήρωμα σπάζοντας την επιφάνεια στα υποδιαστήματα ορισμού:
\begin{align*}
I &= \int_{-3}^4 f(x)\,\d x = \int_{-3}^0 f(x)\,\d x + \int_0^2 f(x)\,\d x + \int_2^4 f(x)\,\d x \\
&= \int_{-3}^0 (x+1)\,\d x + \int_0^2 (-x^2+2x+1)\,\d x + \int_2^4 (-1)\,\d x \\
&= \left[ \frac{x^2}{2} + x \right]_{-3}^0 + \left[ -\frac{x^3}{3} + x^2 + x \right]_0^2 + \left[ -x \right]_2^4 \\
&= \left( 0 - \left( \frac{9}{2} - 3 \right) \right) + \left( -\frac{8}{3} + 4 + 2 - 0 \right) + ( -4 - (-2) ) \\
&= \left( - \frac{3}{2} \right) + \left( 6 - \frac{8}{3} \right) + ( -2 ) \\
&= -\frac{3}{2} + \frac{10}{3} - 2 = -\frac{9}{6} + \frac{20}{6} - \frac{12}{6} = \boldsymbol{-\frac{1}{6}}
\end{align*}

\item Παραγωγισιμότητα στο $x=0$:
Ορίζουμε την παράγωγο της $f$ για τα εσωτερικά σημεία:
\begin{itemize}
    \item Για $-3 < x < 0$: $f'(x) = 1$.
    \item Για $0 < x < 2$: $f'(x) = -2x+2$.
\end{itemize}
Εξετάζουμε τα αναλυτικά πλευρικά όρια της δοθείσας παραγώγου:
- Αριστερό όριο παραγώγου: $\lim_{x\to 0^-} f'(x) = \lim_{x\to 0^-} 1 = 1$.
- Δεξιό όριο παραγώγου: $\lim_{x\to 0^+} f'(x) = \lim_{x\to 0^+} (-2x+2) = 2$.
Επειδή $1 \neq 2$, η συνάρτηση δημιουργεί «\textbf{γωνιακό σημείο}» και επομένως \textbf{δεν είναι παραγωγίσιμη} στο σημείο $x=0$.
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 11ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Δίνεται $f(x) = xe^{-x} + 1$, $x\in\mathbb{R}$.
\begin{erwthma}

\item \textbf{Μονοτονία και Ακρότατα:}
Η $f$ παραγωγίζεται στο $\mathbb{R}$:
\[ f'(x) = (x)' e^{-x} + x (e^{-x})' = 1 \cdot e^{-x} - x e^{-x} = e^{-x} (1 - x) \]

Μηδενίζουμε: $f'(x) = 0 \Leftrightarrow 1 - x = 0 \Leftrightarrow x = 1$ (αφού $e^{-x} > 0$).
Η $f'(x) > 0$ αποκλειστικά για $x < 1$ και $f'(x) < 0$ για $x > 1$.

Η $f$ είναι \textbf{γνησίως αύξουσα} στο $(-\infty, 1]$ και \textbf{γνησίως φθίνουσα} στο $[1, +\infty)$.
Η $f$ παρουσιάζει \textbf{ολικό μέγιστο} στο σημείο $x = 1$, με ενεργή τιμή:
\[ f(1) = 1 \cdot e^{-1} + 1 = \frac{1}{e} + 1 = \frac{e+1}{e} \]

\item \textbf{Ασύμπτωτες:}
$\bullet$ Κατακόρυφες: Το πεδίο ορισμού είναι όλο το $\mathbb{R}$ και η $f$ είναι συνεχής, άρα \textbf{δεν υπάρχουν}.
$\bullet$ Οριζόντιες (όρια στο άπειρο):
Στο $-\infty$:
\[ \lim_{x\to -\infty} (xe^{-x} + 1) = (-\infty) \cdot (+\infty) + 1 = -\infty \]
Άρα ούτε εκεί υφίσταται οριζόντια/πλάγια.

Στο $+\infty$:
\[ \lim_{x\to+\infty} (xe^{-x} + 1) = \lim_{x\to+\infty} \frac{x}{e^x} + 1 \]
Επεξεργαζόμαστε το όριο με \eng{D.L.H.}:
\[ \lim_{x\to+\infty} \frac{x}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{1}{e^x} = 0 \]
Συνολικά, $\lim_{x\to+\infty} f(x) = 0 + 1 = 1$.
Επομένως, η ευθεία \textbf{$\boldsymbol{y = 1}$} είναι \textbf{οριζόντια ασύμπτωτη} της γραφικής παράστασης στο $+\infty$.

\item Διαθέτουμε την εξίσωση $x e^{-x} = 0$.
Επειδή η εκθετική συνάρτηση αληθεύει ισχυρά πως $e^{-x} > 0$ για κάθε $x \in \mathbb{R}$, η εξίσωση απλοποιείται δια της διαίρεσης:
\[ x e^{-x} = 0 \Leftrightarrow \boldsymbol{x = 0} \]
Η ρίζα είναι καθαρά επιλύσιμη και προφανώς αποδεικνύεται \textbf{μοναδική}. (Δε χρειαζόμαστε πολυπλοκότητα στο $y=1$).

\item Ολοκλήρωμα (ολοκλήρωση κατά παράγοντες διότι το $(e^{-x})' = -e^{-x}$):
\begin{align*}
I &= \int_0^1 f(x)\,\d x = \int_0^1 (xe^{-x} + 1)\,\d x = \int_0^1 x e^{-x}\,\d x + \int_0^1 1\,\d x \\
&= \int_0^1 x (-e^{-x})'\,\d x + [x]_0^1 \\
&= \left[ -x e^{-x} \right]_0^1 - \int_0^1 (x)' (-e^{-x})\,\d x + 1 \\
&= (-1 e^{-1} - 0) + \int_0^1 e^{-x}\,\d x + 1 \\
&= -\frac{1}{e} + \left[ -e^{-x} \right]_0^1 + 1 \\
&= -\frac{1}{e} + \left( -e^{-1} - (-e^0) \right) + 1 \\
&= -\frac{1}{e} - \frac{1}{e} + 1 + 1 = \boldsymbol{2 - \frac{2}{e}}
\end{align*}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 11ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Δίνεται $f:\mathbb{R}\to\mathbb{R}$ με $f(0)=0, f(2)=2$ και $f'(x) > 0$.
\begin{erwthma}

\item Διότι $f'(x) > 0$ για κάθε $x \in \mathbb{R}$, η συνάρτηση κατατάσσεται στις \textbf{γνησίως αύξουσες} στο πεδίο της $\mathbb{R}$.
Όλες οι γνησίως μονότονες συναρτήσεις λαμβάνουν κάθε τιμή του πεδίου τους απολύτως μία φορά, είναι συνεπώς ιδιότητας \textbf{1-1}. 
Εξαιτίας αυτής της ιδιότητας, εξάγεται ευθέως ότι η $f$ \textbf{αντιστρέφεται}.

\item Η συνάρτηση $f$ συγκεντρώνει τα εξής γνωρίσματα:
- Είναι δεδομένα \textbf{συνεχής} στο κλειστό $[0,2]$ (ως παραγωγίσιμη παντού).
- Είναι δεδομένα \textbf{παραγωγίσιμη} στο ανοικτό $(0,2)$.
Άρα πληροί στο ακέραιο τις αυστηρές προϋποθέσεις του \textbf{Θεωρήματος Μέσης Τιμής (\eng{M.V.T.})}.
Συνεπώς, υπάρχει δομικά τουλάχιστον ένα $\xi \in (0,2)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(2) - f(0)}{2 - 0} \]
Αντικαθιστώντας τα εξωτερικά δεδομένα ($f(2)=2$ και $f(0)=0$):
\[ f'(\xi) = \frac{2 - 0}{2} = 1 \]
Το ζητούμενο $\boldsymbol{f'(\xi) = 1}$ αποδείχθηκε.

\item Διαθέτουμε τη $g(x) = f(x) - x$. 
Αξιολογούμε τις τιμές στα άκρα:
- $g(0) = f(0) - 0 = 0 - 0 = 0$.
- $g(2) = f(2) - 2 = 2 - 2 = 0$.
Διασφαλίζουμε ότι:  
1. Η $g$ είναι συνεχής στο $[0,2]$.
2. Η $g$ είναι παραγωγίσιμη στο $(0,2)$ με παράγωγο $g'(x) = f'(x) - 1$.
3. $g(0) = g(2)$.
Ικανοποιούνται οι προϋποθέσεις του \textbf{Θεωρήματος \eng{Rolle}}. Επομένως υπάρχει $\xi \in (0,2)$ τέτοιο ώστε $g'(\xi) = 0 \Rightarrow f'(\xi) - 1 = 0 \Rightarrow f'(\xi) = 1$.

\textbf{Γεωμετρική ερμηνεία:} Το γεγονός ότι $f'(\xi)=1$ σημαίνει ότι υπάρχει ένα τουλάχιστον σημείο $(\xi, f(\xi))$ στο γράφημα, η εφαπτομένη του οποίου έχει κλίση $\lambda=1$. Δηλαδή είναι \textbf{ακριβώς παράλληλη στην ευθεία $y=x$} (τη διχοτόμο), και ιδανικά με την τέμνουσα που ενώνει τα άκρα $(0,0)$ και $(2,2)$.

\item Αναζητούμε την ταυτότητα των εμβαδών:
\[ \int_0^2 f(x)\,\d x + \int_0^2 f^{-1}(y)\,\d y = 4 \]
Επεμβαίνουμε στο δεύτερο ολοκλήρωμα $\int_0^2 f^{-1}(y)\,\d y$, εφαρμόζοντας αναδρομική ολοκλήρωση με αντικατάσταση. Θέτουμε τη μεταβλητή:
$y = f(t)$, άρα το διαφορικό δίνει $\d y = f'(t)\,\d t$.
Βρίσκουμε τα νέα άκρα ολοκλήρωσης (για το $t$):
- Όταν $y = 0 \Rightarrow f(t) = 0$. Επειδή $f(0) = 0$ και η $f$ είναι 1-1, έπεται $\boldsymbol{t = 0}$.
- Όταν $y = 2 \Rightarrow f(t) = 2$. Επειδή $f(2) = 2$ και η $f$ είναι 1-1, έπεται $\boldsymbol{t = 2}$.

Άρα το ολοκλήρωμα μετατρέπεται (\eng{substitution}):
\[ \int_0^2 f^{-1}(f(t)) \cdot f'(t)\,\d t = \int_0^2 t \cdot f'(t)\,\d t \]
Χρησιμοποιούμε ολοκλήρωση κατά παράγοντες στο νέο γινόμενο:
\begin{align*}
\int_0^2 t \cdot (f(t))'\,\d t &= \left[ t \cdot f(t) \right]_0^2 - \int_0^2 (t)' f(t)\,\d t \\
&= 2 \cdot f(2) - 0 \cdot f(0) - \int_0^2 f(t)\,\d t \\
&= 2 \cdot 2 - 0 - \int_0^2 f(t)\,\d t \\
&= 4 - \int_0^2 f(t)\,\d t 
\end{align*}
Επειδή η μεταβλητή του ορισμένου ολοκληρώματος είναι βουβή (\eng{dummy}), το $\int f(t)\d t$ ταυτίζεται με το $\int f(x)\d x$. Συνολικά αντικαθιστώντας:
\[ \int_0^2 f(x)\,\d x + \left( 4 - \int_0^2 f(x)\,\d x \right) = 4 \]
Η ταυτότητα αποδείχθηκε αυστηρά $\boldsymbol{4 = 4}$.
\end{erwthma}
\end{thema}
