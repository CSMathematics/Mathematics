\begin{thema}{Α}
\begin{erwthma}
\item Επειδή η $f$ είναι συνεχής στο $[a,\beta]$, από το Θεώρημα Μέγιστης-Ελάχιστης Τιμής παίρνει στο $[a,\beta]$ μέγιστη και ελάχιστη τιμή.

Αν η $f$ είναι σταθερή στο $[a,\beta]$, τότε
\[
f'(x)=0
\]
για κάθε $x\in(a,\beta)$, οπότε το συμπέρασμα ισχύει.

Αν η $f$ δεν είναι σταθερή, τότε, επειδή
\[
f(a)=f(\beta),
\]
τουλάχιστον μία από τις μέγιστες ή ελάχιστες τιμές της λαμβάνεται σε εσωτερικό σημείο $\xi\in(a,\beta)$.

Στο σημείο αυτό η $f$ παρουσιάζει τοπικό ακρότατο και, επειδή είναι παραγωγίσιμη στο $(a,\beta)$, από το Θεώρημα Fermat έχουμε
\[
f'(\xi)=0.
\]
Άρα υπάρχει τουλάχιστον ένα
\[
\xi\in(a,\beta)
\]
τέτοιο ώστε
\[
{f'(\xi)=0}.
\]

\item Μια συνάρτηση $f$ λέγεται γνησίως φθίνουσα σε ένα διάστημα $\varDelta$, όταν για κάθε $x_1,x_2\in\varDelta$ με
\[
x_1<x_2
\]
ισχύει
\[
f(x_1)>f(x_2).
\]

\item Αν η $f$ είναι συνεχής στο $[a,\beta]$ και $F$ είναι μια παράγουσα της $f$ στο $[a,\beta]$, τότε το ορισμένο ολοκλήρωμα της $f$ από το $a$ έως το $\beta$ ορίζεται από τη σχέση
\[
\dint_a^\beta f(x)\,\d x=F(\beta)-F(a).
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Σωστό}. Αν μια συνάρτηση είναι παραγωγίσιμη σε ένα σημείο, τότε είναι και συνεχής στο σημείο αυτό.
\item \textbf{Σωστό}. Από τη μονοτονία του ορισμένου ολοκληρώματος, αν $f(x)\leq g(x)$ για κάθε $x\in[a,\beta]$, τότε
\[
\dint_a^\beta f(x)\,\d x\leq\dint_a^\beta g(x)\,\d x.
\]
\item \textbf{Λάθος}. Μια τμηματικά ορισμένη συνάρτηση μπορεί να είναι παραγωγίσιμη σε σημείο αλλαγής τύπου, αρκεί να συμπίπτουν οι πλευρικές παράγωγοι.
\item \textbf{Λάθος}. Η συνθήκη $f'(x_0)=0$ είναι αναγκαία για τοπικό ακρότατο όταν η $f$ είναι παραγωγίσιμη, αλλά δεν είναι ικανή. Για παράδειγμα, η $f(x)=x^3$ έχει $f'(0)=0$, χωρίς να έχει ακρότατο στο $0$.
\item \textbf{Σωστό}. Ισχύει η αθροιστική ιδιότητα
\[
\dint_a^\gamma f(x)\,\d x
=
\dint_a^\beta f(x)\,\d x
+
\dint_\beta^\gamma f(x)\,\d x.
\]
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=
\begin{cases}
x+1, & -3\leq x\leq0,\\
-x^2+2x+1, & 0<x\leq2,\\
-x+3, & 2<x\leq4.
\end{cases}
\]
\begin{erwthma}
\item Στο $x=0$ έχουμε
\[
f(0)=0+1=1.
\]
Επίσης
\[
\lim_{x\to0^-}f(x)=\lim_{x\to0^-}(x+1)=1
\]
και
\[
\lim_{x\to0^+}f(x)=\lim_{x\to0^+}(-x^2+2x+1)=1.
\]
Άρα
\[
\lim_{x\to0}f(x)=f(0)=1,
\]
οπότε η $f$ είναι συνεχής στο $x=0$.

Στο $x=2$ έχουμε
\[
f(2)=-2^2+2\cdot2+1=1.
\]
Επίσης
\[
\lim_{x\to2^-}f(x)=\lim_{x\to2^-}(-x^2+2x+1)=1
\]
και
\[
\lim_{x\to2^+}f(x)=\lim_{x\to2^+}(-x+3)=1.
\]
Άρα
\[
\lim_{x\to2}f(x)=f(2)=1,
\]
οπότε η $f$ είναι συνεχής στο $x=2$.

\item Στο διάστημα $[-3,0]$ ο τύπος $x+1$ είναι γνησίως αύξων και παίρνει τιμές από
\[
f(-3)=-2
\]
έως
\[
f(0)=1.
\]

Στο $(0,2]$ έχουμε
\[
f(x)=-x^2+2x+1=-(x-1)^2+2.
\]
Άρα η μέγιστη τιμή του κλάδου είναι
\[
2,
\]
η οποία λαμβάνεται στο $x=1$.

Στο $(2,4]$ ο τύπος $-x+3$ είναι γνησίως φθίνων και οι τιμές του κυμαίνονται από τιμές κοντά στο $1$ έως
\[
f(4)=-1.
\]

Συνεπώς το ολικό μέγιστο της $f$ στο $[-3,4]$ είναι
\[
{2}
\]
και παρουσιάζεται στο
\[
{x=1}.
\]
Το ολικό ελάχιστο είναι
\[
{-2}
\]
και παρουσιάζεται στο
\[
{x=-3}.
\]

\item Σπάμε το ολοκλήρωμα στα τρία διαστήματα:
\begin{align*}
\dint_{-3}^{4}f(x)\,\d x
&=\dint_{-3}^{0}(x+1)\,\d x
+\dint_{0}^{2}(-x^2+2x+1)\,\d x
+\dint_{2}^{4}(-x+3)\,\d x.
\end{align*}
Υπολογίζουμε:
\[
\dint_{-3}^{0}(x+1)\,\d x
=\left[\frac{x^2}{2}+x\right]_{-3}^{0}
=-\frac32,
\]
\[
\dint_{0}^{2}(-x^2+2x+1)\,\d x
=\left[-\frac{x^3}{3}+x^2+x\right]_{0}^{2}
=\frac{10}{3},
\]
και
\[
\dint_{2}^{4}(-x+3)\,\d x
=\left[-\frac{x^2}{2}+3x\right]_{2}^{4}
=0.
\]
Άρα
\[
\dint_{-3}^{4}f(x)\,\d x
=-\frac32+\frac{10}{3}
=\frac{11}{6}.
\]
Επομένως
\[
{\dint_{-3}^{4}f(x)\,\d x=\frac{11}{6}}.
\]

\item Για $-3<x<0$ έχουμε
\[
f'(x)=1.
\]
Άρα
\[
\lim_{x\to0^-}f'(x)=1.
\]

Για $0<x<2$ έχουμε
\[
f'(x)=-2x+2.
\]
Άρα
\[
\lim_{x\to0^+}f'(x)=2.
\]
Επειδή
\[
\lim_{x\to0^-}f'(x)\neq \lim_{x\to0^+}f'(x),
\]
η $f$ δεν είναι παραγωγίσιμη στο $x=0$.
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνάρτηση
\[
f(x)=xe^{-x}+1,\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=e^{-x}-xe^{-x}=e^{-x}(1-x).
\]
Επειδή $e^{-x}>0$, το πρόσημο της $f'$ καθορίζεται από το $1-x$.

Έχουμε
\[
f'(x)=0\Leftrightarrow x=1.
\]
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.7,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$1$,$+\infty$}%
\tkzTabLine{,+,z,-,}
\tkzTabLine{,\nearrow,t,\searrow,}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι γνησίως αύξουσα στο $(-\infty,1]$ και γνησίως φθίνουσα στο $[1,+\infty)$.

Επομένως παρουσιάζει ολικό μέγιστο στο $x=1$, με τιμή
\[
f(1)=1+\frac1e.
\]

\item Η $f$ είναι συνεχής σε όλο το $\mathbb{R}$, άρα δεν έχει κατακόρυφες ασύμπτωτες.

Για $x\to+\infty$ έχουμε
\[
\lim_{x\to+\infty}xe^{-x}
=\lim_{x\to+\infty}\frac{x}{e^x}=0.
\]
Άρα
\[
\lim_{x\to+\infty}f(x)=1,
\]
οπότε η ευθεία
\[
{y=1}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

Για $x\to-\infty$ έχουμε
\[
xe^{-x}\to-\infty,
\]
οπότε
\[
f(x)\to-\infty.
\]
Δεν υπάρχει οριζόντια ασύμπτωτη στο $-\infty$.

Επίσης, για πλάγια ασύμπτωτη στο $-\infty$ εξετάζουμε
\[
\lim_{x\to-\infty}\frac{f(x)}{x}
=\lim_{x\to-\infty}\left(e^{-x}+\frac1x\right)
=+\infty.
\]
Άρα δεν υπάρχει πλάγια ασύμπτωτη στο $-\infty$.

\item Από το πρώτο ερώτημα η $f$ είναι γνησίως αύξουσα στο $(-\infty,1]$ και
\[
\lim_{x\to-\infty}f(x)=-\infty,
\quad
f(1)=1+\frac1e>0.
\]
Άρα, από το Θεώρημα Bolzano, η εξίσωση
\[
f(x)=0
\]
έχει τουλάχιστον μία ρίζα στο $(-\infty,1)$.

Επειδή η $f$ είναι γνησίως αύξουσα στο $(-\infty,1]$, η ρίζα αυτή είναι μοναδική στο $(-\infty,1]$.

Για $x\geq1$ η $f$ είναι γνησίως φθίνουσα και
\[
\lim_{x\to+\infty}f(x)=1>0,
\]
ενώ
\[
f(1)>0.
\]
Άρα δεν έχει ρίζα στο $[1,+\infty)$.

Συνεπώς η εξίσωση $f(x)=0$ έχει ακριβώς μία ρίζα στο $\mathbb{R}$.

\item Υπολογίζουμε:
\[
\dint_0^1 f(x)\,\d x
=\dint_0^1(xe^{-x}+1)\,\d x.
\]
Για το ολοκλήρωμα
\[
\dint_0^1 xe^{-x}\,\d x
\]
κάνουμε ολοκλήρωση κατά παράγοντες, με
\[
u=x,\quad \d v=e^{-x}\,\d x.
\]
Τότε
\[
\d u=\d x,\quad v=-e^{-x}.
\]
Άρα
\begin{align*}
\dint_0^1 xe^{-x}\,\d x
&=\left[-xe^{-x}\right]_0^1+\dint_0^1 e^{-x}\,\d x\\
&=-\frac1e+\left[-e^{-x}\right]_0^1\\
&=-\frac1e+\left(-\frac1e+1\right)\\
&=1-\frac{2}{e}.
\end{align*}
Επομένως
\[
\dint_0^1 f(x)\,\d x
=1-\frac{2}{e}+1
=2-\frac{2}{e}.
\]
Άρα
\[
{\dint_0^1 f(x)\,\d x=2-\frac{2}{e}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
f:\mathbb{R}\to\mathbb{R}
\]
με
\[
f(0)=0,\quad f(2)=2
\]
και
\[
f'(x)>0
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Επειδή
\[
f'(x)>0
\]
για κάθε $x\in\mathbb{R}$, η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$.

Άρα η $f$ είναι $1-1$, οπότε αντιστρέφεται.

\item Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$, άρα είναι συνεχής στο $[0,2]$ και παραγωγίσιμη στο $(0,2)$.

Από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(0,2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(2)-f(0)}{2-0}.
\]
Επειδή
\[
f(2)=2
\quad\text{και}\quad
f(0)=0,
\]
παίρνουμε
\[
f'(\xi)=\frac{2}{2}=1.
\]
Άρα
\[
{f'(\xi)=1}
\]
για κάποιο $\xi\in(0,2)$.

\item Για
\[
g(x)=f(x)-x
\]
έχουμε
\[
g(0)=f(0)-0=0
\]
και
\[
g(2)=f(2)-2=0.
\]
Η $g$ είναι συνεχής στο $[0,2]$ και παραγωγίσιμη στο $(0,2)$, αφού η $f$ είναι παραγωγίσιμη.

Επιπλέον,
\[
g(0)=g(2).
\]
Άρα, από το Θεώρημα Rolle, υπάρχει $\xi\in(0,2)$ τέτοιο ώστε
\[
g'(\xi)=0.
\]
Όμως
\[
g'(x)=f'(x)-1.
\]
Άρα
\[
f'(\xi)-1=0
\]
και επομένως
\[
f'(\xi)=1.
\]

Γεωμετρικά, αυτό σημαίνει ότι υπάρχει σημείο της $C_f$ στο οποίο η εφαπτομένη έχει κλίση $1$, δηλαδή είναι παράλληλη στην ευθεία που ενώνει τα σημεία
\[
(0,0)
\quad\text{και}\quad
(2,2).
\]

\item Επειδή η $f$ είναι γνησίως αύξουσα και
\[
f(0)=0,\quad f(2)=2,
\]
η αντίστροφη $f^{-1}$ είναι ορισμένη στο $[0,2]$ και
\[
f^{-1}(0)=0,\quad f^{-1}(2)=2.
\]

Στο ολοκλήρωμα
\[
\dint_0^2 f^{-1}(y)\,\d y
\]
θέτουμε
\[
y=f(x).
\]
Τότε
\[
\d y=f'(x)\,\d x,
\]
και τα άκρα παραμένουν $0$ και $2$. Άρα
\[
\dint_0^2 f^{-1}(y)\,\d y
=\dint_0^2 x f'(x)\,\d x.
\]
Με ολοκλήρωση κατά παράγοντες,
\begin{align*}
\dint_0^2 x f'(x)\,\d x
&=\left[xf(x)\right]_0^2-\dint_0^2 f(x)\,\d x\\
&=2f(2)-0f(0)-\dint_0^2 f(x)\,\d x\\
&=4-\dint_0^2 f(x)\,\d x.
\end{align*}
Επομένως
\[
\dint_0^2 f(x)\,\d x+\dint_0^2 f^{-1}(y)\,\d y
=\dint_0^2 f(x)\,\d x+4-\dint_0^2 f(x)\,\d x
=4.
\]
Άρα
\[
{\dint_0^2 f(x)\,\d x+\dint_0^2 f^{-1}(y)\,\d y=4}.
\]
\end{erwthma}
\end{thema}
