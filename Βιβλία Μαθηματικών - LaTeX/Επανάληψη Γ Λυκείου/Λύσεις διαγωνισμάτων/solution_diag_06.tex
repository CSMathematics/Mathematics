\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Σ.} 
Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού, αφού η $f'$ είναι συνεχής στο $[\alpha, \beta]$, ισχύει πάντα $\dint_\alpha^\beta f'(x)\,\d x = f(\beta) - f(\alpha)$.
\item \textbf{Σ.}
Μια γνησίως μονότονη συνάρτηση (είτε γνησίως αύξουσα είτε γνησίως φθίνουσα) παίρνει κάθε τιμή το πολύ μία φορά, δηλαδή είναι πάντοτε 1-1. Κατά συνέπεια, κάθε γνησίως μονότονη συνάρτηση είναι αυτόματα και αντιστρέψιμη.
\item \textbf{Λ.}
Δύο συναρτήσεις $f, g$ με $f'(x) = g'(x)$ στο $\Delta$ δεν είναι απαραίτητα ακριβώς ίσες, αλλά διαφέρουν κατά μία σταθερά $c$, δηλαδή $f(x) = g(x) + c$. (Μόνο αν η σταθερά είναι $0$ ταυτίζονται).
\item \textbf{Λ.}
Ο κανόνας των ορίων ολοκλήρωσης αναφέρει πως $\dint_\alpha^\alpha f(x)\,\d x = 0$ για \textbf{κάθε} πραγματικό αριθμό $\alpha$ στο πεδίο ορισμού, όχι \textit{αν και εφόσον} το $\alpha=0$.
\item \textbf{Σ.}
Ισχύει βάσει ορισμού· η γραφική παράσταση της αντίστροφης $f^{-1}$ είναι η συμμετρική της γραφικής παράστασης της $f$ ως προς την ευθεία $y = x$, που γεωμετρικά είναι η διχοτόμος του $1$\textsuperscript{ου} και $3$\textsuperscript{ου} τεταρτημορίου.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η $f(x) = (x-1)e^{-x}$, $x \in \mathbb{R}$.
\begin{erwthma}

\item \textbf{Ασύμπτωτες:}
\underline{Στο $-\infty$:}
\[ \lim_{x\to -\infty} f(x) = \lim_{x\to -\infty} \left[ (x-1)e^{-x} \right] = (-\infty) \cdot (+\infty) = -\infty \]
Άρα δεν υπάρχει οριζόντια ασύμπτωτη στο $-\infty$. Καθώς η $f$ είναι συνεχής στο $\mathbb{R}$, δεν υφίστανται ούτε κατακόρυφες ασύμπτωτες.

\underline{Στο $+\infty$:}
Ελέγχουμε για οριζόντια μορφής $\frac{+\infty}{+\infty}$:
\[ \lim_{x\to +\infty} \frac{x-1}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to +\infty} \frac{(x-1)'}{(e^x)'} = \lim_{x\to +\infty} \frac{1}{e^x} = 0 \]
Συνεπώς, η ευθεία $y = 0$ (άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} στο $+\infty$.

\item \textbf{Μονοτονία και Ακρότατα:}
Παραγωγίζουμε στο $\mathbb{R}$:
\[ f'(x) = (x-1)'e^{-x} + (x-1)(e^{-x})' = 1 \cdot e^{-x} - (x-1)e^{-x} = e^{-x} (1 - x + 1) = e^{-x}(2 - x) \]

Πρόσημο $f'$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 2 - x = 0 \Leftrightarrow x = 2$
    \item $f'(x) > 0 \Leftrightarrow 2 - x > 0 \Leftrightarrow x < 2$
    \item $f'(x) < 0 \Leftrightarrow 2 - x < 0 \Leftrightarrow x > 2$
\end{itemize}

Η $f$ είναι γνησίως αύξουσα στο $(-\infty, 2]$ και γνησίως φθίνουσα στο $[2, +\infty)$.
Διαθέτει \textbf{ολικό μέγιστο} στο $x = 2$, με τιμή:
\[ f(2) = (2-1)e^{-2} = \frac{1}{e^2} \]

\item \textbf{Κυρτότητα και Σημεία Καμπής:}
Παραγωγίζουμε στο $\mathbb{R}$:
\[ f''(x) = (e^{-x})'(2-x) + e^{-x}(2-x)' = -e^{-x}(2-x) - e^{-x} = e^{-x} (-2 + x - 1) = e^{-x}(x - 3) \]

Πρόσημο $f''$:
\begin{itemize}
    \item $f''(x) = 0 \Leftrightarrow x - 3 = 0 \Leftrightarrow x = 3$
    \item $f''(x) > 0 \Leftrightarrow x - 3 > 0 \Leftrightarrow x > 3$
    \item $f''(x) < 0 \Leftrightarrow x - 3 < 0 \Leftrightarrow x < 3$
\end{itemize}

Η $f$ είναι \textbf{κοίλη} στο $(-\infty, 3]$ και \textbf{κυρτή} στο $[3, +\infty)$.
Το σημείο καμπής είναι το $A(3, f(3))$, με $f(3) = (3-1)e^{-3} = \frac{2}{e^3}$.
Οπότε \textbf{$\Sigma.\text{K.} \equiv \left(3, \dfrac{2}{e^3}\right)$}.

\item \textbf{Εμβαδόν:}
Για κάθε $x \in [1,2]$, ισχύει $x \ge 1 \Rightarrow x - 1 \ge 0$, άρα $f(x) \ge 0$. Το εμβαδόν είναι:
\begin{align*}
E &= \int_1^2 |f(x)|\, \d x = \int_1^2 (x-1)e^{-x}\, \d x = \int_1^2 (x-1)(-e^{-x})'\, \d x \\
&= \left[ -(x-1)e^{-x} \right]_1^2 - \int_1^2 (x-1)' (-e^{-x})\, \d x \\
&= \left( -1\cdot e^{-2} - (-0\cdot e^{-1}) \right) + \int_1^2 e^{-x}\, \d x \\
&= -e^{-2} + \left[ -e^{-x} \right]_1^2 = -e^{-2} + (-e^{-2} + e^{-1}) \\
&= \frac{1}{e} - \frac{2}{e^2} = \frac{e-2}{e^2} \ \text{τ.μ.}
\end{align*}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Ογκος κυλίνδρου $V = 500$, βασικά κόστη $c_1 = 0{,}02$, παράπλευρα $c_2 = 0{,}01$.
\begin{erwthma}

\item Ο όγκος κυλίνδρου δίνεται από το $V = \pi r^2 h$. Ισχύει $\pi r^2 h = 500 \Rightarrow h = \dfrac{500}{\pi r^2}$.

Το εμβαδόν των δύο βάσεων είναι $E_\beta = 2 \pi r^2$. Άρα το κόστος τους είναι:
\[ K_1 = 0{,}02 \cdot 2\pi r^2 = 0{,}04\pi r^2 \]
Το εμβαδόν της κυλινδρικής παράπλευρης επιφάνειας είναι $E_\pi = 2\pi r h$. Αντικαθιστώντας το $h$:
\[ E_\pi = 2\pi r \frac{500}{\pi r^2} = \frac{1000}{r} \]
Άρα το κόστος της παράπλευρης επιφάνειας είναι:
\[ K_2 = 0{,}01 \cdot \frac{1000}{r} = \frac{10}{r} \]

Το συνολικό κόστος $K(r)$ της κονσέρβας είναι το άθροισμα των παραπάνω:
\[ K(r) = K_1 + K_2 = 0{,}04 \pi r^2 + \frac{10}{r} \qquad (r > 0) \]

\item Παραγωγίζουμε την συνάρτηση κόστους στο πεδίο $(0, +\infty)$:
\[ K'(r) = 0{,}04 \pi (2r) - \frac{10}{r^2} = 0{,}08 \pi r - \frac{10}{r^2} = \frac{0{,}08 \pi r^3 - 10}{r^2} \]

Ψάχνουμε ρίζα για το ρυθμό:
\[ K'(r) = 0 \Leftrightarrow 0{,}08 \pi r^3 - 10 = 0 \Leftrightarrow \pi r^3 = \frac{10}{0{,}08} = \frac{1000}{8} = 125 \Leftrightarrow r^3 = \frac{125}{\pi} \Leftrightarrow r = \frac{5}{\sqrt[3]{\pi}} \]

Για $r > \frac{5}{\sqrt[3]{\pi}} \Rightarrow K'(r) > 0$, ενώ για $r < \frac{5}{\sqrt[3]{\pi}} \Rightarrow K'(r) < 0$. 
Συνεπώς η $K$ είναι γνησίως φθίνουσα στο $(0, \frac{5}{\sqrt[3]{\pi}}]$ και γνησίως αύξουσα στο $[\frac{5}{\sqrt[3]{\pi}}, +\infty)$.
Παρουσιάζει \textbf{ολικό (καθώς και μοναδικό) ελάχιστο} για την ακτίνα $\boxed{r = \frac{5}{\sqrt[3]{\pi}}}$.
Άρα πράγματι υπάρχει μοναδική τιμή για $r$ που ελαχιστοποιεί το κόστος.

\item Υπολογίζουμε την ελάχιστη τιμή του κόστους για τη $r_0 = \frac{5}{\sqrt[3]{\pi}}$:
\begin{align*}
K(r_0) &= 0{,}04 \pi \left( \frac{5}{\pi^{1/3}} \right)^2 + \frac{10}{\frac{5}{\pi^{1/3}}} = 0{,}04 \pi \frac{25}{\pi^{2/3}} + 2\pi^{1/3} \\
&= 1 \cdot \pi^{1-2/3} + 2\pi^{1/3} = \pi^{1/3} + 2\pi^{1/3} = 3\pi^{1/3} \ \text{ευρώ.}
\end{align*}

Επειδή $\pi < 4 \Rightarrow \sqrt[3]{\pi} < \sqrt[3]{4} < \sqrt[3]{8}=2 \Rightarrow 3\sqrt[3]{\pi} < 6 < 15$. Άρα $K(r_0) < 15$.
Στο διάστημα $\Delta = [r_0, +\infty)$, η συνάρτηση είναι \textbf{συνεχής και γνησίως αύξουσα}, με:
\[ \lim_{r \to +\infty} K(r) = \lim_{r \to +\infty} \left( 0{,}04\pi r^2 + \frac{10}{r} \right) = +\infty \]
Επομένως το σύνολο τιμών της $K$ σε αυτόν τον κλάδο $\Delta$ είναι το $[3\sqrt[3]{\pi}, +\infty)$. Το $15$ ανήκει εξόφθαλμα στο εν λόγω εύρος τιμών, και αφού η $K$ είναι γνησίως αύξουσα εκεί, υπάρχει \textbf{ακριβώς ένα (οπότε τουλάχιστον ένα)} $R \in (r_0, +\infty)$ ώστε $K(R) = 15$.

\item Το μέγεθος $h(t)$ δίνεται συναρτήσει του χρόνου από το $h(t) = \frac{500}{\pi r(t)^2}$.
Παραγωγίζοντας ως προς το χρόνο (εφαρμόζοντας τον κανόνα της αλυσίδας):
\[ h'(t) = \frac{500}{\pi} \left( r^{-2}(t) \right)' = \frac{500}{\pi} (-2)r^{-3}(t) r'(t) = -\frac{1000}{\pi r^3(t)} r'(t) \]

Τη χρονική στιγμή $t_0$, αντικαθιστούμε $r(t_0) = 5$ και $r'(t_0) = 0{,}5 = \frac{1}{2}$:
\[ h'(t_0) = -\frac{1000}{\pi \cdot 5^3} \cdot \frac{1}{2} = -\frac{1000}{125\pi} \cdot \frac{1}{2} = -\frac{8}{\pi} \cdot \frac{1}{2} = -\frac{4}{\pi} \ \text{cm/s} \]
Άρα ο ρυθμός μεταβολής του ύψους είναι ίσος με $\boxed{-\dfrac{4}{\pi} \ \text{cm/sec}}$.

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 6ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
$f(1)=e$ και $f'(x) > 2x f(x)$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}

\item Εισάγουμε τη βοηθητική συνάρτηση $g(x) = e^{-x^2} f(x)$, $x \in \mathbb{R}$.
Παραγωγίζουμε την $g$:
\begin{align*}
g'(x) &= (e^{-x^2})' f(x) + e^{-x^2} f'(x) = -2x e^{-x^2} f(x) + e^{-x^2} f'(x) \\
&= e^{-x^2} (f'(x) - 2x f(x))
\end{align*}

Αφού $e^{-x^2} > 0$ και $f'(x) - 2x f(x) > 0$, έπεται ότι $\boldsymbol{g'(x) > 0}$ για κάθε $x \in \mathbb{R}$.
Η συνάρτηση $g$ είναι επομένως \textbf{γνησίως αύξουσα} σε ολόκληρο το $\mathbb{R}$.

Για οποιοδήποτε $x > 1$, από τη γνήσια μονοτονία:
\[ x > 1 \Rightarrow g(x) > g(1) \Rightarrow e^{-x^2} f(x) > e^{-1} f(1) \]
Όμως γνωρίζουμε ότι $f(1) = e$. Άρα:
\[ e^{-x^2} f(x) > e^{-1} e \Rightarrow e^{-x^2} f(x) > 1 \Rightarrow f(x) > e^{x^2} \]
Η ανισότητα αυτή κατέστη αποδεδειγμένη.

\item Θα εξετάσουμε τη μονοτονία της $f$ στο υποδιάστημα $(1, +\infty)$. Στο πεδίο αυτό, το $x$ είναι θετικό. 

Έχουμε ότι $f'(x) > 2x f(x)$. Παράλληλα αποδείξαμε στο ($\Delta_1$) πώς για $x>1$: $f(x)>e^{x^2}>0$.
Εκ τούτου, επειδή $x>0 \Rightarrow 2x > 0$ και $f(x) > 0$, το γινόμενο είναι ξεκάθαρα θετικό:
\[ f'(x) > 2x f(x) > 0 \Rightarrow f'(x) > 0 \]
Συμπεραίνουμε συνεπώς ανενδοίαστα πως η $f$ είναι \textbf{γνησίως αύξουσα} στο εν λόγω διάστημα $(1, +\infty)$.

\textbf{Όριο:}
Καθώς αποδείξαμε πως $f(x) > e^{x^2}$ για $x>1$, εκτελούμε τα όρια συγκριτικά.
Επειδή το όριο $\lim_{x\to +\infty} e^{x^2} = +\infty$, εξάγεται ως άμεση και φυσιολογική συνέπεια της κριτηρίου συγκρίσεως πώς και:
\[ \lim_{x\to +\infty} f(x) = +\infty \]

\item Η $f(x)$ είναι παραγωγίσιμη (άρα και συνεχής) στο κλειστό διάστημα $[1,2]$. 
Δύναται να εφαρμοστεί το \textbf{Θεώρημα Μέσης Τιμής (\eng{M.V.T.})} στο εν λόγω εύρος:
Υπάρχει τουλάχιστον ένα εσωτερικό σημείο $\xi \in (1,2)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(2) - f(1)}{2 - 1} \Rightarrow f'(\xi) = f(2) - f(1) \]

Ζητείται τώρα η ανισοτική ελαχιστοποίηση της διαφοράς.
Αξιοποιούμε τη δοσμένη διαφορική ανισότητα για $x = \xi$: $f'(\xi) > 2\xi f(\xi)$.
Άρα $f(2) - f(1) > 2\xi f(\xi)$.

Όμως, το σημείο $\xi \in (1,2)$ συνεπάγεται ότι $\xi > 1$, και αφού η $f$ είναι γνησίως αύξουσα:
$f(\xi) > f(1) = e$.
Άρα, $2\xi > 2$ και $f(\xi) > e$, οπότε τα γινόμενά τους φέρουν την αυστηρή υπεροχή:
\[ 2\xi f(\xi) > 2 \cdot 1 \cdot e = 2e \]
Καταληκτικά λοιπόν,
\[ f(2) - f(1) = f'(\xi) > 2\xi f(\xi) > 2e \Rightarrow \boxed{f(2) - f(1) > 2e} \]

\item Ζητείται αφετηριακά ο ρυθμός μεταβολής $f'(\alpha)$ στο $\alpha > 1$.
Εφαρμόζοντας τη διαφορική ιδιότητα και την ανισότητα ερωτήματος $\Delta_1$ για το σημείο $x = \alpha > 1$:
\[ f'(\alpha) > 2\alpha f(\alpha) \stackrel{f(\alpha)>e^{\alpha^2}}{>} 2\alpha e^{\alpha^2} \Rightarrow f'(\alpha) > 2\alpha e^{\alpha^2} \]

\textbf{Έλεγχος εφαπτομένης:}
Υποθέτουμε πως παρά ταύτα υπάρχει εφαπτομένη με τετμημένη $x_0 > 1$, η οποία σχηματίζει γωνία $\omega \le 45^\circ$. Επίσης $f'(x_0) > 0$, άρα $0^\circ < \omega \le 45^\circ$.
Η κλίση συνεπάγεται πως $f'(x_0) = \ef\omega \le \ef 45^\circ \Rightarrow f'(x_0) \le 1$.

Όμως, αξιοποιώντας την αποδειχθείσα σχέση για $x = x_0 > 1$:
\[ f'(x_0) > 2x_0 e^{x_0^2} \]
Επειδή $x_0 > 1 \Rightarrow x_0^2 > 1 \Rightarrow e^{x_0^2} > e > 1$.
Οπότε:
\[ f'(x_0) > 2 \cdot 1 \cdot e = 2e \approx 5{,}43 > 1 \Rightarrow f'(x_0) > 1 \]
Αντικρύζουμε μία καθοριστική και ασυμβίβαστη αντίφαση ($f'(x_0) \le 1$ και ταυτόχρονα $f'(x_0) > 1$). 
Επαληθεύεται, συνεπώς, η τελική πρόταση πως δεν υφίσταται καμία απολύτως τέτοια εφαπτομένη για την $C_f$.

\end{erwthma}
\end{thema}
