\begin{thema}{Α}
\begin{erwthma}
\item Μια συνάρτηση $f$ λέγεται συνεχής στο κλειστό διάστημα $[a,\beta]$, όταν:
\begin{itemize}
\item είναι συνεχής σε κάθε σημείο του ανοικτού διαστήματος $(a,\beta)$,
\item είναι δεξιά συνεχής στο $a$, δηλαδή
\[
\lim_{x\to a^+}f(x)=f(a),
\]
\item είναι αριστερά συνεχής στο $\beta$, δηλαδή
\[
\lim_{x\to \beta^-}f(x)=f(\beta).
\]
\end{itemize}

\item Αν μια συνάρτηση $f$ είναι συνεχής στο $[a,\beta]$ και παραγωγίσιμη στο $(a,\beta)$, τότε υπάρχει τουλάχιστον ένα
\[
\xi\in(a,\beta)
\]
τέτοιο ώστε
\[
f'(\xi)=\frac{f(\beta)-f(a)}{\beta-a}.
\]
Γεωμετρικά, υπάρχει τουλάχιστον ένα σημείο της $C_f$ στο οποίο η εφαπτομένη είναι παράλληλη προς τη χορδή που ενώνει τα σημεία
\[
A(a,f(a)),\quad B(\beta,f(\beta)).
\]

\item Αν οι συνεχείς συναρτήσεις $f,g$ τέμνονται στα σημεία με τετμημένες $a$ και $\beta$, τότε το εμβαδόν του χωρίου που περικλείεται από τις $C_f,C_g$ είναι
\[
E=\dint_a^\beta |f(x)-g(x)|\,\d x.
\]
Αν, ειδικότερα, $f(x)\geq g(x)$ για κάθε $x\in[a,\beta]$, τότε
\[
E=\dint_a^\beta \bigl(f(x)-g(x)\bigr)\,\d x.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Το μηδενισμό της παραγώγου δεν αρκεί για την ύπαρξη τοπικού ακροτάτου. Χρειάζεται να εξετάσουμε και αν αλλάζει πρόσημο η $f'$.
\item \textbf{Σωστό}. Αν
\[
\lim_{x\to+\infty}f(x)=0,
\]
τότε η ευθεία $y=0$ είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.
\item \textbf{Σωστό}. Μπορεί η συνάρτηση να είναι ορισμένη στο σημείο της κατακόρυφης ασύμπτωτης. Για παράδειγμα, αν
\[
f(x)=
\begin{cases}
\dfrac{1}{x^2},&x\neq0,\\
0,&x=0,
\end{cases}
\]
τότε το $x=0$ ανήκει στο πεδίο ορισμού, αλλά η ευθεία $x=0$ είναι κατακόρυφη ασύμπτωτη.
\item \textbf{Σωστό}. Ένα ολικό μέγιστο που παρουσιάζεται σε εσωτερικό σημείο ενός διαστήματος είναι ταυτόχρονα και τοπικό μέγιστο.
\item \textbf{Σωστό}. Αν $F'(x)=f(x)$ σε ένα διάστημα, τότε κάθε άλλη παράγουσα της $f$ στο ίδιο διάστημα έχει τη μορφή
\[
F(x)+c,\quad c\in\mathbb{R}.
\]
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=\frac{e^x}{x^2+1},\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Για $x\to-\infty$ έχουμε
\[
\lim_{x\to-\infty}e^x=0
\quad\text{και}\quad
\lim_{x\to-\infty}(x^2+1)=+\infty.
\]
Άρα
\[
\lim_{x\to-\infty}\frac{e^x}{x^2+1}=0.
\]
Επομένως η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.

Για $x\to+\infty$ έχουμε
\[
\lim_{x\to+\infty}\frac{e^x}{x^2+1}=+\infty,
\]
αφού η εκθετική συνάρτηση υπερισχύει κάθε πολυωνυμικής. Άρα στο $+\infty$ δεν υπάρχει οριζόντια ασύμπτωτη.

\item Παραγωγίζουμε:
\begin{align*}
f'(x)
&=\frac{e^x(x^2+1)-e^x\cdot2x}{(x^2+1)^2}\\
&=\frac{e^x(x^2-2x+1)}{(x^2+1)^2}\\
&=\frac{e^x(x-1)^2}{(x^2+1)^2}.
\end{align*}
Επειδή
\[
e^x>0
\quad\text{και}\quad
(x^2+1)^2>0
\]
για κάθε $x\in\mathbb{R}$, έχουμε
\[
f'(x)\geq0
\]
και
\[
f'(x)=0\Leftrightarrow x=1.
\]
Η παράγωγος δεν αλλάζει πρόσημο στο $x=1$. Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$ και δεν παρουσιάζει τοπικά ακρότατα.

\item Εξετάζουμε το όριο
\[
\lim_{x\to+\infty}\frac{f(x)}{x}
=\lim_{x\to+\infty}\frac{e^x}{x(x^2+1)}
=\lim_{x\to+\infty}\frac{e^x}{x^3+x}.
\]
Η εκθετική υπερισχύει της πολυωνυμικής, οπότε
\[
\lim_{x\to+\infty}\frac{e^x}{x^3+x}=+\infty.
\]
Άρα ο συντελεστής διεύθυνσης μιας πιθανής πλάγιας ασύμπτωτης δεν είναι πεπερασμένος πραγματικός αριθμός. Επομένως η $C_f$ δεν έχει πλάγια ασύμπτωτη στο $+\infty$.

\item Το ζητούμενο εμβαδόν είναι
\[
E=\dint_0^1 \frac{e^x}{x^2+1}\,\d x,
\]
αφού $f(x)>0$ για κάθε $x\in\mathbb{R}$.

Για κάθε $x\in[0,1]$ ισχύει
\[
x^2+1\geq1,
\]
άρα
\[
\frac{e^x}{x^2+1}\leq e^x.
\]
Μάλιστα η ανισότητα είναι γνήσια για κάθε $x\in(0,1]$. Επομένως
\[
E=\dint_0^1 \frac{e^x}{x^2+1}\,\d x
<
\dint_0^1 e^x\,\d x
=e-1.
\]
Άρα
\[
{E<e-1}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
g:\mathbb{R}\to\mathbb{R}
\]
με
\[
g(0)=1
\quad\text{και}\quad
g'(x)=1-e^{-x}.
\]
\begin{erwthma}
\item Από τη σχέση
\[
g'(x)=1-e^{-x}
\]
παίρνουμε
\[
g(x)=x+e^{-x}+c.
\]
Επειδή $g(0)=1$, έχουμε
\[
g(0)=0+1+c=1
\]
και άρα
\[
c=0.
\]
Επομένως
\[
{g(x)=x+e^{-x}}.
\]

\item Έχουμε
\[
g'(x)=1-e^{-x}.
\]
Λύνουμε:
\[
g'(x)=0
\Leftrightarrow
1-e^{-x}=0
\Leftrightarrow
e^{-x}=1
\Leftrightarrow
x=0.
\]
Για $x<0$ είναι $e^{-x}>1$, άρα $g'(x)<0$, ενώ για $x>0$ είναι $e^{-x}<1$, άρα $g'(x)>0$.

Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $g'$ και τη μονοτονία της $g$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.7,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$g'(x)$ /0.8,$g(x)$/0.8}%
{$-\infty$,$0$,$+\infty$}%
\tkzTabLine{,-,z,+,}
\tkzTabLine{,\searrow,t,\nearrow,}
\end{tikzpicture}
\end{center}

Άρα η $g$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$. Επομένως παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
g(0)=1.
\]
Άρα
\[
{g(x)\geq1,\quad x\in\mathbb{R}},
\]
και η απόλυτη ελάχιστη τιμή της είναι
\[
{1}.
\]

\item Αναζητούμε πλάγια ασύμπτωτη της μορφής
\[
y=\lambda x+\beta
\]
στο $+\infty$.

Έχουμε
\[
\lambda=\lim_{x\to+\infty}\frac{g(x)}{x}
=\lim_{x\to+\infty}\left(1+\frac{e^{-x}}{x}\right)=1.
\]
Επίσης
\[
\beta=\lim_{x\to+\infty}\bigl(g(x)-\lambda x\bigr)
=\lim_{x\to+\infty}\bigl(x+e^{-x}-x\bigr)=0.
\]
Άρα η πλάγια ασύμπτωτη της $C_g$ στο $+\infty$ είναι
\[
{y=x}.
\]

\item Η $C_g$ βρίσκεται πάνω από την ασύμπτωτη $y=x$, αφού
\[
g(x)-x=e^{-x}>0
\]
για κάθε $x\in\mathbb{R}$.

Άρα το ζητούμενο εμβαδόν είναι
\[
E=\dint_0^{\ln2}\bigl(g(x)-x\bigr)\,\d x
=\dint_0^{\ln2}e^{-x}\,\d x.
\]
Επομένως
\[
E=\left[-e^{-x}\right]_0^{\ln2}
=-e^{-\ln2}+1
=-\frac12+1
=\frac12.
\]
Άρα
\[
{E=\frac12}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
f:\mathbb{R}\to\mathbb{R}
\]
τέτοια ώστε
\[
f^3(x)+3f(x)=x
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Παραγωγίζουμε τη δοσμένη σχέση:
\[
3f^2(x)f'(x)+3f'(x)=1.
\]
Άρα
\[
3f'(x)\bigl(f^2(x)+1\bigr)=1
\]
και επομένως
\[
f'(x)=\frac{1}{3\bigl(f^2(x)+1\bigr)}.
\]
Επειδή
\[
3\bigl(f^2(x)+1\bigr)>0
\]
για κάθε $x\in\mathbb{R}$, έχουμε
\[
f'(x)>0.
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Για $x=0$ παίρνουμε
\[
f^3(0)+3f(0)=0.
\]
Άρα
\[
f(0)\bigl(f^2(0)+3\bigr)=0.
\]
Επειδή
\[
f^2(0)+3>0,
\]
προκύπτει
\[
f(0)=0.
\]

Επομένως
\[
\lim_{x\to0}\frac{f(x)}{x}
=\lim_{x\to0}\frac{f(x)-f(0)}{x-0}
=f'(0).
\]
Από τον τύπο της παραγώγου έχουμε
\[
f'(0)=\frac{1}{3\bigl(f^2(0)+1\bigr)}
=\frac{1}{3}.
\]
Άρα
\[
{\lim_{x\to0}\frac{f(x)}{x}=\frac13}.
\]

\item Επειδή η $f$ είναι γνησίως αύξουσα και $f(0)=0$, για κάθε $x\in[0,4]$ ισχύει
\[
f(x)\geq0.
\]
Άρα το εμβαδόν ισούται με
\[
E=\dint_0^4 f(x)\,\d x.
\]

Θέτουμε
\[
y=f(x).
\]
Τότε από τη δοσμένη σχέση έχουμε
\[
x=y^3+3y
\]
και επομένως
\[
\d x=(3y^2+3)\,\d y.
\]
Για $x=0$ παίρνουμε $y=0$, ενώ για $x=4$ παίρνουμε
\[
y^3+3y=4.
\]
Η εξίσωση έχει λύση $y=1$ και, επειδή η συνάρτηση $y^3+3y$ είναι γνησίως αύξουσα, αυτή είναι η μοναδική λύση.

Άρα
\begin{align*}
\dint_0^4 f(x)\,\d x
&=\dint_0^1 y(3y^2+3)\,\d y\\
&=\dint_0^1(3y^3+3y)\,\d y\\
&=\left[\frac{3y^4}{4}+\frac{3y^2}{2}\right]_0^1\\
&=\frac34+\frac32\\
&=\frac94.
\end{align*}
Επομένως
\[
{E=\dint_0^4 f(x)\,\d x=\frac94}.
\]

\item Η εξίσωση γράφεται, για $x>0$,
\[
f(x)=\frac{1}{x^{2026}}.
\]
Θεωρούμε τη συνάρτηση
\[
K(x)=f(x)-\frac{1}{x^{2026}},\quad x>0.
\]
Η $K$ είναι συνεχής και παραγωγίσιμη στο $(0,+\infty)$ και
\[
K'(x)=f'(x)+\frac{2026}{x^{2027}}.
\]
Επειδή
\[
f'(x)>0
\quad\text{και}\quad
\frac{2026}{x^{2027}}>0
\]
για κάθε $x>0$, έχουμε
\[
K'(x)>0.
\]
Άρα η $K$ είναι γνησίως αύξουσα στο $(0,+\infty)$.

Επίσης, επειδή $f$ είναι συνεχής και $f(0)=0$,
\[
\lim_{x\to0^+}K(x)
=\lim_{x\to0^+}\left(f(x)-\frac{1}{x^{2026}}\right)
=-\infty.
\]

Ακόμη, από τη σχέση
\[
f(x)\bigl(f^2(x)+3\bigr)=x
\]
και καθώς $x\to+\infty$, προκύπτει ότι
\[
f(x)\to+\infty.
\]
Άρα
\[
\lim_{x\to+\infty}K(x)=+\infty.
\]

Επομένως, από τη συνέχεια της $K$, υπάρχει τουλάχιστον μία ρίζα της εξίσωσης $K(x)=0$ στο $(0,+\infty)$. Επειδή η $K$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

Άρα η εξίσωση
\[
f(x)=\frac{1}{x^{2026}}
\]
έχει
\[
{\text{ακριβώς μία ρίζα στο }(0,+\infty)}.
\]
\end{erwthma}
\end{thema}
