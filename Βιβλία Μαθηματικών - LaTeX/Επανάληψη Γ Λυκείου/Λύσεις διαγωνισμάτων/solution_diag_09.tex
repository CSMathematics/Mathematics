\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Λ.} 
Απαιτείται όχι μόνο να μηδενίζεται η παράγωγος, αλλά να \textbf{αλλάζει και πρόσημο} εκατέρωθεν της ρίζας για να συνιστά τοπικό ακρότατο.
\item \textbf{Σ.}
Αν $\lim_{x\to+\infty} f(x) = 0$, αυτό σημαίνει εξ ορισμού ότι η ευθεία $y = 0$ (άξονας $x'x$) είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.
\item \textbf{Λ.}
Για να έχει μια συνάρτηση κατακόρυφη ασύμπτωτη στο $x_0$, πρέπει το όριο στο $x_0$ (από δεξιά ή αριστερά) να είναι άπειρο. Αυτό προϋποθέτει η περιοχή γύρω από το $x_0$ να ανήκει στο πεδίο ορισμού, αλλά το ίδιο το $x_0$ να είναι «ανοικτό άκρο» (ήτοι μη ορισμένο σημείο τουλάχιστον από τον ένα κλάδο). Μία συνεχής συνάρτηση στο $\mathbb{R}$ δεν έχει κατακόρυφες ασύμπτωτες.
\item \textbf{Σ.}
Πράγματι, ένα ολικό μέγιστο (εφόσον παρουσιάζεται στο εσωτερικό του διαστήματος ορισμού) αποτελεί ταυτόχρονα και τοπικό μέγιστο προσφέροντας διπλή ιδιότητα.
\item \textbf{Σ.}
Οποιαδήποτε άλλη αρχική συνάρτηση $G(x)$ θα διαφέρει από την $F(x)$ αποκλειστικά κατά μία σταθερά $c$, δηλαδή $G(x) = F(x) + c$.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 9ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται $f(x) = \dfrac{e^x}{x^2+1}$, σε όλο το $\mathbb{R}$.
\begin{erwthma}

\item \textbf{Όρια και Ασύμπτωτες:}
Στο $-\infty$:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} \frac{e^x}{x^2+1} = \frac{0}{+\infty} = 0 \]
Κατά συνέπεια, η ευθεία $\boldsymbol{y = 0}$ (άξονας των $x$) είναι \textbf{οριζόντια ασύμπτωτη} στο $-\infty$.

Στο $+\infty$ (μορφή $\frac{+\infty}{+\infty}$):
\[ \lim_{x\to+\infty} \frac{e^x}{x^2+1} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{2x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{2} = +\infty \]
Άρα \textbf{δεν υπάρχει} οριζόντια ασύμπτωτη στο $+\infty$.

\item \textbf{Μονοτονία:}
Παραγωγίζουμε χρησιμοποιώντας τον κανόνα πηλίκου για $x \in \mathbb{R}$:
\[ f'(x) = \frac{(e^x)'(x^2+1) - e^x(x^2+1)'}{(x^2+1)^2} = \frac{e^x(x^2+1) - e^x(2x)}{(x^2+1)^2} = \frac{e^x(x^2 - 2x + 1)}{(x^2+1)^2} \]
\[ f'(x) = \frac{e^x(x-1)^2}{(x^2+1)^2} \]

Παρατηρούμε ότι $f'(x) \ge 0$ για κάθε $x \in \mathbb{R}$, ενώ $f'(x) = 0$ μόνο για το επιμέρους σημείο $x = 1$.
Εφόσον η $f$ είναι συνεχής, εξάγεται πως η $f$ είναι \textbf{γνησίως αύξουσα} σε ολόκληρο το $\mathbb{R}$. 
Λόγω της απουσίας αλλαγής προσήμου της $f'$, η συνάρτηση \textbf{δεν εμφανίζει κανένα τοπικό ούτε ολικό ακρότατο}.

\item \textbf{Έλεγχος Πλάγιας Ασύμπτωτης στο $+\infty$:}
Ελέγχουμε την ύπαρξη του συντελεστή διεύθυνσης $\lambda$:
\[ \lambda = \lim_{x\to+\infty} \frac{f(x)}{x} = \lim_{x\to+\infty} \frac{e^x}{x^3+x} \]
Εφαρμόζουμε επαναληπτικά τον κανόνα \eng{De l'Hospital}:
\[ \lim_{x\to+\infty} \frac{e^x}{x^3+x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{3x^2+1} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{6x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to+\infty} \frac{e^x}{6} = +\infty \]
Αφού το όριο δε συγκλίνει σε πεπερασμένο αριθμό ($\lambda \in \mathbb{R}$), συμπεραίνουμε πως \textbf{δεν υπάρχει πλάγια ασύμπτωτη} στο $+\infty$.

\item Το εμβαδόν εκφράζεται αλγεβρικά από το ολοκλήρωμα:
\[ E = \int_0^1 \frac{e^x}{x^2+1}\,\d x \]
Ωστόσο, το αναγραφόμενο ολοκλήρωμα δεν διαθέτει παράγουσα η οποία να εκφράζεται ως συνδυασμός στοιχειωδών αναλυτικών συναρτήσεων. 

\begin{prosoxi}{\linewidth}
Το ολοκλήρωμα $\int \frac{e^x}{x^2+1} \d x$ ανήκει στη κατηγορία των \textbf{μη στοιχειωδών ολοκληρωμάτων} και δεν μπορεί να υπολογιστεί με τη συμβατική σχολική ύλη. Θεωρείται βέβαιο πως λόγω τυπογραφικού (ίσως η $f$ ήταν $e^x(x^2+1)$) παρεισφρέει σφάλμα στην άσκηση. Το αφήνουμε αναλλοίωτο σε μορφή ορισμένου εμβαδού.
\end{prosoxi}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 9ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Δίνεται $g'(x) = 1 - e^{-x}$ με $g(0) = 1$.
\begin{erwthma}

\item Από τη γνωστή παράγωγο βρίσκουμε τις παράγουσες (αόριστο ολοκλήρωμα):
\[ g'(x) = 1 - e^{-x} \Rightarrow g(x) = x - (-e^{-x}) + c \Rightarrow g(x) = x + e^{-x} + c \]
Για να βρούμε της ταυτότητα της σταθεράς $c$ αξιοποιούμε το $g(0) = 1$:
\[ g(0) = 0 + e^0 + c \Rightarrow 1 + c = 1 \Rightarrow c = 0 \]
Κατά συνέπεια ο τύπος της συνάρτησης είναι: $\boldsymbol{g(x) = x + e^{-x}}$.

\item Ζητείται να εξεταστεί η ανισοτική συνθήκη. Ορίζουμε την $K(x) = g(x) - (2 - e^{-x}) = x + e^{-x} - 2 + e^{-x} = x + 2e^{-x} - 2$.
Παραγωγίζουμε:
\[ K'(x) = 1 - 2e^{-x} \]
Λύνουμε $K'(x) = 0 \Leftrightarrow e^{-x} = \frac{1}{2} \Leftrightarrow -x = -\ln 2 \Leftrightarrow x = \ln 2$.
Η $K(x)$ για $x \in [0, \ln 2)$ είναι γνησίως φθίνουσα, άρα φτάνει σε ένα απόλυτο ελάχιστο:
\[ K(\ln 2) = \ln 2 + 2e^{-\ln 2} - 2 = \ln 2 + 2\left(\frac{1}{2}\right) - 2 = \ln 2 + 1 - 2 = \ln 2 - 1 \]
Επειδή $\ln 2 \approx 0{,}69$, η ελάχιστη τιμή $\ln 2 - 1$ είναι $\boldsymbol{< 0}$.
Οπότε δε μπορεί να αληθεύει η ανισότητα $K(x) \ge 0$ για κάθε $x \ge 0$. Το απόλυτο ελάχιστο της $g$ στο συνολικό της πεδίο λαμβάνει χώρα στο $x=0$ (αφού $g'(x) = 1-e^{-x} > 0$ για $x>0$) με τιμή $g(0)=1$.

\begin{erat}
\textbf{Πιθανό Τυπογραφικό:} Από τους λογαριασμούς είναι προφανές ότι η ζητούμενη προς απόδειξη ανισότητα $x+e^{-x} \ge 2-e^{-x}$ καταρρίπτεται ($\Psi\varepsilon\upsilon\delta\acute{\eta}\varsigma$). Υφίσταται αστοχία στα δεδομένα της άσκησης.
\end{erat}

\item Αναζητούμε πλάγια ασύμπτωτη $y = \lambda x + \beta$ στο $+\infty$:
\[ \lambda = \lim_{x\to+\infty} \frac{g(x)}{x} = \lim_{x\to+\infty} \frac{x + e^{-x}}{x} = \lim_{x\to+\infty} \left( 1 + \frac{1}{x e^x} \right) = 1 + 0 = 1 \]
Υπολογίζουμε εν συνεχεία το όριο του $\beta$:
\[ \beta = \lim_{x\to+\infty} (g(x) - \lambda x) = \lim_{x\to+\infty} (x + e^{-x} - x) = \lim_{x\to+\infty} (e^{-x}) = 0 \]
Συνεπώς η γραμμή $\boldsymbol{y = x}$ είναι η ζητούμενη \textbf{πλάγια ασύμπτωτη} στο $+\infty$.

\item Με βάση την ασύμπτωτη από το ερώτημα 3: το εμβαδόν απαρτίζεται από το διάστημα μεταξύ $g$ και $y$:
\[ E = \int_0^{\ln 2} |g(x) - x|\,\d x = \int_0^{\ln 2} |x + e^{-x} - x|\,\d x = \int_0^{\ln 2} e^{-x}\,\d x \]
Καθώς η εκθετική είναι πάντα θετική:
\[ E = \left[ -e^{-x} \right]_0^{\ln 2} = -e^{-\ln 2} - (-e^0) = -\frac{1}{2} + 1 = \boldsymbol{\frac{1}{2} \ \text{τ.μ.}} \]

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 9ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Δίνεται $f^3(x) + 3f(x) = x$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}

\item Εξαιτίας του ότι η $f$ δίνεται παραγωγίσιμη, παραγωγίζουμε τα δύο μέλη (κανόνας αλυσίδας):
\[ (f^3(x))' + 3(f(x))' = (x)' \Leftrightarrow 3f^2(x) f'(x) + 3f'(x) = 1 \]
Εξάγουμε το $3f'(x)$ ως κοινό παράγοντα:
\[ 3f'(x) (f^2(x) + 1) = 1 \Leftrightarrow f'(x) = \frac{1}{3(f^2(x) + 1)} \]
Γνωρίζουμε φυσιολογικά ότι $f^2(x) \ge 0 \Rightarrow f^2(x) + 1 > 0 \Rightarrow 3(f^2(x)+1) > 0$.
Από τον διαιρετικό σχηματισμό προκύπτει ότι $f'(x) > 0$ για \textbf{κάθε} ανεξάρτητη μεταβλητή $x \in \mathbb{R}$. 
Ως αποτέλεσμα, η $f$ είναι αναμφίλεκτα \textbf{γνησίως αύξουσα} σε ολόκληρο το $\mathbb{R}$.

\item Αναζητούμε αρχικά την ρίζα θέτοντας $x=0$ στην αρχική εντολή:
\[ f^3(0) + 3f(0) = 0 \Leftrightarrow f(0) \cdot (f^2(0) + 3) = 0 \]
Επειδή ο όρος της παρένθεσης $f^2(0) + 3 \ge 3 > 0$, η μόνη δυνατή λύση εκμαιεύεται στο $\boldsymbol{f(0) = 0}$.

Για το αιτούμενο όριο $\lim_{x\to 0} \frac{f(x)}{x}$:
Λόγω του ότι $f(0)=0$, το όριο ταυτίζεται με τον επίσημο ορισμό της παραγώγου στο σημείο εξόρμησης $0$:
\[ \lim_{x\to 0} \frac{f(x)}{x} = \lim_{x\to 0} \frac{f(x)-f(0)}{x-0} = f'(0) \]
Αξιοποιώντας τη σχέση από το πρώτο ερώτημα:
\[ f'(0) = \frac{1}{3(f^2(0) + 1)} = \frac{1}{3(0 + 1)} = \boldsymbol{\frac{1}{3}} \]

\item Ζητείται το εμβαδόν μεταξύ της καμπύλης $f$ και του άξονα $x$ από το $x=0$ μέχρι το $x=4$.
Για $x>0 \Rightarrow f(x) > f(0) \Rightarrow f(x) > 0$ (η $f$ είναι αύξουσα). Άρα:
\[ E = \int_0^4 f(x)\,\d x \]
Μετασχηματίζουμε το ολοκλήρωμα (χρήση αντίστροφης): Θέτουμε σταθερά τον όρο $y = f(x)$, άρα $x = y^3 + 3y$.
Παραγωγίζοντας τη σχέση υποκατάστασης: $\d x = (3y^2+3)\d y$. 
Αλλαγή των άκρων:
- Για $x=0 \Rightarrow y^3+3y=0 \Rightarrow y=0$.
- Για $x=4 \Rightarrow y^3+3y=4 \Rightarrow y=1$ (με προφανή ρίζα και μοναδική λόγω μονοτονίας).
Το ολοκλήρωμα ανάγεται σε:
\begin{align*}
\int_0^1 y \cdot (3y^2+3)\,\d y &= \int_0^1 (3y^3 + 3y)\,\d y = \left[ \frac{3y^4}{4} + \frac{3y^2}{2} \right]_0^1 \\
&= \frac{3}{4} + \frac{3}{2} = \frac{3}{4} + \frac{6}{4} = \boldsymbol{\frac{9}{4} \ \text{τ.μ.}}
\end{align*}

\begin{parat}{\linewidth}
\textbf{Αναντιστοιχία δεδομένων:} Η εκφώνηση του διαγωνίσματος δίνει \textbf{εσφαλμένα} ως αποτέλεσμα το $11/4$. Η ορθή μαθηματική λύση ταυτίζεται με το υπολογισμένο $9/4$. 
\end{parat}

\item Η εξίσωση στο $(0, +\infty)$ γράφεται $f(x) - \frac{1}{x^{2026}} = 0$.
Ορίζουμε $K(x) = f(x) - x^{-2026}$, με $x > 0$.
Η $K(x)$ είναι παραγωγίσιμη, άρα συνεχής:
\[ K'(x) = f'(x) + 2026 \cdot x^{-2027} = f'(x) + \frac{2026}{x^{2027}} \]
Αφού $x>0$, το κλάσμα είναι θετικό. Παράλληλα $f'(x) > 0$. Επομένως $K'(x) > 0$.
Εκ τούτου, η $K$ είναι \textbf{γνησίως αύξουσα}.

\textbf{Όρια:}
$\bullet$ Στο $0^+$: $\lim_{x\to 0^+} f(x) = f(0) = 0$. Και $\lim_{x\to 0^+} \frac{1}{x^{2026}} = +\infty \Rightarrow \lim_{x\to 0^+} \left( - \frac{1}{x^{2026}} \right) = -\infty$. Άρα $\lim_{x\to 0^+} K(x) = -\infty$. 
$\bullet$ Στο $+\infty$: Από $f^3(x)+3f(x)=x \Rightarrow f(x)(f^2(x)+3)=x \to +\infty \Rightarrow f(x) \to +\infty$ (καθώς $f>0$).
Και $\lim_{x\to +\infty} \frac{1}{x^{2026}} = 0$. Άρα $\lim_{x\to +\infty} K(x) = +\infty$.

Το σύνολο τιμών της είναι το $K((0, +\infty)) = \mathbb{R}$. Ο αριθμός $0$ περιέχεται οπωσδήποτε σε αυτό, και επειδή η $K$ είναι γνησίως αύξουσα (επομένως 1-1), συμπεραίνουμε πως έχει \textbf{ακριβώς μία και μοναδική διάκριτη ρίζα}.

\end{erwthma}
\end{thema}
