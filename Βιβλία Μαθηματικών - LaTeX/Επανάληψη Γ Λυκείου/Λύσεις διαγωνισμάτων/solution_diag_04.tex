\begin{thema}{Α}
\begin{erwthma}
\item Γνωρίζουμε ότι
\[
\lim_{x\to x_0}x=x_0.
\]
Άρα, για κάθε φυσικό $k$, από τις ιδιότητες των ορίων έχουμε
\[
\lim_{x\to x_0}x^k=x_0^k.
\]
Επομένως
\begin{align*}
\lim_{x\to x_0}P(x)
&=\lim_{x\to x_0}\left(a_\nu x^\nu+a_{\nu-1}x^{\nu-1}+\cdots+a_1x+a_0\right)\\
&=a_\nu x_0^\nu+a_{\nu-1}x_0^{\nu-1}+\cdots+a_1x_0+a_0\\
&=P(x_0).
\end{align*}
Άρα
\[
{\lim_{x\to x_0}P(x)=P(x_0)}.
\]

\item Μια συνάρτηση $f$ λέγεται συνεχής σε ένα κλειστό διάστημα $[a,\beta]$, όταν:
\begin{itemize}
\item είναι συνεχής σε κάθε σημείο του $(a,\beta)$,
\item είναι συνεχής από δεξιά στο $a$, δηλαδή $\lim\limits_{x\to a^+}f(x)=f(a)$,
\item είναι συνεχής από αριστερά στο $\beta$, δηλαδή $\lim\limits_{x\to \beta^-}f(x)=f(\beta)$.
\end{itemize}

\item Η ευθεία
\[
y=\lambda x+\beta
\]
ονομάζεται πλάγια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$, όταν
\[
\lim_{x\to+\infty}\bigl[f(x)-(\lambda x+\beta)\bigr]=0.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Σωστό}. Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού,
\[
\dint_a^\beta f'(x)\,\d x=f(\beta)-f(a).
\]
\item \textbf{Σωστό}. Αν η $f$ τέμνει τον άξονα $x'x$ σε δύο διαφορετικά σημεία, τότε θα υπάρχουν $x_1\neq x_2$ με $f(x_1)=f(x_2)=0$, άτοπο αφού η $f$ είναι $1-1$.
\item \textbf{Σωστό}. Για τη σταθερή συνάρτηση $f(x)=c$ ισχύει
\[
\dint_a^b c\,\d x=c(b-a).
\]
\item \textbf{Λάθος}. Η γραφική παράσταση της $f^{-1}$ είναι συμμετρική της $C_f$ ως προς την ευθεία $y=x$.
\item \textbf{Λάθος}. Το θεώρημα \en{Bolzano} εξασφαλίζει την ύπαρξη τουλάχιστον μίας ρίζας, όχι πεπερασμένο πλήθος ριζών.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Θεωρούμε τη συνάρτηση
\[
f(x)=\frac{\ln x}{x},\quad x>0.
\]
\begin{erwthma}
\item Για κάθε $x>0$ είναι
\[
f'(x)=\left(\frac{\ln x}{x}\right)'
=\frac{\frac{1}{x}\cdot x-\ln x}{x^2}
=\frac{1-\ln x}{x^2}.
\]
Βρίσκουμε ρίζες και πρόσημα της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow 1-\ln x=0\Leftrightarrow x=e$.
\item $f'(x)>0\Leftrightarrow 1-\ln x>0\Leftrightarrow 0<x<e$.
\item $f'(x)<0\Leftrightarrow 1-\ln x<0\Leftrightarrow x>e$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$0$,$e$,$+\infty$}%
\tkzTabLine{d, +, z, -, }
\tkzTabLine{d, \nearrow,t,\searrow,}
\end{tikzpicture}
\end{center}
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $(0,e]$ και \textbf{γνησίως φθίνουσα} στο $[e,+\infty)$.

Άρα παρουσιάζει ολικό μέγιστο στο $x_0=e$, με τιμή
\[
f(e)=\frac{\ln e}{e}=\frac{1}{e}.
\]

Επειδή $\pi>e$ και η $f$ είναι γνησίως φθίνουσα στο $[e,+\infty)$, έχουμε
\[
f(\pi)<f(e)
\Leftrightarrow \frac{\ln\pi}{\pi}<\frac{1}{e}.
\]
Άρα
\[
e\ln\pi<\pi
\Leftrightarrow \ln\pi^e<\ln e^\pi.
\]
Επειδή η $\ln$ είναι γνησίως αύξουσα, προκύπτει
\[
{\pi^e<e^\pi}.
\]

\item Για την κατακόρυφη ασύμπτωτη εξετάζουμε το όριο στο $0^+$:
\[
\lim_{x\to0^+}\frac{\ln x}{x}=-\infty.
\]
Άρα η ευθεία
\[
{x=0}
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Για την οριζόντια ασύμπτωτη στο $+\infty$ έχουμε
\[
\lim_{x\to+\infty}\frac{\ln x}{x}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}\frac{\frac{1}{x}}{1}=0.
\]
Άρα η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

\item Για $\lambda=2$ η εξίσωση γίνεται
\[
x^2=e^x,\quad x>0.
\]
Επειδή τα δύο μέλη είναι θετικά, παίρνουμε λογαρίθμους:
\[
2\ln x=x
\Leftrightarrow \frac{\ln x}{x}=\frac{1}{2}
\Leftrightarrow f(x)=\frac{1}{2}.
\]
Από το πρώτο ερώτημα γνωρίζουμε ότι
\[
f(x)\leq \frac{1}{e}
\]
για κάθε $x>0$. Επειδή $e>2$, έχουμε
\[
\frac{1}{e}<\frac{1}{2}.
\]
Άρα η εξίσωση $f(x)=\frac{1}{2}$ είναι αδύνατη.

Επομένως η εξίσωση
\[
x^2=e^x
\]
έχει
\[
{0\ \text{ρίζες}}
\]
στο $(0,+\infty)$.

\item Στο διάστημα $[1,e]$ ισχύει
\[
\ln x\geq0,
\]
άρα $f(x)\geq0$. Επομένως το ζητούμενο εμβαδόν είναι
\begin{align*}
E&=\dint_1^e f(x)\,\d x
=\dint_1^e\frac{\ln x}{x}\,\d x\\
&=\left[\frac{\ln^2x}{2}\right]_1^e
=\frac{\ln^2e}{2}-\frac{\ln^21}{2}
=\frac{1}{2}.
\end{align*}
Άρα
\[
{E=\frac{1}{2}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνάρτηση
\[
f(x)=\sqrt{x+1}-1.
\]
\begin{erwthma}
\item Η ευθεία $x=a$ τέμνει τον άξονα $Ox$ στο σημείο
\[
B(a,0)
\]
και τη γραφική παράσταση της $f$ στο σημείο
\[
A(a,f(a))\equiv A(a,\sqrt{a+1}-1).
\]
Επειδή $a>0$, έχουμε
\[
\sqrt{a+1}-1>0.
\]
Το τρίγωνο $OAB$ είναι ορθογώνιο στο $B$, με
\[
OB=a
\quad\text{και}\quad
AB=\sqrt{a+1}-1.
\]
Άρα το εμβαδόν του είναι
\[
E(a)=\frac{1}{2}OB\cdot AB
=\frac{1}{2}a(\sqrt{a+1}-1).
\]

\item Έχουμε
\[
E(a)=\frac{1}{2}a(\sqrt{a+1}-1).
\]
Παραγωγίζοντας ως προς $a$, παίρνουμε
\[
E'(a)=\frac{1}{2}\left(\sqrt{a+1}-1+\frac{a}{2\sqrt{a+1}}\right).
\]
Επειδή $a=a(t)$, με τον κανόνα της αλυσίδας έχουμε
\[
\frac{\d E}{\d t}=E'(a)\cdot a'(t).
\]
Τη χρονική στιγμή που $a=3$ είναι $a'(t)=2$, άρα
\begin{align*}
\frac{\d E}{\d t}
&=\frac{1}{2}\left(\sqrt{4}-1+\frac{3}{2\sqrt4}\right)\cdot2\\
&=2-1+\frac{3}{4}
=\frac{7}{4}.
\end{align*}
Επομένως ο ζητούμενος ρυθμός μεταβολής είναι
\[
{\frac{7}{4}\ \text{μον}^2/\si{sec}}.
\]

\item Υπολογίζουμε
\begin{align*}
\dint_0^3 f(x)\,\d x
&=\dint_0^3(\sqrt{x+1}-1)\,\d x\\
&=\left[\frac{2}{3}(x+1)^{\frac{3}{2}}-x\right]_0^3\\
&=\left(\frac{2}{3}\cdot4^{\frac{3}{2}}-3\right)-\frac{2}{3}\\
&=\left(\frac{16}{3}-3\right)-\frac{2}{3}
=\frac{5}{3}.
\end{align*}
Άρα
\[
{\dint_0^3 f(x)\,\d x=\frac{5}{3}}.
\]

\item Έχουμε
\[
\lim_{x\to0}\frac{f(x)}{x}
=\lim_{x\to0}\frac{\sqrt{x+1}-1}{x}.
\]
Πολλαπλασιάζουμε με τη συζυγή παράσταση:
\begin{align*}
\lim_{x\to0}\frac{\sqrt{x+1}-1}{x}
&=\lim_{x\to0}
\frac{(\sqrt{x+1}-1)(\sqrt{x+1}+1)}{x(\sqrt{x+1}+1)}\\
&=\lim_{x\to0}\frac{x+1-1}{x(\sqrt{x+1}+1)}\\
&=\lim_{x\to0}\frac{1}{\sqrt{x+1}+1}
=\frac{1}{2}.
\end{align*}
Άρα
\[
{\lim_{x\to0}\frac{f(x)}{x}=\frac{1}{2}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Έστω $f:\mathbb{R}\to\mathbb{R}$ παραγωγίσιμη συνάρτηση με $f(0)=0$ και
\[
f'(x)-f(x)>0
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση
\[
g(x)=f(x)e^{-x},\quad x\in\mathbb{R}.
\]
Η $g$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\begin{align*}
g'(x)&=f'(x)e^{-x}+f(x)(e^{-x})'\\
&=f'(x)e^{-x}-f(x)e^{-x}\\
&=e^{-x}\bigl(f'(x)-f(x)\bigr).
\end{align*}
Επειδή
\[
e^{-x}>0
\quad\text{και}\quad
f'(x)-f(x)>0,
\]
έχουμε
\[
g'(x)>0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $g$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Από το προηγούμενο ερώτημα, η $g$ είναι γνησίως αύξουσα στο $\mathbb{R}$. Επίσης
\[
g(0)=f(0)e^0=0.
\]
Αν $x<0$, τότε
\[
g(x)<g(0)=0
\Leftrightarrow f(x)e^{-x}<0.
\]
Επειδή $e^{-x}>0$, προκύπτει
\[
f(x)<0.
\]
Αν $x>0$, τότε
\[
g(x)>g(0)=0
\Leftrightarrow f(x)e^{-x}>0.
\]
Επειδή $e^{-x}>0$, προκύπτει
\[
f(x)>0.
\]
Άρα
\[
{f(x)<0\ \text{για }x<0}
\quad\text{και}\quad
{f(x)>0\ \text{για }x>0}.
\]

\item Για κάθε $x>0$, από το προηγούμενο ερώτημα έχουμε
\[
f(x)>0.
\]
Από τη δοσμένη ανισότητα
\[
f'(x)-f(x)>0
\]
παίρνουμε
\[
f'(x)>f(x)>0.
\]
Άρα
\[
f'(x)>0
\]
για κάθε $x>0$.

Έστω $0\leq x_1<x_2$. Από το Θεώρημα Μέσης Τιμής στο $[x_1,x_2]$, υπάρχει $\xi\in(x_1,x_2)$ τέτοιο ώστε
\[
\frac{f(x_2)-f(x_1)}{x_2-x_1}=f'(\xi).
\]
Επειδή $\xi>0$, έχουμε $f'(\xi)>0$. Άρα
\[
f(x_2)-f(x_1)>0
\Leftrightarrow f(x_2)>f(x_1).
\]
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $[0,+\infty)$.

\item Έστω επιπλέον ότι
\[
f(1)=e.
\]
Τότε
\[
g(1)=f(1)e^{-1}=ee^{-1}=1.
\]
Επειδή η $g$ είναι γνησίως αύξουσα, για κάθε $x>1$ έχουμε
\[
g(x)>g(1)
\Leftrightarrow f(x)e^{-x}>1
\Leftrightarrow f(x)>e^x.
\]
Επίσης, για κάθε $x\in(0,1)$ έχουμε
\[
g(x)<g(1)
\Leftrightarrow f(x)e^{-x}<1
\Leftrightarrow f(x)<e^x.
\]

Έστω, προς άτοπο, ότι υπάρχει εφαπτομένη της $C_f$ σε σημείο με τετμημένη $x_0>1$ που σχηματίζει με τον θετικό ημιάξονα $Ox$ γωνία $\omega\leq45^\circ$.

Η κλίση της εφαπτομένης είναι
\[
\lambda=f'(x_0).
\]
Από τη δοσμένη ανισότητα και από το ότι $x_0>1$, έχουμε
\[
f'(x_0)>f(x_0)>e^{x_0}>e>1.
\]
Άρα
\[
\lambda>1.
\]
Επομένως
\[
\ef\omega=\lambda>1=\ef45^\circ,
\]
οπότε
\[
\omega>45^\circ,
\]
άτοπο. Άρα δεν υπάρχει τέτοια εφαπτομένη.
\end{erwthma}
\end{thema}
