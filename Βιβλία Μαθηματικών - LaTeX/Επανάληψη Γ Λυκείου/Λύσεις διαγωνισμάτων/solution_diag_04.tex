\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Σ.} 
Από το Θεμελιώδες Θεώρημα Ολοκληρωτικού Λογισμού, αφού η $f'$ είναι συνεχής, ισχύει $\dint_\alpha^\beta f'(x)\,\d x = f(\beta) - f(\alpha)$.
\item \textbf{Σ.}
Μια συνάρτηση 1-1 παίρνει την κάθε τιμή (άρα και το 0) το πολύ μία φορά. Αν τέμνει τον άξονα των $x$ σε παραπάνω από ένα σημεία, τότε θα υπήρχαν $x_1 \neq x_2$ ώστε $f(x_1) = f(x_2) = 0$, το οποίο αντιφάσκει με το 1-1.
\item \textbf{Σ.}
Ισχύει ότι $\dint_\alpha^\beta c\,\d x = \left[cx\right]_\alpha^\beta = c\beta - c\alpha = c(\beta-\alpha)$.
\item \textbf{Λ.}
Η γραφική παράσταση της αντίστροφης συνάρτησης $f^{-1}$ είναι συμμετρική της $C_f$ ως προς την ευθεία $y=x$ (διχοτόμος του $1$ου και $3$ου τεταρτημορίου), όχι ως προς την $y=-x$.
\item \textbf{Λ.}
Το θεώρημα \eng{Bolzano} εξασφαλίζει την ύπαρξη \textbf{τουλάχιστον μίας} ρίζας. Το πλήθος των ριζών μπορεί να είναι άπειρο. Για παράδειγμα, η $f(x) = x\hm\frac{1}{x}$ με $f(0)=0$ σε ένα κατάλληλο διάστημα μπορεί να έχει άπειρες ρίζες.

\begin{prosoxi}{\linewidth}
Το θεώρημα \eng{Bolzano} δίνει ικανή (αλλά όχι αναγκαία) συνθήκη για να υπάρχει τουλάχιστον μία ρίζα.
\end{prosoxi}

\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 4ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{\ln x}{x}$, $x > 0$.
\begin{erwthma}

\item Παραγωγίζουμε στο $(0,+\infty)$:
\[ f'(x) = \frac{(\ln x)' x - \ln x (x)'}{x^2} = \frac{\frac{1}{x} x - \ln x}{x^2} = \frac{1 - \ln x}{x^2} \]

Πρόσημο της $f'$:
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow 1 - \ln x = 0 \Leftrightarrow \ln x = 1 \Leftrightarrow x = e$
    \item $f'(x) > 0 \Leftrightarrow 1 - \ln x > 0 \Leftrightarrow \ln x < 1 \Leftrightarrow 0 < x < e$
    \item $f'(x) < 0 \Leftrightarrow 1 - \ln x < 0 \Leftrightarrow \ln x > 1 \Leftrightarrow x > e$
\end{itemize}

Μονοτονία και Ακρότατα:
\begin{itemize}
    \item Η $f$ είναι γνησίως αύξουσα στο $(0,e]$ και γνησίως φθίνουσα στο $[e,+\infty)$.
    \item Παρουσιάζει ολικό μέγιστο στο $x=e$, με τιμή $f(e) = \dfrac{\ln e}{e} = \dfrac{1}{e}$.
\end{itemize}

\underline{Σύγκριση $e^\pi$ και $\pi^e$:}

Γνωρίζουμε ότι το $e$ είναι το μοναδικό σημείο ολικού μεγίστου, άρα $f(x) \le f(e)$ για κάθε $x > 0$, και η ισότητα ισχύει μόνο για $x = e$.
Επειδή $\pi \neq e$:
\[ f(\pi) < f(e) \Rightarrow \frac{\ln \pi}{\pi} < \frac{1}{e} \Rightarrow e \ln \pi < \pi \Rightarrow \ln \pi^e < \pi \Rightarrow e^{\ln \pi^e} < e^\pi \Rightarrow \pi^e < e^\pi \]
Επομένως, ο $e^\pi$ είναι μεγαλύτερος του $\pi^e$.

\item \textbf{Κατακόρυφες Ασύμπτωτες:}
Υπολογίζουμε το όριο στο $0$:
\[ \lim_{x\to 0^+} f(x) = \lim_{x\to 0^+} \left( \frac{1}{x} \cdot \ln x \right) = (+\infty) \cdot (-\infty) = -\infty \]
Άρα η ευθεία $x=0$ (δηλαδή ο άξονας $y'y$) αποτελεί \textbf{κατακόρυφη ασύμπτωτη} της γραφικής παράστασης της $f$.

\textbf{Οριζόντιες Ασύμπτωτες:}
Στο $+\infty$ (ελέγχουμε μορφή απροσδιοριστίας $\frac{+\infty}{+\infty}$):
\[ \lim_{x\to +\infty} f(x) = \lim_{x\to +\infty} \frac{\ln x}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to +\infty} \frac{\frac{1}{x}}{1} = 0 \]
Άρα η ευθεία $y=0$ (δηλαδή ο άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} της $C_f$ στο $+\infty$.

\item Η δοσμένη εξίσωση, για $x>0$ και $\lambda=2$, μετασχηματίζεται ως:
\[ x^2 = e^x \Leftrightarrow \ln(x^2) = \ln(e^x) \Leftrightarrow 2\ln x = x \Leftrightarrow \frac{\ln x}{x} = \frac{1}{2} \Leftrightarrow f(x) = \frac{1}{2} \]

Από το (1) ερώτημα γνωρίζουμε ότι η μέγιστη τιμή της $f$ είναι το $\frac{1}{e}$. Επειδή $e > 2 \Rightarrow \frac{1}{e} < \frac{1}{2}$, συμπεραίνουμε ότι για κάθε $x>0$ ισχύει:
\[ f(x) \le \frac{1}{e} < \frac{1}{2} \]
Συνεπώς, η εξίσωση $f(x) = \frac{1}{2}$ είναι αδύνατη στο $(0,+\infty)$, δηλαδή \textbf{έχει 0 ρίζες}.

\item Για $x \in [1,e]$ ισχύει $\ln x \ge 0$, οπότε $f(x) \ge 0$. Το ζητούμενο εμβαδόν είναι:
\[ E = \int_1^e |f(x)|\, \d x = \int_1^e \frac{\ln x}{x}\, \d x = \int_1^e \ln x \cdot (\ln x)'\, \d x \]
\[ E = \left[ \frac{\ln^2 x}{2} \right]_1^e = \frac{\ln^2 e}{2} - \frac{\ln^2 1}{2} = \frac{1}{2} - 0 = \frac{1}{2} \ \text{τ.μ.} \]

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 4ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Μια ευθεία $x = a$ ($a > 0$) τέμνει την $f(x) = \sqrt{x+1} - 1$ στο $A$ και τον $Ox$ στο $B$.
\begin{erwthma}

\item Οι συντεταγμένες των σημείων είναι $O(0,0)$, $B(a,0)$ και $A(a, f(a))$. Το τρίγωνο είναι ορθογώνιο στο $B$.
Για τα μήκη των κάθετων πλευρών έχουμε:
\[ |OB| = a \quad (\text{αφού } a>0) \]
Για το σημείο $A$ έχουμε $f(a) = \sqrt{a+1}-1$. Επειδή $a>0 \Rightarrow a+1>1 \Rightarrow \sqrt{a+1}>1 \Rightarrow f(a)>0$. Άρα:
\[ |AB| = f(a) = \sqrt{a+1}-1 \]
Το εμβαδόν του τριγώνου $OAB$ είναι:
\[ E(a) = \frac{1}{2} \cdot |OB| \cdot |AB| = \frac{1}{2} a (\sqrt{a+1}-1) \]

\item Το εμβαδόν συναρτήσει του χρόνου $t$ (αφού $a=a(t)$) δίνεται από:
\[ E(t) = \frac{1}{2} a(t) \left(\sqrt{a(t)+1}-1\right) \]
Παραγωγίζουμε ως προς $t$ χρησιμοποιώντας τον κανόνα γινομένου:
\begin{align*}
E'(t) &= \frac{1}{2} a'(t) \left(\sqrt{a(t)+1}-1\right) + \frac{1}{2} a(t) \frac{1}{2\sqrt{a(t)+1}} a'(t) \\
&= \frac{1}{2} a'(t) \left[ \sqrt{a(t)+1} - 1 + \frac{a(t)}{2\sqrt{a(t)+1}} \right]
\end{align*}

Τη χρονική στιγμή $t_0$, δίνονται τα στοιχεία $a(t_0) = 3$ και $\alpha'(t_0) = 2$. Αντικαθιστούμε:
\begin{align*}
E'(t_0) &= \frac{1}{2} \cdot 2 \cdot \left[ \sqrt{3+1} - 1 + \frac{3}{2\sqrt{3+1}} \right] \\
&= 1 \cdot \left[ \sqrt{4} - 1 + \frac{3}{2\sqrt{4}} \right] \\
&= 2 - 1 + \frac{3}{4} = 1 + \frac{3}{4} = \frac{7}{4} \ \text{τ.μ./sec}
\end{align*}

\item Υπολογίζουμε το ολοκλήρωμα εκφράζοντας τη ρίζα ως δύναμη:
\begin{align*}
\int_0^3 f(x)\,\d x &= \int_0^3 (\sqrt{x+1} - 1)\,\d x = \int_0^3 \left( (x+1)^{1/2} - 1 \right)\,\d x \\
&= \left[ \frac{(x+1)^{\frac{3}{2}}}{\frac{3}{2}} - x \right]_0^3 = \left[ \frac{2}{3} (x+1)\sqrt{x+1} - x \right]_0^3 \\
&= \left( \frac{2}{3} \cdot 4\sqrt{4} - 3 \right) - \left( \frac{2}{3} \cdot 1\sqrt{1} - 0 \right) \\
&= \left( \frac{16}{3} - 3 \right) - \frac{2}{3} = \frac{14}{3} - \frac{9}{3} = \frac{5}{3}
\end{align*}

\item Θα το υπολογίσουμε με τη μέθοδο της συζυγούς παράστασης:
\begin{align*}
\lim_{x\to 0} \frac{f(x)}{x} &= \lim_{x\to 0} \frac{\sqrt{x+1}-1}{x} = \lim_{x\to 0} \frac{(\sqrt{x+1}-1)(\sqrt{x+1}+1)}{x(\sqrt{x+1}+1)} \\
&= \lim_{x\to 0} \frac{x+1-1}{x(\sqrt{x+1}+1)} = \lim_{x\to 0} \frac{x}{x(\sqrt{x+1}+1)} \\
&= \lim_{x\to 0} \frac{1}{\sqrt{x+1}+1} = \frac{1}{2}
\end{align*}

\begin{parat}{\linewidth}
Εναλλακτικά, παρατηρούμε πως το όριο ταυτίζεται με τον ορισμό της παραγώγου στο σημείο $0$: \[\lim_{x\to 0} \frac{f(x)-f(0)}{x-0} = f'(0) = \frac{1}{2\sqrt{0+1}} = \frac{1}{2}\]
\end{parat}

\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Έστω $f: \mathbb{R} \to \mathbb{R}$ με $f(0)=0$ και $f'(x) - f(x) > 0$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}

\item Θεωρούμε τη βοηθητική συνάρτηση $g(x) = e^{-x} f(x)$, $x \in \mathbb{R}$.

Η συνάρτηση $g$ είναι παραγωγίσιμη σε όλο το $\mathbb{R} \mu\varepsilon$:
\[ g'(x) = (e^{-x})' f(x) + e^{-x} f'(x) = -e^{-x} f(x) + e^{-x} f'(x) = e^{-x} (f'(x) - f(x)) \]

Αφού ισχύει $e^{-x} > 0$ και η δοσμένη συνθήκη $f'(x) - f(x) > 0$, έπεται ότι:
\[ g'(x) > 0 \quad \text{εκπληρώνεται για κάθε } x \in \mathbb{R} \]
Άρα, η συνάρτηση $g$ είναι \textbf{γνησίως αύξουσα} στο σύνολο τῶν πραγματικών αριθμών $\mathbb{R}$.

Για οποιοδήποτε $x > 0$, αφού η $g$ είναι γνησίως αύξουσα, έχουμε τη συνεπαγωγή:
\[ x > 0 \Rightarrow g(x) > g(0) \Rightarrow e^{-x}f(x) > e^0 f(0) \Rightarrow e^{-x}f(x) > 0 \Rightarrow f(x) > 0 \]
(Λάβαμε υπόψη $e^{-x} > 0$ καθώς και το δεδομένο $f(0)=0$).
Επομένως, πράγματι η συνάρτηση $f$ λαμβάνει αυστηρά θετικές τιμές για κάθε $x > 0$.

\item Θα δείξουμε ότι η $f$ είναι γνησίως αύξουσα στο $[0,+\infty)$. Ουσιαστικά επιδιώκουμε να δείξουμε ότι $f'(x) > 0$ για $x \in (0,+\infty)$.

Από τη δοσμένη συνθήκη έχουμε: $f'(x) - f(x) > 0 \Rightarrow f'(x) > f(x)$ για κάθε $x \in \mathbb{R}$.

Ωστόσο, μόλις αποδείχθηκε (ερώτημα 1) πώς για κάθε $x>0 \Rightarrow f(x)>0$. Συνδυάζοντας τα, για κάθε $x>0$ ισχύει:
\[ f'(x) > f(x) > 0 \Rightarrow f'(x) > 0 \]

Επιπλέον, στο σημείο $x=0$, $f'(0) > f(0) = 0 \Rightarrow f'(0) > 0$. Συνεπώς, $f'(x) > 0$ για κάθε $x \in [0,+\infty)$. Αφού η $f$ είναι και συνεχής (λόγω παραγωγισιμότητας), καταλήγουμε στο συμπέρασμα πως η $f$ είναι \textbf{γνησίως αύξουσα} σε ολόκληρο το διάστημα $[0,+\infty)$.

\item Εξετάσαμε ήδη στο $1$\textsuperscript{ο} ερώτημα ότι η βοηθητική $g(x) = f(x)e^{-x}$ είναι γνησίως αύξουσα σε όλο το $\mathbb{R}$, καθώς το πρόσημο της $g'$ είναι πάντα θετικό.

Εξετάζουμε τυχόν διέλευση του άξονα. Για κάθε $x < 0$, παρελθόντας ότι η $g$ είναι γνησίως αύξουσα, λαμβάνουμε:
\[ x < 0 \Rightarrow g(x) < g(0) \Rightarrow e^{-x} f(x) < 0 \Rightarrow f(x) < 0 \]
Το ζητούμενο λοιπόν απεδείχθη, η $f$ παραμένει αμιγώς αρνητική για τα αντίστοιχα $x$.

\item Δίνεται επιπλέον το σημείο διέλευσης της $f$, όπου $f(1) = e$.
Άρα για την $g$ ισχύει ακριβώς: $g(1) = f(1)e^{-1} = e \cdot e^{-1} = 1$.

Χρησιμοποιούμε εκ νέου τη στοθερή γνησίως αύξουσα φύση της $g$ στο $\mathbb{R}$:
Η συνθήκη για κάθε $x > 1$:
\[ x > 1 \Rightarrow g(x) > g(1) \Rightarrow f(x)e^{-x} > 1 \Rightarrow f(x) > e^x \]

Η συνθήκη για κάθε $x \in (0,1)$:
\[ 0 < x < 1 \Rightarrow g(x) < g(1) \Rightarrow f(x)e^{-x} < 1 \Rightarrow f(x) < e^x \]
Αποδείχθηκαν έτσι τα σκέλη ανισοτήτων.

\textbf{Έλεγχος γωνίας εφαπτομένης:}
Υποθέτουμε, προς άτοπο, ότι υπάρχει εφαπτομένη της $C_f$ σε κάποιο σημείο $x_0 > 1$, η οποία να σχηματίζει με τον θετικό ημιάξονα $Ox$ μια γωνία $\omega \le 45^\circ$. Επειδή $\omega > 0$ (η $f$ είναι αύξουσα, άρα $f'(x_0) > 0$), θα πρέπει η κλίση να υπακούει στη σχέση $\ef\omega \le \ef 45^\circ = 1$, δηλαδή $f'(x_0) \le 1$.

Ωστόσο ανατρέποντας στη δοσμένη ανισότητα:
\[ f'(x_0) > f(x_0) \]
Αλλά επειδή $x_0 > 1$, έχουμε αποδείξει πως $f(x_0) > e^{x_0}$. Ταυτόχρονα, αξιοποιώντας την αυστηρή μονοτονία της εκθετικής, όταν $x_0 > 1 \Rightarrow e^{x_0} > e^1 = e$.
Συνοψίζοντας την αλυσίδα:
\[ f'(x_0) > f(x_0) > e^{x_0} > e > 1 \Rightarrow f'(x_0) > 1 \]

Η καταληκτική ανισότητα $f'(x_0) > 1$ έρχεται σε μετωπική και ξεκάθαρη αντίφαση με τη συνθήκη $f'(x_0) \le 1$. Συνεπώς, η αρχική υπόθεση ήταν εσφαλμένη και \textbf{δεν υπάρχει} τέτοια εφαπτομένη.

\end{erwthma}
\end{thema}
