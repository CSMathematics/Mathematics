\begin{thema}{Α}
\begin{erwthma}
\item \textbf{Θεώρημα Μέσης Τιμής.} Έστω μια συνάρτηση $f$ συνεχής στο $[a,\beta]$ και παραγωγίσιμη στο $(a,\beta)$. Τότε υπάρχει ένα τουλάχιστον $\xi\in(a,\beta)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(\beta)-f(a)}{\beta-a}.
\]

Για την απόδειξη θεωρούμε τη συνάρτηση
\[
\varphi(x)=f(x)-\frac{f(\beta)-f(a)}{\beta-a}(x-a),\quad x\in[a,\beta].
\]
Η $\varphi$ είναι συνεχής στο $[a,\beta]$ και παραγωγίσιμη στο $(a,\beta)$. Επιπλέον
\[
\varphi(a)=f(a)
\]
και
\[
\varphi(\beta)=f(\beta)-\frac{f(\beta)-f(a)}{\beta-a}(\beta-a)=f(a).
\]
Άρα
\[
\varphi(a)=\varphi(\beta).
\]
Επομένως, από το θεώρημα \eng{Rolle}, υπάρχει $\xi\in(a,\beta)$ τέτοιο ώστε
\[
\varphi'(\xi)=0.
\]
Όμως
\[
\varphi'(x)=f'(x)-\frac{f(\beta)-f(a)}{\beta-a}.
\]
Άρα
\[
f'(\xi)-\frac{f(\beta)-f(a)}{\beta-a}=0
\]
και επομένως
\[
{f'(\xi)=\frac{f(\beta)-f(a)}{\beta-a}}.
\]

\item Ορισμένο ολοκλήρωμα μιας συνεχούς συνάρτησης $f$ από το $a$ στο $\beta$ ονομάζουμε τον αριθμό
\[
\dint_a^\beta f(x)\,\d x=F(\beta)-F(a),
\]
όπου $F$ είναι μια παράγουσα της $f$ στο διάστημα που περιέχει τα $a,\beta$.

\item Οι τρεις προϋποθέσεις του Θεωρήματος \eng{Rolle} είναι:
\begin{itemize}
\item η $f$ να είναι συνεχής στο $[a,\beta]$,
\item η $f$ να είναι παραγωγίσιμη στο $(a,\beta)$,
\item να ισχύει $f(a)=f(\beta)$.
\end{itemize}
Τότε υπάρχει τουλάχιστον ένα $\xi\in(a,\beta)$ τέτοιο ώστε
\[
f'(\xi)=0.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Σωστό}. Για κάθε $x$ ισχύει $|f(x)|\geq0$.
\item \textbf{Λάθος}. Σε κάθε επιμέρους διάστημα η $f$ είναι σταθερή, αλλά οι σταθερές μπορεί να είναι διαφορετικές.
\item \textbf{Λάθος}. Γενικά η σύνθεση δεν είναι αντιμεταθετική, δηλαδή $g\circ f\neq f\circ g$.
\item \textbf{Λάθος}. Για παράδειγμα, η $f(x)=x^2$ είναι συνεχής στο $\mathbb{R}$, αλλά δεν είναι $1-1$.
\item \textbf{Σωστό}. Αν $f(x)=0$, τότε
\[
\dint_a^\beta f(x)\,\d x=0.
\]
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Θεωρούμε τη συνάρτηση
\[
f(x)=\frac{e^x}{x},\quad x\in\mathbb{R}^*.
\]
\begin{erwthma}
\item Για την κατακόρυφη ασύμπτωτη εξετάζουμε τα πλευρικά όρια στο $0$:
\[
\lim_{x\to0^+}\frac{e^x}{x}=+\infty
\]
και
\[
\lim_{x\to0^-}\frac{e^x}{x}=-\infty.
\]
Άρα η ευθεία
\[
{x=0}
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Στο $-\infty$ έχουμε
\[
\lim_{x\to-\infty}\frac{e^x}{x}=0,
\]
άρα η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $-\infty$.

Στο $+\infty$ έχουμε
\[
\lim_{x\to+\infty}\frac{e^x}{x}
\xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}}
\lim_{x\to+\infty}e^x=+\infty.
\]
Άρα δεν υπάρχει οριζόντια ασύμπτωτη στο $+\infty$.

\item Για κάθε $x\in\mathbb{R}^*$ είναι
\[
f'(x)=\left(\frac{e^x}{x}\right)'
=\frac{xe^x-e^x}{x^2}
=\frac{e^x(x-1)}{x^2}.
\]
Επειδή $e^x>0$ και $x^2>0$ για κάθε $x\neq0$, το πρόσημο της $f'$ είναι το πρόσημο του $x-1$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow x=1$.
\item $f'(x)<0\Leftrightarrow x<1$, με $x\neq0$.
\item $f'(x)>0\Leftrightarrow x>1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$1$,$+\infty$}%
\tkzTabLine{,-,d,-,z,+,}
\tkzTabLine{,\searrow,d,\searrow,t,\nearrow,}
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι \textbf{γνησίως φθίνουσα} στα $(-\infty,0)$ και $(0,1]$, ενώ είναι \textbf{γνησίως αύξουσα} στο $[1,+\infty)$.

Στο $x=1$ η $f'$ αλλάζει πρόσημο από $-$ σε $+$, άρα η $f$ παρουσιάζει τοπικό ελάχιστο, με τιμή
\[
f(1)=e.
\]

\item Η εξίσωση
\[
e^x=\lambda x
\]
είναι ισοδύναμη, για $x\neq0$, με
\[
\frac{e^x}{x}=\lambda
\Leftrightarrow f(x)=\lambda.
\]
Από τη μελέτη της μονοτονίας και των ορίων έχουμε:
\[
f\bigl((-\infty,0)\bigr)=(-\infty,0)
\]
και
\[
f\bigl((0,+\infty)\bigr)=[e,+\infty).
\]
Επομένως:
\begin{itemize}
\item Αν $\lambda<0$, η εξίσωση έχει \textbf{μία} ρίζα.
\item Αν $0\leq\lambda<e$, η εξίσωση δεν έχει ρίζα.
\item Αν $\lambda=e$, η εξίσωση έχει \textbf{μία} ρίζα, την $x=1$.
\item Αν $\lambda>e$, η εξίσωση έχει \textbf{δύο} ρίζες.
\end{itemize}

\item Έχουμε
\[
\left(\frac{e^x}{x}\right)'=\frac{x e^x-e^x}{x^2}
=\frac{x-1}{x^2}e^x.
\]
Άρα
\begin{align*}
\dint_2^3\frac{x-1}{x^2}e^x\,\d x
&=\dint_2^3\left(\frac{e^x}{x}\right)'\,\d x\\
&=\left[\frac{e^x}{x}\right]_2^3
=\frac{e^3}{3}-\frac{e^2}{2}.
\end{align*}
Επομένως
\[
{\dint_2^3\frac{x-1}{x^2}e^x\,\d x=\frac{e^3}{3}-\frac{e^2}{2}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Από το σχήμα της $f'$ έχουμε
\[
f'(x)=
\begin{cdcases}
x+2, & -2\leq x\leq0\\
2-x, & 0\leq x\leq3
\end{cdcases}.
\]
\begin{erwthma}
\item Από το σχήμα παρατηρούμε ότι
\[
f'(x)>0\quad\text{για }x\in(-2,2)
\]
και
\[
f'(x)<0\quad\text{για }x\in(2,3).
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $[-2,2]$ και \textbf{γνησίως φθίνουσα} στο $[2,3]$.

Στο $x=2$ η $f'$ αλλάζει πρόσημο από $+$ σε $-$, άρα η $f$ παρουσιάζει τοπικό μέγιστο.

Στα άκρα του πεδίου ορισμού, η $f$ παρουσιάζει τοπικά ελάχιστα στα
\[
x=-2
\quad\text{και}\quad
x=3.
\]

\item Η κυρτότητα της $f$ καθορίζεται από το πρόσημο της $f''$, δηλαδή από τη μονοτονία της $f'$.

Στο $(-2,0)$ η $f'$ είναι γνησίως αύξουσα, άρα
\[
f''(x)>0
\]
και η $f$ είναι κυρτή.

Στο $(0,3)$ η $f'$ είναι γνησίως φθίνουσα, άρα
\[
f''(x)<0
\]
και η $f$ είναι κοίλη.

Επομένως η $f$ αλλάζει κυρτότητα στο $x=0$. Από το επόμενο ερώτημα θα έχουμε $f(0)=2$, άρα το σημείο καμπής είναι
\[
{A(0,2)}.
\]

\item Για $x\in[-2,0]$ έχουμε
\[
f'(x)=x+2.
\]
Άρα
\[
f(x)=\frac{x^2}{2}+2x+c_1.
\]
Επειδή $f(-2)=0$, παίρνουμε
\[
2-4+c_1=0
\Leftrightarrow c_1=2.
\]
Άρα
\[
f(x)=\frac{x^2}{2}+2x+2,\quad -2\leq x\leq0.
\]

Για $x\in[0,3]$ έχουμε
\[
f'(x)=2-x.
\]
Άρα
\[
f(x)=2x-\frac{x^2}{2}+c_2.
\]
Επειδή η $f$ είναι συνεχής στο $0$ και από τον πρώτο τύπο έχουμε $f(0)=2$, προκύπτει
\[
c_2=2.
\]
Άρα
\[
f(x)=2x-\frac{x^2}{2}+2,\quad 0\leq x\leq3.
\]
Επομένως
\[
{
f(x)=
\begin{cdcases}
\dfrac{x^2}{2}+2x+2, & -2\leq x\leq0\\
-\dfrac{x^2}{2}+2x+2, & 0\leq x\leq3
\end{cdcases}}
\]

\item Στο διάστημα $(0,3)$ έχουμε
\[
f'(x)=2-x,
\]
άρα
\[
f''(x)=-1.
\]
Επομένως
\[
\dint_0^3 x f''(x)\,\d x
=\dint_0^3 (-x)\,\d x
=\left[-\frac{x^2}{2}\right]_0^3
=-\frac{9}{2}.
\]
Άρα
\[
{\dint_0^3 x f''(x)\,\d x=-\frac{9}{2}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται συνεχώς παραγωγίσιμη συνάρτηση $f:\mathbb{R}\to\mathbb{R}$ με
\[
f(1)=e
\]
και
\[
f'(x)<f(x)
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Θεωρούμε τη συνάρτηση
\[
g(x)=f(x)e^{-x},\quad x\in\mathbb{R}.
\]
Η $g$ είναι παραγωγίσιμη και
\[
g'(x)=e^{-x}\bigl(f'(x)-f(x)\bigr).
\]
Επειδή
\[
e^{-x}>0
\quad\text{και}\quad
f'(x)-f(x)<0,
\]
έχουμε
\[
g'(x)<0
\]
για κάθε $x\in\mathbb{R}$. Άρα η $g$ είναι γνησίως φθίνουσα στο $\mathbb{R}$.

Για κάθε $x>1$ έχουμε
\[
g(x)<g(1)
\Leftrightarrow f(x)e^{-x}<f(1)e^{-1}
\Leftrightarrow f(x)e^{-x}<1.
\]
Άρα
\[
{f(x)<e^x,\quad x>1}.
\]

\item Για $x>1$ θέτουμε
\[
K(x)=e^x-f(x).
\]
Τότε
\[
K'(x)=e^x-f'(x).
\]
Από το προηγούμενο ερώτημα και τη δοσμένη ανισότητα έχουμε
\[
f'(x)<f(x)<e^x.
\]
Άρα
\[
K'(x)=e^x-f'(x)>0.
\]
Επομένως η $K$ είναι \textbf{γνησίως αύξουσα} στο $(1,+\infty)$.

\item Η εξίσωση
\[
f(x)=e^x-1
\]
είναι ισοδύναμη με
\[
e^x-f(x)=1
\Leftrightarrow K(x)=1.
\]
Επειδή η $K$ είναι γνησίως αύξουσα στο $(1,+\infty)$, είναι $1-1$ στο διάστημα αυτό. Άρα η εξίσωση
\[
K(x)=1
\]
έχει το πολύ μία ρίζα στο $(1,+\infty)$.

\item Έστω επιπλέον ότι
\[
\lim_{x\to+\infty}f(x)=0.
\]
Η εξίσωση
\[
f(x)=e^x-2026
\]
είναι ισοδύναμη με
\[
K(x)=2026.
\]
Η $K$ είναι συνεχής και γνησίως αύξουσα στο $(1,+\infty)$. Επίσης
\[
\lim_{x\to1^+}K(x)=K(1)=e-f(1)=0
\]
και
\[
\lim_{x\to+\infty}K(x)=\lim_{x\to+\infty}(e^x-f(x))=+\infty.
\]
Άρα υπάρχει τουλάχιστον ένα $x_0\in(1,+\infty)$ τέτοιο ώστε
\[
K(x_0)=2026.
\]
Επειδή η $K$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

Επομένως η εξίσωση
\[
f(x)=e^x-2026
\]
έχει ακριβώς μία ρίζα στο $(1,+\infty)$.
\end{erwthma}
\end{thema}
