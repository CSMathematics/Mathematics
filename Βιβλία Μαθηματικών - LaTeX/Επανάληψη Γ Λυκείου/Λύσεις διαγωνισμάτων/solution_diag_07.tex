%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Σ.} 
Η απόλυτη τιμή ορίζεται να λαμβάνει πάντοτε μη αρνητικές τιμές ($|f(x)| \ge 0$). Επομένως η γραφική της παράσταση δεν επεκτείνεται ποτέ κάτω από τον άξονα των $x$.
\item \textbf{Λ.}
Αν $\Delta = \Delta_1 \cup \Delta_2$, τα συμπεράσματα συνέπειας του Θ.Μ.Τ. ισχύουν ξεχωριστά σε κάθε διάστημα. Η $f$ θα είναι σταθερή $c_1$ στο $\Delta_1$ και σταθερή $c_2$ στο $\Delta_2$, αλλά γενικά $c_1 \neq c_2$, οπότε δεν είναι απαραίτητα απολύτως σταθερή σε όλο το $\Delta$.
\item \textbf{Λ.}
Η πράξη της σύνθεσης συναρτήσεων δεν είναι γενικά αντιμεταθετική, δηλαδή συνήθως $g(f(x)) \neq f(g(x))$.
\item \textbf{Λ.}
Η συνέχεια εξασφαλίζει ότι το γράφημα δεν έχει «όπες» ή «άλματα». Μια συνεχής συνάρτηση π.χ. $f(x)=x^2$ δεν είναι 1-1 (αφού $f(-1)=f(1)=1$). Μόνο αν είναι \textbf{γνησίως μονότονη} θα είναι και 1-1.
\item \textbf{Σ.}
Πράγματι, αν $f(x) = 0$ για κάθε $x$, τότε $\dint_a^b 0\,\d x = 0$. Η γεωμετρική ερμηνεία είναι πως μια ευθεία (άξονας) δεν σχηματίζει εμβαδόν.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται $f(x) = \dfrac{e^x}{x}$ για $x \in \mathbb{R}^*$.
\begin{erwthma}

\item \textbf{Κατακόρυφες Ασύμπτωτες:}
Υπολογίζουμε τα πλευρικά όρια στο σημείο εκτός πεδίου ορισμού, το $x_0 = 0$:
\[ \lim_{x\to 0^+} \frac{e^x}{x} = (+1) \cdot (+\infty) = +\infty \]
\[ \lim_{x\to 0^-} \frac{e^x}{x} = (+1) \cdot (-\infty) = -\infty \]
Συνεπώς, η ευθεία $x=0$ (ο άξονας $y'y$) αποτελεί \textbf{κατακόρυφη ασύμπτωτη} της $C_f$.

\textbf{Οριζόντιες Ασύμπτωτες:}
$\bullet$ Στο $-\infty$:
\[ \lim_{x\to -\infty} \frac{e^x}{x} = \lim_{x\to -\infty} \left( e^x \cdot \frac{1}{x} \right) = 0 \cdot 0 = 0 \]
Άρα η ευθεία $y=0$ (ο άξονας $x'x$) είναι \textbf{οριζόντια ασύμπτωτη} στο $-\infty$.

$\bullet$ Στο $+\infty$:
\[ \lim_{x\to +\infty} \frac{e^x}{x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to +\infty} \frac{e^x}{1} = +\infty \]
Άρα δεν υπάρχει οριζόντια ασύμπτωτη στο $+\infty$.

\item Παραγωγίζουμε στο $\mathbb{R}^*$:
\[ f'(x) = \frac{(e^x)'x - e^x(x)'}{x^2} = \frac{x e^x - e^x}{x^2} = \frac{e^x(x-1)}{x^2} \]

Πρόσημο της $f'$: (Αφού $e^x > 0$ και $x^2 > 0$ για $x \neq 0$)
\begin{itemize}
    \item $f'(x) = 0 \Leftrightarrow x-1 = 0 \Leftrightarrow x = 1$
    \item $f'(x) > 0 \Leftrightarrow x-1 > 0 \Leftrightarrow x > 1$
    \item $f'(x) < 0 \Leftrightarrow x-1 < 0 \Leftrightarrow x < 1$ (και $x \neq 0$)
\end{itemize}

Μονοτονία: Η συνάρτηση είναι \textbf{γνησίως φθίνουσα} σε καθένα από τα διαστήματα $(-\infty, 0)$ και $(0, 1]$. Επίσης είναι \textbf{γνησίως αύξουσα} στο $[1, +\infty)$.
Աκρότατα: Η $f$ παρουσιάζει \textbf{τοπικό ελάχιστο} στο $x=1$, με τιμή $f(1) = \dfrac{e^1}{1} = e$.

\item Η εξίσωση γράφεται ισοδύναμα (επειδή $x=0 \Rightarrow 1=0$ είναι αδύνατο, άρα $x \neq 0$):
\[ e^x = \lambda x \Leftrightarrow \frac{e^x}{x} = \lambda \Leftrightarrow f(x) = \lambda \]

Βρίσκουμε το σύνολο τιμών $f(\Delta_i)$ για κάθε διάστημα μονοτονίας:
$\bullet$ $\Delta_1 = (-\infty, 0)$: Η $f$ φθίνει από $0$ έως $-\infty$, άρα $f(\Delta_1) = (-\infty, 0)$.
$\bullet$ $\Delta_2 = (0, 1]$: Η $f$ φθίνει από $+\infty$ στο $e$, άρα $f(\Delta_2) = [e, +\infty)$.
$\bullet$ $\Delta_3 = [1, +\infty)$: Η $f$ αυξάνει από $e$ έως $+\infty$, άρα $f(\Delta_3) = [e, +\infty)$.

Με βάση τα παραπάνω συμπεραίνουμε για το πλήθος των ριζών (των οριζόντιων τομών της $C_f$ με την $y=\lambda$):
\begin{itemize}
    \item Αν $\boldsymbol{\lambda < 0}$: η εξίσωση έχει \textbf{1 ρίζα} (ακριβώς στο $\Delta_1$).
    \item Αν $\boldsymbol{0 \le \lambda < e}$: η εξίσωση έχει \textbf{0 ρίζες} (το $\lambda$ δεν ανήκει σε κανένα υποσύνολο τιμών).
    \item Αν $\boldsymbol{\lambda = e}$: η εξίσωση έχει \textbf{1 ρίζα} (το $x=1$ που ανήκει στα $\Delta_2, \Delta_3$ ως κοινό κορυφαίο σημείο-ελάχιστο).
    \item Αν $\boldsymbol{\lambda > e}$: η εξίσωση έχει \textbf{2 ρίζες} (μία ρίζα στο $\Delta_2$ και μία στο $\Delta_3$).
\end{itemize}

\item Το ζητούμενο ολοκλήρωμα μετασχηματίζεται ως εξής:
\[ I = \int_2^3 \frac{x-1}{x^2}e^x\,\d x = \int_2^3 \left( \frac{1}{x} - \frac{1}{x^2} \right)e^x\,\d x \]
Παρατηρούμε ότι ισχύει η σχέση παραγώγισης $\left(\dfrac{1}{x}\right)' = -\dfrac{1}{x^2}$. Συνεπώς αναγνωρίζουμε τον κανόνα παραγώγισης γινομένου $\left(\dfrac{1}{x} e^x\right)'$:
\[ \left( \frac{e^x}{x} \right)' = \left( \frac{1}{x} e^x \right)' = \left(\frac{1}{x}\right)' e^x + \frac{1}{x} (e^x)' = -\frac{1}{x^2}e^x + \frac{1}{x}e^x = \left(\frac{1}{x} - \frac{1}{x^2}\right) e^x \]
Άρα:
\[ I = \int_2^3 \left( \frac{e^x}{x} \right)'\,\d x = \left[ \frac{e^x}{x} \right]_2^3 = \frac{e^3}{3} - \frac{e^2}{2} = \frac{2e^3 - 3e^2}{6} \]

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
\begin{erwthma}

\item \textbf{Μονοτονία:} Το πρόσημο της παραγώγου $f'$ καθορίζει τη μονοτονία της $f$. Τονίζουμε ότι το γράφημα απεικονίζει αυτούσια την \textbf{παράγωγο} $f'$, όχι την $f$.
Από το γράφημα παρατηρούμε πως:
\begin{itemize}
    \item Η $f'(x) > 0$ για $x \in (-2, 2)$ γιατί η γραμμή βρίσκεται πάνω από τον άξονα $x$.
    \item Η $f'(x) < 0$ για $x \in (2, 3]$ γιατί η γραμμή βρίσκεται κάτω από τον άξονα $x$.
    \item Η $f'(x) = 0$ στα σημεία $-2$ και $2$.
\end{itemize}
Εφόσον η $f$ είναι συνεχής, είναι \textbf{γνησίως αύξουσα} στο διάστημα $[-2, 2]$ και \textbf{γνησίως φθίνουσα} στο διάστημα $[2, 3]$. 

\textbf{Ακρότατα:} Αφού η $f'$ αλλάζει πρόσημο από θετικό σε αρνητικό γύρω από το 2, στο $\boldsymbol{x = 2}$ αντιστοιχεί \textbf{τοπικό μέγιστο}.
Επιπρόσθετα, στα κλειστά άκρα εμφανίζονται τοπικά ελάχιστα: η $f$ είναι αύξουσα αμέσως μετά το $\boldsymbol{x = -2}$ (\textbf{τοπικό ελάχιστο}) και φθίνουσα ακριβώς πριν το $\boldsymbol{x = 3}$ (\textbf{τοπικό ελάχιστο}). 

\item \textbf{Κυρτότητα:} Η κυρτότητα καθορίζεται από τη δεύτερη παράγωγο $f''(x)$, η οποία είναι η \textbf{κλίση} (παράγωγος) της ήδη σχεδιασμένης $f'(x)$.
Εξετάζοντας τα επιμέρους ευθύγραμμα τμήματα της $f'$:
\begin{itemize}
    \item Στο $\Delta_1 = (-2, 0)$, το τμήμα είναι αύξον, συνεπώς η κλίση του $f''(x) > 0$. Η $f(x)$ είναι \textbf{κυρτή} στο $[-2, 0]$.
    \item Στο $\Delta_2 = (0, 3)$, το τμήμα είναι φθίνον, συνεπώς η κλίση του $f''(x) < 0$. Η $f(x)$ είναι \textbf{κοίλη} στο $[0, 3]$.
\end{itemize}
Στο σημείο ενότητας $x = 0$, η συνάρτηση αλλάζει κυρτότητα ενώ παραμένει συνεχής, αποτελώντας συνεπώς \textbf{σημείο καμπής}. Η ακριβής τετμημένη είναι η \textbf{$\boldsymbol{x = 0}$}.

\item Χρησιμοποιώντας τα σημεία που διέρχονται τα δύο ευθύγραμμα τμήματα για την $f'(x)$:
$\bullet$ Για $x \in [-2, 0]$: Περνά από $(-2,0)$ και $(0,2)$. Εξίσωση: $y - 0 = \frac{2-0}{0-(-2)}(x+2) \Rightarrow y = x+2$.
Άρα $f'(x) = x+2$. Υπολογίζοντας το αόριστο ολοκλήρωμα:
\[ f(x) = \frac{x^2}{2} + 2x + c_1 \]
Δίνεται ως δεδομένο ότι $f(-2) = 0$.
\[ \frac{(-2)^2}{2} + 2(-2) + c_1 = 0 \Leftrightarrow \frac{4}{2} - 4 + c_1 = 0 \Leftrightarrow c_1 = 2 \]
Συνεπώς $\boldsymbol{f(x) = \dfrac{x^2}{2} + 2x + 2}$ για $x \in [-2, 0]$.

$\bullet$ Για $x \in (0, 3]$: Περνά από $(0,2)$ και $(3,-1)$. 
Κλίση $\lambda = \frac{-1-2}{3-0} = -1$. Εξίσωση $y - 2 = -1(x - 0) \Rightarrow y = -x + 2$.
Άρα $f'(x) = -x+2$. Με ολοκλήρωση:
\[ f(x) = -\frac{x^2}{2} + 2x + c_2 \]
Εξαιτίας του ότι η $f$ είναι συνεχής σε όλο το διάστημα, πρέπει στο $0$ τα πλευρικά όρια (άρα και οι τιμές τους διότι είναι πολυωνυμικές αριστερά και δεξιά) να συμπίπτουν: $f(0^-) = f(0^+) \Rightarrow 2 = c_2$.
Συνεπώς $\boldsymbol{f(x) = -\dfrac{x^2}{2} + 2x + 2}$ για $x \in (0, 3]$.

\item Τοποθετώντας τον τύπο της $f''(x)$ στο διάστημα $(0,3)$, έχουμε είδη διαπιστώσει ότι πρόκειται για σταθερή κλίση, με $f''(x) = -1$.
(Η τιμή της κλίσης του ευθύγραμμου τμήματος της $f'$ στο αντίστοιχο διάστημα).

Υπολογίζουμε εν συνεχεία άμεσα το ολοκλήρωμα:
\begin{align*}
\int_0^3 x f''(x)\,\d x &= \int_0^3 x (-1)\,\d x = \int_0^3 (-x)\,\d x \\
&= \left[ -\frac{x^2}{2} \right]_0^3 = -\frac{9}{2} - \left( -\frac{0}{2} \right) = \boldsymbol{-\frac{9}{2}}
\end{align*}

\begin{parat}{\linewidth}
Εναλλακτικά, θα ήταν απόλυτα σωστή και η εφαρμογή ολοκλήρωσης κατά παράγοντες $\int_0^3 x f''(x)\d x = \left[x f'(x)\right]_0^3 - \int_0^3 f'(x)\d x$, η οποία επιστρέφει το ίδιο ακριβώς αποτέλεσμα (αντικαθιστώντας από τους γνωστούς τύπους).
\end{parat}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 7ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Δίνεται $f(1)=e$ και $f'(x) < f(x)$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}

\item Θέτουμε τη βοηθητική συνάρτηση $g(x) = e^{-x} f(x)$, με $x \in \mathbb{R}$.
Παραγωγίζουμε την $g$:
\[ g'(x) = (e^{-x})' f(x) + e^{-x} f'(x) = -e^{-x} f(x) + e^{-x} f'(x) = e^{-x} (f'(x) - f(x)) \]

Διότι $e^{-x} > 0$ και $f'(x) - f(x) < 0$ (από τη συνθήκη), προκύπτει ξεκάθαρα πως $g'(x) < 0$ σε ολόκληρο τον $\mathbb{R}$. Η συνάρτηση $g$ παίζει ως \textbf{γνησίως φθίνουσα}.

Για τη διάταξη των $x > 1$, λαμβάνουμε:
\[ x > 1 \stackrel{g \searrow}{\Longrightarrow} g(x) < g(1) \]
\[ e^{-x} f(x) < e^{-1} f(1) \]
Εφαρμόζοντας το δεδομένο $f(1) = e$:
\[ e^{-x} f(x) < e^{-1} \cdot e \Rightarrow e^{-x} f(x) < 1 \Rightarrow \boldsymbol{f(x) < e^x} \]
Η ζητούμενη ανισότητα αποδείχθηκε επιτυχώς.

\item Διαθέτουμε την $K(x) = e^x - f(x)$ για $x \in (1, +\infty)$.
Εκτελούμε παραγώγιση:
\[ K'(x) = e^x - f'(x) \]

Από τη δοσμένη διαφορική συνθήκη παρατηρούμε πως $f'(x) < f(x)$ σε ολόκληρο το εδραίωμα. Παράλληλα, μόλις αποδείχθηκε (ερώτημα 1) πώς για κάθε ανεξάρτητη μεταβλητή $x > 1$, ισχύει $f(x) < e^x$.
Συρράπτοντας αυτά τα δύο για $x > 1$:
\[ f'(x) < f(x) < e^x \Rightarrow f'(x) < e^x \Rightarrow e^x - f'(x) > 0 \]
Καταλήγουμε στο $K'(x) > 0$ αποκλειστικά για το διάστημα $(1, +\infty)$. Η $K(x)$ λογίζεται επομένως ως \textbf{γνησίως αύξουσα} σε αυτό το διάστημα ανοιχτής διεύρυνσης.

\item Η προς εξέταση εξίσωση $f(x) = e^x - 1$ ισοδυναμεί απόλυτα με την:
\[ e^x - f(x) = 1 \Leftrightarrow K(x) = 1 \]

Αφού στο προηγούμενο ερώτημα ορίσαμε την $K(x)$ ως \textbf{γνησίως αύξουσα} στο $(1, +\infty)$, η συνάρτηση διαθέτει την ιδιότητα 1-1 στο παρόν διάστημα, αρά αποκλείεται να επιτύχει την ίδια τιμή δύο ξεχωριστές φορές. Ένεκα αυτού, η εξίσωση $K(x) = 1$ δέχεται \textbf{το πολύ μία ρίζα}.

\begin{parat}{\linewidth}
Όταν ζητείται η απόδειξη του "το πολύ μία ρίζα", επαληθεύουμε αποκλειστικά τη μέθοδο της μονοτονίας/1-1. Δεν επενδύουμε σε έλεγχο θεώρηματος \eng{Bolzano} αν δεν ζητείται ρητά ύπαρξη.
\end{parat}

\item Η εξίσωση που επικροτείται είναι η $f(x) = e^x - 2026 \Leftrightarrow K(x) = 2026$.
Η $K(x) = e^x - f(x)$ είναι παραγωγίσιμη, οπότε \textbf{συνεχής} στο $\Delta = (1, +\infty)$ και \textbf{γνησίως αύξουσα}.

Οι οριακές τιμές του διαστήματος αναλύονται ως εξής:
$\bullet$ Στο $1^+$ (λόγω της ασφαλούς εξασφαλισμένης συνέχειας στο 1):
\[ \lim_{x\to 1^+} K(x) = K(1) = e^1 - f(1) = e - e = 0 \]
$\bullet$ Στο $+\infty$:
Η αποστολή ορίου $\lim_{x\to+\infty} K(x) = \lim_{x\to+\infty} (e^x - f(x))$.
Γνωρίζουμε ως απαρέγκλιτο δεδομένο ότι $\lim_{x\to+\infty} f(x) = 0$ και επειδή $\lim_{x\to+\infty} e^x = +\infty$, εξάγεται:
\[ \lim_{x\to+\infty} K(x) = +\infty - 0 = +\infty \]

Κατά συνέπεια, το εδραιωμένο σύνολο τιμών της $K$ στο συγκεκριμένο διάστημα αντιστοιχεί στο $K(\Delta) = (0, +\infty)$.
Το ζητούμενο νούμερο $2026 \in K(\Delta) = (0, +\infty)$. Έτσι εξασφαλίζεται η διασταύρωση, με άλλα λόγια υπάρχει \textbf{ακριβώς μία ρίζα} στο διάστημα $(1, +\infty)$ διότι η συνάρτηση συνδράμει στη μονοτονία του ολικού 1-1.
\end{erwthma}
\end{thema}
