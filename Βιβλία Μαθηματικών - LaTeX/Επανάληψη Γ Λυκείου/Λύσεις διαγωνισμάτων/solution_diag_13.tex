\begin{thema}{Α}
\begin{erwthma}
\item Έστω $\eta$ ένας αριθμός μεταξύ των $f(a)$ και $f(\beta)$.

Επειδή η $f$ είναι συνεχής στο $[a,\beta]$, από το Θεώρημα Ενδιάμεσων Τιμών, η $f$ παίρνει κάθε τιμή μεταξύ των $f(a)$ και $f(\beta)$.

Άρα υπάρχει τουλάχιστον ένα
\[
x_0\in(a,\beta)
\]
τέτοιο ώστε
\[
f(x_0)=\eta.
\]
Επομένως, για κάθε ενδιάμεση τιμή $\eta$ μεταξύ των $f(a)$ και $f(\beta)$, υπάρχει σημείο $x_0\in(a,\beta)$ στο οποίο η $f$ παίρνει την τιμή αυτή.

\item Μια συνάρτηση $f$ λέγεται κοίλη, ή λέμε ότι στρέφει τα κοίλα προς τα κάτω σε ένα διάστημα $\varDelta$, όταν για κάθε $x_1,x_2\in\varDelta$ και κάθε $\lambda\in[0,1]$ ισχύει
\[
f\bigl(\lambda x_1+(1-\lambda)x_2\bigr)
\geq
\lambda f(x_1)+(1-\lambda)f(x_2).
\]
Γεωμετρικά, η γραφική της παράσταση βρίσκεται πάνω από κάθε χορδή της.

\item Αν η συνάρτηση $f$ είναι παραγωγίσιμη στο $x_0$, τότε εφαπτομένη της $C_f$ στο σημείο
\[
A(x_0,f(x_0))
\]
ονομάζουμε την ευθεία που διέρχεται από το $A$ και έχει συντελεστή διεύθυνσης $f'(x_0)$.

Η εξίσωσή της είναι
\[
y-f(x_0)=f'(x_0)(x-x_0).
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Η γνήσια μονοτονία συνεπάγεται ότι η συνάρτηση είναι $1-1$, αλλά το αντίστροφο δεν ισχύει γενικά χωρίς πρόσθετες υποθέσεις.
\item \textbf{Σωστό}. Είναι το Θεώρημα Ενδιάμεσων Τιμών.
\item \textbf{Λάθος}. Η $f(x)=|x|$ δεν είναι παραγωγίσιμη στο $0$, αφού η αριστερή παράγωγος είναι $-1$ και η δεξιά παράγωγος είναι $1$.
\item \textbf{Σωστό}. Η σταθερά βγαίνει έξω από το ορισμένο ολοκλήρωμα:
\[
\dint_a^\beta cf(x)\,\d x
=c\dint_a^\beta f(x)\,\d x.
\]
\item \textbf{Σωστό}. Κάθε γνησίως αύξουσα συνάρτηση σε ένα διάστημα είναι $1-1$.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=x^2e^{-x},\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=2xe^{-x}-x^2e^{-x}
=x(2-x)e^{-x}.
\]
Επειδή $e^{-x}>0$, το πρόσημο της $f'$ καθορίζεται από το $x(2-x)$.

Έχουμε
\[
f'(x)=0\Leftrightarrow x=0\ \text{ή}\ x=2.
\]
Στον παρακάτω πίνακα φαίνονται τα πρόσημα της $f'$ και η μονοτονία της $f$:
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=2,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ /0.8,$f'(x)$ /0.8,$f(x)$/0.8}%
{$-\infty$,$0$,$2$,$+\infty$}%
\tkzTabLine{,-,z,+,z,-,}
\tkzTabLine{,\searrow,t,\nearrow,t,\searrow,}
\end{tikzpicture}
\end{center}

Άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$, γνησίως αύξουσα στο $[0,2]$ και γνησίως φθίνουσα στο $[2,+\infty)$.

Παρουσιάζει τοπικό ελάχιστο στο $x=0$, με
\[
f(0)=0,
\]
και τοπικό μέγιστο στο $x=2$, με
\[
f(2)=4e^{-2}=\frac{4}{e^2}.
\]

\item Για $x\to+\infty$ έχουμε
\[
\lim_{x\to+\infty}x^2e^{-x}
=\lim_{x\to+\infty}\frac{x^2}{e^x}=0.
\]
Άρα η ευθεία
\[
{y=0}
\]
είναι οριζόντια ασύμπτωτη της $C_f$ στο $+\infty$.

Για $x\to-\infty$ έχουμε
\[
x^2\to+\infty
\quad\text{και}\quad
e^{-x}\to+\infty,
\]
οπότε
\[
f(x)\to+\infty.
\]
Δεν υπάρχει οριζόντια ασύμπτωτη στο $-\infty$.

Για πλάγια ασύμπτωτη στο $-\infty$ εξετάζουμε
\[
\lim_{x\to-\infty}\frac{f(x)}{x}
=\lim_{x\to-\infty}xe^{-x}
=-\infty.
\]
Άρα δεν υπάρχει πλάγια ασύμπτωτη στο $-\infty$.

\item Έχουμε
\[
f'(x)=e^{-x}(2x-x^2).
\]
Άρα
\begin{align*}
f''(x)
&=e^{-x}(2-2x)-e^{-x}(2x-x^2)\\
&=e^{-x}(x^2-4x+2).
\end{align*}
Επειδή $e^{-x}>0$, το πρόσημο της $f''$ καθορίζεται από το
\[
x^2-4x+2.
\]
Οι ρίζες του είναι
\[
x=2-\sqrt2
\quad\text{και}\quad
x=2+\sqrt2.
\]
Άρα η $f$ είναι κυρτή στα διαστήματα
\[
(-\infty,2-\sqrt2)
\quad\text{και}\quad
(2+\sqrt2,+\infty)
\]
και κοίλη στο διάστημα
\[
(2-\sqrt2,2+\sqrt2).
\]

Επομένως τα σημεία καμπής είναι
\[
\left(2-\sqrt2,(2-\sqrt2)^2e^{-(2-\sqrt2)}\right)
\]
και
\[
\left(2+\sqrt2,(2+\sqrt2)^2e^{-(2+\sqrt2)}\right).
\]

\item Υπολογίζουμε με παραγοντική ολοκλήρωση:
\begin{align*}
\dint_0^2 x^2e^{-x}\,\d x
&=\left[-x^2e^{-x}\right]_0^2+\dint_0^2 2xe^{-x}\,\d x\\
&=-\frac{4}{e^2}+2\dint_0^2 xe^{-x}\,\d x.
\end{align*}
Για το δεύτερο ολοκλήρωμα:
\begin{align*}
\dint_0^2 xe^{-x}\,\d x
&=\left[-xe^{-x}\right]_0^2+\dint_0^2 e^{-x}\,\d x\\
&=-\frac{2}{e^2}+\left[-e^{-x}\right]_0^2\\
&=-\frac{2}{e^2}+1-\frac{1}{e^2}\\
&=1-\frac{3}{e^2}.
\end{align*}
Άρα
\[
\dint_0^2 x^2e^{-x}\,\d x
=-\frac{4}{e^2}+2\left(1-\frac{3}{e^2}\right)
=2-\frac{10}{e^2}.
\]
Επομένως
\[
{\dint_0^2 x^2e^{-x}\,\d x=2-\frac{10}{e^2}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Η $C_f$ αποτελείται από τα ευθύγραμμα τμήματα που ενώνουν τα σημεία
\[
(0,0),\quad (2,4),\quad (6,0).
\]
\begin{erwthma}
\item Στο διάστημα $[0,2]$ η ευθεία περνά από τα σημεία $(0,0)$ και $(2,4)$, άρα έχει κλίση
\[
\frac{4-0}{2-0}=2.
\]
Επομένως ο τύπος της είναι
\[
f(x)=2x,\quad 0\leq x\leq2.
\]

Στο διάστημα $[2,6]$ η ευθεία περνά από τα σημεία $(2,4)$ και $(6,0)$, άρα έχει κλίση
\[
\frac{0-4}{6-2}=-1.
\]
Επομένως ο τύπος της είναι
\[
f(x)=-x+6,\quad 2<x\leq6.
\]
Άρα
\[
{
f(x)=
\begin{cases}
2x, & 0\leq x\leq2,\\
-x+6, & 2<x\leq6.
\end{cases}}
\]

\item Το ολοκλήρωμα
\[
\dint_0^6 f(x)\,\d x
\]
ισούται με το εμβαδόν του τριγώνου με βάση $6$ και ύψος $4$.
Άρα
\[
\dint_0^6 f(x)\,\d x
=\frac12\cdot6\cdot4=12.
\]
Επομένως
\[
{\dint_0^6 f(x)\,\d x=12}.
\]

\item Στο $x=2$ έχουμε
\[
f(2)=4.
\]
Η αριστερή παράγωγος είναι η κλίση του πρώτου ευθύγραμμου τμήματος:
\[
f'_-(2)=2.
\]
Η δεξιά παράγωγος είναι η κλίση του δεύτερου ευθύγραμμου τμήματος:
\[
f'_+(2)=-1.
\]
Επειδή
\[
f'_-(2)\neq f'_+(2),
\]
η $f$ δεν είναι παραγωγίσιμη στο $x_0=2$.

\item Έχουμε
\[
\frac{f(6)-f(0)}{6}
=\frac{0-0}{6}=0.
\]
Για $x\in(0,2)$ είναι
\[
f'(x)=2,
\]
ενώ για $x\in(2,6)$ είναι
\[
f'(x)=-1.
\]
Στο $x=2$ η $f$ δεν είναι παραγωγίσιμη.

Άρα δεν υπάρχει $\xi\in(0,6)$ τέτοιο ώστε
\[
f'(\xi)=0=\frac{f(6)-f(0)}{6}.
\]

Αυτό δεν έρχεται σε αντίφαση με το Θεώρημα Μέσης Τιμής, γιατί η $f$ δεν είναι παραγωγίσιμη σε όλο το $(0,6)$, αφού δεν είναι παραγωγίσιμη στο $x=2$. Επομένως δεν ικανοποιεί όλες τις προϋποθέσεις του θεωρήματος.
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
f:\mathbb{R}\to\mathbb{R}
\]
με
\[
f(0)=1
\]
και
\[
f'(x)\geq f(x)
\]
για κάθε $x\geq0$.
\begin{erwthma}
\item Θεωρούμε
\[
g(x)=f(x)e^{-x}.
\]
Τότε
\[
g'(x)=f'(x)e^{-x}-f(x)e^{-x}
=e^{-x}\bigl(f'(x)-f(x)\bigr).
\]
Για κάθε $x\geq0$ έχουμε
\[
f'(x)-f(x)\geq0
\]
και
\[
e^{-x}>0.
\]
Άρα
\[
{g'(x)\geq0,\quad x\geq0}.
\]

\item Από το προηγούμενο ερώτημα η $g$ είναι αύξουσα στο $[0,+\infty)$.
Άρα, για κάθε $x\geq0$,
\[
g(x)\geq g(0).
\]
Όμως
\[
g(0)=f(0)e^0=1.
\]
Άρα
\[
f(x)e^{-x}\geq1.
\]
Επειδή $e^x>0$, παίρνουμε
\[
{f(x)\geq e^x,\quad x\geq0}.
\]

\item Από το προηγούμενο ερώτημα, για κάθε $x\in[0,1]$ ισχύει
\[
f(x)\geq e^x.
\]
Ολοκληρώνοντας κατά μέλη στο $[0,1]$, παίρνουμε
\[
\dint_0^1 f(x)\,\d x
\geq
\dint_0^1 e^x\,\d x
=e-1.
\]
Άρα
\[
{\dint_0^1 f(x)\,\d x\geq e-1}.
\]

\item Αν επιπλέον
\[
f(1)=e,
\]
τότε
\[
g(1)=f(1)e^{-1}=1.
\]
Από το πρώτο ερώτημα η $g$ είναι αύξουσα στο $[0,1]$, ενώ
\[
g(0)=1
\quad\text{και}\quad
g(1)=1.
\]
Επομένως, για κάθε $x\in[0,1]$ ισχύει
\[
1=g(0)\leq g(x)\leq g(1)=1.
\]
Άρα
\[
g(x)=1
\]
για κάθε $x\in[0,1]$.

Δηλαδή
\[
f(x)e^{-x}=1
\]
και επομένως
\[
{f(x)=e^x,\quad x\in[0,1]}.
\]
\end{erwthma}
\end{thema}
