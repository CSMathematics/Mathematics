\begin{thema}{Α}
\begin{erwthma}
\setcounter{enumi}{3}
\item \ \\
\begin{alist}
\item \textbf{Σ.} 
Μία (απλώς) αύξουσα συνάρτηση, χωρίς το επίθετο «γνησίως», επιτρέπεται να παραμένει σταθερή σε κάποια ενδιάμεσα υποδιαστήματα ($f(x_1) \le f(x_2)$). Σε αυτά τα διαστήματα δίνει ίδιες τιμές, οπότε δεν είναι αναγκαστικά 1-1, και κατά συνέπεια μπορεί να μην αντιστρέφεται.
\item \textbf{Σ.}
Αποτελεί το Θεώρημα Ενδιάμεσων Τιμών. Μια συνεχής συνάρτηση στο κλειστό λαμβάνει υποχρεωτικά όλες τις ενδιάμεσες τιμές ανάμεσα στο ελάχιστο και στο μέγιστό της (άρα και από τα άκρα της).
\item \textbf{Λ.}
Στο υποψήφιο σημείο $x=0$, η απόλυτη τιμή αλλάζει κλάδο και σχηματίζει τη γνωστή γωνία (κορυφή του $V$). Η αριστερή παράγωγος είναι $-1$ και η δεξιά παράγωγος είναι $1$. Επειδή δε συμφωνούν, δεν είναι παραγωγίσιμη εκεί.
\item \textbf{Σ.}
Ισχύει. Μια σταθερά επιτρέπεται να εξάγεται ως κοινός παράγοντας απόλυτα έξω από το σύμβολο του ορισμένου ολοκληρώματος.
\item \textbf{Σ.}
Πράγματι, η γνήσια μονοτονία αποτρέπει οποιεσδήποτε δύο ανεξάρτητες τιμές να απεικονιστούν στην ίδια τιμή $y$. Εξασφαλίζει πλήρως τη συνθήκη 1-1.
\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 13ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 e^{-x}$ (ίδια συνάρτηση με το διαγώνισμα 10).
\begin{erwthma}

\item \textbf{Μονοτονία και Ακρότατα:}
Η $f$ παραγωγίζεται με $(uv)'$:
\[ f'(x) = 2x e^{-x} - x^2 e^{-x} = x(2-x) e^{-x} \]
- Επειδή $e^{-x} > 0$, οι ρίζες της $f'$ ταυτίζονται με τις ρίζες του $x(2-x)=0$, δηλαδή $x=0$ και $x=2$.
- Στο διάστημα $(0, 2)$ ισχύει $f'(x) > 0$.
- Στα διαστήματα $(-\infty, 0)$ και $(2, +\infty)$ ισχύει $f'(x) < 0$.

Επομένως, παρατηρείται:
$\bullet$ \textbf{Ολικό/Τοπικό ελάχιστο} στο $x=0$, με $f(0) = 0$.
$\bullet$ \textbf{Τοπικό μέγιστο} στο $x=2$, με $f(2) = 4 e^{-2} = \frac{4}{e^2}$.

\item \textbf{Ασύμπτωτες:}
Στο $-\infty$:
\[ \lim_{x\to -\infty} f(x) = \lim (-\infty)^2 e^{+\infty} = (+\infty)(+\infty) = +\infty \]
Άρα ούτε οριζόντια, ούτε πλάγια (διότι $f(x)/x \to -\infty$).

Στο $+\infty$:
\[ \lim_{x\to +\infty} x^2 e^{-x} = \lim_{x\to +\infty} \frac{x^2}{e^x} \xlongequal[\text{\eng{DLH}}]{\frac{+\infty}{+\infty}} \lim_{x\to +\infty} \frac{2x}{e^x} \xlongequal[\text{\eng{DLH}}]{\dots} 0 \]
Κατά συνέπεια, η $C_f$ διαθέτει \textbf{οριζόντια ασύμπτωτη} την ευθεία $\boldsymbol{y = 0}$ στο $+\infty$.

\item \textbf{Κυρτότητα και Σημεία Καμπής:}
Παραγωγίζουμε την $f'$:
\begin{align*}
f''(x) &= (2x - x^2)' e^{-x} - (2x - x^2) e^{-x} = (2 - 2x) e^{-x} - (2x - x^2) e^{-x} \\
&= e^{-x} (x^2 - 4x + 2)
\end{align*}
Οι ρίζες του πολυωνύμου είναι τα $\rho_{1,2} = 2 \pm \sqrt{2}$.

- $f''(x) > 0$ για $x \in (-\infty, 2-\sqrt{2}) \cup (2+\sqrt{2}, +\infty) \Rightarrow f$ \textbf{κοίλα άνω}.
- $f''(x) < 0$ για $x \in (2-\sqrt{2}, 2+\sqrt{2}) \Rightarrow f$ \textbf{κοίλα κάτω}.

Η γραφική παράσταση παρουσιάζει \textbf{δύο σημεία καμπής} στα ακριβή σημεία αλλαγής κυρτότητας $x = 2-\sqrt{2}$ και $x = 2+\sqrt{2}$.

\item Υπολογισμός ορισμένου ολοκληρώματος με διαδοχικές ολοκληρώσεις κατά παράγοντες (επειδή $-e^{-x} = (e^{-x})'$):
\begin{align*}
I &= \int_0^2 x^2 (-e^{-x})'\,\d x = \left[ -x^2 e^{-x} \right]_0^2 - \int_0^2 (x^2)' (-e^{-x})\,\d x \\
&= \left( -4e^{-2} - 0 \right) + \int_0^2 2x e^{-x}\,\d x \\
&= -\frac{4}{e^2} + \int_0^2 2x (-e^{-x})'\,\d x \\
&= -\frac{4}{e^2} + \left[ -2x e^{-x} \right]_0^2 - \int_0^2 (2x)' (-e^{-x})\,\d x \\
&= -\frac{4}{e^2} + \left( -4e^{-2} - 0 \right) + 2 \int_0^2 e^{-x}\,\d x \\
&= -\frac{8}{e^2} + 2 \left[ -e^{-x} \right]_0^2 = -\frac{8}{e^2} + 2 \left( -e^{-2} + 1 \right) \\
&= -\frac{8}{e^2} - \frac{2}{e^2} + 2 = \boldsymbol{2 - \frac{10}{e^2}}
\end{align*}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 13ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Συνάρτηση $f:[0, 6] \to \mathbb{R}$ οριζόμενη από τα διαδοχικά σημεία $(0,0) \to (2,4) \to (6,0)$.
\begin{erwthma}

\item Από το γράφημα εξάγουμε τις εξισώσεις των ευθειών που στεγάζουν τα τμήματα:
- Στο $[0, 2]$, η ευθεία περνά από το $(0,0)$ και το $(2,4)$. Είναι της μορφής $y = \lambda x$ με κλίση $\lambda = \frac{4-0}{2-0} = 2$. Άρα $y = 2x$.
- Στο $[2, 6]$, η ευθεία περνά από το $(2,4)$ και το $(6,0)$. Συντελεστής διεύθυνσης $\lambda = \frac{0-4}{6-2} = \frac{-4}{4} = -1$.
Εξίσωση: $y - 0 = -1(x - 6) \Rightarrow y = -x + 6$.
Συνεκτικά:
\[
f(x) = \begin{cases} 
2x, & 0 \leq x \leq 2 \\ 
-x + 6, & 2 < x \leq 6 
\end{cases}
\]

\item Το εμβαδόν αντιστοιχεί χωροταξικά αποκλειστικά σε ένα τρίγωνο με βάση $[0, 6]$, δηλαδή μήκος $b=6$, και ύψος την κορυφή $(2,4)$, δηλαδή $h=4$.
Εφαρμόζουμε τον κανόνα γεωμετρίας:
\[ \int_0^6 f(x)\,\d x = \frac{1}{2} \cdot b \cdot h = \frac{1}{2} \cdot 6 \cdot 4 = \frac{24}{2} = \boldsymbol{12 \ \text{τ.μ.}} \]

\item Ελέγχουμε την ύπαρξη της παραγώγου στο σημείο κάμψης $x=2$:
$\lim_{x\to 2^-} \frac{f(x)-f(2)}{x-2} = \lim_{x\to 2^-} \frac{2x-4}{x-2} = \lim \frac{2(x-2)}{x-2} = 2$ (η κλίση $\lambda_1$).
$\lim_{x\to 2^+} \frac{f(x)-f(2)}{x-2} = \lim_{x\to 2^+} \frac{-x+6-4}{x-2} = \lim \frac{-(x-2)}{x-2} = -1$ (η κλίση $\lambda_2$).
Αφού $2 \neq -1$, η $f$ \textbf{δεν είναι παραγωγίσιμη} στο $x=2$. 
Τέτοια σημεία αποκαλούνται γεωμετρικά \textbf{γωνιακά σημεία}. Οι πλευρικές εφαπτόμενες (ημιευθείες) δημιουργούν μία ξεκάθαρη γωνία στο επίπεδο (της οποίας η εφαπτομένη τριγωνομετρική είναι $\varepsilon\phi\phi = \left| \frac{2 - (-1)}{1 + 2(-1)} \right| = 3$).

\item Ζητείται να εφαρμοστεί το Θ.Μ.Τ στο συνολικό εύρος $[0, 6]$.
Αν ίσχυε πλήρως το Θ.Μ.Τ. για τη συγκεκριμένη $f$, θα υπήρχε κάποιο $\xi \in (0,6)$ για το οποίο:
\[ f'(\xi) = \frac{f(6) - f(0)}{6 - 0} = \frac{0 - 0}{6} = 0 \]
Όμως, γνωρίζουμε ήδη ότι για οποιοδήποτε $\xi \in (0,2)$ ο λόγος (παράγωγος) είναι $2$, και για οποιοδήποτε $\xi \in (2,6)$ η κλίση είναι σταθερά $-1$. Στο δε $x=2$ το παράγωγο δεν ορίζεται. 
Έτσι, πράγματι \textbf{δεν υφίσταται} κανένα τέτοιο $\xi$ όπου να μηδενίζεται η παράγωγος. 

\textbf{Εξήγηση αντίφασης:} Αυτό δεν αντιφάσκει διόλου με τα Θεωρήματα του Διαφορικού. Το Θ.Μ.Τ. προϋποθέτει ρητά ότι η συνάρτηση είναι **παραγωγίσιμη** στο ολόκληρο ανοιχτό διάστημα $(0,6)$. Η δικιά μας $f$ παρουσιάζει γωνία (μη παραγωγισιμότητα) στο $x=2 \in (0,6)$, παραβιάζοντας έτσι τις συνθήκες επιβολής του θεωρήματος.
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται $f:\mathbb{R} \to \mathbb{R}$ παραγωγίσιμη με $f(0)=1$ και $f'(x) \geq f(x)$ για $x \geq 0$.
\begin{erwthma}

\item Ορίζουμε την $g(x) = f(x) e^{-x}$ για το πεδίο των θετικών πραγματικών.
Ως γινόμενο παραγωγίσιμων, η $g$ παραγωγίζεται:
\[ g'(x) = f'(x) e^{-x} + f(x) (e^{-x})' = f'(x) e^{-x} - f(x) e^{-x} = e^{-x} \left( f'(x) - f(x) \right) \]
Μας δίνεται η απόλυτη ανίσωση πως αν $x \geq 0 \Rightarrow f'(x) \geq f(x) \Leftrightarrow f'(x) - f(x) \geq 0$.
Δεδομένου επιπρόσθετα ότι η εκθετική $e^{-x} > 0$, ο πολλαπλασιασμός διατηρεί το πρόσημο:
\[ \boldsymbol{g'(x) \geq 0 \ \text{για κάθε } x \geq 0} \]

\item Εξαιτίας του ότι $g'(x) \geq 0$, η συνάρτηση $g(x)$ αποδεικνύεται \textbf{αύξουσα} (non-decreasing) στο εν λόγω διάστημα $[0, +\infty)$.
Επομένως διεκδικεί ότι για κάθε $x \geq 0$:
\[ g(x) \geq g(0) \]
Υπολογίζουμε την τιμή αναφοράς:
\[ g(0) = f(0) e^0 = 1 \cdot 1 = 1 \]
Αντικαθιστούμε πίσω στη σχέση ορισμού:
\[ f(x) e^{-x} \geq 1 \Rightarrow \boldsymbol{f(x) \geq e^x \ \text{για κάθε } x \geq 0} \]

\item Έχουμε ήδη αποδείξει την ανισότητα $f(x) \geq e^x$ η οποία ισχύει για όλα τα όρια κατά μήκος της ολοκλήρωσης, καθώς εργαζόμαστε στο περιορισμένο $[0,1]$ που ανήκει στο $[0, +\infty)$.
Ολοκληρώνουμε κατά μέλη διατηρώντας ορθά τη φορά:
\[ \int_0^1 f(x)\,\d x \geq \int_0^1 e^x\,\d x \]
Ο γνωστός υπολογισμός: $\int_0^1 e^x\,\d x = [e^x]_0^1 = e^1 - e^0 = e - 1$.
Επομένως, εξακριβώνεται πλήρως ότι $\boldsymbol{\dint_0^1 f(x)\,\d x \geq e - 1}$.

\item Θεωρούμε γνωστό το γεγονός ότι εάν επιπλέον $f(1) = e$, τότε το $g(1)$ δίνει:
\[ g(1) = f(1) e^{-1} = e \cdot \frac{1}{e} = 1 \]
Ωστόσο, είχαμε προγενέστερα αποδείξει πως η $g$ είναι αυξουσα, έχοντας $\boldsymbol{1} \leq g(x)$ για όλα τα $x\in[0,1]$.
Αφού στο δεξί όριο η αυξουσα συνάρτηση τερματίζει πάλι στο $\boldsymbol{1}$, είναι απόλυτα αδύνατο η $g$ να ανέβηκε γεωμετρικά και μετά να κατέβηκε (αφού απαγορεύεται να φθίνει).
Αυτό μεταφράζεται αναγκαστικά στο ότι η $g$ είναι \textbf{απόλυτα σταθερή} στο ενδιάμεσο διάστημα $[0,1]$:
\[ g(x) = 1 \quad \text{για κάθε } x \in [0,1] \]
Αντικαθιστώντας:
\[ f(x) e^{-x} = 1 \Rightarrow \boldsymbol{f(x) = e^x \quad \text{για κάθε } x \in [0,1]} \]

\end{erwthma}
\end{thema}
