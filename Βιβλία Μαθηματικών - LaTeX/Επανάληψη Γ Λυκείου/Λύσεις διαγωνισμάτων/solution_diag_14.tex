\begin{thema}{Α}
\begin{erwthma}
\item Έστω ότι η $f$ παρουσιάζει τοπικό μέγιστο στο $x_0$. Τότε υπάρχει $\delta>0$ τέτοιο ώστε, για κάθε
\[
x\in(x_0-\delta,x_0+\delta),
\]
να ισχύει
\[
f(x)\leq f(x_0).
\]
Για $h>0$ αρκετά μικρό έχουμε
\[
\frac{f(x_0+h)-f(x_0)}{h}\leq0,
\]
ενώ για $h<0$ αρκετά μικρό, επειδή $h<0$, έχουμε
\[
\frac{f(x_0+h)-f(x_0)}{h}\geq0.
\]
Επειδή η $f$ είναι παραγωγίσιμη στο $x_0$, τα δύο πλευρικά όρια είναι ίσα με $f'(x_0)$. Άρα
\[
f'(x_0)\leq0
\quad\text{και}\quad
f'(x_0)\geq0,
\]
οπότε
\[
f'(x_0)=0.
\]
Αν η $f$ παρουσιάζει τοπικό ελάχιστο στο $x_0$, η απόδειξη είναι ανάλογη, με τις ανισότητες αντίστροφες. Επομένως, σε κάθε περίπτωση τοπικού ακροτάτου,
\[
{f'(x_0)=0}.
\]

\item Μια συνάρτηση $f$ λέγεται αντιστρέψιμη, όταν είναι $1-1$, δηλαδή όταν για κάθε $x_1,x_2$ του πεδίου ορισμού της ισχύει
\[
f(x_1)=f(x_2)\Rightarrow x_1=x_2.
\]
Τότε ορίζεται η αντίστροφη συνάρτηση $f^{-1}$ με πεδίο ορισμού το σύνολο τιμών της $f$ και ισχύει
\[
f^{-1}(f(x))=x.
\]

\item Αν η $f$ είναι συνεχής στο $[a,\beta]$ και
\[
f(x)\geq0
\]
για κάθε $x\in[a,\beta]$, τότε το ορισμένο ολοκλήρωμα
\[
\dint_a^\beta f(x)\,\d x
\]
εκφράζει το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις ευθείες
\[
x=a,\quad x=\beta.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Σωστό}. Αν η $f$ είναι κυρτή, τότε η γραφική της παράσταση βρίσκεται πάνω από κάθε εφαπτομένη της, εκτός από το σημείο επαφής.
\item \textbf{Σωστό}. Κάθε πολυωνυμική συνάρτηση άρτιου βαθμού έχει ίδια συμπεριφορά στα δύο άκρα του άξονα, οπότε, λόγω συνέχειας, παρουσιάζει τουλάχιστον ένα ολικό και άρα τοπικό ακρότατο.
\item \textbf{Σωστό}. Με αλλαγή της σειράς των άκρων ολοκλήρωσης αλλάζει το πρόσημο:
\[
\dint_a^\beta f(x)\,\d x
=-\dint_\beta^a f(x)\,\d x.
\]
\item \textbf{Σωστό}. Αν η $f$ είναι $1-1$, τότε η αντίστροφη της αντίστροφης είναι η ίδια η $f$, δηλαδή
\[
(f^{-1})^{-1}=f.
\]
\item \textbf{Λάθος}. Η συνθήκη $f'(x_0)>0$ αφορά μόνο την παράγωγο στο σημείο $x_0$ και δεν εξασφαλίζει υποχρεωτικά ότι η $f$ είναι γνησίως αύξουσα σε ολόκληρη περιοχή του $x_0$.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση
\[
f(x)=(x-1)\ln x,\quad x>0.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=\ln x+\frac{x-1}{x}
=\ln x+1-\frac1x.
\]
Επίσης
\[
f''(x)=\frac1x+\frac{1}{x^2}>0
\]
για κάθε $x>0$. Άρα η $f'$ είναι γνησίως αύξουσα στο $(0,+\infty)$.

Επειδή
\[
f'(1)=\ln1+1-1=0,
\]
και η $f'$ είναι γνησίως αύξουσα, έχουμε
\[
f'(x)<0\quad\text{για }0<x<1
\]
και
\[
f'(x)>0\quad\text{για }x>1.
\]

Άρα η $f$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.

Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
f(1)=0.
\]

\item Από το προηγούμενο ερώτημα, η ελάχιστη τιμή της $f$ στο $(0,+\infty)$ είναι
\[
f(1)=0.
\]
Άρα
\[
{f(x)\geq0,\quad x>0}.
\]

\item Για $x_0=e$ έχουμε
\[
f(e)=(e-1)\ln e=e-1.
\]
Επίσης
\[
f'(e)=\ln e+1-\frac1e
=2-\frac1e.
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $e$ είναι
\[
y-(e-1)=\left(2-\frac1e\right)(x-e).
\]
Ισοδύναμα,
\[
{y=\frac{2e-1}{e}x-e}.
\]

\item Υπολογίζουμε
\[
I=\dint_1^e (x-1)\ln x\,\d x.
\]
Θέτουμε
\[
u=\ln x,\quad \d v=(x-1)\,\d x.
\]
Τότε
\[
\d u=\frac1x\,\d x,\quad v=\frac{x^2}{2}-x.
\]
Άρα
\begin{align*}
I
&=\left[\left(\frac{x^2}{2}-x\right)\ln x\right]_1^e
-\dint_1^e\left(\frac{x^2}{2}-x\right)\frac1x\,\d x\\
&=\frac{e^2}{2}-e-\dint_1^e\left(\frac{x}{2}-1\right)\,\d x\\
&=\frac{e^2}{2}-e-\left[\frac{x^2}{4}-x\right]_1^e\\
&=\frac{e^2}{2}-e-\left(\frac{e^2}{4}-e-\frac14+1\right)\\
&=\frac{e^2}{4}-\frac34.
\end{align*}
Επομένως
\[
{\dint_1^e f(x)\,\d x=\frac{e^2-3}{4}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Η γραφική παράσταση της $f'$ αποτελείται από τα ευθύγραμμα τμήματα που ενώνουν τα σημεία
\[
(-2,-4),\quad (1,2),\quad (4,-1).
\]
Άρα
\[
f'(x)=
\begin{cases}
2x, & -2\leq x\leq1,\\
-x+3, & 1<x\leq4.
\end{cases}
\]
Επίσης δίνεται ότι
\[
f(1)=3.
\]
\begin{erwthma}
\item Από το πρόσημο της $f'$ έχουμε:
\[
f'(x)<0\quad\text{για }-2<x<0,
\]
\[
f'(x)>0\quad\text{για }0<x<3,
\]
και
\[
f'(x)<0\quad\text{για }3<x<4.
\]
Άρα η $f$ είναι γνησίως φθίνουσα στο $[-2,0]$, γνησίως αύξουσα στο $[0,3]$ και γνησίως φθίνουσα στο $[3,4]$.

Επομένως παρουσιάζει τοπικό ελάχιστο στο $x=0$ και τοπικό μέγιστο στο $x=3$.

\item Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού,
\[
\dint_{-2}^{1}f'(x)\,\d x=f(1)-f(-2).
\]
Το ολοκλήρωμα υπολογίζεται γεωμετρικά από το γράφημα της $f'$:
\[
\dint_{-2}^{1}f'(x)\,\d x
=-4+1=-3.
\]
Άρα
\[
-3=3-f(-2),
\]
οπότε
\[
{f(-2)=6}.
\]

\item Πρώτα υπολογίζουμε το $f(4)$:
\[
f(4)-f(1)=\dint_1^4 f'(x)\,\d x.
\]
Γεωμετρικά,
\[
\dint_1^4 f'(x)\,\d x
=2-\frac12=\frac32.
\]
Άρα
\[
f(4)=3+\frac32=\frac92.
\]

Η κλίση της χορδής που ενώνει τα σημεία
\[
A(-2,f(-2)),\quad B(4,f(4))
\]
είναι
\[
\lambda_{AB}
=\frac{f(4)-f(-2)}{4-(-2)}
=\frac{\frac92-6}{6}
=-\frac14.
\]

Ζητάμε σημείο $\xi$ με
\[
f'(\xi)=-\frac14.
\]
Στο πρώτο τμήμα,
\[
2\xi=-\frac14
\Leftrightarrow
\xi=-\frac18\in(-2,1).
\]
Άρα υπάρχει τουλάχιστον ένα τέτοιο $\xi$, συγκεκριμένα
\[
{\xi=-\frac18}.
\]

\item Για $x<1$ η γραφική παράσταση της $f'$ έχει κλίση $2$, άρα η $f'$ είναι γνησίως αύξουσα και η $f$ είναι κυρτή.

Για $x>1$ η γραφική παράσταση της $f'$ έχει κλίση $-1$, άρα η $f'$ είναι γνησίως φθίνουσα και η $f$ είναι κοίλη.

Επομένως στο $x_0=1$ η $f$ αλλάζει κυρτότητα, από κυρτή σε κοίλη. Άρα η γραφική παράσταση της $f$ παρουσιάζει σημείο καμπής στο σημείο με τετμημένη
\[
{x_0=1}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται η συνάρτηση
\[
f(x)=x-\ln x,\quad x>0.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=1-\frac1x=\frac{x-1}{x}.
\]
Επειδή $x>0$, το πρόσημο της $f'$ καθορίζεται από το $x-1$.

Άρα η $f$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.

Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με
\[
f(1)=1-\ln1=1.
\]

Για την κυρτότητα,
\[
f''(x)=\frac{1}{x^2}>0
\]
για κάθε $x>0$. Άρα η $f$ είναι κυρτή στο $(0,+\infty)$.

\item Από το ολικό ελάχιστο του προηγούμενου ερωτήματος έχουμε
\[
f(x)\geq f(1)=1
\]
για κάθε $x>0$. Δηλαδή
\[
x-\ln x\geq1.
\]
Άρα
\[
{\ln x\leq x-1,\quad x>0}.
\]

\item Από την ανισότητα
\[
\ln x\leq x-1,\quad x>0,
\]
θέτουμε
\[
x=\frac{a}{b}.
\]
Επειδή $a,b>0$, έχουμε $\frac{a}{b}>0$, άρα
\[
\ln\frac{a}{b}\leq \frac{a}{b}-1.
\]
Δηλαδή
\[
\ln a-\ln b\leq \frac{a-b}{b}.
\]
Πολλαπλασιάζοντας με $b>0$, παίρνουμε
\[
{b\ln a-b\ln b\leq a-b}.
\]

\item Θεωρούμε τη συνάρτηση
\[
H(x)=\dint_1^x f(t)\,\d t-\frac{e-1}{2},\quad x\in[1,e].
\]
Η $H$ είναι συνεχής στο $[1,e]$ και παραγωγίσιμη στο $(1,e)$ με
\[
H'(x)=f(x)=x-\ln x.
\]
Από το δεύτερο ερώτημα έχουμε
\[
f(x)\geq1>0
\]
για κάθε $x>0$. Άρα
\[
H'(x)>0
\]
στο $(1,e)$, οπότε η $H$ είναι γνησίως αύξουσα.

Επίσης
\[
H(1)=-\frac{e-1}{2}<0.
\]
Ακόμη,
\[
\dint_1^e f(t)\,\d t>\dint_1^e 1\,\d t=e-1,
\]
αφού $f(t)\geq1$ και η ισότητα ισχύει μόνο για $t=1$.
Άρα
\[
H(e)=\dint_1^e f(t)\,\d t-\frac{e-1}{2}>0.
\]

Από το Θεώρημα Bolzano υπάρχει τουλάχιστον μία ρίζα της $H$ στο $(1,e)$. Επειδή η $H$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

Επομένως η εξίσωση
\[
\dint_1^x f(t)\,\d t=\frac{e-1}{2}
\]
έχει
\[
{\text{ακριβώς μία ρίζα στο }(1,e)}.
\]
\end{erwthma}
\end{thema}
