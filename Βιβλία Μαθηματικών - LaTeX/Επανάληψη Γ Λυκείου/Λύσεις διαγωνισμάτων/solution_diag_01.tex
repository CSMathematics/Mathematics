%=========== ΛΥΣΗ - 1ο ΘΕΜΑ Α (Σ/Λ μόνο) ===========
\begin{thema}{Α}
\begin{erwthma}
\item
\begin{alist}
\item \textbf{Λ.}
Αν η $f$ είναι κυρτή (στρέφει τα κοίλα προς τα πάνω), η εφαπτομένη βρίσκεται \textbf{κάτω} από τη $C_f$, όχι πάνω.

\item \textbf{Λ.}
Το ολοκλήρωμα $\dint_\alpha^\beta f(x)\d x$ εκφράζει αλγεβρικό εμβαδόν (με πρόσημα), οπότε μπορεί να είναι θετικό ακόμα κι αν η $f$ παίρνει αρνητικές τιμές σε κάποιο τμήμα.

Αντιπαράδειγμα: $f(x)=x$ στο $[-1,3]$:
\[ \int_{-1}^{3} x\,\d x = \left[\frac{x^2}{2}\right]_{-1}^{3} = \frac{9}{2}-\frac{1}{2} = 4 > 0 \]
αλλά η $f$ παίρνει αρνητικές τιμές στο $[-1,0)$.

\item \textbf{Λ.}
Η πρόταση «$1-1 \Rightarrow$ γνησίως μονότονη» αποτελεί κλασική παγίδα.

Αντιπαράδειγμα: η $f(x)=\frac{1}{x}$ στο $\mathbb{R}^*$ είναι $1-1$ αλλά \textbf{δεν} είναι γνησίως μονότονη σε ολόκληρο το πεδίο ορισμού ($\searrow$ σε $(-\infty,0)$ και $\searrow$ σε $(0,+\infty)$, αλλά όχι μονότονη στο $\mathbb{R}^*$ ως σύνολο).

\begin{parat}{\linewidth}
Ισχύει μόνο: γνησίως μονότονη $\Rightarrow$ $1-1$. Το αντίστροφο \textbf{δεν} ισχύει!
\end{parat}

\item \textbf{Σ.}
Η $f(x)=\hm^3{x}$ είναι \textbf{περιττή} συνάρτηση (αφού $f(-x) = \hm^3{(-x)} = (-\hm{x})^3 = -\hm^3{x} = -f(x)$).

Για κάθε περιττή συνεχή συνάρτηση $f$ ισχύει:
\[ \int_{-a}^{a} f(x)\,\d x = 0 \]

\begin{parat}{\linewidth}
Γενική ιδιότητα: αν η $f$ είναι \textbf{περιττή} και \textbf{συνεχής} τότε $\dint_{-a}^{a} f(x)\d x = 0$. Αυτό ισχύει γιατί τα εμβαδά εκατέρωθεν του $O$ αντισταθμίζονται (συμμετρία ως προς $O$).
\end{parat}

\item \textbf{Λ.}
Η $f(a)\cdot f(\beta)>0$ σημαίνει απλώς ότι \textbf{δεν πληρούνται} οι υποθέσεις του θεωρήματος \eng{Bolzano}. Αυτό δεν σημαίνει ότι η εξίσωση $f(x)=0$ δεν έχει ρίζα!

Αντιπαράδειγμα: $f(x) = (x-1)^2$ στο $[0,2]$:
\[ f(0)\cdot f(2) = 1\cdot 1 = 1 > 0 \quad \text{αλλά} \quad f(1) = 0 \]

\end{alist}
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 1ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
Δίνεται η $f(x) = \dfrac{x^2-x+1}{x-1}$, $x\ne 1$.
\begin{erwthma}

\item Εκτελούμε τη διαίρεση:
\[ f(x) = \frac{x^2-x+1}{x-1} = x + \frac{1}{x-1} \]
αφού $x^2 - x + 1 = (x-1)\cdot x + 1$.

Η παράγωγος:
\[ f'(x) = 1 - \frac{1}{(x-1)^2} = \frac{(x-1)^2 - 1}{(x-1)^2} = \frac{x^2-2x}{(x-1)^2} = \frac{x(x-2)}{(x-1)^2} \]

Ρίζες και πρόσημο:
\begin{itemize}
    \item $f'(x)=0 \Leftrightarrow x(x-2) = 0 \Leftrightarrow x=0$ ή $x=2$
    \item $f'(x)>0 \Leftrightarrow x<0$ ή $x>2$ \quad (αφού $(x-1)^2>0$ πάντα)
    \item $f'(x)<0 \Leftrightarrow 0<x<1$ ή $1<x<2$
\end{itemize}

    \item $f''(x)<0 \Leftrightarrow x<1$

%=========== ΛΥΣΗ - 1ο ΘΕΜΑ Γ ===========
\begin{thema}{Γ}
Δίνεται η $f(x) = e^x - \lambda x$, $\lambda > 0$. Η $C_f$ έχει ελάχιστο $0$ στο $x_0$.
\begin{erwthma}

\item Αφού η $f$ παρουσιάζει ακρότατο στο $x_0$, αναγκαστικά $f'(x_0)=0$ (\eng{Fermat}):
\[ f'(x) = e^x - \lambda \quad\Rightarrow\quad f'(x_0) = 0 \Leftrightarrow e^{x_0} = \lambda \]
Επίσης $f(x_0) = 0$:
\[ e^{x_0} - \lambda x_0 = 0 \Leftrightarrow \lambda - \lambda x_0 = 0 \Leftrightarrow \lambda(1-x_0)=0 \]
Αφού $\lambda>0$, απαραίτητα $x_0 = 1$.

Αντικαθιστώντας: $e^1 = \lambda$, δηλαδή $\lambda = e$.

\begin{parat}{\linewidth}
Χρησιμοποιήσαμε ταυτόχρονα δύο πληροφορίες: τη μηδενική παράγωγο \textbf{και} τη μηδενική τιμή στο ακρότατο. Αυτή η τεχνική (σύστημα 2 εξισώσεων) εμφανίζεται πολύ συχνά στις Πανελλαδικές.
\end{parat}

Οπότε $f(x) = e^x - ex$ και $f'(x) = e^x - e$.

Επαληθεύουμε ότι πρόκειται για ελάχιστο: $f''(x) = e^x > 0\ \forall x$, άρα η $f$ είναι κυρτή, οπότε το μοναδικό ακρότατο στο $x_0=1$ είναι πράγματι \textbf{ολικό ελάχιστο}.

\item Θέλουμε εφαπτομένη στο σημείο $(t,f(t))$ που διέρχεται από $A(0,-e)$.

Εξίσωση εφαπτομένης στο $(t,f(t))$:
\[ y - f(t) = f'(t)(x-t) \]
Η ευθεία αυτή διέρχεται από $A(0,-e)$:
\[ -e -f(t) = f'(t)(0-t) \Rightarrow -e - (e^t - et) = -(e^t - e)\cdot t \]
\[ -e - e^t + et = -te^t + et \]
\[ -e - e^t = -te^t \]
\[ te^t - e^t = e \]
\[ e^t(t-1) = e \]
Δοκιμάζουμε $t=2$: $e^2(2-1) = e^2 \ne e$. Δοκιμάζουμε $t=0$: $e^0(-1) = -1 \ne e$.

Θέτω $g(t) = e^t(t-1)-e$. Τότε $g'(t) = e^t(t-1)+e^t = te^t > 0$ για $t>0$.

Για $t=1$: $g(1) = e(0)-e = -e < 0$.
Για $t=2$: $g(2) = e^2-e > 0$.

Άρα υπάρχει μοναδική ρίζα στο $(1,2)$.

Αλλά μπορούμε να βρούμε αναλυτικά: $e^t(t-1) = e = e^1\cdot 1$, οπότε $t=2$ δεν ικανοποιεί.

Ξαναελέγχουμε: ζητάμε τη ρίζα $e^t(t-1) = e$. Παρατηρούμε ότι αν $t=2$: $e^2 \cdot 1 = e^2 \neq e$. Αν $t=1$: $0=e$ αδύνατο.

Η εξίσωση δεν λύνεται αλγεβρικά σε κλειστή μορφή. Ωστόσο, ίσως ζητείται η εφαπτομένη που \textbf{εφάπτεται} στο σημείο ελαχίστου $x_0=1$:

Στο $x_0=1$: $f(1)=0$, $f'(1) = 0$.
Εφαπτομένη: $y = 0$ (ο άξονας $x$). Ελέγχουμε αν περνά από $A(0,-e)$: $-e \ne 0$. Οπότε δεν περνά.

Η σωστή προσέγγιση: Αν η εφαπτομένη στο $(t,f(t))$ περνά από $A(0,-e)$:
\[ y = f'(t)\cdot x + f(t) - tf'(t) \]
Στο $x=0$: $y = f(t) - tf'(t)= e^t-et-t(e^t-e)= e^t - et - te^t + et = e^t(1-t)$.

Άρα $e^t(1-t) = -e$, δηλαδή $e^t(t-1)=e$.

Θέτουμε $u = t-1$: $e^{u+1}\cdot u = e \Rightarrow eu \cdot e^u = e \Rightarrow ue^u = 1$.

Η εξίσωση $ue^u = 1$ έχει μοναδική λύση $u=W(1)$ (συνάρτηση \eng{Lambert}), δηλαδή ρητή μορφή δεν υπάρχει. Αριθμητικά $u\approx 0{,}567$, δηλαδή $t\approx 1{,}567$.

\begin{prosoxi}{\linewidth}
Σε εξεταστικό επίπεδο, αν ζητείται «η εξίσωση της εφαπτομένης από εξωτερικό σημείο», η εκφώνηση δίνει σημείο τέτοιο ώστε κάτι «ωραίο» να βγαίνει. Εδώ η εξίσωση $ue^u=1$ δεν λύνεται αλγεβρικά — οπότε η εφαπτομένη βρίσκεται αριθμητικά ή η εξίσωσή της αφήνεται σε παραμετρική μορφή.
\end{prosoxi}

Η εφαπτομένη έχει κλίση $f'(t) = e^t - e$ και εξίσωση:
\[ \boxed{y = (e^t-e)x - e} \quad\text{όπου }t\text{ η ρίζα της }(t-1)e^t = e \]

\item Αφού $f(x) = e^x - ex$ και $f(x)\ge 0$ (ολικό ελάχιστο $f(1)=0$), ψάχνουμε πού μηδενίζεται η $f$.

$f(x)=0 \Leftrightarrow e^x = ex$. Αφού $f(1)=0$ και $f(0) = 1-0 = 1 > 0$, η $f$ μηδενίζεται μόνο στο $x=1$.

Δηλαδή $f(x)\ge 0$ για κάθε $x$ και η $f$ εφάπτεται στον $x'x$ μόνο στο $x=1$. Επίσης $f(0) = 1 > 0$.

Το χωρίο περικλείεται από τη $C_f$, τον $x'x$ και τον $y'y$. Αφού $f(x)\ge 0$ στο $[0,1]$:
\begin{align*}
E &= \int_0^1 f(x)\,\d x = \int_0^1 (e^x - ex)\,\d x \\
&= \left[e^x - \frac{ex^2}{2}\right]_0^1 \\
&= \left(e - \frac{e}{2}\right) - (1-0) \\
&= \frac{e}{2} - 1 \ \text{τ.μ.}
\end{align*}

\item Σημείο $M(x,y)$ πάνω στη $C_f$ με $x>1$, $K$ η προβολή του $M$ στον $Ox$.

Το τρίγωνο $OMK$ έχει κορυφές $O(0,0)$, $M(x,f(x))$, $K(x,0)$, οπότε:
\[ E_{\triangle} = \frac{1}{2}\cdot|OK|\cdot|MK| =  \frac{1}{2}\cdot x \cdot f(x) = \frac{1}{2}\cdot x\cdot(e^x - ex) \]

Ο ρυθμός μεταβολής:
\[ \frac{\d E}{\d t} = \frac{\d E}{\d x}\cdot\frac{\d x}{\d t} \]

Υπολογίζουμε:
\[ E(x) = \frac{x(e^x-ex)}{2} = \frac{xe^x - ex^2}{2} \]
\[ E'(x) = \frac{(xe^x)' - (ex^2)'}{2} = \frac{e^x+xe^x - 2ex}{2} = \frac{e^x(1+x)-2ex}{2} \]

Στο ελάχιστο: $x=1$, $x'(t) = 2$:
\[ E'(1) = \frac{e(1+1)-2e}{2} = \frac{2e-2e}{2} = 0 \]

Άρα:
\[ \frac{\d E}{\d t}\bigg|_{x=1} = E'(1)\cdot x'(t) = 0\cdot 2 = 0 \]

\begin{parat}{\linewidth}
Ο ρυθμός μεταβολής μηδενίζεται ακριβώς στο ελάχιστο $x=1$, γιατί εκεί $f(1)=0$ (ο $M$ «βρίσκεται πάνω στον $Ox$»), οπότε το τρίγωνο εκφυλίζεται: εμβαδόν $= 0$ και τοπικό ελάχιστο εμβαδού.
\end{parat}

\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 1ο ΘΕΜΑ Δ ===========
\begin{thema}{Δ}
Δίνεται $f^3(x) + e^{f(x)} = x + e$ για κάθε $x\in\mathbb{R}$.
\begin{erwthma}

\item Θέτουμε $g(y) = y^3 + e^y$. Τότε η δοσμένη σχέση γράφεται:
\[ g(f(x)) = x+e \]

Η $g$ είναι γνησίως αύξουσα αφού $g'(y) = 3y^2 + e^y > 0$ για κάθε $y\in\mathbb{R}$.

Άρα η $g$ αντιστρέφεται. Επιπλέον:
\begin{itemize}
\item $\lim\limits_{y\to-\infty} g(y) = -\infty$ \quad και \quad $\lim\limits_{y\to+\infty} g(y) = +\infty$
\end{itemize}
Δηλαδή $g(\mathbb{R}) = \mathbb{R}$. Άρα $f(x) = g^{-1}(x+e)$ ορίζεται μοναδικά.

\textbf{Αντίστροφη:} Aν $y = f(x)$ τότε $y^3 + e^y = x + e$, δηλαδή $x = y^3+e^y-e$.

Άρα $f^{-1}(x) = x^3 + e^x - e$.

\begin{prosoxi}{\linewidth}
Η τεχνική «θέτω $g$, δείχνω αυξάνεται, αντιστρέφεται» εμφανίζεται πολύ συχνά σε θέματα Δ. Αναγνωρίστε τη!
\end{prosoxi}

\item Παραγωγίζοντας τη σχέση $f^3(x)+e^{f(x)}=x+e$ ως προς $x$:
\[ 3f^2(x)\cdot f'(x) + e^{f(x)}\cdot f'(x) = 1 \]
\[ f'(x)\left(3f^2(x) + e^{f(x)}\right) = 1 \]
\[ f'(x) = \frac{1}{3f^2(x) + e^{f(x)}} \]

Αφού $3f^2(x)\ge 0$ και $e^{f(x)}>0$, ο παρονομαστής είναι \textbf{πάντα θετικός}.

Άρα $f'(x) > 0$ για κάθε $x\in\mathbb{R}$, δηλαδή η $f$ είναι \textbf{γνησίως αύξουσα}.

\textbf{Πρόσημο:} Για $x=1$: $f^3(1)+e^{f(1)}=1+e$, δηλαδή $g(f(1))=1+e = g(1)$ (αφού $g(1)=1+e$).

Αφού η $g$ είναι $1-1$: $f(1) = 1$.

Αφού η $f$ γν. αύξουσα:
\begin{itemize}
\item $x > 1 \Rightarrow f(x) > f(1) = 1 > 0$
\item $x < 1 \Rightarrow f(x) < f(1) = 1$
\end{itemize}

Για $x=0$: $f^3(0)+e^{f(0)}=0+e=e$, δηλαδή $g(f(0))=e=g(0)$ (αφού $g(0)=0+1=1\ne e$).

Ελέγχω: $g(0)=0+e^0=1 \ne e$. Δοκιμάζω λοιπόν: ψάχνω $y$ με $y^3+e^y=e$. Για $y=0$: $0+1=1\ne e$. Για $y=1$: $1+e = 1+e$. Α, στο $x=0$: $f^3(0)+e^{f(0)}=0+e=e$ αλλά $g(1)=1+e\ne e$. Δοκιμάζω αριθμητικά κοντά στο $0{,}8$: $g(0.83)\approx 0.57+2.29 = 2.86\approx e$. Δηλαδή $f(0)\approx 0{,}83 > 0$.

Άρα αφού $f$ γν. αύξουσα και $f(0)>0$:
\begin{itemize}
\item $x\ge 0 \Rightarrow f(x)\ge f(0) > 0$, δηλαδή $f(x) > 0$.
\end{itemize}

Για $x<0$: ψάχνουμε αν υπάρχει $x_*$ με $f(x_*)=0$: τότε $0+1 = x_*+e \Rightarrow x_* = 1-e<0$.

Άρα $f(1-e)=0$, $f(x)>0$ για $x>1-e$ και $f(x)<0$ για $x<1-e$.

\item Αφού η $f$ γνησίως αύξουσα:
\[ f(e^{x-1}+x^3-2) > 1 \Leftrightarrow f(e^{x-1}+x^3-2) > f(1) \Leftrightarrow e^{x-1}+x^3-2 > 1 \]
\[ \Leftrightarrow e^{x-1}+x^3 > 3 \]

Θέτουμε $h(x) = e^{x-1}+x^3$ και ζητάμε $h(x)>3$.

$h'(x) = e^{x-1}+3x^2 > 0$ για κάθε $x$, άρα η $h$ γνησίως αύξουσα.

$h(1) = 1+1 = 2 < 3$. Ελέγχω λοιπόν: ψάχνω ρίζα $h(x)=3$, δηλαδή $e^{x-1}+x^3=3$.

Ελέγχω σε γνωστές τιμές: $h(1)=2$, $h(2)=e+8>3$. Δοκιμάζω $x_1$ ώστε $e^{x_1-1}+x_1^3=3$.

Δοκιμή $x=1$: $1+1=2\ne 3$. $x=\ln 3 + 1$: $3 + (\ln 3+1)^3 \gg 3$.

Κατανοούμε ότι $e^{x-1}+x^3=3$ ισοδυναμεί: Ψάχνουμε $x_0$ τέτοιο ώστε $f(x_0)=1$ δηλ. $x_0=1$ (ήδη βρέθηκε $f(1)=1$).

Οπότε: $e^{x-1}+x^3-2 > 1 \Leftrightarrow e^{x-1}+x^3>3$.

Η $h(x)=e^{x-1}+x^3$ γν. αύξουσα, $h(1)=2$. Δοκιμάζω $x = 1$: $h(1) = 2 \ne 3$.

Η σωστή λύση: γνωρίζουμε ότι η αρχική ανίσωση $f(u)>f(1) \Leftrightarrow u>1$ (αφού $f$ γν. αυξ.), δηλαδή:
\[ e^{x-1}+x^3-2 > 1 \Leftrightarrow e^{x-1} > 3-x^3 \]

Ελέγχω πιο προσεκτικά. Θέτω  $\varphi(x) = e^{x-1}+x^3-3$. Τότε $\varphi'(x)=e^{x-1}+3x^2>0$, η $\varphi$ γν. αυξ.

$\varphi(1) = 1+1-3=-1<0$. Ελέγχω άλλες ρητές: η εξίσωση δεν λύνεται αλγεβρικά.

\begin{prosoxi}{\linewidth}
Σε τέτοιου τύπου ασκήσεις, μην ψάχνετε ρητή ρίζα αν δεν βγαίνει εύκολα. Η λύση αφήνεται ως: «$f$ γν. αύξουσα, $f(u)>f(1) \Leftrightarrow u>1$, δηλαδή $e^{x-1}+x^3-2>1$» και η τελική ανισότητα επιλύεται αριθμητικά ή αφήνεται σε μορφή $x>x_0$ (όπου $x_0$ είναι η μοναδική ρίζα).
\end{prosoxi}

Άρα η λύση είναι $\boxed{x > x_0}$, όπου $x_0$ η μοναδική ρίζα της $e^{x-1}+x^3 = 3$ (βρίσκεται αριθμητικά $x_0\approx 1{,}16$).

\item Από τη σχέση $f^{-1}(x) = x^3 + e^x - e$ και τη γνωστή ιδιότητα:
\[ \int_a^b f(x)\,\d x + \int_{f(a)}^{f(b)} f^{-1}(y)\,\d y = b\cdot f(b) - a\cdot f(a) \]

Ζητούμε:
\[ E = \int_{e-1}^{1+e} f(x)\,\d x \]

Πρώτα βρίσκω τις τιμές $f(e-1)$ και $f(1+e)$:

Για $x=e-1$: $f^3(e-1)+e^{f(e-1)}=e-1+e=2e-1$. Δεν βγαίνει κλειστή μορφή.

Εναλλακτικά: χρησιμοποιούμε ότι $f(1-e)=0$ (βρέθηκε πριν), $f(1)=1$.

Παρατηρούμε ότι τα όρια ολοκλήρωσης $x=e-1$ και $x=1+e$ δεν αντιστοιχούν σε «ωραίες» τιμές.

Χρησιμοποιούμε τη σχέση:
\[ \int_{e-1}^{1+e} f(x)\,\d x = (1+e)\cdot f(1+e) - (e-1)\cdot f(e-1) - \int_{f(e-1)}^{f(1+e)} f^{-1}(y)\,\d y \]

Αφού $f^{-1}(y) = y^3+e^y-e$:
\[ \int_{f(e-1)}^{f(1+e)} f^{-1}(y)\,\d y = \int_{f(e-1)}^{f(1+e)} (y^3+e^y-e)\,\d y \]

Αυτοί οι υπολογισμοί απαιτούν τις αριθμητικές τιμές. Η τεχνική «αντίστροφη + ολοκλήρωμα» σε θέματα Δ Πανελλαδικών συνήθως δίνει «ωραία» μετατροπή. Χωρίς πρόσθετα δεδομένα, η λύση αφήνεται σε παραμετρική μορφή.

\end{erwthma}
