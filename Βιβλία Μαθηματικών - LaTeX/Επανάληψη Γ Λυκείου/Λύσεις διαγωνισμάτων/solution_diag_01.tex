\begin{thema}{Α}
\begin{erwthma}
\item Έστω $x_1,x_2\in\varDelta$ με $x_1<x_2$. Η $f$ είναι συνεχής στο $[x_1,x_2]$ και παραγωγίσιμη στο $(x_1,x_2)$, αφού $x_1,x_2$ είναι σημεία του διαστήματος $\varDelta$.

Επομένως, από το Θεώρημα Μέσης Τιμής, υπάρχει $\xi\in(x_1,x_2)$ τέτοιο ώστε
\[
\frac{f(x_2)-f(x_1)}{x_2-x_1}=f'(\xi).
\]
Επειδή $f'(\xi)>0$ και $x_2-x_1>0$, έχουμε
\[
f(x_2)-f(x_1)=f'(\xi)(x_2-x_1)>0.
\]
Άρα
\[
f(x_2)>f(x_1).
\]
Επομένως η $f$ είναι \textbf{γνησίως αύξουσα} στο $\varDelta$.

\item Μια συνάρτηση $f:A\to\mathbb{R}$ λέγεται συνάρτηση $1-1$, όταν για οποιαδήποτε $x_1,x_2\in A$ ισχύει:
\[
x_1\neq x_2 \Rightarrow f(x_1)\neq f(x_2).
\]
Ισοδύναμα:
\[
f(x_1)=f(x_2)\Rightarrow x_1=x_2.
\]

\item Η ευθεία $y=l$ λέγεται οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$, όταν
\[
\lim_{x\to+\infty}f(x)=l.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Αν η $f$ είναι κυρτή, η εφαπτομένη της βρίσκεται κάτω από τη γραφική της παράσταση.
\item \textbf{Λάθος}. Ένα θετικό ολοκλήρωμα δεν σημαίνει ότι η συνάρτηση παίρνει μόνο θετικές τιμές.
\item \textbf{Λάθος}. Μια συνάρτηση $1-1$ δεν είναι υποχρεωτικά γνησίως μονότονη.
\item \textbf{Σωστό}. Η συνάρτηση $\hm^3x$ είναι περιττή, άρα
\[
\dint_{-a}^{a}\hm^3x\,\d x=0.
\]
\item \textbf{Λάθος}. Από τη σχέση $f(a)f(\beta)>0$ δεν αποκλείεται η ύπαρξη ρίζας στο $(a,\beta)$.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Η συνάρτηση έχει πεδίο ορισμού
\[
D_f=\mathbb{R}\setminus\{1\}.
\]
Επιπλέον
\[
f(x)=\frac{x^2-x+1}{x-1}=x+\frac{1}{x-1}.
\]
\begin{erwthma}
\item Για κάθε $x\in D_f$ είναι
\[
f'(x)=\left(x+\frac{1}{x-1}\right)'=1-\frac{1}{(x-1)^2}
=\frac{(x-1)^2-1}{(x-1)^2}
=\frac{x(x-2)}{(x-1)^2}.
\]
Βρίσκουμε ρίζες και πρόσημα της $f'$:
\begin{itemize}
\item $f'(x)=0\Leftrightarrow x(x-2)=0\Leftrightarrow x=0$ ή $x=2$.
\item Επειδή $(x-1)^2>0$ για κάθε $x\neq1$, το πρόσημο της $f'$ είναι το πρόσημο του $x(x-2)$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε τα πρόσημα της $f'$ και τη μονοτονία της $f$:
\begin{center}

\tkzTabLine{ , \nearrow,t,\searrow,d,\searrow,t,\nearrow,}
\end{tikzpicture}
\end{center}
Επομένως η $f$ είναι:
\begin{itemize}
\item \textbf{γνησίως αύξουσα} στο $(-\infty,0]$,
\item \textbf{γνησίως φθίνουσα} στα $[0,1)$ και $(1,2]$,
\item \textbf{γνησίως αύξουσα} στο $[2,+\infty)$.
\end{itemize}
Επειδή το πρόσημο της $f'$ αλλάζει από $+$ σε $-$ στο $x=0$, η $f$ παρουσιάζει τοπικό μέγιστο στο $x=0$, με τιμή
\[
f(0)=\frac{1}{-1}=-1.
\]
Επειδή το πρόσημο της $f'$ αλλάζει από $-$ σε $+$ στο $x=2$, η $f$ παρουσιάζει τοπικό ελάχιστο στο $x=2$, με τιμή
\[
f(2)=2+\frac{1}{1}=3.
\]

\item Για κάθε $x\in D_f$ είναι
\[
f''(x)=\left(1-\frac{1}{(x-1)^2}\right)'=\frac{2}{(x-1)^3}.
\]
Βρίσκουμε το πρόσημο της $f''$:
\begin{itemize}
\item $f''(x)>0\Leftrightarrow \frac{2}{(x-1)^3}>0\Leftrightarrow x>1$.
\item $f''(x)<0\Leftrightarrow \frac{2}{(x-1)^3}<0\Leftrightarrow x<1$.
\end{itemize}
Στον παρακάτω πίνακα βλέπουμε την κυρτότητα της $f$:
\begin{center}

{$-\infty$,$1$,$+\infty$}%

\tkzTabLine{ ,\curvearrowright,d,\rotatebox[origin= c]{180}{$\curvearrowleft$}, }
\end{tikzpicture}
\end{center}
Άρα η $f$ είναι κοίλη στο $(-\infty,1)$ και κυρτή στο $(1,+\infty)$.

Η $f''$ δεν μηδενίζεται σε σημείο του πεδίου ορισμού της και το $x=1$ δεν ανήκει στο $D_f$. Επομένως η $C_f$ δεν έχει σημεία καμπής.

\item Για την κατακόρυφη ασύμπτωτη έχουμε
\[
\lim_{x\to1^-}f(x)=\lim_{x\to1^-}\left(x+\frac{1}{x-1}\right)=-\infty
\]
και
\[
\lim_{x\to1^+}f(x)=\lim_{x\to1^+}\left(x+\frac{1}{x-1}\right)=+\infty.
\]
Άρα η ευθεία
\[
{x=1}
\]
είναι κατακόρυφη ασύμπτωτη της $C_f$.

Επιπλέον
\[
\lim_{x\to\pm\infty}\bigl(f(x)-x\bigr)
=\lim_{x\to\pm\infty}\frac{1}{x-1}=0.
\]
Άρα η ευθεία
\[
{y=x}
\]
είναι πλάγια ασύμπτωτη της $C_f$ στο $-\infty$ και στο $+\infty$.

Οριζόντια ασύμπτωτη δεν υπάρχει, αφού
\[
\lim_{x\to\pm\infty}f(x)=\pm\infty.
\]

Για το σύνολο τιμών, με βάση τον πίνακα μεταβολών:
\begin{itemize}
\item Στο $(-\infty,1)$ η μέγιστη τιμή είναι $f(0)=-1$, άρα
\[
f\bigl((-\infty,1)\bigr)=(-\infty,-1].
\]
\item Στο $(1,+\infty)$ η ελάχιστη τιμή είναι $f(2)=3$, άρα
\[
f\bigl((1,+\infty)\bigr)=[3,+\infty).
\]
\end{itemize}
Επομένως
\[
{f(D_f)=(-\infty,-1]\cup[3,+\infty)}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
\begin{erwthma}
\item Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ και
\[
f'(x)=e^x-\lambda.
\]
Αφού η $C_f$ παρουσιάζει ελάχιστο ίσο με $0$ στο $x_0$, σύμφωνα με το θεώρημα \eng{Fermat} ισχύει
\[
f'(x_0)=0\Rightarrow e^{x_0}-\lambda=0\Rightarrow \lambda=e^{x_0}.
\]
Επίσης
\[
f(x_0)=0\Rightarrow e^{x_0}-\lambda x_0=0.
\]
Με αντικατάσταση του $\lambda=e^{x_0}$ προκύπτει
\[
e^{x_0}-e^{x_0}x_0=0
\Leftrightarrow e^{x_0}(1-x_0)=0
\Leftrightarrow x_0=1.
\]
Άρα
\[
\lambda=e^{x_0}=e.
\]

\item Από το προηγούμενο ερώτημα έχουμε
\[
f(x)=e^x-ex
\quad\text{και}\quad
f'(x)=e^x-e.
\]
Έστω $M(\alpha,f(\alpha))$ το σημείο επαφής. Η εφαπτομένη στο $M$ έχει εξίσωση
\[
y-f(\alpha)=f'(\alpha)(x-\alpha).
\]
Επειδή διέρχεται από το $A(0,-e^2)$, έχουμε
\begin{align*}
-e^2-f(\alpha)&=f'(\alpha)(0-\alpha)\\
-e^2-\left(e^\alpha-e\alpha\right)&=-\alpha(e^\alpha-e)\\
-e^2&=e^\alpha(1-\alpha).
\end{align*}
Άρα
\[
e^\alpha(\alpha-1)=e^2.
\]
Η τιμή $\alpha=2$ επαληθεύει την εξίσωση, ενώ για $\alpha>1$ η συνάρτηση $e^\alpha(\alpha-1)$ είναι γνησίως αύξουσα. Επομένως $\alpha=2$.

Η ζητούμενη εφαπτομένη είναι
\begin{align*}
y-f(2)&=f'(2)(x-2)\\
y-\left(e^2-2e\right)&=(e^2-e)(x-2)\\
y&=(e^2-e)x-e^2.
\end{align*}
Άρα
\[
{y=(e^2-e)x-e^2}.
\]

\item Από το πρώτο ερώτημα η $f$ παρουσιάζει ελάχιστο ίσο με $0$ στο $x=1$, άρα
\[
f(x)\geq0
\]
για κάθε $x\in\mathbb{R}$. Το χωρίο που περικλείεται από την $C_f$, τον άξονα $x'x$ και τον άξονα $y'y$ βρίσκεται στο διάστημα $[0,1]$.

Επομένως το εμβαδόν είναι
\begin{align*}
E&=\int_0^1 f(x)\,\d x
=\int_0^1 (e^x-ex)\,\d x\\
&=\left[e^x-\frac{e x^2}{2}\right]_0^1
=\left(e-\frac{e}{2}\right)-1
=\frac{e}{2}-1.
\end{align*}
Άρα
\[
{E=\frac{e}{2}-1}.
\]

\item Έστω $x(t)$ η τετμημένη και $y(t)$ η τεταγμένη του $M$ τη χρονική στιγμή $t$. Τότε
\[
y(t)=f(x(t))=e^{x(t)}-ex(t).
\]
Το εμβαδόν του τριγώνου $OMK$ είναι
\[
E(t)=\frac{1}{2}x(t)y(t)=\frac{1}{2}x(t)f(x(t)).
\]
Παραγωγίζοντας ως προς $t$, έχουμε
\[
E'(t)=\frac{1}{2}x'(t)\left[f(x(t))+x(t)f'(x(t))\right].
\]
Τη χρονική στιγμή που $x=2$ είναι $x'(t)=2$, ενώ
\[
f(2)=e^2-2e
\quad\text{και}\quad
f'(2)=e^2-e.
\]
Άρα
\begin{align*}
E'(t)&=\frac{1}{2}\cdot2\left[f(2)+2f'(2)\right]\\
&=e^2-2e+2(e^2-e)\\
&=3e^2-4e.
\end{align*}
Επομένως ο ζητούμενος ρυθμός μεταβολής είναι
\[
{3e^2-4e\ \text{μον}^2/\si{sec}}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Θεωρούμε τη συνάρτηση
\[
g(t)=t^3+e^t,\quad t\in\mathbb{R}.
\]
Για κάθε $t\in\mathbb{R}$ είναι
\[
g'(t)=3t^2+e^t>0,
\]
άρα η $g$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.
\begin{erwthma}
\item Από τη δοσμένη σχέση έχουμε
\[
g(f(x))=x+e.
\]
Επειδή η $g$ είναι γνησίως αύξουσα, είναι συνάρτηση $1-1$.

Έστω $x_1,x_2\in\mathbb{R}$ με $f(x_1)=f(x_2)$. Τότε
\[
g(f(x_1))=g(f(x_2))
\Leftrightarrow x_1+e=x_2+e
\Leftrightarrow x_1=x_2.
\]
Άρα η $f$ είναι συνάρτηση $1-1$, οπότε αντιστρέφεται.

Αν $y=f(x)$, τότε
\[
y^3+e^y=x+e
\Leftrightarrow x=y^3+e^y-e.
\]
Αντίστροφα, για κάθε $y\in\mathbb{R}$, αν
\[
x=y^3+e^y-e,
\]
τότε
\[
g(f(x))=x+e=y^3+e^y=g(y).
\]
Επειδή η $g$ είναι $1-1$, παίρνουμε $f(x)=y$. Άρα $f(\mathbb{R})=\mathbb{R}$.

Άρα
\[
{f^{-1}(y)=y^3+e^y-e,\quad y\in\mathbb{R}}.
\]

\item Έστω $x_1,x_2\in\mathbb{R}$ με $x_1<x_2$. Τότε
\[
x_1+e<x_2+e
\Leftrightarrow g(f(x_1))<g(f(x_2)).
\]
Επειδή η $g$ είναι γνησίως αύξουσα, προκύπτει
\[
f(x_1)<f(x_2).
\]
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

Για το πρόσημο της $f$ βρίσκουμε πρώτα τη ρίζα της:
\[
f(x)=0
\Leftrightarrow 0^3+e^0=x+e
\Leftrightarrow 1=x+e
\Leftrightarrow x=1-e.
\]
Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε:
\[
\begin{array}{c|ccc}
x & (-\infty,1-e) & 1-e & (1-e,+\infty)\\ \hline
f(x) & - & 0 & +
\end{array}
\]

\item Επειδή η $f$ είναι γνησίως αύξουσα, έχουμε
\[
f(e^{x-1}+x^3-1)>1
\Leftrightarrow e^{x-1}+x^3-1>f^{-1}(1).
\]
Είναι
\[
f^{-1}(1)=1^3+e^1-e=1.
\]
Άρα η ανίσωση γίνεται
\[
e^{x-1}+x^3-1>1
\Leftrightarrow e^{x-1}+x^3-2>0.
\]
Θεωρούμε
\[
h(x)=e^{x-1}+x^3-2.
\]
Για κάθε $x\in\mathbb{R}$ είναι
\[
h'(x)=e^{x-1}+3x^2>0,
\]
άρα η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$. Επίσης
\[
h(1)=e^0+1-2=0.
\]
Επομένως
\[
h(x)>0\Leftrightarrow x>1.
\]
Άρα η λύση της ανίσωσης είναι
\[
{x\in(1,+\infty)}.
\]

\item Από το Δ2 ερώτημα, η $f$ είναι μη αρνητική στο διάστημα $[1-e,1]$. Επομένως το ζητούμενο εμβαδόν είναι
\[
E=\int_{1-e}^{1}f(x)\,\d x.
\]
Θέτουμε
\[
y=f(x)\Rightarrow x=f^{-1}(y)=y^3+e^y-e.
\]
Τότε
\[
\d x=(3y^2+e^y)\,\d y.
\]
Επιπλέον
\[
x=1-e\Rightarrow y=f(1-e)=0
\]
και
\[
x=1\Rightarrow y=f(1)=1,
\]
αφού
\[
f^{-1}(1)=1.
\]
Άρα
\begin{align*}
E&=\int_0^1 y(3y^2+e^y)\,\d y\\
&=\int_0^1 3y^3\,\d y+\int_0^1 y e^y\,\d y\\
&=\left[\frac{3y^4}{4}\right]_0^1+\left[e^y(y-1)\right]_0^1\\
&=\frac{3}{4}+\left(0-(-1)\right)\\
&=\frac{7}{4}.
\end{align*}
Επομένως
\[
{E=\frac{7}{4}}.
\]
\end{erwthma}
\end{thema}
