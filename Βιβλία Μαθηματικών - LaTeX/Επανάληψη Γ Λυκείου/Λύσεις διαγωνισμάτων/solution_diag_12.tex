\begin{thema}{Α}
\begin{erwthma}
\item Έστω $x_1,x_2\in\varDelta$ με
\[
x_1<x_2.
\]
Επειδή η $f$ είναι συνεχής στο διάστημα $\varDelta$ και παραγωγίσιμη σε κάθε εσωτερικό σημείο του, εφαρμόζουμε το Θεώρημα Μέσης Τιμής στο $[x_1,x_2]$.

Άρα υπάρχει $\xi\in(x_1,x_2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(x_2)-f(x_1)}{x_2-x_1}.
\]
Από την υπόθεση έχουμε
\[
f'(\xi)>0.
\]
Επειδή
\[
x_2-x_1>0,
\]
παίρνουμε
\[
f(x_2)-f(x_1)>0.
\]
Άρα
\[
f(x_1)<f(x_2).
\]
Επομένως η $f$ είναι γνησίως αύξουσα στο $\varDelta$.

\item Η γραφική παράσταση μιας συνάρτησης $f$ παρουσιάζει σημείο καμπής στο σημείο της με τετμημένη $x_0$, όταν στο σημείο αυτό η $C_f$ αλλάζει κυρτότητα, δηλαδή η $f$ μεταβαίνει από κυρτή σε κοίλη ή από κοίλη σε κυρτή.

Αν η $f''$ υπάρχει σε μια περιοχή του $x_0$, τότε συνήθως ελέγχουμε αν η $f''$ αλλάζει πρόσημο εκατέρωθεν του $x_0$.

\item Αν οι συναρτήσεις $u,v$ είναι παραγωγίσιμες και οι παράγωγοί τους είναι συνεχείς στο $[a,\beta]$, τότε ο τύπος της παραγοντικής ολοκλήρωσης για το ορισμένο ολοκλήρωμα είναι
\[
\dint_a^\beta u(x)v'(x)\,\d x
=
\left[u(x)v(x)\right]_a^\beta
-
\dint_a^\beta u'(x)v(x)\,\d x.
\]

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Λάθος}. Από το Θεώρημα Bolzano προκύπτει ότι υπάρχει τουλάχιστον μία ρίζα στο $(a,\beta)$, όχι υποχρεωτικά ακριβώς μία.
\item \textbf{Σωστό}. Από το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού,
\[
\dint_a^\beta f'(x)\,\d x=f(\beta)-f(a),
\]
εφόσον η $f'$ είναι συνεχής στο $[a,\beta]$.
\item \textbf{Λάθος}. Αν η $f$ είναι κοίλη, τότε η γραφική της παράσταση βρίσκεται κάτω από κάθε εφαπτομένη της, όχι πάνω από αυτήν.
\item \textbf{Σωστό}. Κάθε πολυωνυμική συνάρτηση 3ου βαθμού έχει όρια με αντίθετη συμπεριφορά στα $-\infty$ και $+\infty$, οπότε από το Θεώρημα Bolzano έχει τουλάχιστον μία πραγματική ρίζα.
\item \textbf{Λάθος}. Η συνθήκη $f''(x_0)=0$ δεν αρκεί για σημείο καμπής. Χρειάζεται αλλαγή κυρτότητας.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Η $C_f$ αποτελείται από το άνω ημικύκλιο με κέντρο την αρχή και ακτίνα $2$ στο $[-2,2]$ και από το ευθύγραμμο τμήμα από το $(2,0)$ έως το $(5,3)$.
\begin{erwthma}
\item Στο ημικύκλιο οι τιμές της $f$ είναι από $0$ έως $2$, δηλαδή
\[
[0,2].
\]
Στο ευθύγραμμο τμήμα από το $(2,0)$ έως το $(5,3)$ οι τιμές της $f$ είναι από $0$ έως $3$, δηλαδή
\[
[0,3].
\]
Άρα το σύνολο τιμών της $f$ είναι
\[
{[0,3]}.
\]

\item Το ολοκλήρωμα
\[
\dint_{-2}^{2}f(x)\,\d x
\]
ισούται με το εμβαδόν ημικυκλίου ακτίνας $\rho=2$. Άρα
\[
\dint_{-2}^{2}f(x)\,\d x
=\frac12\pi\rho^2
=\frac12\pi\cdot4
=2\pi.
\]

Το ολοκλήρωμα
\[
\dint_{2}^{5}f(x)\,\d x
\]
ισούται με το εμβαδόν ορθογωνίου τριγώνου με βάση
\[
5-2=3
\]
και ύψος $3$. Άρα
\[
\dint_{2}^{5}f(x)\,\d x
=\frac12\cdot3\cdot3
=\frac92.
\]

\item Στο διάστημα $[-2,2]$ η συνάρτηση έχει τύπο
\[
f(x)=\sqrt{4-x^2}.
\]
Άρα, για $-2<x<2$,
\[
f'(x)=\frac{-x}{\sqrt{4-x^2}}.
\]
Επομένως
\[
\lim_{x\to2^-}f'(x)
=\lim_{x\to2^-}\frac{-x}{\sqrt{4-x^2}}
=-\infty.
\]

Στο διάστημα $[2,5]$ η γραφική παράσταση είναι ευθύγραμμο τμήμα με κλίση
\[
\frac{3-0}{5-2}=1,
\]
άρα
\[
\lim_{x\to2^+}f'(x)=1.
\]
Επειδή η αριστερή παράγωγος δεν είναι πεπερασμένος αριθμός και δεν ισούται με τη δεξιά, η $f$ δεν είναι παραγωγίσιμη στο $x=2$.

\item Το σημείο της $C_f$ με τετμημένη $x_0=0$ είναι
\[
(0,2).
\]
Στο ημικύκλιο έχουμε
\[
f(x)=\sqrt{4-x^2},
\]
οπότε
\[
f'(x)=\frac{-x}{\sqrt{4-x^2}}
\]
και
\[
f'(0)=0.
\]
Η εφαπτομένη στο $(0,2)$ έχει εξίσωση
\[
y-2=0(x-0),
\]
δηλαδή
\[
{y=2}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνάρτηση
\[
f(x)=\ln(x^2+1)-x+1,\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=\frac{2x}{x^2+1}-1
=\frac{2x-x^2-1}{x^2+1}
=-\frac{(x-1)^2}{x^2+1}.
\]
Επειδή
\[
x^2+1>0
\]
και
\[
(x-1)^2\geq0,
\]
έχουμε
\[
f'(x)\leq0
\]
για κάθε $x\in\mathbb{R}$, με ισότητα μόνο για $x=1$.

Άρα η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$ και δεν παρουσιάζει τοπικά ακρότατα.

\item Επειδή η $f$ είναι γνησίως φθίνουσα, για κάθε $x\in[0,1]$ ισχύει
\[
f(x)\geq f(1).
\]
Όμως
\[
f(1)=\ln(1^2+1)-1+1=\ln2>0.
\]
Άρα
\[
f(x)\geq\ln2>0
\]
για κάθε $x\in[0,1]$. Επομένως
\[
{f(x)>0,\quad x\in[0,1]}.
\]

\item Η εξίσωση
\[
\ln(x^2+1)=x-1
\]
γράφεται ισοδύναμα
\[
f(x)=0.
\]
Έχουμε
\[
f(1)=\ln2>0.
\]
Επίσης
\[
\lim_{x\to+\infty}\bigl(\ln(x^2+1)-x+1\bigr)=-\infty,
\]
αφού ο γραμμικός όρος $x$ υπερισχύει του λογαριθμικού όρου.

Άρα υπάρχει $b>1$ τέτοιο ώστε
\[
f(b)<0.
\]
Επειδή η $f$ είναι συνεχής στο $[1,b]$, από το Θεώρημα Bolzano υπάρχει τουλάχιστον μία ρίζα στο $(1,b)$.

Επιπλέον, επειδή η $f$ είναι γνησίως φθίνουσα στο $\mathbb{R}$, η ρίζα αυτή είναι μοναδική.

Άρα η εξίσωση έχει
\[
{\text{ακριβώς μία ρίζα στο }(1,+\infty)}.
\]

\item Το σημείο κινείται πάνω στην $C_f$, άρα
\[
y(t)=f(x(t)).
\]
Παραγωγίζοντας ως προς $t$, παίρνουμε
\[
y'(t)=f'(x(t))x'(t).
\]
Δίνεται
\[
x'(t)=1.
\]
Όταν $x=1$, έχουμε
\[
f'(1)=-\frac{(1-1)^2}{1^2+1}=0.
\]
Άρα
\[
y'(t)=f'(1)\cdot1=0.
\]
Επομένως ο ρυθμός μεταβολής της τεταγμένης εκείνη τη στιγμή είναι
\[
{0\ \text{μονάδες}/\sec}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται η συνάρτηση
\[
f(x)=e^x-x-1,\quad x\in\mathbb{R}.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=e^x-1.
\]
Έχουμε
\[
f'(x)=0\Leftrightarrow e^x=1\Leftrightarrow x=0.
\]
Επίσης
\[
f'(x)<0\quad\text{για }x<0
\]
και
\[
f'(x)>0\quad\text{για }x>0.
\]
Άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty,0]$ και γνησίως αύξουσα στο $[0,+\infty)$.

Επομένως παρουσιάζει ολικό ελάχιστο στο $x=0$, με τιμή
\[
f(0)=e^0-0-1=0.
\]
Άρα, για κάθε $x\in\mathbb{R}$,
\[
f(x)\geq0.
\]
Δηλαδή
\[
e^x-x-1\geq0
\]
και επομένως
\[
{e^x\geq x+1,\quad x\in\mathbb{R}}.
\]

\item Για $x\neq0$ έχουμε
\[
g(x)=\frac{e^x-1}{x}.
\]
Εξετάζουμε το όριο στο $0$:
\[
\lim_{x\to0}\frac{e^x-1}{x}
=1,
\]
αφού είναι η παράγωγος της $e^x$ στο $0$.

Επειδή
\[
g(0)=1,
\]
παίρνουμε
\[
\lim_{x\to0}g(x)=g(0).
\]
Άρα η $g$ είναι συνεχής στο $x_0=0$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής στη συνάρτηση $f$ στο διάστημα $[0,1]$.

Η $f$ είναι συνεχής στο $[0,1]$ και παραγωγίσιμη στο $(0,1)$, άρα υπάρχει $\xi\in(0,1)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(1)-f(0)}{1-0}.
\]
Έχουμε
\[
f(1)=e-2
\quad\text{και}\quad
f(0)=0.
\]
Επομένως
\[
f'(\xi)=e-2.
\]
Όμως
\[
f'(\xi)=e^\xi-1.
\]
Άρα
\[
e^\xi-1=e-2
\]
και τελικά
\[
{e^\xi=e-1}.
\]

\item Πρώτα υπολογίζουμε
\[
\dint_0^1 f(t)\,\d t
=\dint_0^1(e^t-t-1)\,\d t.
\]
Έχουμε
\begin{align*}
\dint_0^1(e^t-t-1)\,\d t
&=\left[e^t-\frac{t^2}{2}-t\right]_0^1\\
&=\left(e-\frac12-1\right)-1\\
&=e-\frac52.
\end{align*}
Η εξίσωση γίνεται
\[
2f(x)=e-\frac52.
\]
Θεωρούμε τη συνάρτηση
\[
H(x)=2f(x)-\left(e-\frac52\right),\quad x\in[0,1].
\]
Η $H$ είναι συνεχής στο $[0,1]$ και
\[
H'(x)=2f'(x)=2(e^x-1)>0
\]
για κάθε $x\in(0,1)$. Άρα η $H$ είναι γνησίως αύξουσα στο $[0,1]$.

Επιπλέον,
\[
H(0)=2f(0)-\left(e-\frac52\right)
=-\left(e-\frac52\right)<0,
\]
αφού $e>\frac52$, ενώ
\[
H(1)=2(e-2)-\left(e-\frac52\right)
=e-\frac32>0.
\]
Από το Θεώρημα Bolzano υπάρχει τουλάχιστον μία ρίζα στο $(0,1)$. Επειδή η $H$ είναι γνησίως αύξουσα, η ρίζα αυτή είναι μοναδική.

Άρα η εξίσωση
\[
2f(x)=\dint_0^1 f(t)\,\d t
\]
έχει
\[
{\text{ακριβώς μία ρίζα στο }(0,1)}.
\]
\end{erwthma}
\end{thema}
