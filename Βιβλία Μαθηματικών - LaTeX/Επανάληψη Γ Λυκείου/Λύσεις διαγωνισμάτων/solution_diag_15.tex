\begin{thema}{Α}
\begin{erwthma}
\item Έστω $x_1,x_2\in\varDelta$ με
\[
x_1<x_2.
\]
Η $f$ είναι παραγωγίσιμη στο $\varDelta$, άρα είναι συνεχής στο $[x_1,x_2]$ και παραγωγίσιμη στο $(x_1,x_2)$.

Από το Θεώρημα Μέσης Τιμής υπάρχει $\xi\in(x_1,x_2)$ τέτοιο ώστε
\[
f'(\xi)=\frac{f(x_2)-f(x_1)}{x_2-x_1}.
\]
Από την υπόθεση έχουμε
\[
f'(\xi)=0.
\]
Άρα
\[
\frac{f(x_2)-f(x_1)}{x_2-x_1}=0.
\]
Επειδή $x_2-x_1\neq0$, παίρνουμε
\[
f(x_2)=f(x_1).
\]
Επομένως η $f$ είναι σταθερή στο $\varDelta$.

\item Μια συνάρτηση $f$ λέγεται κυρτή, ή λέμε ότι στρέφει τα κοίλα προς τα άνω σε ένα διάστημα $\varDelta$, όταν για κάθε $x_1,x_2\in\varDelta$ και κάθε $\lambda\in[0,1]$ ισχύει
\[
f\bigl(\lambda x_1+(1-\lambda)x_2\bigr)
\leq
\lambda f(x_1)+(1-\lambda)f(x_2).
\]
Γεωμετρικά, η γραφική της παράσταση βρίσκεται κάτω από κάθε χορδή της.

\item Μια συνάρτηση $F$ λέγεται αρχική συνάρτηση ή παράγουσα της $f$ στο διάστημα $\varDelta$, όταν είναι παραγωγίσιμη στο $\varDelta$ και ισχύει
\[
F'(x)=f(x)
\]
για κάθε $x\in\varDelta$.

\item Οι χαρακτηρισμοί είναι:
\begin{alist}
\item \textbf{Σωστό}. Κάθε γνησίως μονότονη συνάρτηση σε ένα διάστημα είναι $1-1$.
\item \textbf{Λάθος}. Αν τα άκρα $a,\beta$ είναι σταθερά, τότε το
\[
\dint_a^\beta f(x)\,\d x
\]
είναι σταθερός αριθμός, οπότε η παράγωγός του είναι $0$, όχι $f(x)$.
\item \textbf{Σωστό}. Αν $f$ είναι συνεχής, $f(x)\geq0$ στο $[a,\beta]$ και το ολοκλήρωμά της είναι $0$, τότε δεν μπορεί να είναι θετική σε κανένα σημείο. Άρα
\[
f(x)=0
\]
για κάθε $x\in[a,\beta]$.
\item \textbf{Σωστό}. Αν $f$ και $g$ είναι $1-1$ και ορίζεται η σύνθεση $f\circ g$, τότε και η $f\circ g$ είναι $1-1$.
\item \textbf{Σωστό}. Από το Θεώρημα Μέγιστης-Ελάχιστης Τιμής, κάθε συνεχής συνάρτηση σε κλειστό διάστημα $[a,\beta]$ παίρνει μέγιστη και ελάχιστη τιμή.
\end{alist}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνονται οι συναρτήσεις
\[
f(x)=e^x-1
\]
και
\[
g(x)=\ln\left(\frac{x+1}{x-1}\right).
\]
\begin{erwthma}
\item Η $f$ έχει πεδίο ορισμού
\[
D_f=\mathbb{R}.
\]

Για την $g$ πρέπει
\[
\frac{x+1}{x-1}>0.
\]
Με πίνακα προσήμων ή απευθείας έλεγχο παίρνουμε
\[
x<-1
\quad\text{ή}\quad
x>1.
\]
Άρα
\[
D_g=(-\infty,-1)\cup(1,+\infty).
\]

\item Για $x\in D_g$ έχουμε
\begin{align*}
h(x)
&=(f\circ g)(x)\\
&=f(g(x))\\
&=e^{g(x)}-1\\
&=e^{\ln\left(\frac{x+1}{x-1}\right)}-1\\
&=\frac{x+1}{x-1}-1\\
&=\frac{2}{x-1}.
\end{align*}
Επομένως
\[
{h(x)=\frac{2}{x-1},\quad x\in(-\infty,-1)\cup(1,+\infty)}.
\]

\item Για $x\in(1,+\infty)$ έχουμε
\[
\phi(x)=\frac{2}{x-1}.
\]
Η $\phi$ είναι παραγωγίσιμη και
\[
\phi'(x)=-\frac{2}{(x-1)^2}<0
\]
για κάθε $x>1$. Άρα η $\phi$ είναι γνησίως φθίνουσα στο $(1,+\infty)$, επομένως είναι $1-1$ και αντιστρέφεται.

Για την αντίστροφη, θέτουμε
\[
y=\frac{2}{x-1}.
\]
Τότε
\[
y(x-1)=2
\]
και
\[
x=1+\frac{2}{y}.
\]
Επειδή
\[
\phi((1,+\infty))=(0,+\infty),
\]
η αντίστροφη είναι
\[
{\phi^{-1}(x)=1+\frac{2}{x},\quad x>0}.
\]

\item Υπολογίζουμε
\[
\dint_2^3 \phi(x)\,\d x
=\dint_2^3 \frac{2}{x-1}\,\d x.
\]
Άρα
\[
\dint_2^3 \phi(x)\,\d x
=2\left[\ln(x-1)\right]_2^3
=2\ln2.
\]
Επομένως
\[
{\dint_2^3 \phi(x)\,\d x=2\ln2}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνάρτηση
\[
f(x)=x\ln x-x+1,\quad x>0.
\]
\begin{erwthma}
\item Παραγωγίζουμε:
\[
f'(x)=\ln x+x\cdot\frac1x-1=\ln x.
\]
Έχουμε
\[
f'(x)=0\Leftrightarrow \ln x=0\Leftrightarrow x=1.
\]
Επίσης
\[
f'(x)<0\quad\text{για }0<x<1
\]
και
\[
f'(x)>0\quad\text{για }x>1.
\]
Άρα η $f$ είναι γνησίως φθίνουσα στο $(0,1]$ και γνησίως αύξουσα στο $[1,+\infty)$.

Επομένως παρουσιάζει ολικό ελάχιστο στο $x=1$, με τιμή
\[
f(1)=1\cdot\ln1-1+1=0.
\]
Άρα
\[
{f(x)\geq0,\quad x>0}.
\]

\item Για $x_0=e$ έχουμε
\[
f(e)=e\ln e-e+1=1
\]
και
\[
f'(e)=\ln e=1.
\]
Άρα η εφαπτομένη της $C_f$ στο σημείο με τετμημένη $e$ είναι
\[
y-1=1(x-e),
\]
δηλαδή
\[
{y=x-e+1}.
\]

\item Η εξίσωση
\[
x\ln x=x-1
\]
γράφεται
\[
x\ln x-x+1=0
\]
δηλαδή
\[
f(x)=0.
\]
Από το πρώτο ερώτημα έχουμε
\[
f(x)\geq0
\]
για κάθε $x>0$, και η ελάχιστη τιμή $0$ λαμβάνεται μόνο στο $x=1$.

Άρα η εξίσωση έχει ακριβώς μία ρίζα στο $(0,+\infty)$, την
\[
{x=1}.
\]

\item Αν $\theta$ είναι η γωνία που σχηματίζει η εφαπτομένη με τον άξονα $x'x$, τότε
\[
\tan\theta(t)=f'(x(t)).
\]
Παραγωγίζοντας ως προς $t$, παίρνουμε
\[
\frac{1}{\cos^2\theta(t)}\theta'(t)=f''(x(t))x'(t).
\]
Επειδή
\[
\frac{1}{\cos^2\theta}=1+\tan^2\theta,
\]
έχουμε
\[
\bigl(1+(f'(x(t)))^2\bigr)\theta'(t)=f''(x(t))x'(t).
\]

Τη στιγμή που $x=e$, έχουμε
\[
f'(e)=1,\quad f''(x)=\frac1x,\quad f''(e)=\frac1e,
\]
και δίνεται
\[
x'(t)=1.
\]
Άρα
\[
(1+1^2)\theta'(t)=\frac1e\cdot1,
\]
οπότε
\[
\theta'(t)=\frac{1}{2e}.
\]
Επομένως
\[
{\theta'(t)=\frac{1}{2e}\ \text{rad}/\sec}.
\]
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται παραγωγίσιμη συνάρτηση
\[
f:\mathbb{R}\to\mathbb{R}
\]
με
\[
f(0)=0
\]
και
\[
f'(x)+f(x)>0
\]
για κάθε $x\in\mathbb{R}$.
\begin{erwthma}
\item Θεωρούμε
\[
h(x)=e^x f(x).
\]
Τότε
\[
h'(x)=e^x f(x)+e^x f'(x)
=e^x\bigl(f'(x)+f(x)\bigr).
\]
Επειδή
\[
e^x>0
\]
και
\[
f'(x)+f(x)>0,
\]
έχουμε
\[
{h'(x)>0}
\]
για κάθε $x\in\mathbb{R}$.

Άρα η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$.

\item Έχουμε
\[
h(0)=e^0f(0)=0.
\]
Επειδή η $h$ είναι γνησίως αύξουσα:
\begin{itemize}
\item αν $x>0$, τότε $h(x)>h(0)=0$,
\item αν $x<0$, τότε $h(x)<h(0)=0$.
\end{itemize}
Όμως
\[
h(x)=e^x f(x)
\]
και
\[
e^x>0.
\]
Άρα το $h(x)$ έχει το ίδιο πρόσημο με το $f(x)$.

Επομένως
\[
{f(x)>0,\quad x>0}
\]
και
\[
{f(x)<0,\quad x<0}.
\]

\item Γνωρίζουμε ότι
\[
f(0)=0,
\]
άρα το $x=0$ είναι ρίζα της εξίσωσης $f(x)=0$.

Από το προηγούμενο ερώτημα, για $x>0$ είναι $f(x)>0$, ενώ για $x<0$ είναι $f(x)<0$. Άρα δεν υπάρχει άλλη ρίζα.

Συνεπώς η εξίσωση $f(x)=0$ έχει μοναδική ρίζα στο $\mathbb{R}$, την
\[
{x=0}.
\]

\item Δίνεται επιπλέον ότι
\[
f(1)=\frac1e.
\]
Τότε
\[
h(1)=e\cdot f(1)=1.
\]
Επειδή η $h$ είναι γνησίως αύξουσα και $h(0)=0$, για κάθε $x\in(0,1)$ ισχύει
\[
0<h(x)<h(1)=1.
\]
Άρα
\[
0<e^x f(x)<1.
\]
Επειδή $e^x>0$, παίρνουμε
\[
0<f(x)<e^{-x}
\]
για κάθε $x\in(0,1)$.

Ολοκληρώνοντας στο $[0,1]$, έχουμε
\[
0<\dint_0^1 f(x)\,\d x<\dint_0^1 e^{-x}\,\d x.
\]
Όμως
\[
\dint_0^1 e^{-x}\,\d x
=\left[-e^{-x}\right]_0^1
=1-\frac1e.
\]
Άρα
\[
{0<\dint_0^1 f(x)\,\d x<1-\frac1e}.
\]
\end{erwthma}
\end{thema}
