\section{Όρια - Συνέχεια}

\subsection{Ισότητα συναρτήσεων}
\begin{askhsh}{Ισότητα συναρτήσεων}
Για να εξετάσουμε αν δύο συναρτήσεις είναι ίσες εργαζόμαστε ως εξής:

\bhmata
\begin{bhma}
\item Βρίσκουμε τα πεδία ορισμού $D_f$ και $D_g$.
\item Ελέγχουμε αν $D_f = D_g$. 
\begin{itemize}
    \item Αν $D_f \neq D_g$, τότε οι συναρτήσεις \textbf{δεν είναι ίσες} ($f \neq g$).
    \item Αν $D_f = D_g$, τότε ελέγχουμε αν για κάθε $x$ του κοινού πεδίου ορισμού ισχύει $f(x) = g(x)$ (συνήθως απλοποιώντας τον έναν ή και τους δύο τύπους). Αν ισχύει, τότε \textbf{$f=g$}.
\end{itemize}
\end{bhma}
\textbf{Υποσύνολο στο οποίο είναι ίσες:} \\
Αν $D_f \neq D_g$ (άρα $f \neq g$), αλλά οι τύποι τους απλοποιούνται στην ίδια μορφή, τότε οι συναρτήσεις είναι ίσες στο σύνολο $D_f \cap D_g$.
\end{askhsh}

\Paradeigma{Ισότητα Συναρτήσεων}
\bmath{Να εξετάσετε αν οι παρακάτω συναρτήσεις είναι ίσες. Αν δεν είναι, να βρείτε το ευρύτερο υποσύνολο του $\mathbb{R}$ στο οποίο είναι ίσες.
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](1)
\task $f(x) = \ln(x^2 - 1)$ \quad και \quad $g(x) = \ln(x-1) + \ln(x+1)$
\task $f(x) = \dfrac{x^2+x}{x^2+2x+1}$ \quad και \quad $g(x) = \dfrac{x}{x+1}$
\end{tasks}}

\lysh
\begin{alist}
\item Για το πεδίο ορισμού της $f$ πρέπει:
\[ x^2 - 1 > 0 \Leftrightarrow x \in (-\infty, -1) \cup (1, +\infty) \]
Άρα $D_f = (-\infty, -1) \cup (1, +\infty)$.\\
Για το πεδίο ορισμού της $g$ πρέπει να συναληθεύουν οι περιορισμοί:
\[ x - 1 > 0 \Leftrightarrow x > 1 \quad \text{και} \quad x + 1 > 0 \Leftrightarrow x > -1 \]
Η συναλήθευση δίνει $x > 1$. Άρα $D_g = (1, +\infty)$. Επειδή $D_f \neq D_g$, οι συναρτήσεις \textbf{δεν είναι ίσες} ($f \neq g$). Για να βρούμε το υποσύνολο στο οποίο είναι ίσες, βρίσκουμε την τομή των πεδίων ορισμού:
\[ D_f \cap D_g = (1, +\infty) \]
Για κάθε $x \in (1, +\infty)$ ισχύει:
\[ g(x) = \ln(x-1) + \ln(x+1) = \ln\left((x-1)(x+1)\right) = \ln(x^2 - 1) = f(x) \]
Επομένως, οι συναρτήσεις $f$ και $g$ είναι ίσες στο διάστημα $(1, +\infty)$.

\item Για το πεδίο ορισμού της $f$ πρέπει ο παρονομαστής να μην μηδενίζεται:
\[ x^2+2x+1 \neq 0 \Leftrightarrow (x+1)^2 \neq 0 \Leftrightarrow x \neq -1 \]
Άρα $D_f = \mathbb{R} - \{-1\}$. Για το πεδίο ορισμού της $g$ πρέπει:
\[ x+1 \neq 0 \Leftrightarrow x \neq -1 \]
Άρα $D_g = \mathbb{R} - \{-1\}$. Παρατηρούμε ότι τα πεδία ορισμού ταυτίζονται, δηλαδή $D_f = D_g$. Τώρα θα ελέγξουμε αν οι τύποι τους είναι ίδιοι. Για κάθε $x \in \mathbb{R} - \{-1\}$ έχουμε:
\[ f(x) = \dfrac{x^2+x}{x^2+2x+1} = \dfrac{x(x+1)}{(x+1)^2} = \dfrac{x}{x+1} = g(x) \]
Αφού $D_f = D_g$ και $f(x) = g(x)$ για κάθε $x \in D_f$, τότε οι συναρτήσεις $f$ και $g$ είναι ίσες.
\end{alist}

\subsection{Πράξεις συναρτήσεων}
\begin{askhsh}{Πράξεις συναρτήσεων}
Για να ορίσουμε το άθροισμα $f+g$, τη διαφορά $f-g$, το γινόμενο $f\cdot g$ ή το πηλίκο $\dfrac{f}{g}$ δύο συναρτήσεων $f$ και $g$, βρίσκουμε πρώτα τα πεδία ορισμού τους $D_f$ και $D_g$.\\
\bhmata
\begin{bhma}
\item Το πεδίο ορισμού του αθροίσματος, της διαφοράς και του γινομένου είναι η τομή των πεδίων ορισμού τους:
\[ D_{f+g}=D_{f-g}=D_{f\cdot g} = D_f \cap D_g \]
\item Το πεδίο ορισμού του πηλίκου είναι η τομή των πεδίων ορισμού τους, εξαιρώντας τα $ x $ που μηδενίζουν τον παρονομαστή:
\[ D_{\frac{f}{g}} = \{x \in D_f \cap D_g \ \ \text{και}\ \ g(x) \neq 0\} \]
\end{bhma}
\end{askhsh}

\Paradeigma{Πράξεις συναρτήσεων}
\bmath{Δίνονται οι συναρτήσεις $f(x)=\sqrt{x-2}$ και $g(x)=\sqrt{x-5}$. Να ορίσετε τις συναρτήσεις
\begin{multicols}{2}
\begin{alist}
\item $f+g$
\item $\dfrac{f}{g}$
\end{alist}
\end{multicols}}
\lysh
Η συνάρτηση $f$ ορίζεται όταν $x-2\geq 0\Rightarrow x\geq 2$, άρα $D_f=[2,+\infty)$. Η συνάρτηση $g$ ορίζεται όταν $x-5\geq 0\Rightarrow x\geq 5$, οπότε $D_g=[5,+\infty)$. Η τομή των πεδίων ορισμού είναι $D_f \cap D_g = [2,+\infty) \cap [5,+\infty) = [5,+\infty)$.

\begin{alist}
\item Η συνάρτηση $f+g$ έχει πεδίο ορισμού $D_{f+g}=D_f \cap D_g = [5,+\infty)$ και τύπο:
\[(f+g)(x)=f(x)+g(x)=\sqrt{x-2}+\sqrt{x-5}\]

\item Η συνάρτηση $\dfrac{f}{g}$ έχει πεδίο ορισμού
\[ D_{\frac{f}{g}} = \{x\in D_f\cap D_g \ \ \text{και}\ \ g(x)\neq 0\} \]
Όμως $g(x)=0 \Leftrightarrow \sqrt{x-5}=0 \Leftrightarrow x-5=0 \Leftrightarrow x=5$. Επομένως εξαιρούμε το $5$ από την τομή, άρα $D_{\frac{f}{g}} = (5,+\infty)$. Ο τύπος της είναι:
\[\left(\dfrac{f}{g}\right)(x)=\dfrac{f(x)}{g(x)}=\dfrac{\sqrt{x-2}}{\sqrt{x-5}}=\sqrt{\dfrac{x-2}{x-5}}\]
\end{alist}

\subsection{Σύνθεση συναρτήσεων}
\begin{askhsh}{Εύρεση σύνθεσης}
Για να οριστεί η συνάρτηση $ f\circ g $ πρέπει να βρούμε το πεδίο ορισμού της και τον τύπο της.\\
\bhmata
\begin{bhma}
\item Ο τύπος της $ f\circ g $ θα ισούται με
\[ (f\circ g)(x)=f(g(x)) \]
που σημαίνει ότι στον τύπο της $ f $ αντικαθιστούμε το $ x $ με $ g(x) $.
\item Για το πεδίο ορισμού ισχύουν οι σχέσεις
\[ x\in D_g\ \ \textrm{ και }\ \ g(x)\in D_f \]
Οι περιορισμοί αυτοί μας οδηγούν σε εξισώσεις και ανισώσεις. Οι κοινές λύσεις σχηματίζουν το πεδίο ορισμού.
\end{bhma}
Εντελώς ανάλογα εργαζόμαστε για τις συναρτήσεις $ g\circ f, f\circ f\ldots $
\end{askhsh}

\item $f\circ g$
\item $g\circ f$
\item $f\circ f$
\end{alist}
\end{multicols}}
\lysh

\subsection{Συνάρτηση \bmath{$ 1-1 $} - Αντίστροφη}
\begin{askhsh}{Συνάρτηση $ 1-1 $}
\tropoi\begin{tropos}
\item Αποδεικνύουμε ότι η $ f $ είναι γνησίως μονότονη στο $ D_f $. (Ο τρόπος αυτός ενδείκνυται όταν το πεδίο ορισμού της $ f $ είναι ένα διάστημα.)
\item Αν το $D_f$ είναι ένωση διαστημάτων, αποδεικνύουμε ότι η $f$ είναι γνησίως μονότονη σε κάθε διάστημα, βρίσκουμε τις εικόνες των διαστημάτων και δείχνουμε ότι είναι ξένες μεταξύ τους.
\item Με τη βοήθεια του ορισμού της $ 1-1 $ συνάρτησης
\[ \textrm{Για κάθε }x_1,x_2\in D_f \ \ : \ \  f(x_1)=f(x_2)\Rightarrow x_1=x_2 \]
(Ο τρόπος αυτός ενδείκνυται για συναρτήσεις με πεδίο ορισμού ένωση διαστημάτων, αρκεί ο τύπος να επιτρέπει την επίλυση της εξίσωσης.)
\item Με τη βοήθεια της γραφικής παράστασης της $ f $. Κάθε οριζόντια ευθεία πρέπει να τέμνει τη $ C_f $ σε ένα το πολύ σημείο.
\item Αν η εξίσωση $ y=f(x) $ έχει μοναδική λύση ως προς $ x $ για κάθε $ y\in f(D_f) $ και η λύση ανήκει στο $ D_f $ τότε η $ f $ είναι $ 1-1 $.
\item Με απαγωγή σε άτοπο. Υποθέτουμε δηλαδή ότι η $ f $ δεν είναι $ 1-1 $.
\end{tropos}
\end{askhsh}
\Paradeigma{Απόδειξη $1-1$ με Μονοτονία (Παράγωγος)}
\bmath{Να δείξετε ότι η συνάρτηση $f(x) = x^3 + 3x - 2$ είναι $1-1$.}\\\\
\lysh
Η συνάρτηση $f$ είναι πολυωνυμική με πεδίο ορισμού $D_f = \mathbb{R}$. Η $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο:
\[ f'(x) = (x^3 + 3x - 2)' = 3x^2 + 3 \]
Παρατηρούμε ότι για κάθε $x \in \mathbb{R}$ ισχύει $3x^2 \ge 0$, άρα:
\[ f'(x) = 3x^2 + 3 \ge 3 > 0 \]
Επειδή $f'(x) > 0$ για κάθε $x \in \mathbb{R}$, η συνάρτηση $f$ είναι {γνησίως αύξουσα} στο $\mathbb{R}$. Συνεπώς είναι και {1-1}.\\\\
\Paradeigma{Απόδειξη $1-1$ με τον Ορισμό}
\bmath{Να δείξετε ότι η συνάρτηση $f(x) = \dfrac{2x-1}{x+3}$ είναι $1-1$.}

\lysh
Η συνάρτηση ορίζεται για $x+3 \neq 0 \Leftrightarrow x \neq -3$. Άρα $D_f = \mathbb{R} - \{-3\}$.
\\
Έστω $x_1, x_2 \in D_f$ με $f(x_1) = f(x_2)$. Έχουμε:
\begin{align*}
\dfrac{2x_1-1}{x_1+3} &= \dfrac{2x_2-1}{x_2+3} \\
(2x_1-1)(x_2+3) &= (2x_2-1)(x_1+3) \\
2x_1x_2 + 6x_1 - x_2 - 3 &= 2x_2x_1 + 6x_2 - x_1 - 3\\
6x_1 - x_2 &= 6x_2 - x_1 \\
6x_1 + x_1 &= 6x_2 + x_2 \\
7x_1 &= 7x_2 \\
x_1 &= x_2
\end{align*}
Αφού $f(x_1) = f(x_2) \Rightarrow x_1 = x_2$, η συνάρτηση $f$ είναι {$1-1$}.

\begin{askhsh}{Εύρεση αντίστροφης συνάρτησης}
\bhmata\begin{bhma}
\item Δείχνουμε ότι η $f$ είναι $1-1$.
\item Εύρεση συνόλου τιμών της $ f $ με τη βοήθεια μονοτονίας.
\item Επίλυση της εξίσωσης $ y=f(x) $ ως προς $ x $ με $ x\in D_f $.
\end{bhma}
\end{askhsh}
\Paradeigma{Εύρεση αντίστροφης}
\bmath{Δίνεται η συνάρτηση $f(x)=\ln{(x-2)-\ln{(5-x)}}$. Να δείξετε ότι η $f$ είναι αντιστρέψιμη και να ορίσετε τη συνάρτηση $f^{-1}$.}\\\\
\lysh
Η $f$ ορίζεται όταν
\[
x-2>0\Rightarrow x>2\ \text{και}\ 5-x>0\Rightarrow x<5
\]
άρα $D_f=(2,5)$. Για κάθε $x\in(2,5)$ είναι:
\[ f'(x)=\left[\ln{(x-2)-\ln{(5-x)}}\right]'=\dfrac{(x-2)'}{x-2}-\dfrac{(5-x)'}{5-x}=\dfrac{1}{x-2}+\dfrac{1}{5-x}>0 \]
επομένως η $f$ είναι γνησίως αύξουσα στο $(2,5)$, άρα και $1-1$ οπότε αντιστρέφεται. Η $f^{-1}$ έχει πεδίο ορισμού το σύνολο τιμών της $f$:
\[ D_{f^{-1}}=f(D_f)=f((2,5))\stackrel{f\nearrow}{=}\left(\lim\limits_{x\to2^+}{f(x)},\lim\limits_{x\to5^-}{f(x)}\right) \]
\begin{itemize}
\item $\lim\limits_{x\to2^+}{f(x)}=\lim\limits_{x\to2^+}{\left(\ln{(x-2)}-\ln{(5-x)}\right)}=-\infty-\ln{3}=-\infty$.
\item $\lim\limits_{x\to5^-}{f(x)}=\lim\limits_{x\to5^-}{\left(\ln{(x-2)}-\ln{(5-x)}\right)}=\ln{3}-(-\infty)=+\infty$.
\end{itemize}
οπότε $D_{f^{-1}}=\mathbb{R}$. Για την εύρεση του τύπου έχουμε $y=f(x)\Leftrightarrow x=f^{-1}(y)$. Είναι λοιπόν
\begin{align*}
y=f(x)&\Leftrightarrow y=\ln{(x-2)}-\ln{(5-x)}\Leftrightarrow y=\ln{\dfrac{x-2}{5-x}}\Leftrightarrow\\
&\Leftrightarrow e^y=\dfrac{x-2}{5-x}\Leftrightarrow e^y(5-x)=x-2\Leftrightarrow\\
&\Leftrightarrow 5e^y-xe^y=x-2\Leftrightarrow x+xe^y=5e^y+2\Leftrightarrow\\
&\Leftrightarrow x\left(e^y+1\right)=5e^y+2\Leftrightarrow x=\dfrac{5e^y+2}{e^y+1}\Leftrightarrow f^{-1}(y)=\dfrac{5e^y+2}{e^y+1}
\end{align*}
άρα η αντίστροφη της $f$ είναι $f^{-1}(x)=\dfrac{5e^x+2}{e^x+1}\ ,\ D_{f^{-1}}=\mathbb{R}$.

\Paradeigma{Εύρεση αντίστροφης σε ένωση διαστημάτων}
\bmath{Δίνεται η συνάρτηση $f(x)=\dfrac{2x-1}{x-1}$. Να δείξετε ότι η $f$ είναι αντιστρέψιμη και να βρείτε την αντίστροφή της.}\\\\
\lysh
Η συνάρτηση $f$ ορίζεται όταν $x-1\neq 0\Rightarrow x\neq 1$, άρα $D_f=(-\infty,1)\cup(1,+\infty)$. Η $f$ είναι παραγωγίσιμη στο $D_f$ με
\[ f'(x)=\left(\dfrac{2x-1}{x-1}\right)'=\dfrac{2(x-1)-(2x-1)\cdot 1}{(x-1)^2}=\dfrac{2x-2-2x+1}{(x-1)^2}=\dfrac{-1}{(x-1)^2}<0 \]
Επομένως η $f$ είναι γνησίως φθίνουσα σε καθένα από τα διαστήματα $\varDelta_1=(-\infty,1)$ και $\varDelta_2=(1,+\infty)$. 
Άρα η $f$ είναι $1-1$ σε καθένα από τα $\varDelta_1$ και $\varDelta_2$.
\begin{itemize}
\item Για το $\varDelta_1=(-\infty,1)$, η $f$ είναι γνησίως φθίνουσα, οπότε 
\[ f(\varDelta_1)=\left(\lim\limits_{x\to 1^-}f(x),\lim\limits_{x\to-\infty}f(x)\right)=(-\infty,2) \]
διότι $\lim\limits_{x\to-\infty}\dfrac{2x-1}{x-1}=\lim\limits_{x\to-\infty}\dfrac{2x}{x}=2$ και $\lim\limits_{x\to 1^-}\dfrac{2x-1}{x-1}=\dfrac{1}{0^-}=-\infty$.
\item Για το $\varDelta_2=(1,+\infty)$, η $f$ είναι γνησίως φθίνουσα, οπότε
\[ f(\varDelta_2)=\left(\lim\limits_{x\to+\infty}f(x),\lim\limits_{x\to 1^+}f(x)\right)=(2,+\infty) \]
διότι $\lim\limits_{x\to+\infty}\dfrac{2x-1}{x-1}=\lim\limits_{x\to+\infty}\dfrac{2x}{x}=2$ και $\lim\limits_{x\to 1^+}\dfrac{2x-1}{x-1}=\dfrac{1}{0^+}=+\infty$.
\end{itemize}
Για να δείξουμε ότι η $f$ είναι $1-1$ μπορούμε να εργαστούμε με δύο τρόπους:
\begin{tropos}
\item 
\textbf{Με μονοτονία και σύνολο τιμών:}\\
Για να είναι η $f$ συνάρτηση $1-1$ σε όλο το πεδίο ορισμού της, πρέπει τα σύνολα τιμών $f(\varDelta_1)$ και $f(\varDelta_2)$ να είναι ξένα μεταξύ τους. Επειδή $f(\varDelta_1)\cap f(\varDelta_2)=(-\infty,2)\cap(2,+\infty)=\varnothing$, η συνάρτηση $f$ είναι $1-1$.

\item \textbf{Με τον ορισμό:}\\
Έστω $x_1,x_2\in D_f$ με $f(x_1)=f(x_2)$. Τότε:
\begin{align*}
\dfrac{2x_1-1}{x_1-1}&=\dfrac{2x_2-1}{x_2-1}\Rightarrow\\
(2x_1-1)(x_2-1)&=(2x_2-1)(x_1-1)\Rightarrow\\
2x_1x_2-2x_1-x_2+1&=2x_1x_2-2x_2-x_1+1\Rightarrow\\
-2x_1-x_2&=-2x_2-x_1\Rightarrow\\
-x_1&=-x_2\Rightarrow \\x_1&=x_2
\end{align*}
\end{tropos}
Άρα η $f$ είναι $1-1$ οπότε αντιστρέφεται. Με οποιονδήποτε τρόπο κι αν εργαστούμε έχουμε βρει ότι το σύνολο τιμών της $f$ άρα και πεδίο ορισμού της αντίστροφης είναι
\[f(D_f)=D_{f^{-1}}=f(\varDelta_1)\cup f(\varDelta_2)=(-\infty,2)\cup(2,+\infty)\]
Για την εύρεση της $f^{-1}$ θέτουμε $y=f(x)$ και λύνουμε ως προς $x$:
\begin{align*}
y=\dfrac{2x-1}{x-1}&\Leftrightarrow y(x-1)=2x-1\Leftrightarrow\\
&\Leftrightarrow yx-y=2x-1\Leftrightarrow yx-2x=y-1\Leftrightarrow\\
&\Leftrightarrow x(y-2)=y-1 \Leftrightarrow x=\dfrac{y-1}{y-2}\Leftrightarrow f^{-1}(y)=\dfrac{y-1}{y-2}
\end{align*}
Η αντίστροφη συνάρτηση είναι:
\[ f^{-1}(x)=\dfrac{x-1}{x-2}\ ,\ D_{f^{-1}}=\mathbb{R}-\{2\} \]
\subsection{Όρια μορφής \bmath{$\frac{0}{0}$}}
\begin{askhsh}{Όριο $ \frac{0}{0} $ ρητής}
Αν ένα όριο $\lim\limits_{x\to x_0}{\dfrac{P(x)}{Q(x)}}$, όπου $P(x),Q(x)$ πολυώνυμα, έχει μορφή $\frac{0}{0}$ τότε εργαζόμαστε ως εξής:
\begin{tropos}
\item \kerkissans{\textbf{Παραγοντοποίηση}}
\\\bhmata\begin{bhma}
\item Παραγοντοποιούμε αριθμητή και παρονομαστή.
\item Απλοποιούμε τις παραστάσεις $x-x_0$ και αντικαθιστούμε όπου $x$ το $x_0$.
\end{bhma}
\item \kerkissans{\textbf{Κανόνας \en{De L'Hospital}}}\\
Εφαρμόζουμε τον κανόνα \en{De L'Hospital} όσες φορές χρειαστεί έως ότου φύγει η απροσδιοριστία.
\end{tropos}
\end{askhsh}
\Paradeigma{Όριο $\frac{0}{0}$ ρητής - Παραγοντοποίηση}
\bmath{Υπολογίστε τα παρακάτω όρια
\begin{multicols}{4}
\begin{alist}
\item $\lim\limits_{x\to 2}{\dfrac{x^2-4}{x^2-2x}}$
\item $\lim\limits_{x\to -1}{\dfrac{x^2-x-2}{x^3+1}}$
\item $\lim\limits_{x\to 1}{\dfrac{x^3-7x+6}{x-x^2}}$
\item $\lim\limits_{x\to \frac{1}{2}}{\dfrac{4x^2-4x+1}{4x^2-1}}$
\end{alist}
\end{multicols}}
\lysh\vspace{-5mm}
\begin{alist}
\item $\lim\limits_{x\to 2}{\dfrac{x^2-4}{x^2-2x}}\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to 2}{\dfrac{(x-2)(x+2)}{x(x-2)}}=\lim\limits_{x\to 2}{\dfrac{x+2}{x}}=\dfrac{2+2}{2}=2$.
\item $\lim\limits_{x\to -1}{\dfrac{x^2-x-2}{x^3+1}}\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to -1}{\dfrac{(x+1)(x-2)}{(x+1)\left(x^2-x+1\right)}}=\lim\limits_{x\to -1}{\dfrac{x-2}{x^2-x+1}}=\dfrac{-1-2}{1+1+1}=\dfrac{-3}{3}=-1$.
\item $\lim\limits_{x\to 1}{\dfrac{x^3-7x+6}{x-x^2}}\xlongequal[\text{\eng{Horner}}]{\frac{0}{0}}\lim\limits_{x\to 1}{\dfrac{(x-1)\left(x^2+x-6\right)}{x\left(1-x\right)}}=\lim\limits_{x\to 1}{\dfrac{-\left(x^2+x-6\right)}{x}}=\dfrac{-1-1+6}{1}=4$
\item $\lim\limits_{x\to\frac{1}{2}}{\dfrac{4x^2-4x+1}{4x^2-1}}\xlongequal[{\footnotesize \text{ταυτότητα}}]{\frac{0}{0}}\lim\limits_{x\to\frac{1}{2}}{\dfrac{\left(2x-1\right)^2}{(2x-1)(2x+1)}}=\lim\limits_{x\to\frac{1}{2}}{\dfrac{2x-1}{2x+1}}=\dfrac{2\cdot\dfrac{1}{2}-1}{2\cdot\frac{1}{2}+1}=0$
\end{alist}
\Paradeigma{Όριο $\frac{0}{0}$ ρητής - Κανόνας \en{De L'Hospital}}
\bmath{Υπολογίστε τα παρακάτω όρια
\begin{multicols}{2}
\begin{alist}
\item $\lim\limits_{x\to 2}{\dfrac{x^2+3x-10}{x^3-8}}$
\item $\lim\limits_{x\to -1}{\dfrac{x^2+2x+1}{x^2-x}}$
\end{alist}
\end{multicols}}
\lysh\vspace{-5mm}
\begin{alist}
\item $\lim\limits_{x\to 2}{\dfrac{x^2+3x-10}{x^3-8}}\dlh{\frac{0}{0}}\lim\limits_{x\to 2}{\dfrac{\left(x^2+3x-10\right)'}{\left(x^3-8\right)'}}=\lim\limits_{x\to 2}{\dfrac{2x+3}{3x^2}}=\dfrac{2\cdot 2+3}{3\cdot 2^2}=\dfrac{7}{12}$
\item $\lim\limits_{x\to-1}{\dfrac{x^2+2x+1}{x^2-x}}\dlh{\frac{0}{0}}\lim\limits_{x\to-1}{\dfrac{\left(x^2+2x+1\right)'}{\left(x^2-x\right)'}}=\lim\limits_{x\to-1}{\dfrac{2x+2}{2x-1}}=\dfrac{2\cdot\left(-1\right)+2}{2\cdot(-1)-1}=0$
\end{alist}
\begin{askhsh}{Όριο $ \frac{0}{0} $ άρρητης}
\bhmata\begin{bhma}
\item Πολλαπλασιάζουμε αριθμητή και παρονομαστή με τη συζυγή παράσταση του όρου που περιέχει τη ρίζα. Αν ρίζες υπάρχουν και στους δύο όρους, πολλαπλασιάζουμε με τις συζυγείς και των δύο.
\item Σχηματίζουμε την ταυτότητα $(a-\beta)(a+\beta)=a^2-\beta^2$ για να διώξουμε τη ρίζα.
\item Κάνουμε τις πράξεις και παραγοντοποιούμε.
\item Απλοποιούμε τον κοινό παράγοντα $x-x_0$ και υπολογίζουμε το όριο με αντικατάσταση.
\end{bhma}
\end{askhsh}
\Paradeigma{Όριο $\frac{0}{0}$ άρρητης}
\bmath{Υπολογίστε τα παρακάτω όρια
\begin{multicols}{2}
\begin{alist}
\item $\lim\limits_{x\to 2}{\dfrac{\sqrt{3x-2}-2}{x^2-4}}$
\item $\lim\limits_{x\to 2}{\dfrac{\sqrt{x+1}-\sqrt{2x-1}}{\sqrt{x+2}-2}}$
\end{alist}
\end{multicols}}
\lysh\vspace{-5mm}
\begin{alist}
\item 
$\begin{aligned}[t]
\lim\limits_{x\to 2}{\dfrac{\sqrt{3x-2}-2}{x^2-2}}&\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to 2}{\dfrac{\left(\sqrt{3x-2}-2\right)\left(\sqrt{3x-2}+2\right)}{\left(x^2-4\right)\left(\sqrt{3x-2}+2\right)}}=\\&=\lim\limits_{x\to 2}{\dfrac{3x-2-4}{(x-2)(x+2)\left(\sqrt{3x-2}+2\right)}}=\\&=\lim\limits_{x\to 2}{\dfrac{3(x-2)}{(x-2)(x+2)\left(\sqrt{3x-2}+2\right)}}=\lim\limits_{x\to 2}{\dfrac{3}{(x-2)\left(\sqrt{3x-2}+2\right)}}=\dfrac{3}{32}
\end{aligned}$
\item 
$\begin{aligned}[t]
\lim\limits_{x\to 2}{\dfrac{\sqrt{x+1}-\sqrt{2x-1}}{\sqrt{x+2}-2}}&\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to 2}{\dfrac{\left(\sqrt{x+1}-\sqrt{2x-1}\right)\left(\sqrt{x+1}+\sqrt{2x-1}\right)\left(\sqrt{x+2}+2\right)}{\left(\sqrt{x+2}-2\right)\left(\sqrt{x+1}+\sqrt{2x-1}\right)\left(\sqrt{x+2}+2\right)}}=\\&=\lim\limits_{x\to 2}{\dfrac{\left(x+1-2x+1\right)\left(\sqrt{x+2}+2\right)}{\left(x+2-4\right)\left(\sqrt{x+1}+\sqrt{2x-1}\right)}}=\\&=\lim\limits_{x\to 2}{\dfrac{\left(2-x\right)\left(\sqrt{x+2}+2\right)}{\left(x-2\right)\left(\sqrt{x+1}+\sqrt{2x-1}\right)}}=\lim\limits_{x\to 2}{\dfrac{-\sqrt{x+2}-2}{\sqrt{x+1}+\sqrt{2x-1}}}=\dfrac{-4}{2\sqrt{3}}=\dfrac{-2\sqrt{3}}{3}
\end{aligned}$
\end{alist}
\begin{askhsh}{Όριο $ \frac{0}{0} $ ρητής με απόλυτες τιμές}
\bhmata\begin{bhma}
\item Υπολογίζω τα όρια των παραστάσεων μέσα στις απόλυτες τιμές.
\item Διώχνω τις απόλυτες τιμές με τον παρακάτω κανόνα
\begin{gather*}
\lim\limits_{x\to x_0}{f(x)}>0\Rightarrow f(x)>0\ \textrm{κοντά στο }x_0\\
\lim\limits_{x\to x_0}{f(x)}<0\Rightarrow f(x)<0\  \textrm{κοντά στο }x_0
\end{gather*}
Αν κάποια απόλυτη τιμή μηδενίζεται στο $ x_0 $ τότε υπολογίζω πλευρικά όρια.
\item Υπολογίζω όριο ρητής $ \frac{0}{0} $
\end{bhma}
\end{askhsh}
\Paradeigma{Όριο $ \frac{0}{0} $ - Η απόλυτη τιμή δεν μηδενίζεται}
\bmath{Να υπολογίσετε το όριο: $\displaystyle \lim\limits_{x\to 3} \dfrac{|x^2-10| - 1}{x-3}$}

\lysh
Για $x=3$ έχουμε: $|3^2-10|-1 = |-1|-1 = 1-1=0$ και $3-3=0$. Υπολογίζουμε το όριο της παράστασης μέσα στην απόλυτη τιμή:
\[ \lim\limits_{x\to 3} (x^2-10) = 9 - 10 = -1 < 0 \]
Επειδή το όριο είναι αρνητικό, η παράσταση είναι αρνητική κοντά στο 3. Άρα: ${|x^2-10| = -(x^2-10) = -x^2+10}$.
\begin{align*}
\lim\limits_{x\to 3} \dfrac{(-x^2+10) - 1}{x-3} &= \lim\limits_{x\to 3} \dfrac{-x^2 + 9}{x-3} \\
&= \lim\limits_{x\to 3} \dfrac{-(x^2 - 9)}{x-3} \\
&= \lim\limits_{x\to 3} \dfrac{-(x-3)(x+3)}{x-3} \\
&= \lim\limits_{x\to 3} [-(x+3)] = -(3+3) = {-6}
\end{align*}
\Paradeigma{Όριο $ \frac{0}{0} $ = Η απόλυτη τιμή μηδενίζεται (Πλευρικά Όρια)}
\bmath{Να υπολογίσετε (αν υπάρχει) το όριο: $\displaystyle \lim\limits_{x\to 1} \dfrac{x^2 - x}{|x-1|}$.}\\\\
\lysh
Για $x=1$ έχουμε μορφή $\frac{0}{0}$. Η παράσταση μέσα στην απόλυτη τιμή ($x-1$) μηδενίζεται, οπότε παίρνουμε πλευρικά όρια.

\begin{itemize}
    \item \textbf{Όριο στο $1^+$ ($x > 1$):}
    \\
    Είναι $x-1 > 0$, άρα $|x-1| = x-1$.
    \[ \lim\limits_{x\to 1^+} \dfrac{x(x-1)}{x-1} = \lim\limits_{x\to 1^+} x = {1} \]

    \item \textbf{Όριο στο $1^-$ ($x < 1$):}
    \\
    Είναι $x-1 < 0$, άρα $|x-1| = -(x-1)$.
    \[ \lim\limits_{x\to 1^-} \dfrac{x(x-1)}{-(x-1)} = \lim\limits_{x\to 1^-} (-x) = {-1} \]
\end{itemize}
Επειδή τα πλευρικά όρια είναι διαφορετικά, το όριο {δεν υπάρχει}.
\begin{askhsh}{Όριο $ \frac{0}{0} $ με τριγωνομετρικές παραστάσεις}
\begin{tropos}
\item Κατασκευάζω και χρησιμοποιώ τριγωνομετρικές ταυτότητες.
\item Κατασκευάζω με πράξεις κάποιο βασικό τριγωνομετρικό όριο, αρκεί όταν $ x\to x_0 $ να μηδενίζεται η γωνία του τριγωνομετρικού αριθμού.
\item Κανόνας \en{De L' Hospital}. (Χρειάζεται προσοχή εδώ γιατί μπορεί η εφαρμογή του κανόνα να με οδηγήσει σε δυσκολότερο όριο.)
\end{tropos}
\end{askhsh}
\Paradeigma{Τριγωνομετρικά όρια $\frac{0}{0}$}
\bmath{Να υπολογίσετε τα όρια
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](3)
\task $\lim\limits_{x\to 0}{\dfrac{\syn{x}-1}{x^2}}$
\task $\lim\limits_{x\to 0}{\dfrac{\ef{x}}{x}}$
\task $\lim\limits_{x\to 0}{\dfrac{x\cdot\hm{x}}{x^2+\hm{x}}}$
\end{tasks}
}
\lysh
\vspace{-5mm}
\begin{alist}
\item $
\begin{aligned}[t]
\lim\limits_{x\to 0}{\dfrac{\syn{x}-1}{x^2}}&\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to 0}{\dfrac{\left(\syn{x}-1\right)\left(\syn{x}+1\right)}{x^2\left(\syn{x}+1\right)}}=\lim\limits_{x\to 0}{\dfrac{\syn^2{x}-1}{x^2\left(\syn{x}+1\right)}}=\\&=\lim\limits_{x\to 0}{\dfrac{-\hm^2{x}}{x^2\left(\syn{x}+1\right)}}=\lim\limits_{x\to 0}{\left[\left(\dfrac{\hm{x}}{x}\right)^2\dfrac{-1}{\syn{x}+1}\right]}=1\cdot\dfrac{-1}{2}=-\dfrac{1}{2}
\end{aligned}
$
\item $\lim\limits_{x\to 0}{\dfrac{\ef{x}}{x}}\stackrel{\frac{0}{0}}{=}\lim\limits_{x\to 0}{\dfrac{\dfrac{\hm{x}}{\syn{x}}}{x}}=\lim\limits_{x\to 0}{\dfrac{\hm{x}}{x\cdot\syn{x}}}=\lim\limits_{x\to 0}{\left(\dfrac{\hm{x}}{x}\cdot\dfrac{1}{\syn{x}}\right)}=1\cdot\dfrac{1}{2}=\dfrac{1}{2}$
\item $\lim\limits_{x\to 0}{\dfrac{x\cdot\hm{x}}{x^2+\hm{x}}}\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}\lim\limits_{x\to 0}{\dfrac{\hm{x}+x\cdot\syn{x}}{2x+\syn{x}}}=\dfrac{0}{1}=0$
\end{alist}

\begin{askhsh}{Όριο $ \frac{0}{0} $ με εκθετικές, λογαριθμικές και συνδυασμό αυτών}
Εφαρμογή του κανόνα \eng{De L' Hospital}.
\end{askhsh}
\Paradeigma{Όριο $\frac{0}{0}$ με εκθετικές και λογαριθμικές}
\bmath{Υπολογίστε τα παρακάτω όρια
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](3)
\task $\lim\limits_{x\to +\infty}{\dfrac{\ln{x}}{x}}$
\task $\lim\limits_{x\to 0}{\dfrac{x^2-x}{e^x-1}}$
\task $\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}}{\ln{x}}}$
\end{tasks}}
\lysh\vspace{-7mm}
\begin{alist}
\item $\lim\limits_{x\to 0}{\dfrac{\ln{(x+1)}}{x}}\xlongequal[\text{\eng{DLH}}]{\frac{0}{0}}\lim\limits_{x\to 0}{\dfrac{\frac{1}{x+1}}{1}}=\lim\limits_{x\to0}{\dfrac{1}{x+1}}=1$.
\item $\lim\limits_{x\to 0}{\dfrac{x^2-x}{e^x-1}}\xlongequal[\text{\en{DLH}}]{\frac{0}{0}}\lim\limits_{x\to 0}{\dfrac{2x-1}{e^x}}=\dfrac{-1}{1}=-1$.
\item $\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}}{\ln{x}}}\xlongequal[\text{\en{DLH}}]{\frac{+\infty}{-\infty}}\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}\cdot\left(-\frac{1}{x^2}\right)}{\frac{1}{x}}}=\lim\limits_{x\to 0^+}{\left(-\dfrac{e^{\frac{1}{x}}}{x}\right)}=\lim\limits_{x\to0^+}{\left(-\dfrac{1}{x}e^{\frac{1}{x}}\right)}=(-\infty)\cdot(+\infty)=-\infty$.
\end{alist}
\subsection{Όρια με απροσδιοριστία $ \frac{\pm\infty}{\pm\infty} $}
\begin{askhsh}{Όρια $ \dfrac{\pm\infty}{\pm\infty} $ ρητής}
Με ρητή συνάρτηση όταν $ x\to\pm\infty $ υπολογίζουμε το όριο του κλάσματος μόνο με τους μεγιστοβάθμιους όρους.
\end{askhsh}
\Paradeigma{Όριο ρητής στο $-\infty$}
\bmath{Να υπολογίσετε τα όρια: 
\begin{multicols}{2}
\begin{alist}
\item $\lim\lim\limits_{x\to -\infty} \dfrac{3x^3 - 5x + 2}{2x^3 + x^2 - 1}$
\item $\lim\limits_{x\to +\infty} \dfrac{x^2 + 1}{x^3 - x}$
\end{alist}
\end{multicols}
}
\lysh
\begin{alist}
\item Το όριο είναι της μορφής $\frac{-\infty}{-\infty}$. Εφαρμόζουμε τον κανόνα των μεγιστοβάθμιων όρων:
\[
\lim\limits_{x\to -\infty} \dfrac{3x^3 - 5x + 2}{2x^3 + x^2 - 1} = \lim\limits_{x\to -\infty} \dfrac{3x^3}{2x^3} = \dfrac{3}{2}
\]
\item Ομοίως, κρατάμε τους μεγιστοβάθμιους:
\[
\lim\limits_{x\to +\infty} \dfrac{x^2 + 1}{x^3 - x}=\lim\limits_{x\to +\infty} \dfrac{x^2}{x^3} = \lim\limits_{x\to +\infty} \dfrac{1}{x} = 0
\]
\end{alist}
\begin{askhsh}{Όρια $ \frac{\pm\infty}{\pm\infty} $ με ρίζες}
Για τον υπολογισμό ορίων της μορφής $\dfrac{\pm\infty}{\pm\infty}$ που περιέχουν ρίζες, εφαρμόζουμε τη \textbf{μέθοδο του κοινού παράγοντα}:
\bhmata\begin{bhma}
\item Μέσα σε κάθε ρίζα βγάζουμε κοινό παράγοντα τη μεγαλύτερη δύναμη του $x$ (π.χ. σε τετραγωνική ρίζα το $x^2$).
\item Βγάζουμε τη δύναμη αυτή έξω από τη ρίζα με τη βοήθεια της ιδιότητας $\sqrt{x^2} = |x|$ (γενικότερα $\sqrt[\nu]{x^\nu} = |x|$ για άρτιο $\nu$).
\item Διώχνουμε την απόλυτη τιμή ανάλογα με το πού τείνει το $x$:
\begin{itemize}
    \item Αν $x \to +\infty$, τότε $x > 0$ άρα $|x| = x$.
    \item Αν $x \to -\infty$, τότε $x < 0$ άρα $|x| = -x$.
\end{itemize}
\item Βγάζουμε κοινό παράγοντα τη μεγαλύτερη δύναμη του $x$ συνολικά σε αριθμητή και παρονομαστή.
\item Απλοποιούμε τους κοινούς παράγοντες και υπολογίζουμε το όριο που απομένει. (Υπενθύμιση: $\lim\limits_{x\to \pm\infty}\dfrac{c}{x^\nu} = 0$).
\end{bhma}
\end{askhsh}
\Paradeigma{Όριο με ρίζα στο $-\infty$}
\bmath{Να υπολογίσετε το όριο: $\displaystyle \lim\limits_{x\to -\infty} \dfrac{\sqrt{4x^2+x+1} - 3x}{x+2}$}

\lysh
Βγάζουμε κοινό παράγοντα το $x^2$ μέσα στη ρίζα:
\[
L=\lim\limits_{x\to -\infty} \dfrac{\sqrt{x^2(4+\frac{1}{x}+\frac{1}{x^2})} - 3x}{x\left(1+\frac{2}{x}\right)} = \lim\limits_{x\to -\infty} \dfrac{|x|\sqrt{4+\frac{1}{x}+\frac{1}{x^2}} - 3x}{x\left(1+\frac{2}{x}\right)}
\]
Επειδή $x \to -\infty$, ισχύει $|x| = -x$. Άρα:
\[
L= \lim\limits_{x\to -\infty} \dfrac{-x\sqrt{4+\frac{1}{x}+\frac{1}{x^2}} - 3x}{x\left(1+\frac{2}{x}\right)} = \lim\limits_{x\to -\infty} \frac{x\left( -\sqrt{4+\frac{1}{x}+\frac{1}{x^2}} - 3 \right)}{x\left(1+\frac{2}{x}\right)}
= \dfrac{-\sqrt{4+0+0} - 3}{1+0} = \dfrac{-2-3}{1} = -5
\]
\begin{askhsh}{Όρια $ \frac{\pm\infty}{\pm\infty} $ Διάφορες συναρτήσεις}
Εφαρμόζουμε τον κανόνα \en{De L' Hospital}
\end{askhsh}
\Paradeigma{Όρια $ \frac{\pm\infty}{\pm\infty} $ Διάφορες συναρτήσεις}
\bmath{Υπολογίστε τα παρακάτω όρια
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](3)
\task $\lim\limits_{x\to +\infty}{\dfrac{\ln{x}}{x}}$
\task $\lim\limits_{x\to +\infty} \dfrac{e^x + x}{x^2 + 1}$
\task $\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}}{\ln{x}}}$
\end{tasks}}
\lysh\vspace{-7mm}
\begin{alist}
\item $\lim\limits_{x\to +\infty}{\dfrac{\ln{x}}{x}}\dlh{\frac{-\infty}{+\infty}}\lim\limits_{x\to +\infty}{\dfrac{\frac{1}{x}}{1}}=\lim\limits_{x\to+\infty}{\frac{1}{x}}=0$.
\item $\lim\limits_{x\to +\infty} \dfrac{e^x + x}{x^2 + 1}\dlh{\frac{+\infty}{+\infty}}\lim\limits_{x\to +\infty} \dfrac{(e^x + x)'}{(x^2 + 1)'} = \lim\limits_{x\to +\infty} \dfrac{e^x + 1}{2x}\dlh{\frac{+\infty}{+\infty}} \lim\limits_{x\to +\infty} \dfrac{e^x}{2} = +\infty$.
\item $\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}}{\ln{x}}}\dlh{\frac{+\infty}{-\infty}}\lim\limits_{x\to 0^+}{\dfrac{e^{\frac{1}{x}}\cdot\left(-\frac{1}{x^2}\right)}{\frac{1}{x}}}=\lim\limits_{x\to 0^+}{\left(-\frac{e^{\frac{1}{x}}}{x}\right)}=\lim\limits_{x\to0^+}{\left(-\frac{1}{x}e^{\frac{1}{x}}\right)}=(-\infty)\cdot(+\infty)=-\infty$.
\end{alist}
\subsection{Όρια με απροσδιοριστία \bmath{$ 0\cdot(\pm\infty) $}}
\begin{askhsh}{Όρια $ 0\cdot(\pm\infty) $ - Γενική μέθοδος}
\bhmata\begin{bhma}
\item Γράφουμε το γινόμενο με μορφή σύνθετου κλάσματος ως εξής
\[ \lim\limits_{x\to x_0}{\left(f(x)\cdot g(x)\right)}=\lim\limits_{x\to x_0}{\dfrac{f(x)}{\frac{1}{g(x)}}}\ \ \textrm{ή}\ \ \lim\limits_{x\to x_0}{\dfrac{g(x)}{\frac{1}{f(x)}}} \]
\item Το όριο παίρνει τη μορφή $ \frac{0}{0} $ ή $ \frac{\pm\infty}{\pm\infty} $ οπότε εφαρμόζουμε κανόνα \en{De L' Hospital}
\end{bhma}
\end{askhsh}

\Paradeigma{Όρια \bmath{$ 0\cdot(\pm\infty) $}}
\bmath{Να υπολογίσετε τα όρια: 
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
\task $\displaystyle \lim\limits_{x\to 0^+} (x \ln x)$
\task $\displaystyle \lim\limits_{x\to +\infty} (x \cdot e^{-x})$
\end{tasks}}
\lysh\vspace{-7mm}
\begin{alist}
\item Έχουμε απροσδιοριστία της μορφής $0 \cdot (-\infty)$. Επιλέγουμε να κρατήσουμε τον λογάριθμο στον αριθμητή και διαιρούμε με $\dfrac{1}{x}$:
\[ \lim\limits_{x\to 0^+} (x \ln x) = \lim\limits_{x\to 0^+} \dfrac{\ln x}{\frac{1}{x}} \dlh{\frac{-\infty}{+\infty}} \lim\limits_{x\to 0^+} \dfrac{(\ln x)'}{(\dfrac{1}{x})'} = \lim\limits_{x\to 0^+} \dfrac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim\limits_{x\to 0^+} \left( \dfrac{1}{x} \cdot (-x^2) \right) = \lim\limits_{x\to 0^+} (-x) = {0} \]
\begin{center}
\begin{parat}{15cm}
\begin{itemize}[leftmargin=2mm]
\item Στον αριθμητή του κλάσματος κρατάμε τη «δύσκολη» συνάρτηση. (ως προς την παράγωγο)
\item Μετά από κάθε εφαρμογή του κανόνα \eng{DLH}, \textbf{πρώτα} απλοποιούμε την παράσταση και μετά ελέγχουμε αν διορθώθηκε η απροσδιοριστία.
\end{itemize}
\end{parat}
\end{center}
\item 
Γράφουμε το $e^{-x}$ ως $\dfrac{1}{e^x}$.
\[ \lim\limits_{x\to +\infty} (x \cdot e^{-x}) = \lim\limits_{x\to +\infty} \dfrac{x}{e^x} \dlh{\frac{+\infty}{+\infty}} \lim\limits_{x\to +\infty} \dfrac{(x)'}{(e^x)'} = \lim\limits_{x\to +\infty} \dfrac{1}{e^x} = \lim\limits_{x\to +\infty} \dfrac{1}{e^x} = {0} \]
\end{alist}
\begin{askhsh}{Όρια με απροσδιοριστία $ 0^0,1^{\pm\infty},(\pm\infty)^0 $}
\bhmata\begin{bhma}
\item Χρησιμοποιούμε τον παρακάτω κανόνα
\[ \lim\limits_{x\to x_0}{f(x)^{g(x)}}=\lim\limits_{x\to x_0}{e^{\ln{f(x)^{g(x)}}}}=\lim\limits_{x\to x_0}{e^{g(x)\cdot\ln{f(x)}}} \]
\item Υπολογίζουμε το όριο του εκθέτη το οποίο έχει απροσδιοριστία $ 0\cdot(\pm\infty) $. Στη συνέχεια με αντικατάσταση υπολογίζουμε το αρχικό όριο.
\end{bhma}
\end{askhsh}
\Paradeigma{Όρια με απροσδιοριστία $0^0$ και $1^{+\infty}$}
\bmath{Να υπολογίσετε τα παρακάτω όρια:
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
\task $\displaystyle \lim\limits_{x\to 0^+} x^x$
\task $\displaystyle \lim\limits_{x\to +\infty} \left(1+\dfrac{1}{x}\right)^x$
\end{tasks}}
\lysh\vspace{-7mm}
\begin{alist}
\item Έχουμε απροσδιοριστία της μορφής $0^0$:
\[ L=\lim\limits_{x\to 0^+} x^x =\lim\limits_{x\to 0^+}{e^{\ln{x^x}}} \lim\limits_{x\to 0^+} e^{x\ln x} \]
Υπολογίζουμε το όριο του εκθέτη που έχει μορφή $0\cdot(-\infty)$:
\[ \lim\limits_{x\to 0^+} (x\ln x) = \lim\limits_{x\to 0^+} \dfrac{\ln x}{\frac{1}{x}} \dlh{\frac{-\infty}{+\infty}} \lim\limits_{x\to 0^+} \dfrac{\frac{1}{x}}{-\frac{1}{x^2}} = \lim\limits_{x\to 0^+} (-x) = 0 \]
Επομένως, το αρχικό όριο γίνεται:
\[ L=\lim\limits_{x\to 0^+} e^{x\ln x} = e^0 = 1 \]
\item Έχουμε απροσδιοριστία της μορφής $1^{+\infty}$. Έχουμε:
\[ \lim\limits_{x\to +\infty} \left(1+\frac{1}{x}\right)^x =\lim\limits_{x\to +\infty} e^{\ln\left(1+\frac{1}{x}\right)^x}= \lim\limits_{x\to +\infty} e^{x\ln\left(1+\frac{1}{x}\right)} \]
Υπολογίζουμε το όριο του εκθέτη:
\[ \lim\limits_{x\to +\infty} x\ln\left(1+\dfrac{1}{x}\right) = \lim\limits_{x\to +\infty} \dfrac{\ln\left(1+\frac{1}{x}\right)}{\frac{1}{x}} \dlh{\frac{0}{0}} \lim\limits_{x\to +\infty} \dfrac{\frac{1}{1+\frac{1}{x}}\cdot\left(-\frac{1}{x^2}\right)}{-\frac{1}{x^2}} = \lim\limits_{x\to +\infty} \frac{1}{1+\frac{1}{x}} = \frac{1}{1+0} = 1 \]
Επομένως, το αρχικό όριο γίνεται:
w\[ \lim\limits_{x\to +\infty} e^{x\ln\left(1+\frac{1}{x}\right)} = e^1 = e \]
\end{alist}
\subsection{Όρια με απροσδιοριστία \bmath{$ +\infty-\infty $}}
\begin{askhsh}{Όρια $ +\infty-\infty $ - Γενική μέθοδος}
\bhmata
\begin{bhma}
\item Βγάζουμε κοινό παράγοντα μια από τις δύο συναρτήσεις.
\[ \lim\limits_{x\to x_0}{(f(x)-g(x))}=\lim\limits_{x\to x_0}{\left[f(x)\left(1-\dfrac{g(x)}{f(x)}\right)\right]} \]
\item Στη συνέχεια υπολογίζουμε το όριο του κλάσματος $ \dfrac{g(x)}{f(x)} $ με μορφή $ \dfrac{\pm\infty}{\pm\infty} $.
\item Επιστρέφουμε στο αρχικό όριο αντικαθιστώντας.
\end{bhma}
\end{askhsh}
\Paradeigma{Υπολογισμός ορίου με εξαγωγή κοινού παράγοντα}
\bmath{Να υπολογίσετε το όριο: \[\lim\limits_{x\to +\infty} (e^x - x^2 + 2x - 1)\]}

\lysh
Το όριο παρουσιάζει απροσδιοριστία μορφής $+\infty - \infty$ (αφού $e^x \to +\infty$ και $-x^2 \to -\infty$). Εφαρμόζουμε τη μεθοδολογία βγάζοντας κοινό παράγοντα τον  $e^x$.
\[ 
f(x) = e^x - x^2 + 2x - 1 = e^x \left( 1 - \dfrac{x^2}{e^x} + \dfrac{2x}{e^x} - \dfrac{1}{e^x} \right)
\]
Υπολογίζουμε τα επιμέρους όρια:
\begin{itemize}
    \item $\displaystyle \lim\limits_{x\to +\infty} \dfrac{x^2}{e^x} \stackrel{\frac{+\infty}{+\infty}}{=} \lim\limits_{x\to +\infty} \dfrac{(x^2)'}{(e^x)'} = \lim\limits_{x\to +\infty} \dfrac{2x}{e^x} \stackrel{\frac{+\infty}{+\infty}}{=} \lim\limits_{x\to +\infty} \dfrac{2}{e^x} = 0$.
    \item Ομοίως, $\displaystyle \lim\limits_{x\to +\infty} \dfrac{2x}{e^x} = 0$ και $\displaystyle \lim\limits_{x\to +\infty} \dfrac{1}{e^x} = 0$.
\end{itemize}
Επομένως:
\[ 
\lim\limits_{x\to +\infty} f(x) = +\infty \left( 1 - 0 + 0 - 0 \right)= +\infty 
\]
\begin{askhsh}{Όρια $ +\infty-\infty $ - Κλάσματα}
Αν στο όριο $\lim\limits_{x\to x_0}{(f(x)-g(x))}$ οι συναρτήσεις $ f(x),g(x) $ είναι κλάσματα τότε τα κάνουμε ομώνυμα και οδηγούμαστε σε μία από τις μορφές $\dfrac{0}{0}$,$\dfrac{\pm\infty}{\pm\infty}$ ή $\dfrac{a}{0}$.
\end{askhsh}
\Paradeigma{Υπολογισμός ορίου διαφοράς κλασμάτων}
\bmath{Να υπολογίσετε το όριο: 
\[\lim\limits_{x\to 1^+} \left( \dfrac{1}{\ln x} - \dfrac{1}{x-1} \right)\]}
\lysh
Παρατηρούμε ότι: $\displaystyle \lim\limits_{x\to 1^+} \dfrac{1}{\ln x} = +\infty$  και $\displaystyle \lim\limits_{x\to 1^+} \dfrac{1}{x-1} = +\infty$.
Έχουμε μορφή $+\infty - \infty$ οπότε κάνουμε τα κλάσματα ομώνυμα:
\begin{align*}
\lim\limits_{x\to 1^+} \left( \frac{1}{\ln x} - \frac{1}{x-1} \right)&=\lim\limits_{x\to 1} \frac{x-1-\ln x}{(x-1)\ln x}\dlh{\frac{0}{0}}\lim\limits_{x\to 1} \frac{(x-1-\ln x)'}{((x-1)\ln x)'} = \lim\limits_{x\to 1} \frac{1 - \frac{1}{x}}{1\cdot \ln x + (x-1)\frac{1}{x}} =\\&= \lim\limits_{x\to 1} \dfrac{\frac{x-1}{x}}{\ln x + 1 - \frac{1}{x}} = \lim\limits_{x\to 1} \frac{x-1}{x\ln x + x - 1}\dlh{\frac{0}{0}}\lim\limits_{x\to 1} \frac{(x-1)'}{(x\ln x + x - 1)'} =\\&= \lim\limits_{x\to 1} \frac{1}{1\cdot \ln x + x\cdot \frac{1}{x} + 1} = \lim\limits_{x\to 1} \frac{1}{\ln x + 1 + 1} = \frac{1}{0 + 2} = \frac{1}{2}
\end{align*}
\begin{askhsh}{Όρια $ +\infty-\infty $ - Διαφορά λογαρίθμων}
Σχηματίζουμε διαφορά λογαρίθμων και χρησιμοποιούμε την ιδιότητα
\[ \ln{a}-\ln{\beta}=\ln{\dfrac{a}{\beta}} \]
Στη συνέχεια υπολογίζουμε το όριο του κλάσματος το οποίο έχει απροσδιοριστία $\dfrac{\pm\infty}{\pm\infty}$.
\end{askhsh}
\Paradeigma{Υπολογισμός ορίου με διαφορά λογαρίθμων}
\bmath{Να υπολογίσετε το όριο:
\[ \lim\limits_{x\to +\infty} \left[ \ln(2x^2+3) - \ln(x^2+x+1) \right]\]}

\lysh
Έχουμε μορφή $+\infty - \infty$. Χρησιμοποιούμε την ιδιότητα των λογαρίθμων:

\[
\lim\limits_{x\to +\infty} \left[ \ln(2x^2+3) - \ln(x^2+x+1) \right] = \lim\limits_{x\to+\infty}\ln \left( \dfrac{2x^2+3}{x^2+x+1} \right)
\]
Θέτουμε $u = \dfrac{2x^2+3}{x^2+x+1}$. Υπολογίζουμε πρώτα το όριο του $u$ στο $+\infty$:
\[
u_0=\lim\limits_{x\to +\infty} u(x) = \lim\limits_{x\to +\infty} \dfrac{2x^2+3}{x^2+x+1} = \lim\limits_{x\to +\infty} \dfrac{2x^2}{x^2} = 2
\]
Επομένως, το αρχικό όριο γίνεται:
\[
\lim\limits_{u\to 2} \ln u = \ln 2
\]
\begin{askhsh}{Όρια της μορφής $ \frac{a}{0} $}
\bhmata\begin{bhma}
\item Παραγοντοποιούμε τον παρονομαστή.
\item Γράφουμε σε ξεχωριστό κλάσμα τον παράγοντα που μηδενίζεται.
\item Αν αυτός ο παράγοντας έχει σταθερό πρόσημο τότε προχωράμε στον υπολογισμό. Αν όχι υπολογίζουμε πλευρικά όρια.
\end{bhma}
\end{askhsh}
\Paradeigma{Μελέτη ορίων $\frac{a}{0}$}
\bmath{Να υπολογίσετε (αν υπάρχουν) τα παρακάτω όρια:}
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
\task $\displaystyle \lim\limits_{x\to 1} \dfrac{x^2+2}{(x-1)^2}$
\task $\displaystyle \lim\limits_{x\to 2} \dfrac{x+3}{x^2-4}$
\end{tasks}
\lysh\vspace{-7mm}
\begin{alist}
\item  1η Περίπτωση: Σταθερό Πρόσημο Παρονομαστή
\begin{itemize}
    \item Για τον αριθμητή ισχύει: $\lim\limits_{x\to 1} (x^2+2) = 1^2 + 2 = 3 > 0$.
    \item Για τον παρονομαστή ισχύει: $\lim\limits_{x\to 1} (x-1)^2 = 0$.
\end{itemize}
Παρατηρούμε ότι για κάθε $x \neq 1$, ισχύει $(x-1)^2 > 0$ (θετικό κοντά στο 1).
\\
Επομένως:
\[ \lim\limits_{x\to 1} \dfrac{1}{(x-1)^2} = +\infty \]
Άρα για το συνολικό όριο έχουμε:
\[ \lim\limits_{x\to 1} \dfrac{x^2+2}{(x-1)^2} = \lim\limits_{x\to 1} \left[ (x^2+2) \cdot \dfrac{1}{(x-1)^2} \right] = 3 \cdot (+\infty) = +\infty \]

\vspace{0.5cm}
\item  2η Περίπτωση: Αλλαγή Προσήμου (Πλευρικά Όρια)
\\
Έχουμε το όριο $\displaystyle \lim\limits_{x\to 2} \dfrac{x+3}{x^2-4}$.
\begin{itemize}
    \item Για τον αριθμητή: $\lim\limits_{x\to 2} (x+3) = 5 > 0$.
    \item Για τον παρονομαστή: $\lim\limits_{x\to 2} (x^2-4) = 0$.
\end{itemize}
Παραγοντοποιούμε τον παρονομαστή: $x^2-4 = (x-2)(x+2)$. Το πρόσημό του εξαρτάται από το $x-2$, το οποίο αλλάζει πρόσημο στο 2. Άρα εξετάζουμε πλευρικά όρια.
\begin{itemize}
    \item \textbf{Δεξί όριο στο $2$:}
    \\
    Όταν $x \to 2^+$, είναι $x > 2 \Rightarrow x^2 > 4 \Rightarrow x^2-4 > 0$.
    \[ \lim\limits_{x\to 2^+} \dfrac{x+3}{x^2-4} = \dfrac{5}{0^+} = +\infty \]
    
    \item \textbf{Αριστερό όριο στο $2$:}
    \\
    Όταν $x \to 2^-$, είναι $x < 2 \Rightarrow x^2 < 4 \Rightarrow x^2-4 < 0$.
    \[ \lim\limits_{x\to 2^-} \dfrac{x+3}{x^2-4} = \dfrac{5}{0^-} = -\infty \]
\end{itemize}
Επειδή τα πλευρικά όρια είναι διαφορετικά, το όριο \textbf{δεν υπάρχει}.
\end{alist}
\begin{askhsh}{Όρια με τριγωνομετρικές συναρτήσεις $ \hm{f(x)},\syn{f(x)} $ - Μηδενική επί φραγμένη}
Αν το όριο περιέχει σύνθετες τριγωνομετρικές συναρτήσεις με γωνία $ f(x) $ και $ \lim\limits_{x\to x_0}{f(x)}=\pm\infty $ τότε
\\\bhmata\begin{bhma}
\item Γράφουμε τη συνάρτηση μέσα στο όριο ως γινόμενο συναρτήσεων. 
\item Κλείνουμε τη συνάρτηση του ορίου σε απόλυτη τιμή και σχηματίζουμε διπλή ανισότητα ώστε να εφαρμοστεί κριτήριο παρεμβολής.
\end{bhma}
\end{askhsh}
\Paradeigma{Μηδενική επί φραγμένη}
\bmath{Να υπολογίσετε το όριο: $\displaystyle \lim\limits_{x\to +\infty} \dfrac{x\hm{x}}{x^2+1}$}

\lysh
Γράφουμε τη συνάρτηση ως γινόμενο: $\displaystyle f(x) = \dfrac{x}{x^2+1}\cdot\hm{x}$.
Έχουμε $|\hm{x}| \le 1$ για κάθε $x\in\mathbb{R}$. Κλείνουμε την $f(x)$ σε απόλυτη τιμή:
\[ |f(x)| = \left| \dfrac{x}{x^2+1}\hm{x} \right| = \left| \dfrac{x}{x^2+1} \right| |\hm{x}| \le \left| \dfrac{x}{x^2+1} \right| \cdot 1 \]
Για $x>0$ ισχύει $\dfrac{x}{x^2+1} > 0$, οπότε $\left| \dfrac{x}{x^2+1} \right| = \dfrac{x}{x^2+1}$. Επομένως:
\[ |f(x)| \le \dfrac{x}{x^2+1} \Leftrightarrow -\dfrac{x}{x^2+1} \le f(x) \le \dfrac{x}{x^2+1} \]
Επειδή $\displaystyle \lim\limits_{x\to +\infty} \dfrac{x}{x^2+1} = \lim\limits_{x\to +\infty} \dfrac{x}{x^2} = \lim\limits_{x\to +\infty} \dfrac{1}{x} = 0$, ισχύει επίσης ότι $\displaystyle \lim\limits_{x\to +\infty} \left(-\dfrac{x}{x^2+1}\right) = 0$.
Σύμφωνα με το Κριτήριο Παρεμβολής προκύπτει ότι $\displaystyle \lim\limits_{x\to +\infty} f(x) = 0$.
\begin{askhsh}{Γνωστό όριο που περιέχει την $ f(x) $ - Βοηθητική συνάρτηση}
\bhmata\begin{bhma}
\item Θέτουμε $ g(x) $ τη συνάρτηση του ορίου και λύνουμε ως προς $ f(x) $.
\item Υπολογίζουμε το όριο της $ f $ στο $ x_0 $.
\end{bhma}
\end{askhsh}
\Paradeigma{Χρήση βοηθητικής συνάρτησης}
\bmath{Δίνεται συνάρτηση $f:\mathbb{R}\to\mathbb{R}$ για την οποία ισχύει $\displaystyle \lim\limits_{x\to 1} \dfrac{f(x)-2}{x-1} = 4$. Να υπολογίσετε:
\begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
\task $\displaystyle \lim\limits_{x\to 1} f(x)$
\task $\displaystyle \lim\limits_{x\to 1} \dfrac{f(x)-2}{\sqrt{x}-1}$
\end{tasks}}
\lysh\vspace{-7mm}
\begin{alist}
\item Θέτουμε $\displaystyle g(x) = \dfrac{f(x)-2}{x-1}$, με $\lim\limits_{x\to 1} g(x) = 4$. Λύνουμε ως προς $f(x)$ για $x \neq 1$:
\[ f(x) - 2 = (x-1)g(x) \Rightarrow f(x) = (x-1)g(x) + 2 \]
Επομένως:
\[ \lim\limits_{x\to 1} f(x) = \lim\limits_{x\to 1} \left[ (x-1)g(x) + 2 \right] = 0 \cdot 4 + 2 = 2 \]
\item Χρησιμοποιώντας την παραπάνω σχέση $f(x)-2 = (x-1)g(x)$ έχουμε:
\begin{align*}
\lim\limits_{x\to 1} \dfrac{f(x)-2}{\sqrt{x}-1} &= \lim\limits_{x\to 1} \dfrac{(x-1)g(x)}{\sqrt{x}-1} = \lim\limits_{x\to 1} \dfrac{(\sqrt{x}-1)(\sqrt{x}+1)g(x)}{\sqrt{x}-1} \\
&= \lim\limits_{x\to 1} \left[ (\sqrt{x}+1)g(x) \right] = (1+1) \cdot \lim\limits_{x\to 1} g(x) = 2 \cdot 4 = 8
\end{align*}
\end{alist}
\begin{askhsh}{Κριτήριο παρεμβολής}
Το κριτήριο παρεμβολής για τον υπολογισμό ορίων εφαρμόζεται σε ανισότητες της μορφής
\[ g(x)\leq f(x)\leq h(x)\ \ \textrm{ή}\ \ |f(x)|\leq g(x)\Rightarrow -g(x)\leq f(x)\leq g(x) \]
\end{askhsh}
\Paradeigma{Κριτήριο Παρεμβολής}
\bmath{Έστω συνάρτηση $f:\mathbb{R}\to\mathbb{R}$ τέτοια ώστε για κάθε $x\in\mathbb{R}$ να ισχύει:
\[ 4x-1 \leq f(x) \leq 2x^2+1 \]
Να υπολογίσετε το $\displaystyle \lim\limits_{x\to 1} f(x)$.}

\lysh
Έχουμε την ανισότητα $g(x) \leq f(x) \leq h(x)$ όπου $g(x) = 4x-1$ και $h(x) = 2x^2+1$.
Υπολογίζουμε τα όρια των ακραίων συναρτήσεων στο $x_0=1$:
\begin{itemize}
\item $\displaystyle \lim\limits_{x\to 1} g(x) = \lim\limits_{x\to 1} (4x-1) = 4\cdot 1 - 1 = 3$
\item $\displaystyle \lim\limits_{x\to 1} h(x) = \lim\limits_{x\to 1} (2x^2+1) = 2\cdot 1^2 + 1 = 3$
\end{itemize}
Αφού $\lim\limits_{x\to 1} g(x) = \lim\limits_{x\to 1} h(x) = 3$, σύμφωνα με το Κριτήριο Παρεμβολής προκύπτει ότι:
\[ \lim\limits_{x\to 1} f(x) = 3 \]
\subsection{Συνέχεια συναρτήσεων}
\begin{askhsh}{Συνεχής συνάρτηση}
Αν μας ζητείται να εξετάσουμε αν μια συνάρτηση $f$ είναι συνεχής:
\begin{itemize}
\item \textbf{Σε ένα σημείο $x_0$} του πεδίου ορισμού της, τότε υπολογίζουμε την τιμή $f(x_0)$ και το όριο $\lim\limits_{x\to x_0}{f(x)}$ (ή τα πλευρικά όρια αν αλλάζει τύπος εκατέρωθεν του $x_0$) και εξετάζουμε αν $\lim\limits_{x\to x_0}{f(x)}=f(x_0)$.
\item \textbf{Σε όλο το πεδίο ορισμού της} (ή σε ένα ευρύτερο διάστημα), τότε:
  \begin{itemize}
    \item Αιτιολογούμε τη συνέχεια της συνάρτησης στα ανοιχτά διαστήματα που ορίζεται κάθε τύπος της (ως αποτέλεσμα πράξεων ή σύνθεσης γνωστών βασικών συνεχών συναρτήσεων).
    \item Εξετάζουμε ξεχωριστά τη συνέχεια στα σημεία αλλαγής του τύπου της (αν είναι συνάρτηση πολλαπλών κλάδων), χρησιμοποιώντας πλευρικά όρια όπως στο προηγούμενο βήμα.
  \end{itemize}
\end{itemize}
\end{askhsh}

\Paradeigma{Έλεγχος συνέχειας σε σημείο}
\bmath{Δίνεται η συνάρτηση:
\[ f(x) = \begin{cdcases} \dfrac{x^2-4}{x-2}, & x \neq 2 \\[2mm] 4, & x = 2 \end{cdcases} \]
Να εξετάσετε αν η $f$ είναι συνεχής στο σημείο $x_0 = 2$.}

\lysh
Υπολογίζουμε πρώτα την τιμή της συνάρτησης στο σημείο $x_0 = 2$. Έχουμε:
\[ f(2) = 4 \]
Στη συνέχεια υπολογίζουμε το όριο της συνάρτησης καθώς το $x$ τείνει στο $2$. Αφού $x \to 2$, σημαίνει ότι $x \neq 2$, άρα χρησιμοποιούμε τον πρώτο κλάδο. Έχουμε απροσδιοριστία μορφής $\frac{0}{0}$:
\[ \lim\limits_{x\to 2} f(x) = \lim\limits_{x\to 2} \dfrac{x^2-4}{x-2} = \lim\limits_{x\to 2} \dfrac{(x-2)(x+2)}{x-2} = \lim\limits_{x\to 2} (x+2) = 2+2 = 4 \]
Παρατηρούμε ότι ισχύει:
\[ \lim\limits_{x\to 2} f(x) = f(2) = 4 \]
Επομένως, η συνάρτηση $f$ είναι συνεχής στο σημείο $x_0 = 2$.

\Paradeigma{Έλεγχος συνέχειας σε όλο το πεδίο ορισμού}
\bmath{Δίνεται η συνάρτηση:
\[ f(x) = \begin{cdcases} \dfrac{\hm{x}}{x}, & x < 0 \\[2mm] x^2 - x + 1, & x \ge 0 \end{cdcases} \]
Να εξετάσετε την $f$ ως προς τη συνέχειά της στο πεδίο ορισμού της.}

\lysh
Το πεδίο ορισμού της συνάρτησης είναι το $D_f = \mathbb{R}$. Θα ελέγξουμε τη συνέχεια της $f$ στα επιμέρους διαστήματα και στο σημείο αλλαγής τύπου.
\begin{itemize}
\item Στο διάστημα $(-\infty, 0)$ η $f$ είναι συνεχής ως πηλίκο των συνεχών συναρτήσεων.
\item Στο διάστημα $(0, +\infty)$ η $f$ είναι συνεχής ως πολυωνυμική.
\item Στο σημείο $x_0 = 0$ θα ελέγξουμε αν τα πλευρικά όρια ταυτίζονται με την τιμή $f(0)$.
\begin{itemize}
    \item $f(0) = 0^2 - 0 + 1 = 1$.
    \item $\displaystyle \lim\limits_{x\to 0^-} f(x) = \lim\limits_{x\to 0^-} \dfrac{\hm{x}}{x} = 1$.
    \item $\displaystyle \lim\limits_{x\to 0^+} f(x) = \lim\limits_{x\to 0^+} (x^2 - x + 1) = 0^2 - 0 + 1 = 1$.
\end{itemize}
Επειδή $\displaystyle \lim\limits_{x\to 0^-} f(x) = \lim\limits_{x\to 0^+} f(x) = f(0) = 1$, η συνάρτηση $f$ είναι συνεχής στο σημείο $x_0=0$.
\end{itemize}
Αφού η $f$ είναι συνεχής σε κάθε εσωτερικό σημείο των διαστημάτων της και στο σημείο αλλαγής τύπου, συμπεραίνουμε ότι είναι \textbf{συνεχής σε όλο το πεδίο ορισμού της} ($\mathbb{R}$).
\subsection{Θεώρημα \en{Bolano}}
\begin{askhsh}{Θεώρημα \en{Bolzano}}
\begin{tropos}
\item \kerkissans{\textbf{Ύπαρξη λύσης εξίσωσης (Βασική μέθοδος)}}
\\\bhmata\begin{bhma}
\item Μεταφέρουμε όλους τους όρους στο $1^\text{ο}$ μέλος, ώστε η εξίσωση να πάρει τη μορφή $f(x)=0$.
\item Ορίζουμε τη συνάρτηση $f$ με τύπο το $1^\text{ο}$ μέλος και βρίσκουμε το πεδίο ορισμού της.
\item Εφαρμόζουμε το Θεώρημα \en{Bolzano} για την $f$ σε κατάλληλο κλειστό διάστημα $[\alpha, \beta]$. Δηλαδή αιτιολογούμε τη συνέχεια της $f$ στο $[\alpha, \beta]$, υπολογίζουμε τα $f(\alpha), f(\beta)$ και δείχνουμε ότι $f(\alpha)\cdot f(\beta) < 0$.
\end{bhma}

\item \kerkissans{\textbf{Ύπαρξη λύσης κλασματικής εξίσωσης}}\\
Κάνουμε απαλοιφή παρονομαστών (πολλαπλασιάζοντας με το Ε.Κ.Π.) και θεωρούμε τη νέα εξίσωση χωρίς παρονομαστές και συνεχίζουμε όπως στην $1^\text{η}$ μέθοδο. Προσοχή, στο τέλος ελέγχουμε αν η ρίζα που βρήκαμε ανήκει στο πεδίο ορισμού της αρχικής εξίσωσης.

\item \kerkissans{\textbf{Ύπαρξη σημείου $x_0$ όπου ισχύει μια σχέση (π.χ. $g(x_0)=h(x_0)$)}}\\
Θέτουμε όπου $x_0$ το $x$, μεταφέρουμε όλους τους όρους στο $1^\text{ο}$ μέλος ώστε να προκύψει $g(x)-h(x)=0$ και συνεχίζουμε όπως στην $1^\text{η}$ μέθοδο ορίζοντας ως $f(x)$ αυτή τη διαφορά.

\item \kerkissans{\textbf{Ύπαρξη ρίζας σε κλειστό διάστημα $x_0 \in [\alpha, \beta]$}}\\
Σχηματίζουμε την $f(x)$ και συνεχίζουμε όπως στην $1^\text{η}$ μέθοδο, καταλήγοντας στον υπολογισμό του $f(\alpha)\cdot f(\beta)$. Συνήθως όμως εδώ προκύπτει $f(\alpha)\cdot f(\beta) \le 0$. Διακρίνουμε τις εξής περιπτώσεις:
\begin{itemize}
    \item Αν $f(\alpha)\cdot f(\beta) < 0$, τότε υπάρχει ρίζα στο ανοιχτό $(\alpha, \beta)$, άρα και στο κλειστό $[\alpha, \beta]$.
    \item Αν $f(\alpha)\cdot f(\beta) = 0$, τότε $f(\alpha)=0$ (άρα το $\alpha$ είναι ρίζα) ή $f(\beta)=0$ (άρα το $\beta$ είναι ρίζα). Και στις δύο περιπτώσεις υπάρχει ρίζα στο κλειστό $[\alpha, \beta]$.
\end{itemize}

\item \kerkissans{\textbf{Μοναδική ρίζα}}\\
Εφαρμόζουμε το Θεώρημα \en{Bolzano} για να δείξουμε ότι υπάρχει τουλάχιστον μία ρίζα  ($1^\text{η}$ μέθοδος). Επιπλέον, εξετάζουμε τη μονοτονία της $f$ (συνήθως με τη βοήθεια της παραγώγου $f'$). Εφόσον η $f$ είναι γνησίως μονότονη, συμπεραίνουμε ότι η ρίζα είναι μοναδική.

\item \kerkissans{\textbf{Τουλάχιστον $\nu$ ρίζες}}\\
Επιλέγουμε $\nu$ κατάλληλα, ξένα μεταξύ τους, κλειστά διαστήματα και συνεχίζουμε εφαρμόζοντας την $1^\text{η}$ μέθοδο ξεχωριστά για το καθένα από αυτά. Θα προκύψει έτσι τουλάχιστον μία ρίζα σε κάθε ανοιχτό υποδιάστημα.

\item \kerkissans{\textbf{Θεώρημα \en{Bolzano} με όριο}}\\
Αν το διάστημα είναι ανοιχτό (π.χ. $(\alpha, \beta)$) και δεν γνωρίζουμε τις τιμές της συνάρτησης στα άκρα, υπολογίζουμε τα όρια $\lim\limits_{x\to \alpha^+} f(x)$ και $\lim\limits_{x\to \beta^-} f(x)$. Αν τα όρια αυτά είναι ετερόσημα (π.χ. $+\infty$ και $-\infty$), τότε υπάρχουν κατάλληλες κοντινές τιμές $x_1, x_2 \in (\alpha, \beta)$ τέτοιες ώστε $f(x_1) > 0$ και $f(x_2) < 0$ και συνεχίζουμε όπως στην $1^\text{η}$ μέθοδο στο νέο κλειστό διάστημα $[x_1, x_2]$.
\end{tropos}
\end{askhsh}

\Paradeigma{Ύπαρξη λύσης εξίσωσης (Βασική μέθοδος)}
\bmath{Να αποδείξετε ότι η εξίσωση $x^3 + 2x - 1 = 0$ έχει τουλάχιστον μία ρίζα στο διάστημα $(0, 1)$.}

\lysh
Όλοι οι όροι βρίσκονται ήδη στο $1^\text{ο}$ μέλος. Θεωρούμε τη συνάρτηση 
\[ f(x) = x^3 + 2x - 1 \]
η οποία ορίζεται στο $\mathbb{R}$. Θα εφαρμόσουμε το Θεώρημα \en{Bolzano} στο κλειστό διάστημα $[0, 1]$:
\begin{itemize}
    \item Η συνάρτηση $f$ είναι συνεχής στο διάστημα $[0, 1]$ ως πολυωνυμική.
    \item Υπολογίζουμε τις τιμές της στα άκρα του διαστήματος:
    \begin{align*}
        f(0) &= 0^3 + 2\cdot 0 - 1 = -1 < 0 \\
        f(1) &= 1^3 + 2\cdot 1 - 1 = 1 + 2 - 1 = 2 > 0
    \end{align*}
\end{itemize}
Παρατηρούμε ότι ισχύει $f(0) \cdot f(1) = (-1) \cdot 2 = -2 < 0$. 

Επομένως, σύμφωνα με το Θεώρημα \en{Bolzano}, υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $f(x_0) = 0$.
Άρα, η εξίσωση $x^3 + 2x - 1 = 0$ έχει τουλάχιστον μία ρίζα στο διάστημα $(0, 1)$.\\

\Paradeigma{Ύπαρξη λύσης κλασματικής εξίσωσης}
\bmath{Να αποδείξετε ότι η εξίσωση $\dfrac{x^2+1}{x-2} + \dfrac{x+3}{x-1} = 0$ έχει τουλάχιστον μία ρίζα στο διάστημα $(1, 2)$.}

\lysh
Η εξίσωση ορίζεται όταν $x-2 \neq 0 \Rightarrow x \neq 2$ και $x-1 \neq 0 \Rightarrow x \neq 1$.
Επομένως, το πεδίο ορισμού της είναι το $\mathbb{R} - \{1, 2\}$.
Για την απαλοιφή των παρονομαστών, πολλαπλασιάζουμε κάθε όρο με το Ε.Κ.Π. που είναι το $(x-1)(x-2)$:
\begin{gather*}
(x-1)(x-2) \dfrac{x^2+1}{x-2} + (x-1)(x-2) \dfrac{x+3}{x-1} = 0 \Rightarrow \\
(x-1)(x^2+1) + (x-2)(x+3) = 0
\end{gather*}
Θεωρούμε τη νέα συνάρτηση 
\[ f(x) = (x-1)(x^2+1) + (x-2)(x+3) \]
η οποία ορίζεται σε όλο το $\mathbb{R}$. Θα εφαρμόσουμε το Θεώρημα \en{Bolzano} στο κλειστό διάστημα $[1, 2]$:
\begin{itemize}
    \item Η συνάρτηση $f$ είναι συνεχής στο διάστημα $[1, 2]$ ως πολυωνυμική.
    \item Υπολογίζουμε τις τιμές της στα άκρα:
    \begin{align*}
        f(1) &= (1-1)(1^2+1) + (1-2)(1+3) = 0 \cdot 2 + (-1) \cdot 4 = -4 < 0 \\
        f(2) &= (2-1)(2^2+1) + (2-2)(2+3) = 1 \cdot 5 + 0 \cdot 5 = 5 > 0
    \end{align*}
\end{itemize}
Εφόσον $f(1) \cdot f(2) = (-4) \cdot 5 = -20 < 0$, σύμφωνα με το Θεώρημα \en{Bolzano}, υπάρχει τουλάχιστον ένα $x_0 \in (1, 2)$ τέτοιο ώστε $f(x_0) = 0$.\\

\Paradeigma{Ύπαρξη σημείου όπου ισχύει σχέση}
\bmath{Να αποδείξετε ότι οι γραφικές παραστάσεις των συναρτήσεων $g(x) = e^x$ και $h(x) = -x$ τέμνονται σε ένα τουλάχιστον σημείο με τετμημένη $x_0 \in (-1, 0)$.}

\lysh
Το σημείο τομής των δύο γραφικών παραστάσεων, εφόσον υπάρχει, θα έχει τετμημένη $x_0$ τη λύση της εξίσωσης:
\[ g(x_0) = h(x_0) \Rightarrow e^{x_0} = -x_0 \Rightarrow e^{x_0} + x_0 = 0 \]
Ορίζουμε τη συνάρτηση
\[ f(x) = e^x + x \]
η οποία ορίζεται σε όλο το $\mathbb{R}$. Θα εφαρμόσουμε το Θεώρημα \en{Bolzano} για την $f$ στο κλειστό διάστημα $[-1, 0]$:
\begin{itemize}
    \item Η συνάρτηση $f$ είναι συνεχής στο διάστημα $[-1, 0]$ ως άθροισμα συνεχών συναρτήσεων.
    \item Επίσης
    \begin{align*}
        f(-1) &= e^{-1} + (-1) = \dfrac{1}{e} - 1 = \dfrac{1-e}{e} < 0 \quad (\text{αφού } e > 1) \\
        f(0) &= e^0 + 0 = 1 > 0
    \end{align*}
\end{itemize}
άρα $f(-1) \cdot f(0) < 0$. Επομένως, σύμφωνα με το Θεώρημα \en{Bolzano}, υπάρχει τουλάχιστον ένα $x_0 \in (-1, 0)$ τέτοιο ώστε $f(x_0) = 0\Leftrightarrow g(x_0) = h(x_0)$. 
Άρα οι γραφικές παραστάσεις των συναρτήσεων τέμνονται σε τουλάχιστον ένα σημείο με τετμημένη στο διάστημα $(-1, 0)$.\\

\Paradeigma{Ύπαρξη ρίζας σε κλειστό διάστημα}
\bmath{Δίνεται η συνεχής συνάρτηση $f: [1, 3] \to \mathbb{R}$ για την οποία ισχύει $f(1) + 2f(3) = 0$. Να αποδείξετε ότι η εξίσωση $f(x) = 0$ έχει τουλάχιστον μία ρίζα στο κλειστό διάστημα $[1, 3]$.}

\lysh
Η $f$ είναι συνεχής στο $[a,\beta]$. Από τη δοσμένη σχέση έχουμε $f(1) = -2f(3)$ άρα προκύπτει:
\[ f(1) \cdot f(3) = -2f(3) \cdot f(3) = -2f^2(3)\le 0 \]
Διακρίνουμε τις εξής δύο περιπτώσεις:
\begin{itemize}
    \item \textbf{Περίπτωση 1}: Αν ${f(1) \cdot f(3) < 0}$ τότε σύμφωνα με το Θ. \en{Bolzano} υπάρχει τουλάχιστον ένα $x_0 \in (1, 3)$ τέτοιο ώστε $f(x_0) = 0$.
    
    \item \textbf{Περίπτωση 2}: Αν ${f(1) \cdot f(3) = 0}$ τότε ισχύει είτε $f(1) = 0$, είτε $f(3) = 0$.
\end{itemize}
Οπότε, σε κάθε περίπτωση η εξίσωση $f(x) = 0$ έχει τουλάχιστον μία ρίζα στο κλειστό διάστημα $[1, 3]$.\\

\Paradeigma{Μοναδική ρίζα}
\bmath{Να αποδείξετε ότι η εξίσωση $e^x + x - 2 = 0$ έχει ακριβώς μία ρίζα στο διάστημα $(0, 1)$.}

\lysh
Θεωρούμε τη συνάρτηση $f(x) = e^x + x - 2$, η οποία ορίζεται στο $\mathbb{R}$.
\\\\\textbf{Ύπαρξη ρίζας:} Θα εφαρμόσουμε το Θεώρημα \en{Bolzano} για την $f$ στο κλειστό διάστημα $[0, 1]$:
\begin{itemize}
    \item Η συνάρτηση $f$ είναι συνεχής στο διάστημα $[0, 1]$ ως άθροισμα συνεχών συναρτήσεων.
    \item Υπολογίζουμε τις τιμές της στα άκρα του διαστήματος:
    \begin{align*}
        f(0) &= e^0 + 0 - 2 = 1 - 2 = -1 < 0 \\
        f(1) &= e^1 + 1 - 2 = e - 1 > 0 \quad (\text{αφού } e \approx 2.71 > 1)
    \end{align*}
\end{itemize}
Παρατηρούμε ότι ισχύει $f(0) \cdot f(1) < 0$. 
Επομένως, σύμφωνα με το Θεώρημα \en{Bolzano}, η εξίσωση $f(x) = 0$ έχει τουλάχιστον μία ρίζα $x_0 \in (0, 1)$.

\textbf{Μοναδικότητα ρίζας:} Η συνάρτηση $f$ είναι παραγωγίσιμη στο $\mathbb{R}$ με παράγωγο:
\[ f'(x) = e^x + 1 \]
Επειδή $e^x > 0$ για κάθε $x \in \mathbb{R}$, ισχύει $f'(x) > 0$ για κάθε $x \in \mathbb{R}$. 
Άρα, η συνάρτηση $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, συνεπώς είναι και $1-1$. 
Η ιδιότητα αυτή εξασφαλίζει ότι η ρίζα $x_0$ που βρήκαμε στο διάστημα $(0, 1)$ είναι μοναδική.\\

\Paradeigma{Τουλάχιστον $\nu$ ρίζες (π.χ. 2 ρίζες)}
\bmath{Να αποδείξετε ότι η εξίσωση $x^3 - 3x + 1 = 0$ έχει τουλάχιστον δύο ρίζες στο διάστημα $(-2, 1)$.}

\lysh
Θεωρούμε τη συνάρτηση $f(x) = x^3 - 3x + 1$, η οποία ορίζεται στο $\mathbb{R}$. 
Για να δείξουμε ότι η εξίσωση έχει τουλάχιστον δύο ρίζες, θα χωρίσουμε το αρχικό διάστημα σε δύο κατάλληλα, ξένα μεταξύ τους υποδιαστήματα και θα εφαρμόσουμε το Θεώρημα \en{Bolzano} στο καθένα από αυτά.

Επιλέγουμε το εσωτερικό σημείο $0 \in (-2, 1)$ και σχηματίζουμε τα διαστήματα $[-2, 0]$ και $[0, 1]$.
Η συνάρτηση $f$ είναι συνεχής σε όλο το $\mathbb{R}$ ως πολυωνυμική, άρα είναι συνεχής και στα δύο αυτά κλειστά διαστήματα.
Υπολογίζουμε τις τιμές της στα άκρα:
\begin{align*}
    f(-2) &= (-2)^3 - 3(-2) + 1 = -8 + 6 + 1 = -1 < 0 \\
    f(0) &= 0^3 - 3\cdot 0 + 1 = 1 > 0 \\
    f(1) &= 1^3 - 3\cdot 1 + 1 = 1 - 3 + 1 = -1 < 0
\end{align*}
Εφαρμόζουμε το Θεώρημα \en{Bolzano}:
\begin{itemize}
    \item \textbf{Στο διάστημα $\mathbf{[-2, 0]}$:} Ισχύει $f(-2) \cdot f(0) = (-1) \cdot 1 = -1 < 0$. Επομένως, υπάρχει τουλάχιστον μία ρίζα $x_1 \in (-2, 0)$.
    \item \textbf{Στο διάστημα $\mathbf{[0, 1]}$:} Ισχύει $f(0) \cdot f(1) = 1 \cdot (-1) = -1 < 0$. Ομοίως, υπάρχει τουλάχιστον μία ρίζα $x_2 \in (0, 1)$.
\end{itemize}
Επειδή τα ανοιχτά διαστήματα $(-2, 0)$ και $(0, 1)$ είναι προφανώς ξένα μεταξύ τους, οι ρίζες $x_1$ και $x_2$ είναι διακριτές ($x_1 \neq x_2$). 
Άρα, η εξίσωση έχει τουλάχιστον δύο ρίζες στο αρχικό διάστημα $(-2, 1)$.\\

\Paradeigma{Θεώρημα \en{Bolzano} με όριο}
\bmath{Να αποδείξετε ότι η εξίσωση $e^x + \ln x = 0$ έχει τουλάχιστον μία ρίζα στο διάστημα $(0, 1)$.}

\lysh
Θεωρούμε τη συνάρτηση $f(x) = e^x + \ln x$, η οποία έχει πεδίο ορισμού το $(0, +\infty)$. 
Θέλουμε να βρούμε ρίζα στο ανοιχτό διάστημα $(0, 1)$. Ωστόσο, η συνάρτηση δεν ορίζεται στο $x=0$, άρα δεν μπορούμε να εφαρμόσουμε άμεσα το Θεώρημα \en{Bolzano} στο κλειστό διάστημα $[0, 1]$.
Για να ξεπεράσουμε το πρόβλημα, θα αναζητήσουμε ένα σημείο «πολύ κοντά» στο $0$ από τα δεξιά:
\begin{itemize}
    \item Η συνάρτηση $f$ είναι συνεχής στο $(0, 1]$ ως άθροισμα συνεχών συναρτήσεων.
    \item Υπολογίζουμε την τιμή της στο δεξί άκρο: 
    \[ f(1) = e^1 + \ln 1 = e > 0 \]
    \item Για το αριστερό άκρο, υπολογίζουμε το πλευρικό όριο στο 0:
    \[ \lim\limits_{x\to 0^+} f(x) = \lim\limits_{x\to 0^+} (e^x + \ln x) = 1 + (-\infty) = -\infty \]
\end{itemize}
Αφού $\lim\limits_{x\to 0^+} f(x) = -\infty$, γνωρίζουμε από τον ορισμό του ορίου ότι υπάρχουν τιμές του $x$ όσο θέλουμε κοντά στο 0 (από τα δεξιά) για τις οποίες η συνάρτηση παίρνει αρνητικές τιμές. Δηλαδή, μπορούμε να βρούμε έναν πολύ μικρό θετικό αριθμό $\alpha \in (0, 1)$ τέτοιο ώστε να ισχύει $f(\alpha) < 0$.

Τώρα, εφαρμόζουμε το Θεώρημα \en{Bolzano} για την $f$ στο νέο κλειστό διάστημα $[\alpha, 1]$:
\begin{itemize}
    \item Η $f$ είναι συνεχής στο $[\alpha, 1]$.
    \item Ισχύει $f(\alpha) \cdot f(1) < 0$ (επειδή $f(\alpha) < 0$ και $f(1) = e > 0$).
\end{itemize}
Επομένως, υπάρχει τουλάχιστον ένα $x_0 \in (\alpha, 1)$ τέτοιο ώστε $f(x_0) = 0$ άρα η εξίσωση έχει πράγματι τουλάχιστον μία ρίζα στο $(a,1)\subseteq(0, 1)$.

\begin{askhsh}{Συνδυασμός Θ.Μ.Ε.Τ. και Θ.Ε.Τ.}
Για να δείξουμε ότι υπάρχει σημείο $x_0 \in [\alpha, \beta]$ στο οποίο η συνάρτηση $f$ παίρνει μια τιμή που είναι «σταθμισμένος μέσος» (π.χ. $f(x_0) = \dfrac{f(x_1)+f(x_2)}{2}$) άλλων τιμών της:

\bhmata
\begin{bhma}
\item Εφαρμόζουμε το \textbf{Θεώρημα Μέγιστης και Ελάχιστης Τιμής (Θ.Μ.Ε.Τ.)}: Αφού η $f$ είναι συνεχής στο κλειστό $[\alpha, \beta]$, έχει ελάχιστη τιμή $m$ και μέγιστη τιμή $M$. Άρα $m \le f(x) \le M$ για κάθε $x \in [\alpha, \beta]$.
\item Γράφουμε αυτή την ανισότητα για τα συγκεκριμένα $x_i$ ($x_1, x_2, \ldots$) που εμφανίζονται.
\item Πολλαπλασιάζουμε τις ανισότητες με τους αντίστοιχους (θετικούς) συντελεστές και τις προσθέτουμε κατά μέλη για να κατασκευάσουμε την παράσταση $y_0$.
\item Καταλήγουμε στη σχέση $m \le y_0 \le M$.
\item Εφόσον $m \le y_0 \le M$, από το \textbf{Θεώρημα Ενδιάμεσων Τιμών} (ή το σύνολο τιμών της συνεχούς $f$ που είναι το $[m, M]$), συμπεραίνουμε ότι υπάρχει τουλάχιστον ένα $x_0 \in [\alpha, \beta]$ τέτοιο ώστε $f(x_0) = y_0$.
\end{bhma}
\end{askhsh}

\Paradeigma{Συνδυασμός Θ.Μ.Ε.Τ. και Θ.Ε.Τ.}
\bmath{Έστω $f: [\alpha, \beta] \to \mathbb{R}$ μια συνεχής συνάρτηση και $x_1, x_2 \in [\alpha, \beta]$. Να αποδείξετε ότι υπάρχει τουλάχιστον ένα $x_0 \in [\alpha, \beta]$ τέτοιο ώστε:
\[ f(x_0) = \dfrac{f(x_1) + 2f(x_2)}{3} \]}

\lysh
Επειδή η συνάρτηση $f$ είναι συνεχής στο κλειστό διάστημα $[\alpha, \beta]$, σύμφωνα με το \textbf{Θεώρημα Μέγιστης και Ελάχιστης Τιμής (Θ.Μ.Ε.Τ.)}, η $f$ παρουσιάζει στο $[\alpha, \beta]$ ελάχιστο $m$ και μέγιστο $M$. Επομένως, για κάθε $x \in [\alpha, \beta]$ ισχύει:
\[ m \le f(x) \le M \]
Εφαρμόζουμε την ανισότητα για τα σημεία $x_1, x_2 \in [\alpha, \beta]$:
\begin{align*}
m \le f(x_1) &\le M \\
m \le f(x_2) &\le M \Rightarrow 2m \le 2f(x_2) \le 2M
\end{align*}
Προσθέτοντας κατά μέλη τις παραπάνω ανισότητες έχουμε:
\[ 3m \le f(x_1) + 2f(x_2) \le 3M \Rightarrow m \le \dfrac{f(x_1) + 2f(x_2)}{3} \le M \]
Ο αριθμός $\gamma = \dfrac{f(x_1) + 2f(x_2)}{3}$ ανήκει στο διάστημα $[m, M]$. Αφού η $f$ είναι συνεχής, το σύνολο τιμών της είναι το $[m, M]$ (συνέπεια του \textbf{Θεωρήματος Ενδιάμεσων Τιμών}). Άρα, υπάρχει τουλάχιστον ένα $x_0 \in [\alpha, \beta]$ τέτοιο ώστε:
\[ f(x_0) = \gamma \Rightarrow f(x_0) = \dfrac{f(x_1) + 2f(x_2)}{3} \]\\

\begin{askhsh}{Εύρεση προσήμου συνάρτησης}
Για να βρούμε το πρόσημο μιας συνεχούς συνάρτησης $f$ ακολουθούμε τα εξής βήματα:

\bhmata
\begin{bhma}
\item Βρίσκουμε το πεδίο ορισμού της συνάρτησης $f$.
\item Λύνουμε την εξίσωση $f(x)=0$ για να βρούμε τις ρίζες της.
\item Οι ρίζες (μαζί με τα σημεία ασυνέχειας ή τα άκρα των διαστημάτων του πεδίου ορισμού) χωρίζουν το πεδίο ορισμού σε ανοιχτά υποδιαστήματα. Η $f$ διατηρεί σταθερό πρόσημο σε καθένα από αυτά τα διαστήματα.
\item Επιλέγουμε έναν οποιονδήποτε αριθμό σε κάθε διάστημα, υπολογίζουμε την τιμή της $f$ και βρίσκουμε το πρόσημό της. Το πρόσημο αυτό θα είναι και το πρόσημο της $f$ σε ολόκληρο το αντίστοιχο διάστημα.
\item Κατασκευάζουμε τον τελικό πίνακα προσήμων.
\end{bhma}
\end{askhsh}

\Paradeigma{Εύρεση προσήμου συνάρτησης}
\bmath{Να βρείτε το πρόσημο της συνάρτησης $f(x) = \ln^2 x - \ln x$ για τις διάφορες τιμές του $x$.}

\lysh
Η συνάρτηση $f$ ορίζεται όταν $x > 0$. Επομένως, το πεδίο ορισμού της είναι το $D_f = (0, +\infty)$. Επιπλέον, η $f$ είναι συνεχής στο πεδίο ορισμού της ως πράξεις συνεχών συναρτήσεων.

Λύνουμε την εξίσωση $f(x) = 0$:
\begin{align*}
    \ln^2 x - \ln x &= 0 \\
    \ln x (\ln x - 1) &= 0
\end{align*}
Έχουμε λοιπόν δύο ρίζες:
\begin{itemize}
    \item $\ln x = 0 \Leftrightarrow x = 1$
    \item $\ln x = 1 \Leftrightarrow x = e$
\end{itemize}

Οι ρίζες $x=1$ και $x=e$ χωρίζουν το πεδίο ορισμού $(0, +\infty)$ στα εξής ανοιχτά διαστήματα: $(0, 1)$, $(1, e)$ και $(e, +\infty)$. Επειδή η $f$ είναι συνεχής και δεν μηδενίζεται μέσα σε αυτά τα διαστήματα, διατηρεί σταθερό πρόσημο στο καθένα από αυτά.

Επιλέγουμε έναν τυχαίο (βολικό) αριθμό από κάθε διάστημα για να βρούμε το πρόσημο:
\begin{itemize}
    \item Για το διάστημα $(0, 1)$, επιλέγουμε το $x = \dfrac{1}{e}$:
    \[ f\left(\dfrac{1}{e}\right) = \ln^2\left(e^{-1}\right) - \ln\left(e^{-1}\right) = (-1)^2 - (-1) = 1 + 1 = 2 > 0 \]
    Άρα $f(x) > 0$ για κάθε $x \in (0, 1)$.
    
    \item Για το διάστημα $(1, e)$, επιλέγουμε το $x = \sqrt{e}$:
    \[ f(\sqrt{e}) = \ln^2\left(e^{1/2}\right) - \ln\left(e^{1/2}\right) = \left(\dfrac{1}{2}\right)^2 - \dfrac{1}{2} = \dfrac{1}{4} - \dfrac{1}{2} = -\dfrac{1}{4} < 0 \]
    Άρα $f(x) < 0$ για κάθε $x \in (1, e)$.
    
    \item Για το διάστημα $(e, +\infty)$, επιλέγουμε το $x = e^2$:
    \[ f(e^2) = \ln^2\left(e^2\right) - \ln\left(e^2\right) = 2^2 - 2 = 4 - 2 = 2 > 0 \]
    Άρα $f(x) > 0$ για κάθε $x \in (e, +\infty)$.
\end{itemize}

\begin{askhsh}{Γνωρίζω την $f^2$ και ψάχνω την $f$}
Αν δίνεται μια σχέση της μορφής $f^2(x) = g(x)$ (ή γενικότερα μια σχέση από την οποία μπορούμε να λύσουμε ως προς $f^2(x)$) και γνωρίζουμε ότι η συνάρτηση $f$ είναι συνεχής, για να βρούμε τον τύπο της $f$ ακολουθούμε τα εξής βήματα:

\bhmata
\begin{bhma}
\item Λύνουμε τη δοσμένη σχέση ως εξής:
\[f^2(x)=g(x)\Leftrightarrow \sqrt{f^2(x)}=\sqrt{g(x)}\Leftrightarrow |f(x)|=\sqrt{g(x)}\]
\item Λύνουμε την εξίσωση $f(x) = 0 \Leftrightarrow f^2(x) = 0 \Leftrightarrow g(x) = 0$ για να βρούμε τις ρίζες της συνάρτησης.
\item Οι ρίζες αυτές χωρίζουν το πεδίο ορισμού σε ανοιχτά υποδιαστήματα. Σε καθένα από αυτά, η $f$ δεν μηδενίζεται. Επειδή είναι και συνεχής, \textbf{διατηρεί σταθερό πρόσημο} σε κάθε τέτοιο υποδιάστημα:
\[ f(x) = \sqrt{g(x)} \quad \text{για κάθε } x \in \Delta \quad \text{ή} \quad f(x) = -\sqrt{g(x)} \quad \text{για κάθε } x \in \Delta \]
\item Συνδυάζουμε τους πιθανούς κλάδους (θετικός ή αρνητικός) για καθένα από τα διαστήματα. Προκύπτουν έτσι όλες οι δυνατές συναρτήσεις $f$.
\item Αν δίνεται κάποια αρχική συνθήκη $f(x_0) = y_0$, υπολογίζουμε το πρόσημο της $f$ σε εκείνο το διάστημα και βρίσκουμε το ζητούμενο τύπο της $f$.
\end{bhma}
\end{askhsh}

\Paradeigma{Εύρεση τύπων από το τετράγωνο της συνάρτησης}
\bmath{Να βρείτε όλες τις συνεχείς συναρτήσεις $f: \mathbb{R} \to \mathbb{R}$ για τις οποίες ισχύει $f^2(x) = x^2$ για κάθε $x \in \mathbb{R}$.}

\lysh
Από τη δοσμένη σχέση έχουμε:
\[ f^2(x) = x^2 \Leftrightarrow |f(x)| = \sqrt{x^2} \Leftrightarrow |f(x)| = |x| \]
Βρίσκουμε τις ρίζες της συνάρτησης $f$:
\[ f(x) = 0 \Leftrightarrow f^2(x) = 0 \Leftrightarrow x^2 = 0 \Leftrightarrow x = 0 \]

Η ρίζα $x=0$ χωρίζει το πεδίο ορισμού $\mathbb{R}$ στα ανοιχτά διαστήματα $(-\infty, 0)$ και $(0, +\infty)$. Η συνάρτηση $f$ είναι συνεχής στο $\mathbb{R}$, άρα και στα διαστήματα αυτά, και επιπλέον δεν μηδενίζεται σε αυτά. Συνεπώς διατηρεί σταθερό πρόσημο σε καθένα από τα διαστήματα $(-\infty, 0)$ και $(0, +\infty)$. 
\begin{itemize}
    \item Στο διάστημα $(-\infty, 0)$ ισχύει $x < 0$, άρα $|x| = -x$. Οπότε $|f(x)| = -x$.
    Επειδή η $f$ διατηρεί σταθερό πρόσημο, έχουμε δύο ενδεχόμενα:
    \[ f(x) = -x \quad \text{ή} \quad f(x) = -(-x) = x \quad \text{για κάθε } x \in (-\infty, 0) \]
    \item Στο διάστημα $(0, +\infty)$ ισχύει $x > 0$, άρα $|x| = x$. Οπότε $|f(x)| = x$.
    Επειδή η $f$ διατηρεί σταθερό πρόσημο, έχουμε επίσης δύο ενδεχόμενα:
    \[ f(x) = x \quad \text{ή} \quad f(x) = -x \quad \text{για κάθε } x \in (0, +\infty) \]
\end{itemize}

Γνωρίζουμε επίσης ότι στο σημείο μηδενισμού ισχύει $f(0) = 0$. 
Συνδυάζοντας όλες τις παραπάνω περιπτώσεις για τους κλάδους του $(-\infty, 0)$ και του $(0, +\infty)$ μαζί με την τιμή στο $f(0)=0$, προκύπτουν $\mathbf{4}$ πιθανές συναρτήσεις:
\begin{multicols}{2}
\begin{enumerate}
    \item $f_1(x) = \begin{cases} -x, & x \le 0 \\ x, & x > 0 \end{cases} \quad \Rightarrow \quad f_1(x) = |x|$
    \item $f_2(x) = \begin{cases} x, & x \le 0 \\ -x, & x > 0 \end{cases} \quad \Rightarrow \quad f_2(x) = -|x|$
    \item $f_3(x) = \begin{cases} x, & x \le 0 \\ x, & x > 0 \end{cases} \quad \Rightarrow \quad f_3(x) = x$
    \item $f_4(x) = \begin{cases} -x, & x \le 0 \\ -x, & x > 0 \end{cases} \quad \Rightarrow \quad f_4(x) = -x$
\end{enumerate}
\end{multicols}

\Paradeigma{Εύρεση της $f$ από την $f^2$ με γνωστή τιμή}
\bmath{Δίνεται συνεχής συνάρτηση $f:\mathbb{R}\to\mathbb{R}$ με $f(0)=3$ για την οποία ισχύει
\[f^2(x)-x^2=9\ ,\ \text{για κάθε }x\in\mathbb{R}\]
Να βρείτε τον τύπο της $f$.}

\lysh
Από τη δοσμένη σχέση παίρνουμε για κάθε $x\in\mathbb{R}$:
\[f^2(x)-x^2=9\Leftrightarrow f^2(x)=x^2+9\Leftrightarrow\sqrt{f^2(x)}=\sqrt{x^2+9}\Leftrightarrow |f(x)|=\sqrt{x^2+9}\]
Η $f$ μηδενίζεται όταν:
\[ f(x) = 0 \Leftrightarrow f^2(x) = 0 \Leftrightarrow x^2 + 9 = 0 \]
Όμως $x^2 + 9 \geq 9 > 0$ για κάθε $x \in \mathbb{R}$, άρα η εξίσωση δεν έχει λύση. Η συνάρτηση $f$ \textbf{δεν μηδενίζεται πουθενά}.

Επειδή η $f$ είναι συνεχής στο $\mathbb{R}$ και δεν μηδενίζεται, διατηρεί σταθερό πρόσημο σε \textbf{όλο} το $\mathbb{R}$.  Έτσι:
\[ f(x) = \sqrt{x^2+9} \quad \text{για κάθε } x \in \mathbb{R} \quad \text{ή} \quad f(x) = -\sqrt{x^2+9} \quad \text{για κάθε } x \in \mathbb{R} \]

Από τη δοσμένη αρχική συνθήκη $f(0) = 3 > 0$, η $f$ έχει θετικό πρόσημο. Άρα απορρίπτουμε τον αρνητικό κλάδο.\\

Επομένως, ο τύπος της $f$ είναι:
\[ f(x) = \sqrt{x^2+9} \quad ,\quad D_f = \mathbb{R} \]

\subsection{Εύρεση παραμέτρων}\label{paragrafos:eyresh_parametrwn_oria}
Η γενική μέθοδος για την εύρεση μιας παραμέτρου είναι να κατασκευάσουμε μια εξίσωση ή ανίσωση που να την περιέχει, ώστε λύνοντάς την να την προσδιορίσουμε. Κάποια συνθήκη της υπόθεσης είναι αυτή που θα μας οδηγήσει σ' αυτή την εξίσωση-ανίσωση. 
\begin{center}
\begin{mytblr}{m{7.5cm}m{7.5cm}}
\textbf{Συνθήκη} & \textbf{Εξίσωση - Ανίσωση} \\
Το σημείο $ A(a,\beta) $  ανήκει στη $ C_f $ & $ f(a)=\beta $ \\
Γνωστό όριο που περιέχει παραμέτρους $ a,\beta\ldots $ & Βοηθητική συνάρτηση \\
Η $ f $ είναι συνεχής σε σημείο $ x_0\in D_f $ & $ \lim\limits_{x\to x_0}f(x)=f(x_0) $ \\
\end{mytblr}
\end{center}

\Paradeigma{Εύρεση παραμέτρων}
\bmath{Δίνεται η συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ με τύπο:
\[ f(x) = \begin{cases} x^2 + ax + \beta, & x \le 1 \\ g(x), & x > 1 \end{cases} \]
όπου $a, \beta \in \mathbb{R}$ και $g: (1, +\infty) \to \mathbb{R}$ μια συνάρτηση. Αν γνωρίζουμε ότι:
\begin{itemize}
    \item Το σημείο $A(-1, 3)$ ανήκει στη γραφική παράσταση της $f$.
    \item Ισχύει $\lim\limits_{x\to 1^+} \dfrac{g(x) - 2x}{x-1} = 4$.
    \item Η συνάρτηση $f$ είναι συνεχής στο $x_0 = 1$.
\end{itemize}
Να βρείτε τις τιμές των παραμέτρων $a$ και $\beta$.
}

\lysh
Θα αξιοποιήσουμε ένα-ένα τα δεδομένα κατασκευάζοντας εξισώσεις, όπως αναφέρει η μεθοδολογία:
\begin{itemize}
\item \textbf{1ο Δεδομένο (Το σημείο $A(-1, 3)$ ανήκει στη $C_f$):} \\
Το σημείο $A(-1, 3)$ ανήκει στη $C_f$ άρα $f(-1) = 3$. Από τον πρώτο κλάδο της $f$:
\[ f(-1) = 3 \Rightarrow (-1)^2 + a(-1) + \beta = 3 \Rightarrow 1 - a + \beta = 3 \Rightarrow \beta - a = 2 \quad (1) \]

\item \textbf{2ο Δεδομένο (Γνωστό όριο - Βοηθητική συνάρτηση):} \\
Έχουμε το όριο $\lim\limits_{x\to 1^+} \dfrac{g(x) - 2x}{x-1} = 4$. Θέτουμε συνάρτηση:
\[ h(x) = \dfrac{g(x) - 2x}{x-1} \quad \text{με} \quad \lim\limits_{x\to 1^+} h(x) = 4 \]
Λύνουμε ως προς $g(x)$:
\[ g(x) - 2x = (x-1)h(x) \Rightarrow g(x) = (x-1)h(x) + 2x \]
Τώρα μπορούμε να βρούμε το δεξιό πλευρικό όριο της $f$ στο $1$:
\[ \lim\limits_{x\to 1^+} f(x) = \lim\limits_{x\to 1^+} g(x) = \lim\limits_{x\to 1^+} \left[ (x-1)h(x) + 2x \right] = 0 \cdot 4 + 2(1) = 2 \]

\item \textbf{3ο Δεδομένο (Η $f$ είναι συνεχής στο $x_0 = 1$):} \\
Για να είναι η $f$ συνεχής στο $x_0=1$, πρέπει να ισχύει:
\[ \lim\limits_{x\to 1^-} f(x) = \lim\limits_{x\to 1^+} f(x) = f(1) \]
Βρίσκουμε το αριστερό πλευρικό όριο και την τιμή της $f$ στο 1:
\[ \lim\limits_{x\to 1^-} f(x) = \lim\limits_{x\to 1^-} (x^2 + ax + \beta) = 1 + a + \beta \]
\[ f(1) = 1^2 + a(1) + \beta = 1 + a + \beta \]
Οπότε από την εξίσωση της συνέχειας προκύπτει:
\[ 1 + a + \beta = 2 \Rightarrow a + \beta = 1 \quad (2) \]
\end{itemize}

Από το σύστημα των εξισώσεων (1) και (2) προκύπτει:
\[ \begin{cases} \beta - a = 2 \\ \beta + a = 1 \end{cases} \]
Προσθέτοντας κατά μέλη τις δύο εξισώσεις έχουμε: 
\[ 2\beta = 3 \Rightarrow \beta = \dfrac{3}{2} \]
Αντικαθιστώντας στη (2): 
\[ a + \dfrac{3}{2} = 1 \Rightarrow a = 1 - \dfrac{3}{2} \Rightarrow a = -\dfrac{1}{2} \]

Άρα, οι ζητούμενες τιμές των παραμέτρων είναι $a = -\dfrac{1}{2}$ και $\beta = \dfrac{3}{2}$.
