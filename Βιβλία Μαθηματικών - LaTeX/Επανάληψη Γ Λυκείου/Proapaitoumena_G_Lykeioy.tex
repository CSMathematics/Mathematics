\section{Ιδιότητες δυνάμεων}

Αν $a, \beta \in \mathbb{R}$ και $\mu, \nu \in \mathbb{Z}$, τότε ισχύουν οι παρακάτω ιδιότητες (εφόσον οι δυνάμεις ορίζονται):
\begin{multicols}{2}
\begin{itemize}[label=\faEdit]
    \item $a^\mu \cdot a^\nu = a^{\mu+\nu}$
    \item $\dfrac{a^\mu}{a^\nu} = a^{\mu-\nu}$, με $a \neq 0$
    \item $a^\mu \cdot \beta^\mu = (a \cdot \beta)^\mu$
    \item $\dfrac{a^\mu}{\beta^\mu} = \left(\dfrac{a}{\beta}\right)^\mu$, με $\beta \neq 0$
    \item $(a^\mu)^\nu = a^{\mu \cdot \nu}$
    \item $a^0 = 1$, με $a \neq 0$
    \item $a^{-\nu} = \dfrac{1}{a^\nu}$, με $a \neq 0$
    \item $\left(\dfrac{a}{\beta}\right)^{-\nu} = \left(\dfrac{\beta}{a}\right)^\nu$, με $a, \beta \neq 0$
\end{itemize}
\end{multicols}

\begin{itemize}
    \item Για να απλοποιήσουμε παραστάσεις με δυνάμεις, προσπαθούμε να δημιουργήσουμε κοινές βάσεις ή κοινούς εκθέτες, αναλύοντας τους αριθμούς σε γινόμενο πρώτων παραγόντων.
    \item Όταν έχουμε αρνητικούς εκθέτες, αντιστρέφουμε τη βάση.
    \item Προσοχή στο πρόσημο όταν η βάση είναι αρνητική και υψώνεται σε δύναμη: $(-a)^{2\nu} = a^{2\nu}$ (άρτιος εκθέτης δίνει θετικό αποτέλεσμα), ενώ $(-a)^{2\nu+1} = -a^{2\nu+1}$ (περιττός εκθέτης κρατάει το μείον).
\end{itemize}


\Paradeigma{Απλοποίηση παράστασης δυνάμεων}
\bmath{Να απλοποιηθεί η παράσταση: $A = \dfrac{(x^2 y^3)^2 \cdot (x^{-1} y^2)^{-3}}{x^5 y^4}$.}\\
\lysh
\[ A = \dfrac{(x^2 y^3)^2 \cdot (x^{-1} y^2)^{-3}}{x^5 y^4} = \dfrac{x^{2\cdot2} y^{3\cdot2} \cdot x^{(-1)\cdot(-3)} y^{2\cdot(-3)}}{x^5 y^4} \]
\[ A = \dfrac{x^4 y^6 \cdot x^3 y^{-6}}{x^5 y^4} = \dfrac{x^{4+3} y^{6-6}}{x^5 y^4} = \dfrac{x^7 y^0}{x^5 y^4} = \dfrac{x^7 \cdot 1}{x^5 y^4} = x^{7-5} y^{-4} = \dfrac{x^2}{y^4} \]

\section{Βασικές ταυτότητες}

Οι κυριότερες αλγεβρικές ταυτότητες που θα χρησιμοποιήσουμε είναι:
\begin{multicols}{2}
\begin{enumerate}[label=\arabic*.]
\item \kerkissans{Τετράγωνο αθροίσματος:}\\ $(a+\beta)^2=a^2+2a\beta+\beta^2$
\item \kerkissans{Τετράγωνο διαφοράς:}\\ $(a-\beta)^2=a^2-2a\beta+\beta^2$
\item \kerkissans{Κύβος αθροίσματος:}\\ $(a+\beta)^3=a^3+3a^2\beta+3a\beta^2+\beta^3$
\item \kerkissans{Κύβος διαφοράς:}\\ $(a-\beta)^3=a^3-3a^2\beta+3a\beta^2-\beta^3$
\item \kerkissans{Γινόμενο αθροίσματος επί διαφορά:}\\ $(a+\beta)(a-\beta)=a^2-\beta^2$
\item \kerkissans{Άθροισμα κύβων:}\\ $(a+\beta)(a^2-a\beta+\beta^2)=a^3+\beta^3$
\item \kerkissans{Διαφορά κύβων:}\\ $(a-\beta)(a^2+a\beta+\beta^2)=a^3-\beta^3$
\end{enumerate}
\end{multicols}

\begin{itemize}
    \item Η ανάπτυξη ταυτοτήτων χρησιμεύει για τον υπολογισμό και την απλοποίηση διαφόρων αλγεβρικών παραστάσεων.
    \item Η διαφορά τετραγώνων είναι από τις πιο χρήσιμες ταυτότητες στη Γ Λυκείου, ειδικά σε διαδικασίες όπως ο υπολογισμός ορίων με συζυγή παράσταση.
\end{itemize}


\Paradeigma{Χρήση ταυτοτήτων για απλοποίηση}
\bmath{Να απλοποιήσετε την παράσταση: $B = (\sqrt{x} - 1)(\sqrt{x} + 1) - (\sqrt{x} - 2)^2$.}\\
\lysh
Χρησιμοποιώντας τη διαφορά τετραγώνων για τον πρώτο όρο και το τετράγωνο διαφοράς για τον δεύτερο:
\[ B = (\sqrt{x})^2 - 1^2 - \left( (\sqrt{x})^2 - 2\cdot\sqrt{x}\cdot 2 + 2^2 \right) \]
\[ B = x - 1 - (x - 4\sqrt{x} + 4) \]
\[ B = x - 1 - x + 4\sqrt{x} - 4 \]
\[ B = 4\sqrt{x} - 5 \]

\section{Τρόποι παραγοντοποίησης}

Η παραγοντοποίηση είναι η διαδικασία μετατροπής ενός αθροίσματος σε γινόμενο παραγόντων. Οι βασικές μέθοδοι είναι:
\begin{enumerate}[label=\arabic*.]
\item \kerkissans{Κοινός παράγοντας:}\\ $a\beta+a\gamma=a(\beta+\gamma)$
\item \kerkissans{Ομαδοποίηση}\\ $ ax+ay+\beta x+\beta y=a(x+y)+\beta(x+y)=(x+y)(a+\beta) $
\begin{multicols}{2}
\item \kerkissans{Διαφορά τετραγώνων:}\\ $a^2-\beta^2=(a+\beta)(a-\beta)$
\item \kerkissans{Άθροισμα κύβων:}\\ $a^3+\beta^3=(a+\beta)(a^2-a\beta+\beta^2)$
\item \kerkissans{Διαφορά κύβων:}\\ $a^3-\beta^3=(a-\beta)(a^2+a\beta+\beta^2)$
\item \kerkissans{Ανάπτυγμα αθροίσματος:}\\
$a^2+2a\beta+\beta^2=(a+\beta)^2$
\end{multicols}
\item \kerkissans{Ανάπτυγμα διαφοράς:}\\
$a^2-2a\beta+\beta^2=(a-\beta)^2$
\item \kerkissans{Τριώνυμο}\\
$ax^2+\beta x+\gamma=\begin{cdcases}
a(x-x_1)(x-x_2) & ,\textrm{αν }\varDelta>0\\
a(x-x_0)^2 & ,\textrm{αν }\varDelta=0\\
\textrm{δεν παραγοντοποιείται} & ,\textrm{αν }\varDelta<0
\end{cdcases}$
\item \kerkissans{Τέλεια διαίρεση πολυωνύμων}\\
$P(x)=\delta(x)\cdot\pi(x)$ όπου $\delta(x)$ ο διαιρέτης και $\pi(x)$ το πηλίκο.
\end{enumerate}


\begin{itemize}
    \item Η παραγοντοποίηση είναι απαραίτητη για την απλοποίηση κλασμάτων, την επίλυση εξισώσεων/ανισώσεων, την εύρεση ορίων κ.λπ.
    \item Πάντα ξεκινάμε ελέγχοντας αν υπάρχει \textbf{κοινός παράγοντας} σε όλους τους όρους. Αν υπάρχει, τον βγάζουμε πρώτο.
    \item Στη συνέχεια εξετάζουμε αν εφαρμόζεται κάποια \textbf{ταυτότητα} ή αν είναι \textbf{τριώνυμο}.
\end{itemize}


\Paradeigma{Παραγοντοποίηση αλγεβρικής παράστασης}
\bmath{Να παραγοντοποιήσετε την παράσταση: $P(x) = x^3 - 4x^2 - x + 4$.}\\
\lysh
Με τη μέθοδο της ομαδοποίησης:
\[ P(x) = x^3 - 4x^2 - x + 4 = x^2(x - 4) - (x - 4) \]
Βγάζουμε κοινό παράγοντα το $(x - 4)$:
\[ P(x) = (x - 4)(x^2 - 1) \]
Εφαρμόζουμε τη διαφορά τετραγώνων:
\[ P(x) = (x - 4)(x - 1)(x + 1) \]

\section{Ιδιότητες απόλυτης τιμής}

Η απόλυτη τιμή $|x|$ εκφράζει την απόσταση του αριθμού $x$ από το 0 στον άξονα των πραγματικών αριθμών. Ισχύει ότι:
\[ |x| = \begin{cases} x, & \text{αν } x \geq 0 \\ -x, & \text{αν } x < 0 \end{cases} \]
\textbf{Βασικές Ιδιότητες:}
\begin{multicols}{2}
\begin{itemize}[label=\faEdit]
    \item $|x| \geq 0$, με το ίσον να ισχύει μόνο όταν $x = 0$.
    \item $|-x| = |x|$
    \item $|x \cdot y| = |x| \cdot |y|$
    \item $\left| \dfrac{x}{y} \right| = \dfrac{|x|}{|y|}$, με $y \neq 0$
    \item $|x|^2 = x^2$
    \item $|x + y| \leq |x| + |y|$ (Τριγωνική ανισότητα)
\end{itemize}
\end{multicols}

\begin{itemize}
    \item Για να 'βγάλουμε' ένα απόλυτο σε μια παράσταση, εξετάζουμε το πρόσημο της ποσότητας που βρίσκεται μέσα στο απόλυτο.
    \item Αν η ποσότητα είναι θετική (ή μηδέν), το απόλυτο παραλείπεται. Αν είναι αρνητική, το απόλυτο παραλείπεται και αλλάζουμε το πρόσημο.
    \item Η ιδιότητα $\sqrt{x^2} = |x|$ χρησιμοποιείται πάρα πολύ συχνά στην απλοποίηση ριζών.
\end{itemize}


\Paradeigma{Απλοποίηση με απόλυτες τιμές}
\bmath{Αν $x \in (1, 3)$, να απλοποιήσετε την παράσταση: $\Gamma = |x-1| + |x-3| - \sqrt{(x-4)^2}$.}\\
\lysh
Αφού $x \in (1, 3)$, έχουμε:
\begin{itemize}
    \item $x > 1 \Leftrightarrow x - 1 > 0$, άρα $|x-1| = x-1$.
    \item $x < 3 \Leftrightarrow x - 3 < 0$, άρα $|x-3| = -(x-3) = -x+3$.
    \item $x < 3 < 4 \Leftrightarrow x - 4 < 0$, άρα $\sqrt{(x-4)^2} = |x-4| = -(x-4) = -x+4$.
\end{itemize}
Άρα:
\[ \Gamma = (x-1) + (-x+3) - (-x+4) = x - 1 - x + 3 + x - 4 = x - 2 \]

\section{Ιδιότητες ριζών}
\textbf{\kerkissans{Ιδιότητες Τετραγωινκής ρίζας}}
\begin{multicols}{2}
\begin{itemize}[label*=\faEdit]
\item $ \left(\sqrt{x}\;\right)^2=x\;\;,\;\; x\geq0  $
\item  $ \sqrt{x^2}=|x|\;\;,\;\; x\in\mathbb{R} $
\item $ \sqrt{x\cdot y}=\!\sqrt{x}\cdot\!\sqrt{y}\;\;,\;\; x,y\geq0 $
\item $ \sqrt{\dfrac{x}{y}}\;=\dfrac{\sqrt{x}}{\sqrt{y}}\;\;,\;\; x\geq0\textrm{ και }y>0 $
\item $ \sqrt{x\pm y}\neq\!\sqrt{x}\pm\!\sqrt{y}\;\;,\;\; x,y\geq0 $
\end{itemize}
\end{multicols}
\textbf{\kerkissans{Ιδιότητες ν-οστής ρίζας}}
\begin{multicols}{2}
\begin{itemize}[label*=\faEdit]
\item $ \left(\sqrt[\nu]{x}\;\right)^\nu=x\;\;,\;\; x\geq0  $ 
\item $ \sqrt[\nu]{x^\nu}=\begin{cases}
|x|&  ,x\in\mathbb{R}\textrm{ αν }\nu\textrm{ άρτιος}\\x&  ,x\geq0\textrm{ και } \nu\in\mathbb{N}\end{cases} $
\item $ \sqrt[\nu]{x\cdot y}=\!\sqrt[\nu]{x}\cdot\!\sqrt[\nu]{y}\;\;,\;\; x,y\geq0 $ 
\item $ \sqrt[\nu]{\dfrac{x}{y}}\;=\dfrac{\sqrt[\nu]{x}}{\sqrt[\nu]{y}}\;\;,\;\; x\geq0\textrm{ και }y>0 $
 \item $ \sqrt[\mu]{\!\sqrt[\nu]{x}}=\!\sqrt[\nu\cdot\mu]{x}\;\;,\;\; x\geq0 $ 
\item $ \sqrt[\nu]{x^\nu\cdot y}=x\!\sqrt[\nu]{y}\;\;,\;\; x,y\geq0  $
\end{itemize}
\end{multicols}

\begin{itemize}
    \item Προσοχή: $\sqrt{x^2} = |x|$ ισχύει για κάθε $x \in \mathbb{R}$, ενώ $\sqrt{x}^2 = x$ ισχύει μόνο για $x \geq 0$.
    \item Επίσης,  $\sqrt{x + y} \neq \sqrt{x} + \sqrt{y}$. Η ρίζα «σπάει» μόνο στον πολλαπλασιασμό και τη διαίρεση, \textbf{ποτέ} στην πρόσθεση και την αφαίρεση.
    \item Όταν έχουμε ρίζες στον παρονομαστή, συνήθως κάνουμε \textbf{ρητοποίηση} πολλαπλασιάζοντας αριθμητή και παρονομαστή με την κατάλληλη παράσταση (π.χ. τη συζυγή σε περιπτώσεις του τύπου $\sqrt{a} \pm \sqrt{\beta}$).
\end{itemize}


\Paradeigma{Ρητοποίηση παρονομαστή}
Να μετατρέψετε το κλάσμα ώστε να έχει ρητό παρονομαστή: $\dfrac{2}{3 - \sqrt{5}}$.

\lysh
Πολλαπλασιάζουμε αριθμητή και παρονομαστή με τη \textbf{συζυγή παράσταση} του παρονομαστή, δηλαδή το $3 + \sqrt{5}$:
\[ \dfrac{2(3 + \sqrt{5})}{(3 - \sqrt{5})(3 + \sqrt{5})} = \dfrac{2(3 + \sqrt{5})}{3^2 - (\sqrt{5})^2} = \dfrac{2(3 + \sqrt{5})}{9 - 5} = \dfrac{2(3 + \sqrt{5})}{4} = \dfrac{3 + \sqrt{5}}{2} \]

\section{Ιδιότητες λογαρίθμων}

Ο λογάριθμος του $\theta > 0$ με βάση $a$ ($0 < a \neq 1$), συμβολίζεται $\log_a \theta$, και είναι ο εκθέτης στον οποίο πρέπει να υψωθεί το $a$ για να προκύψει το $\theta$:
\[ \log_a \theta = x \Leftrightarrow a^x = \theta \]
\textbf{Βασικοί λογάριθμοι:}
\begin{itemize}
    \item Δεκαδικός λογάριθμος ($\log x$): Βάση το $10$. Ισχύει: $\log x \Leftrightarrow \log_{10} x$
    \item Φυσικός λογάριθμος ($\ln x$): Βάση το $e \approx 2.718$. Ισχύει: $\ln x \Leftrightarrow \log_{e} x$
\end{itemize}
\textbf{Ιδιότητες:} 
Για $x, y > 0$, $\kappa \in \mathbb{R}$ και $a > 0$ με $a \neq 1$:

\begin{itemize}[label=\faEdit]
\begin{multicols}{2}
    \item $\log_a 1 = 0 \quad (\ln 1 = 0)$
    \item $\log_a a = 1 \quad (\ln e = 1)$
    \item $a^{\log_a x} = x \quad (e^{\ln x} = x)$
    \item $\log_a(a^x) = x \quad (\ln(e^x) = x)$
\end{multicols}
    \item $\log_a(x \cdot y) = \log_a x + \log_a y \quad (\ln(xy) = \ln x + \ln y)$
    \item $\log_a \left(\dfrac{x}{y}\right) = \log_a x - \log_a y \quad \left(\ln\left(\dfrac{x}{y}\right) = \ln x - \ln y\right)$
    \item $\log_a(x^\kappa) = \kappa \log_a x \quad (\ln(x^\kappa) = \kappa \ln x)$
\end{itemize}


\begin{itemize}
    \item Οι ιδιότητες (5), (6) και (7) χρησιμοποιούνται και «ανάποδα» για να «μαζέψουμε» λογαριθμικές παραστάσεις. Π.χ. $\ln x + \ln y = \ln(xy)$ και $2\ln x = \ln(x^2)$.
    \item Προσοχή: $\ln(x \cdot y) = \ln x + \ln y$ (σωστό), αλλά $\ln(x + y) \neq \ln x + \ln y$ (λάθος).
    \item $\ln(x^2)$ ορίζεται για $x \neq 0$ και ισούται με $2\ln|x|$. Ενώ το $(\ln x)^2 = \ln^2 x$ είναι το τετράγωνο της συνάρτησης και γράφεται $\ln^2 x$.
\end{itemize}


\Paradeigma{Απλοποίηση λογαριθμικής παράστασης}
Να βρείτε την τιμή της παράστασης: $\Delta = 2\ln\sqrt{e} + 3\ln\left(\dfrac{1}{e}\right) - \ln(e^2)$.

\lysh
\[ \Delta = 2\ln(e^{\dfrac{1}{2}}) + 3\ln(e^{-1}) - \ln(e^2) \]
\[ \Delta = 2 \cdot \dfrac{1}{2}\ln e + 3(-1)\ln e - 2\ln e \]
Αφού $\ln e = 1$:
\[ \Delta = 1 - 3 - 2 = -4 \]

\section{Σύνολα αριθμών}

\begin{itemize}
    \item \textbf{Φυσικοί ($\mathbb{N}$):} $\mathbb{N} = \{0, 1, 2, 3, \dots\}$. To $\mathbb{N}^* = \{1, 2, 3, \dots\}$ δεν περιέχει το $0$.
    \item \textbf{Ακέραιοι ($\mathbb{Z}$):} $\mathbb{Z} = \{\dots, -2, -1, 0, 1, 2, \dots\}$. Περιλαμβάνουν τους φυσικούς.
    \item \textbf{Ρητοί ($\mathbb{Q}$):} $\mathbb{Q} = \left\lbrace \dfrac{\mu}{\nu} : \mu \in \mathbb{Z}, \nu \in \mathbb{Z}^*\right\rbrace .$ Οι αριθμοί που μπορούν να γραφτούν ως κλάσματα ακεραίων.
    \item \textbf{Άρρητοι:} Οι αριθμοί που δεν είναι ρητοί (π.χ. $\sqrt{2}, \pi, e$). Έχουν άπειρα και μη περιοδικά δεκαδικά ψηφία.
    \item \textbf{Πραγματικοί ($\mathbb{R}$):} $\mathbb{R} = $ Ρητοί $\cup$ Άρρητοι. Όλοι οι αριθμοί πάνω στον άξονα. Ισχύει: $\mathbb{N} \subset \mathbb{Z} \subset \mathbb{Q} \subset \mathbb{R}$.
\end{itemize}

\Paradeigma{Σύνολα και Λύσεις}
Να βρεθούν οι πραγματικές λύσεις της εξίσωσης $x^2 = 3$. Τι είδους αριθμοί είναι?

\lysh
Λύνουμε στο $\mathbb{R}$:
\[ x^2 = 3 \Leftrightarrow x = \pm\sqrt{3} \]
Οι ρίζες $\sqrt{3}, -\sqrt{3}$ ανήκουν στους πραγματικούς ($\mathbb{R}$) και πιο συγκεκριμένα στο σύνολο των άρρητων αριθμών.

\section{Εξισώσεις 1ου βαθμού}

Μία εξίσωση της μορφής $a x + \beta = 0$, με άγνωστο το $x$, διακρίνεται στις εξής περιπτώσεις:
\begin{itemize}
    \item Αν $a \neq 0$: τότε έχει \textbf{μοναδική λύση} $x = -\dfrac{\beta}{a}$.
    \item Αν $a = 0$ και $\beta \neq 0$: η εξίσωση γίνεται $0x = -\beta$ (όπου $-\beta \neq 0$) και είναι \textbf{αδύνατη}.
    \item Αν $a = 0$ και $\beta = 0$: η εξίσωση γίνεται $0x = 0$ που αληθεύει για κάθε $x \in \mathbb{R}$ (δηλαδή είναι \textbf{ταυτότητα}).
\end{itemize}

\bhmata
\begin{bhma}
    \item Διώχνουμε παρονομαστές και παρενθέσεις.
    \item Χωρίζουμε γνωστούς από αγνώστους (φέρνουμε όλα τα $x$ σε ένα μέλος).
    \item Φέρνουμε την εξίσωση στη μορφή $a x = \beta$.
    \item Αν οι συντελεστές είναι παράμετροι, μελετάμε προσεκτικά τις περιπτώσεις $a = 0$ και $a \neq 0$.
\end{bhma}


\Paradeigma{Απλή εξίσωση πρώτου βαθμού}
Να λυθεί η εξίσωση: $\dfrac{x-1}{2} - \dfrac{2x+3}{3} = x - \dfrac{5}{6}$.

\lysh
Βρίσκουμε ΕΚΠ$(2, 3, 6) = 6$ και πολλαπλασιάζουμε κάθε όρο με $6$:
\[ 6 \cdot \dfrac{x-1}{2} - 6 \cdot \dfrac{2x+3}{3} = 6 \cdot x - 6 \cdot \dfrac{5}{6} \]
\[ 3(x-1) - 2(2x+3) = 6x - 5 \]
\[ 3x - 3 - 4x - 6 = 6x - 5 \]
\[ -x - 9 = 6x - 5 \]
\[ -x - 6x = -5 + 9 \]
\[ -7x = 4 \Leftrightarrow x = -\dfrac{4}{7} \]

\section{Ανισώσεις 1ου βαθμού}

Είναι ανισότητες της μορφής $a x + \beta > 0$ (ή $<, \ge, \le$). Λύνονται με τα ίδια βήματα που λύνουμε μια εξίσωση 1ου βαθμού.
Η \textbf{μοναδική} τους διαφορά είναι ο \textbf{βασικός κανόνας των ανισοτήτων}: 
\begin{center}
\begin{prosoxi}{10cm}
\textit{Αν πολλαπλασιάσουμε ή διαιρέσουμε και τα δύο μέλη μιας ανισότητας με \textbf{αρνητικό} αριθμό, τότε \textbf{αλλάζει η φορά} της ανισότητας.}
\end{prosoxi}
\end{center}

\bhmata
\begin{bhma}
    \item Διώχνουμε παρονομαστές (πολλαπλασιάζοντας με το ΕΚΠ αν είναι θετικό νούμερο). Αν ο παρονομαστής έχει άγνωστο ή παράμετρο των οποίων το πρόσημο δεν ξέρουμε, χρειάζονται μεγαλύτερη προσοχή (βλ. κλασματικές ανισώσεις).
    \item Χωρίζουμε γνωστούς από αγνώστους.
    \item Καταλήγουμε στη μορφή $a x > \beta$:
      \begin{itemize}
          \item Αν $a > 0$, τότε $x > \dfrac{\beta}{a}$
          \item Αν $a < 0$, τότε $x < \dfrac{\beta}{a}$ (αλλαγή φοράς)
      \end{itemize}
\end{bhma}


\Paradeigma{Ανίσωση 1ου βαθμού}
Να βρεθούν οι τιμές του $x$ για τις οποίες ισχύει: $-(x - 2) + 3(1 - 2x) \leq x - 5$.

\lysh
\[ -x + 2 + 3 - 6x \leq x - 5 \]
\[ -7x + 5 \leq x - 5 \]
\[ -7x - x \leq -5 - 5 \]
\[ -8x \leq -10 \]
Διαιρούμε με το $-8$ άρα \textbf{αλλάζει η φορά}:
\[ x \geq \dfrac{-10}{-8} \Leftrightarrow x \geq \dfrac{5}{4} \]
Δηλαδή $x \in \left[\dfrac{5}{4}, +\infty\right)$.

\section{Εξισώσεις με απόλυτη τιμή}

Οι εξισώσεις που περιέχουν την άγνωστη ποσότητα σε απόλυτη τιμή λύνονται με βάση τις παρακάτω βασικές μορφές:
\begin{itemize}
    \item $|f(x)| = a \quad$ με $a > 0 \Leftrightarrow f(x) = a$ ή $f(x) = -a$
    \item $|f(x)| = a \quad$ με $a = 0 \Leftrightarrow f(x) = 0$
    \item $|f(x)| = a \quad$ με $a < 0 \Leftrightarrow \text{Αδύνατη}$

\end{itemize}

\end{bhma}

\[ \Delta = \beta^2 - 4a\gamma \]
Ανάλογα με το πρόσημο της διακρίνουσας έχουμε:
\begin{itemize}
    \item Αν $\Delta > 0$: 2 πραγματικές και άνισες ρίζες, τις $x_{1,2} = \dfrac{-\beta \pm \sqrt{\Delta}}{2a}$.
    \item Αν $\Delta = 0$: 1 πραγματική διπλή ρίζα, την $x = -\dfrac{\beta}{2a}$.
    \item Αν $\Delta < 0$: Δεν έχει πραγματικές ρίζες (αδύνατη στο $\mathbb{R}$).
\end{itemize}

\bhmata
\begin{bhma}
    \item Φέρνουμε την εξίσωση στη μορφή $a x^2 + \beta x + \gamma = 0$.
    \item Σε «ελλιπείς» εξισώσεις (όπου $\beta = 0$ ή $\gamma = 0$), δε χρειαζόμαστε διακρίνουσα:
    \begin{itemize}
        \item Αν $\gamma = 0$: $a x^2 + \beta x = 0 \Leftrightarrow x(a x + \beta) = 0 \Leftrightarrow x = 0$ ή $x = -\dfrac{\beta}{a}$.
        \item Αν $\beta = 0$: $a x^2 + \gamma = 0 \Leftrightarrow x^2 = -\dfrac{\gamma}{a}$. Αν $-\dfrac{\gamma}{a} \geq 0$ βγάζουμε ρίζα, αλλιώς είναι αδύνατη.
    \end{itemize}
\end{bhma}

\section{Εξισώσεις 3ου+ βαθμού}

Οι εξισώσεις ανώτερου βαθμού που δεν λύνονται άμεσα με παραγοντοποίηση λύνονται με τη μέθοδο \textbf{\eng{Horner}}.
Έστω ότι έχουμε το πολυώνυμο $P(x) = a_\nu x^\nu + \dots + a_1 x + a_0$.
\begin{itemize}
    \item Οι \textbf{πιθανές ακέραιες ρίζες} του είναι οι \textbf{διαιρέτες του σταθερού όρου} $a_0$.
    \item Αν $\rho$ είναι ρίζα του, το $P(x)$ έχει παράγοντα το $(x - \rho)$. 
    \item Οπότε $P(x) = 0 \Leftrightarrow (x - \rho) \cdot \pi(x) = 0 \Leftrightarrow x = \rho$ ή $\pi(x) = 0$, όπου το $\pi(x)$ είναι μικρότερου βαθμού και το βρίσκουμε με το Σχήμα \eng{Horner}.
\end{itemize}

\bhmata
\begin{bhma}
    \item Πάντα ξεκινάμε φέρνοντας όλους τους όρους στο 1o μέλος : $P(x) = 0$.
    \item Πριν δοκιμάσουμε \eng{Horner}, ελέγχουμε για \textbf{κοινός παράγοντας} ή \textbf{ομαδοποίηση}.
	\item Αν όχι,εφαρμόζουμε σχήμα \eng{Horner}.
\end{bhma}

\section{Εξισώσεις της μορφής $x^\nu=a$ }

Μία εξίσωση της μορφής $x^\nu = a$, με $\nu \in \mathbb{N}^*$, διακρίνεται σε 4 περιπτώσεις ανάλογα με το αν ο εκθέτης $\nu$ είναι άρτιος ή περιττός, και αν το $a$ είναι θετικό ή αρνητικό (ή $0$):
\begin{center}
\begin{mytblr}{}
 & $a>0$ & $a<0$\\
$\nu$: άρτιος & $x=\pm\sqrt[\nu]{a}$ & Αδύνατη \\
$\nu$: περιττός & $x=\sqrt[\nu]{a}$ & $x=-\sqrt[\nu]{|a|}$ 
\end{mytblr}
\end{center}
(Αν $a = 0$, ισχύει πάντα $x^\nu = 0 \Leftrightarrow x = 0$)


\begin{itemize}
    \item Αν συναντήσουμε παραστάσεις του τύπου $(f(x))^\nu = a$, χρησιμοποιούμε τον πίνακα για να λύσουμε ως προς $f(x)$.
    \item Προσοχή όταν ο εκθέτης είναι \textbf{άρτιος} και \textbf{θετικό} δεύτερο μέλος. Πρέπει να βάλουμε το $\pm$, αλλιώς «χάνουμε» τη μία (αρνητική) ρίζα.
\end{itemize}

\section{Κλασματικές εξισώσεις}

Μια εξίσωση λέγεται κλασματική όταν περιέχει άγνωστο στον παρονομαστή.
Για να έχει νόημα, \textbf{πρέπει υποχρεωτικά οι παρονομαστές να είναι διάφοροι του μηδενός}.

\bhmata
\begin{bhma}
    \item Παραγοντοποιούμε τους παρονομαστές.
    \item Βρίσκουμε το Ελάχιστο Κοινό Πολλαπλάσιο (ΕΚΠ) των παρονομαστών.
	\item Παίρνουμε περιορισμό $\text{ΕΚΠ}\neq 0$.
    \item Πολλαπλασιάζουμε κάθε όρο της εξίσωσης με το ΕΚΠ, ώστε να απαλειφθούν οι παρονομαστές.
    \item Λύνουμε την πολυωνυμική εξίσωση που προκύπτει.
    \item \textbf{Ελέγχουμε} αν οι λύσεις που βρήκαμε ικανοποιούν τους περιορισμούς του Βήματος 3.
\end{bhma}


\Paradeigma{Κλασματική εξίσωση}
Να λυθεί η εξίσωση: 
\[\dfrac{x}{x - 1} - \dfrac{1}{x + 1} = \dfrac{x+1}{x^2 - 1}\]

\lysh
Η εξίσωση γράφεται
\[\dfrac{x}{x - 1} - \dfrac{1}{x + 1} = \dfrac{x+1}{(x-1)(x+1)}\]
Το ΕΚΠ είναι το $(x - 1)(x + 1)$. Πρέπει ΕΚΠ$\neq 0$ άρα
\[\text{ΕΚΠ}\neq0\Leftrightarrow (x-1)(x+1)\neq 0\Leftrightarrow x-1\neq0\ \ \text{και}\ \ x+1\neq 0\Leftrightarrow x\neq1\ \ \text{και}\ \ x\neq-1\]
Πολλαπλασιάζουμε με το ΕΚΠ:
\[ (x - 1)(x + 1) \dfrac{x}{x - 1} - (x - 1)(x + 1) \dfrac{1}{x + 1} = (x - 1)(x + 1) \dfrac{x+1}{(x - 1)(x + 1)} \]
\[ x(x + 1) - 1(x - 1) = x+1 \]
\[ x^2 + x - x + 1 = x+1 \Leftrightarrow x^2-x = 0 \Leftrightarrow x(x-1)=0 \Leftrightarrow x=0 \ \text{ή}\ x=1 \]
\textbf{Έλεγχος:}
Λόγω των περιορισμών ($x \neq \pm 1$), η λύση $x=1$ \textbf{απορρίπτεται}, ενώ η λύση $x=0$ είναι \textbf{δεκτή}.

\section{Κλασματικές ανισώσεις}

Μια ανίσωση λέγεται κλασματική όταν έχει άγνωστο σε παρονομαστή κλάσματος. Η βασική μορφή είναι $\dfrac{A(x)}{B(x)} \lessgtr 0$.
Η βασική ιδιότητα που χρησιμοποιούμε είναι ότι το πηλίκο $\dfrac{A(x)}{B(x)}$ και το γινόμενο $A(x) \cdot B(x)$ έχουν \textbf{το ίδιο πρόσημο}:
\[ \dfrac{A(x)}{B(x)} > 0 \Leftrightarrow A(x) \cdot B(x) > 0 \]
(Ομοίως για $<0$). Αν έχουμε $\geq$ ή $\leq$ τότε προσθέτουμε και τον απαραίτητο περιορισμό:
\[ \dfrac{A(x)}{B(x)} \geq 0 \Leftrightarrow A(x) \cdot B(x) \geq 0 \ \text{\textbf{και} } B(x) \neq 0 \]

\bhmata
\begin{bhma}
    \item \textbf{Δεν κάνουμε απαλοιφή παρονομαστών} όταν δεν ξέρουμε το πρόσημό τους, διότι αν πολλαπλασιάσουμε με αρνητικό αριθμό αλλάζει η φορά, ενώ με θετικό παραμένει.
    \item Μεταφέρουμε όλους τους όρους στο ένα μέλος (ώστε στο άλλο να έχουμε 0).
    \item Κάνουμε τα κλάσματα ομώνυμα ώστε να προκύψει ένα ενιαίο κλάσμα $\dfrac{A(x)}{B(x)} \gtrless 0$.
    \item Μετατρέπουμε το πηλίκο σε γινόμενο: $\dfrac{A(x)}{B(x)} \gtrless 0 \Leftrightarrow A(x) \cdot B(x) \gtrless 0$. Αναζητούμε το πρόσημο με πίνακα!
    \item Αν υπάρχει $\geq$ ή $\leq$, βάζουμε οπωσδήποτε τον περιορισμό $B(x) \neq 0$.
\end{bhma}

\section{Εξισώσεις με 1 ρίζα (Άρρητες εξισώσεις)}

Μία άρρητη εξίσωση περιέχει τον άγνωστο κάτω από ρίζα.
Λύνεται υψώνοντας και τα δύο μέλη σε κατάλληλη δύναμη για να φύγει η ρίζα (συνήθως στο τετράγωνο). \textbf{Ο έλεγχος των λύσεων είναι υποχρεωτικός.}
\bhmata
\begin{bhma}
    \item \textbf{(Περιορισμοί):} Για να ορίζεται η $\sqrt{A(x)}$, πρέπει αναγκαστικά $A(x) \geq 0$.
    \item Απομονώνουμε στο ένα μέλος τη ρίζα: $\sqrt{A(x)} = B(x)$.
    \item \textbf{(Επιπλέον περιορισμός:} Αφού $\sqrt{A(x)} \geq 0$, πρέπει να είναι και το 2ο μέλος θετικό ή μηδέν, άρα δηλώνουμε ότι πρέπει $B(x) \geq 0$.
    \item Υψώνουμε στο τετράγωνο: $(\sqrt{A(x)})^2 = (B(x))^2 \Leftrightarrow A(x) = (B(x))^2$.
    \item Λύνουμε την εξίσωση που προκύπτει.
    \item Ελέγχουμε τις λύσεις αν ικανοποιούν τους περιορισμούς (ή κάνουμε επαλήθευση στην αρχική).
\end{bhma}


\Paradeigma{Άρρητη εξίσωση}
Να λυθεί η εξίσωση: $\sqrt{2x + 3} = x$.

\lysh
\textbf{Περιορισμοί:}
Πρέπει $2x + 3 \geq 0 \Leftrightarrow x \geq -\dfrac{3}{2}$ και $x \geq 0$ άρα έχουμε τελικό περιορισμό: $x \geq 0$.

Υψώνουμε στο τετράγωνο:
\[ (\sqrt{2x + 3})^2 = x^2 \Leftrightarrow 2x + 3 = x^2 \Leftrightarrow x^2 - 2x - 3 = 0 \]
\Quadradic{1,-2,-3}{x}
\textbf{Έλεγχος:}
Το 3 ικανοποιεί τον περιορισμό $x \geq 0$, άρα είναι \textbf{δεκτό}.
Το $-1$ απορρίπτεται γιατί $-1 < 0$. Άρα μοναδική λύση $x = 3$.

\section{Ανισώσεις με 1 ρίζα}

Είναι ανισώσεις της μορφής $\sqrt{A(x)} > B(x)$ ή $\sqrt{A(x)} < B(x)$. 
Η λύση τους απαιτεί μεγάλη προσοχή στα πρόσημα των δύο μελών πριν υψώσουμε στο τετράγωνο. \textbf{Δεν} μπορούμε να υψώσουμε στο τετράγωνο μια ανίσωση αν δεν ξέρουμε ότι και τα δύο μέλη είναι \textbf{μη αρνητικά}. Ας εξετάσουμε τις 2 συχνότερες περιπτώσεις:

\bmath{1η Περίπτωση: $\sqrt{A(x)} < B(x)$ (ή $\leq$)}

\bhmata
\begin{bhma}
\item Για να υψώσουμε στο τετράγωνο, πρέπει πρώτα να θέσουμε τους περιορισμούς:
$A(x) \geq 0$ και $B(x) > 0$.
\item Με αυτούς τους περιορισμούς, υψώνουμε στο τετράγωνο και παίρνουμε $A(x) < B^2(x)$.
\item Ελέγχουμε αν οι λύσεις ικανοποιούν τους περιορισμούς.
\end{bhma}


\bmath{2η Περίπτωση: $\sqrt{A(x)} > B(x)$ (ή $\geq$)}

Εδώ διακρίνουμε σε δύο περιπτώσεις για το πρόσημο του $B(x)$:
\begin{enumerate}
\item Αν το $B(x)$ είναι αρνητικό ($B(x) < 0$), η ανίσωση ισχύει πάντα αρκεί να ορίζεται η ρίζα, άρα πρέπει $A(x) \geq 0$.
Λύνουμε τις ανισώσεις: $A(x) \geq 0$ και $B(x) < 0$.
\item Αν το $B(x)$ είναι $\geq 0$, μπορούμε να υψώσουμε στο τετράγωνο.
Λύνουμε τις ανισώσεις: $A(x) \geq 0$, $B(x) \geq 0 $ και $ A(x) > (B(x))^2 $.
Η τελική λύση είναι η \textbf{ένωση} των λύσεων των (1) και (2).
\end{enumerate}

\Paradeigma{Άρρητη ανίσωση της μορφής $\sqrt{A} > B$}
Να λυθεί η ανίσωση $\sqrt{x-1} > x-3$.

\lysh
Πρέπει να ισχύει $x-1 \ge 0 \Leftrightarrow x \ge 1$.
Εξετάζουμε 2 περιπτώσεις για το $x-3$:

\bmath{1η περίπτωση: $x - 3 < 0 \Leftrightarrow x < 3$}
Η ανίσωση $\sqrt{x-1} > x-3$ έχει α' μέλος θετικό ή μηδέν και β' μέλος αρνητικό, οπότε ισχύει πάντα (εφόσον ορίζεται η ρίζα). 
Άρα συναληθεύουμε τους περιορισμούς: $\begin{cases} x \geq 1 \\ x < 3 \end{cases}$. Αυτό δίνει \textbf{$x \in [1, 3)$}.

Τώρα και τα 2 μέλη είναι $\geq 0$, οπότε επιτρέπεται να υψώσουμε στο τετράγωνο:
\[ (\sqrt{x-1})^2 > (x-3)^2 \Leftrightarrow x-1 > x^2 - 6x + 9 \Leftrightarrow x^2 - 7x + 10 < 0 \]

Οι βασικές τριγωνομετρικές εξισώσεις λύνονται με τους παρακάτω τύπους (όπου $\kappa \in \mathbb{Z}$):
\begin{itemize}
    \item $\hm{x} = \hm{\theta} \Leftrightarrow x = 2\kappa\pi + \theta$ ή $x = 2\kappa\pi + (\pi - \theta)$
    \item $\syn{x} = \syn{\theta} \Leftrightarrow x = 2\kappa\pi \pm \theta$
    \item $\ef{x} = \ef{\theta} \Leftrightarrow x = \kappa\pi + \theta$
    \item $\syf{x} = \syf{\theta} \Leftrightarrow x = \kappa\pi + \theta$
\end{itemize}
\textbf{Ειδικές περιπτώσεις (για $\hm{x}, \syn{x}$):} \\
$\hm{x} = 0 \Leftrightarrow x = \kappa\pi \qquad \hm{x} = 1 \Leftrightarrow x = 2\kappa\pi + \dfrac{\pi}{2} \qquad \hm{x} = -1 \Leftrightarrow x = 2\kappa\pi - \dfrac{\pi}{2}$ \\
$\syn{x} = 0 \Leftrightarrow x = \kappa\pi + \dfrac{\pi}{2} \qquad \syn{x} = 1 \Leftrightarrow x = 2\kappa\pi \qquad \syn{x} = -1 \Leftrightarrow x = 2\kappa\pi + \pi$

\bhmata
\begin{bhma}
    \item Μεταφέρουμε όλους τους όρους, παραγοντοποιούμε ή αντικαθιστούμε κάνοντας χρήση τριγωνομετρικών ταυτοτήτων μέχρι να καταλήξουμε σε απλή μορφή.
    \item Σε εξισώσεις $\hm{A(x)} = \hm{B(x)}$ κ.λπ., εφαρμόζουμε απ’ ευθείας τους τύπους βάζοντας $A(x)$ στη θέση του $x$ και $B(x)$ στη θέση του $\theta$.
    \item Αν μας δίνεται διάστημα (π.χ. $x \in [0, 2\pi]$), βρίσκουμε τη γενική λύση και μετά λύνουμε την ανίσωση (π.χ. $0 \leq 2\kappa\pi + \theta \leq 2\pi$) για να βρούμε ποιες ακέραιες τιμές του $\kappa$ είναι δεκτές.
\end{bhma}

\section{Εκθετικές εξισώσεις}

Εκθετική ονομάζεται η εξίσωση που ο άγνωστος βρίσκεται στον εκθέτη. 
Οι ιδιότητες (με $a > 0$, $a \neq 1$) που χρησιμοποιούμε είναι:
\[ a^{f(x)} = a^{g(x)} \Leftrightarrow f(x) = g(x) \]
(λόγω του ότι η $a^x$ είναι 1-1). 

\begin{itemize}
    \item \textbf{1ος τρόπος (Ίδιες βάσεις):} Αναλύουμε τους αριθμούς σε γινόμενο πρώτων παραγόντων ώστε να καταλήξουμε σε μορφή $a^{f(x)} = a^{g(x)}$. Στη συνέχεια εξισώνουμε τους εκθέτες.
    \item \textbf{2ος τρόπος (Λογαρίθμιση):} Αν τα δύο μέλη είναι θετικά αλλά δεν μπορούν να γίνουν ίδια βάση (π.χ. $2^x = 3$), λογαριθμίζουμε και τα δύο μέλη συνήθως με $\ln$: $\ln(2^x) = \ln 3 \Rightarrow x \ln 2 = \ln 3$.
    \item \textbf{3ος τρόπος (Αντικατάσταση):} Αν υπάρχουν προσθαφαιρέσεις δυνάμεων (π.χ. $e^{2x} - 3e^x + 2 = 0$), θέτουμε $\omega = a^x > 0$ (π.χ. $\omega = e^x > 0$), οπότε $a^{2x} = \omega^2$. Έτσι προκύπτει πολυωνυμική εξίσωση.
\end{itemize}


\Paradeigma{Εκθετικές εξισώσεις}
Να λυθούν οι εξισώσεις: 
\begin{multicols}{2}
\begin{alist}
\item $2^{x^2 - 1} = 8$
\item $e^{2x} - 5e^x + 6 = 0$.
\end{alist}
\end{multicols}

\lysh\vspace{-5mm}
\begin{alist}
\item Είναι: $8 = 2^3$. Άρα:
\[ 2^{x^2 - 1} = 2^3 \Leftrightarrow x^2 - 1 = 3 \Leftrightarrow x^2 = 4 \Leftrightarrow x = \pm 2 \]
\item Θέτουμε $e^x = \omega > 0$. Η εξίσωση γράφεται $\omega^2-5\omega+6=0$ και έχει ρίζες $\omega=2$ και $\omega=3$.Επομένως: 
\begin{itemize}
\item $e^x = 2 \Leftrightarrow x = \ln 2$
\item $e^x = 3 \Leftrightarrow x = \ln 3$.
\end{itemize}
\end{alist}

\section{Εκθετικές ανισώσεις}

Μία ανίσωση $a^{f(x)} > a^{g(x)}$ λύνεται αξιοποιώντας τη μονοτονία της εκθετικής συνάρτησης $a^x$:
\begin{itemize}
    \item Αν ${a > 1}$, τότε η $a^x$ είναι γν. αύξουσα, δηλαδή η φορά διατηρείται: $a^{f(x)} > a^{g(x)} \Leftrightarrow f(x) > g(x)$.
    \item Αν ${0 < a < 1}$, τότε η $a^x$ είναι γν. φθίνουσα, δηλαδή η φορά αλλάζει: $a^{f(x)} > a^{g(x)} \Leftrightarrow f(x) < g(x)$.
\end{itemize}

\Paradeigma{Εκθετική ανίσωση}
Να λυθεί η ανίσωση: $\left(\dfrac{1}{2}\right)^{3x} < 16$.

\lysh
Γράφουμε και τα δύο μέλη ως δυνάμεις της ίδιας βάσης. (Το 16 είναι $2^4$, το $\dfrac{1}{2}$ είναι $2^{-1}$).
\[ (2^{-1})^{3x} < 2^4 \Leftrightarrow 2^{-3x} < 2^4 \]
Επειδή η βάση είναι το 2 ($2 > 1$), η φορά \textbf{παραμένει}. Στη συνέχεια:
\[ -3x < 4 \Leftrightarrow x > -\dfrac{4}{3} \]

\section{Λογαριθμικές εξισώσεις}
Χρησιμοποιούμε τις ιδιότητες των λογαρίθμων ώστε να προκύψει η μορφή:
\[ \ln A(x) = \ln B(x) \Leftrightarrow A(x) = B(x) \]
Αν χρειαστεί μετατρέπουμε το $\ln f(x) = a$ σε εκθετική μορφή από τον ορισμό:
\[ \ln f(x) = a \Leftrightarrow f(x) = e^a \]

\bhmata
\begin{bhma}
    \item Παίρνουμε περιορισμούς (κάθε ποσότητα μέσα σε $\log$ ή $\ln$ πρέπει να είναι $> 0$). Οι ρίζες πρέπει να ικανοποιούν αυτούς τους περιορισμούς.
    \item \textbf{1ος τρόπος:} Μαζεύουμε πολλά $\ln$ σε ένα μέχρι η εξίσωση να γραφτεί στη μορφή $\ln A = \ln B$.
    \item \textbf{2ος τρόπος:} Αν εμφανίζονται δυνάμεις του $\ln x$, θέτουμε $\omega = \ln x$. 
\end{bhma}


\Paradeigma{Λογαριθμική εξίσωση}
Να λυθεί η εξίσωση: $\ln(x-1) + \ln(x-2) = \ln(2x-4)$.

\lysh
\textbf{Περιορισμοί:}
Πρέπει 
\begin{itemize}
\item $x-1 > 0 \Rightarrow x > 1$
\item $x-2 > 0 \Rightarrow x > 2$
\item $2x-4 > 0 \Rightarrow 2x > 4 \Rightarrow x > 2$
\end{itemize}
 Άρα πρέπει {$x > 2$}. Στο 1ο μέλος έχουμε άθροισμα λογαρίθμων:
\[ \ln[(x-1)(x-2)] = \ln(2x-4) \]
Αφού η $\ln(x)$ είναι $1-1$:
\[ (x-1)(x-2) = 2x-4 \Leftrightarrow x^2 - 2x - x + 2 = 2x - 4 \Leftrightarrow x^2 - 5x + 6 = 0 \]
Η εξίσωση έχει ρίζες $x = 2$ ή $x = 3$.
\textbf{Έλεγχος:} Το 2 απορρίπτεται λόγω του περιορισμού ($x > 2$). Συνεπώς η μόνη λύση είναι δεκτή η $x = 3$.

\section{Λογαριθμικές ανισώσεις}

Οι λογαριθμικές ανισώσεις αντιμετωπίζονται παρόμοια με τις λογαριθμικές εξισώσεις σε ότι αφορά τους περιορισμούς. Αφού καταλήξουμε στη μορφή $\ln A > \ln B$, εκμεταλλευόμαστε τη μονοτονία της λογαριθμικής συνάρτησης $\log_a x$.
\begin{itemize}
    \item Αν ${a > 1}$ (όπως στα $\ln x, \log x$), η συνάρτηση είναι γνησίως αύξουσα. Άρα: $\ln f(x) > \ln g(x) \Leftrightarrow f(x) > g(x)$. ({διατήρηση φοράς})
    \item Αν ${0 < a < 1}$, η συνάρτηση είναι αυστηρώς φθίνουσα (η φορά αλλάζει).
\end{itemize}

\bhmata
\begin{bhma}
    \item Βρίσκουμε τους \textbf{περιορισμούς} για τους οποίους ορίζεται η ανίσωση.
    \item Χρησιμοποιούμε ιδιότητες (ή αντικατάσταση με $\omega = \ln x$) για να λύσουμε την ανίσωση.
    \item \textbf{Συναληθεύουμε} τη λύση που βρήκαμε με τους αρχικούς περιορισμούς. Η τελική λύση προκύπτει από την τομή αυτών των διαστημάτων.
\end{bhma}


\Paradeigma{Λογαριθμική ανίσωση}
Να λυθεί η ανίσωση: $\ln(3x - 1) \leq 0$.

\lysh
\textbf{Περιορισμοί:}
Πρέπει $3x - 1 > 0 \Leftrightarrow 3x > 1 \Leftrightarrow x > \dfrac{1}{3}$.
Γράφουμε το $0$ ως $\ln 1$:
\[ \ln(3x - 1) \leq \ln 1 \]
Επειδή η $\ln x$ είναι γνησίως αύξουσα, η φορά διατηρείται:
\[ 3x - 1 \leq 1 \Leftrightarrow 3x \leq 2 \Leftrightarrow x \leq \dfrac{2}{3} \]
Κάνουμε συναλήθευση της λύσης $x \leq \dfrac{2}{3}$ με τον περιορισμό $x > \dfrac{1}{3}$.
Τελική λύση: $x \in \left(\dfrac{1}{3}, \dfrac{2}{3}\right]$.

\section{Εξίσωση ευθείας}

Κάθε εξίσωση της μορφής $a x + \beta y + \gamma = 0$ (με $a \neq 0$ ή $\beta \neq 0$) παριστάνει ευθεία γραμμή.
Ειδικές μορφές:
\begin{itemize}
    \item $y = \lambda x + \beta$: Ευθεία με κλίση (συντελεστή διεύθυνσης) $\lambda$ που τέμνει τον $y'y$ στο $(0, \beta)$. (Για $\lambda = \ef{\omega}$, όπου $\omega$ η γωνία που σχηματίζει η ευθεία με τον $x'x$).
    \item $y = y_0 + \lambda(x - x_0)$: Εξίσωση ευθείας που διέρχεται από το σημείο $M(x_0, y_0)$ με συντελεστή $\lambda$.
    \item $x = x_0$: Κατακόρυφη ευθεία, κάθετη στον $x'x$.
    \item $y = y_0$: Οριζόντια ευθεία, παράλληλη στον $x'x$.
\end{itemize}
Συντελεστής διεύθυνσης ευθείας που διέρχεται από δύο σημεία $A(x_1, y_1)$ και $B(x_2, y_2)$: 
\[ \lambda = \dfrac{y_2 - y_1}{x_2 - x_1} \quad (x_1 \neq x_2) \]

\textbf{Σχετικές θέσεις ευθειών:} Αν $\varepsilon_1: y = \lambda_1 x + \beta_1$ και $\varepsilon_2: y = \lambda_2 x + \beta_2$:
\begin{itemize}
    \item $\varepsilon_1 \parallel \varepsilon_2 \Leftrightarrow \lambda_1 = \lambda_2$
    \item $\varepsilon_1 \bot \varepsilon_2 \Leftrightarrow \lambda_1 \cdot \lambda_2 = -1$
    \item Συμπίπτουν αν $\lambda_1 = \lambda_2$ και $\beta_1 = \beta_2$.
\end{itemize}


\Paradeigma{Εξίσωση ευθείας από σημεία}
Να βρεθεί η εξίσωση της ευθείας που διέρχεται από τα σημεία $A(-1, 2)$ και $B(3, 10)$.

\lysh
Ο συντελεστής διεύθυνσης είναι:
\[ \lambda = \dfrac{y_B - y_A}{x_B - x_A} = \dfrac{10 - 2}{3 - (-1)} = \dfrac{8}{4} = 2 \]
Η εξίσωση της ευθείας (διά του $A$) είναι:
\[ y - y_A = \lambda(x - x_A) \Leftrightarrow y - 2 = 2(x - (-1)) \]
\[ y - 2 = 2(x + 1) \Leftrightarrow y = 2x + 2 + 2 \Leftrightarrow y = 2x + 4 \]

Συνήθως τα ζητούμενα μεγέθη (π.χ. $E, V$) εξαρτώνται από \textbf{δύο} μεταβλητές (π.χ. $x, y$ ή $R, \upsilon$). Ο \textbf{περιορισμός} που δίνεται στο πρόβλημα (π.χ. το ορθογώνιο βρίσκεται εγγεγραμμένο σε κύκλο, ή η περίμετρος είναι σταθερή κ.λπ.) μας επιτρέπει να \textbf{λύσουμε ως προς μία μεταβλητή} και να την αντικαταστήσουμε. Έτσι, το $E$ ή το $V$ γίνεται \textbf{συνάρτηση μίας μόνο μεταβλητής} της οποίας μπορούμε πια να μελετήσουμε τη μονοτονία και τα ακρότατα (με παραγώγιση).\\

\Paradeigma{Χρήση γεωμετρικών τύπων}
Θέλουμε να περιφράξουμε με συρματόπλεγμα 100 μέτρων ένα ορθογώνιο οικόπεδο. Εκφράστε το εμβαδόν του σε συνάρτηση της μίας πλευράς του.

\lysh
Έστω $x, y$ οι διαστάσεις του. Το μήκος της περίφραξης είναι η περίμετρος, άρα:
\[ 2x + 2y = 100 \Leftrightarrow x + y = 50 \Leftrightarrow y = 50 - x \]
Πρέπει επίσης $x > 0$ και $y > 0 \Rightarrow 50 - x > 0 \Rightarrow x < 50$. Δηλαδή $x \in (0, 50)$. Το εμβαδόν δίνεται από τον τύπο $E = x \cdot y$. Αντικαθιστώντας το $y$:
\[ E(x) = x(50 - x) \Leftrightarrow E(x) = -x^2 + 50x, \quad \text{για } x \in (0, 50). \]
