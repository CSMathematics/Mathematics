\section{Ιδιότητες}
\idiothta{Σύνθεση συναρτήσεων}
Έστω $f,g,h$ συναρτήσεις.
\begin{itemize}
\item Οι συναρτήσεις $f\circ g$ και $g\circ f$ δεν είναι υποχρεωτικά ίσες: $f\circ g\neq g\circ f$.
\item Αν ορίζεται η $h\circ(g\circ f)$ τότε ορίζεται και η $(h\circ g)\circ f$ και ισχύει
\[ h\circ(g\circ f)=(h\circ g)\circ f \]
\end{itemize}

\idiothta{Συνάρτηση \bmath{$1-1$}}
Έστω $f:A\to \mathbb{R} $ μια $1-1$ συνάρτηση. Τότε ισχύει ότι
\begin{itemize}
\item για κάθε $y$ του συνόλου τιμών της $f$ η εξίσωση $y=f(x)$ έχει μοναδική λύση ως προς $x$.
\item Κάθε οριζόντια ευθεία τέμνει την $C_f$ το πολύ σε ένα σημείο.
\end{itemize}

\idiothta{Αντίστροφη συνάρτηση}
Έστω $f:A\to\mathbb{R}$ μια $1-1$ συνάρτηση. Για την αντίστροφη συνάρτηση $f^{-1}$ ισχύουν τα εξής:
\begin{itemize}[itemsep=0mm]
\item Το πεδίο ορισμού της $ f^{-1} $ είναι το σύνολο τιμών $ f(A) $ της $ f $.
\item Το σύνολο τιμών της $ f^{-1} $ είναι το πεδίο ορισμού $ A $ της $ f $.
\item Ισχύει ότι $ y=f(x)\Leftrightarrow x=f^{-1}(y) $ για κάθε $x\in A$ και $ y\in f(A) $.
\item $f^{-1}(f(x))=x$ για κάθε $x\in A$.
\item $f(f^{-1}(y))=y$ για κάθε $y\in f(A)$.
\end{itemize}

\idiothta{Όρια}
Δίνονται συναρτήσεις $f.g$ με πεδία ορισμού $A,B$ αντίστοιχα και $x_0\in A\cap B$. Αν υπάρχουν τα όρια $\lim\limits_{x\to x_0}{f(x)}=l_1$ και $\lim\limits_{x\to x_0}{g(x)}=l_2$ τότε ισχύουν οι παρακάτω ιδιότητες:
\begin{itemize}
\item  $ \displaystyle{\lim_{x\rightarrow x_0}\left( f(x)\pm g(x)\right)=\displaystyle{\lim_{x\rightarrow x_0}f(x)}\pm\displaystyle{\lim_{x\rightarrow x_0}g(x)}}=l_1\pm l_2 $
\item $ \displaystyle{\lim_{x\rightarrow x_0}\left( k\cdot f(x)\right) }=k\cdot\displaystyle{\lim_{x\rightarrow x_0}f(x)}=k\cdot l_1\;\;,\;\; k\in\mathbb{R} $ 
\item $ \displaystyle{\lim_{x\rightarrow x_0}\left( f(x)\cdot g(x)\right)=\displaystyle{\lim_{x\rightarrow x_0}f(x)}\cdot\displaystyle{\lim_{x\rightarrow x_0}g(x)}}=l_1\cdot l_2 $
\item $ \displaystyle{\lim_{x\rightarrow x_0}\left(\dfrac{ f(x)} {g(x)}\right)=\dfrac{\displaystyle{\lim_{x\rightarrow x_0}f(x)}}{\displaystyle{\lim_{x\rightarrow x_0}g(x)}}}=\frac{l_1}{l_2}\;\;,\;\;l_2\neq0 $
\item $ \displaystyle{\lim_{x\rightarrow x_0}|f(x)|}=\left| \displaystyle{\lim_{x\rightarrow x_0}f(x)}\right|=|l_1|  $ 
\item $ \displaystyle{\lim_{x\rightarrow x_0}\!\!\sqrt[\kappa]{f(x)}}=\!\sqrt[\kappa]{\displaystyle{\lim_{x\rightarrow x_0}f(x)}}\;\;=\!\!\sqrt[\kappa]{l_1}\;\;,\;\;l_1\geq0 $ 
\item $ \displaystyle{\lim_{x\rightarrow x_0}f^\nu(x)}=\left( \displaystyle{\lim_{x\rightarrow x_0}f(x)}\right)^\nu=l_1^\nu  $
\end{itemize}
\idiothta{Βασικά τριγωνομετρικά όρια}
\vspace{-5mm}
\begin{multicols}{2}
\begin{alist}
\item $\lim\limits_{x\to 0}{\dfrac{\hm{x}}{x}}=1$
\item $\lim\limits_{x\to 0}{\dfrac{\syn{x}-1}{x}}=0$
\end{alist}
\end{multicols}

\idiothta{Όριο με αντικατάσταση}
Αν $g(x)\neq u_0$ κοντά στο $x_0$ τότε
\[\lim\limits_{x\to x_0}{f(g(x))}=\lim\limits_{u\to u_0}{f(u)}\]
όπου $u=g(x)$ και $u_0=\lim\limits_{x\to x_0}{g(x)}$.


\idiothta{Μη πεπερασμένο όριο}
\vspace{-5mm}
\begin{itemize}
\item Αν $\lim\limits_{x\to x_0}{f(x)}=+\infty$ τότε $f(x)>0$ κοντά στο $x_0$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=-\infty$ τότε $f(x)<0$ κοντά στο $x_0$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=+\infty$ τότε $\lim\limits_{x\to x_0}{(-f(x))}=-\infty$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=-\infty$ τότε $\lim\limits_{x\to x_0}{(-f(x))}=+\infty$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=\pm\infty$ τότε $\lim\limits_{x\to x_0}{\frac{1}{f(x)}}=0$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=\pm\infty$ τότε $\lim\limits_{x\to x_0}{|f(x)|}=+\infty$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=+\infty$ τότε $\lim\limits_{x\to x_0}{\sqrt[\nu]{f(x)}}=+\infty$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=0$ και $f(x)>0$ κοντά στο $x_0$ τότε $\lim\limits_{x\to x_0}{\frac{1}{f(x)}}=+\infty$.
\item Αν $\lim\limits_{x\to x_0}{f(x)}=0$ και $f(x)<0$ κοντά στο $x_0$ τότε $\lim\limits_{x\to x_0}{\frac{1}{f(x)}}=-\infty$.
\item $\lim\limits_{x\to 0}{\frac{1}{x^2}}=+\infty$ και γενικά $\lim\limits_{x\to 0}{\frac{1}{x^{2\nu}}}=0$, όπου $\nu\in\mathbb{N}$.
\item Αν $\nu\in\mathbb{N}$ τότε τα όρια $\lim\limits_{x\to 0}{\frac{1}{x^{2\nu+1}}}$ δεν υπάρχουν.
\end{itemize}

\idiothta{Όρια στο $\pm\infty$}
\begin{itemize}
\item $\lim\limits_{x\to +\infty}{x^{\nu}}=+\infty$ για κάθε $\nu\in\mathbb{N}^*$
\item $\lim\limits_{x\to -\infty}{x^{\nu}}=\ccases{+\infty & , \text{αν $\nu$ άρτιος}\\-\infty & , \text{αν $\nu$ περιττός}}$ για κάθε $\nu\in\mathbb{N}^*$
\item $\lim\limits_{x\to \pm\infty}{\dfrac{1}{x^{\nu}}}=0$ για κάθε $\nu\in\mathbb{N}^*$.
\end{itemize}

\idiothta{Όριο πολυωνυμικής και ρητής στο \bmath{$\pm\infty$}}
Έστω $P(x)=a_{\nu}x^{\nu}+a_{\nu-1}x^{\nu-1}+\ldots+a_1x+a_0$ και $Q(x)=\beta_{\nu}x^{\mu}+\beta_{\mu-1}x^{\mu-1}+\ldots+\beta_1x+\beta_0$ δύο πολυώνυμα με $a_\nu,\beta_\mu\neq 0$. Ισχύει ότι:
\begin{multicols}{2}
\begin{itemize}
\item $\lim\limits_{x\to\pm\infty}{P(x)}=\lim\limits_{x\to\pm\infty}{\left(a_{\nu}x^{\nu}\right)}$
\item $\lim\limits_{x\to \pm\infty}{\dfrac{P(x)}{Q(x)}}=\lim\limits_{x\to\pm\infty}{\dfrac{a_{\nu}x^{\nu}}{\beta_{\mu}x^{\mu}}}$.
\end{itemize}
\end{multicols}

\idiothta{Όρια εκθετικών συναρτήσεων}
Δίνεται η εκθετική συνάρτηση $f(x)=a^x,\ x\in\mathbb{R}$ με $0<a\neq 1$.
\begin{itemize}
\item Αν $a>1$ τότε
\begin{multicols}{2}
\begin{alist}
\item $\lim\limits_{x\to-\infty}{a^x}=0$
\item $\lim\limits_{x\o+\infty}{a^x}=+\infty$
\end{alist}
\end{multicols}
\item Αν $0<a<1$ τότε
\begin{multicols}{2}
\begin{alist}
\item $\lim\limits_{x\to-\infty}{a^x}=+\infty$
\item $\lim\limits_{x\o+\infty}{a^x}=0$
\end{alist}
\end{multicols}
\end{itemize}

\idiothta{Όρια λογαριθμικών συναρτήσεων}
\begin{multicols}{2}
\begin{itemize}
\item $\lim\limits_{x\to0^+}{\log{x}}=-\infty$
\item $\lim\limits_{x\to+\infty}{\log{x}}=+\infty$
\item $\lim\limits_{x\to0^+}{\ln{x}}=-\infty$
\item $\lim\limits_{x\to+\infty}{\ln{x}}=+\infty$
\end{itemize}
\end{multicols}

\idiothta{Ασύμπτωτες}
\begin{itemize}
\item Οι πολυωνυμικές συναρτήσεις 2ου$+$ βαθμού δεν έχουν ασύμπτωτες.
\item Οι ρητές συναρτήσεις με βαθμό αριθμητή τουλάχιστον $2$ μονάδες μεγαλύτερο από το βαθμό του παρονομαστή δεν έχουν πλάγιες ασύμπτωτες.
\end{itemize}
\idiothta{Αρχική συνάρτηση}
Αν οι $F,G$ είναι παράγουσες δύο συνεχών συναρτήσεων $f,g$ σε ένα διάστημα $\Delta$ και $\lambda\in\mathbb{R}$ τότε
\begin{itemize}
\item η $F+G$ είναι μια παράγουσα της συνάρτησης $f+g$ στο $\Delta$.
\item η $\lambda F$ είναι μια παράγουσα της $\lambda f$ στο $\Delta$.
\end{itemize}
\idiothta{Ορισμένο ολοκλήρωμα}
\vspace{-5mm}
\begin{itemize}
\begin{multicols}{2}
\item $\displaystyle\int_{a}^{\beta}{f(x)\d x}=-\int_{\beta}^{a}{f(x)\d x}$
\item $\displaystyle\int_{a}^{a}{f(x)\d x}=0$
\end{multicols}
\item $\displaystyle\int_{a}^{\beta}{(f(x)+g(x))\d x}=\int_{a}^{\beta}{f(x)\d x}+\int_{a}^{\beta}{g(x)\d x}$
\item $\displaystyle\int_{a}^{\beta}{\lambda f(x)\d x}=\lambda\int_{a}^{\beta}{f(x)\d x}$\ \ ,\ \ $\lambda\in\mathbb{R}$
\item $\displaystyle\int_{a}^{\beta}{(\lambda f(x)+\mu g(x))\d x}=\lambda\int_{a}^{\beta}{f(x)\d x}+\mu\int_{a}^{\beta}{g(x)\d x}$\ \ ,\ \ $\lambda,\mu\in\mathbb{R}$.
\item $\displaystyle\int_{a}^{\beta}{f(x)\d x}=\int_{a}^{\gamma}{f(x)\d x}+\int_{\gamma}^{\beta}{f(x)\d x}$\ \ ,\ \ $a,\beta,\gamma\in\Delta$.
\item $\displaystyle\int_{a}^{\beta}{c\d x}=c(\beta-a)$ για κάθε $c\in\mathbb{R}$.
\end{itemize}

\idiothta{Παραγοντική ολοκλήρωση}
Αν οι συναρτήσεις $f',g'$ είναι συνεχείς στο διάστημα $[a,\beta]$ τότε
\[\int_{a}^{\beta}{f(x)g'(x)\d x}=\left[f(x)g(x)\right]_{a}^{\beta}-\int_{a}^{\beta}{f'(x)g(x)\d x}\]

\idiothta{Ολοκλήρωση με αντικατάσταση}
Αν οι συναρτήσεις $f,g'$ είναι συνεχείς στο διάστημα $[a,\beta]$ τότε
\[\int_{a}^{\beta}{f(g(x))g'(x)\d x}=\int_{u_1}^{u_2}{f(u)\d u}\]
όπου $u=g(x), \d u=g'(x)\d x$ και $u_1=g(a)$ και $u_2=g(\beta)$.

\idiothta{Εμβαδόν χωρίου μεταξύ \bmath{$C_f,C_g,x=a,x=\beta$}}
Δίνονται συνεχής συνάρτηση $ f $ ορισμένη σε ένα διάστημα $ [a,\beta] $. Το εμβαδόν του χωρίου $ \varOmega $ μεταξύ $ C_f $, του άξονα $x'x$ και των ευθειών $ x=a,x=\beta $ δίνεται από τον τύπο
\[ E(\varOmega)=\int_{a}^{\beta}{|f(x)|\d x} \]

\idiothta{Εμβαδόν χωρίου μεταξύ \bmath{$C_f,C_g,x=a,x=\beta$}}
Δίνονται δύο συνεχείς συναρτήσεις $ f,g $ ορισμένες σε ένα διάστημα $ [a,\beta] $ Το εμβαδόν του χωρίου $ \varOmega $ μεταξύ των γραφικών παραστάσεων $ C_f,C_g $ και των ευθειών $ x=a,x=\beta $ δίνεται από τον τύπο
\[ E(\varOmega)=\int_{a}^{\beta}{|f(x)-g(x)|\d x} \]
