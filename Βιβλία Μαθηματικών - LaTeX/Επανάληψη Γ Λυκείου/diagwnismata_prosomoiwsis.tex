\subsection{ΔΙΑΓΩΝΙΣΜΑ ΠΡΟΣΟΜΟΙΩΣΗΣ 1}
\begin{thema}{Α}
\begin{erwthma}
    \item Έστω μια συνάρτηση $f$ ορισμένη σε ένα διάστημα $\Delta$. Αν $f'(x)>0$ σε κάθε εσωτερικό σημείο του $\Delta$ και η $f$ είναι συνεχής σε όλο το $\Delta$, να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα στο $\Delta$. \monades{7}
    \item Πότε μια συνάρτηση $f:A\to\mathbb{R}$ λέγεται συνάρτηση $1-1$? \monades{4}
    \item Πότε η ευθεία $y = l$ λέγεται οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$?\monades{4}
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό},
αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη:
    \begin{alist}
        \item Αν η $f$ είναι κυρτή στο $\mathbb{R}$, η εφαπτομένη της βρίσκεται πάντα πάνω από τη γραφική της παράσταση.
        \item Αν $\dint_a^\beta f(x) \d x > 0$, τότε αναγκαστικά η $f$ παίρνει μόνο θετικές τιμές στο διάστημα $[a,\beta]$.
        \item Αν μία συνάρτηση είναι $1-1$, τότε είναι υποχρεωτικά γνησίως μονότονη.
        \item Ισχύει ότι $\dint_{-a}^a \hm^3x \,\d x = 0$ για κάθε $a \in \mathbb{R}$.
        \item Αν η $f$ είναι συνεχής στο $[a,\beta]$ και $f(a) \cdot f(\beta) > 0$, τότε η εξίσωση $f(x)=0$ είναι αδύνατο να έχει ρίζα στο $(a,\beta)$.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}
\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x^2 - x + 1}{x - 1}$ με $x \ne 1$.
\begin{erwthma}
    \item Να μελετήσετε την $f(x)$ ως προς τη μονοτονία και τα τοπικά ακρότατα. \monades{6}
    \item Να εξετάσετε την $f$ ως προς την κυρτότητα και τα σημεία καμπής. \monades{6}
    \item Να βρείτε, εάν υπάρχουν, τις κατακόρυφες, οριζόντιες ή πλάγιες ασύμπτωτες της γραφικής παράστασης $C_f$. 

\monades{6}
    \item Να σχεδιάσετε τη γραφική παράσταση της $f$ και, με βάση αυτή, να προσδιορίσετε το σύνολο τιμών της. \monades{7}
\end{erwthma}
\end{thema}
\begin{thema}{Γ}
Δίνεται η συνάρτηση $f(x) = e^x - \lambda x$, $x \in \mathbb{R}$ και $\lambda > 0$. Η $C_f$ παρουσιάζει ελάχιστο ίσο με $0$ σε ένα σημείο $x_0$.
\begin{erwthma}
    \item Να αποδείξετε ότι $x_0 = 1$ και ότι $\lambda = e$. \monades{6}
    \item Να βρείτε την εξίσωση της εφαπτομένης της $C_f$ η οποία διέρχεται από το σημείο $A(0, -e^2)$.

 \monades{6}
    \item Υπολογίστε το εμβαδόν του χωρίου που περικλείεται από την $C_f$, τον άξονα $x'x$ και τον άξονα $y'y$. 

\monades{6}
    \item Ένα σημείο $M(x, y)$ κινείται πάνω στη γραφική παράσταση της $f$ (για $x \ge 1$). Αν ο ρυθμός μεταβολής της τετμημένης του είναι $x'(t) = 2 \, \text{μον}/\si{sec}$, να βρείτε το ρυθμό μεταβολής του εμβαδού του τριγώνου $O M K$, όπου $K$ είναι η προβολή του $M$ στον θετικό ημιάξονα $Ox$, τη χρονική στιγμή που το $M$ διέρχεται από το σημείο της καμπύλης με τετμημένη $x=2$. \monades{7}
\end{erwthma}
\end{thema}
\begin{thema}{Δ}
Θεωρούμε τη συνάρτηση $f: \mathbb{R} \to \mathbb{R}$ για την οποία ικανοποιείται η σχέση:
\[ f^3(x) + e^{f(x)} = x + e \quad \text{για κάθε } x \in \mathbb{R} \]
\begin{erwthma}
    \item Να αποδείξετε ότι η $f$ αντιστρέφεται και να βρείτε τον τύπο της αντίστροφης συνάρτησης $f^{-1}$. \monades{6}
    \item Να αποδείξετε ότι η $f$ είναι γνησίως αύξουσα και να βρείτε το πρόσημό της στο πεδίο ορισμού της. \monades{6}
    \item Να λύσετε την ανίσωση $f(e^{x-1} + x^3 - 1) > 1$. \monades{6}
    \item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x = 1 - e$ και $x = 1$. \monades{7}
\end{erwthma}
\end{thema}

\subsection{ΔΙΑΓΩΝΙΣΜΑ ΠΡΟΣΟΜΟΙΩΣΗΣ 2}
\begin{thema}{Α}
\begin{erwthma}
    \item Έστω μια συνάρτηση $f$ παραγωγίσιμη σε ένα διάστημα $(a, \beta)$, με εξαίρεση ίσως ένα σημείο του $x_0$, στο οποίο όμως η $f$ είναι συνεχής. Αν $f'(x) > 0$ στο $(a, x_0)$ και $f'(x) < 0$ στο $(x_0, \beta)$, τότε να αποδείξετε ότι το $f(x_0)$ είναι τοπικό μέγιστο της $f$. \monades{7}
    \item Πότε μία συνάρτηση $f$ λέγεται γνησίως αύξουσα σε ένα διάστημα $\Delta$? \monades{4}
    \item Τι ονομάζουμε αρχική συνάρτηση ή παράγουσα της $f$ στο $\Delta$? \monades{4}
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό},
αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη:
    \begin{alist}
        \item Αν η συνάρτηση $f$ είναι συνεχής στο $x_0$ και $f(x_0) < 0$, τότε $f(x) < 0$ για κάθε $x$ σε μια περιοχή του $x_0$.
        \item Αν η $f$ είναι παραγωγίσιμη στο $[a, \beta]$ με $f(a) = f(\beta)$, τότε απαραίτητα $\dint_{a}^{\beta} f'(x) \d x = 0$.
        \item Υπάρχουν συναρτήσεις που είναι $1-1$ αλλά δεν είναι γνησίως μονότονες.
        \item Κάθε κατακόρυφη ευθεία τέμνει τη γραφική παράσταση μιας συνάρτησης σε τουλάχιστον ένα σημείο.
        \item Αν $f''(x) > 0$ για κάθε $x \in \mathbb{R}$, τότε η $f$ δεν έχει σημεία καμπής.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}
\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = \dfrac{e^x}{e^x + 1}$ και $g(x) = \ln x$.
\begin{erwthma}
    \item Να βρείτε το πεδίο ορισμού και τον τύπο της συνάρτησης $h(x) = (f \circ g)(x)$. \monades{6}
    \item Αν $h(x) = \dfrac{x}{x+1}$ για $x > 0$, να αποδείξετε ότι η $h$ αντιστρέφεται και να προσδιορίσετε την αντίστροφη συνάρτηση $h^{-1}(x)$. \monades{6}
    \item Να υπολογίσετε τα όρια $\displaystyle\lim_{x \to 0^+} h(x)$ και $\displaystyle\lim_{x \to +\infty} h(x)$. \monades{6}
    \item Να αποδείξετε ότι η εξίσωση $h(x) = e^{-x}$ έχει ακριβώς μία ρίζα στο διάστημα $(0, 1)$. \monades{7}
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνεχής συνάρτηση $f: (0, +\infty) \to \mathbb{R}$ για την οποία ισχύει $f(e) = \frac{2}{e}$ και ικανοποιεί τη σχέση:
\[ x^2 f^2(x) - 2x \ln x \cdot f(x) + \ln^2 x - 1 = 0 \quad \text{για κάθε } x > 0 \]
\begin{erwthma}
    \item Δείξτε ότι ο τύπος της $f$ είναι $f(x) = \dfrac{\ln x + 1}{x}$. \monades{6}
    \item Να μελετήσετε τη συνάρτηση $f$ ως προς τη μονοτονία της και τα ακρότατά της.

 \monades{6}
    \item Να βρείτε, εάν υπάρχει, την οριζόντια ασύμπτωτη της γραφικής παράστασης της $f$ στο $+\infty$. \monades{6}
    \item Ένα σημείο $M(x(t), y(t))$ κινείται πάνω στη γραφική παράσταση της $f$ με τετμημένη ($x > 1$). Τη χρονική στιγμή $t_0$ που διέρχεται από τη θέση με τετμημένη $x(t_0) = e$, ο ρυθμός μεταβολής της τετμημένης του είναι $x'(t_0) = 2e^2 \, \text{μον/\si{sec}}$. Αν $E(t)$ είναι το εμβαδόν του ορθογωνίου με διαγώνιο την απόσταση του $M$ από την αρχή των αξόνων $O(0,0)$ (δηλαδή $E(t) = x(t)y(t)$), να βρείτε τον ρυθμό μεταβολής του εμβαδού $E(t)$ τη στιγμή $t_0$. \monades{7}
\end{erwthma}
\end{thema}
\begin{thema}{Δ}
Θεωρούμε την παραγωγίσιμη συνάρτηση $f: (0, +\infty) \to \mathbb{R}$ για την οποία ικανοποιείται η σχέση:
\[ f(x) + e^{f(x)} = e^{x-1} - \ln x + \frac{e^{e^{x-1}}}{x} \quad \text{για κάθε } x > 0 \]
\begin{erwthma}
    \item Να αποδείξετε ότι ο τύπος της συνάρτησης είναι $f(x) = e^{x-1} - \ln x$. \monades{6}
    \item Να αποδείξετε ότι για οποιαδήποτε $a, \beta \in \mathbb{R}$ με $1 < a < \beta$, ισχύει η ανισότητα \[\frac{f(\beta) - f(a)}{\beta - a} > e^{a-1} - \frac{1}{a}.\] \monades{6}
    \item Να αποδείξετε ότι υπάρχει ακριβώς ένα $x_0 \in (1, 2)$ τέτοιο ώστε η εφαπτομένη της γραφικής παράστασης $C_f$ στο σημείο $A(x_0, f(x_0))$ να διέρχεται από την αρχή των αξόνων $O(0,0)$. \monades{6}
    \item Να αποδείξετε την ανισότητα $\dint_1^e f(x) \d x < e^{e-1} - 1$. \monades{7}
\end{erwthma}
\end{thema}

\subsection{ΔΙΑΓΩΝΙΣΜΑ ΠΡΟΣΟΜΟΙΩΣΗΣ 3}
\begin{thema}{Α}
\begin{erwthma}
    \item Να αποδείξετε ότι αν μια συνάρτηση $f$ είναι παραγωγίσιμη σε ένα σημείο $x_0$ του πεδίου ορισμού της, τότε είναι και συνεχής στο σημείο αυτό. \monades{7}
    \item Τι ονομάζουμε ολικό μέγιστο (ή απλά μέγιστο) μιας συνάρτησης $f$ με πεδίο ορισμού $A$? \monades{4}
    \item Πότε το σημείο $A(x_0, f(x_0))$ λέγεται σημείο καμπής της γραφικής παράστασης της $f$? \monades{4}
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό},
αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη:
    \begin{alist}
        \item Αν $f(x) < g(x)$ για κάθε $x \in [a, \beta]$ όπου $f,g$ συνεχείς, τότε απαραίτητα $\dint_a^\beta f(x) \d x < \dint_a^\beta g(x) \d x$.
        \item Η παράγωγος $f'$ μιας παραγωγίσιμης συνάρτησης $f$ μηδενίζεται σε κάθε θέση τοπικού ακρότατου.
        \item Αν $\dint_a^\beta f(x)\d x = 0$, τότε υποχρεωτικά ισχύει $a = \beta$.
        \item Αν $f, g$ συναρτήσεις τέτοιες ώστε η $(g \circ f)(x)$ να ορίζεται, το πεδίο ορισμού της είναι η τομή των πεδίων ορισμού τους.
        \item Αν μία συνάρτηση είναι $1-1$, τότε η γραφική παράσταση της αντίστροφης $f^{-1}$ είναι συμμετρική της $C_f$ ως προς την ευθεία $y=x$.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \begin{cdcases} \frac{\sqrt{x^2+3} - 2}{x-1}, & x < 1 \\ a x + \beta, & x \ge 1 \end{cdcases}$.
\begin{erwthma}
    \item Να υπολογίσετε το όριο $\displaystyle \lim_{x \to 1^-} f(x)$. \monades{6}
    \item Αν η γραφική παράσταση της $f$ διέρχεται από το σημείο $\left(2,\frac{3}{4}\right)$ και επιπλέον είναι συνεχής στο $x_0 = 1$, να βρείτε τις τιμές των παραμέτρων $a$ και $\beta$. \monades{6}
    \item Για τις τιμές $a = \frac{1}{4}$ και $\beta = \frac{1}{4}$, να υπολογίσετε τα όρια $\displaystyle\lim_{x\to-\infty} f(x)$ και $\displaystyle\lim_{x\to+\infty} f(x)$. \monades{6}
    \item Να υπολογίσετε το εμβαδόν του χωρίου που περικλείεται από τη γραφική παράσταση της $f$, τον άξονα $x'x$ και τις κατακόρυφες ευθείες $x=1, x=3$. \monades{7}
\end{erwthma}
\end{thema}

\begin{thema}{Γ}

    \item Να μελετήσετε την $f$ ως προς την κυρτότητα στο πεδίο ορισμού της και να βρείτε την εξίσωση της εφαπτομένης της στο $x_0 = 1$. \monades{6}

    \item Ένα σημείο $M$ κινείται πάνω στη γραφική παράσταση της $f$ για $x > e$. Η τετμημένη του αυξάνεται με σταθερό ρυθμό $x'(t) = e \, \text{μον/\si{sec}}$.  Αν $\theta$ είναι η γωνία που σχηματίζει η $OM$ με τον θετικό ημιάξονα των $x$, να υπολογίσετε τον ρυθμό μεταβολής της γωνίας $\theta$, τη χρονική στιγμή $t_0$ που η τετμημένη του $M$ περνά από το σημείο $x = e^2$. \monades{7}

\begin{thema}{Α}
\begin{erwthma}
    \item Έστω το πολυώνυμο $P(x) = a_\nu x^\nu + a_{\nu-1} x^{\nu-1} + \dots + a_1 x + a_0$. Να αποδείξετε ότι ισχύει $\displaystyle\lim_{x\to x_0} P(x) = P(x_0)$. \monades{7}
    \item Πότε μια συνάρτηση $f$ λέγεται συνεχής σε ένα κλειστό διάστημα $[a, \beta]$? \monades{4}
    \item Πότε η ευθεία $y = \lambda x + \beta$ ονομάζεται πλάγια ασύμπτωτη της γραφικής παράστασης μιας συνάρτησης στο $+\infty$? \monades{4}
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό},

    \begin{alist}

        \item Μια συνάρτηση $1-1$, δεν μπορεί ποτέ να τέμνει τον άξονα των $x$ σε περισσότερα από $1$ σημεία.
        \item Το ολοκλήρωμα της σταθερής συνάρτησης $f(x)=c$ ισούται πάντοτε με το $c (b-a)$.
        \item Η γραφική παράσταση της $f^{-1}$ είναι συμμετρική της $C_f$ ως προς την ευθεία $y=-x$.
        \item Το θεώρημα \en{Bolzano} εξασφαλίζει την ύπαρξη πεπερασμένου πλήθους ριζών.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}

\begin{thema}{Γ}

 \monades{6}
    \item Υπολογίστε τον ρυθμό μεταβολής του εμβαδού τη χρονική στιγμή κατά την οποία η τετμημένη είναι ίση με $a=3$. \monades{6}
    \item Υπολογίστε, το ολοκλήρωμα $\dint_0^3 f(x)\d x$. 

\monades{6}

        \item Αν μια συνάρτηση $f$ είναι γνησίως μονότονη σε ένα διάστημα $\Delta$, τότε είναι υποχρεωτικά και $1-1$ στο $\Delta$.
        \item Ισχύει ότι $\left(\dint_a^\beta f(x)\d x\right)' = f(x)$, όπου $a,\beta$ είναι σταθερές και $f$ συνεχής συνάρτηση.
        \item Αν για μια συνεχή συνάρτηση $f$ ισχύει $f(x)\geq 0$ για κάθε $x \in [a,\beta]$ και $\dint_a^\beta f(x)\d x = 0$, τότε υποχρεωτικά $f(x)=0$ για κάθε $x \in [a,\beta]$.
        \item Η σύνθεση δύο $1-1$ συναρτήσεων, εφόσον ορίζεται, είναι πάντα συνάρτηση $1-1$.
        \item Κάθε συνεχής συνάρτηση $f$ σε ένα κλειστό διάστημα $[a,\beta]$ παίρνει υποχρεωτικά μια μέγιστη και μια ελάχιστη τιμή.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = e^x - 1$ και $\displaystyle g(x) = \ln\left(\frac{x+1}{x-1}\right)$.
\begin{erwthma}
    \item Να βρείτε τα πεδία ορισμού των συναρτήσεων $f$ και $g$. \monades{4}
    \item Να προσδιορίσετε τη συνάρτηση $h = f \circ g$. \monades{7}
    \item Έστω η συνάρτηση $\phi(x) = h(x)$ για $x \in (1, +\infty)$. Να αποδείξετε ότι η $\phi$ αντιστρέφεται και να βρείτε την αντίστροφή της, $\phi^{-1}$. \monades{7}
    \item Να υπολογίσετε το ορισμένο ολοκλήρωμα $\dint_2^3 \phi(x)\,\d x$. \monades{7}
\end{erwthma}
\end{thema}

\begin{thema}{Γ}
Δίνεται η συνάρτηση $f(x) = x\ln x - x + 1$, με $x>0$.
\begin{erwthma}
    \item Να μελετήσετε την $f$ ως προς τη μονοτονία και τα ακρότατα, και να αποδείξετε ότι $f(x)\geq 0$ για κάθε $x > 0$. \monades{6}
    \item Να βρείτε την εξίσωση της εφαπτομένης της γραφικής παράστασης της $f$ στο σημείο της με τετμημένη $x_0 = e$. \monades{6}
    \item Να αποδείξετε ότι η εξίσωση $x\ln x = x - 1$ έχει ακριβώς μία ρίζα στο $(0,+\infty)$. \monades{6}
    \item Ένα υλικό σημείο $M$ κινείται επί της $C_f$ και η τετμημένη του αυξάνεται με ρυθμό $x'(t)=1 \text{ μονάδα/sec}$. Να βρείτε τον ρυθμό μεταβολής της γωνίας $\theta$ που σχηματίζει η εφαπτομένη της $C_f$ στο $M$ με τον άξονα $x'x$, τη χρονική στιγμή που το σημείο διέρχεται από τη θέση με τετμημένη $x=e$. \monades{7}
\end{erwthma}
\end{thema}

\begin{thema}{Δ}
Δίνεται η παραγωγίσιμη συνάρτηση $f:\mathbb{R}\to\mathbb{R}$, για την οποία ισχύει $f(0)=0$ και $f'(x) + f(x) > 0$ για κάθε $x \in \mathbb{R}$.
\begin{erwthma}
    \item Θεωρούμε τη συνάρτηση $h(x)=e^x f(x)$. Να αποδείξετε ότι $h'(x) > 0$ και να συμπεράνετε ότι η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$. \monades{6}
    \item Να αποδείξετε ότι $f(x)>0$ για κάθε $x>0$ και $f(x)<0$ για κάθε $x<0$. \monades{6}
    \item Να αποδείξετε ότι η εξίσωση $f(x)=0$ έχει μοναδική ρίζα στο $\mathbb{R}$. \monades{6}
    \item Αν επιπλέον δίνεται ότι $f(1)=\frac{1}{e}$, να αποδείξετε ότι $0 < \dint_0^1 f(x)\,\d x < 1 - \frac{1}{e}$. \monades{7}
\end{erwthma}
\end{thema}
