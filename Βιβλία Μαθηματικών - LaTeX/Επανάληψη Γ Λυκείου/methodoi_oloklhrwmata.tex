\section{Ολοκληρωτικός Λογισμός}

\subsection{Υπολογισμός απλών ολοκληρωμάτων}
\begin{askhsh}{Βασικές Ιδιότητες Ολοκληρωμάτων}
Για τον υπολογισμό ενός απλού ορισμένου ολοκληρώματος $\dint_a^{\beta}{f(x)\, \d x}$ βασιζόμαστε κυρίως στις ιδιότητες γραμμικότητας και στον πίνακα των αρχικών συναρτήσεων.

\bhmata
\begin{bhma}
    \item Βρίσκουμε την αρχική συνάρτηση $F(x)$ της $f(x)$.
    \item Εφαρμόζουμε το Θεμελιώδες Θεώρημα του Ολοκληρωτικού Λογισμού: \[\dint_a^\beta f(x)\d x = [F(x)]_a^\beta = F(\beta) - F(a)\]
\end{bhma}
\end{askhsh}
Για το 1ο βήμα της μεθόδου θα χρειαστούμε τον παρακάτω πίνακα με τις παράγουσες των βασικών συναρτήσεων:
\begin{center}
\begin{mytblr}{cell{1}{3}={secondcolor!90!black},cell{even[2-Z]}{3} = {secondcolor!70!white},cell{odd[3-Z]}{3} = {secondcolor!30!white},column{3}={8.7cm},row{2-Z} = {10.7mm}}
Συνάρτηση & Παράγουσα & Ορισμένο ολοκλήρωμα\\
$c$ & $cx$ & $\displaystyle\int_{a}^{\beta}{c\mathrm{d}x}=[cx]_{a}^{\beta}=c(\beta-a)$\\
$x$ & $\dfrac{x^2}{2}$ & $\displaystyle\int_{a}^{\beta}{x\mathrm{d}x}=\left[\dfrac{x^2}{2}\right]_{a}^{\beta}=\dfrac{\beta^2}{2}-\dfrac{a^2}{2}$  \\
$x^{\nu}$ & $\dfrac{x^{\nu+1}}{\nu+1}$ & $\displaystyle\int_{a}^{\beta}{x^{\nu}\mathrm{d}x}=\left[\dfrac{x^{\nu+1}}{\nu+1}\right]_{a}^{\beta}=\dfrac{\beta^{\nu+1}}{\nu+1}-\dfrac{a^{\nu+1}}{\nu+1}$\\
$\dfrac{1}{2\sqrt{x}}$ & $\sqrt{x}$ & $\displaystyle\int_{a}^{\beta}{\frac{1}{2\sqrt{x}}\mathrm{d}x}=\left[\sqrt{x}\right]_{a}^{\beta}=\sqrt{\beta}-\sqrt{a}$ \\
$\sqrt[\nu]{x^{\mu}}$ & $\dfrac{x^{\frac{\mu}{\nu}+1}}{\frac{\mu}{\nu}+1}$ & $\displaystyle\int_{a}^{\beta}{\sqrt[\nu]{x^{\nu}}\mathrm{d}x}=\left[\dfrac{x^{\frac{\mu}{\nu}+1}}{\frac{\mu}{\nu}+1}\right]_{a}^{\beta}=\dfrac{\beta^{\frac{\mu}{\nu}+1}}{\frac{\mu}{\nu}+1}-\dfrac{a^{\frac{\mu}{\nu}+1}}{\frac{\mu}{\nu}+1}$ \\
$\dfrac{1}{x^2}$ & $-\dfrac{1}{x}$ & $\displaystyle\int_{a}^{\beta}{\frac{1}{x^2}\mathrm{d}x}=\left[-\dfrac{1}{x}\right]_{a}^{\beta}=-\dfrac{1}{\beta}+\dfrac{1}{a}$ \\
$\hm{x}$ & $-\syn{x}$ & $\displaystyle\int_{a}^{\beta}{\hm{x}\mathrm{d}x}=\left[-\syn{x}\right]_{a}^{\beta}=-\syn{\beta}+\syn{a}$ \\
$\syn{x}$ & $\hm{x}$ & $\displaystyle\int_{a}^{\beta}{\syn{x}\mathrm{d}x}=\left[\hm{x}\right]_{a}^{\beta}=\hm{\beta}-\hm{a}$  \\
$\dfrac{1}{\syn^2{x}}$ & $\ef{x}$ & $\displaystyle\int_{a}^{\beta}{\frac{1}{\syn^2{x}}\mathrm{d}x}=\left[\ef{x}\right]_{a}^{\beta}=\ef{\beta}-\ef{a}$\\
$\dfrac{1}{\hm^2{x}}$ & $-\syf{x}$ & $\displaystyle\int_{a}^{\beta}{\frac{1}{\hm^2{x}}\mathrm{d}x}=\left[-\syf{x}\right]_{a}^{\beta}=-\syf{\beta}+\syf{a}$\\
$e^x$ & $e^x$ & $\displaystyle\int_{a}^{\beta}{e^x\mathrm{d}x}=\left[e^{x}\right]_{a}^{\beta}=e^{\beta}-e^{a}$ \\
$a^x$ & $\dfrac{a^x}{\ln{a}}$ & $\displaystyle\int_{a}^{\beta}{{\mu}^x\mathrm{d}x}=\left[\frac{{\mu}^x}{\ln{\mu}}\right]_{a}^{\beta}=\frac{{\mu}^{\beta}}{\ln{\mu}}-\frac{\mu^{a}}{\ln{\mu}}$  \\
$\dfrac{1}{x}$ & $\ln{|x|}$ & $\displaystyle\int_{a}^{\beta}{\frac{1}{x}\mathrm{d}x}=\left[\ln{|x|}\right]_{a}^{\beta}=\ln|\beta|-\ln{|a|}$
\end{mytblr}
\end{center}
\Paradeigma{Απλά ολοκληρώματα}
\bmath{Να υπολογιστεί το ολοκλήρωμα $I = \dint_1^2 \left( 3x^2 - \dfrac{4}{x^2} + \dfrac{2}{x} - e^x \right) \d x$.}

\lysh
Έχουμε:
\begin{align*}
I &= \dint_1^2 \left( 3x^2 - \frac{4}{x^2} + \frac{2}{x} - e^x \right) \d x=\\&= 3 \left[\frac{x^3}{3} - 4 \frac{x^{-1}}{-1}+ 2 \ln|x|-e^x\right]_1^2=\left[\frac{x^3}{3} +\frac{4}{x}+ 2 \ln{x}-e^x\right]_1^2=\\&=\left(2^3+\frac{4}{2}+2\ln{2}-e^2\right)-\left(1^3+\frac{4}{1}+2\ln{1}-e^1\right)=5+2\ln{2}-e^2+e
\end{align*}

Ομοίως εργαζόμαστε και για τον υπολογισμό ολοκληρωμάτων που περιέχουν σύνθετες συναρτήσεις, οι παράγουσες των οποίων βρίσκονται στον παρακάτω πίνακα:
\begin{center}
\begin{mytblr}{row{2-Z} = {10.7mm},cell{1}{2,4}={secondcolor!90!black},cell{even[2-Z]}{2,4} = {secondcolor!70!white},cell{odd[3-Z]}{2,4} = {secondcolor!30!white}}
Συνάρτηση & Παράγουσα & Συνάρτηση & Παράγουσα \\
$f^{\nu}(x)\cdot f'(x)$ & $\dfrac{f^{\nu+1}(x)}{\nu+1}$&
$\dfrac{f'(x)}{2\sqrt{f(x)}}$ & $\sqrt{f(x)}$\\
$\dfrac{f'(x)}{f^2(x)}$ & $-\dfrac{1}{f(x)}$&
$\hm{f(x)}\cdot f'(x)$ & $-\syn{f(x)}$\\
$\syn{f(x)}\cdot f'(x)$ & $\hm{f(x)}$&
$\dfrac{f'(x)}{\syn^2{f(x)}}$ & $\ef{f(x)}$\\
$\dfrac{f'(x)}{\hm^2{f(x)}}$ & $-\syf{f(x)}$&
$e^{f(x)}\cdot f'(x)$ & $e^{f(x)}$\\
$a^{f(x)}\cdot f'(x)$ & $\dfrac{a^{f(x)}}{\ln{a}}$&
$\dfrac{f'(x)}{f(x)}$ & $\ln{f(x)}$
\end{mytblr}
\end{center}
\Paradeigma{Ολοκληρώματα σύνθετων συναρτήσεων}
\bmath{Να υπολογιστούν τα ολοκληρώματα:
\begin{multicols}{2}
\begin{alist}
\item $I = \dint_0^1 (x^2+1)^3 \cdot 2x \, \d x$
\item $I = \dint_1^e \dfrac{3x^2+1}{x^3+x} \, \d x$
\item $I = \dint_0^1 (2x+1)e^{x^2+x} \, \d x$
\item $I = \dint_0^{\frac{\pi}{4}} \dfrac{\hm{x}}{\syn^2{x}} \, \d x$
\end{alist}
\end{multicols}
}
\lysh\vspace{-3mm}
\begin{alist}
\item Παρατηρούμε ότι ο παράγοντας $2x$ είναι η παράγωγος της βάσης της δύναμης $x^2+1$, οπότε το ολοκλήρωμα είναι της μορφής $f^3(x)\cdot f'(x)$, με παράγουσα $\dfrac{f^4(x)}{4}$:
\begin{align*}
I &= \dint_0^1 (x^2+1)^3 \cdot 2x \, \d x = \left[\frac{(x^2+1)^4}{4}\right]_0^1 = \frac{(1+1)^4}{4} - \frac{(0+1)^4}{4} = \frac{16}{4} - \frac{1}{4} = \frac{15}{4}
\end{align*}

\item Παρατηρούμε ότι ο αριθμητής $3x^2+1$ είναι η παράγωγος του παρονομαστή $f(x)=x^3+x$, οπότε το ολοκλήρωμα είναι της μορφής $\dfrac{f'(x)}{f(x)}$, με παράγουσα $\ln|f(x)|$:
\begin{align*}
I &= \dint_1^e \frac{3x^2+1}{x^3+x} \, \d x = \left[\ln|x^3+x|\right]_1^e = \ln(e^3+e) - \ln(1+1) = \ln\bigl(e(e^2+1)\bigr) - \ln 2 \\
&= 1 + \ln(e^2+1) - \ln 2 = 1 + \ln\!\left(\frac{e^2+1}{2}\right)
\end{align*}

\item Βλέπουμε όπως και πριν ότι ο παράγοντας $2x+1$ είναι η παράγωγος του εκθέτη $x^2+x$, οπότε το ολοκλήρωμα είναι της μορφής $e^{f(x)}\cdot f'(x)$, με παράγουσα $e^{f(x)}$:
\begin{align*}
I &= \dint_0^1 (2x+1)e^{x^2+x} \, \d x = \left[e^{x^2+x}\right]_0^1 = e^{1+1} - e^{0+0} = e^2 - 1
\end{align*}

\item Έχουμε ότι $\dfrac{\hm{x}}{\syn^2{x}}=-\dfrac{(\syn{x})'}{\syn^2{x}}$, οπότε το ολοκλήρωμα είναι της μορφής $-\dfrac{f'(x)}{f^2(x)}$, με παράγουσα $\dfrac{1}{f(x)}$:
\begin{align*}
I &= \dint_0^{\frac{\pi}{4}} \frac{\hm{x}}{\syn^2{x}} \, \d x = \left[\dfrac{1}{\syn{x}}\right]_0^{\frac{\pi}{4}} = \frac{1}{\syn{\frac{\pi}{4}}}-\frac{1}{\syn{0}}=\sqrt{2}-1
\end{align*}
\end{alist}

\subsection{Παραγοντική ολοκλήρωση}
\begin{askhsh}{Μέθοδος Παραγοντικής Ολοκλήρωσης}
Αν μέσα σε ένα ολοκλήρωμα έχουμε γινόμενο δύο συναρτήσεων $f(x)\cdot g(x)$, για να εφαρμοστεί ο κανόνας της παραγοντικής ολοκλήρωσης τότε

\bhmata
\begin{bhma}
\item γράφουμε μία από τις δύο συναρτήσεις ως παράγωγο, βρίσκοντας την αρχική της:
\[\int_{a}^{\beta}{f(x)\cdot g(x)\, \d x}=\int_{a}^{\beta}{F'(x)\cdot g(x)\, \d x}\]
\item Εφαρμόζουμε τον κανόνα της παραγοντικής ολοκλήρωσης. Επαναλαμβάνουμε όσες φορές χρειάζεται.
\[\int_{a}^{\beta}{F'(x)g(x)\d x}=[F(x)g(x)]_a^\beta-\int_{a}^{\beta}{F(x)g'(x)\d x}\]
\end{bhma}
\end{askhsh}
\Paradeigma{Ολοκληρώματα της μορφής $P(x)\cdot e^{\kappa x+\lambda}$}
\bmath{Να υπολογιστεί το ακόλουθο ολοκλήρωμα:
\[\int_{1}^{2}{x^2e^x\, \d x}\]
}
\lysh
Γράφουμε $e^x = (e^x)'$ οπότε είναι:
\begin{align*}
I &= \dint_{1}^{2}{x^2e^x\, \d x} = \dint_{1}^{2}{x^2(e^x)'\, \d x} \\
&= \left[x^2e^x\right]_1^2 - \dint_{1}^{2}{(x^2)'e^x\, \d x} = (4e^2 - e) - \dint_{1}^{2}{2xe^x\, \d x} \\
&= 4e^2 - e - 2\dint_{1}^{2}{x(e^x)'\, \d x} = 4e^2 - e - 2\left( \left[xe^x\right]_1^2 - \dint_{1}^{2}{(x)'e^x\, \d x} \right) \\
&= 4e^2 - e - 2\left( (2e^2 - e) - \dint_{1}^{2}{e^x\, \d x} \right) = 4e^2 - e - 4e^2 + 2e + 2\left[e^x\right]_1^2 \\
&= e + 2(e^2 - e) = 2e^2 - e
\end{align*}

\Paradeigma{Ολοκληρώματα της μορφής $P(x)\cdot \hm{(\kappa x+\lambda)}$ ή $P(x)\cdot \syn{(\kappa x+\lambda)}$}
\bmath{Να υπολογιστεί το ακόλουθο ολοκλήρωμα:
\[\int_{0}^{\pi}{x^3\hm{(2x)}\, \d x}\]
}
\lysh
Σε αυτά τα ολοκληρώματα ενσωματώνουμε το τριγωνομετρικό μέρος στην παράγωγο. Εφαρμόζουμε τον κανόνα 3 φορές για να μειωθεί ο βαθμός του πολυωνύμου:
\begin{align*}
I &= \dint_{0}^{\pi}{x^3\hm{(2x)}\, \d x} = \dint_{0}^{\pi}{x^3\left(-\frac{\syn{(2x)}}{2}\right)'\, \d x} \\
&= \left[-x^3\frac{\syn{(2x)}}{2}\right]_0^{\pi} - \dint_{0}^{\pi}{(x^3)'\left(-\frac{\syn{(2x)}}{2}\right)\, \d x} = \left(-\frac{\pi^3 \syn{(2\pi)}}{2} - 0\right) + \frac{3}{2}\dint_{0}^{\pi}{x^2\syn{(2x)}\, \d x} \\
&= -\frac{\pi^3}{2} + \frac{3}{2}\dint_{0}^{\pi}{x^2\left(\frac{\hm{(2x)}}{2}\right)'\, \d x} = -\frac{\pi^3}{2} + \frac{3}{4}\left[x^2\hm{(2x)}\right]_0^{\pi} - \frac{3}{4}\dint_{0}^{\pi}{(x^2)'\hm{(2x)}\, \d x} \\
&= -\frac{\pi^3}{2} + \frac{3}{4}(0 - 0) - \frac{3}{2}\dint_{0}^{\pi}{x\hm{(2x)}\, \d x} = -\frac{\pi^3}{2} - \frac{3}{2}\dint_{0}^{\pi}{x\left(-\frac{\syn{(2x)}}{2}\right)'\, \d x} \\
&= -\frac{\pi^3}{2} + \frac{3}{4}\left[x\syn{(2x)}\right]_0^{\pi} - \frac{3}{4}\dint_{0}^{\pi}{(x)'\syn{(2x)}\, \d x} = -\frac{\pi^3}{2} + \frac{3}{4}\left(\pi\syn{(2\pi)} - 0\right) - \frac{3}{4}\left[\frac{\hm{(2x)}}{2}\right]_0^{\pi} \\
&= -\frac{\pi^3}{2} + \frac{3\pi}{4} - \frac{3}{8}(0 - 0) = \frac{3\pi - 2\pi^3}{4}
\end{align*}

\Paradeigma{Ολοκληρώματα της μορφής $P(x)\cdot\ln{(ax)}$}
\bmath{Να υπολογιστεί το ακόλουθο ολοκλήρωμα:
\[\int_{1}^{e}{x\ln{x}\, \d x}\]
}
\lysh
Σε αυτά τα ολοκληρώματα, βρίσκουμε αρχική του πολυωνύμου. Έχουμε:
\begin{align*}
I &= \dint_1^e x \ln{x} \, \d x = \dint_1^e \left(\frac{x^2}{2}\right)' \ln{x} \, \d x \\
&= \left[\frac{x^2}{2} \ln{x}\right]_1^e - \dint_1^e \frac{x^2}{2} (\ln{x})' \, \d x \\
&= \left( \frac{e^2}{2}\ln{e} - \frac{1^2}{2}\ln{1} \right) - \dint_1^e \frac{x^2}{2} \cdot \frac{1}{x} \, \d x \\
&= \frac{e^2}{2} - \frac{1}{2}\dint_1^e x \, \d x = \frac{e^2}{2} - \frac{1}{2} \left[\frac{x^2}{2}\right]_1^e \\
&= \frac{e^2}{2} - \frac{1}{2} \left( \frac{e^2}{2} - \frac{1}{2} \right) = \frac{e^2}{2} - \frac{e^2}{4} + \frac{1}{4} = \frac{e^2 + 1}{4}
\end{align*}

\Paradeigma{Ολοκληρώματα της μορφής $e^{\mu x+\nu}\cdot \hm{(\kappa x+\lambda)}$ ή $e^{\mu x+\nu}\cdot \syn{(\kappa x+\lambda)}$}
\bmath{Να υπολογιστεί το ακόλουθο ολοκλήρωμα:
\[\int_{0}^{\pi}{e^x\hm{x}\, \d x}\]
}
\lysh
Αυτά τα ολοκληρώματα είναι \textbf{κυκλικά}. Εφαρμόζουμε την παραγοντική ολοκλήρωση δύο φορές, με αποτέλεσμα να εμφανιστεί ξανά το αρχικό ολοκλήρωμα $I$, το οποίο στη συνέχεια λύνουμε ως προς $I$.
\begin{align*}
I &= \dint_{0}^{\pi}{e^x \hm{x}\, \d x} = \dint_{0}^{\pi}{(e^x)' \hm{x}\, \d x} = \left[e^x \hm{x}\right]_0^{\pi} - \dint_{0}^{\pi}{e^x (\hm{x})'\, \d x} \\
&= (e^{\pi}\hm{\pi} - e^0\hm{0}) - \dint_{0}^{\pi}{e^x \syn{x}\, \d x} = 0 - \dint_{0}^{\pi}{(e^x)' \syn{x}\, \d x} \\
&= -\left( \left[e^x \syn{x}\right]_0^{\pi} - \dint_{0}^{\pi}{e^x (\syn{x})'\, \d x} \right) = -\left( (e^{\pi}\syn{\pi} - e^0\syn{0}) - \dint_{0}^{\pi}{e^x (-\hm{x})\, \d x} \right) \\
&= -\left( -e^{\pi} - 1 + \dint_{0}^{\pi}{e^x \hm{x}\, \d x} \right) = e^{\pi} + 1 - I
\end{align*}
Έτσι καταλήγουμε στην εξίσωση:
\[ I = e^{\pi} + 1 - I \iff 2I = e^{\pi} + 1 \iff I = \frac{e^{\pi} + 1}{2} \]

\subsection{Ολοκλήρωση με αντικατάσταση}
\begin{askhsh}{Αλλαγή μεταβλητής στο Ορισμένο Ολοκλήρωμα}
Χρησιμοποιούμε αντικατάσταση κυρίως όταν έχουμε μια σύνθετη συνάρτηση πολλαπλασιασμένη με την παράγωγο της εσωτερικής συνάρτησης (ή κάτι το ανάλογο που μπορεί να κατασκευαστεί).

\bhmata
\begin{bhma}
    \item Θέτουμε τη βοηθητική μεταβλητή (συνήθως το "μέσα" μέρος) ίση με μια νέα μεταβλητή $u$: π.χ. $u = g(x)$.
    \item Υπολογίζουμε με παραγώγιση: $\d u = g'(x)\d x$ οπότε αντικαθιστούμε έμμεσα και το $\d x$.
    \item Αλλάζουμε τα όρια ολοκλήρωσης με βάση τη σχέση ισότητας $u=g(x)$:
    \begin{itemize}
        \item Για το κάτω όριο: $x = a \Rightarrow u_1 = g(a)$
        \item Για το άνω όριο: $x = \beta \Rightarrow u_2 = g(\beta)$
    \end{itemize}
    \item Αντικαθιστούμε τα πάντα στο ολοκλήρωμα, απλοποιούμε ώστε τα $x$ να εξαφανιστούν, και λύνουμε ως προς τη νέα μεταβλητή $u$.
\end{bhma}
\end{askhsh}

\Paradeigma{Ολοκληρώματα με ριζικά $\sqrt[n]{g(x)}$}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I = \dint_0^3 \frac{x}{\sqrt{x+1}} \, \d x \]
}
\lysh
Σε ολοκληρώματα όπου εμφανίζεται ρίζα και είναι δύσκολο να βρούμε κατευθείαν παράγουσα, μια πολύ καλή επιλογή είναι να θέσουμε ολόκληρη τη ρίζα (ή το υπόρριζο) ως μια νέα μεταβλητή $u$. 
Θέτουμε $u = \sqrt{x+1}$.
Υψώνοντας στο τετράγωνο, έχουμε $u^2 = x+1 \Rightarrow x = u^2 - 1$. Παραγωγίζοντας και τα δύο μέλη (το αριστερό ως προς $u$ και το δεξί ως προς $x$), παίρνουμε :
$2u \, \d u = 1 \, \d x \Rightarrow \d x = 2u \, \d u$.
\begin{itemize}
\item Για $x = 0$, το νέο όριο είναι $u = \sqrt{0+1} = 1$.
\item Για $x = 3$, το νέο όριο είναι $u = \sqrt{3+1} = 2$.
\end{itemize}
οπότε το ολοκλήρωμα γράφεται:
\begin{align*}
I &= \dint_0^3 \frac{x}{\sqrt{x+1}} \, \d x = \dint_1^2 \frac{u^2 - 1}{u} \cdot 2u \, \d u \\
&= \dint_1^2 2(u^2 - 1) \, \d u = 2\left[ \frac{u^3}{3} - u \right]_1^2 \\
&= 2\left( \left( \frac{8}{3} - 2 \right) - \left( \frac{1}{3} - 1 \right) \right)= 2\left( \frac{2}{3} - \left( -\frac{2}{3} \right) \right) = 2 \left( \frac{4}{3} \right) = \frac{8}{3}
\end{align*}

\Paradeigma{Ολοκληρώματα με δυνάμεις της μορφής $(x+a)^n$}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I = \dint_1^2 x(x-1)^4 \, \d x \]
}
\lysh
Θέτουμε $u = x-1\Leftrightarrow x = u+1$. Παραγωγίζουμε $\d x = \d u$.
\begin{itemize}
\item Για $x = 1$, έχουμε $u = 1 - 1 = 0$.
\item Για $x = 2$, έχουμε $u = 2 - 1 = 1$.
\end{itemize}
Άρα αντικαθιστώντας παίρνουμε:
\begin{align*}
I &= \dint_1^2 x(x-1)^4 \, \d x = \dint_0^1 (u+1)u^4 \, \d u \\
&= \dint_0^1 (u^5 + u^4) \, \d u = \left[ \frac{u^6}{6} + \frac{u^5}{5} \right]_0^1 = \left( \frac{1}{6} + \frac{1}{5} \right) - 0 = \frac{5}{30} + \frac{6}{30} = \frac{11}{30}
\end{align*}

\Paradeigma{Τριγωνομετρικές αντικαταστάσεις}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I = \dint_0^{\frac{\pi}{2}} \frac{\syn{x}}{(1+\hm{x})^2} \, \d x \]
}
\lysh
Όταν έχουμε μια τριγωνομετρική συνάρτηση στον παρονομαστή ή σε κάποια δύναμη, και στον αριθμητή (ή δίπλα στο $\d x$) υπάρχει η παράγωγός της, είναι κλασική περίπτωση αντικατάστασης.

Θέτουμε $u = 1 + \hm{x}$.
Τότε παραγωγίζοντας είναι $\d u = (1+\hm{x})' \, \d x \Rightarrow \d u = \syn{x} \, \d x$. Παρατηρούμε ότι το $\d u$ «κρύβεται» ακριβώς στον αριθμητή μας.
\begin{itemize}
\item Για $x = 0$, $u = 1 + \hm{0} = 1 + 0 = 1$.
\item Για $x = \dfrac{\pi}{2}$, $u = 1 + \hm{\dfrac{\pi}{2}} = 1 + 1 = 2$.
\end{itemize}
άρα έχουμε:
\begin{align*}
I &= \dint_0^{\frac{\pi}{2}} \frac{1}{(1+\hm{x})^2} \cdot \syn{x} \, \d x = \dint_1^2 \frac{1}{u^2} \, \d u \\
&= \dint_1^2 u^{-2} \, \d u = \left[ \frac{u^{-1}}{-1} \right]_1^2 = \left[ -\frac{1}{u} \right]_1^2 = \left( -\frac{1}{2} \right) - \left( -\frac{1}{1} \right) = 1 - \frac{1}{2} = \frac{1}{2}
\end{align*}

\Paradeigma{Διάφορα ολοκληρώματα}
\bmath{Να υπολογιστούν τα ολοκληρώματα:
\begin{multicols}{2}
\begin{alist}
\item $ I = \dint_1^e \dfrac{\ln{x}}{x} \, \d x $ 
\item $ I = \dint_0^1 \dfrac{e^x}{e^x + 1} \, \d x $
\end{alist}
\end{multicols}
}
\lysh\vspace{-5mm}
\begin{alist}
\item Συχνά όταν βλέπουμε το $\ln{x}$ μέσα σε ένα ολοκλήρωμα και ταυτόχρονα το $x$ στον παρονομαστή, δηλαδή το $\dfrac{1}{x}$ που είναι η παράγωγός του, θέτουμε τη λογαριθμική συνάρτηση ίση με $u$.

Θέτουμε $u = \ln{x}$. Παραγωγίζοντας είναι $\d u = (\ln{x})' \, \d x \Rightarrow \d u = \dfrac{1}{x} \, \d x$.\\
Αλλάζουμε τα όρια: για $x=1 \Rightarrow u=\ln{1}=0$, και για $x=e \Rightarrow u=\ln{e}=1$.
\begin{align*}
I &= \dint_1^e \frac{\ln{x}}{x} \, \d x = \dint_1^e \ln{x} \cdot \frac{1}{x} \, \d x = \dint_0^1 u \, \d u \\
&= \left[ \frac{u^2}{2} \right]_0^1 = \frac{1^2}{2} - \frac{0^2}{2} = \frac{1}{2}
\end{align*}

\item 
Ομοίως με τις εκθετικές συναρτήσεις. Εδώ ο αριθμητής είναι ακριβώς η παράγωγος του παρονομαστή.

Θέτουμε $u = e^x + 1$. Παραγωγίζοντας είναι $\d u = (e^x + 1)' \, \d x \Rightarrow \d u = e^x \, \d x$.\\
Αλλάζουμε τα όρια: για $x=0 \Rightarrow u=e^0 + 1 = 2$, και για $x=1 \Rightarrow u=e^1 + 1 = e+1$.
\begin{align*}
I &= \dint_0^1 \frac{e^x}{e^x + 1} \, \d x = \dint_2^{e+1} \frac{1}{u} \, \d u \\
&= \left[ \ln|u| \right]_2^{e+1} = \ln(e+1) - \ln{2} = \ln\left(\frac{e+1}{2}\right)
\end{align*}
\end{alist}

\subsection{Ολοκληρώματα ρητών συναρτήσεων}
\begin{askhsh}{Ανάλυση σε απλά κλάσματα}
Όταν έχουμε να ολοκληρώσουμε συνάρτηση της μορφής ρητού κλάσματος $f(x) = \frac{P(x)}{Q(x)}$ επεμβαίνουμε στους βαθμούς:

\bhmata
\begin{bhma}
    \item \textbf{Πλεονάζων Βαθμός:} Αν ο βαθμός του αριθμητή $P(x)$ είναι $ \ge $ παρονομαστή $Q(x)$, κάνουμε Ευκλείδεια Διαίρεση πολυωνύμων: $P(x) = \pi(x)Q(x) + \upsilon(x)$. Τότε το κλάσμα γράφεται $\dfrac{P(x)}{Q(x)} = \pi(x) + \dfrac{\upsilon(x)}{Q(x)}$.
    \item \textbf{Παραγοντοποίηση:} Αν ο παρονομαστής $Q(x)$ είναι 2ου βαθμού με πραγματικές, διακριτές ρίζες $\rho_1, \rho_2$ ($\Delta > 0$), τον αναλύουμε πλήρως. π.χ. $Q(x) = (x-\rho_1)(x-\rho_2)$.
    \item \textbf{Ανάλυση:} Χωρίζουμε το κλάσμα σε μερικά αθροίσματα: $\dfrac{A}{x-\rho_1} + \dfrac{B}{x-\rho_2}$. Κάνουμε ομώνυμα και εξισώνουμε τους νέους αριθμητές με τον αρχικό για να βρούμε ακριβώς τους πραγματικούς αριθμούς $A$ και $B$.
    \item \textbf{Ολοκλήρωση:} Ολοκληρώνουμε τους όρους ξεχωριστά.
\end{bhma}
\end{askhsh}

\Paradeigma{Ρητό ολοκλήρωμα - Διάσπαση κλάσματος}
\bmath{Να υπολογιστεί το ολοκλήρωμα $I = \dint_2^3 \dfrac{2x - 1}{x^2 - x} \d x$.}

\lysh
Ο βαθμός του αριθμητή είναι μικρότερος του παρονομαστή, άρα δεν απαιτείται διαίρεση. Ο παρονομαστής παραγοντοποιείται: $x^2 - x = x(x-1)$.
\[ \dfrac{2x - 1}{x(x-1)} = \dfrac{A}{x} + \dfrac{B}{x-1} \Rightarrow 2x - 1 = A(x-1) + Bx \Rightarrow 2x - 1 = (A+B)x - A \]
Από την ισότητα των ομοίων όρων των πολυωνύμων έχουμε το σύστημα:
$A+B = 2$ \ (για τον όρο $x$)\\ 
$-A = -1$ \ (για το σταθερό όρο), 
οπότε $A = 1$ και με αντικατάσταση $B = 1$.
Επομένως η συνάρτηση γράφεται: $\dfrac{2x-1}{x^2-x} = \dfrac{1}{x} + \dfrac{1}{x-1}$.
\[ I = \dint_2^3 \left( \frac{1}{x} + \frac{1}{x-1} \right) \d x = \left[\ln|x| + \ln|x-1|\right]_2^3 = (\ln 3 + \ln 2) - (\ln 2 + \ln 1) = \ln 3 \]

\Paradeigma{Ρητό ολοκλήρωμα (με διαίρεση)}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I = \dint_2^3 \frac{x^2 - 3x + 3}{x - 1} \, \d x \]
}
\lysh
Εδώ ο βαθμός του αριθμητή (2) είναι μεγαλύτερος από το βαθμό του παρονομαστή (1). Επομένως, θα εκτελέσουμε την Ευκλείδεια διαίρεση: $(x^2 - 3x + 3) : (x - 1)$. Κάνοντας τη διαίρεση, βρίσκουμε:
\begin{center}
\Horner[remainder=secondcolor,box=secondcolor]{1,-3,3}{1}
\end{center}
Άρα, από την ταυτότητα της διαίρεσης $P(x) = \delta(x)\pi(x) + \upsilon(x)=(x-1)(x-2)+1$, το κλάσμα γράφεται:
\[ \frac{x^2 - 3x + 3}{x - 1} =\dfrac{(x-1)(x-2)+1}{x-1}= (x - 2) + \frac{1}{x - 1} \]
Επομένως το ολοκλήρωμα γράφεται:
\begin{align*}
I &= \dint_2^3 \left( x - 2 + \frac{1}{x - 1} \right) \, \d x = \left[ \frac{x^2}{2} - 2x + \ln|x - 1| \right]_2^3 \\
&= \left( \frac{3^2}{2} - 2\cdot3 + \ln|3 - 1| \right) - \left( \frac{2^2}{2} - 2\cdot2 + \ln|2 - 1| \right) \\
&= \left( \frac{9}{2} - 6 + \ln{2} \right) - (2 - 4 + \ln{1}) \\
&= \left( -\frac{3}{2} + \ln{2} \right) - (-2 + 0) = -\frac{3}{2} + 2 + \ln{2} = \frac{1}{2} + \ln{2}
\end{align*}

\Paradeigma{Ρητό ολοκλήρωμα με διπλή ρίζα στον παρονομαστή ($\Delta = 0$)}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I_3 = \dint_0^1 \frac{2x - 1}{(x - 2)^2} \, \d x \]
}
\lysh
Ο παρονομαστής έχει διπλή ρίζα το $x=2$. Αντί για κλασική ανάλυση σε κλάσματα, ένας καλός και γρήγορος τρόπος είναι να «εμφανίσουμε» τον παράγοντα $(x-2)$ στον αριθμητή. Εναλλακτικά, θα μπορούσαμε να θέσουμε $u = x - 2$. Προσπαθούμε να δημιουργήσουμε το $x-2$ στον αριθμητή:
\[ 2x - 1 = 2(x - 2) + 4 - 1 = 2(x - 2) + 3 \]
Άρα το κλάσμα διασπάται ως εξής:
\[ \frac{2x - 1}{(x - 2)^2} = \frac{2(x - 2) + 3}{(x - 2)^2} = \frac{2(x - 2)}{(x - 2)^2} + \frac{3}{(x - 2)^2} = \frac{2}{x - 2} + \frac{3}{(x - 2)^2} \]

Γνωρίζοντας ότι $(x-2)^{-2} = \left[ \dfrac{(x-2)^{-1}}{-1}\right]' = \left[ -\dfrac{1}{x-2}\right]' $, προκύπτει:
\begin{align*}
I_3 &= \dint_0^1 \left( \frac{2}{x - 2} + 3(x - 2)^{-2} \right) \, \d x = \left[ 2\ln|x - 2| - \frac{3}{x - 2} \right]_0^1 \\
&= \left( 2\ln|1 - 2| - \frac{3}{1 - 2} \right) - \left( 2\ln|0 - 2| - \frac{3}{0 - 2} \right) \\
&= \left( 2\ln1 - \frac{3}{-1} \right) - \left( 2\ln2 - \left(-\frac{3}{2}\right) \right) = 3 - \frac{3}{2} - 2\ln{2} = \frac{3}{2} - 2\ln{2}
\end{align*}
\textit{Σημείωση:} Το ίδιο ακριβώς αποτέλεσμα θα προέκυπτε με την αντικατάσταση $u = x - 2 \Rightarrow \d x = \d u$.\\

\Paradeigma{Ρητό ολοκλήρωμα με $\Delta < 0$ (Αναγωγή σε λογάριθμο)}
\bmath{Να υπολογιστεί το ολοκλήρωμα:
\[ I_4 = \dint_1^2 \frac{4x + 2}{x^2 + x + 1} \, \d x \]
}
\lysh
Όταν το τριώνυμο του παρονομαστή έχει $\Delta < 0$ (εδώ $\Delta = 1^2 - 4\cdot 1\cdot 1 = -3$), δεν παραγοντοποιείται σε πραγματικούς αριθμούς. Σε αυτές τις περιπτώσεις, στα πλαίσια της ύλης της Γ' Λυκείου, ο αριθμητής αποτελεί σχεδόν πάντα πολλαπλάσιο της παραγώγου του παρονομαστή. 
Η παράγωγος του παρονομαστή είναι $(x^2 + x + 1)' = 2x + 1$.
Ο αριθμητής είναι $4x + 2 = 2(2x + 1)$. Επομένως παρατηρούμε πως:
\[ \frac{4x + 2}{x^2 + x + 1} = 2 \frac{(x^2 + x + 1)'}{x^2 + x + 1} \]

Έχουμε:
\begin{align*}
I_4 &= \dint_1^2 2 \frac{2x + 1}{x^2 + x + 1} \, \d x = 2 \left[ \ln|x^2 + x + 1| \right]_1^2 \\
&= 2(\ln(2^2 + 2 + 1) - \ln(1^2 + 1 + 1)) = 2\ln\left(\frac{7}{3}\right)
\end{align*}

\subsection{Ολοκληρώματα και Ανισότητες}
\begin{askhsh}{Φράγματα εμβαδών μέσω μονοτονίας και ακροτάτων}
\bhmata
\begin{bhma}
    \item Εάν μας ζητείται να αποδείξουμε μια ανισοτική σχέση φραγής εμβαδού από σταθερές τιμές της μορφής $K \le \dint_a^\beta f(x) \d x \le \Lambda$, τότε αρχικά μελετάμε την ίδια την $f(x)$ ως προς τα ελάχιστα-μέγιστα στο $[a, \beta]$.
    \item Αν αποδείξουμε αναλυτικά ότι ολικό ελάχιστο είναι το $m$ και ολικό μέγιστο το $M$, τότε $m \le f(x) \le M$.
    \item Ολοκληρώνουμε έπειτα την ανισότητα αυτή στα δοσμένα όρια $[a, \beta]$. Προκύπτει ευθέως: \[\dint_a^\beta m \, \d x \le \dint_a^\beta f(x) \, \d x \le \dint_a^\beta M \, \d x \Rightarrow m(\beta-a) \le \dint_a^\beta f(x) \d x \le M(\beta-a)\]
    \item Για απλές σχέσεις διατήρησης του προσήμου ($>0$ ή $<0$), αν $f(x) \ge 0$ και η $f$ δεν είναι παντού ταυτοτικά $0$, ισχύει αναγκαία ότι $\dint_a^\beta f(x) \d x > 0$.
\end{bhma}
\end{askhsh}

\Paradeigma{Ανισότητες χωρίς υπολογισμό της παράγουσας}
\bmath{Να αποδείξετε ότι ισχύει η διπλή ανισότητα: \ $1 \le \dint_0^1 e^{x^2} \d x \le e$.}

\lysh
Είναι αδύνατον να βρεθεί τύπος παράγουσας για το $e^{x^2}$. Άρα καταφεύγουμε σε ανισότητες ολοκληρωμάτων. Θεωρούμε τη συνάρτηση εσωτερικά $f(x) = e^{x^2}$ στο απαιτούμενο διάστημα $[0, 1]$.
Η παράγωγός της είναι $f'(x) = 2x e^{x^2}$. Γνωρίζουμε ότι ο εκθετικός $e^{x^2} > 0$ και για $x \in (0, 1]$ το $2x > 0$.
Άρα συνολικά $f'(x) > 0$ στο $(0, 1]$, οπότε η $f$ είναι γνησίως αύξουσα στο $[0, 1]$.
Άρα για κάθε $x \in [0, 1]$ έχουμε εφαρμόζοντας τη μονοτονία:
\[ 0 \le x \le 1 \xRightarrow{f \uparrow} f(0) \le f(x) \le f(1) \Rightarrow e^0 \le e^{x^2} \le e^1 \Rightarrow 1 \le e^{x^2} \le e \]
Εφόσον η ανισότητα αποτελείται από συνεχείς ως προς το $x$ συναρτήσεις, ολοκληρώνοντας από το $0$ έως το $1$ προκύπτει το αναγκαίο συμπερασμα:
\[ \dint_0^1 1 \, \d x \le \dint_0^1 e^{x^2} \, \d x \le \dint_0^1 e \, \d x \Rightarrow [x]_0^1 \le \dint_0^1 e^{x^2} \d x \le [e\cdot x]_0^1 \Rightarrow 1 \le \dint_0^1 e^{x^2} \d x \le e \]

\subsection{Υπολογισμός Εμβαδού Χωρίων}
\begin{askhsh}{Εμβαδόν μεταξύ $C_f$ και $x'x$ (3 περιπτώσεις) - Εμβαδόν μεταξύ $C_f$ και $C_g$ (3 περιπτώσεις)}
Το εμβαδόν του χωρίου μεταξύ της γραφικής παράστασης $C_f$ και του άξονα $x'x$ σε διάστημα $[a,\beta]$ δίνεται από:
\[E=\dint_a^\beta |f(x)|\,\d x\]
\bhmata
\begin{bhma}
\item Λύνουμε την εξίσωση $f(x)=0$ και βρίσκουμε τις ρίζες $\rho_1<\rho_2<\ldots<\rho_\nu$.
\item Βρίσκουμε τα πρόσημα της συνάρτησης $f$.
\item \begin{itemize}
\item Αν μας δίνονται \textbf{και τα δύο άκρα} το ζητούμενο εμβαδόν είναι
\[E=\int_{a}^{\beta}{|f(x)|\d x}\]
    \item Αν δεν δίνονται τα άκρα τότε βάζουμε άκρα τη μέγιστη και την ελάχιστη ρίζα:
    \[E=\dint_{\rho_1}^{\rho_\nu}|f(x)|\,\d x\]
    \item Αν μας δίνεται το \textbf{ένα} από τα δύο άκρα τότε
    \[E=\dint_{\min(a,\rho_1)}^{\max(a,\rho_\nu)}|f(x)|\,\d x\]
\end{itemize}
\end{bhma}
\begin{center}
\bmath{Αν αντί για εμβαδόν μεταξύ $C_f$ και $x'x$ ψάχνουμε εμβαδόν μεταξύ $C_f$ και $C_g$ τότε ακολουθούμε τα ίδια βήματα για τη συνάρτηση $h(x)=f(x)-g(x)$.}
\end{center}
\end{askhsh}

\Paradeigma{Εμβαδόν μεταξύ $C_f$ και $x'x$ (3 περιπτώσεις)}
\bmath{\vspace{-5mm}
\begin{alist}
\item Να υπολογιστεί το εμβαδόν του χωρίου μεταξύ της $C_f$ με $f(x)=x^2-1$ και του άξονα $x'x$ για $x\in[-2,2]$.
\item Να υπολογιστεί το εμβαδόν του χωρίου που περικλείεται από την $C_f$ με $f(x)=x^2-4x+3$ και τον άξονα $x'x$.
\item Να υπολογιστεί το εμβαδόν του χωρίου μεταξύ της $C_f$ με $f(x)=\sqrt{x}-1$, του άξονα $x'x$ και της ευθείας $x=0$.
\end{alist}
}
\lysh\vspace{-5mm}
\begin{alist}
\item 
Έχουμε γνωστά άκρα $-2$ και $2$. Βρίσκουμε τα μηδενικά: $x^2-1=0 \Rightarrow x=\pm 1$. Στα διαστήματα $[-2,-1]\cup[1,2]$ ισχύει $f(x)\ge 0$, ενώ στο $[-1,1]$ ισχύει $f(x)\le 0$. Άρα:
\begin{align*}
E&=\dint_{-2}^{-1}(x^2-1)\,\d x-\dint_{-1}^{1}(x^2-1)\,\d x+\dint_{1}^{2}(x^2-1)\,\d x=\\
&=\left[\frac{x^3}{3}-x\right]_{-2}^{-1}-\left[\frac{x^3}{3}-x\right]_{-1}^{1}+\left[\frac{x^3}{3}-x\right]_{1}^{2}=\\
&=\frac{4}{3}-\left(-\frac{4}{3}\right)+\frac{4}{3}=4
\end{align*}

\item Δεν δίνονται άκρα, άρα βρίσκουμε σημεία τομής με $x'x$: $x^2-4x+3=0 \Rightarrow x=1,3$. 
\begin{center}
\begin{tikzpicture}
\tkzTabInit[color,lgt=3,espcl=1.5,colorC = maincolor!50!white,colorL = maincolor!20!white,colorV = maincolor!50!white]%
{$x$ / 0.8 , $x^2-4x+3$ / 0.8}{$-\infty$, $1$, $3$, $+\infty$}
\tkzTabLine{ , + , z , - , z , + , }
\end{tikzpicture}
\end{center}
Στο $[1,3]$ ισχύει $f(x)\le 0$ άρα:
\[
E=\int_{1}^{3}{|f(x)|\d x}=-\dint_{1}^{3}(x^2-4x+3)\,\d x=\left[\frac{x^3}{3}-2x^2+3x\right]_{1}^{3}=0-\frac{4}{3}=\frac{4}{3}
\]
\item Δίνεται 1 άκρο $x=0$ και το άλλο άκρο προκύπτει από τομή με $x'x$: $\sqrt{x}-1=0 \Rightarrow \sqrt{x}=1 \Rightarrow x=1$. Στο $[0,1]$ ισχύει $\sqrt{x}-1\le 0$, άρα:
\begin{align*}
E&=-\dint_{0}^{1}(\sqrt{x}-1)\,\d x=\dint_{0}^{1}(1-\sqrt{x})\,\d x=\left[x-\frac{2}{3}x^{\frac{3}{2}}\right]_{0}^{1}=1-\frac{2}{3}=\frac{1}{3}
\end{align*}
\end{alist}

\Paradeigma{Εμβαδόν μεταξύ $C_f$ και $C_g$ (3 περιπτώσεις)}
\bmath{\vspace{-5mm}
\begin{alist}
\item Να υπολογιστεί το εμβαδόν του χωρίου μεταξύ των $C_f, C_g$ με $f(x)=x$ και $g(x)=x^2$ για $x\in[0,2]$.
\item Να υπολογιστεί το εμβαδόν του χωρίου που περικλείεται από τις $C_f, C_g$ με $f(x)=x^2$ και $g(x)=2x$.
\item Να υπολογιστεί το εμβαδόν του χωρίου μεταξύ των $C_f, C_g$ με $f(x)=\sqrt{x}$ και $g(x)=x$, του άξονα $y'y$ (δηλ. $x=0$) και της κοινής τους τομής.
\end{alist}
}
\lysh\vspace{-5mm}
\begin{alist}
\item Έχουμε γνωστά άκρα $0$ και $2$. Βρίσκουμε τα σημεία τομής στο εσωτερικό λύνοντας $f(x)=g(x)$:
\[
x=x^2 \Rightarrow x(x-1)=0 \Rightarrow x=0,1
\]
Στο $[0,1]$ ισχύει $x\ge x^2$, ενώ στο $[1,2]$ ισχύει $x^2\ge x$. Άρα:
\[
E=\dint_{0}^{1}(x-x^2)\,\d x+\dint_{1}^{2}(x^2-x)\,\d x
=\left[\frac{x^2}{2}-\frac{x^3}{3}\right]_{0}^{1}+\left[\frac{x^3}{3}-\frac{x^2}{2}\right]_{1}^{2}=\frac{1}{6}+\frac{5}{6}=1
\]
\item Δεν δίνονται άκρα, άρα βρίσκουμε τα σημεία τομής:
\[
x^2=2x \Rightarrow x(x-2)=0 \Rightarrow x=0,2
\]
Στο $(0,2)$ ισχύει $2x-x^2>0$, άρα $g(x)\ge f(x)$ στο $[0,2]$ και:
\[
E=\dint_{0}^{2}(2x-x^2)\,\d x=\left[x^2-\frac{x^3}{3}\right]_{0}^{2}=4-\frac{8}{3}=\frac{4}{3}
\]
\item Δίνεται 1 άκρο το $x=0$ (άξονας $y'y$). Το άλλο άκρο είναι κοινή τομή $f(x)=g(x)$:
\[
\sqrt{x}=x \Rightarrow x=x^2 \Rightarrow x=0,1
\]
Επιλέγουμε το $\beta=1$. Στο $[0,1]$ ισχύει $\sqrt{x}\ge x$, άρα:
\[
E=\dint_{0}^{1}(\sqrt{x}-x)\,\d x=\left[\frac{2}{3}x^{\frac{3}{2}}-\frac{x^2}{2}\right]_{0}^{1}=\frac{2}{3}-\frac{1}{2}=\frac{1}{6}
\]
\end{alist}
\begin{askhsh}{Εμβαδόν χωρίου μεταξύ $C_f$ και εφαπτομένης}
Το εμβαδόν μεταξύ δύο συνεχών συναρτήσεων $f$ και $g$ για $x \in [a, \beta]$ δίνεται αναλυτικά από το ολοκλήρωμα \[E = \dint_a^\beta |f(x) - g(x)| \d x\]
\bhmata
\begin{bhma}
    \item \textbf{Βρίσκουμε τα όρια του χωρίου:} Αν δεν δίνονται οι κατακόρυφες ευθείες που κλείνουν το χωρίο, τότε βρίσκουμε τα σημεία τομής της καμπύλης με την εφαπτομένη λύνοντας την εξίσωση $f(x)=y_{\varepsilon\varphi}$. Αυτά τα σημεία δίνουν τα όρια ολοκλήρωσης.
    \item \textbf{Χρησιμοποιούμε την κυρτότητα:} Επειδή η μία από τις δύο γραφικές είναι εφαπτομένη, η σχετική θέση τους φαίνεται από την κυρτότητα της $f$.
    \begin{itemize}
        \item Αν $f''(x)>0$, τότε η $C_f$ είναι πάνω από την εφαπτομένη της.
        \item Αν $f''(x)<0$, τότε η $C_f$ είναι κάτω από την εφαπτομένη της.
    \end{itemize}
    Έτσι αφαιρούμε σωστά τις δύο συναρτήσεις και υπολογίζουμε το ολοκλήρωμα.
\end{bhma}
\end{askhsh}

\Paradeigma{Εμβαδόν χωρίου χωρίς εξωτερικά άκρα και εμβαδόν με κυρτότητα}
\bmath{\vspace{-5mm}
\begin{alist}
\item Να υπολογιστεί το εμβαδόν του χωρίου που περικλείεται από τις γραφικές παραστάσεις $f(x)=x^2$ και $g(x)=x+2$.
\item Να υπολογιστεί το εμβαδόν μεταξύ της $h(x)=e^x$, της εφαπτομένης της στο $x_0=1$, και του άξονα $y'y$.
\end{alist}
}

\lysh\vspace{-5mm}
\begin{alist}
\item Αναζητούμε αρχικά τα σημεία τομής καθώς μας λείπουν τα άκρα του ολοκληρώματος:
\[ f(x) = g(x) \Rightarrow x^2 = x + 2 \Rightarrow x^2 - x - 2 = 0 \]
Οι ρίζες του τριωνύμου είναι $x = -1$ και $x = 2$.
Στο εσωτερικό των ριζών $(-1, 2)$ η διαφορά τους διατηρεί ως τριώνυμο σταθερό αρνητικό πρόσημο $x^2-x-2\le 0$ άρα $g(x) \ge f(x)$.

\[ E = \dint_{-1}^2 (g(x) - f(x)) \d x = \dint_{-1}^2 (x + 2 - x^2) \d x = \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^2 \]
\[ = \left( \frac{2^2}{2} + 2\cdot 2 - \frac{8}{3} \right) - \left( \frac{1^2}{2} - 2 + \frac{(-1)^3}{3} \right) = \left(6 - \frac{8}{3}\right) - \left(-\frac{3}{2} - \frac{1}{3}\right) = \frac{10}{3} + \frac{9}{6} = \frac{20+9}{6} = \frac{9}{2} \text{ τ.μ.} \]
\item 
Για κάθε $x\in\mathbb{R}$ έχουμε: $h'(x)=h''(x)=e^x>0$. Υπολογίζουμε την εξίσωση της εφαπτομένης της $h$: 

Για $x=1$: $h(1)=e$ και $h'(1)=e$. Η εφαπτομένη έχει εξίσωση
\begin{gather*}
y-f(1)=f'(1)(x-1)\Leftrightarrow\\
y-e=e(x-1)\Leftrightarrow y=ex
\end{gather*}

Το χωρίο με άκρα τον άξονα $y'y$ δηλαδή την $x=0$, και το σημείο επαφής στα δεξιά $x=1$, υπολογίζεται άμεσα λόγω κυρτότητας. Επειδή η $h(x)=e^x$ είναι κυρτή βρίσκεται πάνω από την εφαπτομένη της με εξαίρεση το σημείο επαφής: $e^x \ge ex$.
