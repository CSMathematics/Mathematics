\begin{thema}{Α}
\begin{erwthma}
    \item Έστω μια συνάρτηση $f$ παραγωγίσιμη σε ένα διάστημα $(a, \beta)$, με εξαίρεση ίσως ένα σημείο $x_0$, στο οποίο όμως η $f$ είναι συνεχής. Αν $f'(x) > 0$ στο $(a, x_0)$ και $f'(x) < 0$ στο $(x_0, \beta)$, τότε να αποδείξετε ότι το $f(x_0)$ είναι τοπικό μέγιστο της $f$. \monades{7}
    \item Πότε δύο συναρτήσεις $f$, $g$ λέγονται ίσες; \monades{4}
    \item Να διατυπώσετε το θεώρημα μέσης τιμής του διαφορικού λογισμού και να το ερμηνεύσετε γεωμετρικά. \monades{4}
    \item Να χαρακτηρίσετε τις προτάσεις που ακολουθούν, γράφοντας στο τετράδιό σας, δίπλα στο γράμμα που αντιστοιχεί σε κάθε πρόταση, τη λέξη \textbf{Σωστό}, αν η πρόταση είναι σωστή, ή \textbf{Λάθος}, αν η πρόταση είναι λανθασμένη.
    \begin{alist}
        \item Για κάθε συνεχή συνάρτηση $f : [a, \beta] \to \mathbb{R}$, αν $G$ είναι μια παράγουσα της $f$ στο $[a, \beta]$, τότε $\displaystyle\int_{a}^{\beta} f(t)\,\d t = G(a) - G(\beta)$.
        \item Αν οι συναρτήσεις $f$, $g$ έχουν όριο στο $x_0$ και ισχύει $f(x) \leq g(x)$ κοντά στο $x_0$, τότε $\displaystyle\lim_{x \to x_0} f(x) \leq \lim_{x \to x_0} g(x)$.
        \item Κάθε συνάρτηση $f$, για την οποία ισχύει $f'(x) = 0$ για κάθε $x \in (a, x_0) \cup (x_0, \beta)$, είναι σταθερή στο $(a, x_0) \cup (x_0, \beta)$.
        \item Μια συνάρτηση $f$ είναι $1$-$1$, αν και μόνο αν, για κάθε στοιχείο $y$ του συνόλου τιμών της, η εξίσωση $y = f(x)$ έχει ακριβώς μία λύση ως προς $x$.
        \item Αν η $f$ είναι συνεχής στο $[a, \beta]$, τότε η $f$ παίρνει στο $[a, \beta]$ μια μέγιστη τιμή $M$ και μια ελάχιστη τιμή $m$.
    \end{alist} \monades{10}
\end{erwthma}
\end{thema}

    \item Έστω μια συνάρτηση $f$ συνεχής σε ένα διάστημα $\Delta$ και δύο φορές παραγωγίσιμη στο εσωτερικό του $\Delta$. Αν $f''(x) > 0$ για κάθε εσωτερικό σημείο $x$ του $\Delta$, τότε η $f$ είναι κυρτή στο $\Delta$.
    \item Αν η συνάρτηση $g$ είναι παραγωγίσιμη στο $x_0$ και η συνάρτηση $f$ είναι παραγωγίσιμη στο $g(x_0)$, τότε η συνάρτηση $f \circ g$ είναι παραγωγίσιμη στο $x_0$ και ισχύει
    \begin{equation*}
    (f \circ g)'(x_0) = f'(g(x_0)) \cdot g'(x_0)
    \end{equation*}
    \item[ε)] Η συνάρτηση $f(x) = \syf x$ είναι παραγωγίσιμη στο $\mathbb{R} - \{x \mid \hm x = 0\}$ και ισχύει $(\syf x)' = \frac{1}{\hm^2 x}$.
\end{alist}
\monades{10}
\end{erwthma}
\end{thema}
