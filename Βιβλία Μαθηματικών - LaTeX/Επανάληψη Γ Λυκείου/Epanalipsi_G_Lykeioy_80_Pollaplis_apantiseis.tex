\section*{Απαντήσεις Πολλαπλής Επιλογής}

\subsection*{Κεφάλαιο 1: Όρια - Συνέχεια}
\begin{enumerate}[label=\textbf{1.\arabic*.}]
\item \textbf{β}. Θέλουμε $1-x^2>0$, άρα $-1<x<1$.
\item \textbf{β}. Συμμετρία ως προς την αρχή των αξόνων σημαίνει περιττή συνάρτηση.
\item \textbf{β}. Με $u=2x$ έχουμε
\[
\frac{f(2x)}{x}=2\frac{f(u)}{u}\to 2\cdot2=4.
\]
\item \textbf{γ}. Ισχύει
\[
\frac{\hm 3x}{x}=3\frac{\hm 3x}{3x}\to3.
\]
\item \textbf{γ}. Ο αριθμητής είναι βαθμού $3$ και ο παρονομαστής βαθμού $2$, άρα το πηλίκο τείνει στο $+\infty$.
\item \textbf{β}. Από Bolzano αρκεί $f(a)f(\beta)<0$.
\item \textbf{β}. Σε ανοιχτό διάστημα παίρνουμε ως άκρα του συνόλου τιμών τα πλευρικά όρια στα άκρα.
\item \textbf{β}. Για να ορίζεται η $\sqrt{\ln x}$ πρέπει $\ln x\geq0$, άρα $x\geq1$.
\item \textbf{α}. Γινόμενο δύο περιττών συναρτήσεων είναι άρτια συνάρτηση.
\item \textbf{γ}. Το $f(-x)$ δίνει συμμετρία ως προς $y'y$ και το πρόσημο $-$ έπειτα συμμετρία ως προς $x'x$, συνολικά συμμετρία ως προς την αρχή.
\item \textbf{β}. Η συνθήκη ότι η εξίσωση $f(x)=y$ έχει το πολύ μία λύση είναι ακριβώς η ιδιότητα $1-1$.
\item \textbf{β}. Οι $C_f$ και $C_{f^{-1}}$ μπορεί να τέμνονται και εκτός της $y=x$, π.χ. όταν η $f$ είναι γνησίως φθίνουσα.
\item \textbf{β}. Ρητοποιώντας,
\[
\frac{\sqrt{x+3}-2}{x-1}=\frac{1}{\sqrt{x+3}+2}\to\frac14.
\]
\item \textbf{γ}. Επειδή $|\hm(1/x)|\leq1$, έχουμε
\[
\left|x\hm\frac1x\right|\leq |x|\to0.
\]
\item \textbf{β}. Η συνέχεια και η αλλαγή προσήμου στα άκρα εξασφαλίζουν τουλάχιστον μία ρίζα.
\item \textbf{β}. Το θεώρημα μέγιστης-ελάχιστης τιμής απαιτεί συνέχεια σε κλειστό διάστημα.
\item \textbf{α}. Αν $f\circ g$ είναι $1-1$, τότε αναγκαστικά η $g$ είναι $1-1$.
\item \textbf{β}. Άθροισμα τετραγώνων ίσο με $0$ σημαίνει ότι και τα δύο τετράγωνα είναι $0$.
\item \textbf{α}. Για $x\to1^-$ είναι $|x-1|=1-x$, άρα ο αριθμητής είναι $0$ και το πηλίκο είναι $0$.
\item \textbf{β}. Θέτοντας $u=\frac1x$, παίρνουμε
\[
x\hm\frac1x=\frac{\hm u}{u}\to1.
\]
\item \textbf{β}. Η πολυωνυμική είναι συνεχής και αλλάζει πρόσημο στο $[1,2]$, άρα έχει τουλάχιστον μία ρίζα.
\item \textbf{β}. Αν η άρτια $f$ έχει ρίζα το $2$, τότε έχει ρίζα και το $-2$.
\item \textbf{β}. Συνεχής συνάρτηση χωρίς ρίζες διατηρεί πρόσημο. Αφού $f(2026)=1>0$, είναι παντού θετική.
\item \textbf{β}. Επειδή $1-\syn x\to0$ και ειδικά $1-\syn x\sim1$ δεν ισχύει εδώ· έχουμε $\syn0=1$, άρα
\[
\frac{1-\syn x}{x}\to0.
\]
\item \textbf{α}. Αν $|f(x)|\to0$, τότε αναγκαστικά $f(x)\to0$.
\end{enumerate}

\subsection*{Κεφάλαιο 2: Διαφορικός Λογισμός}
\begin{enumerate}[label=\textbf{2.\arabic*.}]
\item \textbf{γ}. Από κανόνα αλυσίδας,
\[
(\hm^2 x)'=2\hm x\syn x.
\]
\item \textbf{β}. Οριζόντια εφαπτομένη σημαίνει συντελεστής διεύθυνσης $0$, δηλαδή $f'(x_0)=0$.
\item \textbf{γ}. Η συνέχεια δεν συνεπάγεται παραγωγισιμότητα, π.χ. $|x|$ στο $0$.
\item \textbf{β}. Ο κανόνας αλυσίδας δίνει
\[
(f(g(x)))'=f'(g(x))g'(x).
\]
\item \textbf{β}. Από Fermat, σε παραγωγίσιμο εσωτερικό τοπικό ακρότατο ισχύει $f'(x_0)=0$.
\item \textbf{β}. Το Rolle εξασφαλίζει τουλάχιστον ένα σημείο με $f'(\xi)=0$, δηλαδή οριζόντια εφαπτομένη.
\item \textbf{γ}. Το Θ.Μ.Τ. δίνει εφαπτομένη παράλληλη στη χορδή των άκρων.
\item \textbf{β}. Αν $f'=g'$ στο $\mathbb{R}$, τότε $f-g$ είναι σταθερή.
\item \textbf{γ}. Αν $f'>0$ σε διάστημα και η $f$ είναι συνεχής, τότε η $f$ είναι γνησίως αύξουσα.
\item \textbf{β}. Κρίσιμα είναι τα εσωτερικά σημεία όπου η $f'$ μηδενίζεται ή δεν υπάρχει.
\item \textbf{γ}. Μηδενισμός της $f'$ και αλλαγή προσήμου δίνουν θέση τοπικού ακροτάτου.
\item \textbf{β}. Τα τοπικά ακρότατα συγκρίνουν τιμές μόνο τοπικά, άρα τέτοια διάταξη είναι δυνατή.
\item \textbf{γ}. Σε κλειστό διάστημα ελέγχουμε κρίσιμα σημεία στο εσωτερικό και τα άκρα.
\item \textbf{β}. Αν $f'(x_0)=0$ και $f''(x_0)>0$, τότε έχουμε τοπικό ελάχιστο.
\item \textbf{δ}. Σε κυρτή συνάρτηση η εφαπτομένη βρίσκεται κάτω από τη γραφική παράσταση.
\item \textbf{γ}. Σημείο καμπής υπάρχει όταν υπάρχει εφαπτομένη και αλλάζει η κυρτότητα.
\item \textbf{δ}. Αν $f''<0$, η $f$ στρέφει τα κοίλα κάτω, δηλαδή είναι κοίλη.
\item \textbf{β}. Αν η $f$ είναι δύο φορές παραγωγίσιμη και έχει σημείο καμπής στο $x_0$, τότε $f''(x_0)=0$.
\item \textbf{β}. Το $f''(x_0)=0$ δεν αρκεί· χρειάζεται αλλαγή προσήμου της $f''$.
\item \textbf{γ}. Ο κανόνας De L'Hospital εφαρμόζεται άμεσα σε μορφές $\frac00$ και $\frac{\pm\infty}{\pm\infty}$.
\item \textbf{β}. Ο ορισμός πλάγιας ασύμπτωτης είναι
\[
\lim_{x\to+\infty}[f(x)-(\lambda x+\beta)]=0.
\]
\item \textbf{γ}. Ο παρονομαστής μηδενίζεται στο $x=3$, άρα κατακόρυφη ασύμπτωτη είναι η $x=3$.
\item \textbf{β}. Μπορεί να έχει το πολύ δύο οριζόντιες ασύμπτωτες, μία στο $+\infty$ και μία στο $-\infty$.
\item \textbf{β}. Ο στιγμιαίος ρυθμός μεταβολής της $f$ στο $x_0$ είναι η $f'(x_0)$.
\item \textbf{β}. Παράγωγος περιττής παραγωγίσιμης συνάρτησης είναι άρτια.
\item \textbf{β}. Γνησίως αύξουσα παραγωγίσιμη συνάρτηση μπορεί να έχει $f'(x_0)=0$, π.χ. $x^3$ στο $0$.
\item \textbf{γ}. Κάθε πολυώνυμο περιττού βαθμού $\geq3$ έχει τουλάχιστον ένα σημείο καμπής.
\item \textbf{γ}. Ισχύει
\[
(a^x)'=a^x\ln a.
\]
\item \textbf{β}. Το όριο του λόγου μεταβολής, όταν υπάρχει ως πραγματικός αριθμός, είναι η παράγωγος $f'(x_0)$.
\item \textbf{β}. Ο σωστός κανόνας γινομένου είναι
\[
(fg)'=f'g+fg'.
\]
\item \textbf{γ}. Από το γράφημα της $f'$ έχουμε αλλαγή από $+$ σε $-$ στο $x=2$, άρα τοπικό μέγιστο εκεί.
\item \textbf{β}. Το $f''$ αλλάζει πρόσημο στο $x=1$, άρα η $f$ έχει σημείο καμπής στο $x_0=1$.
\item \textbf{β}. Εφόσον $f'(x)=x^2\geq0$ και μηδενίζεται μόνο στο $0$, η $f$ είναι γνησίως αύξουσα.
\item \textbf{α}. Η διακεκομμένη οριζόντια ασύμπτωτη $y=1$ δείχνει ότι
\[
\lim_{x\to+\infty}f(x)=1.
\]
\item \textbf{β}. Σε δύο φορές παραγωγίσιμο τοπικό ελάχιστο τύπου βαθουλώματος έχουμε $f'(2)=0$ και $f''(2)>0$.
\end{enumerate}

\subsection*{Κεφάλαιο 3: Ολοκληρωτικός Λογισμός}
\begin{enumerate}[label=\textbf{3.\arabic*.}]
\item \textbf{β}. Η αλλαγή των άκρων αλλάζει πρόσημο:
\[
\dint_a^\beta f(x)\,\d x=-\dint_\beta^a f(x)\,\d x.
\]
\item \textbf{α}. Για ίδια άκρα ολοκλήρωσης ισχύει
\[
\dint_a^a f(x)\,\d x=0.
\]
\item \textbf{γ}. Όταν $f\geq0$, το εμβαδόν είναι το ολοκλήρωμα
\[
\dint_a^\beta f(x)\,\d x.
\]
\item \textbf{δ}. Έχουμε
\[
\dint_a^\beta c\,\d x=c(\beta-a).
\]
\item \textbf{β}. Για άρτια συνάρτηση,
\[
\dint_{-a}^a f(x)\,\d x=2\dint_0^a f(x)\,\d x.
\]
\item \textbf{α}. Για περιττή συνάρτηση, το ολοκλήρωμα στο συμμετρικό διάστημα είναι $0$.
\item \textbf{γ}. Η προσθετική ιδιότητα ισχύει για οποιουσδήποτε πραγματικούς $a,\beta,\gamma$.
\item \textbf{α}. Είναι ο τύπος παραγοντικής ολοκλήρωσης στο ορισμένο ολοκλήρωμα.
\item \textbf{β}. Αν κρατήσουμε τη νέα μεταβλητή $u$, τα όρια αλλάζουν σε $g(a)$ και $g(\beta)$.
\item \textbf{β}. Συνεχής μη αρνητική συνάρτηση με μηδενικό ολοκλήρωμα είναι μηδενική σε όλο το διάστημα.
\item \textbf{γ}. Αν $f\geq g$, το εμβαδόν μεταξύ τους είναι
\[
\dint_a^\beta (f(x)-g(x))\,\d x.
\]
\item \textbf{β}. Ισχύει
\[
\dint_1^e \frac1x\,\d x=\ln e-\ln1=1.
\]
\item \textbf{γ}. Από το Θεμελιώδες Θεώρημα,
\[
\dint_a^\beta f'(x)\,\d x=f(\beta)-f(a).
\]
\item \textbf{γ}. Έχουμε
\[
\dint_0^\pi \hm x\,\d x=[-\syn x]_0^\pi=2.
\]
\item \textbf{α}. Από τη μονοτονία του ολοκληρώματος,
\[
m(\beta-a)\leq \dint_a^\beta f(x)\,\d x\leq M(\beta-a).
\]
\item \textbf{β}. Το εμβαδόν επίπεδου χωρίου είναι πάντοτε μη αρνητικός πραγματικός αριθμός.
\item \textbf{γ}. Θετικό ή μη αρνητικό ολοκλήρωμα δεν σημαίνει ότι η συνάρτηση είναι παντού μη αρνητική.
\item \textbf{β}. Από
\[
\dint_2^1 g(x)\,\d x=5
\]
παίρνουμε
\[
\dint_1^2 g(x)\,\d x=-5.
\]
Άρα το ζητούμενο είναι $3-5=-2$.
\item \textbf{γ}. Το ορισμένο ολοκλήρωμα εξαρτάται από τη συνάρτηση και τα άκρα, όχι από το όνομα της μεταβλητής.
\item \textbf{β}. Το ορισμένο ολοκλήρωμα συνεχούς συνάρτησης σε κλειστό διάστημα είναι ένας μοναδικός πραγματικός αριθμός.
\end{enumerate}
