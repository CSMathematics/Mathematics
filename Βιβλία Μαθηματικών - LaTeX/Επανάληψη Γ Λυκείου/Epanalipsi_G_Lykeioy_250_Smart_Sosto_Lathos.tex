\section{Ερωτήσεις Σωστό - Λάθος}
\subsection{Όρια συνέχεια (85 Ερωτήσεις)}
\begin{enumerate}[label=\textbf{1.\arabic*.}]
    \item Το πεδίο ορισμού της συνάρτησης $f(x) = \ln(x^2)$ είναι το $(0, +\infty)$.
    \item Οι συναρτήσεις $f(x) = \sqrt{x^2}$ και $g(x) = x$ είναι ίσες στο $\mathbb{R}$.
    \item Αν δύο συναρτήσεις έχουν το ίδιο πεδίο ορισμού και την ίδια ελάχιστη και μέγιστη τιμή, τότε είναι αναγκαστικά ίσες.
    \item Αν η συνάρτηση άθροισμα $f+g$ έχει πεδίο ορισμού το $\mathbb{R}$, τότε υποχρεωτικά και οι δύο συναρτήσεις $f, g$ ορίζονται σε όλο το $\mathbb{R}$.
    \item Το πεδίο ορισμού της $f+g$ ταυτίζεται γενικά με την ένωση των πεδίων ορισμού των $f$ και $g$.
    \item Η συνάρτηση $f(x) = \dfrac{x^2-1}{x-1}$ ταυτίζεται με τη συνάρτηση $g(x) = x+1$.
    \item Για κάθε $x>0$, ισχύει $\ln x^2 = 2\ln x$.
    \item Για κάθε $x \neq 0$, ισχύει $\ln x^2 = 2\ln x$.
    \item Αν οι συναρτήσεις $f, g$ είναι άρτιες, τότε η συνάρτηση γινόμενο $f \cdot g$ είναι άρτια.
    \item Αν η $f$ είναι περιττή, το $0$ ανήκει στο πεδίο ορισμού της, τότε αναγκαστικά $f(0)=0$.
    \item Μια συνάρτηση $f$ μπορεί να τέμνει τον άξονα $y'y$ σε περισσότερα από ένα σημεία.
    \item Μια άρτια συνάρτηση είναι αδύνατον να είναι 1-1, εκτός αν το πεδίο ορισμού της έχει μόνο ένα στοιχείο.
    \item Κάθε περιοδική συνάρτηση με μη μηδενική περίοδο, δεν είναι 1-1.
    \item Αν $f^2(x) = x^2$ για κάθε $x \in \mathbb{R}$, τότε απαραίτητα ισχύει $f(x)=x$ για κάθε $x \in \mathbb{R}$ ή $f(x)=-x$ για κάθε $x \in \mathbb{R}$.
    \item Αν η $f$ είναι γνησίως αύξουσα σε ένα διάστημα $\Delta$, τότε η $\dfrac{1}{f}$ είναι σίγουρα γνησίως φθίνουσα στο $\Delta$.
    \item Αν οι συναρτήσεις $f,g$ είναι γνησίως φθίνουσες τότε η συνάρτηση $g\circ f$ είναι γνησίως αύξουσα.
    \item Αν η συνάρτηση σύνθεση $f \circ g$ είναι 1-1, τότε υποχρεωτικά η $g$ είναι 1-1.
    \item Αν η συνάρτηση σύνθεση $f \circ g$ είναι 1-1, τότε υποχρεωτικά και η $f$ είναι 1-1.
    \item Αν η $f$ είναι γνησίως αύξουσα και η $g$ γνησίως φθίνουσα, η $f+g$ είναι πάντα σταθερή.
    \item Η γραφική παράσταση της $-f(x)$ είναι συμμετρική της $C_f$ ως προς τον άξονα $x'x$.
    \item Η γραφική παράσταση της $f(-x)$ είναι συμμετρική της $C_f$ ως προς τον άξονα $x'x$.
    \item Αν η $f$ έχει αντίστροφη, η εξίσωση $f(x)=y$ έχει ακριβώς μία λύση ως προς $x$ για κάθε $y$ του συνόλου τιμών της.
    \item Οι γραφικές παραστάσεις των $f$ και $f^{-1}$ τέμνονται αποκλειστικά και μόνο πάνω στην ευθεία $y=x$.
    \item Η γραφική παράσταση της συνάρτησης $f(x) = \sqrt{x-2}$ έχει άξονα συμμετρίας τον $y'y$.
    \item Αν η $f$ έχει αντίστροφη και ισχύει $f(a)=\beta$, τότε απαραίτητα $f^{-1}(\beta)=a$.
    \item Αν μια συνάρτηση $f$ είναι 1-1, τότε είναι υποχρεωτικά γνησίως μονότονη στο πεδίο ορισμού της.
    \item Αν $\lim\limits_{x \to x_0} f(x) = \ell$, τότε η $f$ πρέπει υποχρεωτικά να ορίζεται στο $x_0$.
    \item Αν $f(x)<g(x)$ κοντά στο $x_0$, τότε σίγουρα $\lim\limits_{x \to x_0} f(x) < \lim\limits_{x \to x_0} g(x)$.
    \item Αν $\lim\limits_{x \to x_0} f(x) > 0$, τότε $f(x) > 0$ για κάθε $x$ του πεδίου ορισμού της.
    \item Αν $\lim\limits_{x \to x_0} |f(x)| = |\ell| \neq 0$, τότε υποχρεωτικά $\lim\limits_{x \to x_0} f(x) = \ell$ ή $\lim\limits_{x \to x_0} f(x) = -\ell$.
    \item Αν $\lim\limits_{x \to x_0} f(x) = +\infty$, τότε υποχρεωτικά $f(x) > 0$ κοντά στο $x_0$.
    \item Ισχύει ότι $\lim\limits_{x \to +\infty} \dfrac{\hm x}{x} = 1$.
    \item Το $\lim\limits_{x \to x_0} \sqrt{f(x)}$ υπάρχει στο $\mathbb{R}$ αν και μόνο αν $\lim\limits_{x \to x_0} f(x) \ge 0$.
    \item Αν $\lim\limits_{x\to x_0}f(x)=0$ και $f(x)<0$ κοντά στο $x_0$, το $\lim\limits_{x \to x_0} \sqrt{f(x)}$ είναι 0.
    \item Το όριο $\lim\limits_{x \to 0} \dfrac{1}{x^3}$ δεν υπάρχει στο $\mathbb{R} \cup \{-\infty, +\infty\}$.
    \item Ισχύει ότι $\lim\limits_{x \to 0} \dfrac{1}{|x|} = +\infty$.
    \item Αν $\lim\limits_{x \to x_0} f(x) = +\infty$, τότε η $f$ παίρνει υποχρεωτικά μόνο θετικές τιμές στο πεδίο ορισμού της.
    \item Αν $\lim\limits_{x \to +\infty} f(x) = -\infty$, τότε υπάρχουν αρνητικές τιμές της $f$.
    \item Η μορφή $0 \cdot (+\infty)$ καταλήγει πάντα στο 0.
    \item Η μορφή $(+\infty) + (+\infty)$ αποτελεί απροσδιοριστία.
    \item Η μορφή $(+\infty) - (+\infty)$ αποτελεί απροσδιοριστία.
    \item Αν $\lim\limits_{x \to x_0} f(x) = 0$ και η $g$ είναι φραγμένη κοντά στο $x_0$, τότε $\lim\limits_{x \to x_0} [f(x)g(x)] = 0$.
    \item Αν $\lim\limits_{x \to x_0} g(x) = u_0$ και $\lim\limits_{u \to u_0} f(u) = \ell$, τότε πάντοτε ισχύει $\lim\limits_{x \to x_0} f(g(x)) = \ell$.
    \item Η ευθεία $x=x_0$ λέγεται κατακόρυφη ασύμπτωτη της $C_f$, αν τουλάχιστον ένα από τα πλευρικά όρια στο $x_0$ είναι $+\infty$ ή $-\infty$.
    \item Μια συνάρτηση μπορεί η γραφική της παράσταση να τέμνει μια κατακόρυφη ασύμπτωτή της.
    \item Μια πολυωνυμική συνάρτηση βαθμού $\nu \ge 1$ μπορεί να έχει κατακόρυφη ασύμπτωτη.
    \item Μια ρητή συνάρτηση έχει σίγουρα κατακόρυφη ασύμπτωτη σε κάθε σημείο που μηδενίζεται ο παρονομαστής της.
    \item Η ευθεία $y = \lambda x + \beta$ είναι πλάγια ασύμπτωτη της $C_f$ στο $+\infty$ αν $\lim\limits_{x \to +\infty} [f(x) - (\lambda x + \beta)] = 0$.
    \item Μια συνάρτηση μπορεί να έχει δύο διαφορετικές οριζόντιες ασύμπτωτες στο $+\infty$.
    \item Μια συνάρτηση μπορεί να έχει ταυτόχρονα και πλάγια και οριζόντια ασύμπτωτη στο $+\infty$.
    \item Μια συνάρτηση μπορεί να έχει ταυτόχρονα δύο διαφορετικές πλάγιες ασύμπτωτες στο $+\infty$.
    \item Το όριο $\lim\limits_{x \to -\infty} e^x$ ισούται με 0.
    \item Το όριο $\lim\limits_{x \to 0^+} \ln x$ ισούται με $-\infty$.
    \item Κάθε συνάρτηση που είναι συνεχής στο πεδίο ορισμού της, είναι αναγκαστικά συνεχής στο $\mathbb{R}$.
    \item Αν μια συνάρτηση είναι συνεχής στο $x_0$, τότε απαραίτητα ορίζεται σε ανοιχτό διάστημα $(x_0-\delta, x_0+\delta)$.
    \item Αν $\lim\limits_{x \to x_0^+} f(x) = \lim\limits_{x \to x_0^-} f(x)$, τότε η $f$ είναι οπωσδήποτε συνεχής στο $x_0$.
    \item Αν $f$ συνεχής στο $x_0$ και $f(x_0) > 0$, τότε $f(x) > 0$ σε ένα διάστημα γύρω από το $x_0$.
    \item Η σύνθεση δύο συνεχών συναρτήσεων είναι συνεχής συνάρτηση στο πεδίο ορισμού της.
    \item Αν το άθροισμα $f+g$ είναι συνεχής συνάρτηση, τότε αναγκαστικά και οι $f, g$ είναι συνεχείς.
    \item Αν η συνάρτηση $|f|$ είναι συνεχής στο $x_0$, τότε και η $f$ είναι σίγουρα συνεχής στο $x_0$.
    \item Μια συνάρτηση συνεχής σε ένα κλειστό διάστημα $[a, \beta]$ έχει πάντοτε μέγιστη και ελάχιστη τιμή.
    \item Μια συνάρτηση συνεχής σε ένα ανοιχτό διάστημα $(a, \beta)$ αποκλείεται να παρουσιάζει μέγιστη και ελάχιστη τιμή.
    \item Το Θεώρημα \en{Bolzano} εξασφαλίζει την ύπαρξη ακριβώς μιας ρίζας της $f(x)=0$ στο $(a, \beta)$.
    \item Αν η $f$ είναι συνεχής στο $[a, \beta]$ και δεν μηδενίζεται σε αυτό, τότε $f(a) \cdot f(\beta) > 0$.
    \item Αν για τη συνεχή συνάρτηση $f$ ισχύει $f(a) \cdot f(\beta) = 0$, τότε το Θεώρημα \en{Bolzano} εξασφαλίζει ότι υπάρχει ρίζα της $f$ στο $(a, \beta)$.
    \item Η εικόνα ενός κλειστού διαστήματος $[a, \beta]$ μέσω μιας συνεχούς και μη σταθερής συνάρτησης είναι πάντα κλειστό διάστημα.
    \item Σύμφωνα με το Θεώρημα Ενδιάμεσων Τιμών, μια συνεχής συνάρτηση σε κλειστό διάστημα λαμβάνει υποχρεωτικά όλες τις τιμές ανάμεσα στην ελάχιστη και τη μέγιστη τιμή της.
    \item Αν η $f$ είναι συνεχής και γνησίως αύξουσα στο $[a, \beta]$, τότε το σύνολο τιμών της είναι το $[f(a), f(\beta)]$.
    \item Η συνάρτηση $f(x) = \dfrac{x^2-1}{x-1}$ είναι συνεχής στο $\mathbb{R}$.
    \item Η συνάρτηση που ορίζεται από $f(x) = x$ για $x \neq 0$ και $f(0) = 1$, είναι συνεχής στο 0.
    \item Κάθε πολυωνυμική συνάρτηση είναι συνεχής στο $\mathbb{R}$.
    \item Αν η εξίσωση $f(x)=0$ έχει ρίζα στο $(a, \beta)$, τότε υποχρεωτικά ισχύει $f(a) \cdot f(\beta) < 0$.
    \item Αν $f(a) \cdot f(\beta) < 0$ και η $f$ δεν είναι συνεχής στο $[a, \beta]$, αποκλείεται η $f$ να έχει ρίζα στο $(a, \beta)$.
    \item Αν η $f$ είναι συνεχής και 1-1 στο $[a, \beta]$, τότε είναι υποχρεωτικά γνησίως μονότονη.
    \item Μια συνάρτηση μπορεί να αλλάζει πρόσημο εκατέρωθεν μιας ρίζας της, ακόμα και αν δεν είναι συνεχής στο σημείο αυτό.
    \item Για να ισχύει το Θεώρημα Μέγιστης-Ελάχιστης Τιμής, αρκεί η συνάρτηση να είναι συνεχής σε ανοιχτό διάστημα.
    \item Το σύνολο τιμών μιας συνεχούς συνάρτησης με πεδίο ορισμού ανοιχτό διάστημα $(a, \beta)$ μπορεί να είναι κλειστό διάστημα $[\gamma, \delta]$.
    \item Αν $f, g$ συνεχείς στο $x_0$ με $f(x_0)=0$ και $g(x_0)=0$, τότε το όριο $\lim\limits_{x \to x_0} \dfrac{f(x)}{g(x)}$ είναι πάντα 1.
    \item Αν μια συνάρτηση $f$ είναι συνεχής και γνησίως μονότονη στο $\mathbb{R}$, τότε τέμνει τον άξονα $x'x$ το πολύ σε ένα σημείο.
    \item Αν η $f$ ορίζεται στο $[a,\beta]$ και παίρνει όλες τις ενδιάμεσες τιμές ανάμεσα στα $f(a)$ και $f(\beta)$, τότε είναι υποχρεωτικά συνεχής.
    \item Αν η $f$ είναι συνεχής σε ένα διάστημα $\Delta$ και δεν μηδενίζεται σε αυτό, τότε οι τιμές της $f$ στο $\Delta$ είναι ομόσημες.
    \item Αν μια συνάρτηση $f$ είναι άρτια και συνεχής στο $\mathbb{R}$, τότε η εξίσωση $f(x)=0$ έχει είτε καμία ρίζα, είτε περιττό πλήθος ριζών αν το $0$ είναι ρίζα της, είτε άρτιο πλήθος ριζών αν το $0$ δεν είναι ρίζα της.
    \item Ισχύει ότι $\lim\limits_{x \to -\infty} \dfrac{\hm x}{x} = -\infty$.
    \item Αν η συνάρτηση $f$ είναι συνεχής στο $[a,\beta]$, $f(a)=f(\beta)$ και υπάρχει σημείο $\gamma \in (a,\beta)$ ώστε $f(\gamma) \neq f(a)$, τότε η $f$ δεν είναι 1-1.
    \item Η εξίσωση $x^4 + 3x^2 + 5 = 0$ έχει τουλάχιστον μία πραγματική ρίζα σύμφωνα με το Θεώρημα \en{Bolzano}.
\end{enumerate}

\subsection{Διαφορικός Λογισμός (100 Ερωτήσεις)}
\begin{enumerate}[label=\textbf{2.\arabic*.}]
    \item Αν μια συνάρτηση είναι συνεχής στο $x_0$, τότε είναι υποχρεωτικά και παραγωγίσιμη στο $x_0$.
    \item Αν η συνάρτηση $f$ δεν είναι παραγωγίσιμη στο $x_0$, τότε σίγουρα δεν είναι συνεχής στο $x_0$.
    \item Κάθε πολυωνυμική συνάρτηση είναι παραγωγίσιμη στο $\mathbb{R}$.
    \item Αν μια συνάρτηση είναι παραγωγίσιμη σε ένα κλειστό διάστημα $[a, \beta]$, τότε η παράγωγός της είναι υποχρεωτικά συνεχής συνάρτηση στο $[a, \beta]$.
    \item Ο ρυθμός μεταβολής της ταχύτητας ενός κινητού είναι η παράγωγος της επιτάχυνσής του.
    \item Ο συντελεστής διεύθυνσης της εφαπτομένης της $C_f$ στο $M(x_0, f(x_0))$ είναι η τιμή $f'(x_0)$.
    \item Η εφαπτομένη μιας γραφικής παράστασης αποκλείεται να έχει άλλα κοινά σημεία με την καμπύλη.
    \item Μια ευθεία είναι δυνατόν να εφάπτεται σε δύο διαφορετικά σημεία της ίδιας γραφικής παράστασης.
    \item Η παράγωγος της εκθετικής συνάρτησης $f(x) = 5^x$ είναι $f'(x) = x \cdot 5^{x-1}$.
    \item Η εξίσωση της εφαπτομένης της γραφικής παράστασης της $f(x) = e^x$ στο σημείο $A(0, 1)$ είναι η $y = x + 1$.
    \item Αν $f, g$ είναι παραγωγίσιμες συναρτήσεις, τότε $(f(x) \cdot g(x))' = f'(x) \cdot g'(x)$.
    \item Ισχύει ότι $\left( \dfrac{1}{f(x)} \right)' = -\dfrac{1}{(f(x))^2}$ για κάθε $x$ όπου η $f$ ορίζεται και δεν μηδενίζεται.
    \item Η παράγωγος του σταθερού αριθμού $e^{\pi}$ είναι $e^{\pi}$.
    \item Αν $(f(x))^2 + (g(x))^2 = 1$ για κάθε $x \in \mathbb{R}$ με $f,g$ παραγωγίσιμες, τότε $f(x)f'(x) + g(x)g'(x) = 0$.
    \item Αν μια συνάρτηση είναι άρτια και παραγωγίσιμη στο $\mathbb{R}$, η παράγωγός της είναι επίσης άρτια συνάρτηση.
    \item Αν η $f$ είναι περιττή και παραγωγίσιμη στο $\mathbb{R}$, τότε αναγκαστικά $f'(0) = 0$.
    \item Αν η $f$ είναι παραγωγίσιμη και 1-1 στο $\mathbb{R}$, τότε η αντίστροφή της είναι υποχρεωτικά παραγωγίσιμη σε όλο το πεδίο ορισμού της.
    \item Αν δύο συναρτήσεις τέμνονται σε ένα σημείο, τότε υποχρεωτικά έχουν στο κοινό τους σημείο κοινή εφαπτομένη.
    \item Στα σημεία όπου η $C_f$ τέμνει τον άξονα $x'x$, οι εφαπτομένες (αν υπάρχουν) είναι οριζόντιες.
    \item Κρίσιμα σημεία ονομάζονται μόνο τα εσωτερικά σημεία του πεδίου ορισμού όπου η παράγωγος μηδενίζεται.
    \item Τα κρίσιμα σημεία μιας συνάρτησης που είναι ορισμένη σε κλειστό διάστημα, αναζητούνται και στα άκρα του διαστήματος.
    \item Σύμφωνα με το θεώρημα \en{Fermat}, σε κάθε τοπικό ακρότατο η παράγωγος της συνάρτησης μηδενίζεται.
    \item Αν $f'(x_0) = 0$, τότε το $x_0$ είναι απαραίτητα θέση τοπικού ακροτάτου της $f$.
    \item Αν μια συνάρτηση $f$ παρουσιάζει τοπικό ακρότατο σε εσωτερικό σημείο $x_0$ του πεδίου ορισμού της και είναι παραγωγίσιμη σε αυτό, τότε υποχρεωτικά $f'(x_0) = 0$.
    \item Αν η παράγωγος μηδενίζεται σε ένα εσωτερικό σημείο, η εφαπτομένη της καμπύλης σε αυτό το σημείο είναι παράλληλη στον άξονα $x'x$.
    \item Το Θεώρημα Rolle εγγυάται την ύπαρξη ακριβώς μιας οριζόντιας εφαπτομένης.
    \item Αν για μια παραγωγίσιμη συνάρτηση στο $[a,\beta]$ ισχύει $f(a) = f(\beta)$, τότε η $f$ είναι αναγκαστικά σταθερή.
    \item Αν μια παραγωγίσιμη συνάρτηση έχει δύο διαφορετικές ρίζες σε ένα διάστημα, τότε η παράγωγός της έχει τουλάχιστον μία ρίζα ανάμεσά τους.
    \item Το Θεώρημα Μέσης Τιμής εφαρμόζεται αποκλειστικά σε γνησίως μονότονες συναρτήσεις.
    \item Το Θεώρημα \en{Rolle} αποτελεί μια ειδική περίπτωση του Θεωρήματος Μέσης Τιμής.
    \item Το Θ.Μ.Τ. εξασφαλίζει τουλάχιστον ένα σημείο όπου η εφαπτομένη της γραφικής παράστασης είναι παράλληλη προς τη χορδή που ενώνει τα άκρα.
    \item Αν $f'(x) = 0$ σε μια ένωση ξένων διαστημάτων, τότε η $f$ είναι υποχρεωτικά σταθερή σε ολόκληρο το πεδίο ορισμού της.
    \item Αν $f'(x) = g'(x)$ για κάθε $x \in \mathbb{R}^*$, τότε υπάρχει μια ενιαία σταθερά $c$ ώστε $f(x) = g(x) + c$ για κάθε $x \in \mathbb{R}^*$.
    \item Αν η $f'(x) > 0$ σε ένα ανοιχτό διάστημα $\Delta$, τότε η $f$ είναι γνησίως αύξουσα στο $\Delta$.
    \item Αν η $f$ είναι γνησίως αύξουσα σε ένα διάστημα $\Delta$ και παραγωγίσιμη σε αυτό, τότε υποχρεωτικά $f'(x) > 0$ για κάθε εσωτερικό σημείο του $\Delta$.
    \item Υπάρχει συνάρτηση γνησίως αύξουσα στο $\mathbb{R}$ της οποίας η παράγωγος μηδενίζεται σε άπειρα μεμονωμένα σημεία.
    \item Αν η εξίσωση $f'(x) = 0$ έχει μόνο μια ρίζα, τότε η $f$ παρουσιάζει υποχρεωτικά ένα τοπικό ακρότατο.
    \item Κάθε τοπικό ελάχιστο μιας συνάρτησης είναι πάντοτε μικρότερο από κάθε τοπικό μέγιστο της ίδιας συνάρτησης.
    \item Η μέγιστη τιμή μιας συνεχούς συνάρτησης σε κλειστό διάστημα λαμβάνεται είτε σε εσωτερικό κρίσιμο σημείο είτε στα άκρα του διαστήματος.
    \item Μια συνάρτηση είναι δυνατόν να έχει άπειρα τοπικά ακρότατα.
    \item Το σημείο $(x_0, f(x_0))$ λέγεται σημείο καμπής αν η εφαπτομένη διαπερνά την καμπύλη της $f$ σε αυτό το σημείο.
    \item Αν το σημείο $x_0$ είναι σημείο καμπής, τότε υποχρεωτικά ισχύει $f''(x_0) = 0$.
    \item Αν $f''(x_0) = 0$, τότε το $x_0$ είναι υποχρεωτικά θέση σημείου καμπής.
    \item Αν η $f''(x) > 0$ σε ένα διάστημα $\Delta$, τότε η $f$ στρέφει τα κοίλα άνω στο $\Delta$.
    \item Η γραφική παράσταση μιας κυρτής συνάρτησης βρίσκεται πάντοτε πάνω από την εφαπτομένη της σε κάθε σημείο του διαστήματος, με εξαίρεση το σημείο επαφής.
    \item Αν η $f$ είναι παραγωγίσιμη στο $[a, \beta]$ και $f(a) = f(\beta)$, τότε το Θεώρημα \en{Rolle} εξασφαλίζει ότι υπάρχει ακριβώς ένα $x_0 \in (a, \beta)$ με $f'(x_0) = 0$.

\begin{enumerate}[label=\textbf{3.\arabic*.}]

    \item Αν ισχύει $\dint_1^3 f(x) \d x > 0$, τότε υποχρεωτικά υπάρχει τουλάχιστον ένα $x_0 \in [1,3]$ με $f(x_0) > 0$.
    \item Αν η $f'$ είναι συνεχής, τότε $\dint_{a}^{\beta} e^{f(x)} \cdot f'(x) \d x = e^{f(\beta)} - e^{f(a)}$.
    \item Το εμβαδόν του χωρίου μεταξύ της καμπύλης $y=\ln x$, του άξονα $x'x$ και της ευθείας $x=e$ δίνεται από το ολοκλήρωμα $\dint_1^e \ln x \d x$.
    \item Ισχύει ότι $\dint_0^{\pi} \syn x \d x = 0$.
    \item Το ολοκλήρωμα $\dint_1^e \dfrac{\ln x}{x} \d x$ ισούται με $\dfrac{1}{2}$.
    \item Αν δύο συναρτήσεις έχουν ίσα ορισμένα ολοκληρώματα στο διάστημα $[a,\beta]$, τότε οι συναρτήσεις είναι αναγκαστικά ίσες.
    \item Το ολοκλήρωμα $\dint_0^{2\pi} \hm x \d x = 0$ επειδή τα επιμέρους εμβαδά πάνω και κάτω από τον άξονα είναι ίσα.
    \item Αν η $f(x)$ είναι η σταθερή συνάρτηση $f(x)=c$ ($c>0$), το εμβαδόν του αντίστοιχου χωρίου της στο $[a, \beta]$ ταυτίζεται γεωμετρικά με το εμβαδόν ενός ορθογωνίου.
    \item Η μέθοδος αντικατάστασης $u = g(x)$ στο ορισμένο ολοκλήρωμα απαιτεί πάντοτε την αλλαγή των ορίων ολοκλήρωσης με βάση τη νέα μεταβλητή.
    \item Αν η συνάρτηση $f$ είναι συνεχής, τότε ισχύει $\dint_a^\beta f(x) \d x + \dint_\beta^a f(x) \d x = 0$.
    \item Η σχέση $\dint_0^1 f(x)\d x = \dint_0^1 f(1-x)\d x$ ισχύει γενικά για κάθε συνεχή συνάρτηση $f$ στο $[0,1]$.
    \item Ο τύπος για τον υπολογισμό του εμβαδού μεταξύ δύο συναρτήσεων $f, g$ είναι απαραίτητα $\dint_{a}^{\beta} (|f(x)| - |g(x)|) \d x$.
\end{enumerate}
