\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{\sqrt{x+1}-1}{x}$ για $x \ge -1$ και $x \ne 0$.
\begin{erwthma}
\item Για τον υπολογισμό του ορίου στο 0, πολλαπλασιάζουμε με τη συζυγή παράσταση:
\[ \lim_{x\to0} f(x) = \lim_{x\to0} \frac{\sqrt{x+1}-1}{x} = \lim_{x\to0} \frac{(\sqrt{x+1}-1)(\sqrt{x+1}+1)}{x(\sqrt{x+1}+1)} = \lim_{x\to0} \frac{x+1-1}{x(\sqrt{x+1}+1)} \]
\[ = \lim_{x\to0} \frac{x}{x(\sqrt{x+1}+1)} = \lim_{x\to0} \frac{1}{\sqrt{x+1}+1} = \frac{1}{\sqrt{1}+1} = \frac{1}{2} \]

\item Αφού η $g$ αποτελεί συνεχή επέκταση της $f$ στο $[-1, +\infty)$, πρέπει η τιμή της στο $x_0 = 0$ να ισούται με το όριό της. Δηλαδή $g(0) = \lim\limits_{x\to0} f(x) = \frac{1}{2}$.
Ο τύπος της $g$ είναι:
\[ g(x) = \begin{cases} \frac{\sqrt{x+1}-1}{x}, & x \in [-1, 0) \cup (0, +\infty) \\ \frac{1}{2}, & x = 0 \end{cases} \]

\item Εξετάζουμε τη συνεχή(ως επέκταση) συνάρτηση $g$ στο κλειστό διάστημα $[0, 3]$.
Υπολογίζουμε τις τιμές στα άκρα:
- $g(0) = \frac{1}{2} = 0,5$.
- $g(3) = \frac{\sqrt{3+1}-1}{3} = \frac{2-1}{3} = \frac{1}{3} \approx 0,33$.
Παρατηρούμε ότι το $\frac{2}{5} = 0,4$ βρίσκεται ακριβώς ανάμεσα στις τιμές $g(3)$ και $g(0)$, αφού $0,33 < 0,4 < 0,5$. 
Συγκεκριμένα: $g(3) < \frac{2}{5} < g(0)$.
Αφού η $g$ είναι συνεχής στο $[0, 3]$, το Θεώρημα Ενδιάμεσων Τιμών εξασφαλίζει ότι υπάρχει τουλάχιστον ένα $\xi \in (0, 3)$ τέτοιο ώστε $g(\xi) = \frac{2}{5}$.

\item Η συνάρτηση $g$ (που ταυτίζεται με την $f$) είναι παραγωγίσιμη στο $(-1, 0) \cup (0, +\infty)$. Υπολογίζουμε την $g'$:
\[ g'(x) = \frac{ \frac{1}{2\sqrt{x+1}}\cdot x - (\sqrt{x+1}-1)\cdot 1 }{x^2} = \frac{ x - 2\sqrt{x+1}(\sqrt{x+1}-1) }{2x^2\sqrt{x+1}} = \frac{ x - 2(x+1) + 2\sqrt{x+1} }{2x^2\sqrt{x+1}} \]
\[ g'(x) = \frac{ -x - 2 + 2\sqrt{x+1} }{2x^2\sqrt{x+1}} \]
Για τον αριθμητή, ισχύει από βασική ταυτότητα: $(\sqrt{x+1}-1)^2 > 0 \Rightarrow x+1 - 2\sqrt{x+1} + 1 > 0 \Rightarrow x+2 > 2\sqrt{x+1}$.
Οπότε $-x-2+2\sqrt{x+1} < 0$. Αφού ο παρονομαστής είναι θετικός, $g'(x) < 0$ για κάθε $x \neq 0$.
Επειδή η $g$ είναι συνεχής και στο 0, συνάγεται ότι είναι γνησίως φθίνουσα σε όλο το $[-1, +\infty)$. Επειδή είναι γνησίως μονότονη, είναι "1-1" κι επομένως αντιστρέφεται στο πεδίο ορισμού της.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η πολυωνυμική συνάρτηση $f(x) = x^4 - 2x^2 + 1$.
\begin{erwthma}
\item Η συνάρτηση $f$ ικανοποιεί τις προϋποθέσεις του Θεωρήματος Rolle στο διάστημα $[-1, 1]$:
- Είναι συνεχής στο $[-1, 1]$ (ως πολυωνυμική).
- Είναι παραγωγίσιμη στο $(-1, 1)$.
- Ισχύει $f(-1) = (-1)^4 - 2(-1)^2 + 1 = 1 - 2 + 1 = 0$ και $f(1) = 1^4 - 2(1)^2 + 1 = 1 - 2 + 1 = 0$. Δηλαδή $f(-1) = f(1)$.
Άρα σύμφωνα με το Θεώρημα Rolle υπάρχει τουλάχιστον ένα $\xi \in (-1, 1)$ τέτοιο ώστε $f'(\xi) = 0$. Η εφαπτομένη στο $\xi$ θα είναι παράλληλη στον άξονα $x'x$, άρα οριζόντια.

\item Υπολογίζουμε την παράγωγο της $f$:
\[ f'(x) = 4x^3 - 4x = 4x(x^2 - 1) \]
Η εξίσωση $f'(x) = 0 \Leftrightarrow 4x(x-1)(x+1) = 0$ έχει ρίζες τα $x_1 = 0$, $x_2 = 1$ και $x_3 = -1$.
Από τις ρίζες αυτές, μόνο η τιμή $\xi = 0$ ανήκει στο ανοιχτό διάστημα $(-1, 1)$. Οπότε η ζητούμενη τιμή είναι το $\xi = 0$.

\item Το πρόσημο της $f'$ φαίνεται στον παρακάτω πίνακα μεταβολών:

Εντοπίζονται τα ακόλουθα ακρότατα:
- Τοπικό ελάχιστο στα $x=-1$ και $x=1$, με τιμές $f(-1) = f(1) = 0$.
- Τοπικό μέγιστο στο $x=0$, με τιμή $f(0) = 1$.

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x) = (4x^3 - 4x)' = 12x^2 - 4 = 4(3x^2 - 1) \]
Η $f''$ μηδενίζεται όταν $3x^2 - 1 = 0 \Leftrightarrow x^2 = \frac{1}{3} \Leftrightarrow x = \pm \frac{1}{\sqrt{3}} = \pm \frac{\sqrt{3}}{3}$.

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^x(x-1) + 1$, με $x \in \mathbb{R}$.
\begin{erwthma}
\item Για το όριο στο $-\infty$, γράφουμε το γινόμενο σε μορφή πηλίκου:
\[ \lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} (e^x(x-1) + 1) = \lim_{x\to-\infty} \frac{x-1}{e^{-x}} + 1 \]
Επειδή $\lim_{x\to-\infty} (x-1) = -\infty$ και $\lim_{x\to-\infty} e^{-x} = +\infty$, το κλάσμα δημιουργεί απροσδιοριστία $\frac{-\infty}{+\infty}$. Εφαρμόζουμε τον κανόνα De L' Hospital:
\[ \lim_{x\to-\infty} \frac{x-1}{e^{-x}} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to-\infty} \frac{(x-1)'}{(e^{-x})'} = \lim_{x\to-\infty} \frac{1}{-e^{-x}} = \lim_{x\to-\infty} (-e^x) = 0 \]
Συνεπώς το ζητούμενο όριο ισούται με $\lim_{x\to-\infty} f(x) = 0 + 1 = 1$.

\item Υπολογίζουμε την πρώτη και δεύτερη παράγωγο της $f$:
\[ f'(x) = e^x(x-1) + e^x(1) + 0 = x e^x - e^x + e^x = x e^x \]
\[ f''(x) = 1 \cdot e^x + x e^x = e^x(x+1) \]
\textbf{Μονοτονία:}
Ρίζα της $f'$ είναι το $x=0$.
- $f'(x) > 0 \Leftrightarrow x > 0$, άρα η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$.
- $f'(x) < 0 \Leftrightarrow x < 0$, άρα η $f$ είναι γνησίως φθίνουσα στο $(-\infty, 0]$.
Παρουσιάζει ολικό ελάχιστο στο $x=0$ με τιμή $f(0) = e^0(0-1)+1 = 0$.
\textbf{Κυρτότητα:}
Ρίζα της $f''$ είναι το $x=-1$.
- $f''(x) > 0 \Leftrightarrow x > -1$, άρα η $f$ είναι κυρτή στο $[-1, +\infty)$.
- $f''(x) < 0 \Leftrightarrow x < -1$, άρα η $f$ είναι κοίλη στο $(-\infty, -1]$.
Παρουσιάζει σημείο καμπής στο σημείο $x=-1$ με τιμή $f(-1) = e^{-1}(-2) + 1 = 1 - \frac{2}{e}$.

\item Θεωρούμε τη συνάρτηση $h(x) = f(x) - \frac{1}{2}$ στο κλειστό $[0, 1]$.
Η $h$ είναι συνεχής. Στα άκρα:
- $h(0) = f(0) - 0,5 = 0 - 0,5 = -0,5 < 0$.
- $h(1) = f(1) - 0,5 = 1 - 0,5 = 0,5 > 0$.
Επειδή $h(0)h(1) < 0$, σύμφωνα με το Θεώρημα Bolzano υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $h(x_0)=0 \Leftrightarrow f(x_0) = \frac{1}{2}$.
Διαπιστώσαμε ότι η $f$ είναι γνησίως αύξουσα στο $[0, +\infty)$, άρα και στο υποσύνολό του $[0, 1]$. Έτσι το $x_0$ είναι μοναδικό.

\item Το σημείο επαφής είναι το $M(0, f(0))$, δηλαδή το $(0, 0)$.
Ο συντελεστής διεύθυνσης της εφαπτομένης είναι $f'(0) = 0 \cdot e^0 = 0$. Επομένως, πρόκειται για οριζόντια εφαπτομένη.
Η εξίσωσή της δίνεται από τον τύπο $y - f(0) = f'(0)(x-0) \Leftrightarrow y - 0 = 0(x-0) \Leftrightarrow y = 0$. 
Οπότε η εφαπτομένη ταυτίζεται με τον άξονα $x'x$.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{x - \hm x}{x + \hm x}$ με $x \neq 0$. (Η εκφώνηση υπονοεί $x>0$ ή ανοικτό διάστημα γύρω από το 0 αποκλείοντας το 0 και άλλα σημεία μηδενισμού, πχ $x=-\hm x$. Στην περιοχή του $0$ ισχύει.)
\begin{erwthma}
\item Για το όριο στο 0 διαιρούμε αριθμητή και παρονομαστή με $x \neq 0$:
\[ \lim_{x\to0} f(x) = \lim_{x\to0} \frac{x - \hm x}{x + \hm x} = \lim_{x\to0} \frac{\frac{x}{x} - \frac{\hm x}{x}}{\frac{x}{x} + \frac{\hm x}{x}} = \lim_{x\to0} \frac{1 - \frac{\hm x}{x}}{1 + \frac{\hm x}{x}} \]
Γνωρίζοντας το βασικό όριο $\lim_{x\to0} \frac{\hm x}{x} = 1$, έχουμε:
\[ \lim_{x\to0} f(x) = \frac{1 - 1}{1 + 1} = \frac{0}{2} = 0 \]

\item Για το όριο στο $+\infty$, ακολουθούμε παρόμοια τεχνική βγάζοντας κοινό παράγοντα το $x$:
\[ \lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} \frac{x\left(1 - \frac{\hm x}{x}\right)}{x\left(1 + \frac{\hm x}{x}\right)} = \lim_{x\to+\infty} \frac{1 - \frac{\hm x}{x}}{1 + \frac{\hm x}{x}} \]
Από την ιδιότητα $|\hm x| \le 1 \Rightarrow -1 \le \hm x \le 1$. Διαιρώντας με $x > 0$:
\[ -\frac{1}{x} \le \frac{\hm x}{x} \le \frac{1}{x} \]
Επειδή $\lim_{x\to+\infty} \left(-\frac{1}{x}\right) = 0$ και $\lim_{x\to+\infty} \frac{1}{x} = 0$, σύμφωνα με το Κριτήριο Παρεμβολής ισχύει $\lim_{x\to+\infty} \frac{\hm x}{x} = 0$.
Επομένως το ζητούμενο όριο είναι:
\[ \lim_{x\to+\infty} f(x) = \frac{1 - 0}{1 + 0} = 1 \]

\item Η συνάρτηση $h(x) = x - \hm x$ είναι παραγωγίσιμη στο $\mathbb{R} $ με:
\[ h'(x) = 1 - \syn x \]
Γνωρίζουμε ότι $\syn x \le 1 \Rightarrow 1 - \syn x \ge 0$.
Άρα $h'(x) \ge 0$ για κάθε $x \in \mathbb{R}$. Το $h'(x)$ μηδενίζεται μόνο στα απομονωμένα σημεία $x = 2\kappa\pi$ ($\kappa \in \mathbb{Z}$). 
Δεδομένου ότι η παράγωγος δεν είναι μηδέν σε κανένα διάστημα (μη πεπερασμένου πλάτους), η $h$ είναι γνησίως αύξουσα στο $\mathbb{R}$.

\item Ζητείται η μοναδικότητα της ρίζας για την εξίσωση $x = \hm x \Leftrightarrow x - \hm x = 0 \Leftrightarrow h(x) = 0$.
Παρατηρούμε ότι για $x = 0$, ισχύει $h(0) = 0 - \hm 0 = 0$. Οπότε το $x=0$ είναι προφανής ρίζα της εξίσωσης.
Επιπροσθέτως, όπως αποδείχθηκε στο (Β3), η συνάρτηση $h(x)$ είναι γνησίως αύξουσα σε όλο το $\mathbb{R}$. 
Επομένως, η $h(x)$ τέμνει τον άξονα $x'x$ το πολύ σε ένα σημείο. 
Άρα το $x=0$ αποτελεί τη μοναδική ρίζα της εξίσωσης στο $\mathbb{R}$.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \ln x$ με πεδίο ορισμού $(0, +\infty)$.
\begin{erwthma}
\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για τη συνάρτηση $f(t) = \ln t$ στο κλειστό διάστημα $[x, x+1]$ με $x > 0$:
- Η $f$ είναι συνεχής στο $[x, x+1]$.
- Η $f$ είναι παραγωγίσιμη στο $(x, x+1)$ με $f'(t) = \frac{1}{t}$.
Κατά συνέπεια, existe τουλάχιστον ένα $\xi \in (x, x+1)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(x+1) - f(x)}{x+1 - x} \Leftrightarrow \frac{1}{\xi} = \frac{\ln(x+1) - \ln x}{1} \Leftrightarrow \frac{1}{\xi} = \ln(x+1) - \ln x \]

\item Από την προηγούμενη σχέση έχουμε: 
\[ \ln(x+1) - \ln x = \ln\left( \frac{x+1}{x} \right) = \ln\left( 1 + \frac{1}{x} \right) \]
Άρα $\ln\left( 1 + \frac{1}{x} \right) = \frac{1}{\xi}$.
Επίσης, η επιλογή του $\xi$ μάς διασφαλίζει ότι $x < \xi < x+1$.
Επειδή όλες οι ποσότητες είναι θετικές, η αντιστροφή των όρων αναστρέφει τη φορά των ανισοτήτων:
\[ \frac{1}{x+1} < \frac{1}{\xi} < \frac{1}{x} \]
Αντικαθιστώντας το $\frac{1}{\xi}$ με $\ln\left( 1 + \frac{1}{x} \right)$:
\[ \frac{1}{x+1} < \ln\left( 1 + \frac{1}{x} \right) < \frac{1}{x} \]
Το ζητούμενο αποδείχθηκε.

\item Πολλαπλασιάζουμε την προκύπτουσα ανισότητα με $x > 0$:
\[ x \cdot \frac{1}{x+1} < x \cdot \ln\left( 1 + \frac{1}{x} \right) < x \cdot \frac{1}{x} \]
\[ \frac{x}{x+1} < x \ln\left( 1 + \frac{1}{x} \right) < 1 \]
Υπολογίζουμε τα όρια των ακραίων μελών καθώς το $x \to +\infty$:
- $\lim_{x\to+\infty} \frac{x}{x+1} = \lim_{x\to+\infty} \frac{x}{x} = 1$.
- $\lim_{x\to+\infty} 1 = 1$.
Άρα, από το Κριτήριο Παρεμβολής, προκύπτει ότι:
\[ \lim_{x \to +\infty} \left[ x \ln\left( 1 + \frac{1}{x} \right) \right] = 1 \]

\item Ως δεύτερος τρόπος για το όριο, θέτουμε $u = \frac{1}{x}$. Όταν $x \to +\infty$, ισχύει $u \to 0^+$.
Το όριο μετασχηματίζεται:
\[ \lim_{x\to+\infty} \left[ x \ln\left( 1 + \frac{1}{x} \right) \right] = \lim_{u\to0^+} \left[ \frac{1}{u} \ln(1+u) \right] = \lim_{u\to0^+} \frac{\ln(1+u)}{u} \]
Το όριο αυτό οδηγεί σε απροσδιοριστία $\frac{0}{0}$. Μπορούμε να εφαρμόσουμε τον κανόνα De L' Hospital:
\[ \lim_{u\to0^+} \frac{(\ln(1+u))'}{(u)'} = \lim_{u\to0^+} \frac{\frac{1}{1+u}}{1} = \frac{1}{1+0} = 1 \]
Ο δεύτερος τρόπος επιβεβαιώνει απολύτως το αποτέλεσμα.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = e^{-x^2}$.
\begin{erwthma}
\item Εφαρμόζουμε το Θεώρημα Rolle για την $f$ στο διάστημα $[-1, 1]$:
- Η $f$ είναι συνεχής στο $[-1, 1]$.
- Η $f$ είναι παραγωγίσιμη στο $(-1, 1)$.
- Ισχύει $f(-1) = e^{-(-1)^2} = e^{-1}$ και $f(1) = e^{-1^2} = e^{-1}$, δηλαδή $f(-1) = f(1)$.
Επομένως υπάρχει τουλάχιστον ένα $\xi \in (-1, 1)$ τέτοιο ώστε $f'(\xi) = 0$. Η εφαπτομένη σε τέτοιο σημείο είναι οριζόντια.
Πράγματι, $f'(x) = -2x e^{-x^2}$. Η ρίζα είναι προφανώς το $\xi = 0$, το οποίο ανήκει στο $(-1, 1)$. Το σημείο επαφής είναι το $(0, 1)$.

\item Το πρόσημο της παραγώγου $f'(x) = -2x e^{-x^2}$ εξαρτάται μόνο από το $-2x$, αφού $e^{-x^2} > 0$.
- $f'(x) > 0 \Leftrightarrow -2x > 0 \Leftrightarrow x < 0$.
- $f'(x) < 0 \Leftrightarrow -2x < 0 \Leftrightarrow x > 0$.

Η $f$ είναι γνησίως αύξουσα στο $(-\infty, 0]$ και γνησίως φθίνουσα στο $[0, +\infty)$.

\item Υπολογίζουμε τη δεύτερη παράγωγο:
\[ f''(x) = (-2x)' e^{-x^2} - 2x(e^{-x^2})' = -2 e^{-x^2} - 2x(-2x e^{-x^2}) = -2 e^{-x^2} + 4x^2 e^{-x^2} = 2e^{-x^2}(2x^2 - 1) \]
Η δεύτερη παράγωγο μηδενίζεται:
\[ f''(x) = 0 \Leftrightarrow 2x^2 - 1 = 0 \Leftrightarrow x^2 = \frac{1}{2} \Leftrightarrow x = \pm \frac{1}{\sqrt{2}} = \pm \frac{\sqrt{2}}{2} \]

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = \dfrac{e^x}{x}$ με $x \neq 0$.
\begin{erwthma}
\item Εξετάζουμε τα σημεία τομής με τον άξονα $x'x$. Θέτουμε $f(x) = 0$:
\[ \frac{e^x}{x} = 0 \Leftrightarrow e^x = 0 \]
Η εξίσωση $e^x = 0$ είναι \textbf{αδύνατη} στο $\mathbb{R}$, καθώς η εκθετική συνάρτηση λαμβάνει μόνο θετικές τιμές ($e^x > 0$ για κάθε $x \in \mathbb{R}$). 
Άρα η $C_f$ δεν έχει κανένα κοινό σημείο με τον άξονα $x'x$.

\item Υπολογίζουμε την παράγωγο της $f$:
\[ f'(x) = \frac{(e^x)'x - e^x(x)'}{x^2} = \frac{xe^x - e^x}{x^2} = \frac{e^x(x-1)}{x^2} \]
Η ρίζα της παραγώγου προκύπτει από τον αριθμητή: $x - 1 = 0 \Leftrightarrow x = 1$.
Το πρόσημο εξαρτάται μόνο από το $x-1$ αφού $e^x > 0$ και $x^2 > 0$ (για $x \neq 0$).
- $f'(x) > 0 \Leftrightarrow x > 1$.
- $f'(x) < 0 \Leftrightarrow x < 1$ (και προφανώς $x \ne 0$).

\item \textbf{Όρια στο $\pm\infty$:}
- Στο $-\infty$: $\lim_{x\to-\infty} f(x) = \lim_{x\to-\infty} \frac{e^x}{x} = 0$, διότι $\lim e^x = 0$ και $\lim \frac{1}{x} = 0$ (άρα $0 \cdot 0 = 0$).

- Στο $0^+$: $\lim_{x\to0^+} \frac{e^x}{x} = \frac{e^0}{0^+} = \frac{1}{0^+} = +\infty$.
- Στο $0^-$: $\lim_{x\to0^-} \frac{e^x}{x} = \frac{e^0}{0^-} = \frac{1}{0^-} = -\infty$.

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^3 + x - 1$ με $x \in \mathbb{R}$.
\begin{erwthma}
\item Η συνάρτηση $f$ είναι πολυωνυμική, άρα παραγωγίσιμη στο $\mathbb{R}$ με:
\[ f'(x) = 3x^2 + 1 \]
Επειδή $3x^2 \ge 0 \Rightarrow 3x^2+1 \ge 1 > 0$, έχουμε $f'(x) > 0$ για κάθε $x \in \mathbb{R}$.
Άρα η $f$ είναι \textbf{γνησίως αύξουσα} σε όλο το $\mathbb{R}$.

\item Θεωρούμε την $f$ στο κλειστό διάστημα $[0, 1]$.
Η $f$ είναι συνεχής στο $[0, 1]$.
Υπολογίζουμε τις τιμές στα άκρα:
- $f(0) = 0^3 + 0 - 1 = -1 < 0$.
- $f(1) = 1^3 + 1 - 1 = 1 > 0$.
Αφού $f(0)f(1) < 0$, σύμφωνα με το Θεώρημα Bolzano έχει τουλάχιστον μία ρίζα $x_0 \in (0, 1)$ ώστε $f(x_0)=0$.
Δεδομένου ότι η $f$ είναι γνησίως αύξουσα στο $\mathbb{R}$, η ρίζα αυτή είναι \textbf{ακριβώς μία}.

\item Εξετάζουμε το πρόσημο της $f(x)$:
- Για $x < x_0$, επειδή η $f$ είναι γνησίως αύξουσα, ισχύει $f(x) < f(x_0) \Rightarrow f(x) < 0$.
- Για $x > x_0$, ομοίως ισχύει $f(x) > f(x_0) \Rightarrow f(x) > 0$.

\item Υπολογίζουμε το ζητούμενο όριο: $L = \lim_{x\to+\infty} \left[ f(x) \hm \left( \frac{1}{f(x)} \right) \right]$.
Κατ' αρχάς, $\lim_{x\to+\infty} f(x) = \lim_{x\to+\infty} (x^3 + x - 1) = \lim_{x\to+\infty} x^3 = +\infty$.
Θέτουμε $u = \frac{1}{f(x)}$. Καθώς το $x \to +\infty$, το $f(x) \to +\infty$, άρα $u \to 0^+$.
Τότε το $f(x) = \frac{1}{u}$ και το όριο μετατρέπεται στο εξής:
\[ L = \lim_{u\to0^+} \left[ \frac{1}{u} \hm u \right] = \lim_{u\to0^+} \frac{\hm u}{u} = 1 \]
\end{erwthma}
\end{thema}

\begin{thema}{Β}
Δίνεται η συνάρτηση $f(x) = x^2 \ln x$ με $x > 0$.
\begin{erwthma}
\item Υπολογίζουμε το όριο στο $0^+$ γράφοντας το γινόμενο ως πηλίκο για να εφαρμόσουμε De L' Hospital:
\[ \lim_{x\to0^+} f(x) = \lim_{x\to0^+} (x^2 \ln x) = \lim_{x\to0^+} \frac{\ln x}{\frac{1}{x^2}} \xlongequal[\text{DLH}]{\frac{-\infty}{+\infty}} \lim_{x\to0^+} \frac{\frac{1}{x}}{-\frac{2}{x^3}} = \lim_{x\to0^+} \left( \frac{1}{x} \cdot \frac{x^3}{-2} \right) = \lim_{x\to0^+} \left( -\frac{x^2}{2} \right) = 0 \]

\item Για να αποτελεί η $g$ συνεχή επέκταση της $f$ στο $0$, πρέπει να ισχύει $g(0) = \lim_{x\to0^+} f(x)$.
Σύμφωνα με το προηγούμενο ερώτημα $\lim_{x\to0^+} f(x) = 0$, συνεπώς ορίζουμε $g(0) = 0$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για την $f$ στο διάστημα $[1, e]$:
- Η $f$ είναι συνεχής στο $[1, e]$.
- Η $f$ είναι παραγωγίσιμη στο $(1, e)$.
Ορίζονται τα σημεία $A(1, f(1))$ και $B(e, f(e))$. 
Υπολογίζουμε $f(1) = 1^2 \ln 1 = 0 \Rightarrow A(1, 0)$ και $f(e) = e^2 \ln e = e^2 \Rightarrow B(e, e^2)$.
Ο συντελεστής διεύθυνσης της τέμνουσας $AB$ είναι: $\lambda_{AB} = \frac{f(e) - f(1)}{e - 1} = \frac{e^2 - 0}{e-1} = \frac{e^2}{e-1}$.
Από το Θ.Μ.Τ. υπάρχει τουλάχιστον ένα $\xi \in (1, e)$ τέτοιο ώστε $f'(\xi) = \frac{e^2}{e-1} \Leftrightarrow f'(\xi) = \lambda_{AB}$.
Από τη γεωμετρική ερμηνεία, η εφαπτομένη της $C_f$ στο $x=\xi$ είναι παράλληλη προς την τέμνουσα $AB$.

\item Υπολογίζουμε την πρώτη και zweite παράγωγο:
\[ f'(x) = 2x \ln x + x^2 \frac{1}{x} = 2x \ln x + x \]
\[ f''(x) = 2\ln x + 2x \frac{1}{x} + 1 = 2\ln x + 2 + 1 = 2\ln x + 3 \]
Η δεύτερη παράγωγος μηδενίζεται: $f''(x) = 0 \Leftrightarrow 2\ln x = -3 \Leftrightarrow \ln x = -1,5 \Leftrightarrow x = e^{-3/2}$.
- Για $x > e^{-3/2}$, ισχύει $\ln x > -1,5 \Rightarrow 2\ln x + 3 > 0 \Rightarrow f''(x) > 0$. Η $f$ είναι \textbf{κυρτή} στο $[e^{-3/2}, +\infty)$.
- Για $0 < x < e^{-3/2}$, ισχύει $f''(x) < 0$. Η $f$ είναι \textbf{κοίλη} στο $(0, e^{-3/2}]$.

\begin{thema}{Β}
Δίνονται οι συναρτήσεις $f(x) = e^x + 2x$ και $g(x) = x^2 + 1$.
\begin{erwthma}
\item Το πεδίο ορισμού της $f$ είναι το $\mathbb{R}$, και της $g$ είναι επίσης το $\mathbb{R}$.
Για να ορίζεται η $f \circ g$, πρέπει: $x \in A_g \Rightarrow x \in \mathbb{R}$ και $g(x) \in D_f \Rightarrow x^2+1 \in \mathbb{R}$. 
Αυτό ισχύει για κάθε πραγματικό $x$. Άρα το $A_{f\circ g} = \mathbb{R}$.
Ο τύπος της σύνθεσης είναι:
\[ (f \circ g)(x) = f(g(x)) = f(x^2+1) = e^{x^2+1} + 2(x^2+1) = e^{x^2+1} + 2x^2 + 2 \]

\item Θεωρούμε τη συνάρτηση $h(x) = f(x) - 3 = e^x + 2x - 3$ στο διάστημα $[0, 1]$.
Η $h$ είναι συνεχής στο $[0, 1]$.
Βρίσκουμε τις τιμές της στα άκρα:
- $h(0) = f(0) - 3 = e^0 + 0 - 3 = 1 - 3 = -2 < 0$.
- $h(1) = f(1) - 3 = e^1 + 2(1) - 3 = e - 1 > 0$ (αφού $e \approx 2,71 > 1$).
Δεδομένου ότι $h(0)h(1) < 0$, το Θεώρημα Bolzano εξασφαλίζει ότι υπάρχει τουλάχιστον ένα $x_0 \in (0, 1)$ τέτοιο ώστε $h(x_0) = 0 \Leftrightarrow f(x_0) = 3$.

\item Υπολογίζουμε την παράγωγο της $f$:
\[ f'(x) = e^x + 2 \]
Γνωρίζουμε ότι η εκθετική συνάρτηση είναι πάντοτε αυστηρά θετική ($e^x > 0$) για κάθε $x \in \mathbb{R}$.
Επομένως, $f'(x) = e^x + 2 > 0 + 2 = 2 > 0$. 
Άρα η εξίσωση $f'(x) = 0$ \textbf{δεν έχει προφανώς καμία λύση} στο $\mathbb{R}$.
Το συμπέρασμα για τη μονοτονία είναι ότι, αφού $f'(x) > 0$ για κάθε $x \in \mathbb{R}$, η συνάρτηση $f$ είναι \textbf{γνησίως αύξουσα} στο $\mathbb{R}$.

\item Εφαρμόζουμε το Θεώρημα Μέσης Τιμής (Θ.Μ.Τ.) για την $f$ στο κλειστό διάστημα $[0, x_0]$ (όπου $x_0 \in (0, 1)$ είναι η ρίζα που βρήκαμε στο ερώτημα 2):
- Η $f$ είναι συνεχής στο $[0, x_0]$.
- Η $f$ είναι παραγωγίσιμη στο $(0, x_0)$.
Σύμφωνα με το Θ.Μ.Τ., υπάρχει τουλάχιστον ένα $\xi \in (0, x_0)$ τέτοιο ώστε:
\[ f'(\xi) = \frac{f(x_0) - f(0)}{x_0 - 0} \]
Από το ερώτημα (Β2) γνωρίζουμε ότι $f(x_0) = 3$ και $f(0) = 1$. Οπότε αντικαθιστώντας λαμβάνουμε:
\[ f'(\xi) = \frac{3 - 1}{x_0} = \frac{2}{x_0} \]
Το ζητούμενο αποδείχθηκε.
\end{erwthma}
\end{thema}
