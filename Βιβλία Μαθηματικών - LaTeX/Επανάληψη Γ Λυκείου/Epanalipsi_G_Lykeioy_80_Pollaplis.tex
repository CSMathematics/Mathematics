    \item Κάθε πολυωνυμική συνάρτηση $P(x)$ περιττού βαθμού $\ge 3$ έχει οπωσδήποτε:

    \item Η παράγωγος της συνάρτησης $f(x) = a^x$ ($a > 0, a \neq 1$) είναι:

        \task Την παράγουσα της $f$.

\section*{ΚΕΦΑΛΑΙΟ 3: Ολοκληρωτικός Λογισμός (20 Ερωτήσεις)}

\begin{enumerate}[label=\textbf{3.\arabic*.}, itemsep=4mm]
    \item Αν $a, \beta \in \mathbb{R}$ και $f$ συνεχής, ισχύει πάντα η ιδιότητα:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task $\dint_a^\beta f(x) \d x = \dint_\beta^a f(x) \d x$
        \task $\dint_a^\beta f(x) \d x = - \dint_\beta^a f(x) \d x$
        \task $\dint_a^\beta f(x) \d x = 0$
        \task $\dint_a^a f(x) \d x = 1$
    \end{tasks}

    \item Για κάθε συνεχή συνάρτηση $f$, ισχύει:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](4)
        \task $\dint_a^a f(x) \d x = 0$
        \task $\dint_a^a f(x) \d x = f(a)$
        \task $\dint_a^a f(x) \d x = a$
        \task $\dint_a^a f(x) \d x = 1$
    \end{tasks}

    \item Αν $f(x) \ge 0$ για κάθε $x \in [a, \beta]$ (με $a < \beta$), το εμβαδόν του χωρίου μεταξύ της $C_f$, του $x'x$ και των $x=a, x=\beta$ είναι:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](4)
        \task $\dint_\beta^a f(x) \d x$
        \task $-\dint_a^\beta f(x) \d x$
        \task $\dint_a^\beta f(x) \d x$
        \task $f(\beta) - f(a)$
    \end{tasks}

    \item Το ορισμένο ολοκλήρωμα μιας σταθεράς $c$ στο διάστημα $[a, \beta]$ ισούται με:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](4)
        \task $c$
        \task $c \cdot x$
        \task $c(a - \beta)$
        \task $c(\beta - a)$
    \end{tasks}

    \item Αν η $f$ είναι συνεχής και \textbf{άρτια} συνάρτηση, τότε για κάθε $a > 0$ ισχύει:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task $\dint_{-a}^a f(x) \d x = 0$
        \task $\dint_{-a}^a f(x) \d x = 2\dint_0^a f(x) \d x$
        \task $\dint_{-a}^a f(x) \d x = f(a)$
        \task $\dint_{-a}^a f(x) \d x = -2\dint_0^a f(x) \d x$
    \end{tasks}

    \item Αν η $f$ είναι συνεχής και \textbf{περιττή} συνάρτηση, τότε για κάθε $a > 0$ ισχύει:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task $\dint_{-a}^a f(x) \d x = 0$
        \task $\dint_{-a}^a f(x) \d x = 2\dint_0^a f(x) \d x$
        \task $\dint_{-a}^a f(x) \d x = 2f(a)$
        \task Δεν υπολογίζεται.
    \end{tasks}

    \item Για τρεις πραγματικούς αριθμούς $a, \beta, \gamma$ και $f$ συνεχή, ισχύει η σχέση $\dint_a^\beta f(x)\d x + \dint_\beta^\gamma f(x)\d x = \dint_a^\gamma f(x)\d x$:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task Μόνο αν $a < \beta < \gamma$.
        \task Μόνο αν η $f$ παίρνει θετικές τιμές.
        \task Για οποιουσδήποτε αριθμούς $a, \beta, \gamma$.
        \task Ποτέ.
    \end{tasks}

    \item Ο τύπος ολοκλήρωσης κατά παράγοντες για το ορισμένο ολοκλήρωμα είναι:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](1)
        \task $\dint_a^\beta f'(x)g(x)\d x = [f(x)g(x)]_a^\beta - \dint_a^\beta f(x)g'(x)\d x$
        \task $\dint_a^\beta f'(x)g'(x)\d x = f(x)g(x)$
        \task $\dint_a^\beta f'(x)g(x)\d x = f(x)g(x) - \dint_a^\beta f(x)g'(x)\d x$
        \task $\dint_a^\beta (f(x)g(x))'\d x = 0$
    \end{tasks}

    \item Κατά την εφαρμογή της μεθόδου αντικατάστασης $u = g(x)$ στο ορισμένο ολοκλήρωμα $\dint_a^\beta f(g(x))g'(x)\d x$, πρέπει απαραιτήτως:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task Να διατηρηθούν τα όρια ολοκλήρωσης $a$ και $\beta$.
        \task Να αλλάξουν τα όρια σε $u_1 = g(a)$ και $u_2 = g(\beta)$.
        \task Να αφαιρεθεί η συνάρτηση $g'(x)$ χωρίς άλλη αλλαγή.
        \task Να υπολογιστεί το εμβαδόν αντί του ολοκληρώματος.
    \end{tasks}

    \item Αν η $f$ είναι συνεχής, $f(x) \ge 0$ για κάθε $x \in [a, \beta]$ (με $a<\beta$) και $\dint_a^\beta f(x)\d x = 0$, τότε:

    \end{tasks}

    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task $f(x) \ge 0$ για κάθε $x \in [a, \beta]$.

        \task Δεν εξάγεται το συμπέρασμα ότι $f(x) \ge 0$ παντού.

    \end{tasks}

    \item Αν $\dint_1^2 f(x) \d x = 3$ και $\dint_2^1 g(x) \d x = 5$, τότε το $\dint_1^2 (f(x) + g(x)) \d x$ ισούται με:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](4)
        \task $8$
        \task $-2$
        \task $2$
        \task $-8$
    \end{tasks}

    \item Το ολοκλήρωμα $\dint_a^\beta f(x) \d x$:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task Εξαρτάται από τη μεταβλητή ολοκλήρωσης (π.χ. αν είναι $x$ ή $t$).
        \task Ανεξαρτητοποιείται από τα όρια $a, \beta$.
        \task Εξαρτάται αποκλειστικά από τη συνάρτηση $f$ και τα όρια $a, \beta$.
        \task Είναι πάντα ίσο με μηδέν.
    \end{tasks}

    \item Το ορισμένο ολοκλήρωμα μιας συνεχούς συνάρτησης σε ένα κλειστό διάστημα:
    \begin{tasks}[label=\let\textdexiakeraia\relax\alph*.](2)
        \task Είναι μια οικογένεια συναρτήσεων.
        \task Είναι ένας μοναδικός πραγματικός αριθμός.
        \task Είναι πάντα θετικός αριθμός.
        \task Παριστάνει πάντα εμβαδόν.
    \end{tasks}

\end{enumerate}
