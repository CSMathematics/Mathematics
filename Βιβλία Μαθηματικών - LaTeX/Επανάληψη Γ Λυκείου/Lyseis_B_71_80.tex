%=========== ΛΥΣΗ - 76ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
(Πρόβλημα Μεγιστοποίησης - Παράθυρο)
\begin{erwthma}
\item Το πλάτος της βάσης του παραθύρου είναι $2x$. Το ύψος των κατακόρυφων πλευρών είναι $y$. Το ημικύκλιο έχει διάμετρο $2x$ άρα ακτίνα $x$.
Το συνολικό μήκος του πλαισίου (χωρίς τη βάση) ανέρχεται στα 10 μέτρα, άρα:
\[ \text{Περίμετρος Δυο Κατακόρυφων \& Ημικυκλίου} = y + y + \text{Τόξο} = 10 \]
Το μήκος ενός ημικυκλίου είναι $\frac{1}{2}(2\pi r) = \pi x$.
Άρα: $2y + \pi x = 10 \Leftrightarrow 2y = 10 - \pi x - 2x \Leftrightarrow y = \frac{10 - (2+\pi)x}{2} = 5 - x - \frac{\pi}{2}x$.
Το συνολικό εμβαδόν απαρτίζεται από το ορθογώνιο ($E_{ορθ} = \beta \cdot \upsilon = 2x \cdot y$) και το ημικύκλιο ($E_{ημικ} = \frac{1}{2}\pi r^2 = \frac{1}{2}\pi x^2$).
\[ E(x) = 2xy + \frac{\pi}{2}x^2 = 2x \left( 5 - x - \frac{\pi}{2}x \right) + \frac{\pi}{2}x^2 = 10x - 2x^2 - \pi x^2 + \frac{\pi}{2}x^2 = 10x - 2x^2 - \frac{\pi}{2}x^2 \]
Για τους περιορισμούς του $x$, προφανώς το πλάτος πρέπει να είναι θετικό ($x > 0$) αλλά και το ύψος επίσης θετικό ($y > 0$):
\[ 10 - (2+\pi)x > 0 \Leftrightarrow (2+\pi)x < 10 \Leftrightarrow x < \frac{10}{2+\pi} \]
Άρα ο τύπος ισχύει για $x \in \left(0, \frac{10}{2+\pi} \right)$.

\item Παραγωγίζουμε τo εμβαδόν ώς προς $x$:
\[ E'(x) = 10 - 4x - \pi x = 10 - (4+\pi)x \]
Μηδενίζουμε την παράγωγο: $E'(x) = 0 \Leftrightarrow (4+\pi)x = 10 \Leftrightarrow x_0 = \frac{10}{4+\pi}$. (Το οποίο ανήκει στο πεδίο ορισμού, καθώς $\frac{10}{4+\pi} < \frac{10}{2+\pi}$).
- Για $x < x_0 \Rightarrow (4+\pi)x < 10 \Rightarrow E'(x) > 0$. Η $E(x)$ είναι γνησίως αύξουσα.
- Για $x > x_0 \Rightarrow E'(x) < 0$. Η $E(x)$ είναι γνησίως φθίνουσα.
Επομένως η συνάρτηση παρουσιάζει **ολικό μέγιστο** (η βέλτιστη λύση που μεγιστοποιεί το εμβαδόν) για $x_0 = \frac{10}{4+\pi}$.
Παρατηρούμε τη σχέση με το ύψος $y$ σε εκείνη τη θέση:
\[ 2y = 10 - (2+\pi)x_0 = 10 - (2+\pi)\frac{10}{4+\pi} = \frac{40+10\pi - 20 - 10\pi}{4+\pi} = \frac{20}{4+\pi} \]
Άρα $y = \frac{10}{4+\pi}$. Κατά συνέπεια, $y = x_0$. Το ύψος του ορθογωνίου πλαισίου ισούται αναγκαστικά με την ακτίνα του τόξου για να επιτύχουμε μέγιστο εμβαδόν!

\item Η συνάρτηση εμβαδού είναι συνεχής στο διάστημα $A = \left(0, \frac{10}{2+\pi}\right)$.
Λαμβάνοντας τα αντίστοιχα όρια στα άκρα του πεδίου:
- $\lim_{x\to0^+} E(x) = 0$.
- Για $x \to x_{max} = \frac{10}{2+\pi}$, το $y$ μηδενίζεται, άρα η περιοχή εκφυλίζεται σε ημικύκλιο με εμβαδόν:
\[ E\left(\frac{10}{2+\pi}\right) = \frac{\pi}{2} \left( \frac{10}{2+\pi} \right)^2 = \frac{50\pi}{(2+\pi)^2} \]
Η μέγιστη τιμή σταθμίζεται στο $x_0$:
\[ E(x_0) = 10x_0 - x_0^2\left(\frac{4+\pi}{2}\right) = x_0 \left( 10 - \frac{4+\pi}{2} x_0 \right) = \frac{10}{4+\pi} \left( 10 - \frac{4+\pi}{2} \frac{10}{4+\pi} \right) = \frac{10}{4+\pi} (10 - 5) = \frac{50}{4+\pi} \]
Η αυστηρή μελέτη μας φανερώνει ότι η $E$ αυξάνεται από το $0$ μέχρι το $\frac{50}{4+\pi}$ και μετά μειώνεται φτάνοντας στο $\frac{50\pi}{(2+\pi)^2}$ (που είναι ένα αυστηρά θετικό σημείο, μικρότερο από το μέγιστο). Το συνολικό **σύνολο τιμών** είναι $\left(0, \frac{50}{4+\pi}\right]$.

\item Η ερώτηση αναζητά το πλήθος των ριζών της εξίσωσης $E(x) = 5$.
Εξετάζουμε πού εντάσσεται το 5 μέσα στο σύνολο τιμών.
Το μέγιστο καθίσταται: $\frac{50}{4+\pi} \approx \frac{50}{7,14} \approx 7 > 5$. Άρα το 5 ξεπερνιέται.
Εξετάζουμε και το κάτω φράγμα του δεύτερου κλάδου: $\frac{50\pi}{(2+\pi)^2}$. Αφού $\pi \approx 3,14$, $\frac{50\pi}{(2+\pi)^2} \approx \frac{157}{(5,14)^2} \approx \frac{157}{26,4} \approx 5,94$.
Επομένως το ελάχιστο της φθίνουσας περιοχής είναι αυστηρά μεγαλύτερο του 5 ($5,94 > 5$).
Δεδομένου τούτου, η οριζόντια ευθεία $y=5$ τέμνει μόνο τον πρώτο κλάδο (που αυξάνεται από το 0 έως το $\approx 7$), τον οποίο τον τέμνει ακριβώς μία φορά. Τον δεύτερο κλάδο (που κατεβαίνει από το $7$ έως το $\approx 5,94$) δεν πρόκειται να τον φθάσει.
Άρα \textbf{μπορεί να κατασκευαστεί μοναδικό μόνο} παράθυρο με εμβαδόν 5 τ.μ.
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 77ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
(Γεωμετρική απόσταση - Ασύμπτωτες)
\begin{erwthma}
\item Το σημείο $M$ βρίσκεται πάνω στην καμπύλη $g(x) = \sqrt{x}$, άρα έχει συντεταγμένες $M(x, \sqrt{x})$ με $x \ge 0$. Το σημείο $A$ έχει συντεταγμένες $A(2, 0)$.
Η απόσταση των δύο σημείων (ευθύγραμμο τμήμα $(AM)$) δίνεται από τον τύπο $\sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}$:
\[ d(x) = \sqrt{(x - 2)^2 + (\sqrt{x} - 0)^2} = \sqrt{x^2 - 4x + 4 + x} = \sqrt{x^2 - 3x + 4} \]

\item Παραγωγίζουμε τη συνάρτηση απόστασης:
\[ d'(x) = \left(\sqrt{x^2 - 3x + 4}\right)' = \frac{1}{2\sqrt{x^2-3x+4}} \cdot (x^2 - 3x + 4)' = \frac{2x - 3}{2\sqrt{x^2-3x+4}} \]
(Το τριώνυμο κάτω από τη ρίζα έχει αρνητική διακρίνουσα $\Delta = 9 - 16 = -7 < 0$, άρα είναι συνεπώς πάντα θετικό και η ρίζα καλώς ορίζεται, παραγωγίσιμη παντού).
Για ακρότατα, μηδενίζουμε:
\[ d'(x) = 0 \Leftrightarrow 2x - 3 = 0 \Leftrightarrow x_0 = \frac{3}{2} \]
Προσδιορισμός μονοτονίας:
- Για $x \in \left[0, \frac{3}{2}\right)$, η παράγωγος είναι αρνητική (αριθμητής αρνητικός). Άρα η $d(x)$ είναι γνησίως φθίνουσα.
- Για $x \in \left(\frac{3}{2}, +\infty\right)$, η παράγωγος είναι θετική (αριθμητής θετικός). Άρα η $d(x)$ είναι γνησίως αύξουσα.
Η απόσταση παρουσιάζει **ολικό ελάχιστο** στη θέση $x_0 = \frac{3}{2}$. Εκεί τα σημεία βρίσκονται πιο κοντά.
Οι συντεταγμένες του σημείου $M_0$ είναι: $M_0\left(\frac{3}{2}, \sqrt{\frac{3}{2}}\right) = M_0\left(\frac{3}{2}, \frac{\sqrt{6}}{2}\right)$.

\item Αναζητούμε πλάγια ασύμπτωτη της μορφής $y = \lambda x + \beta$ στο $+\infty$.
Ο συντελεστής $\lambda$ είναι:
\[ \lambda = \lim_{x\to+\infty} \frac{d(x)}{x} = \lim_{x\to+\infty} \frac{\sqrt{x^2 - 3x + 4}}{x} \]
Στο άπειρο: $\lim_{x\to+\infty} \frac{x\sqrt{1 - \frac{3}{x} + \frac{4}{x^2}}}{x} = \lim_{x\to+\infty} \sqrt{1 - \frac{3}{x} + \frac{4}{x^2}} = \sqrt{1 - 0 + 0} = 1$. Οπότε $\lambda = 1$.
Ο συντελεστής $\beta$ είναι:
\[ \beta = \lim_{x\to+\infty} (d(x) - \lambda x) = \lim_{x\to+\infty} \left( \sqrt{x^2 - 3x + 4} - x \right) \]
Μορφή $\infty - \infty$. Πολλαπλασιάζουμε με τη συζυγή παράσταση:
\[ \beta = \lim_{x\to+\infty} \frac{(\sqrt{x^2-3x+4} - x)(\sqrt{x^2-3x+4} + x)}{\sqrt{x^2-3x+4} + x} = \lim_{x\to+\infty} \frac{(x^2-3x+4) - x^2}{\sqrt{x^2-3x+4} + x} = \lim_{x\to+\infty} \frac{-3x+4}{x\left(\sqrt{1-\frac{3}{x}+\frac{4}{x^2}} + 1\right)} \]
\[ \beta = \lim_{x\to+\infty} \frac{x\left(-3+\frac{4}{x}\right)}{x\left(\sqrt{1-\frac{3}{x}+\frac{4}{x^2}} + 1\right)} = \frac{-3 + 0}{\sqrt{1-0+0} + 1} = \frac{-3}{1+1} = -\frac{3}{2} \]
Επομένως η $d(x)$ έχει \textbf{πλάγια ασύμπτωτη στο $+\infty$} την ευθεία $y = x - \frac{3}{2}$.

\item Ζητείται το εμβαδόν μεταξύ της $g(x) = \sqrt{x}$, του άξονα $x'x$ και της κατακόρυφης ευθείας $x = \frac{3}{2}$.
Καθώς $\sqrt{x} \ge 0$ στο $\left[0, \frac{3}{2}\right]$, το εμβαδόν είναι:
\[ E = \int_0^{3/2} \sqrt{x} dx = \int_0^{3/2} x^{1/2} dx = \left[ \frac{x^{3/2}}{3/2} \right]_0^{3/2} = \frac{2}{3} \cdot \left( \frac{3}{2} \right)^{3/2} = \frac{2}{3} \cdot \left(\frac{3}{2} \sqrt{\frac{3}{2}}\right) = \sqrt{\frac{3}{2}} = \frac{\sqrt{6}}{2} \text{ τ.μ.} \]
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 78ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
(Κριτήριο Παρεμβολής - Παραγωγισιμότητα)
\begin{erwthma}
\item Γνωρίζουμε από την εκφώνηση: $1 - x^2 \le f(x) \le 1 + x^2$.
Ελέγχουμε τα όρια των φραγμάτων:
$\lim_{x\to 0}(1 - x^2) = 1 - 0 = 1$
$\lim_{x\to 0}(1 + x^2) = 1 + 0 = 1$
Επειδή τα δύο όρια ταυτίζονται, σύμφωνα με το **Κριτήριο Παρεμβολής** συνάγουμε ότι και το μεσαίο όριο θα ισούται με την ίδια τιμή, δηλαδή $\lim_{x\to 0} f(x) = 1$.
Εάν η συνάρτηση είναι συνεχής στο σημείο 0, τότε εξ ορισμού η τιμή της ισούται με το όριό της. Συνεπώς $f(0) = \lim_{x\to0} f(x) = 1$.

\item Για να είναι παραγωγίσιμη στo $0$, εξυπακούεται ότι το όριο του λόγου μεταβολής $\lim_{x\to 0} \frac{f(x)-f(0)}{x-0}$ πρέπει να υπάρχει στο $\mathbb{R}$.
Χτίζουμε τον λόγο από την αρχική ανισότητα, αντικαθιστώντας το $f(0)=1$:
\[ 1 - x^2 \le f(x) \le 1 + x^2 \Leftrightarrow -x^2 \le f(x) - 1 \le x^2 \]
Χωρίζουμε σε δύο περιπτώσεις για τη διαίρεση με το $x$:
- Αν $x > 0$: $\frac{-x^2}{x} \le \frac{f(x)-1}{x} \le \frac{x^2}{x} \Leftrightarrow -x \le \frac{f(x)-1}{x} \le x$.
Καθώς $\lim_{x\to 0^+}(-x) = 0$ και $\lim_{x\to 0^+} x = 0$, από το κριτήριο παρεμβολής το δεξί πλευρικό όριο ισούται με $0$.
- Αν $x < 0$ η φορά της ανισότητας αλλάζει: $\frac{-x^2}{x} \ge \frac{f(x)-1}{x} \ge \frac{x^2}{x} \Leftrightarrow -x \ge \frac{f(x)-1}{x} \ge x$.
Με την ίδια λογική, το αριστερό πλευρικό όριο ισούται με $0$.
Αφού και τα δύο όρια ταυτίζονται με το $0$, η $f$ είναι παραγωγίσιμη με ταχύτητα μεταβολής $\mathbf{f'(0) = 0}$.

\item Η εξίσωση της εφαπτομένης της $C_f$ στο σημείο της $x_0=0$ εξάγεται κατά τα γνωστά:
\[ y - f(0) = f'(0)(x - 0) \Leftrightarrow y - 1 = 0(x) \Leftrightarrow y = 1 \]
*(Παρατηρούμε γεωμετρικά πως η $C_f$ «στριμώχνεται» μεταξύ των παραβολών $1-x^2$ (κοίλη, εφαπτομένη $y=1$ στην κορυφή) και $1+x^2$ (κυρτή, εφαπτομένη $y=1$ στην κορυφή). Αναγκαστικά αποκτά κι η ίδια κοινή οριζόντια εφαπτομένη την $y=1$.)*

\item 
*(Σημείωση: Στην εκφώνηση αναγράφεται το όριο του $f(x) \cdot \hm\left(\frac{1}{x}\right)$. Όμως η $\lim_{x\to0} f(x) = 1$, ενώ η συνάρτηση $\hm\left(\frac{1}{x}\right)$ ταλαντώνεται απροσδόκητα έντονα μεταξύ του $[-1, 1]$ δίχως να κατέχει όριο. Έτσι το γινόμενο $1 \cdot (\tau\alpha\lambda\alpha\nu\tau\omega\sigma\eta)$ **δεν** συγκλίνει και το όριο **δεν υπάρχει** στο 0.*
*Ωστόσο, ως τυπική άσκηση εξετάσεων, προφανώς υπήρχε τυπογραφικό κι εννοεί είτε το όριο του  $\big(f(x)-1\big)\hm\left(\frac{1}{x}\right)$ είτε του $x f(x)\hm\left(\frac{1}{x}\right)$. Ας υπολογίσουμε το δεύτερο ως παράδειγμα:)*
Επιζητούμε το $\lim_{x\to0} \left[ x f(x) \cdot \hm\left(\frac{1}{x}\right)\right]$.
Γνωρίζουμε την ιδιότητα του φραγμένου: $|\hm\left(\frac{1}{x}\right)| \le 1$.
Επιπρόσθετα, γνωρίζουμε ότι $\lim_{x\to0} (x f(x)) = 0 \cdot 1 = 0$.
Η κατάσταση λοιπόν είναι $(Μηδενική) \times (Φραγμένη) = Μηδενική$. Συνεπώς, εάν υφίστατο αυτός ο παράγοντας ισότητας, το όριο θα ήταν $0$. 
Για το όριο της εκφώνησης (χωρίς το $x$): \textbf{Δεν υπάρχει}.
\end{erwthma}
\end{thema}

\begin{thema}{Β}
(Κριτήριο Παρεμβολής και Ασύμπτωτες)
\begin{erwthma}
\item Δίνεται η ανισοτική σχέση για κάθε $x > 0$:
\[ \frac{x^2 - x}{x+1} \le f(x) \le x - 2 + \frac{2}{x} \]
Διαιρούμε την ανισότητα με $x > 0$:
\[ \frac{x^2 - x}{x(x+1)} \le \frac{f(x)}{x} \le \frac{x - 2 + \frac{2}{x}}{x} \]
\[ \frac{x(x - 1)}{x(x+1)} \le \frac{f(x)}{x} \le 1 - \frac{2}{x} + \frac{2}{x^2} \]
\[ \frac{x - 1}{x+1} \le \frac{f(x)}{x} \le 1 - \frac{2}{x} + \frac{2}{x^2} \]
Υπολογίζουμε τα όρια των φραγμάτων καθώς το $x \to +\infty$:
$\lim_{x\to+\infty} \frac{x - 1}{x+1} = \lim_{x\to+\infty} \frac{x}{x} = 1$.
$\lim_{x\to+\infty} \left( 1 - \frac{2}{x} + \frac{2}{x^2} \right) = 1 - 0 + 0 = 1$.
Από το **Κριτήριο Παρεμβολής**, προκύπτει ότι:
\[ \lim_{x\to+\infty} \frac{f(x)}{x} = 1 \]

\item Για το δεύτερο όριο, αφαιρούμε από όλους τους όρους της αρχικής ανισότητας το $x$:
\[ \frac{x^2 - x}{x+1} - x \le f(x) - x \le x - 2 + \frac{2}{x} - x \]
Απλοποιούμε το αριστερό μέλος:
\[ \frac{x^2 - x - x(x+1)}{x+1} = \frac{x^2 - x - x^2 - x}{x+1} = \frac{-2x}{x+1} \]
Απλοποιούμε το δεξί μέλος:
\[ -2 + \frac{2}{x} \]
Οπότε παίρνουμε:
\[ \frac{-2x}{x+1} \le f(x) - x \le -2 + \frac{2}{x} \]
Υπολογίζουμε ξανά τα όρια στο $+\infty$:
$\lim_{x\to+\infty} \frac{-2x}{x+1} = \lim_{x\to+\infty} \frac{-2x}{x} = -2$.
$\lim_{x\to+\infty} \left( -2 + \frac{2}{x} \right) = -2 + 0 = -2$.
Εκ νέου από το **Κριτήριο Παρεμβολής**, συμπεραίνουμε ότι:
\[ \lim_{x\to+\infty} (f(x) - x) = -2 \]

\item Από τα προηγούμενα δύο ερωτήματα, διαπιστώσαμε ότι $\lim_{x\to+\infty} \frac{f(x)}{x} = \lambda = 1$ και $\lim_{x\to+\infty} (f(x) - \lambda x) = \beta = -2$.
Συνεπώς, η ευθεία με εξίσωση $y = \lambda x + \beta \Rightarrow y = x - 2$ αποτελεί **πλάγια ασύμπτωτη** της γραφικής παράστασης της $f$ στο $+\infty$.

\item Δίνεται επιπλέον το όριο: $\lim_{x\to1} \frac{f(x)}{x-1} = 2$.
Θέτουμε $g(x) = \frac{f(x)}{x-1}$, οπότε $f(x) = (x-1)g(x)$ με $\lim_{x\to1} g(x) = 2$.
Παίρνοντας όρια έχουμε: $\lim_{x\to1} f(x) = \lim_{x\to1} [(x-1)g(x)] = 0 \cdot 2 = 0$.
Καθώς η $f$ δηλώνεται **συνεχής** (στο $(0, +\infty)$ άρα και στο $x_0=1$), ισχύει: $f(1) = \lim_{x\to1} f(x) = 0$.
Με αυτήν την τιμή, το δοσμένο όριο μετασχηματίζεται άμεσα σε ορισμό της παραγώγου:
\[ \lim_{x\to1} \frac{f(x) - f(1)}{x-1} = 2 \Rightarrow f'(1) = 2 \]
Η εφαπτομένη της $C_f$ στο σημείο $x_0=1$ δίνεται από την εξίσωση:
\[ y - f(1) = f'(1)(x - 1) \Leftrightarrow y - 0 = 2(x - 1) \Leftrightarrow y = 2x - 2 \]
\end{erwthma}
\end{thema}

%=========== ΛΥΣΗ - 80ο ΘΕΜΑ Β ===========
\begin{thema}{Β}
(Συναρτησιακή Σχέση και Μελέτη $f$)
\begin{erwthma}
\item Δίνεται για κάθε $x \in \mathbb{R}$:
\[ f(x-2) = (x-2)e^{3-x} \]
Για να βρούμε τον τύπο της $f$, θέτουμε μια νέα μεταβλητή (π.χ. $u = x-2$).
Από αυτή τη σχέση ελκύουμε $x = u + 2$.
Αντικαθιστούμε στην αρχική:
\[ f(u) = u e^{3-(u+2)} = u e^{3-u-2} = u e^{1-u} \]
Άρα αντικαθιστώντας και πάλι το $u$ με το χαρακτηριστικό $x$, ο τύπος της συνάρτησης είναι πράγματι $f(x) = x e^{1-x}$ για κάθε πραγματικό αριθμό $x$.

\item Η $f(x)$ είναι παραγωγίσιμη ώς γινόμενο:
\[ f'(x) = (x)' e^{1-x} + x (e^{1-x})' = 1\cdot e^{1-x} + x \cdot e^{1-x}(-1) = e^{1-x} - x e^{1-x} = e^{1-x}(1-x) \]
Το πρόσημο εξαρτάται αποκλειστικά από τoν όρο $(1-x)$ (αφού $e^{1-x} > 0$):
- $f'(x) > 0 \Leftrightarrow 1-x > 0 \Leftrightarrow x < 1$. Άρα η $f$ είναι **γνησίως αύξουσα** στο $(-\infty, 1]$.
- $f'(x) < 0 \Leftrightarrow 1-x < 0 \Leftrightarrow x > 1$. Άρα η $f$ είναι **γνησίως φθίνουσα** στο $[1, +\infty)$.
- Η συνάρτηση παρουσιάζει **ολικό μέγιστο** στη θέση $x=1$ με τιμή:
\[ f(1) = 1 \cdot e^{1-1} = 1 \cdot e^0 = 1 \]

\item Μελετάμε την κυρτότητα υπολογίζοντας τη δεύτερη παράγωγο:
\[ f''(x) = (e^{1-x}(1-x))' = (e^{1-x})'(1-x) + e^{1-x}(1-x)' = -e^{1-x}(1-x) + e^{1-x}(-1) = e^{1-x}(x-1-1) = e^{1-x}(x-2) \]
- Ο μηδενισμός βρίσκεται στο $x=2$.
- $f''(x) > 0 \Leftrightarrow x > 2$ ($\mathbf{\kappa\upsilon\rho\tau\acute{\eta}}$ στο $[2, +\infty)$).
- $f''(x) < 0 \Leftrightarrow x < 2$ ($\mathbf{\kappa o\acute{\iota}\lambda\eta}$ στο $(-\infty, 2]$).
Στο σημείο με τετμημένη $x_0=2$, η δεύτερη παράγωγος μηδενίζεται και αλλάζει πρόσημο.
Συνεπώς, το σημείο αυτό είναι το μοναδικό **σημείο καμπής** της $C_f$.
Οι συντεταγμένες του είναι $\left(2, f(2)\right) = (2, 2e^{-1}) = \left(2, \frac{2}{e}\right)$.

\item Η εξίσωση $f(x) = 2$ ζητά σημεία που η συνάρτηση να φθάσει το $2$.
Ωστόσο, όπως αποδείξαμε στο δεύτερο ερώτημα, η $f$ κατέχει **ολικό μέγιστο** ίσο με $1$.
Αυτό πιστοποιεί πως για κάθε πραγματικό αριθμό $x$, η τιμή της συνάρτησης είναι πάντοτε μικρότερη ή ίση του $1$:
\[ f(x) \le f(1) \Leftrightarrow f(x) \le 1 \]
Καθώς η μέγιστη τιμή λειτουργίας της είναι 1, είναι εντελώς αδύνατο η $f(x)$ να ισούται ποτέ με το 2 ($1 < 2$).
Άρα η εξίσωση είναι θεωρητικά και πρακτικά \textbf{αδύνατη} στο $\mathbb{R}$.
\end{erwthma}
\end{thema}
